```python
import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient(conv, a, T):
    """Compute gradient of T*log(sum(exp(conv_k/T))) w.r.t. a."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # grad_j = 2 * sum_k w_k * a[k-j] (with bounds)
    grad = np.zeros(n)
    # For efficiency, only use peaks with significant weight
    threshold = 1e-10
    significant = w > threshold
    for k in range(len(conv)):
        if not significant[k]:
            continue
        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * wk * a[k - j]
    return grad
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_gradient(conv, a, k_peaks=50, temperature=0.01):
    """Compute soft subgradient of max(conv) using top-k peaks."""
    n = len(a)
    conv_len = len(conv)
    
    # Get top-k peak indices
    k_actual = min(k_peaks, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Softmax weights over peaks
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Compute gradient: grad_j = 2 * sum_i w_i * a[k_i - j]
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            grad[js:je+1] += 2.0 * wi * a[k - js:k - js + (je - js + 1)][::-1]
    
    return grad

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (multiplicative weight)
with full-softmax subgradients computed via np.correlate, periodic symmetrization,
temperature annealing, and multi-length exploration. Replaces LP-based Frank-Wolfe
with faster per-iteration gradient methods to allow many more iterations.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(conv, a, temperature):
    """
    Compute the gradient of T*log(sum(exp(conv_k/T))) w.r.t. a.
    Uses np.correlate for O(n^2) computation with small constant.
    """
    n = len(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform over argmax
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # grad_j = 2 * sum_k w_k * a[k-j] = 2 * (w ⋆ a)[j]
    grad = 2.0 * np.correlate(w, a, 'valid')
    return grad


def symmetrize(f):
    """Symmetrize and renormalize."""
    f = (f + f[::-1]) / 2.0
    s = f.sum()
    if s > 1e-15:
        f = f / s
    return f


def mirror_descent(f0, time_limit, T_init=0.05, T_min=0.001, T_decay=0.9997,
                   lr_init=0.02, lr_decay=0.99995, sym_interval=5,
                   restart_every=300, verbose=False):
    """
    Mirror descent (multiplicative weight) with temperature annealing
    and periodic symmetrization. Minimizes max(conv(a,a)) on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    f = symmetrize(f)
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Compute gradient with current temperature
        grad = compute_gradient(conv, f, T)

        # Multiplicative weight update
        log_step = -lr * grad
        # Clip to avoid overflow/underflow
        log_step = np.clip(log_step, -50.0, 50.0)
        f = f * np.exp(log_step)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        # Periodic symmetrization
        if it % sym_interval == 0:
            f = symmetrize(f)

        conv_new = fast_convolve(f)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal temperature
        T = max(T * T_decay, T_min)
        lr *= lr_decay

        # Restart if stuck
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 43 + 11)
            noise_scale = 0.003 if n < 1000 else 0.001
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = symmetrize(f)
            lr = lr_init * 0.5
            T = max(T_init * 0.5, T_min)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"      MD iter {it}: obj={best_obj:.10f}, lr={lr:.2e}, T={T:.5f}")

        if lr < 1e-9:
            break

    # Final symmetrization
    best_f = symmetrize(best_f)
    best_obj = fast_convolve(best_f).max()
    return best_f, best_obj


def projected_subgrad(f0, time_limit, T_init=0.01, lr_init=1.0,
                      sym_interval=10, verbose=False):
    """
    Projected subgradient descent on the simplex with symmetrization.
    Uses backtracking-style adaptive step size.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    f = symmetrize(f)
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        grad = compute_gradient(conv, f, T)

        # Projected gradient step
        f_trial = f - lr * grad
        f_trial = project_simplex(f_trial)

        if it % sym_interval == 0:
            f_trial = symmetrize(f_trial)

        obj_trial = fast_convolve(f_trial).max()

        if obj_trial < M - 1e-14:
            f = f_trial
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_f = f.copy()
                no_improve = 0
            lr = min(lr * 1.02, 5.0)
        else:
            no_improve += 1
            lr *= 0.9

        T = max(T * 0.9995, 1e-4)

        if no_improve > 200:
            rng = np.random.default_rng(it * 37 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            f = symmetrize(f)
            lr = lr_init
            T = T_init
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"      PG iter {it}: obj={best_obj:.10f}, lr={lr:.3f}")

        if lr < 1e-10:
            break

    best_f = symmetrize(best_f)
    best_obj = fast_convolve(best_f).max()
    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: reduce the largest auto-convolution
    peaks by adjusting contributing elements.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    f = symmetrize(f)
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find top peaks
        top_k = min(12, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            j_max = idx[np.argmax(products)]

            old_val = f[j_max]
            if old_val < 1e-12:
                continue

            best_trial = None
            best_trial_obj = M

            for mult in [0.4, 0.6, 0.8, 0.9, 0.95, 1.1, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                trial = f.copy()
                trial[j_max] = new_val
                diff = old_val - new_val
                if abs(diff) > 1e-15:
                    if diff > 0:
                        # Redistribute to smallest elements (excluding j_max)
                        weights = np.ones(n) - trial
                        weights[j_max] = 0
                        weights = np.maximum(weights, 0)
                        wsum = weights.sum()
                        if wsum > 1e-15:
                            trial += diff * weights / wsum
                    else:
                        # Remove proportionally from largest elements
                        weights = trial.copy()
                        weights[j_max] = 0
                        wsum = weights.sum()
                        if wsum > 1e-15:
                            removal = min(-diff, 0.9 * wsum) * weights / wsum
                            trial -= removal
                trial = np.maximum(trial, 0)
                trial = symmetrize(trial)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = obj
                    best_trial = trial

            if best_trial is not None:
                f = best_trial
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"      PeakShave iter {it}: obj={best_obj:.10f}")

    best_f = symmetrize(best_f)
    best_obj = fast_convolve(best_f).max()
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    # Choose lengths to try based on current best
    lengths_to_try = []
    n_cur = len(base)
    if n_cur <= 768:
        lengths_to_try = [1024, 1536, 2048]
    elif n_cur <= 1024:
        lengths_to_try = [1536, 2048, 3072]
    elif n_cur <= 1536:
        lengths_to_try = [2048, 3072]
    else:
        lengths_to_try = [2048, 1536]

    # Also try a shorter length (sometimes shorter is better per-element)
    if n_cur > 512:
        lengths_to_try.append(512)

    for target_n in lengths_to_try:
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break

        f_interp = interpolate_sequence(base, target_n)
        f_interp = symmetrize(f_interp)
        sc = try_seq(f_interp, f"interp_n{target_n}")

        # Optimize at this length
        t_alloc = min(rem * 0.35, 100)
        if t_alloc > 20:
            print(f"  MD n={target_n}: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(f_interp, time_limit=t_alloc,
                                        T_init=0.03, T_min=0.001, T_decay=0.9997,
                                        lr_init=0.015, lr_decay=0.9999,
                                        sym_interval=5, restart_every=200)
            try_seq(a_opt, f"md_n{target_n}")

    # === Phase 4: Projected subgradient on best ===
    print(f"\n=== Phase 4: Projected Subgradient ===")
    base = best_seq if best_seq is not None else parent_sym
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem, 100)
        print(f"  PG: t={t_alloc:.0f}s, n={len(base)}")
        a_opt, obj = projected_subgrad(base, time_limit=t_alloc,
                                       T_init=0.005, lr_init=0.5,
                                       sym_interval=10, verbose=True)
        try_seq(a_opt, "proj_subgrad")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    base = best_seq if best_seq is not None else parent_sym
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_alloc = min(rem, 80)
        print(f"  PeakShave: t={t_alloc:.0f}s")
        a_opt, obj = peak_shaving(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "peak_shave")

    # === Phase 6: Fine mirror descent (very low temp, small steps) ===
    print(f"\n=== Phase 6: Fine MD Polish ===")
    base = best_seq if best_seq is not None else parent_sym
    rem = deadline - time.monotonic() - 50
    if rem > 20:
        t_alloc = min(rem, 60)
        print(f"  MD fine: t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent(base, time_limit=t_alloc,
                                    T_init=0.005, T_min=0.0005, T_decay=0.9999,
                                    lr_init=0.005, lr_decay=0.99998,
                                    sym_interval=3, restart_every=400, verbose=True)
        try_seq(a_opt, "md_fine")

    # === Phase 7: Another peak shave ===
    print(f"\n=== Phase 7: Final Peak Shave ===")
    base = best_seq if best_seq is not None else parent_sym
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(rem, 30)
        print(f"  Final PeakShave: t={t_alloc:.0f}s")
        a_opt, obj = peak_shaving(base, time_limit=t_alloc)
        try_seq(a_opt, "final_peak_shave")

    # === Phase 8: Ultra-fine MD ===
    print(f"\n=== Phase 8: Ultra-fine MD ===")
    base = best_seq if best_seq is not None else parent_sym
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem, 20)
        a_opt, obj = mirror_descent(base, time_limit=t_alloc,
                                    T_init=0.002, T_min=0.0002, T_decay=0.99995,
                                    lr_init=0.002, lr_decay=0.99999,
                                    sym_interval=2, restart_every=500)
        try_seq(a_opt, "md_ultrafine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
