Now, for improvement strategies:

1. **Spectral approach**: In the frequency domain, the self-convolution a*a has DFT |A(ω)|^2. We want to minimize the max of the inverse DFT of |A(ω)|^2. This is related to designing signals with low peak factor.

2. **Phase retrieval / constant modulus**: If we could make |A(ω)|^2 very flat, the time-domain signal would be very flat (close to a delta). But a*a has non-negative entries (it's a Gram matrix), so its DFT is non-negative, meaning |A(ω)|^2 >= 0 for all ω. A very flat |A(ω)|^2 would give a very spread-out time-domain signal, which is what we want (to minimize the peak).

Actually, the inverse DFT of a constant is a delta function. So if |A(ω)|^2 is constant, then a*a = delta, which is impossible for a finite sequence (a*a has support 2n-1, not just 1).

3. **Welch bound / Sidelnikov sequences**: There are known constructions from Coding Theory for sequences with low autocorrelation. These might provide good starting points.

4. **Key observation from the previous code**: The LP-FW approach works by:
   - Finding the direction g that maximizes sum(g) subject to (f*g)_k <= M for all k
   - This is like finding the "best" next point on the constraint boundary
   - Then doing a line search

   - Use a better LP solver or formulation
   - Add more sophisticated line search

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

    return e / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fenchel dual subgradient method with FISTA inner solver (primary - novel)
2. LP-based Frank-Wolfe (polishing)
3. Adaptive log-sum-exp L-BFGS (diversity and escape from local minima)
4. Multi-length exploration with interpolation
5. Final coordinate descent + subgradient polishing
Key innovation: The dual reformulates max-constraint as a weighted sum, making
the inner problem a convex QP solvable via O(n log n) FFT-based matvec products.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_Tw(w, a, fft_size):
    """Compute T_w * a = cross-correlation of w and a, first len(a) components.
    (T_w a)_i = sum_k w(k) * a(k-i) for valid indices."""
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(np.conj(W) * A, fft_size)[:len(a)]


def solve_inner_qp(w, a_init, n_iters, fft_size, use_fista=True):
    """Solve min a^T T_w a s.t. a in simplex using FISTA or projected gradient."""
    n = len(a_init)
    a = a_init.copy()
    a_prev = a_init.copy()
    t = 1.0
    L = 2.0
    step = 1.0 / L

    for i in range(n_iters):
        if use_fista:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + (t / t_new - 1.0) * (a - a_prev)
            # Ensure y is valid (clip to [0, 1] range roughly)
            y = np.maximum(y, 0.0)
            y_sum = y.sum()
            if y_sum < 1e-15:
                y = a.copy()
            else:
                y = y / y_sum
        else:
            y = a

        Tw_y = compute_Tw(w, y, fft_size)
        grad = 2.0 * Tw_y
        a_new = project_simplex(y - step * grad)

        a_prev = a
        a = a_new
        if use_fista:
            t = t_new

    return a


def dual_subgradient_optimize(a0, time_limit, n_inner=40, verbose=False, seed=42):
    """
    Dual subgradient method for min_a max_k b_k(a) s.t. a in simplex.
    The dual is max_w Q(w) where Q(w) = min_a a^T T_w a.
    Subgradient of Q at w is b(a*(w)).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    b_len = 2 * n - 1
    fft_size_cross = 1 << (3 * n - 3).bit_length()

    a = a0.copy()
    # Initialize dual weights concentrated on peak convolution values
    b_init = fast_convolve(a)
    temp_init = b_init.max() * 0.1
    w = softmax_fn(b_init / max(temp_init, 1e-10))

    best_obj = b_init.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    eta = 0.5

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Solve inner QP
        a_inner = solve_inner_qp(w, a, n_inner, fft_size_cross, use_fista=True)

        # Evaluate actual objective
        b = fast_convolve(a_inner)
        obj = b.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_inner.copy()
            no_improve = 0
            if verbose and iter_count % 100 == 0:
                print(f"    Dual iter {iter_count}: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1

        a = a_inner

        # Dual update: subgradient ascent
        w = project_simplex(w + eta * b)

        # Decay step size
        eta = 0.5 / np.sqrt(iter_count)
        eta = max(eta, 0.001)

        # Periodic: sharpen dual weights toward active constraints
        if iter_count % 200 == 0:
            b_curr = fast_convolve(a)
            peak = b_curr.max()
            active = b_curr > 0.9 * peak
            w_blend = 0.5 * w + 0.5 * (active / active.sum())
            w = project_simplex(w_blend)

        # Random restart if stuck
        if no_improve > 500:
            noise_scale = 0.005 + 0.005 * min(no_improve // 500, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            w = project_simplex(rng.random(b_len))
            eta = 0.3
            no_improve = 0
            n_inner = max(20, min(60, n_inner + 5))

    return best_a, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    while time.monotonic() - t0 < time_limit:

        if g_sum < 1e-15:

        g = g / g_sum

            no_improve += 1
            if no_improve > 80:
                break

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)

            if st < 1e-15:

            if val < best_val:

                    hi = mid

            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

                best_obj = M
                best_f = f.copy()
                no_improve = 0

            no_improve = 0

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    eps = 1e-15
    t = np.log(f + eps)

    t0_time = time.monotonic()

    for beta in [10.0, 25.0, 50.0, 100.0, 200.0]:
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break

        try:
            res = minimize(
                _lse_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 200, 'gtol': 1e-8, 'maxls': 30, 'maxfun': 1000}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    LSE-LBFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def _lse_obj_grad(t, beta):
    """Compute log-sum-exp objective and gradient in softmax parameter space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_shifted = beta * (conv - conv.max())
    exp_vals = np.exp(conv_shifted)
    log_sum = conv.max() + np.log(exp_vals.sum()) / beta

    # Gradient with respect to a: weight by softmax of conv
    w = exp_vals / exp_vals.sum()

    # grad_a[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w cross-corr a)[i]
    fft_size = 1 << (3 * n - 3).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    R = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    grad_a = 2.0 * R

    # Chain rule: grad_t = a * (grad_a - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return log_sum, grad_t


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.6, 0.8, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.95, top_k=50, temp=0.0001):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        fft_size = 1 << (3 * n - 3).bit_length()
        W = np.fft.rfft(w.astype(np.float64), fft_size)
        A = np.fft.rfft(f, fft_size)
        R = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
        sg = 2.0 * R

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        temp *= 1.005

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
