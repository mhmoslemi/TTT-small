"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using improved coordinate descent with
wider search, peak-targeting subgradient, pair redistribution, and multi-length
exploration. Key improvements: adaptive coordinate search range, direct peak
shaping via subgradient on top-k peaks, structured pair moves, and broader
n-value exploration.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score_full(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def coord_descent_v2(a, time_limit, wide_range=False):
    """Coordinate descent with adaptive (wider) search range."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score_full(a)
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            
            # Determine search range
            if wide_range:
                lo = 0.0
                hi = min(0.5, max(old_val * 5, 0.05))
            else:
                lo = max(0.0, old_val * 0.2)
                hi = min(0.5, max(old_val * 5, old_val + 0.05))
            
            if hi - lo < 1e-7:
                continue
            
            def obj_v(v):
                v = max(0.0, min(v, 1.0))
                new_a = a.copy()
                new_a[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                scale = (1.0 - v) / other_sum
                new_a[mask] *= scale
                conv = fast_convolve(new_a)
                return conv.max()
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 40})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score_full(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if not improved and pass_count > 3:
            break
        if pass_count > 300:
            break
    
    return best_a, best_score

def peak_targeting_subgrad(a, time_limit, top_k=50, lr0=0.005, decay=0.9998):
    """Subgradient method targeting only the top peaks of b."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score_full(a)
    t0 = time.monotonic()
    step = 0
    cur_lr = lr0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)
        
        # Focus on top-k peaks only
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        # Inverse softmax: weight peaks inversely (push down the highest)
        # Actually, we want to push DOWN the peaks, so weight = pv (higher peaks get more gradient)
        w = pv / pv.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # db[k]/da[j] = 2*a[k-j] for 0 <= k-j < n
            # So sg[j] += w[i] * 2 * a[k-j] for valid j
            # For a given peak k: contributes to sg[j] for j in [max(0,k-n+1), min(k, n-1)]
            # with value 2*a[k-j]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        # Multiplicative update (mirror descent)
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()
        
        sc = compute_score_full(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        
        cur_lr *= decay
        
        if step % 2000 == 0:
            print(f"    PeakSG step {step}: score={best_score:.10f}, lr={cur_lr:.6f}")
        
        # Restart if stuck
        if step % 5000 == 0 and cur_lr < lr0 * 0.05:
            cur_lr = lr0 * 0.3
            a = normalize(best_a + np.random.standard_normal(n) * 0.002)
    
    return best_a, best_score

def pair_redistribution(a, time_limit, n_trials=1000):
    """Find beneficial pair transfers by targeting the max lag of b."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score_full(a)
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_k = np.argmax(conv)
        max_val = conv[max_k]
        
        # Find pairs contributing most to b[max_k]
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        pairs = a[idx] * a[max_k - idx]
        
        # Get top contributors
        top_idx = idx[np.argsort(pairs)[::-1][:10]]
        top_partner = max_k - top_idx
        
        # Try transferring from top contributors to random positions
        for trial in range(min(n_trials, max(1, int((time_limit - (time.monotonic() - t0)) * 50)))):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a source from top contributors or random
            if rng.random() < 0.7:
                src = rng.choice(top_idx)
            else:
                src = rng.integers(0, n)
            
            dst = rng.integers(0, n)
            if src == dst:
                continue
            
            # Try several transfer fractions
            a_src = a[src]
            if a_src < 1e-10:
                continue
            
            best_f = 0
            best_sc = best_score
            for f in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                delta = a_src * f
                new_a = a.copy()
                new_a[src] = a_src - delta
                new_a[dst] += delta
                conv_new = fast_convolve(new_a)
                sc = 2.0 * n * conv_new.max()
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = f
            
            if best_f > 0:
                delta = a[src] * best_f
                a[src] -= delta
                a[dst] += delta
                a = normalize(a)
                sc = compute_score_full(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)

def hanning_sequence(n):
    """Hanning (raised cosine) window."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    a[0] = 0
    a[-1] = 0
    if a.sum() < 1e-15:
        a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return normalize(a)
