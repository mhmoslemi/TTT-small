```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iteration (Von Neumann minimax),
soft-max subgradient polishing, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f):
    """Build A where A[j,k] = f(j-k) if 0 <= j-k < n else 0."""
    n = len(f)
    j_idx = np.arange(2 * n - 1)[:, None]
    k_idx = np.arange(n)[None, :]
    diff = j_idx - k_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = f[diff[mask]]
    return A


def solve_lp_given_f(f, M):
    """Solve: max sum(b) s.t. (f*b)(j) <= M for all j, b >= 0.
    Returns optimal b vector."""
    n = len(f)
    A = build_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return np.ones(n) / n  # fallback


def lp_iteration(f, t=0.1):
    """One LP-based iteration. Returns (new_f, M, lp_optimal_value)."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    b = solve_lp_given_f(f, M)
    b_sum = b.sum()
    if b_sum < 1e-15:
        return f, M, 0.0
    b_norm = b / b_sum
    f_new = (1 - t) * f + t * b_norm
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss < 1e-15:
        f_new = f.copy()
    else:
        f_new = f_new / ss
    return f_new, M, b_sum


def soft_max_subgradient(f, temp, top_k=None):
    """Compute soft-max weighted subgradient."""
    n = len(f)
    conv = fast_convolve(f)
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    if top_k is not None and top_k < len(w):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(w))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    # Subgradient: sg[j] = sum_i w[i+j] * f[i] = correlate(w, f, 'valid')
    sg = np.correlate(w, f, 'valid')
    return sg, conv


def compute_obj(f):
    """Compute the objective value (without the 2n factor, just max(conv) since sum=1)."""
    conv = fast_convolve(f)
    return conv.max()


def compute_full_score(f):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(f)
    return 2.0 * n * conv.max() / (s * s)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'cos':
        x = np.linspace(0, 1, n)
        f = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

                line_search_every=50, verbose=False):

    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    
    n = len(f)
    t = t_init
    best_f = f.copy()
    best_obj = compute_obj(f)
    best_score = compute_full_score(f)
    
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # LP step
        f_new, M, lp_val = lp_iteration(f, t)
        
        # Compute objective
        cur_obj = compute_obj(f_new)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP iter {iter_count}: obj={best_obj:.10f}, t={t:.4f}")
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive t
```

```python
def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f):
    n = len(f)
    j_idx = np.arange(2 * n - 1)[:, None]
    k_idx = np.arange(n)[None, :]
    diff = j_idx - k_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = f[diff[mask]]
    return A


def solve_lp(f, M):
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        x = np.maximum(res.x, 0)
        y = np.maximum(getattr(res.ineqlin, 'marginals', None), 0) if hasattr(res, 'ineqlin') else None
        return x, y
    return np.ones(n) / n, None
```

    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A_matrix(f):
    """Build constraint matrix A where A[j,k] = f(j-k) if 0<=j-k<n else 0."""
    n = len(f)
    j_idx = np.arange(2 * n - 1)[:, None]
    k_idx = np.arange(n)[None, :]
    diff = j_idx - k_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = f[diff[mask]]
    return A


def solve_lp_primal(f, M):
    """Solve: max sum(b) s.t. (f*b)(j) <= M for all j, b >= 0.
    Returns (optimal_b, dual_variables_y or None)."""
    n = len(f)
    A = build_A_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        x = np.maximum(res.x, 0.0)
        y = None
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            if hasattr(res.ineqlin, 'marginals'):
                y = np.maximum(res.ineqlin.marginals, 0.0)
        return x, y
    return np.ones(n) / n, None


def compute_conv_obj(f):
    """Compute max of auto-convolution (assuming f is normalized)."""
    return fast_convolve(f).max()


def compute_score(f):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * fast_convolve(f).max() / (s * s)


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi


def make_init(n, shape='sin2'):
    """Create initialization of given length and shape."""
    x = np.linspace(0, 1, n)
    if shape == 'sin2':
        f = np.sin(np.pi * x) ** 2
    elif shape == 'cos':
        f = 0.5 + 0.5 * np.cos(2.0 * np.pi * (x - 0.5))
    elif shape == 'gauss':
        f = np.exp(-5.0 * (x - 0.5) ** 2)
    elif shape == 'flat':
        f = np.ones(n)
    elif shape == 'sinc':
        f = np.sinc(2.0 * (x - 0.5))
        f = np.maximum(f, 0.01)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def lp_optimize(f0, time_limit, t_init=0.3, t_min=0.005, t_decay=0.997,
                ls_period=25, verbose=False):
    """LP-based (Von Neumann minimax) optimization with adaptive mixing."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f0)) / len(f0)
    n = len(f)

    t = t_init
    best_f = f.copy()
    best_obj = compute_conv_obj(f)

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break

        # LP step
        f_new, M, y_dual = None, 0.0, None
        tp = time.monotonic()
        f_new, M, y_dual = None, 0.0, None
        # Do the actual LP step manually for timing
        conv = fast_convolve(f)
        M = conv.max()
        t_lp_start = time.monotonic()
        b, y_dual = solve_lp_primal(f, M)
        t_lp = time.monotonic() - t_lp_start

        b_sum = b.sum()
        if b_sum < 1e-15:
            f_new = f.copy()
        else:
            b_norm = b / b_sum
            f_new = (1.0 - t) * f + t * b_norm
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss < 1e-15:
                f_new = f.copy()
            else:
                f_new = f_new / ss

        cur_obj = compute_conv_obj(f_new)

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"      LP it={it}, obj={best_obj:.12f}, t={t:.5f}, lp_time={t_lp*1000:.1f}ms")
        else:
            no_improve += 1

        f = f_new
        t = max(t * t_decay, t_min)

        # Periodic line search with LP solution
        if it % ls_period == 0 and no_improve > 3:
            conv_best = fast_convolve(best_f)
            M_best = conv_best.max()
            b_ls, _ = solve_lp_primal(best_f, M_best)
            bs = b_ls.sum()
            if bs > 1e-15:
                b_ls = b_ls / bs
                best_t_obj = best_obj
                best_t = t
                best_f_try = best_f.copy()
                for t_try in [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    f_try = (1.0 - t_try) * best_f + t_try * b_ls
                    f_try = np.maximum(f_try, 0.0)
                    ss = f_try.sum()
                    if ss > 1e-15:
                        f_try = f_try / ss
                        obj_try = compute_conv_obj(f_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_t = t_try
                            best_f_try = f_try.copy()
                if best_f_try is not best_f:
                    t = best_t
                    f = best_f_try.copy()
                    best_obj = best_t_obj
                    best_f = best_f_try.copy()
                    no_improve = 0

        # Escape mechanism
        if no_improve > 250:
            rng_esc = np.random.default_rng(it * 31 + 17)
            f = project_simplex(best_f + rng_esc.standard_normal(n) * 0.004)
            t = max(t_init * 0.4, 0.03)
            no_improve = 0
            it = 0

    return best_f, best_obj


def dual_guided_step(f, y_dual, t):
    """Use dual variables from LP to construct a subgradient step."""
    n = len(f)
    if y_dual is None or len(y_dual) != 2 * n - 1:
        return f
    y_sum = y_dual.sum()
    if y_sum < 1e-15:
        return f
    y_norm = y_dual / y_sum
    # Subgradient: sg[k] = 2 * sum_i f(i) * y_norm(i+k)
    # This is 2 * convolve(f, y_norm)[:n] ... let me compute carefully
    # sg[k] = 2 * sum_i f(i) * y_norm(i + k)
    # = 2 * (f reversed convolved with y_norm) ... 
    # Actually: sum_i f(i) * y_norm(i+k) is the cross-correlation, which is
    # np.correlate(y_norm, f, 'valid') ... no
    # Let me just compute it directly
    # sg[k] = sum_i f(i) * y_norm(i+k) where i ranges over [0, n-1] and i+k over [0, 2n-2]
    # This is the "valid" cross-correlation
    sg = 2.0 * np.correlate(y_norm, f, 'valid')
    f_new = f - t * sg
    f_new = project_simplex(f_new)
    return f_new


def sg_optimize(f0, time_limit, temp_init=0.004, temp_decay=0.9996,
                lr=0.005, top_k=None, verbose=False):
    """Soft-max subgradient optimization with Adam."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f0)) / len(f0)
    n = len(f)

    temp = temp_init
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    best_f = f.copy()
    best_obj = compute_conv_obj(f)

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t_count += 1
        conv = fast_convolve(f)
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws > 1e-300:
            w /= ws
        else:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        if top_k is not None and top_k < len(w):
            idx = np.argpartition(w, -top_k)[-top_k:]
            wm = np.zeros(len(w))
            wm[idx] = w[idx]
            wm /= wm.sum()
            w = wm

        sg = 2.0 * np.correlate(w, f, 'valid')
        m = beta1 * m + (1.0 - beta1) * sg
        v = beta2 * v + (1.0 - beta2) * sg * sg
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        f = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
        temp = max(temp * temp_decay, 1e-7)

        cur_obj = compute_conv_obj(f)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t_count % 200 == 0:
                print(f"      SG it={t_count}, obj={best_obj:.12f}")
        else:
            no_improve += 1

        if no_improve > 600:
            rng_esc = np.random.default_rng(t_count * 19 + 5)
            f = project_simplex(best_f + rng_esc.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            temp = max(temp_init * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: LP-based (Von Neumann minimax) iteration - main workhorse
    Phase 2: Multi-length exploration
    Phase 3: Soft-max subgradient polishing
    Phase 4: Final fine-tuning with alternating LP and subgradient
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_init(768, 'sin2')
        try:
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                best_score = ps
                best_seq = parent.copy()
            print(f"No parent, using sin2 init n=768, score={ps:.8f}")
        except Exception:
            best_score = 2.0
            best_seq = parent.copy()

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: LP-based optimization (main workhorse) ===
    phase1_time = 0.62 * (deadline - time.monotonic())
    print(f"\n=== Phase 1: LP optimization, budget={phase1_time:.0f}s ===")

    # Try multiple initializations at parent length
    inits = {
        'parent': parent.copy(),
        'sin2': make_init(n_parent, 'sin2'),
        'cos': make_init(n_parent, 'cos'),
        'gauss': make_init(n_parent, 'gauss'),
    }

    # Add perturbed parent
    inits['parent_pert'] = project_simplex(parent + rng.standard_normal(n_parent) * 0.02)
    inits['parent_flat'] = project_simplex(0.6 * parent + 0.4 * np.ones(n_parent) / n_parent)

    for name, init in inits.items():
        rem = deadline - time.monotonic() - 0.35 * (deadline - time.monotonic())
        rem = min(rem, phase1_time - (time.monotonic() - (deadline - phase1_time)))
        t_alloc = min(rem * 0.6, 150.0)
        if t_alloc < 15:
            break
        print(f"  LP from '{name}' (t={t_alloc:.0f}s)")
        a_opt, obj = lp_optimize(init, time_limit=t_alloc, t_init=0.3,
                                  t_min=0.005, t_decay=0.997, verbose=False)
        a_opt = np.maximum(a_opt, 0.0)
        ss = a_opt.sum()
        if ss > 0.01:
            a_eval = a_opt.tolist()
            try:
                sc = float(evaluate_sequence(a_eval))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"    NEW BEST [lp_{name}]: {sc:.8f} (n={len(a_opt)})")
            except Exception:
                pass

    # Continue LP optimization from current best for remaining phase 1 time
    if best_seq is not None:
        rem_p1 = (deadline - 0.35 * (deadline - (deadline - phase1_time))) - time.monotonic()
        rem_p1 = phase1_time - (time.monotonic() - (deadline - phase1_time))
        if rem_p1 > 20:
            n_best = len(best_seq)
            print(f"  Extended LP from best (n={n_best}, t={rem_p1:.0f}s)")
            a_opt, _ = lp_optimize(best_seq, time_limit=rem_p1, t_init=0.15,
                                    t_min=0.003, t_decay=0.998, verbose=True)
            a_opt = np.maximum(a_opt, 0.0)
            if a_opt.sum() > 0.01:
                try:
                    sc = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_opt.copy()
                        print(f"    NEW BEST [lp_ext]: {sc:.8f} (n={len(a_opt)})")
                except Exception:
                    pass

    # === Phase 2: Multi-length exploration ===
    phase2_time = 0.17 * (1800 - 30 - (time.monotonic() - (deadline - 1770.0)))
    phase2_time = min(phase2_time, 280.0)
    print(f"\n=== Phase 2: Multi-length exploration, budget={phase2_time:.0f}s ===")

    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = [512, 640, 768, 896, 1024, 1152, 1280]
        target_lengths = [t for t in target_lengths if t != n_best and 256 <= t <= 1536]
        target_lengths.sort(key=lambda x: abs(x - n_best))
        target_lengths = target_lengths[:3]

        for target_n in target_lengths:
            rem = phase2_time - (time.monotonic() - (time.monotonic() - phase2_time))
            rem = deadline - time.monotonic() - 0.13 * 1770
            rem = min(rem, 100.0)
            if rem < 20:
                break
            f_interp = interpolate_seq(best_seq, target_n)
            t_alloc = min(rem * 0.7, 80.0)
            print(f"  LP at n={target_n} (t={t_alloc:.0f}s)")
            a_opt, _ = lp_optimize(f_interp, time_limit=t_alloc, t_init=0.2,
                                    t_min=0.005, t_decay=0.997)
            a_opt = np.maximum(a_opt, 0.0)
            if a_opt.sum() > 0.01:
                try:

                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_opt.copy()
                        print(f"    NEW BEST [len_{target_n}]: {sc:.8f} (n={target_n})")
                except Exception:
                    pass

    # === Phase 3: Subgradient polishing ===
    phase3_time = min(200.0, 0.12 * 1770)

    if best_seq is not None:
        n_best = len(best_seq)
        t_sg = min(phase3_time * 0.6, 120.0)
        print(f"  SG from best (t={t_sg:.0f}s)")
        a_opt, _ = sg_optimize(best_seq, time_limit=t_sg, temp_init=0.003,
                                temp_decay=0.9997, lr=0.004,
                                top_k=max(20, n_best // 12))
        a_opt = np.maximum(a_opt, 0.0)
        if a_opt.sum() > 0.01:
            try:
                sc = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"    NEW BEST [sg]: {sc:.8f} (n={n_best})")
            except Exception:
                pass

        # SG from perturbed best
        rem = phase3_time - (time.monotonic() - (time.monotonic()))
        rem = min(rem, 60.0)
        if rem > 10 and best_seq is not None:
            n_best = len(best_seq)
            perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            a_opt, _ = sg_optimize(perturbed, time_limit=rem, temp_init=0.005,
                                    temp_decay=0.9995, lr=0.005)
            a_opt = np.maximum(a_opt, 0.0)
            if a_opt.sum() > 0.01:

    print(f"\n=== Phase 4: Final fine-tuning ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem_final = deadline - time.monotonic() - 10.0

        # Very fine LP with small t
        if rem_final > 20:
            t_fine = min(rem_final * 0.5, 60.0)
            print(f"  Fine LP (t={t_fine:.0f}s)")
            a_opt, _ = lp_optimize(best_seq, time_limit=t_fine, t_init=0.05,
                                    t_min=0.002, t_decay=0.999, ls_period=15)
            a_opt = np.maximum(a_opt, 0.0)
            if a_opt.sum() > 0.01:
                try:
                    sc = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
