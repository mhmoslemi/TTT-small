import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,
                   lr_decay=0.9999, verbose=False, label=""):
    """
    Fast multiplicative weight mirror descent on the simplex.
    Minimizes max_k (a*a)_k subject to a>=0, sum(a)=1.
    Uses smoothed subgradient with adaptive temperature and backtracking step size.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    obj_history = []

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        obj_history.append(M)

        # Find top peaks
        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        # Softmax weights over peaks
        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        g = compute_subgradient(a, peak_indices, w)

        # Backtracking line search for step size
        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            if trial_lr < 1e-10:
                continue
            log_update = -trial_lr * g
            # Clip for numerical stability
            log_update = np.clip(log_update, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        # Apply update
        log_update = -best_lr * g
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}, lr={best_lr:.6f}")
        else:
            no_improve += 1

        # Adapt temperature (increase slowly to broaden peak set)
        temp = min(temp * 1.002, 0.05)

        # Adapt learning rate
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * lr_decay, 1e-8)

        # Restart if stuck for too long
        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.002 * min(no_improve / 5000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        # Symmetrize periodically
        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)

            sym_obj = fast_convolve(a_sym).max()

                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if lr < 1e-9:
            break

    return best_a, best_obj


def rectilinear_flow(a0, time_limit, top_k=30, temp_init=0.001, dt_init=0.005,
                     verbose=False, label=""):
    """
    Rectilinear (additive) flow on the simplex: da/dt = -(g - mean(g)).
    Different geometry than multiplicative descent, can escape different local minima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    dt = dt_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(a, peak_indices, w)

        # Additive update projected to tangent space
        g_centered = g - g.mean()
        dt_trial = dt
        best_trial_obj = M * 1.01
        for trial_dt in [dt, dt * 0.5, dt * 0.25, dt * 0.1]:
            if trial_dt < 1e-10:
                continue
            a_trial = a - trial_dt * g_centered
            a_trial = project_simplex(a_trial)
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj <= best_trial_obj:
                dt_trial = trial_dt
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        a = a - dt_trial * g_centered
        a = project_simplex(a)

        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} RF iter {it}: obj={best_obj:.10f}, dt={dt_trial:.6f}")
        else:
            no_improve += 1

        temp = min(temp * 1.003, 0.05)
        dt = max(dt * 0.9999, 1e-9)

        if no_improve > 200:
            rng = np.random.default_rng(it * 104729 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            dt = dt_init * 0.3
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = project_simplex(a_sym)
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if dt < 1e-10:
            break

    return best_a, best_obj


def peak_reduction(a0, time_limit, verbose=False, label=""):
    """
    Targeted peak reduction: identify top peaks, find most contributing pairs,
    and redistribute mass from high-peak contributors to low-peak positions.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False

        # Find top peaks
        K = min(8, len(conv))
        peak_indices = np.argsort(conv)[-K:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            j_max = idx[np.argmax(products)]

            # Try reducing a[j_max] and redistributing
            old_val = a[j_max]
            if old_val < 1e-10:
                continue

            best_trial = None
            best_trial_obj = M

            for frac in [0.3, 0.5, 0.7, 0.85, 1.2]:
                new_val = old_val * frac
                if new_val < 0:
                    continue
                trial = a.copy()
                trial[j_max] = new_val
                diff = old_val - new_val
                if abs(diff) < 1e-15:
                    continue

                # Redistribute to positions NOT contributing to peak k
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc = a[non_contrib]
                    nc_sum = nc.sum()
                    if nc_sum > 1e-15:
                        trial[non_contrib] += diff * nc / nc_sum
                        trial = np.maximum(trial, 0.0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_trial_obj - 1e-14:
                            best_trial_obj = obj
                            best_trial = trial

            if best_trial is not None:
                a = best_trial
                improved = True
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    if verbose and it % 10 == 0:
                        print(f"    {label} PR iter {it}: obj={best_obj:.10f}")

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Try broad reduction of all peak-contributing elements
                k = int(np.argmax(fast_convolve(a)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)

                for scale in [0.9, 0.8, 0.7]:
                    trial = a.copy()
                    trial[idx] *= scale
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[idx] = False
                    if non_contrib.sum() > 0:
                        nc = a[non_contrib]
                        nc_sum = nc.sum()
                        if nc_sum > 1e-15:
                            trial[non_contrib] += (a.sum() - trial.sum()) * nc / nc_sum
                            trial = np.maximum(trial, 0.0)
                            trial = trial / trial.sum()
                            obj = fast_convolve(trial).max()
                            if obj < best_obj - 1e-14:
                                a = trial
                                best_obj = obj
                                best_a = trial.copy()
                                no_improve = 0
                                break
                else:
                    break
        else:
            no_improve = 0

    return best_a, best_obj


def fine_coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent: adjust one coordinate at a time with binary search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try several multipliers
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=0.3):
    """Exponentially decaying sequence."""
    a = np.exp(-np.arange(n) / (n * decay))
    a = a / a.sum()
    return a


def make_bimodal(n, center1_frac=0.3, center2_frac=0.7, sigma=0.08):
    """Bimodal Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center1_frac) / sigma) ** 2) + \
        np.exp(-0.5 * ((x - center2_frac) / sigma) ** 2)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Symmetrize a sequence (make it palindromic)."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = 768
    if parent is not None:
        n_base = len(parent)

    # === Phase 1: Diverse Initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []
    for n_try in [n_base, 640, 896, 1024]:
        a = make_gaussian(n_try, 0.25)
        init_candidates.append((a, f"gauss_n{n_try}"))
        a = make_gaussian(n_try, 0.35)
        init_candidates.append((a, f"gauss_w_n{n_try}"))
        a = make_bell(n_try, 0.5)
        init_candidates.append((a, f"bell_n{n_try}"))
        a = make_bimodal(n_try)
        init_candidates.append((a, f"bimodal_n{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))
        init_candidates.append((symmetrize(parent), "parent_sym"))

    # Quick evaluate all initializations
    best_init = None
    best_init_obj = float('inf')
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        obj = try_seq(a_norm, f"init_{label}")
        conv = fast_convolve(a_norm)
        c_obj = 2 * len(a_norm) * conv.max()
        if c_obj < best_init_obj:
            best_init_obj = c_obj
            best_init = a_norm

    if best_init is None:
        best_init = make_gaussian(n_base, 0.3)

    print(f"  Best init obj: {best_init_obj:.6f}")

    # === Phase 2: Main Mirror Descent (largest time allocation) ===
    print(f"\n=== Phase 2: Mirror Descent (main) ===")
    remaining = deadline - time.monotonic()
    t_md = remaining * 0.40
    print(f"  Time allocated: {t_md:.0f}s")
    a_opt, obj = mirror_descent(best_init, time_limit=t_md, top_k=30,
                                temp_init=0.0003, lr_init=0.01,
                                verbose=True, label="MD_main")
    try_seq(a_opt, "MD_main")

    # === Phase 3: Rectilinear Flow ===
    print(f"\n=== Phase 3: Rectilinear Flow ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_rf = remaining * 0.20
        print(f"  Time allocated: {t_rf:.0f}s")
        a_opt, obj = rectilinear_flow(best_seq, time_limit=t_rf, top_k=30,
                                      temp_init=0.001, dt_init=0.003,
                                      verbose=True, label="RF")
        try_seq(a_opt, "RF")

    # === Phase 4: Peak Reduction ===
    print(f"\n=== Phase 4: Peak Reduction ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_pr = remaining * 0.15
        print(f"  Time allocated: {t_pr:.0f}s")
        a_opt, obj = peak_reduction(best_seq, time_limit=t_pr, verbose=True, label="PR")
        try_seq(a_opt, "PR")

    # === Phase 5: Fine Coordinate Descent ===
    print(f"\n=== Phase 5: Fine CD ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = remaining * 0.15
        print(f"  Time allocated: {t_cd:.0f}s")
        a_opt, obj = fine_coordinate_descent(best_seq, time_limit=t_cd,
                                             verbose=True, label="CD")
        try_seq(a_opt, "CD")

    # === Phase 6: Second pass Mirror Descent with different params ===
    print(f"\n=== Phase 6: MD pass 2 (different params) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md2 = remaining * 0.25
        print(f"  Time allocated: {t_md2:.0f}s")

        # Start from symmetrized version
        a_sym = symmetrize(best_seq)
        a_opt, obj = mirror_descent(a_sym, time_limit=t_md2 * 0.5, top_k=50,
                                    temp_init=0.0001, lr_init=0.005,
                                    verbose=True, label="MD_sym")
        try_seq(a_opt, "MD_sym")

        if time.monotonic() < deadline - 60:
            a_opt, obj = mirror_descent(best_seq, time_limit=t_md2 * 0.5, top_k=15,
                                        temp_init=0.0005, lr_init=0.008,
                                        verbose=True, label="MD_fine")
            try_seq(a_opt, "MD_fine")

    # === Phase 7: Multi-restart from best ===
    print(f"\n=== Phase 7: Multi-restart ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 4
        t_per = min(remaining * 0.20 / n_restarts, 60)
        if t_per > 20:
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                if r == 0:
                    f_init = best_seq.copy()
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.002
                    f_init = np.maximum(best_seq + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                elif r == 2:
                    f_init = symmetrize(best_seq)
                    noise = rng.standard_normal(len(f_init)) * 0.001
                    f_init = np.maximum(f_init + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                else:
                    # Blend with uniform
                    f_init = 0.9 * best_seq + 0.1 * (np.ones(len(best_seq)) / len(best_seq))
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per, remaining - 30)
                if t_alloc < 10:
                    break
                a_opt, obj = mirror_descent(f_init, time_limit=t_alloc, top_k=30,
                                            temp_init=0.0002, lr_init=0.006,
                                            verbose=(r == 0), label=f"restart_{r}")
                try_seq(a_opt, f"restart_{r}")

    # === Phase 8: Final polish with very fine MD ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_polish = min(remaining - 10, 80)
            print(f"  Polish time: {t_polish:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_polish, top_k=10,
                                        temp_init=0.00005, lr_init=0.003,
                                        verbose=True, label="polish")
            try_seq(a_opt, "polish")

            # Try symmetrized version as final answer
            a_s = symmetrize(best_seq)
            sc = try_seq(a_s, "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
