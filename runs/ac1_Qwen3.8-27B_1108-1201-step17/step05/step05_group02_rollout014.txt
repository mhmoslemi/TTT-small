with log-sum-exp smoothing, periodic LP-FW refinement, structured initializations,
multi-length exploration, and coordinate descent polishing.
Key improvements: Nesterov acceleration, adaptive peak targeting, faster iteration

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_subgradient(f, conv, top_k, temp):
    """Compute smoothed subgradient of max(conv) using log-sum-exp on top-k peaks."""
    n = len(f)
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    for i in range(len(peaks)):
        if w[i] < 1e-14:

        k_pos = int(peaks[i])
        js = max(0, k_pos - n + 1)
        je = min(n - 1, k_pos)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k_pos - idx]
    return sg


def solve_lp_direction(f, M, n, timeout=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': timeout, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_relax, n, timeout=20)
        if g is None:
            m_relax *= 1.1
            if m_relax > 6.0:
                m_relax = 1.0
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.02, 4.0)
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M

        alphas_coarse = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
        for alpha in alphas_coarse:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            if best_alpha >= 1.0:
                lo, hi = 0.6, 1.0
            elif best_alpha <= 0.02:
                lo, hi = 0.0, 0.1
            else:
                lo = max(0.0, best_alpha * 0.6)
                hi = min(1.0, best_alpha * 1.4)

            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    iter {it} ({elapsed:.0f}s): obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.003 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-15:
                # Try small positive values
                candidates = [0.001, 0.005, 0.01]
            else:
                candidates = [old_val * m for m in [0.3, 0.5, 0.7, 1.4, 2.0, 3.0]]
                candidates = [c for c in candidates if 0 < c < 0.99]

            best_j_val = old_val
            best_j_obj = cur_max

            for new_val in candidates:
                if abs(new_val - old_val) < 1e-12:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
                    cur_max = best_j_obj

        if not improved and passes > 6:
            break

    return best_f, best_obj


def combined_optimize(f0, time_limit, verbose=False):
    """
    Combined optimization: alternates between Nesterov subgradient and LP-FW.
    Subgradient for fast exploration, LP-FW for precise direction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Nesterov state
    y = f.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = 0.006
    temp = 0.004
    top_k = max(20, n // 25)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed

        # Every 200 iterations, do an LP-FW step (if enough time)
        if it % 200 == 0 and remaining > 15:
            M_cur = fast_convolve(y).max()
            g = solve_lp_direction(y, M_cur * 1.003, n, timeout=10)
            if g is None:
                g = solve_lp_direction(y, M_cur * 1.02, n, timeout=15)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    # Try several step sizes
                    best_alpha = 0.0
                    best_val = M_cur
                    for alpha in [0.05, 0.1, 0.2, 0.4, 0.6, 1.0]:
                        ft = y + alpha * (g - y)
                        ft = np.maximum(ft, 0.0)
                        st = ft.sum()
                        if st < 1e-15:
                            continue
                        ft = ft / st
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha
                    if best_alpha > 0:
                        y = y + best_alpha * (g - y)
                        y = np.maximum(y, 0.0)
                        y = y / y.sum()
                        f = y.copy()
                        v = np.zeros(n)
                        t_nest = 1.0
                        conv = fast_convolve(y)
                        if conv.max() < best_obj - 1e-15:
                            best_obj = conv.max()
                            best_f = y.copy()
                            no_improve = 0
                            if verbose:
                                print(f"    [LP-FW] iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
                        continue
            no_improve += 5

        # Subgradient step
        conv_y = fast_convolve(y)
        cur_max = conv_y.max()
        sg = compute_smoothed_subgradient(y, conv_y, top_k, temp)

        # Adaptive step size
        t_nest += 1.0
        gamma = lr / np.sqrt(t_nest)

        v = 0.92 * v - gamma * sg
        y_new = y + v
        y_new = project_simplex(y_new)

        # Nesterov weight update
        t_next = 1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest) / 2.0
        beta = (t_nest - 1.0) / t_next
        f = y_new + beta * (y_new - y)
        f = project_simplex(f)
        y = f.copy()

        conv = fast_convolve(y)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = y.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    [SG] iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive: cool temperature, expand top_k slowly
        temp = max(temp * 0.998, 0.001)
        if no_improve > 50 and it % 10 == 0:
            top_k = min(top_k + 2, n)

        # Decay learning rate
        lr = max(lr * 0.9995, 0.0003)

        # Restart if stuck
        if no_improve > 400:
            rng = np.random.default_rng(it * 31 + 7)
            scale = min(0.005, 0.001 * (no_improve // 100))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            v = np.zeros(n)
            t_nest = 1.0
            lr = max(0.003, lr * 2)
            temp = 0.003
            top_k = max(20, n // 25)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use linear interpolation (cubic can introduce negative values)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, kind='uniform', rng=None):
    """Generate structured initial sequences."""
    if kind == 'uniform':
        f = np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = np.cos(x / 2) ** 2
        f = np.maximum(f, 0)
        f /= f.sum()
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        f = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        f /= f.sum()
    elif kind == 'u_shape':
        x = np.linspace(-1, 1, n)
        f = 1.0 + 0.5 * x ** 2
        f /= f.sum()
    elif kind == 'w_shape':
        x = np.linspace(0, 2 * np.pi, n)
        f = 1.0 + 0.3 * np.cos(2 * x)
        f = np.maximum(f, 0.01)
        f /= f.sum()
    elif kind == 'bimodal':
        center = (n - 1) / 2.0
        sigma = n / 6.0
        offset = n / 4.0
        f = (np.exp(-0.5 * ((np.arange(n) - center + offset) / sigma) ** 2) +
             np.exp(-0.5 * ((np.arange(n) - center - offset) / sigma) ** 2))
        f /= f.sum()
    elif kind == 'decay':
        f = np.exp(-np.arange(n) / (n * 0.3))
        f_rev = np.exp(-np.arange(n - 1, -1, -1) / (n * 0.3))
        f = 0.5 * f + 0.5 * f_rev
        f /= f.sum()
    else:
        f = np.ones(n) / n
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Nesterov subgradient, LP-FW,
    structured initializations, multi-length search, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Quick structured initialization exploration ===
    print(f"\n=== Phase 1: Structured init exploration ===")
    init_kinds = ['uniform', 'cosine', 'gaussian', 'u_shape', 'w_shape', 'bimodal', 'decay']
    for kind in init_kinds:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        f_init = structured_init(n_parent, kind=kind, rng=rng)
        # Quick combined optimization (60s each)
        t_alloc = min(60, rem * 0.15)
        if t_alloc < 10:
            continue
        a_opt, _ = combined_optimize(f_init, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  [{kind}] NEW BEST: {sc:.6f}")

    # Also try parent directly
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(80, rem * 0.3)
        a_opt, _ = combined_optimize(parent, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  [parent] NEW BEST: {sc:.6f}")

    # === Phase 2: Deep optimization on best with combined method ===
    print(f"\n=== Phase 2: Deep combined optimization (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    if rem > 100:
        t_alloc = min(500, rem * 0.4)
        print(f"  t={t_alloc:.0f}s, n={n_best}")
        a_opt, _ = combined_optimize(best_seq, time_limit=t_alloc, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 3: Diverse restarts with LP-FW ===
    print(f"\n=== Phase 3: Diverse LP-FW restarts (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    inits = []
    inits.append(("direct", best_seq.copy()))

    # Small noise
    noise = rng.standard_normal(n_best) * 0.003
    inits.append(("noise_small", project_simplex(best_seq + noise)))

    # Medium noise
    noise = rng.standard_normal(n_best) * 0.01
    inits.append(("noise_med", project_simplex(best_seq + noise)))

    # Symmetrize
    f_sym = 0.5 * best_seq + 0.5 * best_seq[::-1]
    inits.append(("sym", project_simplex(f_sym)))

    # Mix with uniform
    f_mix = 0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best)
    inits.append(("mix_unif", project_simplex(f_mix)))

    # Mix with cosine
    f_cos = structured_init(n_best, 'cosine')
    f_mix2 = 0.7 * best_seq + 0.3 * f_cos
    inits.append(("mix_cos", project_simplex(f_mix2)))

    # Shift
    f_shift = np.roll(best_seq, 1)
    inits.append(("shift1", project_simplex(f_shift)))

    for idx, (label, f_init) in enumerate(inits):
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(120, rem * 0.2)
        if t_alloc < 10:

        print(f"  LP-FW [{label}]: t={t_alloc:.0f}s")

        if np.isfinite(sc) and sc < best_score:

            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.6f}) ===")

        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, rem * 0.25)

        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(f_interp, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (n={target_n}): {sc:.6f}")

    # === Phase 5: Intensive subgradient polishing ===
    print(f"\n=== Phase 5: SG polishing (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t_alloc = min(200, rem * 0.5)
        print(f"  t={t_alloc:.0f}s")
        a_opt, _ = nesterov_subgradient_optimize(
            best_seq, time_limit=t_alloc, verbose=True,
            top_k_init=max(20, n_best // 20), temp_init=0.002)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 60
    if rem > 30:
        t_cd = min(rem * 0.5, 150)
        print(f"  t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 7: Final combined optimization squeeze ===
    print(f"\n=== Phase 7: Final squeeze (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 30:
        t_alloc = min(rem * 0.8, 100)
        print(f"  t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(best_seq, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

        # Tiny perturbations
        rem2 = deadline - time.monotonic() - 10
        if rem2 > 5:
            n_best = len(best_seq)
            for trial in range(12):
                rem3 = deadline - time.monotonic()
                if rem3 < 2:
                    break
                scale = 0.0002 + trial * 0.0001
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                sc = float(evaluate_sequence(f_pert.tolist()))
                if np.isfinite(sc) and sc < best_score:

    # Fallback

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
