Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based subgradient method,
Adam with soft-max subgradients, and multi-phase optimization with length exploration.

import numpy as np
import time
import scipy.optimize as opt
import scipy.sparse as sp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
Phase 1: LP-based directional optimization (main innovation)

import numpy as np

# All helper functions at top level (no closures, no lambdas)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_soft_max_weights(conv, temp, top_k=None):
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(conv))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w

def compute_subgradient(f, w):
    """sg[j] = 2 * sum_i f[i] * w[j+i], computed as 2*correlate(w, f, 'valid')"""
    return 2.0 * np.correlate(w, f, 'valid')

def solve_lp_direction(f, M, n):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, conv(f,g) <= M.
    Returns optimal g or None if infeasible.
    """
    nnz_rows = []
    nnz_cols = []
    nnz_vals = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            v = f[k - j]
            if v > 1e-14:
                nnz_rows.append(k)
                nnz_cols.append(j)
                nnz_vals.append(v)
    
    A = sp.csr_matrix((nnz_vals, (nnz_rows, nnz_cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, 
                          method='highs', options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def lp_optimize(f0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_conv = fast_convolve(f)
    best_obj = best_conv.max()
    
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        
        # Solve LP
        g = solve_lp_direction(f, M, n)
        if g is None or g.sum() < 1e-12:
            no_improve += 50
            if no_improve > 200:
                break
            continue
        
        g = g / g.sum()
        
        # Line search for blend parameter
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            f_new = (1 - t) * f + t * g
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new <= M + 1e-12:
                f = f_new
                improved = True
                break
        
        if not improved:
            no_improve += 1
        else:
            if no_improve > 0:
                no_improve = 0
        
        # Check for stagnation
        conv_check = fast_convolve(f)

            no_improve += 1
        else:
            no_improve = 0

            best_f = f.copy()
            best_obj = fast_convolve(f).max()

def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,

        conv = fast_convolve(f)
        w = get_soft_max_weights(conv, temp, top_k)
        sg = compute_subgradient(f, w)
        
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)
        
        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.05)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit):
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:

            if old_val < 1e-12:

            cur_obj = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_obj
            
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:

                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom

                f_trial[j] = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_symmetric_init(n, shape='bump'):
