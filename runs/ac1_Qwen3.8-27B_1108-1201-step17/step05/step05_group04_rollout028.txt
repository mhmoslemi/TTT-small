def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    """Compute exact gradient of (conv(a,a))[k] with respect to a."""

    hi2 = min(n - 1, k + n - 1)

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows_list.append(k)
            cols_list.append(i)
    return np.array(rows_list), np.array(cols_list)


def solve_lp(a, rows, cols):
    """
    Solve: min M subject to C@b <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt, dual_weights) or (None, None, None).
    """
    n = len(a)
    values = a[rows - cols]

    # A_ub: (2n-1) x (n+1), first n cols are C, last col is -1 (for M)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[rows, cols] = values
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

    if not res.success:
        return None, None, None

    b_opt = np.maximum(res.x[:n], 0.0)
    bs = b_opt.sum()
    if bs < 1e-15:
        return None, None, None
    b_opt /= bs
    M_opt = res.x[n]

    # Extract dual variables for inequality constraints
    if hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = np.abs(res.ineqlin.marginals)
        duals_sum = duals.sum()
        if duals_sum > 1e-15:
            duals /= duals_sum
        else:
            duals = None
    else:
        duals = None

    return b_opt, M_opt, duals


def lp_refine(a0, time_limit, rows, cols, t_init=0.5, t_min=0.1,
              verbose=False):
    """Iterative LP-based refinement."""
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b_opt, M_opt, duals = solve_lp(a, rows, cols)

        if b_opt is None:
            break

        # Determine step size
        t_step = max(t_init * (0.995 ** t), t_min)

        # Update
        a_new = (1 - t_step) * a + t_step * b_opt
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new /= a_new_sum
        a = a_new

        cur_obj = fast_convolve(a).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            # Try a perturbation
            rng = np.random.default_rng(t * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            no_improve = 0
            t = max(t - 20, 0)

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {t}: obj={best_obj:.10f}, t={t_step:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, top_k=50, verbose=False):
    """Adam with exact subgradient of the max convolution term."""
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    cur_lr = lr

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        # Find top-k lags and use weighted gradient
        if top_k is not None and top_k < len(conv):
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
            top_vals = conv[top_idx]
            # Soft distribution over top-k
            top_vals_shifted = top_vals - top_vals.max()
            w = np.exp(top_vals_shifted * 10)
            w /= w.sum()
            sg = np.zeros(n)
            for idx, wi in zip(top_idx, w):
                if wi > 1e-8:
                    sg += wi * compute_exact_gradient(f, idx)
        else:
            k_max = np.argmax(conv)
            sg = compute_exact_gradient(f, k_max)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(f0, time_limit, n_trials_per_coord=5):
    """Coordinate descent with scaling of other elements."""
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_obj = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining exact subgradient Adam, LP-based
    block coordinate descent, length exploration, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        print("No parent, using uniform n=512")

    n_parent = len(parent)

    # Precompute conv matrix indices for parent length
    rows_parent, cols_parent = build_conv_conv_indices(n_parent)

    # === Phase 1: Fast Adam subgradient (120s) ===
    print(f"\n=== Phase 1: Adam subgradient (n={n_parent}) ===")
    t_phase1_end = time.monotonic() + 120

    # Try a few initializations
    inits = {
        'parent': parent.copy(),
        'gauss': None,
        'bump': None,
    }
    x = np.linspace(0, 1, n_parent)
    inits['gauss'] = np.exp(-0.5 * ((x - 0.5) / 0.2) ** 2)
    inits['gauss'] = np.maximum(inits['gauss'], 1e-10)
    inits['gauss'] /= inits['gauss'].sum()
    inits['bump'] = np.sin(np.pi * x) ** 2
    inits['bump'] = np.maximum(inits['bump'], 1e-10)
    inits['bump'] /= inits['bump'].sum()

    phase1_best = None
    phase1_best_obj = float('inf')

    for name, init in inits.items():
        rem = t_phase1_end - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 35)
        print(f"  Adam '{name}', t={t_alloc:.0f}s")
        a_opt, obj = adam_optimize(init, time_limit=t_alloc, lr=0.004,
                                   top_k=max(20, n_parent // 15))
        try_seq(a_opt, f"p1_{name}")
        if obj < phase1_best_obj:
            phase1_best_obj = obj
            phase1_best = a_opt.copy()

    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    # === Phase 2: LP-based refinement (500s) ===
    print(f"\n=== Phase 2: LP refinement (best={best_score:.8f}, n={n_base}) ===")
    t_phase2_end = time.monotonic() + 500

    rows_base, cols_base = build_conv_conv_indices(n_base)

    # Run LP refinement from current best
    rem = t_phase2_end - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.6, 250)
        print(f"  LP refine from best, t={t_alloc:.0f}s")
        a_opt, obj = lp_refine(base, time_limit=t_alloc, rows=rows_base,
                               cols=cols_base, t_init=0.4, t_min=0.1)
        try_seq(a_opt, "p2_lp_best")

    # LP refine from a perturbed version
    rem = t_phase2_end - time.monotonic()
    if rem > 30 and best_seq is not None:
        n_best = len(best_seq)
        rows_b, cols_b = build_conv_conv_indices(n_best)
        perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * 0.02)
        t_alloc = min(rem * 0.5, 150)
        print(f"  LP refine from perturbed, t={t_alloc:.0f}s")
        a_opt, obj = lp_refine(perturbed, time_limit=t_alloc, rows=rows_b,
                               cols=cols_b, t_init=0.3, t_min=0.05)
        try_seq(a_opt, "p2_lp_pert")

    # Another LP run with different step
    rem = t_phase2_end - time.monotonic()
    if rem > 20 and best_seq is not None:
        n_best = len(best_seq)
        rows_b, cols_b = build_conv_conv_indices(n_best)
        t_alloc = min(rem * 0.7, 120)
        print(f"  LP refine (small t), t={t_alloc:.0f}s")
        a_opt, obj = lp_refine(best_seq, time_limit=t_alloc, rows=rows_b,
                               cols=cols_b, t_init=0.15, t_min=0.03)
        try_seq(a_opt, "p2_lp_small")

    # === Phase 3: Length exploration (400s) ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.8f}) ===")
    t_phase3_end = time.monotonic() + 400

    target_lengths = [384, 448, 512, 640, 768, 896, 1024, 1152, 1280]
    target_lengths = [t for t in target_lengths if t != n_base and 128 <= t <= 2048]
    target_lengths.sort(key=lambda x: abs(x - n_base))
    target_lengths = target_lengths[:5]

    for target_n in target_lengths:
        rem = t_phase3_end - time.monotonic()
        if rem < 50:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        rows_t, cols_t = build_conv_conv_indices(target_n)

        # Adam first
        t_alloc_adam = min(rem * 0.25, 40)
        print(f"  n={target_n}: Adam t={t_alloc_adam:.0f}s")
        a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc_adam,
                                 lr=0.005, top_k=max(20, target_n // 15))
        try_seq(a_opt, f"p3_adam_{target_n}")

        # Then LP
        rem = t_phase3_end - time.monotonic()
        if rem > 30:
            t_alloc_lp = min(rem * 0.5, 80)
            print(f"  n={target_n}: LP t={t_alloc_lp:.0f}s")
            a_opt, _ = lp_refine(a_opt, time_limit=t_alloc_lp, rows=rows_t,
                                 cols=cols_t, t_init=0.3, t_min=0.05)
            try_seq(a_opt, f"p3_lp_{target_n}")

    # === Phase 4: Deep LP optimization (400s) ===
    print(f"\n=== Phase 4: Deep LP (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 400

    if best_seq is not None:
        n_best = len(best_seq)
        rows_b, cols_b = build_conv_conv_indices(n_best)

        # Long LP run
        rem = t_phase4_end - time.monotonic()
        if rem > 60:
            t_alloc = min(rem * 0.6, 200)
            print(f"  Deep LP, t={t_alloc:.0f}s")
            a_opt, _ = lp_refine(best_seq, time_limit=t_alloc, rows=rows_b,
                                 cols=cols_b, t_init=0.25, t_min=0.05)
            try_seq(a_opt, "p4_deep_lp")

        # Adam + LP alternation
        rem = t_phase4_end - time.monotonic()
        if rem > 40 and best_seq is not None:
            n_best = len(best_seq)
            rows_b, cols_b = build_conv_conv_indices(n_best)
            a_curr = best_seq.copy()
            alternating_end = time.monotonic() + min(rem * 0.5, 150)
            print(f"  Alternating Adam+LP until {alternating_end - time.monotonic():.0f}s")
            while time.monotonic() < alternating_end:
                # Short Adam
                a_opt, _ = adam_optimize(a_curr, time_limit=15, lr=0.003,
                                         top_k=max(10, n_best // 20))
                s = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_seq = a_opt.copy()
                    a_curr = a_opt.copy()
                    print(f"    Alt improved: {s:.8f}")
                # Short LP
                if time.monotonic() < alternating_end - 30:
                    a_opt2, _ = lp_refine(a_curr, time_limit=20, rows=rows_b,
                                          cols=cols_b, t_init=0.2, t_min=0.05)
                    s2 = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(s2) and s2 < best_score:
                        best_score = s2
                        best_seq = a_opt2.copy()
                        a_curr = a_opt2.copy()
                        print(f"    Alt LP improved: {s2:.8f}")

    # === Phase 5: Coordinate descent + fine Adam (200s) ===
    print(f"\n=== Phase 5: Polishing (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 30:
            t_cd = min(rem * 0.4, 80)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p5_cd")

        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_fine = min(rem * 0.5, 60)
            print(f"  Fine Adam t={t_fine:.0f}s")
            a_opt, _ = adam_optimize(best_seq, time_limit=t_fine, lr=0.001,
                                     top_k=max(5, n_best // 30))
            try_seq(a_opt, "p5_fine")

        # Tiny LP
        rem = deadline - time.monotonic() - 10
        if rem > 20 and best_seq is not None:
            n_best = len(best_seq)
            rows_b, cols_b = build_conv_conv_indices(n_best)
            t_lp_fine = min(rem * 0.6, 50)
            print(f"  Fine LP t={t_lp_fine:.0f}s")
            a_opt, _ = lp_refine(best_seq, time_limit=t_lp_fine, rows=rows_b,
                                 cols=cols_b, t_init=0.05, t_min=0.01)
            try_seq(a_opt, "p5_lp_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
