    Compute the log-sum-exp objective and its gradient with respect to z,
    where a = softmax(z) (so sum(a) = 1, a >= 0).
    """
    a = softmax(z)
    conv = fast_convolve(a)

    bc = beta * conv
    bc_max = bc.max()
    log_sum_exp = bc_max + np.log(np.sum(np.exp(bc - bc_max)))

    obj = 2.0 * n * log_sum_exp

    w = np.exp(bc - bc_max)
    w /= w.sum()

import numpy as np
import time
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << max(1, (size - 1).bit_length())
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def lse_grad_z(z, beta, n):
    """
    Compute gradient of 2n * logsumexp(beta * conv(softmax(z))) w.r.t. z.
    Uses FFT for the cross-correlation step.
    """
    a = softmax(z)
    conv = fast_convolve(a)

    bc = beta * conv
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w /= w.sum()

    # grad_a[j] = sum_k w[k] * a[k-j] for 0<=k-j<n
    # = cross-correlation of a and w
    # Using: grad_a[j] = conv(a_reversed, w)[j + n - 1]
    a_rev = a[::-1]
    fft_len = 1 << max(1, (n + (2 * n - 1) - 1).bit_length())
    A = np.fft.rfft(a_rev, fft_len)
    W = np.fft.rfft(w, fft_len)
    conv_bw = np.fft.irfft(A * W, fft_len)
    grad_a = beta * conv_bw[(n - 1):(n - 1) + n]

    # Chain rule through softmax
    s = np.dot(grad_a, a)
    grad_z = a * (grad_a - s)
    grad_z *= 2.0 * n

    lse_val = bc_max + np.log(w.sum())
    return 2.0 * n * lse_val, grad_z


def adam_optimize(a_init, time_limit, beta_start=3.0, beta_end=40.0,
                  lr0=0.05, restart_every=0, verbose=False):
    """
    Adam optimization on the simplex via softmax parameterization.
    Adaptive beta annealing from beta_start to beta_end.
    """
    n = len(a_init)
    # Initialize z from a_init
    a_init = np.maximum(a_init, 1e-15)
    z = np.log(a_init / a_init.sum())
    z = z - z.mean()  # center for numerical stability

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr0
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-10

    best_a = a_init.copy()
    best_score = compute_score(a_init)
    stale_count = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Batch of steps
        batch = 50
        t_start_batch = time.monotonic()

        for _ in range(batch):
            if time.monotonic() - t0 >= time_limit:
                break
            t += 1

            # Adaptive beta
            progress = min(1.0, t / max(1, time_limit / 3.0))
            beta = beta_start + (beta_end - beta_start) * progress * progress

            obj, grad_z = lse_grad_z(z, beta, n)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Check improvement
            if t % 10 == 0:
                a = softmax(z)
                sc = compute_score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    stale_count = 0
                else:
                    stale_count += 1

        # Adaptive learning rate
        if stale_count > 8:
            lr *= 0.8
            if lr < 0.001:
                # Restart with perturbation
                a_now = softmax(z)
                noise = np.random.standard_normal(n) * 0.005
                a_pert = project_simplex(a_now + noise)
                z = np.log(np.maximum(a_pert, 1e-15))
                z = z - z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr0 * 0.5
                stale_count = 0

        if verbose and (t % 500 == 0):
            print(f"        Adam t={t}, beta={beta:.1f}, lr={lr:.4f}, best={best_score:.10f}")

    return best_a, best_score


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_targeting(a, time_limit, n_peaks=5, verbose=False):
    """
    Targeted reduction of convolution peaks.
    For each peak, try to reduce it by transferring mass from contributing indices.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(a)
        current_score = 2.0 * n * conv.max()

        # Find top peaks
        top_k = min(n_peaks, len(conv) - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[peak_indices > 0]  # skip k=0

        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break

            # conv[pk] = sum_i a[i]*a[pk-i]
            # Find the pair with largest contribution
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[pk - idx]
            max_local = np.argmax(contributions)
            i_star = idx[max_local]
            j_star = pk - idx[max_local]

            if i_star == j_star:
                continue

            # Try transferring from the larger to the smaller
            if a[i_star] > a[j_star]:
                src, dst = i_star, j_star
            else:
                src, dst = j_star, i_star

            # Try several transfer amounts
            best_delta = 0.0
            best_sc = best_score

            for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                delta = a[src] * frac
                new_a = a.copy()
                new_a[src] -= delta
                new_a[dst] += delta
                conv_new = fast_convolve(new_a)
                sc = 2.0 * n * conv_new.max()
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = delta

            if best_delta > 0:
                a[src] -= best_delta
                a[dst] += best_delta
                a = normalize(a)
                if compute_score(a) < best_score:
                    best_score = compute_score(a)
                    best_a = a.copy()
                    improved = True

        if verbose and iteration % 20 == 0:
            print(f"        PT iter={iteration}, score={best_score:.10f}")

        if not improved and iteration > 30:
            break
        if iteration > 2000:
            break

    return best_a, best_score


def coord_descent_polish(a, time_limit, verbose=False):
    """Coordinate descent with golden-section search on the simplex."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if no_improve > 3:
                break

            old_val = a[j]
            if old_val < 1e-14 and pass_count < 3:
                continue

            # Determine search bounds
            lo = max(0.0, old_val * 0.5)
            hi = min(0.5, old_val * 2.0 + 0.01)
            if hi - lo < 1e-8:
                lo = max(0.0, old_val - 0.01)
                hi = min(0.5, old_val + 0.01)

            def obj_v(v):
                v = max(0.0, min(1.0, v))
                new_a = a.copy()
                new_a[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                new_a[mask] *= (1.0 - v) / other_sum
                new_a = np.maximum(new_a, 0.0)
                s = new_a.sum()
                if s < 1e-15:
                    return float('inf')
                conv = fast_convolve(new_a)
                return 2.0 * n * conv.max() / (s * s)

            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 25})
                if res.success and res.fun < best_score - 1e-12:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        new_a[mask] *= (1.0 - v_new) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-13:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        no_improve = 0 if improved else no_improve + 1
        if verbose and pass_count % 5 == 0:
            print(f"        CD pass={pass_count}, score={best_score:.10f}")
        if not improved and pass_count > 5:
            break
        if pass_count > 200:
            break

    return best_a, best_score


def mirror_descent(a_init, time_limit, lr0=0.01, top_k=20, verbose=False):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr0
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)

        # Top-k peak weighting
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        temp = max(1e-8, (pv.max() - pv.min()) * 0.3)
        w = np.exp(pv / temp)
        w /= w.sum()

        # Subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Multiplicative update
        a = np.maximum(a * np.exp(-lr * sg), 1e-15)
        a = a / a.sum()

        sc = compute_score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        lr *= 0.9998
        if lr < lr0 * 0.05:
            lr = lr0 * 0.3
            noise = np.random.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)

        if verbose and step % 2000 == 0:
            print(f"        MD step={step}, score={best_score:.10f}, lr={lr:.5f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    if len(f) == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, len(f))
    new_x = np.linspace(0, 1, new_n)
    return normalize(np.interp(new_x, old_x, f))


def geometric_seq(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    return normalize(a)


def geometric_seq_asym(n, r1, r2):
    """Asymmetric geometric: rises with r1 for first half, falls with r2."""
    a = np.zeros(n)
    half = n // 2
    a[:half] = r1 ** np.arange(half)
    a[half:] = a[half - 1] * r2 ** np.arange(n - half)
    return normalize(a)


def cosine_seq(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def kaiser_seq(n, beta_kw=5.0):
    """Kaiser window."""
    n1 = n - 1
    if n1 <= 0:
        return np.ones(1)
    x = np.linspace(-1, 1, n)
    # Kaiser Bessel window
    from scipy.special import iv
    a = iv(0, beta_kw * np.sqrt(1 - x ** 2)) / iv(0, beta_kw)
    return normalize(a)


def gaussian_seq(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def triangular_seq(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def random_smooth_seq(n, rng, k=20):
    a = rng.random(n)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)


def optimize_full(a_init, time_limit, verbose=False):
    """Multi-stage optimization pipeline."""
    n = len(a_init)
    a_init = normalize(a_init.copy())
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    t0 = time.monotonic()

    # Stage 1: Mirror descent (exploratory)
    t1 = min(time_limit * 0.2, 60)
    if time_limit - (time.monotonic() - t0) > t1 + 20:
        a_md, sc = mirror_descent(a_init, time_limit=t1,
                                   lr0=0.005, top_k=max(10, n // 10))
        if sc < best_score:
            best_score = sc
            best_a = a_md.copy()
        if verbose:
            print(f"      After MD: {best_score:.10f}")

    # Stage 2: Adam with LSE
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.45, 120)
    if rem > t2 + 15:
        a_adam, sc = adam_optimize(best_a, time_limit=t2,
                                    beta_start=3.0, beta_end=35.0,
                                    lr0=0.03)
        if sc < best_score:
            best_score = sc
            best_a = a_adam.copy()
        if verbose:
            print(f"      After Adam: {best_score:.10f}")

    # Stage 3: Peak targeting
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.25, 50)
    if rem > t3 + 10:
        a_pt, sc = peak_targeting(best_a, time_limit=t3, n_peaks=5)
        if sc < best_score:
            best_score = sc
            best_a = a_pt.copy()
        if verbose:
            print(f"      After PT: {best_score:.10f}")

    # Stage 4: Coordinate descent polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_cd, sc = coord_descent_polish(best_a, time_limit=min(rem, 100))
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()
        if verbose:
            print(f"      After CD: {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initializations, gradient-based optimization
    with adaptive LSE smoothing, peak targeting, coordinate descent polish,
    and dimension exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    def track(a, label):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Generate initializations ===
    print("=== Generating initializations ===")
    starts = []

    # Try to use parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent", len(parent)))
            print(f"  Parent: n={len(parent)}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    # Structured initializations for various n
    for n in [512, 640, 768, 896, 1024]:
        for r in [0.998, 0.999, 0.9995]:
            starts.append((geometric_seq(n, r), f"geom_r{r}_n{n}", n))
        starts.append((cosine_seq(n), f"cos_n{n}", n))
        starts.append((kaiser_seq(n, 4.0), f"kaiser_n{n}", n))
        starts.append((gaussian_seq(n, 0.35), f"gauss_n{n}", n))
        starts.append((triangular_seq(n), f"tri_n{n}", n))
        starts.append((geometric_seq_asym(n, 0.999, 0.9985), f"asym_n{n}", n))

    # Random smooth
    for n in [640, 768, 1024]:
        starts.append((random_smooth_seq(n, rng, k=25), f"rand_n{n}", n))

    # Quick evaluation
    print(f"  Evaluating {len(starts)} starts...")
    scored = []
    for a, label, n in starts:
        sc = compute_score(a)
        scored.append((sc, a, label))
    scored.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(scored[:5]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        track(a, label)

    # === Phase 1: Optimize top diverse starts ===
    print(f"\n=== Phase 1: Optimize top starts (budget until {deadline - time.monotonic():.0f}s) ===")

    # Select diverse starts
    selected = []
    used_keys = set()
    for sc, a, label in scored:
        if len(selected) >= 5:
            break
        key = label.split('_n')[0] if '_n' in label else label
        n_val = label.split('_n')[-1] if '_n' in label else 'unknown'
        full_key = f"{key}_{n_val}"
        if full_key not in used_keys:
            selected.append((a, label, sc))
            used_keys.add(full_key)

    # Ensure parent is included
    if not any("parent" in label for _, label, _ in selected):
        for sc, a, label in scored:
            if "parent" in label:
                selected.append((a, label, sc))
                break

    t_per = min((deadline - time.monotonic() - 300) / max(1, len(selected)), 200)

    for a_init, label, init_sc in selected:
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        t_alloc = min(t_per, rem)
        print(f"  Optimizing [{label}] (n={len(a_init)}, init_sc={init_sc:.6f}): t={t_alloc:.0f}s")
        a_opt, sc = optimize_full(a_init, time_limit=t_alloc, verbose=True)
        track(a_opt, label + "_opt")

    # === Phase 2: Dimension exploration ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        for target_n in [512, 640, 768, 896, 1024, 1280]:
            if target_n == len(best_a):
                continue
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 100)
            a_interp = interpolate_sequence(best_a, target_n)
            a_opt, sc = optimize_full(a_interp, time_limit=t_alloc)
            track(a_opt, f"dim_{target_n}")

    # === Phase 3: Final aggressive polish ===
    print(f"\n=== Phase 3: Final polish (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_cd = min(rem * 0.4, 150)
            print(f"  Final CD: t={t_cd:.0f}s")
            a_final, sc = coord_descent_polish(best_a, time_limit=t_cd, verbose=True)
            track(a_final, "final_cd")

            rem = deadline - time.monotonic() - 30
            if rem > 15:
                t_pt = min(rem * 0.5, 80)
                a_pt, sc = peak_targeting(best_a, time_limit=t_pt, n_peaks=8)
                track(a_pt, "final_pt")

            # Adam restart on best
            rem = deadline - time.monotonic() - 20
            if rem > 15:
                t_adam = min(rem * 0.8, 80)
                a_ad, sc = adam_optimize(best_a, time_limit=t_adam,
                                          beta_start=10.0, beta_end=60.0, lr0=0.02)
                track(a_ad, "final_adam")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 15
        if rem > 10 and best_a is not None:
            n_best = len(best_a)
            for trial in range(3):
                rem = deadline - time.monotonic() - 10
                if rem < 10:
                    break
                for noise_scale in [0.001, 0.003, 0.01]:
                    rem = deadline - time.monotonic() - 8
                    if rem < 8:
                        break
                    noise = rng.standard_normal(n_best) * noise_scale
                    a_pert = project_simplex(best_a + noise)
                    a_opt, sc = optimize_full(a_pert, time_limit=min(rem, 30))
                    track(a_opt, f"perturm_{trial}_{noise_scale}")

    # === Fallback ===
    if best_a is None:
        best_a = cosine_seq(768)
        best_score = compute_score(best_a)

    # Final verification
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return final_list
