import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_exact_subgrad(a, conv, threshold_frac=0.99):
    """
    Compute exact subgradient of max(conv) using all lags within
    threshold_frac of the maximum.
    """
    n = len(a)
    max_conv = conv.max()
    # Active lags: those within threshold of max
    active = np.where(conv >= threshold_frac * max_conv)[0]
    w = np.ones(len(active)) / len(active)
    
    grad = np.zeros(n)
    for idx in range(len(active)):
        k = int(active[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += w[idx] * a[k - j]
    
    return grad


def compute_smooth_subgrad(a, conv, tau=50.0, top_k=50):
    """Compute smooth approximation of subgradient using softmax weighting."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0

        w /= ws

        je = min(n - 1, k)

                    best_a = a_try.copy()
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 200:
            lr = max(0.001, lr * 0.97)
            momentum = min(0.97, momentum + 0.003)
            tau = min(300.0, tau * 1.02)
        elif no_improve > 50:
            tau = max(20.0, tau * 0.99)
        else:
            tau = min(200.0, tau * 1.003)

        if no_improve > 800:
            rng = np.random.default_rng(it * 13 + 3)
            noise = rng.standard_normal(n) * (0.003 + 0.003 * min(no_improve // 800, 5))
            a = project_simplex(best_a + noise, 1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.03
            momentum = 0.9
            tau = 50.0

            print(f"  SG iter {it}: obj={best_obj:.10f}, lr={lr:.5f}, tau={tau:.1f}, no_imp={no_improve}")
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, try multiple values
    and pick the best, redistributing mass proportionally.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Try various multipliers
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_obj

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def anti_palindromic_init(n, rng, strength=0.3):
    """
    Initialize with anti-palindromic structure: large on one side, small on the other.
    """
    # Base: uniform
    a = np.ones(n) / n
    
    # Anti-palindromic perturbation: large values on the left, small on the right
    ramp = np.linspace(1 + strength, 1 - strength, n)
    a = (1.0 + strength * 0.5) * a + strength * 0.5 * (ramp - ramp.mean()) / (ramp.max() - ramp.min()) * 2 / n
    
    # Actually, let me do this more carefully
    a = np.ones(n)
    # Add an asymmetric component
    asym = np.linspace(strength, -strength, n)
    a = 1.0 + asym
    a = np.maximum(a, 0.01)  # keep positive
    a /= a.sum()
    
    # Add some random perturbation
    a += rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)
    a /= a.sum()
    
    return a


def raised_cosine_init(n):
    """Raised cosine window."""
    x = np.arange(n) / (n - 1) if n > 1 else np.array([1.0])
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def sine_window_init(n):
    """Sine window."""
    x = np.arange(1, n + 1) / (n + 1)
    a = np.sin(np.pi * x)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def asymmetric_window_init(n, rng, skew=0.5):
    """
    Asymmetric window: power-law decay from one side.
    a[i] = ((i+1)/(n+1))^p
    """
    x = np.arange(1, n + 1) / (n + 1)
    p = rng.uniform(0.5, 2.0)
    a = x ** p
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
