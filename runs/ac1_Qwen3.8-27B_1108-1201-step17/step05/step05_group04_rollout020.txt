```python
"""
Minimize J(a) = 2*n*max(conv(a,a)) / (sum a)^2  over non-negative sequences.

The problem is convex (the smoothed objective is a convex composition), scale
invariant, so we normalize sum(a)=1 and minimize peak auto-correlation.  We use
  * a annealed FISTA (Nesterov accelerated projected gradient) on the smooth
    log-sum-exp surrogate of the peak, with exact FFT gradients;
  * an exact sub-gradient "peak pushing" polish that drives the *true* max
    auto-correlation down (removes the surrogate bias);
  * a Lagrangian / LP finishing step (the active-set idea: maximize sum(g) s.t.
    (a*g)_k <= M) blended into the current iterate;
  * a length sweep to choose the best n.
All helpers are top level; no lambdas / closures / IO.
"""
import time
import numpy as np

import scipy.optimize as _spopt


# --------------------------------------------------------------------------- #
# Core linear algebra helpers
# --------------------------------------------------------------------------- #
def auto_conv(a):
    """Full linear auto-convolution a*a via real FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_beta(b, beta):
    x = np.asarray(b, dtype=np.float64) * beta
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        out = np.zeros_like(x)
        out[np.argmax(b)] = 1.0
        return out
    return e / s


def grad_from_s(a, s):
    """
    Gradient of (1/beta)*logsumexp(beta*b) w.r.t. a, where b = a*a.
    grad_j = 2 * sum_k s_k * a_ext[k-j]  = 2 * linear_crosscorr(a, s)[j].
    """
    n = len(a)
    N = 1 << (3 * n).bit_length()
    fa = np.fft.fft(a, N)
    fs = np.fft.fft(s, N)
    cc = np.fft.ifft(conj(fa) * fs, N).real
    return 2.0 * cc[:n]


def proj_simplex(v):
    """Euclidean projection onto {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (css - 1.0)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (css[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def score(a):
    """True objective value J (scale invariant, assumes sum>0)."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    peak = auto_conv(a).max()
    return float(2.0 * len(a) * peak / (s * s))

    return float(auto_conv(a).max())  # we track true peak, not surrogate


def fista_anneal(a0, deadline, betas=(6.0, 20.0, 80.0, 300.0),
                 per_stage_min=0.05):
    """
    FISTA with backtracking line search on simplex, annealing temperature beta
    from smooth to sharp.  Returns best iterate (true peak tracking).
    """
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    n = len(a)
    best = a.copy()
    best_peak = auto_conv(a).max()

    k = 1
    y = a.copy()
    t_acc = 0.0  # stage counter for sharing

    for beta in betas:
        stage_deadline = min(deadline, time.monotonic() + per_stage_min * 1000)
        # rough Lipschitz start
        L = 4.0 * beta * max(1.0, 2.0 * n)
        while time.monotonic() < deadline:
            b = auto_conv(y)
            s = softmax_beta(b, beta)
            g = grad_from_s(y, s)

            # backtracking to satisfy quadratic upper bound of surrogate
            accepted = False
            for _ in range(40):
                x_new = proj_simplex(y - g / L)
                # surrogate decrease check using smooth value
                def surf(z):
                    zz = auto_conv(z)
                    m = zz.max()
                    return float((m + np.log(np.exp(zz * beta - m * beta).sum())) / beta)
                denom = np.sum((x_new - y) ** 2)
                rhs = surf(y) + float(np.dot(g, x_new - y)) + 0.5 * L * denom
                if surf(x_new) <= rhs + 1e-10:
                    accepted = True
                    break
                L = 2.0 * L
            if not accepted:
                x_new = proj_simplex(y - g / L)

            if k == 1:
                y_new = x_new
            else:
                t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
                y_new = x_new + ((k - 1.0) / t) * (x_new - x_prev)
            x_prev = x_new
            y = y_new
            k += 1

            x = x_new
            peak = auto_conv(x).max()
            if peak < best_peak - 1e-14:
                best_peak = peak
                best = x.copy()
        # end stage
    return best, best_peak

def exact_polish(a0, deadline, max_iter=40000, rel_tol=1e-6):
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    n = len(a)
    best = a.copy()

    step = 0.05

def lp_blend(a0, deadline, steps=6, mix=0.25):
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    n = len(a)
    best = a.copy()
    best_peak = auto_conv(a).max()

    if n > 1600:
        # build A lazily is still fine, but keep it bounded by skipping large n
        return best, best_peak

    for _ in range(steps):
        if time.monotonic() >= deadline:
            break
        b = auto_conv(a)
        M = float(b.max())
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if hi >= lo:

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = _spopt.linprog(c, A_ub=A, b_ub=b_ub,
                             bounds=[(0.0, None)] * n, method='highs')
        if res.status != 0 or res.x is None:
            break
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-12:
            break
        g = g / g.sum()
        a2 = proj_simplex((1.0 - mix) * a + mix * g)
        p2 = auto_conv(a2).max()
        if p2 < best_peak - 1e-14:
            best_peak = p2
            best = a2.copy()
            a = a2
    return best, best_peak

def interp_to(base, new_n):
    old_n = len(base)
    if old_n == new_n:
        return base.copy()
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    out = np.interp(nx, ox, base)
    return proj_simplex(np.maximum(out, 0.0))

# Orchestration
# --------------------------------------------------------------------------- #
def propose_candidate(seed=42, budget_s=1800, **kwargs):
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def consider(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return None
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.size == 0 or a.sum() < 0.01 or not np.isfinite(a).all():
            return None
        try:
            sc = float(evaluate_sequence([float(v) for v in a]))
        except Exception:
            return None
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print("  NEW BEST [%s] score=%.8f n=%d" % (label, sc, a.size))
        return sc

    # ---- try to use the provided parent ----------
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-9:
            parent = p / p.sum()
            consider(parent, "parent")
            print("Parent loaded: n=%d score=%.8f" % (parent.size, best_score))
    except Exception:
        pass

    if best_seq is None:
        base = proj_simplex(np.ones(768))
        consider(base, "uniform768")

    # ---- Stage 1: quick length sweep to pick a good dimension --------
    n_parent = len(best_seq)
    candidates = [256, 320, 384, 448, 512, 640, 768, 896, 1024, 1280, 1536, 2048]
    candidates = [c for c in candidates if 128 <= c <= 2560]
    candidates.sort(key=lambda c: abs(c - n_parent))
    # keep the parent length first, then a handful of neighbours
    sweep_n = []
    for c in candidates:
        if c not in sweep_n:
            sweep_n.append(c)
    sweep_n = sweep_n[:6]

    t_sweep_end = time.monotonic() + min(500.0, 0.32 * float(budget_s))
    per_n = (t_sweep_end - time.monotonic()) / len(sweep_n)
    print("Length sweep over %s, ~%.0fs each" % (sweep_n, per_n))

    for nn in sweep_n:
        if time.monotonic() > t_sweep_end - 5:
            break
        rem = t_sweep_end - time.monotonic()
        if rem < 12:
            break
        start = interp_to(best_seq, nn)
        d = min(rem * 0.9, 120.0)
        a_opt, _ = fista_anneal(start, time.monotonic() + d,
                                betas=(6.0, 25.0, 120.0), per_stage_min=0.02)
        consider(a_opt, "lsweep_n%d" % nn)

    # ---- Stage 2: deep optimization on the current best length -------
    print("Deep optimize: n=%d score=%.8f" % (len(best_seq), best_score))
    n_best = len(best_seq)
    inits = make_inits(n_best, rng, parent if (parent is not None and len(parent) == n_best) else None)
    deep_deadline = time.monotonic() + min(650.0, 0.34 * float(budget_s))
    for name, init in inits.items():
        if time.monotonic() > deep_deadline - 15:
            break
        a_opt, _ = fista_anneal(init, time.monotonic() + min(deep_deadline - time.monotonic() - 5, 180.0),
                                betas=(6.0, 20.0, 80.0, 300.0))
        consider(a_opt, "deep_" + name)

    # ---- Stage 3: exact peak polish + LP finish on the best ----------
    if time.monotonic() < deadline - 60:
        d = min(deadline - time.monotonic() - 40, 320.0)
        a_opt, _ = exact_polish(best_seq, time.monotonic() + d * 0.55)
        consider(a_opt, "polish")
        if time.monotonic() < deadline - 25:
            a_opt, _ = lp_blend(best_seq, time.monotonic() + min(deadline - time.monotonic() - 15, 120.0))
            consider(a_opt, "lpblend")
        # one more polish on the LP result
        if time.monotonic() < deadline - 25:
            a_opt, _ = exact_polish(best_seq, time.monotonic() + min(deadline - time.monotonic() - 12, 120.0))
            consider(a_opt, "polish2")

    # ---- Stage 4: small-perturbation restarts on the best length -----
    if time.monotonic() < deadline - 30:
        for i in range(5):
            if time.monotonic() > deadline - 12:
                break
            noisy = proj_simplex(best_seq + rng.standard_normal(n_best) * 0.004)
            a_opt, _ = exact_polish(noisy, time.monotonic() + min(deadline - time.monotonic() - 8, 40.0))
            consider(a_opt, "pert_%d" % i)

    # ---- final verification & fallback ----------
    if best_seq is None:
        best_seq = proj_simplex(np.ones(768))
    consider(best_seq, "final")

    out = [float(max(0.0, v)) for v in best_seq]
    print("FINAL returned n=%d score=%.8f" % (len(out), best_score))
    return out
```
