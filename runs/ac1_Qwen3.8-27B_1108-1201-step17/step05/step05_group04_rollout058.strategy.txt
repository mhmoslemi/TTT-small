```python
import numpy as np
def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    # 確認輸入是列表
    if not isinstance(sequence, list):
        return np.inf
    # 拒絕空列表
    if not sequence:
        return np.inf
    # 檢查每個元素的有效性與類型（拒絕布林和非數字數值）
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
            
    # 轉換為浮點數確保一致性
    sequence = [float(x) for x in sequence]
    # 保護對稱性和條列化，將負值設為0
    sequence = [max(0, x) for x in sequence]
    # 轉換為確保存在性與安全隱含上界
    for i in range(len(sequence)):
        if sequence[i] > 1000.0:
            sequence[i] = 1000.0
    
    n = len(sequence)
    # 將序列存儲為數組以便快速計算
    seq_arr = np.array(sequence)
    # 使用FFT進行全模式自卷積（不是循環卷積）
    convolve_result = np.convolve(seq_arr, seq_arr, mode="full")
    # 保護data泄漏並正確計算分數
    max_conv = float(convolve_result.max())
    sum_a = float(seq_arr.sum())
    # 最終公式確認：2n * max_conv / (sum_a)^2且至少0.01分子和分母之比
    if n == 0:
        return np.inf
    numerator = 2 * n * max_conv
    denominator = (sum_a ** 2)
    # 確認避免除以零（非嚴格，但仍需保護）
    if denominator < 0.01:
        return np.inf
    return float(numerator / denominator)
```
