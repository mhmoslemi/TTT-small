```python
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, p):
    """
    Compute gradient of smoothed max-convolution objective.
    a: sequence of length n, sum(a) = 1
    p: softmax weights over 2n-1 convolution positions, sum(p) = 1
    Returns: gradient of sum_k p[k]*b[k] wrt a, where b = conv(a,a)
    This equals (a*p)[j] + (a*p)[2n-2-j] for j in 0..n-1
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    ab = np.fft.irfft(A * P, N)
    grad = ab[:n] + ab[2*n-2-n+1:2*n-1][::-1]  # ab[2n-2-j] for j=0..n-1
    # Actually: ab[2n-2-j] for j=0 is ab[2n-2], for j=n-1 is ab[n-1]
    # So grad[j] = ab[j] + ab[2n-2-j]
    # ab has length 3n-2 (indices 0 to 3n-3)
    # We need ab[2n-2-j] for j=0..n-1, which is ab[2n-2], ab[2n-3], ..., ab[n-1]
    # That's ab[n-1:2n-1][::-1]
    grad = ab[:n] + ab[n-1:2*n-1][::-1]
    return grad
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based efficient gradient: grad[j] = (a*p)[j] + (a*p)[2n-2-j]
2. Nesterov-accelerated projected gradient descent with adaptive log-sum-exp temperature

Key innovation: O(n log n) gradient computation via single FFT, combined with
coarse-to-fine temperature annealing and Nesterov momentum for faster convergence.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_grad(a, p):
    """
    Compute gradient of sum_k p[k]*b[k] where b=conv(a,a).
    Uses the identity: grad[j] = (a*p)[j] + (a*p)[2n-2-j]
    a: length n, p: length 2n-1
    Returns: gradient of length n
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    ab = np.fft.irfft(A * P, N)
    grad = ab[:n] + ab[n - 1:2 * n - 1][::-1]
    return grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(b, beta):
    """Compute softmax weights over convolution values."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
        return w
    return exp_shifted / total


def objective_val(a):
    """Compute 2*n*max_b / (sum_a)^2 for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def nesterov_gd(a0, time_limit, beta_start=5.0, beta_end=500.0, lr_start=None,
                 verbose=False, label=""):
    """
    Nesterov-accelerated projected gradient descent with adaptive log-sum-exp.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    if lr_start is None:
        lr_start = n * 0.5

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    a_prev = a.copy()
    momentum = 0.9
    beta = beta_start
    lr = lr_start
    no_improve = 0

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        b_max = b.max()

        p = softmax_weights(b, beta)
        grad = fast_convolve_grad(a, p)

        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        a_candidate = project_simplex(y - lr * grad)

        # Backtracking line search
        cur_obj = fast_convolve(a).max()
        best_alpha = 1.0
        best_trial_obj = fast_convolve(a_candidate).max()

        if best_trial_obj >= cur_obj:
            for alpha in [0.5, 0.25, 0.1, 0.05, 0.01]:
                a_trial = project_simplex(y - lr * alpha * grad)
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_trial_obj:
                    best_trial_obj = obj_trial
                    best_alpha = alpha
                else:
                    break
            a_candidate = project_simplex(y - lr * best_alpha * grad)
            best_trial_obj = fast_convolve(a_candidate).max()

        a_prev = a
        a = a_candidate

        if best_trial_obj < best_obj - 1e-14:
            best_obj = best_trial_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    NesterovGD[{label}] iter {it}: max_b={best_obj:.10f}, "
                      f"beta={beta:.1f}, lr={lr:.4f}")
        else:
            no_improve += 1

        # Anneal beta (coarse to fine)
        beta = min(beta * 1.008, beta_end)

        # Adapt lr
        if no_improve % 100 == 0 and no_improve > 0:
            lr *= 0.92
            momentum = min(momentum + 0.002, 0.99)

        # Restart if stuck
        if no_improve > 600:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.003 + 0.005 * min(no_improve // 600, 10)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            a_prev = a.copy()
            lr = max(lr * 0.5, n * 0.05)
            beta = max(beta_start, beta * 0.3)
            no_improve = 0

        # Hard stop near deadline
        if time.monotonic() - t0 > time_limit - 3:
            break

    return best_a, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                m_exp *= 1.2
                if m_exp > 20.0:
                    m_exp = 1.0
                continue
            g = res.x
        except Exception:
            m_exp *= 1.2
            if m_exp > 20.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.1, 10.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Bisection refinement
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 10.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(no_improve // 80, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

        if time.monotonic() - t0 > time_limit - 5:
            break

    return best_f, best_obj


def coordinate_descent_fine(a0, time_limit, n_trials_per_coord=3):
    """Fine coordinate descent with multiplicative perturbations."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    multipliers = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 2:
                return best_a, best_obj

            old_val = a[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial_sum = a_trial.sum()
                if a_trial_sum < 1e-15:
                    continue
                a_trial = a_trial / a_trial_sum

                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def random_local_search(a0, time_limit, n_restarts=50):
    """Random perturbation local search (zero-temperature Metropolis)."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    scale = 0.002

    while time.monotonic() - t0 < time_limit:
        # Random perturbation
        a_trial = a + rng.standard_normal(n) * scale
        a_trial = project_simplex(a_trial)
        obj_trial = fast_convolve(a_trial).max()

        if obj_trial < best_obj - 1e-14:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial
            # Occasionally increase scale
            if rng.random() < 0.1:
                scale = min(scale * 1.1, 0.05)
        else:
            # Decrease scale
            scale *= 0.995
            if scale < 1e-6:
                scale = 0.005

        # Occasionally do a swap
        if n > 10 and rng.random() < 0.05:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                a_trial = a.copy()
                swap_amount = min(a[i], a[j]) * 0.1
                a_trial[i] -= swap_amount
                a_trial[j] += swap_amount
                if a_trial[i] >= 0 and a_trial[j] >= 0:
                    obj_swap = fast_convolve(a_trial).max()
                    if obj_swap < best_obj - 1e-14:
                        best_obj = obj_swap
                        best_a = a_trial.copy()
                        a = a_trial

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_initialization(n, seed=42):
    """Generate diverse initializations for multi-restart."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    tri = np.arange(1, n + 1)
    inits.append(tri / tri.sum())

    # Exponential decay
    exp_a = np.exp(-np.arange(n) / (n * 0.3))
    inits.append(exp_a / exp_a.sum())

    # Two spikes at ends
    two_spikes = np.zeros(n)
    two_spikes[0] = 0.4
    two_spikes[-1] = 0.4
    two_spikes[:] += 0.001
    inits.append(two_spikes / two_spikes.sum())

    # Half cosine
    cos_a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    cos_a += 1e-6
    inits.append(cos_a / cos_a.sum())

    # Random positive
    rand_a = rng.exponential(1.0, size=n)
    inits.append(rand_a / rand_a.sum())

    # Bump in middle
    bump = np.exp(-0.5 * ((np.arange(n) - n / 2) / (n * 0.15)) ** 2)
    bump += 1e-6
    inits.append(bump / bump.sum())

    # Ramp up then down (asymmetric triangle)
    ramp = np.zeros(n)
    mid = int(n * 0.4)
    ramp[:mid] = np.linspace(0.1, 1.0, mid)
    ramp[mid:] = np.linspace(1.0, 0.1, n - mid)
    inits.append(ramp / ramp.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Nesterov GD on parent (300s)
    Phase 2: LP-FW on best (200s)
    Phase 3: Multi-restart with diverse inits (300s)
    Phase 4: Multi-length exploration (300s)
    Phase 5: Deep optimization (400s)
    Phase 6: Fine local search and polishing (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform initialization n=512")

    n_parent = len(parent)

    # === Phase 1: Nesterov GD on parent ===
    print(f"\n=== Phase 1: Nesterov GD on parent (n={n_parent}) ===")
    t1 = min(300, deadline - time.monotonic() - 1100)
    if t1 > 30:
        a_opt, _ = nesterov_gd(parent, time_limit=t1, beta_start=5.0,
                               beta_end=500.0, verbose=True, label="parent")
        try_seq(a_opt, "nesterov_parent")

        # Also try with different beta schedule
        t1b = min(100, deadline - time.monotonic() - 1000)
        if t1b > 20:
            a_opt, _ = nesterov_gd(parent, time_limit=t1b, beta_start=2.0,
                                   beta_end=200.0, lr_start=n_parent * 0.3,
                                   label="parent_slow")
            try_seq(a_opt, "nesterov_parent_slow")

    # Try reversed parent
    t1r = min(60, deadline - time.monotonic() - 950)
    if t1r > 15:
        a_rev = parent[::-1].copy()
        a_opt, _ = nesterov_gd(a_rev, time_limit=t1r, label="reversed")
        try_seq(a_opt, "nesterov_reversed")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(200, deadline - time.monotonic() - 850)
    if t2 > 30:
        a_opt, _ = lp_fw_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_main")

    # === Phase 3: Multi-restart with diverse initializations ===
    print(f"\n=== Phase 3: Multi-restart (best={best_score:.6f}) ===")
    base_n = len(base)
    inits = diverse_initialization(base_n, seed=seed)
    t3_total = min(300, deadline - time.monotonic() - 550)
    if t3_total > 30:
        t_per_init = max(20, t3_total / len(inits))
        for i, init_a in enumerate(inits):
            rem = deadline - time.monotonic() - 500
            if rem < 20:
                break
            t_alloc = min(t_per_init, rem)
            a_opt, _ = nesterov_gd(init_a, time_limit=t_alloc,
                                    beta_start=3.0, beta_end=300.0,
                                    label=f"init_{i}")
            try_seq(a_opt, f"diverse_{i}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [512, 640, 768, 896, 1024, 1280]
    for target_n in target_lengths:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 350
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(60, rem * 0.35)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_gd(f_interp, time_limit=t_alloc,
                                beta_start=5.0, beta_end=400.0,
                                label=f"len_{target_n}")
        try_seq(a_opt, f"len_{target_n}")

        # Also try LP-FW briefly on the interpolated
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc2 = min(30, rem * 0.3)
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc2)
            try_seq(a_opt, f"len_{target_n}_lpfw")

    # === Phase 5: Deep optimization ===
    print(f"\n=== Phase 5: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: Nesterov GD with fine beta
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(200, rem * 0.5)
            a_opt, _ = nesterov_gd(best_seq, time_limit=t_alloc,
                                    beta_start=50.0, beta_end=2000.0,
                                    lr_start=n_best * 0.2,
                                    verbose=True, label="deep_fine")
            try_seq(a_opt, "deep_nesterov_fine")

        # 5b: Nesterov GD with coarse beta (escape mechanism)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(100, rem * 0.4)
            a_opt, _ = nesterov_gd(best_seq, time_limit=t_alloc,
                                    beta_start=1.0, beta_end=50.0,
                                    lr_start=n_best * 0.5,
                                    label="deep_coarse")
            try_seq(a_opt, "deep_nesterov_coarse")

        # 5c: LP-FW deep
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(100, rem * 0.5)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 5d: Coordinate descent
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(80, rem * 0.6)
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Random local search
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(60, rem * 0.4)
            a_opt, _ = random_local_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "rand_local")

        # Nesterov with very fine settings
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(40, rem * 0.5)
            a_opt, _ = nesterov_gd(best_seq, time_limit=t_alloc,
                                    beta_start=100.0, beta_end=5000.0,
                                    lr_start=n_best * 0.1,
                                    label="ultra_fine")
            try_seq(a_opt, "ultra_fine")

        # Coordinate descent final
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "final_cd")

        # One more Nesterov pass
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = nesterov_gd(best_seq, time_limit=min(rem - 1, 20),
                                    beta_start=80.0, beta_end=2000.0,
                                    lr_start=n_best * 0.15)
            try_seq(a_opt, "final_nesterov")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
