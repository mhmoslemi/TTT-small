"""
Convex-convolution minimizer: exploits the convexity of min max_k conv_k(a)
with improved initializations (central-binomial "flat-conv" sequence), low-temperature
log-sum-exp Adam optimization, exact subgradient polishing, and coordinate descent.
Targets score < 1.5027 for sequences of length ~768-2048.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def binomial_init(n):
    """Sequence a[k] = C(2k,k)/4^k normalized to sum=1.
    This gives approximately flat linear self-convolution."""

    # Use log for numerical stability
    # log(C(2k,k)) = log(2k!) - 2*log(k!)
    # Use the relation: C(2k,k)/4^k = prod_{j=1}^{k} (2j-1)/(2j)
    # Or: a[0]=1, a[k] = a[k-1] * (2k-1)/(2k)
    a = np.ones(n)
    for kk in range(1, n):
        a[kk] = a[kk-1] * (2*kk - 1) / (2*kk)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def geometric_init(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def symmetric_geo_init(n, r):
    """Symmetric rising-falling geometric."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    a = a[:n] / a.sum()
    return a


def cosine_init(n, power=2.0):
    """Cosine window raised to power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()


def power_law_init(n, p):
    """a[k] = (k+1)^(-p) normalized."""
    k = np.arange(1, n + 1, dtype=np.float64)
    a = k ** (-p)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def smooth_grad(f, T):
    """Gradient of logsumexp(conv(f,f)/T)."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    # grad_i = (2/T) * sum_j f[j] * w[i+j] (cross-correlation)
    N = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w


def smooth_optimize(f0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Adam optimization of log-sum-exp smooth approximation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = elapsed / time_limit
        T = T_start * (T_end / T_start) ** (frac ** 0.7)
        T = max(T, T_end)
        lr = 0.012 * (0.3 ** (frac ** 0.5))
        grad, lse_val, conv, w = smooth_grad(f, T)
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    smooth iter {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1
        if no_improve > 4000:
            seed_val = int(elapsed * 1000) % 999983
            rng_p = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.003)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    return best_f, best_obj


def subgradient_polish(f0, n_iters=5000, verbose=False):
    """Exact subgradient method on the simplex (convex problem)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_obj_nonsmooth = best_obj
    # Use diminishing step size
    C = 0.1 * best_obj
    for it in range(n_iters):
        conv = fast_convolve(f)
        cur_max = conv.max()
        # Find active set (positions near max)
        threshold = cur_max * 0.999
        active = np.where(conv >= threshold)[0]
        # Compute subgradient: average gradient over active positions
        # grad_i = sum_{k in active} (1/|active|) * 2 * a[k-i] (approx)
        grad = np.zeros(n)
        for k in active:
            # conv_k = sum_j a[j]*a[k-j], d/d a[i] = 2*a[k-i] (if 0<=k-i<n and i!=k-i)
            # or a[k-i] if i == k-i
            # Simplified: for each active k, the gradient w.r.t. a[i] is:
            # a[k-i] if 0 <= k-i < n, times 2 (or 1 if k-i == i)
            valid_start = max(0, k - n + 1)
            valid_end = min(k, n - 1)
            for i in range(valid_start, valid_end + 1):
                j = k - i
                if j == i:
                    grad[i] += 1.0
                else:
                    grad[i] += 1.0
        grad /= len(active)
        # Actually let me just use the FFT-based approach for the subgradient
        # The subgradient of max_k conv_k is the gradient of conv_{k*} for active k*
        # For a single active k: grad[i] = 2*a[k-i] if 0<=k-i<n, 0 otherwise
        # (simplified, ignoring the symmetric correction)
        grad = np.zeros(n)
        for k in active:
            if k < n:
                grad[:n-k] += f[:n-k] * 2  # Hmm this is getting complicated
            # Let me just use the smooth gradient with very low T
        break  # Skip complex subgradient, use smooth instead

    return best_f, best_obj


def subgrad_polish_fft(f0, n_iters=3000, verbose=False):
    """Subgradient method using FFT to compute gradients of active conv positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    for it in range(n_iters):
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Active set: positions where conv[k] > 0.99 * max
        threshold = cur_max * 0.995
        active = np.where(conv >= threshold)[0]

        # Compute subgradient as weighted average of gradients
        # For conv_k, grad[i] = 2*a[k-i] if 0<=k-i<n (and i in valid range)
        # Using cross-correlation: for active k, the "gradient" at position i is
        # proportional to a[k-i]

        # More efficient: compute the gradient contribution using FFT
        # The gradient of conv_k w.r.t. a is: (P_k + P_k^T) a / 2 * 2 = (P_k + P_k^T) a
        # where (P_k a)[i] = a[k-i] if valid
        # For simplicity, use the fact that for a single k:
        # grad[i] = 2 * a[k-i] for i in [max(0,k-n+1), min(k,n-1)]
        # For multiple active k's, average them.

        grad = np.zeros(n)
        # Use the w-weighted approach (softmax of conv values)
        # This gives the "smooth" subgradient
        scaled_conv = conv / 0.01  # Low temperature
        scaled_conv -= scaled_conv.max()
        w = np.exp(scaled_conv)
        w /= w.sum()

        # grad[i] = (2/T) * sum_j f[j] * w[i+j] for the smooth version
        # For the exact subgradient, use the active set:
        # grad[i] = (2/|active|) * sum_{k in active} a[k-i] (if valid)
        # This is cross-correlation of f with the indicator of active positions
        active_w = np.zeros(2 * n - 1)
        active_w[active] = 1.0 / len(active)
        N = 1 << (3 * n - 2).bit_length()
        F_fft = np.fft.rfft(f, N)
        W_fft = np.fft.rfft(active_w, N)
        grad = np.fft.irfft(F_fft * np.conj(W_fft), N)[:n] * 2.0

        # Projected subgradient step
        # Use diminishing step size
        step = (best_obj * 0.5) / np.sqrt(it + 1)
        step = min(step, 0.1)
        f_new = f - step * grad
        f_new = project_simplex(f_new)

        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()

        f = f_new

        if verbose and it % 500 == 0:
            print(f"    subgrad iter {it}: obj={best_obj:.10f}, |active|={len(active)}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with line search on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            # Try several perturbations
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_local = best_obj
            best_new_val = old_val

            # Try ratio changes
            for frac in [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]:
                new_val = old_val * frac
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                obj = fast_convolve(f_trial).max()
                if obj < best_local:
                    best_local = obj
                    best_new_val = new_val

            if best_new_val != old_val:
                scale = (1.0 - best_new_val) / denom
                f[j] = best_new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = np.maximum(f, 0)
                ss = f.sum()
                if ss > 1e-15:
                    f /= ss
                if best_local < best_obj - 1e-14:
                    best_obj = best_local
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def pairwise_cd(f0, time_limit, verbose=False):
    """Random pairwise coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        n_pairs = min(200 * (n // 100 + 1), 2000)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            best_local = best_obj
            best_ratio = oi / tot

            for ratio in [0.1, 0.3, 0.5, 0.7, 0.9]:
                if abs(ratio - oi / tot) < 0.05:
                    continue
                f[i] = ratio * tot
                f[j] = (1 - ratio) * tot
                obj = fast_convolve(f).max()
                if obj < best_local:
                    best_local = obj
                    best_ratio = ratio
                # Restore
                f[i] = oi
                f[j] = oj

            if abs(best_ratio - oi / tot) > 1e-10:
                f[i] = best_ratio * tot
                f[j] = (1 - best_ratio) * tot
                if best_local < best_obj - 1e-14:
                    best_obj = best_local
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy convex optimizer for min max_k conv_k(a) s.t. sum(a)=1, a>=0.
    Uses: binomial/flat-conv initialization, low-T Adam, subgradient polishing,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old:.6f}]")

    # === Phase 0: Initializations ===
    print("=== Phase 0: Initializations ===")
    t_phase0_end = time.monotonic() + 30

    for n_try in [n_parent, 1024]:
        if time.monotonic() > t_phase0_end:
            break
        # Binomial (flat-conv) init
        b = binomial_init(n_try)
        _try(b, f"binom_n{n_try}")

        # Power law
        for p in [0.3, 0.5, 0.7, 1.0]:
            pl = power_law_init(n_try, p)
            _try(pl, f"pow{p}_n{n_try}")

        # Geometric
        for r in [0.98, 0.99, 0.995, 0.999]:
            g = geometric_init(n_try, r)
            _try(g, f"geo{r}_n{n_try}")

        # Cosine
        for pw in [1.0, 1.5, 2.0, 3.0]:
            c = cosine_init(n_try, pw)
            _try(c, f"cos{pw}_n{n_try}")

    # Blend of binomial and uniform
    if best_seq is not None:
        for alpha in [0.3, 0.5, 0.7]:
            n_b = len(best_seq)
            blend = alpha * best_seq + (1 - alpha) * (np.ones(n_b) / n_b)
            _try(blend, f"blend_a{alpha}")

    print(f"  After Phase 0: best={best_score:.6f}")

    # === Phase 1: Main smooth optimization on best ===
    print(f"\n=== Phase 1: Smooth Opt (best={best_score:.6f}) ===")
    base = best_seq.copy()
    n_base = len(base)

    # Long smooth optimization on the current best
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 400)
        print(f"  Smooth opt n={n_base}, t={t_alloc:.0f}s, T=[0.2, 0.003]")
        a_opt, _ = smooth_optimize(base, time_limit=t_alloc, T_start=0.2, T_end=0.003, verbose=True)
        _try(a_opt, "smooth_main")

    # Smooth opt with different T range
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t_alloc = min(rem, 200)
        print(f"  Smooth opt (low T) n={n_base}, t={t_alloc:.0f}s, T=[0.1, 0.002]")
        a_opt, _ = smooth_optimize(best_seq, time_limit=t_alloc, T_start=0.1, T_end=0.002, verbose=True)
        _try(a_opt, "smooth_lowT")

    # === Phase 2: Subgradient polishing ===
    print(f"\n=== Phase 2: Subgradient Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 10:
        t_sg = min(rem, 120)
        # Convert time to iterations
        n_iter = min(int(t_sg * 50), 5000)
        print(f"  Subgrad: {n_iter} iters")
        a_opt, _ = subgrad_polish_fft(best_seq, n_iters=n_iter, verbose=True)
        _try(a_opt, "subgrad")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 10:
        t_cd = min(rem, 100)
        print(f"  Single CD t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        _try(a_opt, "cd_single")

    rem = deadline - time.monotonic() - 200
    if rem > 10:
        t_pw = min(rem, 80)
        print(f"  Pairwise CD t={t_pw:.0f}s")
        a_opt, _ = pairwise_cd(best_seq, time_limit=t_pw)
        _try(a_opt, "cd_pair")

    # === Phase 4: Longer sequences ===
    print(f"\n=== Phase 4: Longer Sequences (best={best_score:.6f}) ===")
    for target_n in [1024, 1536, 2048]:
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break

        # Interpolate best to new length
        f_interp = interpolate_seq(best_seq, target_n)
        # Also try binomial at this length
        f_binom = binomial_init(target_n)
        # Blend
        f_blend = 0.5 * f_interp + 0.5 * f_binom
        f_blend = project_simplex(f_blend)

        _try(f_interp, f"interp_n{target_n}")
        _try(f_binom, f"binom_n{target_n}")
        _try(f_blend, f"blend_n{target_n}")

        # Smooth optimize the best of these at new length
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t1 = min(rem, 60)
            a_opt, _ = smooth_optimize(f_interp, time_limit=t1, T_start=0.15, T_end=0.005)
            _try(a_opt, f"smooth_n{target_n}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Very low temperature smooth
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_final = min(rem, 80)
            print(f"  Final smooth t={t_final:.0f}s, T=[0.05, 0.001]")
            a_opt, _ = smooth_optimize(best_seq, time_limit=t_final,
                                       T_start=0.05, T_end=0.001, verbose=True)
            _try(a_opt, "smooth_final")

        # Subgradient
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            n_iter = min(int(rem * 40), 2000)
            a_opt, _ = subgrad_polish_fft(best_seq, n_iters=n_iter)
            _try(a_opt, "subgrad_final")

        # CD
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 30))
            _try(a_opt, "cd_final")

        # Random perturbations
        for i in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 5:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_optimize(f_pert, time_limit=min(rem, 15),
                                       T_start=0.03, T_end=0.002)
            _try(a_opt, f"pert{i}")

        # Reversed
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            f_rev = best_seq[::-1].copy()
            a_opt, _ = smooth_optimize(f_rev, time_limit=min(rem, 10),
                                       T_start=0.02, T_end=0.002)
            _try(a_opt, "reversed")

    # Fallback
    if best_seq is None:
        best_seq = binomial_init(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
