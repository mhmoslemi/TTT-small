import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the non-negative simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

    the peak of the convolution, using a linear program.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based refinement,
gradient descent with adaptive active sets, and length exploration.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    """Compute the objective (without calling evaluate_sequence)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def gradient_optimize(f0, time_limit, lr=0.005, temp_init=0.01, temp_decay=0.999):
    """Adam with soft-max subgradient for fast initial optimization."""
    f = f0.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    cur_lr = lr
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        
        # Soft-max weights
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # This is cross_convolve(w, f) but only the valid part
        # Actually, we need: for each j, sg[j] = sum over valid i of w[i+j]*f[i]
        # This is the 'valid' mode correlation
        sg = np.correlate(w, f, 'valid')
        
        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg

        f = project_simplex(f_new)

        cur_lr = max(lr * (0.9999 ** t), lr * 0.05)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.5, 0.003)
            cur_lr = lr
            no_improve = 0
    
    return best_f, best_obj

    LP-based refinement. At each step:
    1. Find active set (convolution values near the max)
    2. Solve LP to find direction that uniformly reduces active peaks
    3. Take step and renormalize
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Active set: indices where conv >= M * (1 - active_frac * log(n))
        # Use a threshold that adapts
        threshold = M - max(1e-10, M * 0.02)
        active = np.where(conv >= threshold)[0]
        
        if len(active) < min_active:
            active = np.argsort(conv)[-min_active:]
        
        # Limit active set size
        if len(active) > max_active:
            active = np.argsort(conv)[-max_active:]
        
        n_act = len(active)
        
        # Build LP
        # Variables: d[0..n-1], t  (n+1 total)
        # Minimize: -t (i.e., maximize t)
        # Subject to:
        #   For each k in active: sum_i a[max(0,k-n+1):min(n,k+1)]...

        # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} d_i * a_{k-i}
        
        c = np.zeros(n + 1)
        c[n] = -1  # minimize -t
        
        # Inequality constraints
        A_ub = np.zeros((n_act, n + 1))
        b_ub = np.zeros(n_act)
        
        for idx, k in enumerate(active):
            # (d*a)[k] = sum_i d[i] * a[k-i] for valid i (0 <= i < n, 0 <= k-i < n)
            i_min = max(0, k - (n - 1))

            A_ub[idx, n] = 1.0  # + t

        bounds = [(-a[j], 1.0) for j in range(n)]
        bounds.append((0.0, None))  # t >= 0
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                         bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        d = res.x[:n]
        t_val = res.x[n]
        
        if t_val < 1e-12:
            break

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        conv_new = fast_convolve(a_new).max()
        
        if conv_new < best_obj - 1e-15:
            best_obj = conv_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def coordinate_descent(a, time_limit, max_passes=50):

            best_val = old_val
            best_local = cur_obj
            
            # Try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:

                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s
                
                val = fast_convolve(a_trial).max()
                if val < best_local - 1e-15:
                    best_local = val
                    best_val = new_val
            
            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    a[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()
                
                if best_local < best_obj - 1e-15:
                    best_obj = best_local
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
gradient descent with soft-max subgradients, multi-phase optimization,
and length exploration. Key improvement over prior work: uses linear programming
to find exact descent directions in the space of convolution peaks, combined
with adaptive active sets and line search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the non-negative simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_conv(a):
    """Compute normalized objective components: (conv, n, sum)."""
    n = len(a)
    s = a.sum()
    conv = fast_convolve(a)
    return conv, n, s


def get_objective(a):
    """Compute the full objective value 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def make_gaussian_init(n, center, sigma, floor=1e-10):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, floor)
    return f / f.sum()


def make_symmetric_init(n, shape):
    """Create a symmetric initialization."""
    x = np.linspace(0, 1, n)
    if shape == 'bump':
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'flat':
        f = np.ones(n)
    elif shape == 'edges':
        f = np.abs(2 * x - 1) ** 2
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization as a sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.15, 0.85)
        s = rng.uniform(0.05, 0.25)
        amp = rng.uniform(0.5, 2.0)
        f += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_skewed_init(n, skew_alpha=2.0):
    """Create a Beta-distribution shaped initialization (skewed)."""
    x = np.linspace(0.01, 0.99, n)
    f = x ** skew_alpha * (1 - x) ** skew_alpha
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def gradient_optimize(f0, time_limit, lr=0.005, temp_init=0.008, temp_decay=0.9994,
                      top_k=None, restart_every=250):
    """
    Adam optimizer with soft-max subgradients and simplex projection.
    Fast method for initial optimization when far from optimum.
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    cur_lr = lr

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        # Soft-max weights over convolution
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        # Subgradient: sg[j] = sum_i w[i+j] * f[i] (correlation)
        sg = np.correlate(w, f, 'valid')

        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.99995 ** t), lr * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > restart_every:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def lp_find_direction(a, active_set):
    """
    Solve LP to find direction d that minimizes the maximum first-order
    increase of active convolution peaks.
    
    Minimize: z
    Subject to: (d*a)[k] - z <= 0 for k in active_set
                sum(d) = 0
                -a[j] <= d[j] <= 1
                z free
    
    Returns (d, z) where d is the direction and z is the optimal objective.
    """
    n = len(a)
    n_act = len(active_set)

    # Variables: d[0..n-1], z  (n+1 total)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize z

    # Inequality constraints: (d*a)[k] - z <= 0
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)

    for idx, k in enumerate(active_set):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A_ub[idx, i] = a[k - i]
        A_ub[idx, n] = -1.0
        b_ub[idx] = 0.0

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    # Bounds: -a[j] <= d[j] <= 1, z free
    bounds = [(-a[j], 1.0) for j in range(n)]
    bounds.append((-np.inf, np.inf))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
    except Exception:
        return None, 0.0

    if not res.success:
        return None, 0.0

    d = res.x[:n]
    z = res.x[n]
    return d, z


def lp_refine(a, time_limit, max_iters=1000, active_frac=0.04,
              min_active=3, max_active=200, line_search_points=4):
    """
    LP-based refinement: iteratively find descent direction via LP,
    perform line search, and update.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Active set: convolution values near the peak
        threshold = M * (1.0 - active_frac)
        active = np.where(conv >= threshold)[0]

        if len(active) < min_active:
            active = np.argsort(conv)[-min_active:]
        if len(active) > max_active:
            # Keep only the top max_active
            top_idx = np.argsort(conv)[-max_active:]
            active = top_idx

        # Get LP direction
        d, z = lp_find_direction(a, active)
        if d is None:
            break

        if z >= -1e-12:
            # No first-order descent direction; try smaller active set
            if len(active) > 2:
                active = np.argsort(conv)[-2:]
                d, z = lp_find_direction(a, active)
                if d is None or z >= -1e-12:
                    break
            else:
                break

        # Determine max feasible step
        if np.any(d < 0):
            alpha_max = np.min(a[d < 0] / (-d[d < 0])) * 0.95
        else:
            alpha_max = 1.0
        alpha_max = min(alpha_max, 1.0)

        if alpha_max < 1e-10:
            break

        # Line search: try several step sizes
        alphas = np.linspace(0.1, alpha_max, line_search_points)
        if alpha_max < 0.5:
            alphas = np.linspace(0.05, alpha_max, line_search_points)

        best_alpha = 0.0
        best_alpha_obj = M

        for alpha in alphas:
            a_trial = a + alpha * d
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            obj = fast_convolve(a_trial).max()
            if obj < best_alpha_obj:
                best_alpha_obj = obj
                best_alpha = alpha

        # If no improvement with standard alphas, try tiny step
        if best_alpha == 0.0 and alpha_max > 1e-8:
            tiny = min(0.01, alpha_max * 0.1)
            a_trial = a + tiny * d
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
                obj = fast_convolve(a_trial).max()
                if obj < M - 1e-14:
                    best_alpha = tiny
                    best_alpha_obj = obj

        if best_alpha == 0.0:
            stagnation += 1
            if stagnation > 20:
                break
            # Perturb and continue
            rng = np.random.default_rng(it * 13 + 7)
            a = project_simplex(a + rng.standard_normal(n) * 0.002)
            continue

        # Update
        a = a + best_alpha * d
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if best_alpha_obj < best_obj - 1e-15:
            best_obj = best_alpha_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Adaptive: tighten active set as we converge
        if it > 100 and M - best_obj < best_obj * 0.01:
            active_frac = max(0.01, active_frac * 0.9)

    return best_a, best_obj

def coordinate_polish(a, time_limit, step_subsamp=3):

    # Build index sampling
    indices = list(range(0, n, max(1, step_subsamp)))

    for _ in range(100):  # Max passes
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for j in indices:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_val = old_val
            best_local = best_obj

            # Try several multipliers
            for mult in [0.4, 0.6, 0.8, 1.25, 1.5, 1.8]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue

                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s

                val = fast_convolve(a_trial).max()
                if val < best_local - 1e-14:
                    best_local = val
                    best_val = new_val

            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    a[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()

                if best_local < best_obj - 1e-14:
                    best_obj = best_local
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Try all coordinates (not just subsampled)
            indices_all = list(range(n))
            improved2 = False
            for j in indices_all:
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                best_val = old_val
                best_local = best_obj
                for mult in [0.7, 0.9, 1.1, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.99:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    if scale < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial /= s
                    val = fast_convolve(a_trial).max()
                    if val < best_local - 1e-14:
                        best_local = val
                        best_val = new_val
                if best_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_val) / denom
                        a[j] = best_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        a[mask] *= scale
                        a /= a.sum()
                    if best_local < best_obj - 1e-14:

                        best_a = a.copy()
                        improved2 = True
            if not improved2:
