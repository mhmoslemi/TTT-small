```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding

# ... helper functions ...

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    n = len(f)
    g = cp.Variable(n)
    
    # Build constraint matrix implicitly
    # conv(f,g)[k] = sum_j f[k-j] * g[j] for valid j
    # We need to express this as A @ g <= M
    
    # For efficiency, build the matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)  # or ECOS, SCS
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    
    if g.value is None:
        return None
    
    return np.maximum(g.value, 0)
```

```python
def compute_subgradient_soft(f, temp, top_k=None):
    """Compute soft-max subgradient of max(conv(f,f))."""
    conv = fast_convolve(f)
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws

    if top_k is not None and top_k < len(w):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(w))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked

    # grad[i] = 2 * sum_j f[j] * w[i+j] = 2 * correlate(w, f, 'valid')[i]
    n = len(f)
    grad = 2.0 * np.correlate(w, f, 'valid')
    return grad, w, conv

def active_set_gradient_optimize(f0, time_limit, lr=0.01, top_k=20):
    """Optimize using only the top-k active convolution positions."""
    f = f0.astype(np.float64).copy()

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    alpha = lr
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Get top-k positions
        if top_k is not None and top_k < len(conv):
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
            # Uniform weights on top-k
            w = np.zeros(len(conv))
            w[top_idx] = 1.0 / top_k
        else:
            # Soft top-k
            T = 0.001
            shifted = conv / T
            shifted -= shifted.max()
            w = np.exp(shifted)
            w /= w.sum()

        # Gradient
        grad = 2.0 * np.correlate(w, f, 'valid')

        # Multiplicative update (mirror descent)
        f_new = f * np.exp(-alpha * grad)
        f_new = f_new / f_new.sum()
        f = np.maximum(f_new, 0)
        f = f / f.sum()

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            # Increase alpha slightly

                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                alpha = lr

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        'gauss': make_gaussian_init(n_parent, 0.5, 0.2),
        'raised': make_symmetric_init(n_parent, 'raised_cosine'),
    }

    for name, init in lp_starts.items():
        rem = t_phase1_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.6, 90)
        print(f"  LP start '{name}', t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(init, time_limit=t_alloc, max_iters=200)
        try_seq(a_opt, f"p1_lp_{name}")

    # Also try Adam from parent with LP assistance
    rem = t_phase1_end - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.8, 80)
        print(f"  Adam+LP from parent, t={t_alloc:.0f}s")
        a_opt, obj = adam_optimize(parent, time_limit=t_alloc,
                                   lr=0.004, temp_init=0.004,
                                   temp_decay=0.9993,
                                   top_k=max(15, n_parent // 20),
                                   use_lp_every=30, lp_step_size=0.1)
        try_seq(a_opt, "p1_adam_lp")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: Smooth gradient optimization ===
    print(f"\n=== Phase 2: Gradient optimization (best={best_score:.8f}) ===")
    t_phase2_end = t_start + 750
    n_base = len(base)

    configs = [
        {'lr': 0.006, 'temp_init': 0.003, 'temp_decay': 0.9994, 'top_k': max(15, n_base // 25)},
        {'lr': 0.004, 'temp_init': 0.006, 'temp_decay': 0.9996, 'top_k': max(20, n_base // 15)},
        {'lr': 0.008, 'temp_init': 0.002, 'temp_decay': 0.9992, 'top_k': max(10, n_base // 30)},
        {'lr': 0.005, 'temp_init': 0.005, 'temp_decay': 0.9995, 'top_k': None},
    ]

    for ci, cfg in enumerate(configs):
        rem = t_phase2_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.8, 100)

        print(f"  Adam config {ci} (lr={cfg['lr']}, T={cfg['temp_init']}), t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(base, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"p2_adam_{ci}")

        # Try active set gradient
        rem = t_phase2_end - time.monotonic()
        if rem > 25:
            t_alloc2 = min(rem * 0.5, 60)
            a_opt, _ = active_set_gradient_optimize(base, time_limit=t_alloc2,
                                                    lr=0.01, top_k=max(10, n_base // 30))
            try_seq(a_opt, f"p2_active_{ci}")

    base = best_seq if best_seq is not None else parent

    # === Phase 3: More LP from best ===
    print(f"\n=== Phase 3: Extended LP (best={best_score:.8f}) ===")
    t_phase3_end = t_start + 1100
    n_best = len(base)

    rem = t_phase3_end - time.monotonic()
    if rem > 40:
        t_alloc = min(rem * 0.7, 200)
        print(f"  Extended LP, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(base, time_limit=t_alloc, max_iters=300)
        try_seq(a_opt, "p3_lp_ext")

    # LP with different step sizes
    rem = t_phase3_end - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.6, 100)
        print(f"  LP small steps, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(base, time_limit=t_alloc, max_iters=200,
                                 step_sizes=[0.1, 0.05, 0.03, 0.02, 0.01, 0.005])
        try_seq(a_opt, "p3_lp_small")

    # Perturbed LP starts
    for pi in range(2):
        rem = t_phase3_end - time.monotonic()
        if rem < 25:
            break
        perturbed = project_simplex(base + rng.standard_normal(len(base)) * 0.01)
        t_alloc = min(rem * 0.5, 60)
        a_opt, obj = lp_optimize(perturbed, time_limit=t_alloc, max_iters=100)
        try_seq(a_opt, f"p3_lp_pert_{pi}")

    base = best_seq if best_seq is not None else parent

    # === Phase 4: Coordinate descent polishing ===
    print(f"\n=== Phase 4: Coordinate descent (best={best_score:.8f}) ===")
    t_phase4_end = t_start + 1350
    n_best = len(base)

    rem = t_phase4_end - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.7, 150)
        print(f"  CD, t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent_fast(base, time_limit=t_cd)
        try_seq(a_opt, "p4_cd")

    # CD from perturbed start
    rem = t_phase4_end - time.monotonic()
    if rem > 20:
        perturbed = project_simplex(base + rng.standard_normal(n_best) * 0.005)
        t_cd = min(rem * 0.6, 80)
        a_opt, _ = coordinate_descent_fast(perturbed, time_limit=t_cd)
        try_seq(a_opt, "p4_cd_pert")

    base = best_seq if best_seq is not None else base

    # === Phase 5: Fine-tuning ===
    print(f"\n=== Phase 5: Fine-tuning (best={best_score:.8f}) ===")
    t_phase5_end = t_start + 1550
    n_best = len(base)

    rem = t_phase5_end - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.7, 100)
        print(f"  Fine Adam, t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(base, time_limit=t_alloc,
                                 lr=0.002, temp_init=0.002,
                                 temp_decay=0.9997, top_k=max(10, n_best // 40))
        try_seq(a_opt, "p5_fine")

    # Very low temperature Adam
    rem = t_phase5_end - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.7, 80)
        a_opt, _ = adam_optimize(base, time_limit=t_alloc,
                                 lr=0.001, temp_init=0.001,
                                 temp_decay=0.9998, top_k=max(5, n_best // 50))
        try_seq(a_opt, "p5_fine2")

    # Active set with very small top_k
    rem = t_phase5_end - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.6, 50)
        a_opt, _ = active_set_gradient_optimize(base, time_limit=t_alloc,
                                                lr=0.005, top_k=5)
        try_seq(a_opt, "p5_active5")

    base = best_seq if best_seq is not None else base

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration (best={best_score:.8f}) ===")
    t_phase6_end = t_start + 1700

    target_lengths = [len(base) - 64, len(base) + 64, 1024, 900]
    target_lengths = [t for t in target_lengths if 128 <= t <= 2048 and t != len(base)]
    target_lengths = list(set(target_lengths))[:3]

    for target_n in target_lengths:
        rem = t_phase6_end - time.monotonic()
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 60)

        print(f"  Length n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc,
                                 lr=0.004, temp_init=0.004,
                                 temp_decay=0.9994, top_k=max(10, target_n // 25))
        try_seq(a_opt, f"p6_n{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_final = len(best_seq)

        # Last Adam pass
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem * 0.8, 40)
            a_opt, _ = adam_optimize(best_seq, time_limit=t_alloc,
                                     lr=0.001, temp_init=0.001,
                                     temp_decay=0.9998, top_k=max(5, n_final // 60))
            try_seq(a_opt, "p7_final")

        # Final CD
        rem = deadline - time.monotonic() - 3
        if rem > 8:
            a_opt, _ = coordinate_descent_fast(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "p7_cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
