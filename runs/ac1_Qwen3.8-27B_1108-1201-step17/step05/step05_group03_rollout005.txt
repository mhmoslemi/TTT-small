def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def score_from_conv(a, conv):
    """Score from precomputed convolution."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(2.0 * len(a) * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def geometric_sequence(n, r, offset=0):
    """Geometric sequence with optional offset."""
    j = np.arange(n)
    a = r ** (j + offset)
    return normalize(a)


def asymmetric_geometric(n, r_left, r_right):
    """Two-sided geometric: decays from center."""
    center = n // 2
    a = np.zeros(n)
    for i in range(n):
        if i <= center:
            a[i] = r_left ** (center - i)
        else:
            a[i] = r_right ** (i - center)
    return normalize(a)


def oscillatory_sequence(n, base_freq, amplitude):
    """Oscillatory modulation of a smooth base."""
    x = np.linspace(0, 1, n)
    base = np.sin(np.pi * x) ** 2
    mod = 1.0 + amplitude * np.cos(2.0 * np.pi * base_freq * x)
    a = base * np.maximum(mod, 0.0)
    return normalize(a)


def two_lobe_sequence(n, frac, sigma, gap_frac):
    """Two Gaussian lobes separated by a gap."""
    x = np.arange(n) / n
    center1 = 0.5 - gap_frac / 2
    center2 = 0.5 + gap_frac / 2
    g1 = np.exp(-0.5 * ((x - center1) / sigma) ** 2)
    g2 = np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    a = frac * g1 + (1 - frac) * g2
    return normalize(a)


def spectral_sequence(n, flatness, rng):
    """Design in frequency domain for flat |A|^2, then clip."""
    A = np.zeros(fft_size_for(n), dtype=np.complex128)
    shape = 0.8 + 0.2 * rng.random()
    freq_mask = np.ones(n)

    spectral = np.ones(n)
    for k in range(1, n // 2):
        spectral[k] = 1.0 + (rng.random() - 0.5) * 2 * (1 - flatness)
    A[:n] = spectral
    A[-n+1:] = spectral[:0:-1] if n > 1 else spectral
    a = np.fft.irfft(A, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def coord_descent(a, time_limit, max_passes=5000, verbose=False):
    """Coordinate descent with bounded scalar minimization per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            v = a[j]
            if v < 1e-10:
                candidates = [0.001, 0.005, 0.01, 0.02, 0.05]
            else:
                lo = max(0.0, v * 0.2)
                hi = min(0.95, v * 5.0)
                if hi - lo < 1e-5:
                    lo = max(0.0, v - 0.1)
                    hi = min(0.95, v + 0.1)
                candidates = [lo, hi, (lo + hi) / 2, v * 0.5, v * 1.5, 0.0]
                candidates = [max(0.0, min(0.95, c)) for c in candidates]

            best_local = best_score
            best_v = v

            for v_new in candidates:
                if abs(v_new - v) < 1e-12:
                    continue
                new_a = a.copy()
                new_a[j] = v_new
                new_a = normalize(new_a)
                conv = fast_convolve(new_a)
                sc = 2.0 * n * conv.max()
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_v = v_new

            if best_v != v:
                a = a.copy()
                a[j] = best_v
                a = normalize(a)
                sc = compute_score(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if verbose and pass_count % 50 == 0:
            print(f"      CD pass {pass_count}: score={best_score:.10f}")

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if pass_count >= max_passes:
            break

    return best_a, best_score


def simulated_annealing(a, time_limit, t0_param=0.05, cooling=0.999, n_iter_max=50000, verbose=False):

        # Random perturbation: pick k coordinates and perturb

        indices = rng.integers(0, n, k_perturb)

        for idx in indices:
            if new_a[idx] > 0:
                new_a[idx] *= np.exp(rng.standard_normal() * T * 2)
            else:
                new_a[idx] = rng.random() * T * 0.1

        new_a = project_simplex(new_a)

        sc_new = compute_score(new_a)
        sc_cur = best_score  # compare against current, not best

        # Actually compare with current score
        if True:
            conv_cur = fast_convolve(a)
            sc_cur = 2.0 * n * conv_cur.max()

        delta = sc_new - sc_cur
        if delta < 0 or rng.random() < np.exp(-max(delta, -10) / max(T, 1e-10)):
            a = new_a
            accept_count += 1
            if sc_new < best_score - 1e-14:

def subgradient_method(a, time_limit, lr0=0.01, decay=0.9995, top_k=50, verbose=False):
    """Averaged subgradient with multiplicative updates."""
    n = len(a)
    a = normalize(a.copy())
    conv = fast_convolve(a)
    best_score = compute_score(a)
    best_a = a.copy()

    avg_a = np.zeros(n)
    avg_count = 0
    cur_lr = lr0
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        temp = max(1e-6, pv.max() - pv.min()) * 0.5
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()

        avg_a += a
        avg_count += 1
        avg_candidate = avg_a / avg_count

        conv = fast_convolve(a)
        sc = compute_score(a)
        sc_avg = compute_score(avg_candidate)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        if sc_avg < best_score - 1e-14:
            best_score = sc_avg
            best_a = avg_candidate.copy()

        cur_lr *= decay

        if step % 10000 == 0 and cur_lr < lr0 * 0.1:
            cur_lr = lr0 * 0.5
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            avg_a = a.copy()
            avg_count = 1

    return best_a, best_score


def block_local_search(a, time_limit, block_size=4, verbose=False):

def crossover(a1, a2, rng, alpha=None):
    """Crossover between two sequences of same length."""
    if a1.shape != a2.shape:
        return a1.copy()
    if alpha is None:
        alpha = rng.random()
    a = alpha * a1 + (1 - alpha) * a2
    return normalize(a)

    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_final, sc_final = coord_descent(a, time_limit=min(rem, 150))
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        if verbose:
            print(f"        After final CD: {best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with diverse initializations, simulated annealing,
    coordinate descent, block search, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Generate diverse starting points ===
    print("=== Generating starting points ===")
    starts = []

    # Parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    # Asymmetric geometric sequences
    for n in [640, 768, 896, 1024]:
        for rl in [0.995, 0.998, 0.999]:
            for rr in [0.99, 0.995, 0.999]:
                if abs(rl - rr) > 0.002:
                    starts.append((asymmetric_geometric(n, rl, rr), f"asym_{rl}_{rr}_n{n}"))

    # Oscillatory sequences
    for n in [768, 1024]:
        for freq in [3, 5, 7, 11]:
            for amp in [0.3, 0.5, 0.7]:
                starts.append((oscillatory_sequence(n, freq, amp), f"osc_f{freq}_a{amp}_n{n}"))

    # Two-lobe sequences
    for n in [768, 1024]:
        for frac in [0.3, 0.4, 0.5, 0.6]:
            for gap in [0.1, 0.2, 0.3]:
                starts.append((two_lobe_sequence(n, frac, 0.15, gap), f"2lobe_f{frac}_g{gap}_n{n}"))

    # Standard shapes
    for n in [640, 768, 896, 1024]:
        starts.append((cosine_sequence(n), f"cos_n{n}"))
        starts.append((gaussian_sequence(n, 0.3), f"gauss_n{n}"))
        starts.append((triangular_sequence(n), f"tri_n{n}"))

    # Random smooth
    for n in [768, 1024]:
        for _ in range(3):
            starts.append((rand_smooth_sequence(n, rng), f"rand_n{n}"))

    print(f"  Total starts: {len(starts)}")

    # Quick evaluate all starts
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(start_scores[:5]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # === Phase 1: Optimize top diverse starts ===
    print(f"\n=== Phase 1: Optimize diverse starts ===")

    selected = []
    selected_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 8:
            break
        base = label.rsplit('_n', 1)[0] if '_n' in label else label
        if base not in selected_types:
            selected.append((a, label))
            selected_types.add(base)

    # Always include parent if available
    parent_included = any("parent" in label for _, label in selected)
    if not parent_included and len(start_scores) > 0:
        for sc, a, label in start_scores:
            if "parent" in label:
                selected.append((a, label))
                break

    n_selected = len(selected)
    t_per = min((deadline - time.monotonic() - 400) / max(1, n_selected), 180)

    hall_of_fame = []  # Store (score, a) pairs

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(t_per, rem)
        print(f"  Optimizing [{label}]: t={t_alloc:.0f}s")
        a_opt, sc_opt = multi_stage_optimize(a_init, time_limit=t_alloc)
        try_seq(a_opt, label + "_opt")
        hall_of_fame.append((sc_opt, a_opt))

    # Sort hall of fame
    hall_of_fame.sort(key=lambda x: x[0])
    print(f"  Hall of fame top 3: {[f'{s:.6f}(n={len(a)})' for s, a in hall_of_fame[:3]]}")

    # === Phase 2: Crossover and re-optimize ===
    print(f"\n=== Phase 2: Crossover exploration ===")
    if len(hall_of_fame) >= 2:
        rem = deadline - time.monotonic() - 200
        n_cross = min(6, max(1, int(rem / 60)))
        for ci in range(n_cross):
            if time.monotonic() > deadline - 150:
                break
            i = ci % len(hall_of_fame)
            j = (ci + 1) % len(hall_of_fame)
            if i == j:
                continue
            _, a1 = hall_of_fame[i]
            _, a2 = hall_of_fame[j]
            if a1.shape != a2.shape:
                a2 = interpolate_sequence(a2, a1.shape[0])

            for alpha in [0.3, 0.5, 0.7]:
                a_cross = crossover(a1, a2, rng, alpha)
                # Quick check
                sc_cross = compute_score(a_cross)
                if sc_cross < best_score * 1.02:
                    t_alloc = min(60, (deadline - time.monotonic() - 100) / 2)
                    if t_alloc > 10:
                        a_opt, sc_opt = multi_stage_optimize(a_cross, time_limit=t_alloc)
                        try_seq(a_opt, f"cross_{ci}_{alpha}")

    # === Phase 3: Try longer sequences ===
    print(f"\n=== Phase 3: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        for target_n in [896, 1024, 1280, 1536]:
            if abs(target_n - best_n) < 50:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 120)
            a_interp = interpolate_sequence(best_a, target_n)
            a_opt, sc = multi_stage_optimize(a_interp, time_limit=t_alloc)
            if sc < best_score:
                try_seq(a_opt, f"dim_{target_n}")

    # === Phase 4: Aggressive local search on best ===
    print(f"\n=== Phase 4: Aggressive local search (best={best_score:.8f}) ===")
    if best_a is not None:
        # Long coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_cd = min(rem * 0.4, 200)
            print(f"  Long CD: t={t_cd:.0f}s")
            a_final, sc = coord_descent(best_a, time_limit=t_cd, max_passes=10000)
            try_seq(a_final, "long_cd")

        # Block local search
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_bl = min(rem * 0.5, 100)
            a_bl, sc = block_local_search(best_a, time_limit=t_bl, block_size=3)
            try_seq(a_bl, "block")

        # Simulated annealing on best
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_sa = min(rem * 0.7, 80)
            a_sa, sc = simulated_annealing(best_a, time_limit=t_sa, t0_param=0.02, cooling=0.9995)
            try_seq(a_sa, "sa_final")

        # Final CD polish
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_final, sc = coord_descent(best_a, time_limit=min(rem, 60))
            try_seq(a_final, "final_polish")

    # === Phase 5: One more round of diverse optimization ===
    print(f"\n=== Phase 5: More exploration (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 30 and best_a is not None:
        # Perturb best and re-optimize
        for trial in range(4):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            n_best = len(best_a)
            noise = rng.standard_normal(n_best) * 0.01
            a_pert = project_simplex(best_a + noise)
            t_alloc = min(rem, 40)
            a_opt, sc = multi_stage_optimize(a_pert, time_limit=t_alloc)
            try_seq(a_opt, f"perturb_{trial}")

    # === Final verification ===
    if best_a is None:
        best_a = cosine_sequence(768)
        best_score = compute_score(best_a)

    # Verify with actual evaluate_sequence
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return final_list
