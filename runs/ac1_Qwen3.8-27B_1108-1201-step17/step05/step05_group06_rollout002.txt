"""
Focused optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses U-shaped/multi-modal initializations, mirror descent with log-sum-exp
smoothing, active-set coordinate descent, and simulated annealing for final polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    """Compute smooth gradient of logsumexp(conv(f,f)/T) via FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w

    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_w_shaped(n, alpha=1.0):
    """W-shaped sequence (two peaks at edges, dip in center)."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.cos(2 * np.pi * x) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_double_hill(n, alpha=0.5):
    """Two-hill sequence."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    """Symmetric geometric sequence."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    a = a / a.sum()
    return a


def smooth_optimize_amsgrad(f0, time_limit, T_start=0.15, T_end=0.003, verbose=False):
    """Optimize using log-sum-exp smooth approximation with AMSGrad."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    v_hat_max = np.zeros(n)
    t_adam = 0

    alpha = 0.006
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = time.monotonic()
    total_time = time_limit
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < total_time:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / total_time, 1.0)
        # Cosine annealing for temperature
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        T = max(T, T_end)
        # Learning rate with cosine decay and warmup
        warmup = min(1.0, elapsed / (total_time * 0.05))
        lr = alpha * warmup * (0.3 + 0.7 * (1.0 + np.cos(np.pi * progress)) / 2.0)

        grad, lse_val, conv, w = smooth_grad(f, T)

        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        v_hat_max = np.maximum(v_hat_max, v_hat)

        f = f - lr * m_hat / (np.sqrt(v_hat_max) + eps)
        f = project_simplex(f)

        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Smooth AMS iter {it}: obj={best_obj:.10f}, T={T:.5f}, lr={lr:.6f}")
        else:
            no_improve += 1

        if no_improve > 2000:
            seed_val = int(elapsed * 1000) % 1000000
            rng_p = np.random.default_rng(seed_val)
            noise_scale = 0.002 + 0.003 * min(no_improve // 2000, 3)
            f = project_simplex(best_f + rng_p.standard_normal(n) * noise_scale)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            v_hat_max = np.zeros(n)
            t_adam = 0
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, T_start=0.1, T_end=0.005, verbose=False):
    """Mirror descent (multiplicative weights) with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    total_time = time_limit
    no_improve = 0
    it = 0
    log_f = np.log(f + 1e-15)

    while time.monotonic() - t0 < total_time:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / total_time, 1.0)
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        T = max(T, T_end)
        lr = 0.05 * (0.3 + 0.7 * (1.0 + np.cos(np.pi * progress)) / 2.0)

        grad, lse_val, conv, w = smooth_grad(f, T)

        # Mirror descent update: a_i <- a_i * exp(-lr * grad_i) / Z
        log_f = log_f - lr * grad
        log_f -= log_f.max()  # numerical stability
        f_new = np.exp(log_f)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s_new
        log_f = np.log(f_new + 1e-15)

        cur_obj = fast_convolve(f_new).max()

        if cur_obj < best_obj - 1e-14:
            f = f_new
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Mirror iter {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1

        if no_improve > 1500:
            seed_val = int(elapsed * 1000) % 1000000
            rng_p = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.003)
            log_f = np.log(f + 1e-15)
            no_improve = 0

    return best_f, best_obj


def active_set_cd(f0, time_limit, verbose=False):
    """Coordinate descent focusing on active convolution constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        conv = fast_convolve(f)
        max_val = conv.max()
        active_threshold = max_val * 0.999
        active_mask = conv >= active_threshold

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue

            old_val = f[j]
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            # Try several candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25,
                          old_val * 1.5, 0.9 * denom, 0.95 * denom]
            candidates = [c for c in candidates if 0 <= c < 1.0 - 1e-10]

            best_c = old_val
            best_val = max_val

            for c in candidates:
                if abs(c - old_val) < 1e-8:
                    continue
                scale = (1.0 - c) / denom
                f_trial = f.copy()
                f_trial[j] = c
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_c = c

            if best_c != old_val:
                scale = (1.0 - best_c) / denom
                f[j] = best_c
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_val < best_obj - 1e-14:
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 5:
            break
        if verbose and passes % 10 == 0:
            print(f"    Active CD pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """Fast simulated annealing: random pairwise mass transfers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    T_sa = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal temperature
        elapsed = time.monotonic() - t0
        T_sa = 0.001 * (0.999 ** it)
        T_sa = max(T_sa, 1e-7)

        # Pick two random indices
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        # Move a small amount of mass
        amount = rng.uniform(0.001, 0.01) * min(f[i], f[j])
        if amount < 1e-14:
            continue

        # Try both directions
        for direction in [1, -1]:
            f_trial = f.copy()
            f_trial[i] -= direction * amount
            f_trial[j] += direction * amount
            if f_trial[i] < 0 or f_trial[j] < 0:
                continue

            val = fast_convolve(f_trial).max()
            delta = val - best_obj

            if delta < 0:
                f = f_trial
                best_obj = val
                best_f = f.copy()
                break
            else:
                # Accept with probability exp(-delta / T_sa)
                if rng.random() < np.exp(-delta / T_sa):
                    f = f_trial
                    break

        if verbose and it % 5000 == 0:
            print(f"    SA iter {it}: obj={best_obj:.10f}, T_sa={T_sa:.8f}")

    return best_f, best_obj


def pairwise_cd_fine(f0, time_limit, verbose=False):
    """Pairwise coordinate descent with fine ratio search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77777)
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        n_pairs = min(500, 2 * n // 3)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break

            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue

            old_i = f[i]
            old_j = f[j]
            total_ij = old_i + old_j
            if total_ij < 1e-14:
                continue

            cur_ratio = old_i / total_ij
            best_ratio = cur_ratio
            best_obj_local = fast_convolve(f).max()

            # Fine grid around current ratio
            for delta in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3,
                          -0.45, -0.25, -0.02, 0.02, 0.25, 0.45]:
                ratio = cur_ratio + delta
                if ratio < 0.01 or ratio > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[i] = ratio * total_ij
                f_trial[j] = (1.0 - ratio) * total_ij
                val = fast_convolve(f_trial).max()
                if val < best_obj_local - 1e-15:
                    best_obj_local = val
                    best_ratio = ratio

            if abs(best_ratio - cur_ratio) > 1e-10:
                f[i] = best_ratio * total_ij
                f[j] = (1.0 - best_ratio) * total_ij
                if best_obj_local < best_obj - 1e-14:
                    best_obj = best_obj_local
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 10:
            break

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-method optimizer combining mirror descent, AMSGrad smooth optimization,
    active-set CD, simulated annealing, and fine pairwise CD.
    Uses diverse initializations including U-shaped and W-shaped profiles.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")

    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Diverse Initializations ===")

    for n_try in [n_parent, 768]:
        # U-shaped
        for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
            g = make_u_shaped(n_try, alpha=alpha)
            _try(g, f"u_a{alpha}_n{n_try}")

        # W-shaped
        for alpha in [0.3, 0.5, 1.0, 1.5]:
            g = make_w_shaped(n_try, alpha=alpha)
            _try(g, f"w_a{alpha}_n{n_try}")

        # Double hill
        for alpha in [0.3, 0.5, 0.8]:
            g = make_double_hill(n_try, alpha=alpha)
            _try(g, f"dh_a{alpha}_n{n_try}")

        # Cosine (hill-shaped)
        for pw in [1.5, 2.0, 3.0, 4.0]:
            g = make_cosine(n_try, power=pw)
            _try(g, f"cos_p{pw}_n{n_try}")

        # Geometric
        for r in [0.97, 0.98, 0.99, 0.995]:
            g = make_geometric(n_try, r)
            _try(g, f"geo_r{r}_n{n_try}")
            g = make_symmetric_geo(n_try, r)
            _try(g, f"symgeo_r{r}_n{n_try}")

    print(f"  After Phase 0: best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 1: Smooth optimization (AMSGrad) on best ===
    print(f"\n=== Phase 1: AMSGrad Smooth Opt (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 60:
        t_alloc = min(rem, 500)
        print(f"  AMSGrad smooth n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize_amsgrad(best_seq, time_limit=t_alloc,
                                           T_start=0.12, T_end=0.004, verbose=True)
        _try(a_opt, "amsgrad_p1")

    # === Phase 2: Mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 60:
        t_alloc = min(rem, 300)
        print(f"  Mirror descent n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_alloc,
                                           T_start=0.1, T_end=0.005, verbose=True)
        _try(a_opt, "mirror_p2")

    # Mirror descent from perturbed start
    rem = deadline - time.monotonic() - 400
    if rem > 40:
        noise = rng.standard_normal(len(best_seq)) * 0.005
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 150)
        a_opt, _ = mirror_descent_optimize(f_pert, time_limit=t_alloc,
                                           T_start=0.08, T_end=0.005)
        _try(a_opt, "mirror_pert")

    # === Phase 3: Active-set CD ===
    print(f"\n=== Phase 3: Active-Set CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem, 250)
        print(f"  Active CD t={t_alloc:.0f}s")
        a_opt, _ = active_set_cd(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "active_cd")

    # === Phase 4: Simulated annealing ===
    print(f"\n=== Phase 4: Simulated Annealing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem, 180)
        print(f"  SA t={t_alloc:.0f}s")
        a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "sa_p4")

    # === Phase 5: Fine pairwise CD ===
    print(f"\n=== Phase 5: Pairwise CD Fine (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_alloc = min(rem, 120)
        print(f"  Pairwise CD t={t_alloc:.0f}s")
        a_opt, _ = pairwise_cd_fine(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "pw_cd")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Very low temperature smooth
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem, 80)
            print(f"  Fine smooth t={t_alloc:.0f}s")
            a_opt, _ = smooth_optimize_amsgrad(best_seq, time_limit=t_alloc,
                                               T_start=0.03, T_end=0.002, verbose=True)
            _try(a_opt, "fine_smooth")

        # Active CD polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem, 60)
            a_opt, _ = active_set_cd(best_seq, time_limit=t_alloc)
            _try(a_opt, "active_cd_polish")

        # SA polish
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 50)
            a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc)
            _try(a_opt, "sa_polish")

        # Reversed
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            f_rev = best_seq[::-1].copy()
            a_opt, _ = smooth_optimize_amsgrad(f_rev, time_limit=min(rem, 30),
                                               T_start=0.02, T_end=0.002)
            _try(a_opt, "reversed")

        # Small perturbation restarts
        for idx in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_optimize_amsgrad(f_pert, time_limit=min(rem - 2, 20),
                                               T_start=0.02, T_end=0.002)
            _try(a_opt, f"perturb_{idx}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
