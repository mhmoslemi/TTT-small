        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Compute subgradient
        sg = smooth_subgrad(a, conv, temperature, top_k)
        
        # Momentum update
        m = momentum * m + sg
        
        # Step
        a_new = a - lr * m
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 100 == 0:
                print(f"      Subgrad iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            # Step size reduction
            lr *= 0.95
            no_improve += 1
        
        temperature *= temp_decay
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)

            lr = lr_init * 0.5
            no_improve = 0
            temperature = temperature_init
        elif no_improve > 100:
            lr = max(lr, lr_init * 0.1)
        
        if lr < 1e-8:
            break
    
    return best_a, best_obj


def coordinate_descent_optimize(a0, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Try different values for a[j]
            # Keep sum = 1 by scaling others
            for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * factor
                if new_val < 0 or new_val > 0.9:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def peak_targeted_optimize(a0, time_limit, verbose=False):
    """Target the top peaks specifically by adjusting contributing elements."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        improved = False
        
        # Get top 5 peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            
            if len(idx) == 0:
                continue
            
            # Find the element in idx with maximum value
            j_max = idx[np.argmax(a[idx])]
            old_val = a[j_max]
            if old_val < 1e-12:
                continue
            
            # Try reducing a[j_max] and redistributing
            best_trial_obj = M
            best_trial = None
            
            for factor in [0.5, 0.7, 0.85, 0.95]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                
                trial = a.copy()
                trial[j_max] = new_val
                # Redistribute to minimum elements
                diff_to_add = diff
                # Add to the smallest elements proportionally
                min_mask = a < np.sort(a)[:max(1, n // 10)]
                if not min_mask.any():
                    min_mask = np.ones(n, dtype=bool)
                    min_mask[j_max] = False
                weights = np.where(min_mask, a, 0.0)
                ws = weights.sum()
                if ws > 1e-15:
                    trial += diff_to_add * weights / ws
                else:
                    # Uniform redistribution
                    trial += diff_to_add / (n - 1)
                    trial[j_max] = new_val
                
                trial = np.maximum(trial, 0)
                trial /= trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = obj
                    best_trial = trial
            
            if best_trial is not None:
                a = best_trial
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_subgrad(a, conv, temperature=0.01, top_k=30):
    """Compute smoothed subgradient of max(conv) w.r.t. a using softmax over top peaks."""
    n = len(a)
    k = min(top_k, len(conv))
    if k <= 0:
        return np.zeros(n)
    peak_idx = np.argpartition(conv, -k)[-k:]
    pv = conv[peak_idx]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peak_idx))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peak_idx)):
        wi = w[i]
        if wi < 1e-14:
            continue
        k_idx = int(peak_idx[i])
        js = max(0, k_idx - n + 1)
        je = min(n - 1, k_idx)
        idx = np.arange(js, je + 1)
        grad[idx] += wi * (a[k_idx - idx] + a[idx])
    return grad

def coord_descent(a0, time_limit, verbose=False, factors=None):
    """Coordinate descent optimizing one element at a time."""
    if factors is None:
        factors = [0.4, 0.6, 0.8, 0.9, 1.1, 1.2, 1.4, 1.8]

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            for factor in factors:
                new_val = old_val * factor
                if new_val < 0 or new_val > 0.95:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj


def sim_anneal_escape(a0, time_limit, rng, T_init=0.002, cooling=0.9995):
    """Simulated annealing to escape local minima."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    current_obj = conv.max()
    best_a = a.copy()
    best_obj = current_obj
    T = T_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Random perturbation
        perturb_idx = rng.integers(0, n, size=max(1, n // 50))
        a_trial = a.copy()
        a_trial[perturb_idx] *= np.exp(rng.standard_normal(len(perturb_idx)) * 0.3)
        a_trial = np.maximum(a_trial, 0)
        ts = a_trial.sum()
        if ts < 1e-15:
            continue
        a_trial /= ts

        new_conv = fast_convolve(a_trial)
        new_obj = new_conv.max()
        delta = new_obj - current_obj

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            current_obj = new_obj
            if current_obj < best_obj - 1e-14:
                best_obj = current_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= cooling
        if no_improve > 500:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
