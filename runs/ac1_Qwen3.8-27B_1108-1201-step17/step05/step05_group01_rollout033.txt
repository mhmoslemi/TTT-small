"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategies:
1. LP-based Frank-Wolfe (primary local optimizer)
2. L-BFGS on smoothed log-sum-exp with extended annealing
3. Gradient-guided simulated annealing (new - escapes local minima)
4. Multi-length exploration with interpolation
5. Subgradient polishing with adaptive peak targeting
6. Coordinate descent for final precision
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.08, 5.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 120:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 120, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def _smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # Gradient via cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):
    """Optimize using L-BFGS on smooth approximation with extended annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)

    # Extended beta schedule with more intermediate values
    betas = [3.0, 5.0, 8.0, 12.0, 20.0, 35.0, 60.0, 100.0, 200.0, 400.0, 800.0]
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)

        try:
            # Scale maxiter based on available time
            maxiter = max(100, int(sub_time * 50))
            res = minimize(
                _smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 50,
                         'maxfun': maxiter * 10}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def gradient_annealing(f0, time_limit, verbose=False):
    """Gradient-guided simulated annealing to escape local minima."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Temperature schedule
    T = 0.005
    T_floor = 0.00005
    cooling_rate = 0.9995
    
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Compute smoothed gradient
        conv = fast_convolve(f)
        conv_max = conv.max()
        beta = 50.0 + 20.0 * (1.0 - T / 0.005)  # adaptive beta
        
        shifted = beta * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # Compute gradient via FFT
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        # Gradient step (on simplex tangent)
        lr = T * 0.5 + 0.001
        dot = np.dot(f, grad_a)
        grad_tangent = grad_a - dot
        f_grad = project_simplex(f - lr * grad_tangent)

        # Random perturbation (size scales with T)
        pert_size = T * 2.0
        num_pert = max(1, n // 100)
        pert_idx = rng.choice(n, num_pert, replace=False)
        f_pert = f_grad.copy()
        f_pert[pert_idx] += rng.standard_normal(num_pert) * pert_size
        f_trial = project_simplex(f_pert)

        obj_trial = fast_convolve(f_trial).max()
        delta = obj_trial - best_obj

        # Metropolis acceptance
        if delta < 0 or (T > T_floor and rng.random() < np.exp(-max(0, delta) / T)):
            f = f_trial
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_f = f.copy()
                no_improve = 0
                if verbose and iter_count % 500 == 0:
                    print(f"    Anneal iter {iter_count}: obj={best_obj:.10f}, T={T:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        T = max(T_floor, T * cooling_rate)

        # Restart if stuck
        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.005
            f = project_simplex(best_f + noise)
            no_improve = 0
            T = max(T, 0.002)

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=None, temp=0.00005):
    """Subgradient polishing with adaptive peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if top_k is None:
        top_k = max(20, n // 10)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # AdaGrad-like adaptive learning rate
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        cur_lr *= 0.9998
        cur_temp *= 1.002

        if no_improve > 300:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):

    betas = [3.0, 5.0, 8.0, 12.0, 20.0, 35.0, 60.0, 100.0, 180.0, 320.0, 600.0, 1200.0]

        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)

        lr = 0.001 + T * 0.3
        dot = np.dot(f, grad_a)
        grad_tangent = grad_a - dot
        f_grad = project_simplex(f - lr * grad_tangent)

        # Random perturbation (scales with temperature)
        pert_size = T * 1.5
        num_pert = max(1, n // 80)
        pert_idx = rng_local.choice(n, num_pert, replace=False)
        f_pert = f_grad.copy()
        f_pert[pert_idx] += rng_local.standard_normal(num_pert) * pert_size
        f_trial = project_simplex(f_pert)

        obj_trial = fast_convolve(f_trial).max()
        delta = obj_trial - best_obj

        # Metropolis acceptance criterion
        if delta < 0 or (T > T_floor and rng_local.random() < np.exp(-max(0.0, delta) / T)):
            f = f_trial
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_f = f.copy()
                no_improve = 0
                if verbose and iter_count % 1000 == 0:
                    print(f"    Anneal iter {iter_count}: obj={best_obj:.10f}, T={T:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        T = max(T_floor, T * cooling_rate)

        # Restart if stuck for too long
        if no_improve > 400:
            noise = rng_local.standard_normal(n) * 0.004
            f = project_simplex(best_f + noise)
            no_improve = 0
            T = max(T, 0.003)

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=None, temp=0.00005):
    """Subgradient polishing with adaptive peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if top_k is None:

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.003

        if no_improve > 350:
            rng_local = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            a_opt, _ = sg_polish(best_seq, time_limit=t_alloc,
                                 lr=0.002, mom=0.95,
                                 top_k=max(20, n_best // 8),
                                 temp=0.00003)
            try_seq(a_opt, "deep_sg")

        # 5d: Anneal deep
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.6, 60)
            a_opt, _ = gradient_annealing(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_anneal")

        # 5e: Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem * 0.7, 40)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

    # === Phase 6: Final polishing (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        for attempt in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break

            if attempt % 4 == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.5, 40)
                a_opt, _ = sg_polish(f_start, time_limit=t_alloc,
                                     lr=0.001, mom=0.96,
                                     top_k=max(10, n_best // 5),
                                     temp=0.00002)
                try_seq(a_opt, f"final_sg_{attempt}")
            elif attempt % 4 == 1:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.5, 40)
                a_opt, _ = lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lbfqg_{attempt}")
            elif attempt % 4 == 2:
                noise = rng.standard_normal(n_best) * 0.003
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.5, 35)
                a_opt, _ = gradient_annealing(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_anneal_{attempt}")
            else:
                t_alloc = min(rem * 0.5, 30)
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_cd_{attempt}")

        # Very final CD pass
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "final_cd_last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
