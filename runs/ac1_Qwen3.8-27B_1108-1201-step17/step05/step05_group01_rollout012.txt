    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. L-BFGS on log-sum-exp smoothed objective with aggressive beta annealing
2. LP-based Frank-Wolfe with golden-section line search and adaptive constraints
3. Peak-targeted subgradient optimization for basin exploration
4. Multi-start with structured perturbations
5. Multi-length exploration via interpolation
6. Final deep optimization alternating LP-FW and L-BFGS
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def lse_fn(t, beta):
    """Log-sum-exp smoothed objective and gradient in t-space (softmax param)."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def lse_fn_topk(t, beta, k):
    """Log-sum-exp over top-k peaks only."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    k_actual = min(k, len(conv))
    top_idx = np.argpartition(conv, -k_actual)[-k_actual:]
    top_vals = conv[top_idx]
    top_max = top_vals.max()
    shifted = beta * (top_vals - top_max)
    exp_shifted = np.exp(shifted)
    w_top = exp_shifted / exp_shifted.sum()

    obj = top_max + np.log(exp_shifted.sum()) / beta

    w_full = np.zeros(len(conv))
    w_full[top_idx] = w_top

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w_full, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def golden_section_min(f_val, a, b, tol=1e-7, max_iter=50):
    """Golden section search for minimum of f_val on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_val(c)
    fd = f_val(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_val(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_val(d)
    return (a + b) / 2.0

            m_exp = min(m_exp * 1.05, 5.0)

            return fast_convolve(ft).max()

                        best_val = opt_val

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_perturb(f, rng, n, strength=0.02):
    """Structured perturbation: peak-targeted + random."""
    conv = fast_convolve(f)
    k_max = int(np.argmax(conv))

    # Peak-targeted: reduce contributions to the max peak
    if k_max > 0:
        reduction = np.zeros(n)
        reduction[:n - k_max] = f[k_max:]
        reduction[k_max:] = f[:n - k_max]
    else:
        reduction = 2.0 * f

    # Normalize reduction to have sum = strength
    rsum = reduction.sum()
    if rsum > 1e-15:
        reduction = reduction / rsum * strength
    else:
        reduction = np.ones(n) / n * strength

    # Combine with random perturbation
    noise = rng.standard_normal(n) * strength * 0.3
    perturbed = f - reduction + noise
    return project_simplex(perturbed)

            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: L-BFGS initialization (150s) ===
    print("\n=== Phase 1: L-BFGS Init ===")
    rem = deadline - time.monotonic()
    t1 = min(150, rem - 1400)
    if t1 > 20:
        a_opt, _ = lbfqg_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lbfqg_init")

    # === Phase 2: LP-FW main optimization (600s) ===
    print(f"\n=== Phase 2: LP-FW Main (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    t2 = min(600, rem - 800)
    if t2 > 50:
        a_opt, _ = lp_fw_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_main")

    # LP-FW on reversed
    rem = deadline - time.monotonic()
    t2r = min(100, rem - 600)
    if t2r > 30:
        a_rev = base[::-1].copy()
        a_opt, _ = lp_fw_optimize(a_rev, time_limit=t2r)
        try_seq(a_opt, "lpfw_reversed")

    # === Phase 3: Top-k L-BFGS (200s) ===
    print(f"\n=== Phase 3: Top-k L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    t3 = min(200, rem - 500)
    if t3 > 30:
        n_topk = max(64, len(base) // 3)
        a_opt, _ = lbfqg_topk_optimize(base, time_limit=t3, top_k=n_topk, verbose=True)
        try_seq(a_opt, "lbfqg_topk")

    # === Phase 4: Peak subgradient for basin exploration (150s) ===
    print(f"\n=== Phase 4: Peak Subgradient (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    t4 = min(150, rem - 300)
    if t4 > 20:
        a_opt, _ = peak_subgradient_optimize(base, time_limit=t4, top_k=80, verbose=True)
        try_seq(a_opt, "peak_sg")

    # === Phase 5: Multi-start with structured perturbations (250s) ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    rem = deadline - time.monotonic()
    t5 = min(250, rem - 200)
    n_starts = max(3, int(t5 / 50))

    for si in range(n_starts):
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        t_per = min(rem / (n_starts - si), 60)

        # Structured perturbation
        pert = structured_perturb(base, rng, n_base, strength=0.015 + 0.01 * si)
        a_opt, _ = lbfqg_optimize(pert, time_limit=max(10, t_per))
        try_seq(a_opt, f"multistart_{si}")

    # === Phase 6: Deep optimization on best (300s) ===
    print(f"\n=== Phase 6: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 6a: LP-FW deep
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.6, 200)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 6b: L-BFGS deep with high beta
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.7, 150)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 6c: Top-k L-BFGS deep
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem * 0.8, 80)
            n_topk = max(32, n_best // 4)
            a_opt, _ = lbfqg_topk_optimize(best_seq, time_limit=t_alloc, top_k=n_topk)
            try_seq(a_opt, "deep_topk")

    # === Phase 7: Multi-length exploration (200s) ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = []
    n_base = len(base)
    for mult in [0.8, 1.125, 1.25, 1.5]:
        tl = int(n_base * mult)
        if tl != n_base and 32 <= tl <= 4096:
            target_lengths.append(tl)

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 60)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lbfqg_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 8: Final polishing (remaining time) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate LP-FW and L-BFGS with small perturbations
        for attempt in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break

            if attempt % 3 == 0:
                pert = structured_perturb(best_seq, rng, n_best, strength=0.005)
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = lp_fw_optimize(pert, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            elif attempt % 3 == 1:
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_lbfqg_{attempt}")
            else:
                t_alloc = min(rem * 0.5, 30)
                n_topk = max(16, n_best // 6)
                a_opt, _ = lbfqg_topk_optimize(best_seq, time_limit=t_alloc, top_k=n_topk)
                try_seq(a_opt, f"final_topk_{attempt}")

        # Final peak subgradient polish
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = peak_subgradient_optimize(best_seq, time_limit=min(rem, 25),
                                                 top_k=max(20, n_best // 5))
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
