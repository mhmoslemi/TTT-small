import numpy as np

def auto_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_topk(a, k=5):
    """Compute a subgradient of max auto-conv using top-k positions."""
    n = len(a)
    conv = auto_convolve(a)
    # Find top k positions
    top_idx = np.argpartition(conv, -k)[-k:]
    # For each top position t, the gradient w.r.t. a[j] is a[t-j] (if valid)
    # Average over top positions for a smoother direction
    sg = np.zeros(n)
    for t in top_idx:
        for j in range(n):
            other = t - j
            if 0 <= other < n:
                sg[j] += a[other]
    sg /= k
    return sg, conv.max()

import numpy as np
import time

def auto_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, top_k=10):
    """Compute subgradient of max auto-conv."""
    n = len(a)
    conv = auto_convolve(a)
    if top_k >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
    
    sg = np.zeros(n)
    for t in top_idx:
        lo = max(0, t - n + 1)
        hi = min(t, n - 1)
        if lo > hi:
            continue
        indices = np.arange(lo, hi + 1)
        sg[indices] += a[t - indices]
    sg /= len(top_idx)
    return sg, conv

def adam_simplex(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, top_k=10):

    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = auto_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    prev_obj = best_obj
    prev_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        sg, conv = compute_subgradient(a, top_k)
        
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a_new)
        
        cur_conv = auto_convolve(a)
        cur_obj = cur_conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(t * 7 + 3)
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
    
    return best_a, best_obj

The idea:
1. Start from a good initial point
2. At each step, propose a random move (e.g., transfer mass between two random coordinates)
3. Accept the move with probability exp(-delta_obj / T) (Metropolis criterion)
4. Slowly decrease T

This can escape local minima and explore the landscape. The key is to choose good moves.

For the move proposal:
- Pick two random coordinates j, m

The primal problem (on the simplex) is:
min_p max_k (p*p)[k]

The Lagrangian is:
L(p, w, t) = t + sum_k w_k * ((p*p)[k] - t)
where w_k >= 0, sum_k w_k = 1

The dual function is:

     = min_{p in simplex} [sum_k w_k * (p*p)[k]]
     (since t cancels: t - t*sum(w_k) = 0 when sum(w_k) = 1)

        = min_{p >= 0, sum(p)=1} p^T W p

where W is a matrix with W[j, m] = sum_k w_k * I(k-j=m) = w[j+m] (if we define w appropriately).

Wait, let me be more careful. (p*p)[k] = sum_j p[j] * p[k-j]. So:
sum_k w_k * (p*p)[k] = sum_k w_k * sum_j p[j]*p[k-j]
= sum_j p[j] * sum_k w_k * p[k-j]

Let's change variable: let l = k-j, so k = j+l.

= sum_{j,l} p[j] * w[j+l] * p[l]
= p^T W p

where W[j,l] = w[j+l] (with w defined on {0, 1, ..., 2n-2}).

So the dual problem is:
max_w g(w) = max_w min_{p in simplex} p^T W(w) p
s.t. w_k >= 0, sum_k w_k = 1

where W(w)[j,l] = w[j+l].

The inner problem (min p^T W p over the simplex) is a quadratic program. If W is PSD, this is convex and can be solved efficiently.

This gives us a way to compute a **lower bound** on the optimal value: max over w of min_{p in simplex} p^T W(w) p.

And we can use this lower bound to guide our primal optimization!

This is a **dual ascent** approach:
1. Start with w = uniform (w_k = 1/(2n-1))
2. Solve the inner QP: p* = argmin_{p in simplex} p^T W(w) p
3. Update w based on the values of (p**p)[k]: increase w_k for large k (ascending on the dual)

This is essentially the **Frank-Wolfe / mirror descent on the dual**!

def dual_ascent(n, time_limit, inner_itr=100):
    """Dual ascent for the auto-convolution min-max problem."""
    n_conv = 2*n - 1
    w = np.ones(n_conv) / n_conv  # dual variable
    
    best_p = np.ones(n) / n
    best_val = auto_convolve(best_p).max()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Build W matrix: W[j,l] = w[j+l]
        # W is n x n, W[j,l] = w[j+l] for j+l in [0, 2n-2]
        W = np.array([[w[j+l] for l in range(n)] for j in range(n)])
        # W is symmetric (W[j,l] = w[j+l] = w[l+j] = W[l,j])
        
        # Solve: min p^T W p s.t. sum(p) = 1, p >= 0
        # Use projected gradient or Frank-Wolfe
        p = solve_simplex_qp(W, inner_itr)
        
        # Update w: dual ascent
        # The subgradient of g(w) w.r.t. w_k is (p*p)[k] - g(w)
        # So we increase w_k where (p*p)[k] is large
        p_conv = auto_convolve(p)
        g_w = p @ W @ p
        
        # Gradient step on w (projected onto simplex)
        lr_w = 0.1 / np.sqrt(time.monotonic() - t0 + 1)
        w_new = w + lr_w * (p_conv - g_w)
        w = project_simplex(w_new)
        
        # Update best
        val = p_conv.max()
        if val < best_val:
            best_val = val
            best_p = p.copy()
    
    return best_p, best_val

    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

    n = len(v)

            if time.monotonic() - t0 > time_limit:
                break

            j = rng.integers(n)
            m = rng.integers(n)
            if j == m:
                continue

            # Determine direction from blame at max position
            km = np.argmax(b)
            kj = km - j
            km2 = km - m
            blame_j = a[kj] if 0 <= kj < n else 0.0
            blame_m = a[km2] if 0 <= km2 < n else 0.0

            if blame_j > blame_m + 1e-12:
                direction = -1.0
                max_d = a[j]
            elif blame_m > blame_j + 1e-12:
                direction = 1.0
                max_d = a[m]
            else:
                direction = 1.0 if rng.random() < 0.5 else -1.0
                max_d = a[j] if direction < 0 else a[m]

            if max_d < 1e-15:
                continue

            # Try multiple step sizes
            did_improve = False
            for frac in (0.3, 0.15, 0.08, 0.04):
                delta = direction * frac * max_d
                if a[j] + delta < -1e-15 or a[m] - delta < -1e-15:
                    continue

                # Compute effect on affected positions only
                lo = max(0, min(j, m))
                hi = min(conv_len - 1, max(j, m) + n - 1)

                # Get old values at affected positions
                affected_old = b[lo:hi + 1].copy()

                # Compute new values
                new_max_candidate = 0.0
                for k in range(lo, hi + 1):
                    vb = b[k]
                    if 0 <= k - j < n:
                        vb += 2.0 * delta * a[k - j]
                    if 0 <= k - m < n:
                        vb += -2.0 * delta * a[k - m]
                    if k == 2 * j:
                        vb += delta * delta
                    if k == 2 * m:
                        vb += delta * delta
                    if k == j + m and j != m:
                        vb += -2.0 * delta * delta
                    new_max_candidate = max(new_max_candidate, vb)

                # The true new max is max(new_max_candidate, old max at unaffected positions)
                # Old max at unaffected <= cur_max
                # So if new_max_candidate < cur_max, we might have improved
                # But we need to check if the old global max was at an affected position
                safe_upper = max(new_max_candidate, cur_max)

                if new_max_candidate < cur_max - 1e-13:
                    # Definitely improved (new max of affected < old global max)
                    # But need to check if old max was at unaffected position
                    # If old argmax is in [lo, hi], it's affected, so new max = new_max_candidate
                    # If old argmax is outside [lo, hi], new max could be old value at that position
                    if km < lo or km > hi:
                        # Old max at unaffected position, new max >= that old value = cur_max
                        # So no actual improvement
                        continue
                    # Apply
                    a[j] += delta
                    a[m] -= delta
                    # Recompute full b (safe and fast for n=768)
                    b = auto_convolve(a)
                    cur_max = b.max()
                    accepted += 1
                    if cur_max < best_max:
                        best_max = cur_max
                        best_a = a.copy()
                    did_improve = True
                    break

        # If no improvement in a whole batch, add tiny perturbation
        if not did_improve and rng.random() < 0.1:
            noise = rng.standard_normal(n) * 0.001
            a = project_simplex(best_a + noise)
            b = auto_convolve(a)
            cur_max = b.max()

    return best_a, best_max


def dual_guided_optimize(a0, time_limit, inner_steps=50):
    """Dual ascent: find weights on auto-conv positions, then solve QP."""
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    best_a = a.copy()
    best_obj = auto_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    # Dual variable: weights on conv positions
    w = np.ones(conv_len) / conv_len

    outer_iter = 0
    while time.monotonic() - t0 < time_limit:
        outer_iter += 1

        # Solve inner QP: min p^T W p s.t. p in simplex
        # W[j,l] = w[j+l], so (Wp)[j] = sum_l w[j+l] * p[l]
        # Use projected gradient on the QP
        p = best_a.copy()
        for _ in range(inner_steps):
            # Compute Wp
            # (Wp)[j] = sum_{l=0}^{n-1} w[j+l] * p[l]
            # This is a correlation: let's compute via direct sum (O(n^2)) or FFT
            # For n=768, O(n^2) = ~600K per step, times 50 steps = 30M per outer iter
            # That might be slow. Let's use FFT.
            # (Wp)[j] = sum_l w[j+l]*p[l] = cross_correl(w, p, lag=j)
            # cross_correl(w, p, lag) = ifft(fft(w) * conj(fft(p))) shifted
            fft_sz = 1 << (conv_len + n - 1 - 1).bit_length()
            Wf = np.fft.rfft(w, fft_sz)
            Pf = np.fft.rfft(p, fft_sz)
            # cross_correl(w, p)[j] = sum_k w[k+j]*p[k] = (w * flip(p))[j + n - 1]... 
            # Actually let's just do it directly for now, n=768 is manageable
            Wp = np.array([np.dot(w[j:j+n], p) for j in range(n)])
            # This is O(n^2) per step. For n=768, that's ~590K per step.
            # 50 steps * 600K = 30M per outer iteration. At ~10ns per op, that's 0.3s per outer iter.
            # In 200s, we get ~600 outer iterations. That's fine.

            grad = 2.0 * Wp
            # Projected gradient step
            lr_q = 2.0 / (np.linalg.norm(W, ord=2) if False else 4.0)
            # Use a fixed small step
            lr_q = 0.5 / max(np.max(np.abs(grad)), 1e-10)
            p_new = p - lr_q * grad
            p = project_simplex(p_new)

        # Evaluate
        conv_p = auto_convolve(p)
        obj_p = conv_p.max()
        if obj_p < best_obj - 1e-14:
            best_obj = obj_p
            best_a = p.copy()

        # Dual ascent: update w
        # Subgradient of dual w.r.t. w_k is (p*p)[k] - min_value
        sg_w = conv_p - obj_p  # centered
        lr_dual = 1.0 / np.sqrt(outer_iter + 1.0)
        w_new = w + lr_dual * sg_w
        w = project_simplex(w_new)

        # Occasionally add noise to w for exploration
        if outer_iter % 20 == 0:
            w = project_simplex(w + rng.standard_normal(conv_len) * 0.01)

    return best_a, best_obj


def make_init_gaussian(n, center=0.5, sigma=0.2):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_init_hanning(n):
    x = np.arange(n)
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_init_sine(n):
    x = np.arange(n)
    f = np.sin(np.pi * (x + 0.5) / n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_init_smooth_random(n, rng, n_comp=4):
    x = np.linspace(0, 1, n)
    f = np.zeros(n)
    for _ in range(n_comp):
        c = rng.uniform(0.15, 0.85)
        s = rng.uniform(0.08, 0.3)
        amp = rng.uniform(0.5, 2.0)
        f += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi
