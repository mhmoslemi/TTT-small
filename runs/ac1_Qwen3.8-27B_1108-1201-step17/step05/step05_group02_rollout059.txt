def project_simplex(v):
    """Projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent
with active-set skipping, multi-length exploration, LP-FW polishing,
and subgradient refinement. Key innovations: (1) skip inactive coordinates
for O(active*n) per pass instead of O(n^2), (2) fast-path feasibility check,
(3) difference-set and structured initializations, (4) adaptive time allocation
across multiple optimization phases.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cd_exact(f_init, n, time_limit, max_passes=2000, verbose=False):
    """
    Exact coordinate descent minimizing max(f*f) for sum(f)=1, f>=0.
    Uses golden section search with active-set skipping.
    """
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f[:] = 1.0 / n
    else:
        f /= s

    conv = fast_convolve(f)
    t0 = time.monotonic()
    best_obj = float(conv.max())
    best_f = f.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve = 0

    for pass_num in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break

        pass_improvement = 0.0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_x = f[j]
            if old_x < 1e-18:
                continue

            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            length = k_hi - k_lo + 1
            f_slice = f[:length]

            # Compute C_fixed (max of unaffected convolution entries)
            C_fixed = 0.0
            if k_lo > 0:
                C_fixed = max(C_fixed, float(np.max(conv[:k_lo])))
            if k_hi < 2 * n - 2:
                C_fixed = max(C_fixed, float(np.max(conv[k_hi + 1:])))

            # Skip if affected region is not the bottleneck
            cur_max_aff = float(np.max(conv[k_lo:k_hi + 1]))
            if cur_max_aff <= C_fixed + 1e-13:
                continue

            # Golden section search for optimal x_new
            x_max = max(3.0 * old_x, 5.0 / n)
            conv_slice = conv[k_lo:k_hi + 1]
            has_quad = (2 * j >= k_lo) and (2 * j <= k_hi)
            quad_idx = 2 * j - k_lo if has_quad else -1
            conv_quad = conv[2 * j] if has_quad else 0.0

            # Fast path: check if we can simply cap at C_fixed
            x_upper = x_max
            feasible = True
            for idx in range(length):
                k = k_lo + idx
                if k == 2 * j:
                    rhs = C_fixed - conv_quad + old_x * old_x
                    if rhs < -1e-15:
                        feasible = False
                        break
                    ub = np.sqrt(max(rhs, 0.0))
                    if ub < x_upper:
                        x_upper = ub
                else:
                    fk = f_slice[idx]
                    if fk > 1e-15:
                        xb = (C_fixed - conv_slice[idx]) / (2.0 * fk) + old_x
                        if xb < x_upper:
                            x_upper = xb
            if x_upper < -1e-12:
                feasible = False

            if feasible and x_upper >= 0:
                # We can achieve max_affected <= C_fixed
                # Set x to min(old_x, x_upper) for minimal change
                x_opt = min(old_x, x_upper)
                x_opt = max(0.0, x_opt)
            else:
                # Full golden section
                a_g = 0.0
                b_g = x_max

                # Evaluate at two interior points
                c_g = b_g - gr * (b_g - a_g)
                d_g = a_g + gr * (b_g - a_g)

                # eval c_g
                delta_c = c_g - old_x
                vals_c = conv_slice + 2.0 * delta_c * f_slice
                if has_quad:
                    vals_c = vals_c.copy()
                    vals_c[quad_idx] = conv_quad + c_g * c_g - old_x * old_x
                fc = max(float(np.max(vals_c)), C_fixed)

                # eval d_g
                delta_d = d_g - old_x
                vals_d = conv_slice + 2.0 * delta_d * f_slice
                if has_quad:
                    vals_d = vals_d.copy()
                    vals_d[quad_idx] = conv_quad + d_g * d_g - old_x * old_x
                fd = max(float(np.max(vals_d)), C_fixed)

                for _ in range(45):
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - gr * (b_g - a_g)
                        delta_c = c_g - old_x
                        vals_c = conv_slice + 2.0 * delta_c * f_slice
                        if has_quad:
                            vals_c = vals_c.copy()
                            vals_c[quad_idx] = conv_quad + c_g * c_g - old_x * old_x
                        fc = max(float(np.max(vals_c)), C_fixed)
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + gr * (b_g - a_g)
                        delta_d = d_g - old_x
                        vals_d = conv_slice + 2.0 * delta_d * f_slice
                        if has_quad:
                            vals_d = vals_d.copy()
                            vals_d[quad_idx] = conv_quad + d_g * d_g - old_x * old_x
                        fd = max(float(np.max(vals_d)), C_fixed)

                x_opt = max(0.0, (a_g + b_g) / 2.0)
                x_opt = min(x_opt, 1.0)

            if abs(x_opt - old_x) < 1e-14:
                continue

            # Apply update
            f[j] = x_opt
            f_sum = f.sum()
            if f_sum > 1e-15:
                f /= f_sum
            conv = fast_convolve(f)
            obj = float(conv.max())

            if obj < best_obj - 1e-14:
                pass_improvement += (best_obj - obj)
                best_obj = obj
                best_f = f.copy()

        if pass_improvement < 1e-13:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
            if verbose and (pass_num + 1) % 50 == 0:
                print(f"      CD pass {pass_num+1}: obj={best_obj:.10f}")

def lp_direction(f, M, n, time_limit_s=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': max(1, time_limit_s), 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f[:] = 1.0 / len(f0)
    else:
        f /= s
    n = len(f)

    conv = fast_convolve(f)
    M = float(conv.max())
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_relax = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        lp_time = min(20, rem * 0.3)
        g = lp_direction(f, M * m_relax, n, time_limit_s=int(lp_time))
        if g is None:
            m_relax = min(m_relax * 1.2, 5.0)
            if m_relax >= 4.5:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.05, 5.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if 0 < best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = float(fast_convolve(ft).max())
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            M = float(fast_convolve(f).max())
            m_relax = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 60, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = float(fast_convolve(f).max())
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=60, temp=0.00004):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f[:] = 1.0 / len(f0)
    else:
        f /= s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = float(conv.max())

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.999
        cur_temp *= 1.004

        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v[:] = 0.0
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def make_diffset_init(n, seed=0):
    """Create initialization inspired by difference sets / flat sequences."""
    rng = np.random.default_rng(seed)
    # Start with a smooth bump and add structure
    x = np.linspace(0, 2 * np.pi, n)
    f = np.cos(0.5 * x) ** 2 + 0.3 * np.ones(n)
    f = np.maximum(f, 0)
    # Add some random perturbation
    f += rng.standard_normal(n) * 0.1
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining exact coordinate descent,
    LP-FW, subgradient polishing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 20.0

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: CD on parent ===
    print(f"\n=== Phase 1: CD on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(200.0, rem * 0.25)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_opt, obj = cd_exact(parent, n_parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  CD result: {sc:.6f} (n={n_parent})")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    target_lengths = []
    for delta in [-100, -50, 50, 100, 200, -200]:
        nl = n_best + delta
        if 128 <= nl <= 2048 and nl != n_best:
            target_lengths.append(nl)
    # Also try some specific lengths
    for nl in [384, 512, 640, 768, 896, 1024]:
        if nl not in target_lengths and 128 <= nl <= 2048:
            target_lengths.append(nl)

    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80.0, rem * 0.12)
        if t_alloc < 15:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = cd_exact(f_interp, target_n, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            n_best = target_n
            print(f"  NEW BEST: {sc:.6f} (n={target_n})")

    # === Phase 3: LP-FW on best ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    t3 = min(300.0, rem * 0.3)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  LP-FW result: {sc:.6f}")

    # === Phase 4: CD after LP-FW ===
    print(f"\n=== Phase 4: CD polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    t4 = min(250.0, rem * 0.3)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        a_opt, _ = cd_exact(best_seq, n_best, time_limit=t4, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  CD polish: {sc:.6f}")

    # === Phase 5: Diverse initializations + CD ===
    print(f"\n=== Phase 5: Diverse inits (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    inits = []
    # Noise perturbations
    for scale in [0.003, 0.008, 0.015]:
        noise = rng.standard_normal(n_best) * scale
        inits.append(("noise_s" + str(scale).replace(".", ""), project_simplex(best_seq + noise)))
    # Reverse
    f_rev = 0.5 * best_seq + 0.5 * best_seq[::-1]
    inits.append(("rev", project_simplex(f_rev)))
    # Mix with structured init
    f_struct = make_diffset_init(n_best, seed=123)
    f_mix = 0.7 * best_seq + 0.3 * f_struct
    inits.append(("mix_struct", project_simplex(f_mix)))
    # Shift
    f_shift = np.roll(best_seq, 3)
    inits.append(("shift3", project_simplex(f_shift)))

    for label, f_init in inits:
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t_alloc = min(100.0, rem * 0.15)
        if t_alloc < 10:
            continue
        print(f"  {label}: t={t_alloc:.0f}s")
        a_opt, _ = cd_exact(f_init, n_best, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f}")

    # === Phase 6: SG polish ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    if rem > 20:
        t_sg = min(60.0, rem * 0.2)
        a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                             lr=0.001, mom=0.93,
                             top_k=max(20, n_best // 8),
                             temp=0.00003)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  SG polish: {sc:.6f}")

    # === Phase 7: Final CD squeeze ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    if rem > 15:
        t_final = min(120.0, rem * 0.7)
        a_opt, _ = cd_exact(best_seq, n_best, time_limit=t_final, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Final CD: {sc:.6f}")

    # === Phase 8: Micro-perturbation search ===
    print(f"\n=== Phase 8: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(15):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0002 + trial * 0.0001
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            # Quick CD polish
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = cd_exact(f_pert, n_best, time_limit=min(rem2 * 0.3, 15))
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_cd]: {sc2:.6f}")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
