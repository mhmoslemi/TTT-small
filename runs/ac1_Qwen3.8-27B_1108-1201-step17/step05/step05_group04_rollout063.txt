```python
"""
Search for non-negative sequences minimizing
    2*n*max(linear_convolution(a,a))/(sum(a))^2.

Algorithm highlights:
- Fast FFT-based evaluation of the objective and of auxiliary shaping metrics.
- Strong local optimization via a sequential linear-programming "best convex
  direction" method:
      minimize predicted next peak
                   max_k [(1-2e)*p_k + 2e*L*(a*g)_k]
      subject to g >= 0, sum(g)=1,
  where p_k = L * conv(a,a)_k (after normalization) and L = 2n-1.
  This is a convex LP. The direction g is then moved along by a line search.
- Diverse initializations: parent, reversed parent, inverse-square-root tails,
  random smooth starts, and a one-parameter family constructed by recursively
  forcing the first half of a*a to be flat.
- Multiplicative/entropy/variance reshaping phases to escape local minima.
- Final coordinate-descent polishing and exact verification.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    from scipy import sparse
    _HAVE_SCIPY = True
except Exception:
    linprog = None
    sparse = None
    _HAVE_SCIPY = False


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    sz = next_pow2(L)
    A = np.fft.rfft(a, sz)
    c = np.fft.irfft(A * A, sz)[:L]
    c = np.maximum(c, 0.0)
    return c


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def interpolate_sequence(a, new_n):
    a = normalize_sequence(a)
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x = np.linspace(0.0, 1.0, old_n)
    y = np.linspace(0.0, 1.0, new_n)
    b = np.interp(y, x, a)
    return normalize_sequence(b)


def score_internal(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    sa = float(np.sum(a))
    if sa <= 1e-300:
        return np.inf
    c = fast_conv(a)
    if c.size == 0:
        return np.inf
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return np.inf
    return float(2.0 * n * cmax / (sa * sa))


def conv_metrics(a, c=None):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf, 0.0, 0.0, None
    if c is None:
        c = fast_conv(a)
    sa = float(np.sum(a))
    cmax = float(np.max(c)) if c.size else 0.0
    if sa <= 1e-300 or cmax <= 0.0 or not np.isfinite(cmax):
        return np.inf, 0.0, 0.0, None

    score = float(2.0 * n * cmax / (sa * sa))
    csum = float(np.sum(c))
    if csum <= 1e-300 or not np.isfinite(csum):
        return score, 0.0, 0.0, None

    L = 2 * n - 1
    p = (c / csum) * float(L)
    p = np.maximum(p, 0.0)

    var = float(np.mean((p - 1.0) * (p - 1.0)))

    p_safe = np.maximum(p, 1e-300)
    kl = float(np.sum(p_safe * np.log(p_safe)) / float(L))
    if not np.isfinite(kl):
        kl = 0.0

    return score, var, max(0.0, kl), p


def unit_norm(v):
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    nrm = float(np.linalg.norm(v))
    if not np.isfinite(nrm) or nrm < 1e-300:
        return np.zeros_like(v)
    return v / nrm


def make_weights(p, beta, w_max, w_var, w_ent):
    p = np.asarray(p, dtype=np.float64)
    L = p.size
    v = np.zeros(L, dtype=np.float64)
    p_safe = np.maximum(p, 1e-300)

    if w_max != 0.0 and beta > 0.0:
        x = beta * (p_safe - float(p_safe.max()))
        sm = np.exp(x)
        sm_sum = float(sm.sum())
        if sm_sum > 1e-300:
            sm = sm / sm_sum
        else:
            sm = np.zeros_like(sm)
            sm[int(np.argmax(p))] = 1.0
        v = v + w_max * unit_norm(sm)

    if w_var != 0.0:
        v = v + w_var * unit_norm(p - 1.0)

    if w_ent != 0.0:
        logp = np.log(p_safe)
        logp = logp - float(logp.mean())
        v = v + w_ent * unit_norm(logp)

    return unit_norm(v)


def multiplicative_update(a, v, eta):
    a = np.asarray(a, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if a.size == 0 or v.size != 2 * a.size - 1:
        return a.copy(), False

    if eta <= 0.0:
        return a.copy(), False

    grad = 2.0 * np.correlate(v, a, "valid")
    grad = np.asarray(grad, dtype=np.float64)
    if grad.size != a.size or not np.all(np.isfinite(grad)):
        return a.copy(), False

    grad = np.clip(grad, -50.0, 50.0)
    logfac = -eta * grad
    logfac -= float(logfac.max())
    new = a * np.exp(logfac)
    new = np.where(np.isfinite(new), new, 0.0)
    new = np.maximum(new, 0.0)
    s = float(new.sum())
    if s <= 1e-300:
        return a.copy(), False
    return new / s, True


def line_search(a, g, current_score):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    best_score = float(current_score)
    best_alpha = 0.0
    best_a = a.copy()

    alphas = np.array(
        [0.002, 0.005, 0.01, 0.02, 0.05, 0.08, 0.12, 0.18,
         0.25, 0.35, 0.5, 0.65, 0.8, 0.9, 1.0],
        dtype=np.float64
    )
    for al in alphas:
        al = float(al)
        if al <= 0.0:
            continue
        trial = normalize_sequence((1.0 - al) * a + al * g)
        sc = score_internal(trial)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_alpha = al
            best_a = trial.copy()

    if best_alpha > 0.0:
        widths = np.array([0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002], dtype=np.float64)
        for w in widths:
            left = max(0.0, best_alpha - float(w))
            right = min(1.0, best_alpha + float(w))
            cand = [left, right]
            local_best = best_score
            local_alpha = best_alpha
            local_a = best_a
            for ca in cand:
                ca = float(ca)
                if ca <= 0.0 or ca >= 1.0:
                    continue
                if abs(ca - best_alpha) < 1e-15:
                    continue
                trial = normalize_sequence((1.0 - ca) * a + ca * g)
                sc = score_internal(trial)
                if np.isfinite(sc) and sc < local_best - 1e-12:
                    local_best = sc
                    local_alpha = ca
                    local_a = trial.copy()
            best_score = local_best
            best_alpha = local_alpha
            best_a = local_a

    return best_a, best_score, best_alpha


def precompute_lp(n):
    n = int(n)
    L = 2 * n - 1
    g_count = n * n
    total_count = g_count + L

    base_col = np.arange(n, dtype=np.int32)
    rows_parts = []
    cols_parts = []
    for d in range(n):
        rows_parts.append(d + base_col)
        cols_parts.append(base_col)

    rows = np.concatenate(rows_parts).astype(np.int32) if rows_parts else np.empty(0, dtype=np.int32)
    cols = np.concatenate(cols_parts).astype(np.int32) if cols_parts else np.empty(0, dtype=np.int32)

    xi_rows = np.arange(L, dtype=np.int32)
    xi_cols = np.full(L, n, dtype=np.int32)

    rows_full = np.concatenate((rows, xi_rows)).astype(np.int32)
    cols_full = np.concatenate((cols, xi_cols)).astype(np.int32)

    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, -1] = 0.0
    b_eq = np.ones((1,), dtype=np.float64)

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0

    bounds = [(0.0, None)] * (n + 1)

    return {
        "n": n,
        "L": L,
        "g_count": g_count,
        "total_count": total_count,
        "rows_full": rows_full,
        "cols_full": cols_full,
        "A_eq": A_eq,
        "b_eq": b_eq,
        "c_obj": c_obj,
        "bounds": bounds,
    }


def solve_lp_direction(a, p_scaled, eps, pre, time_limit=10.0):
    if not _HAVE_SCIPY:
        return None
    n = int(pre["n"])
    L = int(pre["L"])
    if eps <= 1e-8 or eps >= 0.45:
        return None
    a = np.asarray(a, dtype=np.float64)
    p_scaled = np.asarray(p_scaled, dtype=np.float64)
    if a.size != n or p_scaled.size != L:
        return None
    if not np.all(np.isfinite(a)) or not np.all(np.isfinite(p_scaled)):
        return None

    g_count = int(pre["g_count"])
    total_count = int(pre["total_count"])
    data_all = np.empty(total_count, dtype=np.float64)

    data_all[:g_count] = np.repeat(a, n)
    coeff = 2.0 * float(eps) * float(L)
    data_all[:g_count] *= coeff
    data_all[g_count:] = -1.0

    try:
        A_ub = sparse.csr_matrix(
            (data_all, pre["rows_full"], pre["cols_full"]),
            shape=(L, n + 1)
        )
    except Exception:
        return None

    b_ub = (1.0 - 2.0 * float(eps)) * p_scaled
    b_ub = np.maximum(b_ub, 0.0)
    b_ub = np.where(np.isfinite(b_ub), b_ub, 0.0)

    opts_basic = {
        "presolve": True,
        "dual_feasibility_tolerance": 1e-9,
        "primal_feasibility_tolerance": 1e-9,
    }
    opts_time = dict(opts_basic)
    opts_time["time_limit"] = float(max(0.1, time_limit))

    res = None
    try:
        res = linprog(
            pre["c_obj"],
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=pre["A_eq"],
            b_eq=pre["b_eq"],
            bounds=pre["bounds"],
            method="highs",
            options=opts_time,
        )
    except Exception:
        try:
            res = linprog(
                pre["c_obj"],
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=pre["A_eq"],
                b_eq=pre["b_eq"],
                bounds=pre["bounds"],
                method="highs",
                options=opts_basic,
            )
        except Exception:
            res = None

    if res is None or getattr(res, "x", None) is None:
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if gs <= 1e-15:
        return None
    return g / gs


def lp_polish(a, time_limit, pre, rng, eps_start=0.1, tag="lp"):
    if time_limit <= 2.0 or pre is None:
        return normalize_sequence(a), score_internal(a), normalize_sequence(a)

    t0 = time.monotonic()
    current = normalize_sequence(a)
    current_score = score_internal(current)
    best = current.copy()
    best_score = current_score

    eps = float(eps_start)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 12.0:
            break

        score_m, _, _, p = conv_metrics(current)
        if not np.isfinite(score_m):
            break
        current_score = score_m
        if p is None:
            break

        eps_try = [eps]
        if no_improve >= 1:
            eps_try.append(0.03)
        if no_improve >= 2:
            eps_try.extend([0.08, 0.15])

        g = None
        used_eps = eps
        for e in eps_try:
            e = float(e)
            if e <= 0.0 or e >= 0.3:
                continue
            g = solve_lp_direction(current, p, e, pre, time_limit=min(12.0, max(1.0, rem - 5.0)))
            if g is not None:
                used_eps = e
                eps = e
                break

        if g is None:
            no_improve += 1
            if no_improve > 12:
                break
            base = best if best_score <= current_score else current
            mean = float(base.mean())
            scale = float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.1])) * mean
            current = normalize_sequence(base + rng.standard_normal(len(base)) * scale)
            current_score = score_internal(current)
            eps = float(rng.choice([0.03, 0.06, 0.1, 0.15]))
            no_improve = 0
            continue

        trial, trial_score, alpha = line_search(current, g, current_score)
        if trial_score < current_score - 1e-12:
            current = trial
            current_score = trial_score
            no_improve = 0
            if alpha <= 0.0:
                alpha = 0.05
            eps = float(np.clip(alpha * 1.3, 0.02, 0.25))
        elif trial_score < best_score - 1e-12:
            best = trial.copy()
            best_score = trial_score
            current = trial.copy()
            current_score = trial_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 8:
                base = best
                mean = float(base.mean())
                scale = float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.1])) * mean
                current = normalize_sequence(base + rng.standard_normal(len(base)) * scale)
                current_score = score_internal(current)
                eps = float(rng.choice([0.03, 0.05, 0.08, 0.12, 0.2]))
                no_improve = 0
            else:
                eps = float(np.clip(eps * 0.8, 0.02, 0.25))

        it += 1
        if it % 50 == 0:
            print(f"    {tag}: iter={it}, best={best_score:.9f}, cur={current_score:.9f}, eps={eps:.4f}")

    if current_score < best_score - 1e-12:
        best = current.copy()
        best_score = current_score

    return best, best_score, current


def multiplicative_phase(a, time_limit, beta, w_max, w_var, w_ent,
                         q_var, q_ent, score_tol, tag="mp"):
    if time_limit <= 2.0:
        normed = normalize_sequence(a)
        sc = score_internal(normed)
        return normed, sc, normed

    t0 = time.monotonic()
    current = normalize_sequence(a)
    score, var, kl, p = conv_metrics(current)
    if not np.isfinite(score):
        return current, score, current

    best = current.copy()
    best_score = score
    q = score + float(q_var) * var + float(q_ent) * kl
    eta = 0.1
    it = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        w = make_weights(p, beta, w_max, w_var, w_ent)
        if float(np.linalg.norm(w)) < 1e-12:
            break

        new, ok = multiplicative_update(current, w, eta)
        if not ok:
            eta *= 0.5
            if eta < 1e-6:
                break
            continue

        new_score, new_var, new_kl, new_p = conv_metrics(new)
        if not np.isfinite(new_score):
            eta *= 0.5
            if eta < 1e-6:
                break
            continue

        new_q = new_score + float(q_var) * new_var + float(q_ent) * new_kl
        allowed = new_score < (best_score + float(score_tol))

        if allowed and new_q < q - 1e-12:
            current = new
            p = new_p
            q = new_q
            eta = min(eta * 1.02, 1.0)
            it = 0
            if new_score < best_score - 1e-10:
                best = new.copy()
                best_score = new_score
        else:
            eta *= 0.5
            it += 1
            if eta < 1e-6 or it > 200:
                break

        if it % 200 == 0:
            print(f"    {tag}: score={score:.9f}, best={best_score:.9f}, eta={eta:.5f}")

    if score_internal(current) < best_score - 1e-12:
        best = current.copy()
        best_score = score_internal(current)

    return best, best_score, current


def flat_first_sequence(n, A):
    n = int(n)
    p = np.zeros(n, dtype=np.float64)
    p[0] = 1.0
    A = float(A)
    if not np.isfinite(A) or A <= 1e-8:
        A = 1e-4

    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            s = float(np.dot(p[1:k], p[k - 1:0:-1]))
        val = 0.5 * (A - s)
        if not np.isfinite(val) or val < 0.0:
            val = 0.0
        p[k] = val

    return normalize_sequence(p)


def optimize_flat_first(n, time_limit=30.0):
    if time_limit <= 1.0:
        return None, np.inf

    t0 = time.monotonic()
    best_a = None
    best_score = np.inf
    best_A = 1.0

    As = np.concatenate(
        (
            np.geomspace(0.05, 3.0, 20),
            np.array([0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0, 5.0], dtype=np.float64),
        )
    )

    for A in As:
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        try:
            a = flat_first_sequence(n, float(A))
            sc = score_internal(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_A = float(A)
        except Exception:
            pass

    if best_a is not None:
        for _ in range(3):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            lo = max(0.01, best_A * 0.4)
            hi = min(20.0, best_A * 2.5)
            if hi <= lo:
                hi = lo * 2.0
            grid = np.geomspace(lo, hi, 25)
            for A in grid:
                if time.monotonic() - t0 > time_limit - 1.0:
                    break
                try:
                    a = flat_first_sequence(n, float(A))
                    sc = score_internal(a)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        best_A = float(A)
                except Exception:
                    pass

    return best_a, best_score


def make_smooth_random(n, rng):
    x = np.linspace(0.0, 1.0, n)
    f = np.zeros(n, dtype=np.float64)
    for _ in range(6):
        c = float(rng.uniform(0.05, 0.95))
        s = float(rng.uniform(0.05, 0.25))
        amp = float(rng.uniform(0.5, 2.0))
        f += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-12)
    return normalize_sequence(f)


def generate_basic_starts(n, rng, parent):
    out = []
    if parent is not None:
        out.append(("parent", normalize_sequence(parent)))
        out.append(("parent_rev", normalize_sequence(parent[::-1].copy())))

    out.append(("uniform", np.ones(n, dtype=np.float64) / float(n)))

    idx = np.arange(1, n + 1, dtype=np.float64)
    base = 1.0 / np.sqrt(idx)
    out.append(("invsqrt", normalize_sequence(base)))
    out.append(("invsqrt_rev", normalize_sequence(base[::-1].copy())))

    for gamma in (0.5, 1.5):
        arr = 1.0 / np.sqrt(idx + gamma)
        out.append(("isg" + str(gamma), normalize_sequence(arr)))
        out.append(("isg" + str(gamma) + "_rev", normalize_sequence(arr[::-1].copy())))

    for alpha in (0.45, 0.55):
        arr = idx ** (-float(alpha))
        name = "pow" + str(alpha)
        out.append((name, normalize_sequence(arr)))
        out.append((name + "_rev", normalize_sequence(arr[::-1].copy())))

    out.append(("smooth", make_smooth_random(n, rng)))
    return out


def get_parent_sequence():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        return None

    if arr is None or arr.ndim != 1 or arr.size == 0:
        return None

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.sum() <= 1e-300:
        return None

    if arr.size > 1536:
        arr = interpolate_sequence(arr, 1536)

def coordinate_descent_polish(a, time_limit):
    if time_limit <= 2.0:
        normed = normalize_sequence(a)
        return normed, score_internal(normed)

    t0 = time.monotonic()
    current = normalize_sequence(a)
    current_score = score_internal(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    factors = np.array([0.25, 0.5, 0.75, 0.9, 1.1, 1.25, 1.5, 2.0, 4.0], dtype=np.float64)
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old = float(current[j])
            if old <= 1e-12 or old >= 0.9999:
                continue

            denom = 1.0 - old
            if denom <= 1e-12:
                continue

            for f in factors:
                f = float(f)
                x = old * f
                if x < 0.0 or x >= 0.9999:
                    continue

                factor = (1.0 - x) / denom
                trial = current * factor
                trial[j] = x
                trial = normalize_sequence(trial)
                sc = score_internal(trial)

                if np.isfinite(sc) and sc < current_score - 1e-12:
                    current = trial
                    current_score = sc
                    improved = True
                    if current_score < best_score - 1e-10:
                        best = current.copy()
                        best_score = current_score
                    old = float(current[j])
                    denom = 1.0 - old
                    break

        if not improved and passes > 1:
            break

    if current_score < best_score - 1e-12:
        best = current.copy()
        best_score = current_score

    return best, best_score


def safe_evaluate(seq):
    try:
        val = float(evaluate_sequence([float(x) for x in seq]))
        return val
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(120.0, float(budget_s) - 120.0)

    parent = get_parent_sequence()
    if parent is not None:
        n = len(parent)
        parent = normalize_sequence(parent)
    else:
        n = 768
        parent = None

    candidates = []
    if parent is not None:
        candidates.extend(generate_basic_starts(n, rng, parent))
    else:
        candidates.extend(generate_basic_starts(n, rng, None))

    flat_best = None
    flat_time = min(25.0, max(0.0, (deadline - time.monotonic()) - 30.0))
    if flat_time > 5.0:
        flat_best, _ = optimize_flat_first(n, flat_time)
        if flat_best is not None:
            candidates.append(("flat_best", flat_best))
            candidates.append(("flat_best_rev", normalize_sequence(flat_best[::-1].copy())))

    starts_dict = {}
    for name, a in candidates:
        if a is None:
            continue
        a = normalize_sequence(a)
        starts_dict[name] = a

    best_seq = None
    best_score = np.inf
    best_name = "none"

    for name, a in starts_dict.items():
        try:
            sc = score_internal(a)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            best_name = name

    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / float(n)
        best_score = score_internal(best_seq)

    print(f"initial best: name={best_name}, score={best_score:.9f}, n={len(best_seq)}")

    pre = precompute_lp(n)

    rem = max(0.0, deadline - time.monotonic())
    t_main = min(800.0, max(120.0, rem - 700.0))
    t_main = min(t_main, max(0.0, rem - 60.0))
    if t_main > 60.0:
        print(f"Starting main LP polish: {t_main:.1f}s")
        best_seq, best_score, _ = lp_polish(best_seq, t_main, pre, rng, eps_start=0.1, tag="main_lp")
        print(f"after main LP: best={best_score:.9f}")

    rem = max(0.0, deadline - time.monotonic())
    diverse_names = []
    preferred = ["flat_best", "flat_best_rev", "invsqrt", "invsqrt_rev", "smooth", "parent_rev"]
    for nm in preferred:
        if nm in starts_dict and nm != best_name:
            diverse_names.append(nm)
    for nm in starts_dict:
        if nm not in diverse_names and nm != best_name:
            diverse_names.append(nm)

    diverse_budget = min(360.0, max(0.0, rem - 500.0))
    if diverse_budget > 80.0 and diverse_names:
        per = min(diverse_budget / float(len(diverse_names)), 150.0)
        for nm in diverse_names:
            rem = max(0.0, deadline - time.monotonic())
            if rem < 70.0:
                break
            limit = min(per, max(25.0, rem * 0.15))
            print(f"Diverse LP from {nm}, limit={limit:.1f}s")
            a2, sc2, _ = lp_polish(starts_dict[nm], limit, pre, rng, eps_start=0.08, tag="div_" + nm)
            if np.isfinite(sc2) and sc2 < best_score - 1e-10:
                best_seq = a2
                best_score = sc2
                best_name = nm
                print(f"  improved diverse {nm}: {best_score:.9f}")

    rem = max(0.0, deadline - time.monotonic())
    t_mult = min(280.0, max(0.0, rem - 350.0))
    if t_mult > 80.0:
        print(f"Starting multiplicative reshape: {t_mult:.1f}s")
        a_m, sc_m, cur_m = multiplicative_phase(
            best_seq,
            t_mult,
            beta=20.0,
            w_max=1.0,
            w_var=0.2,
            w_ent=0.2,
            q_var=0.5,
            q_ent=1.0,
            score_tol=0.01,
            tag="mp_safe",
        )
        if np.isfinite(sc_m) and sc_m < best_score - 1e-10:
            best_seq = a_m
            best_score = sc_m
            print(f"  multiplicative improved: {best_score:.9f}")

    rem = max(0.0, deadline - time.monotonic())
    t_mult2 = min(160.0, max(0.0, rem - 220.0))
    if t_mult2 > 40.0:
        print(f"Starting entropy/KL exploratory: {t_mult2:.1f}s")
        a_m2, sc_m2, _ = multiplicative_phase(
            best_seq,
            t_mult2,
            beta=0.0,
            w_max=0.5,
            w_var=0.2,
            w_ent=1.0,
            q_var=1.0,
            q_ent=2.0,
            score_tol=0.03,
            tag="mp_kl",
        )
        if np.isfinite(sc_m2) and sc_m2 < best_score - 1e-10:
            best_seq = a_m2
            best_score = sc_m2
            print(f"  KL exploratory improved: {best_score:.9f}")

    rem = max(0.0, deadline - time.monotonic())
    t_final = min(450.0, max(0.0, rem - 80.0))
    if t_final > 60.0:
        print(f"Starting final LP polish: {t_final:.1f}s")
        best_seq, best_score, _ = lp_polish(best_seq, t_final, pre, rng, eps_start=0.05, tag="final_lp")
        print(f"after final LP: best={best_score:.9f}")

    rem = max(0.0, deadline - time.monotonic())
    t_cd = min(120.0, max(0.0, rem - 30.0))
    if t_cd > 25.0:
        print(f"Starting coordinate polish: {t_cd:.1f}s")
        a_cd, sc_cd = coordinate_descent_polish(best_seq, t_cd)
        if np.isfinite(sc_cd) and sc_cd < best_score - 1e-10:
            best_seq = a_cd
            best_score = sc_cd
            print(f"  coordinate polish improved: {best_score:.9f}")

    # Final exact verification over a few saved candidates.
    final_candidates = []
    if best_seq is not None:
        final_candidates.append(("best", normalize_sequence(best_seq)))
        final_candidates.append(("best_rev", normalize_sequence(best_seq[::-1].copy())))

    for key in ("parent", "flat_best", "invsqrt", "smooth"):
        if key in starts_dict:
            final_candidates.append((key, normalize_sequence(starts_dict[key])))

    best_final = None
    best_exact = np.inf
    for _, cand in final_candidates:
        cand = normalize_sequence(cand)
        sc = safe_evaluate(cand.tolist())
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_final = cand

    if best_final is None:
        best_final = np.ones(n, dtype=np.float64) / float(n)

    best_final = np.maximum(np.asarray(best_final, dtype=np.float64), 0.0)
    best_final = normalize_sequence(best_final)

    print(f"FINAL internal score: {best_score:.9f}")
    print(f"FINAL exact score: {best_exact:.9f}")

    return [float(x) for x in best_final]
```
