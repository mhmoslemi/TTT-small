```python
                best_score_new = evaluate_obj_new(best_sequence_candidate)
                
                best_sequence = best_sequence_candidate if best_score_new < best_score else best_sequence
                best_score = best_score_new
                
            except:
                pass
                
    except Exception as e:
        print("Major candidate generation error:", str(e))
        fallback = height_sequence_1 if height_sequence_1 is not None else np.ones(512)/np.sqrt(2.0*512)
        best_sequence = fallback
        best_score = evaluate_obj_new(fallback)
        
    # Final validation
    if best_sequence is not None:
        best_sequence = np.clip(best_sequence, 0, 1000)
        best_score = evaluate_obj_new(best_sequence)
        
    # Lower score is better
    return list(best_sequence) if best_sequence is not None else []
```
