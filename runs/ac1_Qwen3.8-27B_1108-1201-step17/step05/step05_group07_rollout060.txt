```python
def lp_direction(a, M):
    """Solve LP to find best direction."""
    n = len(a)
    # Variables: b_0, ..., b_{n-1} >= 0
    # Constraints: for each k in 0..2n-2: sum_{i} a[i] * b[k-i] <= M
    #   where b[j] = 0 if j < 0 or j >= n
    # Objective: maximize sum(b)
    
    import cvxpy as cp
    
    b = cp.Variable(n, non_neg=True)
    constraints = []
    for k in range(2*n - 1):
        # sum_i a[i] * b[k-i] <= M
        # valid i: max(0, k-(n-1)) <= i <= min(n-1, k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        i_vals = np.arange(i_start, i_end + 1)
        b_idx = k - i_vals  # indices into b
        constraint = a[i_vals] @ b[b_idx] <= M
        constraints.append(constraint)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.GLPK)  # or another solver
    return b.value
```

```python
import numpy as np
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T):
    """Compute soft-max smoothed objective and its gradient w.r.t. z."""
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    
    # Compute grad_a = 2n * (w cross-correlate a)
    # (w cross-correlate a)[i] = sum_j w[i+j] * a[j]
    fft_len = 1 << (2 * n - 1).bit_length()
    W = rfft(w, fft_len)
    A = rfft(a, fft_len)
    cc = irfft(np.conj(A) * W, fft_len)
    grad_a = 2.0 * n * cc[:n]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_direction(a, M, time_limit=30.0):
    """Solve LP to find the best direction to reduce max convolution.
    
    maximize sum(b) subject to C @ b <= M, b >= 0
    where C is the convolution matrix of a.
    
    Returns (b, optimal_value) or (None, None) if infeasible/timeout.
    """
    n = len(a)
    b_var = cp.Variable(n, non_neg=True)
    
    # Build the convolution constraint: cross_conv(a, b)_k <= M for all k
    # cross_conv(a, b)[k] = sum_j a[k-j] * b[j] for valid j
    # This is C @ b where C[k, j] = a[k-j] if 0<=k-j<n else 0
    
    # For efficiency, build the constraint using the structure
    # C @ b = np.convolve(a, b) but only the first 2n-1 elements
    # We can express this as a matrix product
    
    # Build C efficiently
    # C has shape (2n-1, n), C[k,j] = a[k-j] if 0 <= k-j < n else 0
    # Each column j has non-zeros in rows j to j+n-1
    # Use the fact that C is a "sliding window" of a
    
    # For CVXPY, let's build the expression directly
    # conv(a, b) in CVXPY: we can use the matrix form
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    constraints = [C @ b_var <= M, b_var >= 0]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    # Try solvers in order of preference
    solvers = [cp.GLPK, cp.CBC, cp.ECOS, cp.SCS, cp.CVXOPT]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return b_var.value, prob.value
        except Exception:
            continue
    
    # Fallback: use the default
    try:
        prob.solve()
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return b_var.value, prob.value
    except Exception:
        pass
    
    return None, None

def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit enforcement."""

            return 1e15
        obj, _ = soft_max_obj_grad(z, T)
        return obj

            fun=func,
            x0=z0,
            jac=jac,

                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun

def coordinate_descent(a0, time_limit=60.0):

    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False

                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj
```
