def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def _smooth_obj_grad_impl(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space (for L-BFGS)."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # grad_a[i] = 2 * sum_m a[m] * w[m+i]  (cross-correlation)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    # grad_t = a * (grad_a - <a, grad_a>)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

def mwu_optimize(f0, time_limit, verbose=False):
    """Multiplicative Weight Update with smooth gradient and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta_ls = 30.0
    eta = 0.05
    momentum = 0.7
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta_ls * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        v = momentum * v + (1.0 - momentum) * grad_a

        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and (best_obj < 1.51):
                print(f"    MWU: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            eta *= 0.97
            if no_improve > 300:
                # Restart with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                v = np.zeros(n)
                eta = 0.03
                no_improve = 0

        eta = max(eta, 0.002)

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
