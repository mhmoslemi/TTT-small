import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def smooth_obj_and_grad(t, beta):
    """Log-sum-exp smoothed objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    # Gradient via cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t

def solve_minmax_direction(f, n):
    """Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0."""
    # Variables: g[0..n-1], t
    # Total: n+1 variables
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    # Constraints: (f*g)[k] - t <= 0 for k=0..2n-2
    #              sum(g) = 1
    #              g[i] >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: (f*g)[k] - t <= 0
    # A_ub * [g, t] <= b_ub
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[k, j] = f[j]
        A_ub[k, n] = -1.0  # -t
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g[i] >= 0, t free
    # Actually t >= 0 since (f*g)[k] >= 0
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:

    return None, None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def build_conv_matrix(f, n):
    """Build the convolution constraint matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        js = max(0, 0)
        je = min(2 * n - 1, j + n)
        A[j:min(2 * n - 1, j + n), j] = f[:min(n, 2 * n - 1 - j)]
    return A


def solve_minmax_lp(f, n, active_k=None):
    """Solve min-max LP: min t s.t. (f*g)[k] <= t for k in active_k, sum(g)=1, g>=0."""
    if active_k is None:
        active_k = list(range(2 * n - 1))
    
    m = len(active_k)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    for idx, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        A_ub[idx, js:je] = f[:je - js]
        A_ub[idx, n] = -1.0
    
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n].copy(), res.x[n]
    return None, None


def solve_maxsum_lp(f, M, n, active_k=None):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for k in active_k, g >= 0."""
    if active_k is None:
        active_k = list(range(2 * n - 1))
    
    m = len(active_k)
    c = -np.ones(n)
    A_ub = np.zeros((m, n))
    for idx, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        A_ub[idx, js:je] = f[:je - js]
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x.copy()
    return None

def smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def adam_smooth_optimize(f0, time_limit, betas=None, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    t0_time = time.monotonic()
    
    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)
        
        # Adam state
        m_t = np.zeros(n)
        v_t = np.zeros(n)
        step = 0
        lr = 0.5
        b1, b2 = 0.8, 0.999
        eps_adam = 1e-8
        
        t_start = time.monotonic()
        no_improve = 0
        
        while time.monotonic() - t_start < sub_time:
            step += 1
            obj, grad = smooth_obj_grad(t, beta)
            
            # Adam update
            m_t = b1 * m_t + (1 - b1) * grad
            v_t = b2 * v_t + (1 - b2) * grad**2
            m_hat = m_t / (1 - b1**step)
            v_hat = v_t / (1 - b2**step)
            t_new = t - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            
            a_new = softmax_fn(t_new)
            a_new = np.maximum(a_new, 0.0)
            a_sum = a_new.sum()
            if a_sum < 1e-15:

            if real_obj < best_obj - 1e-14:

                best_f = a_new.copy()
                no_improve = 0
                t = t_new

            # Adaptive lr

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local optimization per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-13:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 3.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:

                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            # Try adding small mass to zero coordinates
            zeros = np.where(f < 1e-14)[0]
            if len(zeros) > 0:
                z = zeros[rng_local.integers(0, len(zeros))] if False else zeros[0]
                f[z] = f.max() * 0.01

                obj = fast_convolve(f).max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
            if not improved:
                break
    
    return best_f, best_obj


def sg_polish(f0, time_limit, lr_init=0.003, mom=0.94, top_k=50, temp_init=0.0001):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    
    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9998
        cur_temp *= 1.002
        
        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0005)
            no_improve = 0
    
    return best_f, best_obj

    asymmetric perturbation search, multi-length exploration, and polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        print(f"Using uniform parent: n={len(parent)}")
    
    n_parent = len(parent)
    
    # === Phase 1: Asymmetric perturbation search (200s) ===
    print("\n=== Phase 1: Asymmetric Perturbation (200s) ===")
    t1 = min(200, deadline - time.monotonic() - 1400)
    if t1 > 20:
        a_opt, _ = asymmetric_perturb_search(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "asym_perturb")
    
    # === Phase 2: Min-max LP-FW (500s) ===
    print(f"\n=== Phase 2: LP-FW Min-Max (500s) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(500, deadline - time.monotonic() - 1000)
    if t2 > 40:
        a_opt, _ = lpfw_optimize(base, time_limit=t2, top_k_frac=0.4, verbose=True)
        try_seq(a_opt, "lpfw_minmax")
        
        # Also try from reversed
        t2r = min(100, deadline - time.monotonic() - 800)
        if t2r > 20:
            a_rev = base[::-1].copy()
            a_opt, _ = lpfw_optimize(a_rev, time_limit=t2r, top_k_frac=0.3)
            try_seq(a_opt, "lpfw_reversed")
    
    # === Phase 3: Adam smooth optimization (400s) ===
    print(f"\n=== Phase 3: Adam Smooth (400s) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(400, deadline - time.monotonic() - 600)
    if t3 > 40:
        a_opt, _ = adam_smooth_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "adam_main")
        
        # Adam from perturbed start
        t3b = min(150, deadline - time.monotonic() - 400)
        if t3b > 20:
            noise = rng.standard_normal(len(base)) * 0.02
            f_pert = project_simplex(base + noise)
            a_opt, _ = adam_smooth_optimize(f_pert, time_limit=t3b,
                                            betas=[20, 40, 80, 160, 320])
            try_seq(a_opt, "adam_perturbed")
    
    # === Phase 4: Multi-length exploration (250s) ===
    print(f"\n=== Phase 4: Multi-length (250s) ===")
    base = best_seq if best_seq is not None else parent
    
    target_lengths = []
    # Generate candidate lengths around parent
    for factor in [0.75, 1.25, 1.5, 2.0]:
        cand_n = int(n_parent * factor)
        cand_n = max(64, (cand_n // 64) * 64)  # Round to multiple of 64
        if cand_n not in target_lengths and cand_n <= 4096:
            target_lengths.append(cand_n)
    
    # Always include some standard lengths
    for std_n in [512, 1024, 1536, 2048]:
        if std_n not in target_lengths:
            target_lengths.append(std_n)
    
    target_lengths = sorted(set(target_lengths))[:4]
    
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.45, 80)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        
        # LP-FW on interpolated
        a_opt, _ = lpfw_optimize(f_interp, time_limit=t_alloc * 0.6, top_k_frac=0.3)
        try_seq(a_opt, f"len_{target_n}_lpfw")
        
        # Adam on best of above
        if rem > 60:
            base2 = a_opt if (fast_convolve(a_opt).max() < fast_convolve(f_interp).max()) else f_interp
            t_alloc2 = min(rem * 0.3, 50)
            a_opt2, _ = adam_smooth_optimize(base2, time_limit=t_alloc2,
                                             betas=[20, 50, 100, 200])
            try_seq(a_opt2, f"len_{target_n}_adam")
    
    # === Phase 5: Deep optimization on best found (350s) ===
    print(f"\n=== Phase 5: Deep Opt (350s) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # 5a: LP-FW deep with full constraints
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_alloc = min(rem * 0.4, 150)
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc,
                                     top_k_frac=1.0, verbose=True)
            try_seq(a_opt, "deep_lpfw_full")
        
        # 5b: Asymmetric perturbation + LP-FW
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            a_pert, _ = asymmetric_perturb_search(best_seq, time_limit=min(rem * 0.3, 60))
            t_alloc = min(rem * 0.7, 100)
            a_opt, _ = lpfw_optimize(a_pert, time_limit=t_alloc, top_k_frac=0.5)
            try_seq(a_opt, "deep_asym_lpfw")
        
        # 5c: Adam deep
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem * 0.6, 120)
            a_opt, _ = adam_smooth_optimize(best_seq, time_limit=t_alloc,
                                            betas=[50, 100, 200, 400, 800],
                                            verbose=True)
            try_seq(a_opt, "deep_adam")
        
        # 5d: Subgradient polish
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.7, 80)
            a_opt, _ = sg_polish(best_seq, time_limit=t_alloc,
                                 lr_init=0.002, mom=0.95,
                                 top_k=max(20, n_best // 6),
                                 temp_init=0.00003)
            try_seq(a_opt, "deep_sg")
    
    # === Phase 6: Final polishing (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.4, 60))
            try_seq(a_opt, "final_cd")
        
        # Short LP-FW
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem * 0.5, 30),
                                     top_k_frac=0.5)
            try_seq(a_opt, "final_lpfw")
        
        # Final SG
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 15),
                                 lr_init=0.001, mom=0.96,
                                 top_k=max(10, n_best // 4),
                                 temp_init=0.00001)
            try_seq(a_opt, "final_sg")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
