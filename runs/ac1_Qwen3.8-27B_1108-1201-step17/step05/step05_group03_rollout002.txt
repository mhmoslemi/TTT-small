def fast_score(a):
    """Compute 2*n*max(auto_conv)/sum^2 efficiently using FFT."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = a / s
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    B = np.fft.rfft(b, fft_size)
    conv = np.fft.irfft(B * B, fft_size)[:size]
    return float(2.0 * n * conv.max())


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_conv(a):
    """Compute full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def coord_descent(a, time_limit, verbose=False, label=""):

            order = np.arange(n)
        else:
            order = np.random.permutation(n)

            if time.monotonic() - t0 > time_limit:
                break

                best_sc = best_score

                    others = 1.0 - old_v
                    if others < 1e-12:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    new_a[mask] *= (1.0 - cv) / others
                    sc = fast_score(new_a)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_new_a = new_a
                if best_new_a is not None:
                    a = normalize(best_new_a)
                    if best_sc < best_score:
                        best_score = best_sc
                        best_a = a.copy()
                        any_improved = True
                continue

            # Adaptive range around current value
            lo = max(0.0, old_v * widen_factor)
            hi = min(0.5, old_v * (2.0 - widen_factor))

            # Ensure range is reasonable

                            any_improved = True

def mirror_descent(a, time_limit, lr0=0.005, decay=0.9999, top_k=40, verbose=False, label=""):
    """Multiplicative subgradient (mirror descent) with adaptive peak weighting."""
    n = len(a)
    a = normalize(a.copy())
    best_score = fast_score(a)
    best_a = a.copy()
    cur_lr = lr0
    t0 = time.monotonic()
    step = 0

    # Running average (Nemirovski-Yudin)
    avg_a = np.zeros(n)
    avg_count = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = compute_conv(a)
        peak = conv.max()

        # Adaptive top-k: include all peaks within 1% of max
        threshold = peak * 0.995
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        k_actual = min(len(active), top_k * 2)
        # Get top k_actual from active
        if len(active) > k_actual:
            active_vals = conv[active]
            top_idx = np.argpartition(active_vals, -k_actual)[-k_actual:]
            active = active[top_idx]

        # Softmax weights
        pv = conv[active]
        temp = max(1e-8, pv.max() - pv.min()) * 0.3
        if temp < 1e-10:
            w = np.ones(len(active)) / len(active)
        else:
            scaled = (pv - pv.max()) / temp
            w = np.exp(scaled)
            w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(active)):
            if w[idx] < 1e-14:
                continue
            k = int(active[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            indices = np.arange(js, je + 1)
            sg[indices] += 2.0 * w[idx] * a[k - indices]

        # Multiplicative update (mirror map for entropy)
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-18)
        a = a / a.sum()

        # Track average
        avg_a += a
        avg_count += 1

        sc = fast_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        # Check average too (every 100 steps)
        if step % 100 == 0:
            avg_candidate = avg_a / avg_count
            sc_avg = fast_score(avg_candidate)
            if sc_avg < best_score - 1e-14:
                best_score = sc_avg
                best_a = avg_candidate.copy()

        cur_lr *= decay

        # Restart if learning rate too small
        if cur_lr < lr0 * 0.05:
            cur_lr = lr0 * 0.3
            noise = np.random.standard_normal(n) * 0.001
            a = normalize(best_a + noise)
            avg_a = a.copy()
            avg_count = 1

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD[{label}] step {step}: score={best_score:.10f}, lr={cur_lr:.6f} ({elapsed:.0f}s)")

    return best_a, best_score


def lp_direction(f, M, solver=None):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k.
    Returns the normalized direction g, or None if LP fails.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]

    # Build convolution constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            fs = f[k - js]
            constraints.append(fs @ g[js] <= M)

    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    # Try solvers in order of preference
    solvers = [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]
    if solver is not None:
        solvers = [solver] + [s for s in solvers if s != solver]

    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                g_val = np.asarray(g.value, dtype=np.float64)
                g_val = np.maximum(g_val, 0.0)
                s_val = g_val.sum()
                if s_val > 1e-10:
                    return g_val / s_val
                return None
        except Exception:
            continue

    return None

    return best_a, best_score

def geometric_init(n, r):
    """Geometric sequence, normalized."""
    a = (1.0 - r) * r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_init(n):
    """Cosine-shaped (sin^2) sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped sequence."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def trapezoid_init(n, ratio=0.5):
    """Trapezoidal window."""
    a = np.ones(n)
    m = int(n * ratio / 2)
    if m > 0:
        ramp = np.linspace(0, 1, m)
        a[:m] = ramp
        a[-m:] = ramp[::-1]
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Combines multiple optimization strategies
    with adaptive time allocation across sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_state = {'score': float('inf'), 'a': None}

    print(f"=== Starting optimization (budget={budget_s}s) ===")

    # === Phase 0: Evaluate starting points ===
    print("\n--- Phase 0: Starting points ---")
    candidates = []

    # Parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            candidates.append((parent, "parent"))
            psc = fast_score(parent)
            print(f"  Parent: n={len(parent)}, score={psc:.10f}")
            try_candidate(parent, best_state, "parent_eval")
    except Exception:
        pass

    # Structured inits at various lengths
    for n in [512, 640, 768, 896, 1024]:
        candidates.append((geometric_init(n, 0.999), f"geom_n{n}"))
        candidates.append((cosine_init(n), f"cos_n{n}"))
        candidates.append((gaussian_init(n, 0.35), f"gauss_n{n}"))
        candidates.append((trapezoid_init(n, 0.6), f"trap_n{n}"))

    # Evaluate all quickly
    for a, label in candidates:
        try_candidate(a, best_state, label)

    print(f"  Best after Phase 0: {best_state['score']:.10f}")

    # === Phase 1: LP-based optimization (if available and fast) ===
    print("\n--- Phase 1: LP-based optimization ---")
    if best_state['a'] is not None:
        rem = deadline - time.monotonic() - 600
        # Give LP at most 300s or 30% of remaining time
        t_lp = min(rem, 300)
        if t_lp > 30:
            print(f"  Allocating {t_lp:.0f}s for LP")
            a_lp, sc_lp = lp_optimize(best_state['a'], time_limit=t_lp,
                                       step_size=0.1, verbose=True, label="main")
            try_candidate(a_lp, best_state, "lp_main")

    print(f"  Best after Phase 1: {best_state['score']:.10f}")

    # === Phase 2: Mirror descent (subgradient) exploration ===
    print("\n--- Phase 2: Mirror descent ---")
    if best_state['a'] is not None:
        rem = deadline - time.monotonic() - 400
        t_md = min(rem, 200)
        if t_md > 20:
            print(f"  Allocating {t_md:.0f}s for mirror descent")
            a_md, sc_md = mirror_descent(best_state['a'], time_limit=t_md,
                                         lr0=0.003, decay=0.99995,
                                         top_k=50, verbose=True, label="main")
            try_candidate(a_md, best_state, "md_main")

    print(f"  Best after Phase 2: {best_state['score']:.10f}")

    # === Phase 3: Precise coordinate descent (main workhorse) ===
    print("\n--- Phase 3: Coordinate descent ---")
    if best_state['a'] is not None:
        rem = deadline - time.monotonic() - 200
        t_cd = min(rem, 700)
        if t_cd > 20:
            print(f"  Allocating {t_cd:.0f}s for coordinate descent")
            a_cd, sc_cd = coord_descent(best_state['a'], time_limit=t_cd,
                                         verbose=True, label="main")
            try_candidate(a_cd, best_state, "cd_main")

    print(f"  Best after Phase 3: {best_state['score']:.10f}")

    # === Phase 4: Dimension exploration ===
    print("\n--- Phase 4: Dimension exploration ---")
    if best_state['a'] is not None:
        best_n = len(best_state['a'])
        target_ns = [512, 640, 768, 896, 1024, 1280]
        target_ns = [tn for tn in target_ns if tn != best_n]

        for target_n in target_ns:
            rem = deadline - time.monotonic() - 100
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 120)
            a_interp = interpolate_sequence(best_state['a'], target_n)
            a_opt, sc = coord_descent(a_interp, time_limit=t_alloc,
                                       verbose=False, label=f"n{target_n}")
            try_candidate(a_opt, best_state, f"dim_{target_n}")

    print(f"  Best after Phase 4: {best_state['score']:.10f}")

    # === Phase 5: Pair perturb + final CD polish ===
    print("\n--- Phase 5: Final polish ---")
    if best_state['a'] is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            # Pair perturb
            t_ps = min(rem * 0.3, 100)
            if t_ps > 10:
                a_ps, sc_ps = pair_perturb_local(best_state['a'], time_limit=t_ps,
                                                  n_trials=1000, label="final")
                try_candidate(a_ps, best_state, "pair_perturb")

            # Final CD
            rem = deadline - time.monotonic() - 15
            if rem > 15:
                a_final, sc_final = coord_descent(best_state['a'], time_limit=rem,
                                                   verbose=True, label="final")
                try_candidate(a_final, best_state, "final_cd")

    # === Phase 6: Try a few random perturbations of best ===
    print("\n--- Phase 6: Random perturbations ---")
    if best_state['a'] is not None:
        n_best = len(best_state['a'])
        for trial in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            noise_scale = 0.003 * (trial + 1)
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = normalize(best_state['a'] + noise)
            a_opt, sc = coord_descent(a_pert, time_limit=min(rem, 40),
                                       verbose=False, label=f"pert{trial}")
            try_candidate(a_opt, best_state, f"perturb_{trial}")

    # === Final verification ===
    print("\n--- Final verification ---")
    if best_state['a'] is None:
        # Fallback
        best_state['a'] = cosine_init(768)
        best_state['score'] = fast_score(best_state['a'])

    final_a = best_state['a']
    final_score_internal = fast_score(final_a)
    print(f"  Internal score: {final_score_internal:.10f} (n={len(final_a)})")

    # Verify with actual evaluate_sequence
    try:
        seq_list = [float(x) for x in final_a]
        final_score = float(evaluate_sequence(seq_list))
        if np.isfinite(final_score) and final_score < best_state['score']:
            best_state['score'] = final_score
        elif np.isfinite(final_score):
            best_state['score'] = final_score
        print(f"  Verified score via evaluate_sequence: {best_state['score']:.10f}")
    except Exception as e:
        print(f"  Verification error: {e}")
        best_state['score'] = final_score_internal

    elapsed = time.monotonic() - t_start
    print(f"\n=== DONE in {elapsed:.0f}s === Final score: {best_state['score']:.10f} (n={len(best_state['a'])})")

    return [float(x) for x in best_state['a']]
