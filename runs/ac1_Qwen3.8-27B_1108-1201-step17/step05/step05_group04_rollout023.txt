```python
import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    """Compute the objective 2n * max(conv(a,a)) for normalized a."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def smooth_max_and_grad(a, T):
    """Compute smooth max of auto-convolution and its gradient."""
    b = fast_convolve(a)
    shifted = b / T
    shifted -= shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    obj = T * np.log(e.sum()) + b.max()  # numerically stable log-sum-exp
    # Gradient w.r.t. a
    g = (2.0 / T) * np.correlate(w, a, 'valid')
    return obj, g, w, b

def softmax_seq_to_x(a):
    """Convert sequence to log-space (for softmax parameterization)."""
    return np.log(np.maximum(a, 1e-30))

def x_to_softmax_seq(x):
    """Convert log-space to sequence."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def smooth_obj_and_grad_x(x, T):
    """Objective and gradient w.r.t. x (log-space parameterization)."""
    a = x_to_softmax_seq(x)
    obj_a, g_a, w, b = smooth_max_and_grad(a, T)
    # Chain rule: dL/dx[j] = a[j] * (g_a[j] - sum_i g_a[i]*a[i])
    s = np.dot(g_a, a)
    grad_x = a * (g_a - s)
    return obj_a, grad_x
```

```python
def solve_sao_lp(f, M):
    """Solve the SAO LP: max sum(g) s.t. g>=0, (f*g)[k] <= M."""
    n = len(f)
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i] * g[k-i] where 0 <= i < n and 0 <= k-i < n
    # = sum_{i=max(0,k-n+1)}^{min(k, n-1)} f[i] * g[k-i]
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A_ub[k, k - i] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using combined smooth-max optimization,
SAO LP iterations, and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def smooth_max_and_grad(a, T):
    """Compute smooth max of auto-convolution and gradient w.r.t. a."""
    b = fast_convolve(a)
    b_max = b.max()
    shifted = (b - b_max) / T
    e = np.exp(shifted)
    e_sum = e.sum()
    w = e / e_sum
    obj = b_max + T * np.log(e_sum)
    g = (2.0 / T) * np.correlate(w, a, 'valid')
    return obj, g, w, b


def x_to_seq(x):
    """Convert log-space to normalized sequence."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def seq_to_x(a):
    """Convert normalized sequence to log-space."""
    return np.log(np.maximum(a, 1e-30))


def smooth_obj_grad_x(x, T):
    """Objective and gradient w.r.t. log-space parameterization."""
    a = x_to_seq(x)
    obj, g_a, w, b = smooth_max_and_grad(a, T)
    s = np.dot(g_a, a)
    grad_x = a * (g_a - s)
    return obj, grad_x


def lbfgs_optimize_x(x, T_schedule, time_limit, maxiter_per_T=300):
    """L-BFGS-B optimization with temperature annealing in log-space."""
    best_a = x_to_seq(x)
    best_obj = compute_objective(best_a)
    
    t0 = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        result = minimize(
            smooth_obj_grad_x, x, args=(T,),
            method='L-BFGS-B', jac=True,
            options={'maxiter': maxiter_per_T, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        
        x = result.x
        a = x_to_seq(x)
        obj = compute_objective(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def build_conv_matrix(f):
    """Build matrix A such that A @ g = cross-correlation of f and g."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None].astype(np.float64)
    j = np.arange(n)[None, :].astype(np.float64)
    diff = (k - j).astype(int)
    mask = (diff >= 0) & (diff < n)
    diff_clipped = np.clip(diff, 0, n - 1)
    A = np.where(mask, f[diff_clipped], 0.0)
    return A


def sao_lp_iteration(f, M, t_inter=0.1, time_per_lp=2.0):
    """One SAO LP iteration."""
    n = len(f)
    b = fast_convolve(f)
    
    # Constraint: (f*g)[k] <= (M - b[k]) / (2*t)
    rhs = (M - b) / (2.0 * t_inter)
    rhs = np.maximum(rhs, 0.0)
    
    # Check if all constraints are zero (degenerate case)
    if rhs.max() < 1e-15:
        return None
    
    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = rhs
    bounds = [(0, None)] * n
    
    t_lp_start = time.monotonic()
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': time_per_lp})
    except Exception:
        return None
    
    if result.success and result.x is not None:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None


def sao_optimize(f0, time_limit, t_inter=0.1, max_iters=10000):
    """SAO LP-based optimization."""
    f = f0.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f0)
    else:
        f = f / s
    
    n = len(f)
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = 2.0 * n * b.max()
    
    t0 = time.monotonic()
    no_improve = 0
    consec_fail = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        
        g = sao_lp_iteration(f, M, t_inter=t_inter)
        
        if g is None:
            consec_fail += 1
            if consec_fail > 5:
                # Try with larger t_inter
                t_inter *= 2
                consec_fail = 0
            continue
        
        consec_fail = 0
        
        # Interpolate

        f_new = project_simplex(f_new)
        
        b_new = fast_convolve(f_new)
        obj_new = 2.0 * n * b_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0

                obj_try = 2.0 * n * b_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_f = f_try.copy()
                    f = f_try.copy()
                    no_improve = 0
                    break
            
            if no_improve > 50:
                # Reset t_inter
                t_inter = 0.1
                # Add small noise
                rng = np.random.default_rng(it)
                f = project_simplex(f + rng.standard_normal(n) * 0.001)
                no_improve = 0
    
    return best_f, best_obj
