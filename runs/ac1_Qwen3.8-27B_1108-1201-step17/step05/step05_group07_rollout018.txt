"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-start soft-max L-BFGS
with adaptive temperature annealing, multi-length exploration, and targeted
peak surgery. Optimizes in log-space (softmax parameterization) to maintain
non-negativity. Uses FFT-based convolution for O(n log n) objective/gradient
evaluation.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_soft(a, w, n):
    """
    Compute gradient of (2n) * soft_max_T(conv(a,a)) w.r.t. a.
    w[k] = exp(T*b[k]) / sum(exp(T*b)) normalized weights.
    """
    # c[j] = sum_{m=0}^{n-1} a[m] * w[j+m]  (cross-correlation)
    # This is (w ⋆ a)[j] = sum_n a[n] * w[n+j]
    # FFT: ifft(conj(FFT(w)) * FFT(a))
    N = 1 << (n + (2*n - 1) - 1 + 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    c = irfft(np.conj(W) * A, N)[:n]  # c[j] = sum_m a[m]*w[m+j] for j=0..n-1
    
    # grad[j] = 2n * (2*c[j] - w[2j]*a[j])
    grad = 2.0 * n * (2.0 * c - w[:2*n-1][0::2] * a)  # w[2j] for j=0..n-1 is w[0], w[2], w[4], ...
    
    return grad


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z (log-domain)."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient w.r.t. a
    N = 1 << (n + (2*n - 1) - 1 + 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    c = irfft(np.conj(W) * A, N)[:n]
    
    # w[2j] for j = 0, ..., n-1
    w_diag = w[0:min(2*n, 2*n-1):2]  # w[0], w[2], w[4], ...
    
    grad_a = 2.0 * n * (2.0 * c - w_diag * a)
    
    # Chain rule: grad_z_j = a_j * (grad_a_j - a_j * dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lbgfs_optimize(z0, T, n, time_limit, maxiter=5000):
    """L-BFGS optimization in log-space with time limit."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def peak_surgery(a0, n_peaks=5, n_iter=500, time_limit=60.0):
    """
    Targeted peak reduction: iteratively find the worst peaks and reduce
    them by adjusting contributing elements.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for _ in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top peaks (exclude the very edges which are just a[0]^2 and a[n-1]^2)
        b_masked = b.copy()
        b_masked[0] = -1e10
        b_masked[-1] = -1e10
        peak_idx = np.argsort(b_masked)[::-1][:n_peaks]
        
        improved = False
        for k in peak_idx:
            k = int(k)
            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            
            # Find the pair with the largest contribution
            max_i_local = np.argmax(contribs)
            i = idx[max_i_local]
            j = k - i
            
            # Try reducing a[i] and redistributing
            if a[i] > 1e-12:
                for factor in [0.9, 0.8, 0.7, 0.5]:
                    new_a = a.copy()
                    diff = a[i] * (1 - factor)
                    new_a[i] *= factor
                    # Redistribute to the "other" side of the peak
                    # Elements at positions that DON'T contribute to this peak
                    redist_mask = np.ones(n, dtype=bool)
                    redist_mask[idx] = False  # don't add to contributing elements
                    if redist_mask.sum() > 0:
                        new_a[redist_mask] += diff * new_a[redist_mask] / new_a[redist_mask].sum()
                        new_a = new_a / new_a.sum()
                        new_max = fast_convolve(new_a).max()
                        if new_max < best_obj - 1e-13:
                            a = new_a
                            best_obj = new_max
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            break
    
    return best_a, best_obj


def subgrad_step(a, T, n, lr, top_k=50):
    """One subgradient step with softmax-weighted peaks."""
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    scaled = pv * T
    scaled -= scaled.max()
    w_peaks = np.exp(scaled)
    ws = w_peaks.sum()
    if ws < 1e-15:
        w_peaks = np.zeros(len(peaks))
        w_peaks[np.argmax(pv)] = 1.0
    else:
        w_peaks /= ws
    
    # Compute gradient
    w = np.zeros(2*n - 1)
    for i, k in enumerate(peaks):
        w[int(k)] = w_peaks[i]
    
    # Use the same gradient computation
    N = 1 << (n + (2*n - 1) - 1 + 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    c = irfft(np.conj(W) * A, N)[:n]
    w_diag = w[0:min(2*n, 2*n-1):2]
    grad_a = 2.0 * n * (2.0 * c - w_diag * a)
    
    a_new = a - lr * grad_a
    a_new = project_simplex(a_new)
    return a_new
