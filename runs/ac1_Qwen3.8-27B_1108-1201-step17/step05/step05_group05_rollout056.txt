"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with temperature annealing, symmetry enforcement, and multi-length search.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u - (cssv - target_sum) / rho) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - target_sum) / (idx + 1)
    return np.maximum(v - theta, 0.0)


def soft_dobj(a, tau, conv=None):
    """Compute smoothed objective and its gradient."""
    if conv is None:
        conv = fast_convolve(a)
    n = len(a)
    shifted = tau * (conv - conv.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    obj = (conv.max() + np.log(w_sum)) / tau  # (1/tau)*log(sum exp(tau*conv_k))
    # Gradient via FFT
    w_rev = w[::-1]
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_aw = np.fft.irfft(A * W, fft_size)[:2 * n - 1]
    g = 2.0 * conv_aw[2 * n - 2 : n - 2 : -1]
    # Wait, I need to be careful with indexing

def flat_spectrum_init(n, rng, symmetric=True):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    half = n // 2
    for k in range(n):
        if k == 0:
            A[k] = np.sqrt(n)
        elif k == n - k:  # Nyquist for even n
            A[k] = 1.0
        else:
            A[k] = np.exp(1j * phases[k])
    # Ensure Hermitian symmetry for real output
    for k in range(1, n):
        if k != n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = abs(A[n // 2])  # real at Nyquist
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a /= s
    if symmetric:
        a = (a + a[::-1]) / 2
        a /= a.sum()
    return a


def optimize_smoothed(a0, time_limit, tau_init=200.0, tau_min=20.0, tau_decay=0.9995):
    """Smoothed gradient descent with temperature annealing and backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    # Initial objective
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    tau = tau_init
    lr = 0.05
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Compute smoothed gradient
        obj_smooth, g, conv, w = soft_dobj(a, tau, conv=fast_convolve(a))
        
        # Center gradient (remove mean for simplex compatibility)
        g_centered = g - g.mean()
        
        # Backtracking line search
        step = lr
        for _ in range(8):
            a_trial = project_simplex(a - step * g_centered, target_sum=1.0)
            # Enforce symmetry
            a_trial = (a_trial + a_trial[::-1]) / 2
            a_trial /= a_trial.sum()
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_obj + 1e-12 or obj_trial < obj_smooth:
                break
            step *= 0.5
        
        a = a_trial
        obj_new = fast_convolve(a).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Temperature scheduling
        if no_improve > 500:
            tau = max(tau_min, tau * 0.98)
            lr = max(0.01, lr * 0.95)
            # Restart with perturbation
            if no_improve > 1000:
                rng_r = np.random.default_rng(iter_count * 17 + 3)
                noise = rng_r.standard_normal(n) * 0.003
                a = project_simplex(best_a + noise, target_sum=1.0)
                a = (a + a[::-1]) / 2
                a /= a.sum()
                no_improve = 500
                tau = tau_init * 0.5
                lr = 0.03
        elif no_improve > 200:
            tau = max(tau_min, tau * 0.999)
        else:
            tau = min(tau_init * 2, tau * 1.001)
        
        if iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter {iter_count}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.4f}, no_imp={no_improve} ({elapsed:.0f}s)")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization in softmax
parameterization with temperature annealing on a smoothed (log-sum-exp) objective,
FFT-based gradient computation, symmetry enforcement, multi-length exploration,
and coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute the raw objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def smooth_obj_and_grad_a(z, tau):
    """
    Compute smoothed objective and gradient w.r.t. a (given a=softmax(z)).
    Returns (objective, grad_wrt_a, conv).
    """
    a = softmax(z)
    n = len(a)
    conv = fast_convolve(a)
    shifted = tau * (conv - conv.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    w_norm = w / w_sum
    obj = (conv.max() + np.log(w_sum)) / tau

    # Gradient w.r.t. a: g_i = 2 * sum_j w_{i+j} * a_j
    w_rev = w_norm[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_aw = np.fft.irfft(A * W, fft_size)[:conv_len]
    g_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]

    return obj, g_a, conv


def smooth_obj_and_grad_z(z, tau):
    """
    Compute smoothed objective and gradient w.r.t. z.
    F(z) = (1/tau) * log(sum_k exp(tau * conv(softmax(z), softmax(z))_k))
    """
    obj, g_a, conv = smooth_obj_and_grad_a(z, tau)
    a = softmax(z)
    # dF/dz_j = a_j * (g_a_j - sum_i g_a_i * a_i)
    dot = np.dot(g_a, a)
    g_z = a * (g_a - dot)
    return obj, g_z


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    for attempt in range(20):
        phases = rng.uniform(0, 2 * np.pi, n)
        A = np.zeros(n, dtype=complex)
        A[0] = np.sqrt(n)
        half = n // 2
        for k in range(1, n):
            if k == n - k:
                A[k] = 1.0
            else:
                A[k] = np.exp(1j * phases[k])
        # Enforce Hermitian symmetry
        for k in range(1, (n // 2) + 1):
            if k < n - k:
                A[n - k] = np.conj(A[k])
        if n % 2 == 0:
            A[n // 2] = float(abs(A[n // 2]))
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0)
        # Check quality
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        # Symmetrize
        a = (a + a[::-1]) / 2
        a = a / a.sum()
        obj = fast_convolve(a).max()
        if obj < 2.0 / n:
            break
    return a


def random_nonneg_init(n, rng):
    """Random non-negative initialization with symmetry."""
    half_n = n // 2 + 1
    b = rng.exponential(1.0, half_n)
    if n % 2 == 0:
        a = np.concatenate([b, b[:n // 2 - 1][::-1], b[0:1]])
    else:
        a = np.concatenate([b, b[:n // 2][::-1]])
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a /= a.sum()
    return a


def triangular_init(n):
    """Triangular (Bartlett) window."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(x - mid) / (mid + 0.5)
    a = np.maximum(a, 0)
    return a / a.sum()


def raised_cosine_init(n):
    """Raised cosine (Hann) window."""
    x = np.arange(n) / (n - 1)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    return a / a.sum()


def optimize_lbfqs(z0, time_limit, tau_schedule=None):
    """
    Run L-BFGS-B optimization with temperature annealing.
    z0: initial parameter vector (will be used as softmax input)
    Returns best (a, raw_objective) found.
    """
    n = len(z0)
    z = z0.copy()

    if tau_schedule is None:
        tau_schedule = [300.0, 100.0, 40.0, 15.0, 6.0]

    t0 = time.monotonic()
    a = softmax(z)
    best_a = a.copy()
    best_raw = fast_convolve(a).max()

    for tau in tau_schedule:
        remain = time_limit - (time.monotonic() - t0)
        if remain < 2:
            break

        # Objective and gradient for scipy
        def obj_func(z_local):
            val, grads, _ = smooth_obj_and_grad_z(z_local, tau)
            return val

        def grad_func(z_local):
            _, grads, _ = smooth_obj_and_grad_z(z_local, tau)
            return grads

        # Track time via callback
        state = {'stop': False}
        def callback(zk):
            if time.monotonic() - t0 >= time_limit - 1:
                state['stop'] = True
            return state['stop']

        try:
            result = scipy_minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B', callback=callback,
                options={'maxiter': 2000, 'maxfun': 50000, 'gtol': 1e-10}
            )
            z = result.x
        except Exception:
            pass

        a = softmax(z)
        raw_obj = fast_convolve(a).max()
        if raw_obj < best_raw:
            best_raw = raw_obj
            best_a = a.copy()

        # Symmetrize and renormalize
        a = (a + a[::-1]) / 2
        a = a / a.sum()
        z = np.log(a + 1e-30)
        raw_obj = fast_convolve(a).max()
        if raw_obj < best_raw:
            best_raw = raw_obj
            best_a = a.copy()

    return best_a, best_raw


def coordinate_descent(a0, time_limit):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-12:
                continue
            old_val = a[j]
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    return best_a, best_obj


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
