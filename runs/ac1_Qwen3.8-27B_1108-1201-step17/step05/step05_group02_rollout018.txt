"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with softmax parameterization,
projected subgradient, LP-FW polishing, and multi-length exploration.
Key innovation: smooth optimization via log-sum-exp + L-BFGS, combined with
the previous LP-FW and CD approaches for final polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def lse_on_simplex(a, beta):
    """
    Compute log-sum-exp of conv(a,a) and gradient, given a on simplex.
    Returns (objective, gradient_wrt_x) where a = softmax(x).
    """
    n = len(a)
    conv = fast_convolve(a)
    sum_a = a.sum()  # should be 1
    
    # Log-sum-exp of conv
    conv_max = conv.max()
    shifted = conv - conv_max
    log_sum_exp = conv_max + np.log(np.sum(np.exp(beta * shifted))) / beta
    # Actually: LSE = max + (1/beta)*log(sum(exp(beta*(x - max))))
    LSE = conv_max + (1.0 / beta) * np.log(np.sum(np.exp(beta * shifted)))
    
    # Softmax weights
    p = np.exp(beta * shifted)
    p /= p.sum()
    
    # dLSE/da[j] = sum_k p[k] * a[k-j] * 1(0<=k-j<=n-1)
    # = sum_{m=0}^{n-1} p[j+m] * a[m]
    grad_a = np.zeros(n)
    for m in range(n):
        grad_a += a[m] * p[m:m+n]
    
    # Objective: LSE / sum_a^2 = LSE (since sum_a = 1)
    obj = LSE  # since sum_a = 1
    
    # For the softmax parameterization:
    # a = softmax(x), da/dx[j] = a[j] * (e_j - a)
    # dx_to_obj = sum_j (dobj/da[j]) * a[j] * (e_j - a)
    # = sum_j grad_a[j] * a[j] - a * (sum_j grad_a[j] * a[j])
    
    obj_val = np.sum(grad_a * a)
    grad_x = grad_a * a - a * obj_val
    
    return obj, grad_x


def lbfgs_optimize(x0, time_limit, beta_start=10.0, beta_end=5000.0, verbose=False):
    """
    L-BFGS optimization with softmax parameterization.
    x0 is the log-amplitude vector (a = softmax(x0)).
    """
    n = len(x0)
    a0 = softmax(x0)
    
    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()
    
    t0 = time.monotonic()
    
    # Annealing schedule for beta
    n_anneal = 8
    betas = np.geomspace(beta_start, beta_end, n_anneal)
    
    x = x0.copy()
    
    for bi, beta in enumerate(betas):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        
        t_alloc = min(rem * 0.6, 120)
        
        def fun_and_grad(x):
            a = softmax(x)
            obj, grad = lse_on_simplex(a, beta)
            return obj, grad
        
        try:
            res = minimize(fun_and_grad, x, method='L-BFGS-B',
                          options={'maxiter': 500, 'maxfun': 1000,
                                   'ftol': 1e-15, 'gtol': 1e-10})
            if res.success or res.fun < 1e-10:
                x = res.x
        except Exception as e:
            if verbose:
                print(f"  L-BFGS error at beta={beta}: {e}")
            break
        
        # Evaluate true objective
        a = softmax(x)
        true_obj = fast_convolve(a).max()
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    beta={beta:.1f} ({elapsed:.0f}s): true_obj={best_obj:.12f}")
    
    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.1
            if m_relax > 10.0:
                m_relax = 1.0
                continue
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.01, 8.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            m_relax = 1.0
            f = f_new
            
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 8.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.001 + 0.003 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            M = fast_convolve(f).max()
            m_relax = 1.0
            no_improve = 0
    
    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.005, mom=0.9, top_k=100, temp=0.001):
    """Subgradient optimization with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    f_mom = f.copy()  # Nesterov momentum position
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Use momentum position for gradient computation
        conv_m = fast_convolve(f_mom)
        k_actual = min(top_k, len(conv_m))
        peaks = np.argpartition(conv_m, -k_actual)[-k_actual:]
        
        pv = conv_m[peaks]
        scaled = pv / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_mom[k - idx]
        
        # Nesterov momentum
        v = mom * v - lr * sg
        f_trial = f_mom + v
        f = project_simplex(f_trial)
        f_mom = f + mom * (f - f_mom)  # Nesterov lookahead
        f_mom = np.maximum(f_mom, 0)
        sm = f_mom.sum()
        if sm > 1e-15:
            f_mom /= sm
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay learning rate
        lr *= 0.999
        temp *= 1.01
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            f_mom = f.copy()
            v = np.zeros(n)
            lr = 0.003
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-15:
                continue
                
            multipliers = [0.3, 0.5, 0.7, 1.4, 2.0, 3.0]
            best_j_obj = cur_max
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    cur_conv = fast_convolve(f)
                    cur_max = cur_conv.max()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    f_interp = interp1d(old_x, f, kind='cubic', fill_value='extrapolate')(new_x)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
