            if time.monotonic() - t0 > time_limit:
                break
            
            v = a[j]
            # p_k = a_{k-j} for k = 0,...,2n-2 (zero where invalid)
            p = np.zeros(2 * n - 1)
            p[j:j + n] = a
            
            k_2j = 2 * j
            has_2j = (0 <= k_2j <= 2 * n - 2)
            
            # 1D objective: minimize score as function of new value v'
            def obj_vp(v_new):
                if v_new < -1e-12:
                    return float('inf')
                t = v_new - v
                s_new = s + t
                if s_new < 1e-12:
                    return float('inf')
                b_prime = b + 2.0 * t * p
                if has_2j:
                    b_prime[k_2j] -= t * t
                max_bp = b_prime.max()
                return 2.0 * n * max_bp / (s_new * s_new)
            
            # Bounds
            lo = max(0.0, v - min(v, 0.3 * s))
            hi = min(s * 0.99, v + min(0.5 * s, max(0.1 * v, 0.02 * s)))
            if hi <= lo:
                hi = lo + 0.01 * s
            
            try:
                res = minimize_scalar(obj_vp, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-11, 'maxiter': 50})
                if res.success:
                    v_new = max(0.0, res.x)
                    t = v_new - v
                    if abs(t) > 1e-13:
                        a[j] = v_new
                        s += t
                        # Update b incrementally
                        b += 2.0 * t * p
                        if has_2j:
                            b[k_2j] -= t * t
                        sc = 2.0 * n * b.max() / (s * s)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best_a = a.copy()
                            improved = True
            except Exception:
                pass
        
        if not improved and pass_num > 2:
            break
    
    return best_a, best_score

def lp_multiplicative_step(f, time_limit, n_iters=50, solver=None):
    """
    LP-based multiplicative update.
    At each step: solve max sum(g) s.t. g>=0, f*g <= M.
    Then update f <- (1-t)*f + t*g_normalized.
    """
    n = len(f)
    f = normalize(f.copy())
    M = fast_convolve(f).max()
    best_a = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Try to find a working solver
    if solver is None:
        for s in ['ECOS', 'CLARABEL', 'SCS', 'CVXOPT']:
            try:
                # Quick test
                x = cp.Variable(2)
                prob = cp.Problem(cp.Maximize(x[0] + x[1]), [x >= 0, x[0] + x[1] <= 1])
                prob.solve(solver=s)
                solver = s
                break
            except Exception:
                continue
        if solver is None:
            solver = 'ECOS'
    
    lr = 0.3  # learning rate for the convex combination
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k
        g = cp.Variable(n)
        obj = cp.Maximize(cp.sum(g))
        constraints = [g >= 0]
        
        # Build convolution constraints
        # (f*g)_k = sum_j f_{k-j} * g_j <= M
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min > j_max:
                continue
            # Coefficients: f_{k-j} for j = j_min, ..., j_max
            # k-j ranges from k-j_max to k-j_min
            f_idx_start = k - j_max
            f_idx_end = k - j_min
            coeffs = f[f_idx_start:f_idx_end + 1]
            constraints.append(cp.sum(cp.multiply(coeffs, g[j_min:j_max + 1])) <= M)
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve()  # default solver
            except Exception:
                break
        
        if g.value is None:
            break
        g_val = np.maximum(g.value, 0)
        g_sum = g_val.sum()
        if g_sum < 1e-10:
            break
        
        g_norm = g_val / g_sum  # normalize to sum = 1
        
        # Update
        f_new = (1 - lr) * f + lr * g_norm
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        sc = compute_score(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        
        f = f_new
        M = fast_convolve(f).max()
        
        # Decay learning rate
        lr = max(0.05, lr * 0.99)
    
    return best_a, best_score

def subgradient_step(a, time_limit, lr0=0.01, top_k=30):
    """
    Averaged subgradient method with multiplicative updates.
    """
    n = len(a)
    a = normalize(a.copy())
    conv = fast_convolve(a)
    best_score = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        # Softmax weights
        temp = max(1e-6, pv.max() - pv.min()) * 0.5
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        a = np.maximum(a * np.exp(-lr * sg), 1e-15)
        a = a / a.sum()
        
        conv = fast_convolve(a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        lr *= 0.9995
        if lr < lr0 * 0.05:
            break
    
    return best_a, best_score

def gaussian_init(n, sigma_frac, center_frac=0.5):
    x = np.arange(n) - n * center_frac
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)

def cosine_init(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)

def tapered_init(n, power):
    """Raised-cosine-like taper: (sin(pi*x/n))^power"""
    x = np.arange(n)
    a = (np.sin(np.pi * x / n)) ** power
    return normalize(a)

def geometric_init(n, r):
    a = r ** np.arange(n)
    return normalize(a)

def random_smooth_init(n, rng, kernel_size=20):
    a = rng.random(n)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)

def pair_swap_refined(a, time_limit, n_trials=20000, rng=None):
    """
    Pair swap with binary search for optimal transfer amount.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    if rng is None:
        rng = np.random.default_rng(99)
    t0 = time.monotonic()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        
        # Transfer from j to i: a[i] += delta, a[j] -= delta
        # Find optimal delta via binary search / golden section
        a_i, a_j = a[i], a[j]
        if a_j < 1e-12:
            continue
        
        # Effective range: delta in [0, a_j] (transferring from j to i)
        # or delta in [-a_i, 0] (transferring from i to j)
        
        b = fast_convolve(a)
        s = a.sum()
        
        def obj_delta(delta):
            if delta < -a_i + 1e-12 or delta > a_j - 1e-12:
                return float('inf')
            s_new = s  # sum unchanged
            # b'_k = b_k + delta*(a_{k-i} - a_{k-j}) ... 
            # Actually: a[i] changes by +delta, a[j] changes by -delta
            # b'_k = b_k + delta*(a_{k-i} + a_{k-i}) - delta*(a_{k-j} + a_{k-j}) + cross terms
            # This is complex for 2D, let's just compute directly for small n
            new_a = a.copy()
            new_a[i] = a_i + delta
            new_a[j] = a_j - delta
            conv = fast_convolve(new_a)
            return 2.0 * n * conv.max() / (s * s)
        
        lo = -a_i * 0.5
        hi = a_j * 0.5
        try:
            res = minimize_scalar(obj_delta, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 30})
            if res.success:
                delta = res.x
                new_a = a.copy()
                new_a[i] = a_i + delta
                new_a[j] = a_j - delta
                if new_a[i] >= 0 and new_a[j] >= 0:
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = new_a.copy()
                        a = new_a.copy()
        except Exception:
            pass
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

def multi_stage_optimize(a_init, time_limit, n, verbose=False):
    """
    Run multiple optimization stages on a starting point.
    """
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    
    # Stage 1: Subgradient (exploratory, 20% time)
    t1 = time_limit * 0.2
    if time_limit - (time.monotonic() - t0) > t1 + 5:
        a1, sc1 = subgradient_step(a, time_limit=t1, lr0=0.005, top_k=max(20, n // 5))
        if sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            a = a1.copy()
        if verbose:
            print(f"        After SG: {best_score:.10f}")
    
    # Stage 2: Coordinate descent (40% time)
    rem = time_limit - (time.monotonic() - t0)
    t2 = rem * 0.4
    if rem > t2 + 5:
        a2, sc2 = coord_descent_1d(a, time_limit=t2, max_passes=500, rng=rng)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            a = a2.copy()
        if verbose:
            print(f"        After CD: {best_score:.10f}")
    
    # Stage 3: Pair swap (20% time)
    rem = time_limit - (time.monotonic() - t0)
    t3 = rem * 0.25
    if rem > t3 + 5:
        a3, sc3 = pair_swap_refined(a, time_limit=t3, n_trials=50000, rng=rng)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
            a = a3.copy()
        if verbose:
            print(f"        After PS: {best_score:.10f}")
    
    # Stage 4: Final CD polish (remaining time)
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a4, sc4 = coord_descent_1d(a, time_limit=min(rem, 300), max_passes=500, rng=rng)
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
        if verbose:
            print(f"        After final CD: {best_score:.10f}")
    
    return best_a, best_score
