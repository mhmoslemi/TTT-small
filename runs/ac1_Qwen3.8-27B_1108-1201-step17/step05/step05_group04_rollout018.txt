def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    x = np.linspace(0, 1, n)
    if shape == 'bump':
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'flat':
        f = np.ones(n)
    elif shape == 'cosine':
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Smooth random initialization as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_wiggle_init(n, freq=3, rng=None):
    """Wiggly initialization to break symmetry."""
    x = np.linspace(0, 1, n)
    if rng is None:
        rng = np.random.default_rng(0)
    f = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    f = np.maximum(f, 1e-8)
    return f / f.sum()


def focused_subgrad_optimize(f0, time_limit, lr_init=0.005, temp_init=0.004,
                             temp_decay=0.9997, top_k=5, verbose=False):
    """FISTA focused on top-k active constraints with tight temperature."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    x = f.copy()
    z = f.copy()
    t_k = 1.0
    step = lr_init
    temp = temp_init

    conv = fast_convolve(x)
    best_obj = conv.max()
    best_f = x.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        y = x + ((t_k - 1.0) / (t_k + 1.0)) * (x - z)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = x.copy()

        conv_y = fast_convolve(y)
        # Use very tight top-k for focused optimization
        w = softmax_weights(conv_y, temp, top_k)
        grad = compute_subgrad(y, w)

        x_new = project_simplex(y - step * grad)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x.copy()
        x = x_new
        t_k = t_new

        conv_x = fast_convolve(x)
        cur_obj = conv_x.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = x.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                step *= 0.8

        temp = max(temp * temp_decay, 5e-7)

        if no_improve > 80:
            rng = np.random.default_rng(it * 19 + 11)
            x = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            z = x.copy()
            t_k = 1.0
            step = lr_init
            temp = max(temp_init * 0.8, 0.001)
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    Focused it={it}: obj={best_obj:.10f}")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with FISTA, focused subgradient, pairwise search,
    coordinate refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent, using uniform n=768")

    n_parent = len(parent)
    print(f"Starting best: {best_score:.8f}")

    # === Phase 1: FISTA with multiple configs on parent ===
    print(f"\n=== Phase 1: FISTA multi-config (n={n_parent}) ===")
    t1 = time.monotonic()
    t1_end = t1 + min(400, (deadline - t1) * 0.25)

    configs_p1 = [
        {'lr_init': 0.008, 'temp_init': 0.006, 'temp_decay': 0.9994, 'top_k': max(10, n_parent // 30)},
        {'lr_init': 0.005, 'temp_init': 0.010, 'temp_decay': 0.9996, 'top_k': max(15, n_parent // 20)},
        {'lr_init': 0.012, 'temp_init': 0.004, 'temp_decay': 0.9993, 'top_k': max(8, n_parent // 40)},
        {'lr_init': 0.006, 'temp_init': 0.008, 'temp_decay': 0.9995, 'top_k': None},
    ]

    for ci, cfg in enumerate(configs_p1):
        rem = t1_end - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.6, 120)
        print(f"  FISTA cfg{ci} (lr={cfg['lr_init']}, temp={cfg['temp_init']}), t={t_alloc:.0f}s")
        a_opt, obj = fista_optimize(parent, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"p1_fista_{ci}")

    # Also try plain subgradient
    rem = t1_end - time.monotonic()
    if rem > 30:
        print(f"  Plain subgradient, t={min(rem*0.7, 80):.0f}s")
        a_opt, _ = plain_subgrad_optimize(parent, time_limit=min(rem * 0.7, 80),
                                          lr_init=0.02, temp_init=0.005,
                                          temp_decay=0.9995, top_k=max(10, n_parent // 30))
        try_seq(a_opt, "p1_subgrad")

    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    # === Phase 2: Diverse initializations ===
    print(f"\n=== Phase 2: Diverse starts (n={n_base}) ===")
    t2 = time.monotonic()
    t2_end = t2 + min(350, (deadline - t2) * 0.2)

    inits = {
        'gauss_tight': make_gaussian_init(n_base, 0.5, 0.12),
        'gauss_wide': make_gaussian_init(n_base, 0.5, 0.28),
        'bump': make_symmetric_init(n_base, 'bump'),
        'cosine': make_symmetric_init(n_base, 'cosine'),
        'wiggle': make_wiggle_init(n_base, 3, rng),
        'smooth_rand': make_smooth_random_init(n_base, rng, 7),
        'parent_mix': project_simplex(0.6 * base + 0.4 * make_gaussian_init(n_base, 0.5, 0.2)),
        'perturbed': project_simplex(base + rng.standard_normal(n_base) * 0.02),
    }

    for name, init in inits.items():
        rem = t2_end - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 50)
        a_opt, _ = fista_optimize(init, time_limit=t_alloc, lr_init=0.007,
                                  temp_init=0.007, temp_decay=0.9994,
                                  top_k=max(10, n_base // 25))
        try_seq(a_opt, f"p2_{name}")

    base = best_seq if best_seq is not None else base
    n_base = len(base)

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.8f}) ===")
    t3 = time.monotonic()
    t3_end = t3 + min(350, (deadline - t3) * 0.2)

    target_lengths = sorted(set([512, 640, 704, 768, 832, 896, 1024, 1088, 1152, 1280]),
                           key=lambda x: abs(x - n_base))
    target_lengths = [t for t in target_lengths if t != n_base and 256 <= t <= 1536][:5]

    for target_n in target_lengths:
        rem = t3_end - time.monotonic()
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.5, 80)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = fista_optimize(f_interp, time_limit=t_alloc, lr_init=0.006,
                                  temp_init=0.006, temp_decay=0.9994,
                                  top_k=max(10, target_n // 30))
        try_seq(a_opt, f"p3_n{target_n}")

        rem = t3_end - time.monotonic()
        if rem > 15:
            a_opt, _ = plain_subgrad_optimize(f_interp, time_limit=min(rem * 0.5, 30),
                                              lr_init=0.015, temp_init=0.005,
                                              temp_decay=0.9995)
            try_seq(a_opt, f"p3_sg_n{target_n}")

    base = best_seq if best_seq is not None else base
    n_base = len(base)

    # === Phase 4: Deep focused optimization ===
    print(f"\n=== Phase 4: Deep focused optimization (best={best_score:.8f}) ===")
    t4 = time.monotonic()
    t4_end = t4 + min(400, (deadline - t4) * 0.25)

    # Focused subgradient on best
    for fi in range(3):
        rem = t4_end - time.monotonic()
        if rem < 20:
            break
        start = base.copy()
        if fi > 0:
            start = project_simplex(base + rng.standard_normal(n_base) * 0.003)
        t_alloc = min(rem * 0.5, 100)
        print(f"  Focused run {fi}, t={t_alloc:.0f}s")
        a_opt, _ = focused_subgrad_optimize(start, time_limit=t_alloc, lr_init=0.004,
                                            temp_init=0.003, temp_decay=0.9997,
                                            top_k=5)
        try_seq(a_opt, f"p4_focused_{fi}")

    # FISTA with very tight temperature
    rem = t4_end - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.7, 120)
        print(f"  FISTA tight temp, t={t_alloc:.0f}s")
        a_opt, _ = fista_optimize(base, time_limit=t_alloc, lr_init=0.003,
                                  temp_init=0.003, temp_decay=0.9997,
                                  top_k=max(5, n_base // 50))
        try_seq(a_opt, "p4_tight")

    base = best_seq if best_seq is not None else base
    n_base = len(base)

    # === Phase 5: Pairwise and coordinate refinement ===
    print(f"\n=== Phase 5: Pairwise + coordinate (best={best_score:.8f}) ===")
    t5 = time.monotonic()
    t5_end = t5 + min(300, (deadline - t5) * 0.2)

    rem = t5_end - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.5, 100)
        print(f"  Pairwise, t={t_alloc:.0f}s")
        a_opt, _ = pairwise_optimize(base, time_limit=t_alloc)
        try_seq(a_opt, "p5_pair")

    rem = t5_end - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.6, 80)
        print(f"  Coordinate, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_refine(base, time_limit=t_alloc)
        try_seq(a_opt, "p5_coord")

    base = best_seq if best_seq is not None else base
    n_base = len(base)

    # === Phase 6: Final polishing loop ===
    print(f"\n=== Phase 6: Final polishing (best={best_score:.8f}) ===")
    polish_iter = 0
    while time.monotonic() < deadline - 30:
        polish_iter += 1
        rem = deadline - 30 - time.monotonic()
        if rem < 20:
            break

        # Alternate between focused FISTA and coordinate refinement
        if polish_iter % 2 == 1:
            t_alloc = min(rem * 0.4, 60)
            a_opt, _ = focused_subgrad_optimize(base, time_limit=t_alloc, lr_init=0.003,
                                                temp_init=0.002, temp_decay=0.9998, top_k=3)
            try_seq(a_opt, f"p6_focused_{polish_iter}")
            if best_seq is not None:
                base = best_seq
                n_base = len(base)
        else:
            t_alloc = min(rem * 0.4, 50)
            a_opt, _ = coordinate_refine(base, time_limit=t_alloc)
            try_seq(a_opt, f"p6_coord_{polish_iter}")
            if best_seq is not None:
                base = best_seq
                n_base = len(base)

            # Also try small FISTA
            rem2 = deadline - 30 - time.monotonic()
            if rem2 > 20:
                t_alloc2 = min(rem2 * 0.3, 30)
                a_opt, _ = fista_optimize(base, time_limit=t_alloc2, lr_init=0.002,
                                          temp_init=0.002, temp_decay=0.9998,
                                          top_k=max(5, n_base // 60))
                try_seq(a_opt, f"p6_fista_{polish_iter}")
                if best_seq is not None:
                    base = best_seq
                    n_base = len(base)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
