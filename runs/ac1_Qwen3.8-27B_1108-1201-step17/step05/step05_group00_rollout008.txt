def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, n_peaks=5):
    """Compute weighted subgradient targeting top peaks of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Find top peaks
    k = min(n_peaks, len(conv))
    peak_indices = np.argpartition(conv, -k)[-k:]
    
    # Weight peaks by their excess over the median
    peak_vals = conv[peak_indices]
    median_val = np.median(conv)
    weights = np.maximum(peak_vals - median_val, 0)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(len(peak_indices)) / len(peak_indices)
    else:
        weights /= wsum
    
    # Compute subgradient: sg[j] = sum_k w_k * 2*a[k-j]
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        if weights[i] < 1e-12:
            continue
        kk = int(peak_indices[i])
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * a[kk - idx]
    
    return sg, conv


def subgradient_boost(a0, time_limit, verbose=False, n_peaks=5, step_init=0.01, decay=0.999):
    """
    Penalty-based subgradient method with simplex projection.
    Fast per-iteration (no LP), allows many more iterations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, conv = compute_subgradient(a, n_peaks)
        
        # Subgradient step
        a_new = a - step * sg
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 100 == 0:
                print(f"    SB iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            a = a_new  # Still move, but don't update best
        
        step *= decay
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 13 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            step = step_init * 0.5
            no_improve = 0
        
        if step < 1e-8:
            break
    
    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def peak_targeted_cd(a0, time_limit, verbose=False):
    """
    Coordinate descent that specifically targets the largest auto-convolution peaks.
    At each step, identifies the top peak and reduces the coordinates contributing most to it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False
        
        # Find top 3 peaks
        top_k = min(3, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            
            # Find the coordinate with the largest contribution
            # We want to reduce a[j] where a[j]*a[k-j] is large
            j_max = idx[np.argmax(products)]
            
            # Determine which side to reduce (the larger of a[j] and a[k-j])
            j1 = int(j_max)
            j2 = k - j1
            if j1 == j2:
                candidates = [j1]
            else:
                candidates = [j1, j2] if 0 <= j2 < n else [j1]
            
            best_trial = None
            best_trial_obj = M
            
            for j_red in candidates:
                old_val = a[j_red]
                if old_val < 1e-12:
                    continue
                
                # Try different reduction factors
                for frac in [0.5, 0.7, 0.8, 0.9, 1.2, 1.5]:
                    new_val = old_val * frac
                    if new_val < 0 or new_val > 0.5:
                        continue
                    
                    trial = a.copy()
                    trial[j_red] = new_val
                    diff = old_val - new_val
                    
                    if abs(diff) > 1e-15:
                        # Redistribute to coordinates NOT contributing to top peaks
                        # Build mask of "safe" coordinates
                        safe = np.ones(n, dtype=bool)
                        for pk in peak_indices:
                            pk = int(pk)
                            ps = max(0, pk - n + 1)
                            pe = min(n - 1, pk)
                            safe[ps:pe+1] = False
                        safe[j_red] = False
                        
                        if safe.sum() > 0:
                            sw = a[safe].sum()
                            if sw > 1e-15:
                                trial[safe] += diff * a[safe] / sw
                            else:
                                trial += diff / n
                        else:
                            trial += diff / n
                    
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts
                    
                    obj = fast_convolve(trial).max()
                    if obj < best_trial_obj - 1e-14:
                        best_trial_obj = obj
                        best_trial = trial
            
            if best_trial is not None:
                a = best_trial
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    improved = True
                    if verbose and it % 50 == 0:
                        print(f"    PCD iter {it}: obj={best_obj:.10f}")
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try reducing all contributors to the main peak simultaneously
                k_main = int(np.argmax(fast_convolve(a)))
                js = max(0, k_main - n + 1)
                je = min(n - 1, k_main)
                idx = np.arange(js, je + 1)
                
                trial = a.copy()
                trial[idx] *= 0.9
                diff = a.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je+1] = False
                if non_contrib.sum() > 0:
                    nc_sum = a[non_contrib].sum()
                    if nc_sum > 1e-15:
                        trial[non_contrib] += diff * a[non_contrib] / nc_sum
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    a = trial
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    # Perturb and restart
                    rng = np.random.default_rng(it * 43 + 11)
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.003)

        else:
            no_improve = 0

    return best_a, best_obj


def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.005, decay=0.9995):
    """Mirror descent with multiplicative weight updates."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.001
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        k_actual = min(15, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        lr *= decay
        temperature *= 1.003
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.6
            no_improve = 0

            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Create a bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = (1.0 - t ** 4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_bump(n, separation=0.3):

    a = np.zeros(n)
    center1 = n * (0.5 - separation / 2)
    center2 = n * (0.5 + separation / 2)
    sigma = n * 0.1
    a += np.exp(-0.5 * ((np.arange(n) - center1) / sigma) ** 2)
    a += np.exp(-0.5 * ((np.arange(n) - center2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_curved(n, curvature=2.0):

    target_ns = [512, 640, 768, 896, 1024]
    
    init_best = parent
    init_best_score = best_score
    
    for n_try in target_ns:
        inits = [
            make_gaussian(n_try, 0.25),
            make_bell(n_try, 0.5),
            make_two_bump(n_try, 0.25),
            make_curved(n_try, 1.5),
            make_curved(n_try, 3.0),
            np.ones(n_try) / n_try,
        ]
        for a_init in inits:
            sc = try_seq(a_init, f"init_n{n_try}")
            if sc < init_best_score:
                init_best_score = sc
                init_best = a_init
    
    # Also try interpolating parent to different lengths
    for n_try in target_ns:
        if n_try != len(parent):
            a_interp = interpolate_sequence(parent, n_try)
            sc = try_seq(a_interp, f"interp_n{n_try}")
            if sc < init_best_score:
                init_best_score = sc
                init_best = a_interp
    
    print(f"  Best init score: {init_best_score:.6f}")
    
    # === Phase 2: Subgradient boosting on best init ===
    print("\n=== Phase 2: Subgradient Boosting ===")
    rem = deadline - time.monotonic() - 800
    if rem > 30:
        t_alloc = min(rem, 300)
        print(f"  SB on n={len(init_best)}: t={t_alloc:.0f}s")
        a_opt, obj = subgradient_boost(init_best, time_limit=t_alloc, verbose=True,
                                        n_peaks=5, step_init=0.02, decay=0.9992)
        try_seq(a_opt, "sb_init")
        
        # Also try with different parameters
        if best_seq is not None:
            t_alloc2 = min(rem - t_alloc, 150)
            if t_alloc2 > 20:
                a_opt2, obj2 = subgradient_boost(best_seq, time_limit=t_alloc2, verbose=False,
                                                  n_peaks=8, step_init=0.01, decay=0.9995)
                try_seq(a_opt2, "sb_8peak")
    
    # === Phase 3: LP-FW on current best ===
    print("\n=== Phase 3: LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  LP-FW n={len(best_seq)}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")
    
    # === Phase 4: Peak-targeted CD ===
    print("\n=== Phase 4: Peak-Targeted CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_alloc = min(rem, 150)
            print(f"  PCD n={len(best_seq)}: t={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_cd(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "pcd")
    
    # === Phase 5: Multiplicative descent ===
    print("\n=== Phase 5: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  MD n={len(best_seq)}: t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True,
                                                 lr_init=0.004, decay=0.9996)
            try_seq(a_opt, "md")
    
    # === Phase 6: Try longer/shorter sequences ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    base = best_seq if best_seq is not None else init_best
    for target_n in [1024, 1280, 640, 512]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        sc = try_seq(f_interp, f"expl_n{target_n}")
        
        # Run subgradient boosting on this length
        t_alloc = min(rem * 0.4, 100)
        if t_alloc > 20:
            a_opt, obj = subgradient_boost(f_interp, time_limit=t_alloc, verbose=False,
                                            n_peaks=5, step_init=0.015, decay=0.9993)
            try_seq(a_opt, f"sb_n{target_n}")
    
    # === Phase 7: Iterated refinement on best ===
    print("\n=== Phase 7: Iterated Refinement ===")
    if best_seq is not None:
        for round_idx in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            t_each = min(rem / (3 - round_idx), 80)
            
            # Subgradient boost
            a_opt, obj = subgradient_boost(best_seq, time_limit=t_each * 0.4, verbose=False,
                                            n_peaks=4 + round_idx, step_init=0.008, decay=0.9995)
            try_seq(a_opt, f"refine_sb_{round_idx}")
            
            # Peak CD
            rem2 = deadline - time.monotonic() - 60
            if rem2 > 20:
                t2 = min(rem2 * 0.5, 50)
                a_opt2, obj2 = peak_targeted_cd(best_seq, time_limit=t2, verbose=False)
                try_seq(a_opt2, f"refine_pcd_{round_idx}")
    
    # === Phase 8: Diverse restarts ===
    print("\n=== Phase 8: Diverse Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 40
            if rem < 20:
                break
            t_alloc = min(rem / (4 - p), 60)
            
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Blend with a structured perturbation
                structured = make_gaussian(n_best, 0.2)
                f_init = 0.85 * best_seq + 0.15 * structured
                f_init = f_init / f_init.sum()
            
            a_opt, obj = subgradient_boost(f_init, time_limit=t_alloc, verbose=False,
                                            n_peaks=5, step_init=0.01, decay=0.9994)
            try_seq(a_opt, f"restart_{p}")
    
    # === Phase 9: Final polishing ===
    print("\n=== Phase 9: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 50)
            a_opt, obj = peak_targeted_cd(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "polish_pcd")
        
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 30)
            a_opt, obj = subgradient_boost(best_seq, time_limit=t_alloc, verbose=False,
                                            n_peaks=3, step_init=0.005, decay=0.9996)
            try_seq(a_opt, "polish_sb")
        
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 15)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "polish_lpfw")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
