```python
from scipy.sparse import lil_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(n, f):
    """Build the matrix T such that T @ g = conv(f, g)."""
    T = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n-1, k)
        T[k, i_start:i_end+1] = f[i_start:i_end+1]
    return T.tocsr()

    n = len(f)
    T = build_conv_matrix(n, f)

    return None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
(minimax cross-correlation), line search blending, multi-length exploration,
and smooth L-BFGS polishing. The key innovation is using a linear program at
each iteration to find the best response direction (minimizing the max
cross-correlation with the current sequence), which provides a principled
descent direction for the non-convex auto-convolution peak objective.
"""

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix T such that T @ g = conv(f, g) for any g of same length."""
    n = len(f)
    T = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        T[k, i_start:i_end + 1] = f[i_start:i_end + 1]
    return T.tocsr()


def lp_minimax(f, time_limit=10.0):
    """
    Solve: min t s.t. conv(f,g)[k] <= t for all k, sum(g)=1, g >= 0.
    Returns (g, t_opt).
    """
    n = len(f)
    T = build_conv_matrix(f)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # A_ub @ x <= b_ub: T @ g - t <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T.toarray()
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq @ x == b_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        t_opt = res.x[n]
        return g, t_opt
    return None, None


def line_search_blend(f, g, max_iter=50):
    """
    Find optimal t in [0,1] minimizing max(conv((1-t)f+tg, (1-t)f+tg)).
    Returns (best_t, best_value).
    """
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    ff = irfft(F * F, fft_size)[:size]
    fg = irfft(F * G, fft_size)[:size]
    gg = irfft(G * G, fft_size)[:size]
    
    def obj(t):
        val = (1 - t) ** 2 * ff + 2 * t * (1 - t) * fg + t ** 2 * gg
        return val.max()
    
    # Coarse grid search first
    ts = np.linspace(0, 1, 21)
    vals = [obj(t) for t in ts]
    best_idx = np.argmin(vals)
    
    # Fine search around best
    if best_idx == 0:
        bracket = (0, ts[1])
    elif best_idx == len(ts) - 1:
        bracket = (ts[-2], 1.0)
    else:
        bracket = (ts[best_idx - 1], ts[best_idx + 1])
    
    res = minimize_scalar(obj, bounds=bracket, method='bounded',
                          options={'maxiter': max_iter})
    return res.x, res.fun


def lp_iteration(f0, max_iters=200, time_limit=120.0, verbose=True):
    """
    LP-based iterative refinement with line search.
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    if verbose:
        print(f"  LP iter start: n={n}, obj={best_obj:.8f}")
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        g, t_opt = lp_minimax(f)
        if g is None:
            if verbose:
                print(f"  LP failed at iter {it}")
            break
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Line search
        t_best, val_best = line_search_blend(f, g)
        
        if val_best < best_obj - 1e-12:
            f_new = (1 - t_best) * f + t_best * g
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs < 1e-15:
                break
            f_new = f_new / fs
            
            # Verify
            actual_obj = fast_convolve(f_new).max()
            if actual_obj < best_obj - 1e-12:

                best_obj = actual_obj
                best_f = f.copy()

            else:
                no_improve += 1
        else:

            break

    return best_f, best_obj


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a_i = 2n * sum_k w_k * a_{k-i} (cross correlation)
    size = len(b)
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr: result[i] = sum_k w[k] * a[k-i]
    # This is the correlation, not convolution
    # corr[i] = sum_k w[k] * a[k-i]
    # In FFT: ifft(conj(W) * A) gives correlation
    grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_limit, maxiter=2000):
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxfun': maxiter * 20,
                                'disp': False})
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```
