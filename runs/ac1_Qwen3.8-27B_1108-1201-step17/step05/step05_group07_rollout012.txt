"""
LP-based alternating method to minimize max(auto-convolution) subject to
fixed sum and non-negativity. Uses the key insight that fixing one sequence
makes the cross-convolution constraint linear, enabling efficient LP solves.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def solve_lp(a, M):
    """
    Solve: maximize sum(g) subject to (a*g)_k <= M for all k, g >= 0.
    Returns the optimal g and sum(g).
    """
    n = len(a)
    # Build constraint matrix
    # (a*g)_k = sum_j a[k-j] * g[j] for valid indices
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # For efficiency, build the dense matrix
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    # A[k, j] = a[k-j] if max(0, k-n+1) <= j <= min(n-1, k), else 0
    
    n_k = 2 * n - 1
    # Dense matrix approach for n=768 (1535 x 768 = ~1.1M entries, manageable)
    A_ub = np.zeros((n_k, n))
    for k in range(n_k):
        js_start = max(0, k - n + 1)
        js_end = min(n - 1, k)
        for j in range(js_start, js_end + 1):
            A_ub[k, j] = a[k - j]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(n_k, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, 0.0
    
    return res.x, res.x.sum()

LP-based alternating method to minimize 2*n*max(conv(a,a))/(sum(a))^2.
The key insight: fixing one sequence makes the cross-convolution constraint
linear, enabling efficient LP solves. We alternate between the current
sequence and the LP-optimal response, with line search for safe steps.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_constraint_matrix(a):
    """Build the constraint matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    n_k = 2 * n - 1
    k_idx = np.arange(n_k)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((n_k, n))
    mask = (diff >= 0) & (diff <= n - 1)
    A[mask] = a[diff[mask]]
    return A


def solve_lp_response(a, M):
    """
    Solve LP: maximize sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, sum_g) or (None, 0) if infeasible.
    """
    n = len(a)
    A_ub = build_constraint_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x, res.x.sum()
        elif res.status == 2:  # Infeasible
            return None, 0.0
        else:
            # Try with more iterations
            res2 = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'max_iter': 10000})
            if res2.status == 0:
                return res2.x, res2.x.sum()
            return None, 0.0
    except Exception as e:
        print(f"    LP error: {e}")
        return None, 0.0

def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, x>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_polish(a0, time_limit=60.0, lr=0.01, mom=0.9, top_k=50):
    """Subgradient method with top-k peak weighting."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, fast_convolve(a).max()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        
        co = float(fast_convolve(a).max())
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.998
        cur_temp *= 1.005
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit=60.0):

    a = a0.copy().astype(np.float64)
    s = a.sum()

    n = len(a)

        # Process elements in random order

        np.random.shuffle(order)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try a range of multipliers
            for mult in [0.3, 0.5, 0.7, 0.8, 1.2, 1.4, 1.7, 2.0]:

                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = float(fast_convolve(a_trial).max())

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

One optimization: instead of solving the full LP each time, we could warm-start from the previous solution. But scipy's linprog doesn't support warm-starting directly.

Another idea: since the constraint matrix changes slowly (a changes slowly), we could use an approximate constraint matrix from the previous iteration and add a constraint that ensures the new solution satisfies the exact constraints. But this is complex.

Let me just go with the straightforward approach and see if it's fast enough.

Let me finalize the code:

"""
LP-based alternating method for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the key insight that fixing one sequence makes the cross-convolution
constraint linear, enabling efficient LP solves. Combined with subgradient
polishing, coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_constraint_matrix(a):
    """Build matrix A where (A@g)[k] = (a*g)[k] for full convolution."""
    n = len(a)
    n_k = 2 * n - 1
    k_idx = np.arange(n_k)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((n_k, n))
    mask = (diff >= 0) & (diff <= n - 1)
    A[mask] = a[diff[mask]]
    return A


def solve_lp_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A_ub = build_constraint_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x, res.x.sum()
    except Exception:
        pass
    return None, 0.0


def lp_alternating(a_init, time_limit=120.0, verbose=False):
    """LP-based alternating minimization with line search."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
        s = a.sum()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum / s
    
    M = float(fast_convolve(a).max())
    if verbose:
        print(f"  LP-Alt start: n={n}, M={M:.6f}")
    
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    no_improve = 0
    
    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, g_sum = solve_lp_response(a, M)
        if g is None:
            break
        
        if g_sum <= target_sum * (1.0 + 1e-7):
            if verbose:
                print(f"  LP-Alt converged at iter {iteration}, M={M:.6f}")
            break
        
        g_scaled = g * (target_sum / g_sum)
        
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.075, 0.03]:
            a_trial = (1.0 - t) * a + t * g_scaled
            M_trial = float(fast_convolve(a_trial).max())
            if M_trial <= M * (1.0 + 1e-10):
                a = a_trial
                M = M_trial
                improved = True
                break
        
        if not improved:
            # Try a very tiny step
            t = 0.01
            a = (1.0 - t) * a + t * g_scaled
            M_new = float(fast_convolve(a).max())
            if M_new < M:
                M = M_new
            else:
                a = a / a.sum() * target_sum  # revert
                no_improve += 1
        
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            break
        
        if verbose and iteration % 20 == 0:
            print(f"    iter {iteration}: M={M:.6f}, ratio={g_sum/target_sum:.4f}")
    
    if verbose:
        print(f"  LP-Alt final: M={best_M:.6f} (n={n})")
    
    return best_a, best_M


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_polish(a0, time_limit=60.0, lr=0.01, mom=0.9, top_k=50):
    """Subgradient method with softmax-weighted peaks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, 0.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        
        co = float(fast_convolve(a).max())
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.003
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit=60.0):
    """Cycle through coordinates, trying to improve each."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, 0.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = float(fast_convolve(a_trial).max())
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
