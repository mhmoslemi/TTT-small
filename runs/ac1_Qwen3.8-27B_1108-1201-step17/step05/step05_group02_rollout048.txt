"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP Frank-Wolfe
with adaptive constraint selection, warm starts, multi-length exploration,
and iterative peeling for faster convergence.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, active_lags):
    """Build the constraint matrix for only active lags.
    A_ub[i, j] = f[j] contributes to lag (active_lags[i] - j)
    Actually, (f*g)(k) = sum_j f[j] * g[k-j] for valid j.
    So for a given lag k, the coefficient of g[j] is f[k-j] if 0 <= k-j < n.
    """
    n = len(f)
    K = len(active_lags)
    A = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        # (f*g)(k) = sum_{j} f[j] * g[k-j] where 0 <= j < n and 0 <= k-j < n
        # So g[j] appears with coefficient f[k-j] when 0 <= k-j < n
        # i.e., k-n+1 <= j <= k
        js_start = max(0, k - n + 1)
        js_end = min(n - 1, k)
        for j in range(js_start, js_end + 1):
            A[i, j] = f[k - j]
    return A


def solve_active_lp(f, M, active_lags, n):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in active_lags, g >= 0."""
    c = -np.ones(n)
    A_ub = build_conv_matrix(f, active_lags)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0, None)] * n
    options = {'time_limit': 5, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

def subgradient_step(f, top_k=50, lr=0.01, temp=0.001):
    """One subgradient step on the max-convolution objective."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Select peaks
    actual_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -actual_k)[-actual_k:]
    
    # Softmax weights
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Subgradient of max at the peaks
    # d/d f[j] of (f*f)(k) = sum_i f[k-i] * delta(i=j) + sum_i f[i] * delta(k-i=j)
    # = f[k-j] + f[j-k] (with appropriate boundary handling)
    # More carefully: (f*f)(k) = sum_{i=0}^{n-1} f[i] * f[k-i] (where 0<=k-i<n)
    # d/d f[j] = f[k-j] (if 0<=k-j<n) + f[j-k] (if 0<=j-k<n)
    # But since we're looking at lag k >= 0 in the first half, and the auto-conv is symmetric,
    # let's just compute: for lag k, the gradient wrt f[j] is f[k-j] if 0<=k-j<n
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js_start = max(0, k - n + 1)
        js_end = min(n - 1, k)
        idx = np.arange(js_start, js_end + 1)
        sg[idx] += w[i] * f[k - idx]
    
    # Also account for the symmetric part (lag -k = 2n-1-k)
    # Since conv is symmetric, conv[k] = conv[2n-1-k]
    # The gradient contribution from the symmetric lag is the same
    # So we just double the gradient (but only for k not in the middle)
    # Actually, since we're already summing over all selected peaks (which includes
    # both k and 2n-1-k if both are in the top_k), we're fine.
    
    f_new = f - lr * sg
    f_new = project_simplex(f_new)
    return f_new

        cur_temp *= 1.01

            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

def coordinate_descent(f0, time_limit):
    """Simple coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_max = fast_convolve(f).max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            best_j_val = old_val
            best_j_obj = cur_max

            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                cur_max = best_j_obj
                if cur_max < best_obj - 1e-14:
                    best_obj = cur_max
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP Frank-Wolfe
with adaptive constraint selection, multi-length exploration, diverse
restarts, and iterative refinement for faster convergence.
Key improvement over previous: sparse active-set LP (top K lags only)
enables 5-10x more iterations in the same time budget.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, active_lags, n):
    """Build LP constraint matrix A where A[i,j] = f[k_i - j] if valid."""
    K = len(active_lags)
    A = np.zeros((K, n))
    for i in range(K):
        k = active_lags[i]
        js_start = max(0, k - n + 1)
        js_end = min(n - 1, k)
        if js_start <= js_end:
            A[i, js_start:js_end + 1] = f[k - (js_start:js_end + 1)]
    return A


def solve_lp_direction(f, M, active_lags, n, time_limit_lp=5):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in active_lags, g >= 0."""
    c = -np.ones(n)
    A_ub = build_constraint_matrix(f, active_lags, n)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0, None)] * n
    options = {'time_limit': time_limit_lp, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def active_lp_fw(f0, time_limit, top_k_init=25, top_k_max=120, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    top_k = top_k_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()

        actual_k = min(top_k, 2 * n - 1)
        active_lags = np.argpartition(conv, -actual_k)[-actual_k:]

        g = solve_lp_direction(f, M * 1.001, active_lags, n, time_limit_lp=4)
        if g is None:

def subgradient_polish(f0, time_limit, lr=0.004, top_k=40, temp=0.0005):
    """Subgradient polishing with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        actual_k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -actual_k)[-actual_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * f[k - idx]

        f_new = f - cur_lr * sg
        f = project_simplex(f_new)
        co = fast_convolve(f).max()

        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.999
        cur_temp *= 1.008

        if no_improve > 150:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            cur_lr = max(lr * 0.4, 0.0003)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_max = fast_convolve(f).max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            best_j_val = old_val
            best_j_obj = cur_max

            for mult in [0.5, 0.7, 0.85, 1.15, 1.4, 1.8]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                cur_max = best_j_obj
                if cur_max < best_obj - 1e-14:
                    best_obj = cur_max
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization using active-set LP Frank-Wolfe with:
    - Adaptive constraint selection (top K lags)
    - Multi-length exploration
    - Diverse initializations
    - Subgradient and coordinate descent polishing
    - Micro-perturbation search
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Active LP-FW on parent ===
    print(f"\n=== Phase 1: Active LP-FW n={n_parent} ===")
    rem = deadline - time.monotonic()
    t1 = min(450, rem * 0.3)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_opt, obj = active_lp_fw(parent, time_limit=t1,
                                   top_k_init=25, top_k_max=100, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")
        else:
            print(f"  No improvement (internal obj={obj:.10f}, score={sc:.6f})")

    # === Phase 2: Try nearby lengths ===
    print(f"\n=== Phase 2: Length search (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    candidate_lengths = []
    for delta in [48, -48, 96, -96, 128, -128, 192, -192, 256, -256, 384, -384]:
        nl = n_best + delta
        if 256 <= nl <= 2048 and nl != n_best:
            candidate_lengths.append(nl)

    for nl in [384, 512, 640, 896, 1024, 1280]:
        if 256 <= nl <= 2048 and nl != n_best and nl not in candidate_lengths:
            candidate_lengths.append(nl)

    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(70, rem * 0.25)
        if t_alloc < 20:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = active_lp_fw(f_interp, time_limit=t_alloc,
                                  top_k_init=20, top_k_max=80)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={target_n})")

    # === Phase 3: Deep optimization with diverse inits ===
    print(f"\n=== Phase 3: Diverse inits (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    inits = []
    inits.append(("direct", best_seq.copy()))

    noise = rng.standard_normal(n_best) * 0.003
    inits.append(("noise3k", project_simplex(best_seq + noise)))

    noise = rng.standard_normal(n_best) * 0.01
    inits.append(("noise10k", project_simplex(best_seq + noise)))

    f_sym = 0.5 * best_seq + 0.5 * best_seq[::-1]
    inits.append(("symmetric", project_simplex(f_sym)))

    f_mix = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
    inits.append(("mix_uniform", project_simplex(f_mix)))

    # Edge-concentrated
    f_edge = best_seq.copy()
    edge_w = n_best // 8
    f_edge[:edge_w] *= 1.4
    f_edge[-edge_w:] *= 1.4
    inits.append(("edge_boost", project_simplex(f_edge)))

    # Center-concentrated
    f_center = best_seq.copy()
    cs = n_best // 4
    ce = 3 * n_best // 4
    f_center[cs:ce] *= 1.3
    inits.append(("center_boost", project_simplex(f_center)))

    # Sinusoidal modulation
    x = np.linspace(0, 2 * np.pi, n_best)
    f_sine = best_seq * (1.0 + 0.1 * np.sin(x))
    inits.append(("sine_mod", project_simplex(f_sine)))

    for idx, (label, f_init) in enumerate(inits):
        rem = deadline - time.monotonic() - 180
        if rem < 20:
            break
        t_alloc = min(100, rem * 0.2)
        if t_alloc < 15:
            continue
        print(f"  [{label}] t={t_alloc:.0f}s")
        a_opt, _ = active_lp_fw(f_init, time_limit=t_alloc,
                                  top_k_init=20, top_k_max=80)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f}")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: SG Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_sg = min(rem * 0.5, 60)
        a_opt, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                       lr=0.003, top_k=max(30, len(best_seq) // 5),
                                       temp=0.0003)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 50
    if rem > 30:
        t_cd = min(rem * 0.6, 60)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 6: Final LP-FW with higher K ===
    print(f"\n=== Phase 6: Final LP-FW squeeze (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_sq = min(rem * 0.7, 80)
        a_opt, _ = active_lp_fw(best_seq, time_limit=t_sq,
                                  top_k_init=50, top_k_max=200)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 7: Micro perturbations ===
    print(f"\n=== Phase 7: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(12):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0002 + trial * 0.0001
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = active_lp_fw(f_pert, time_limit=min(rem2 * 0.4, 12),
                                          top_k_init=30, top_k_max=80)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_lp]: {sc2:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
