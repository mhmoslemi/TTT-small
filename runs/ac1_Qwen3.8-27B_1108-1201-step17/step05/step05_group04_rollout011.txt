```python
import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def lse_objective_grad(x, t, n):
    """
    Log-sum-exp smoothed objective and gradient in softmax parameterization.
    Objective: 2n * (1/t) * log(sum_k exp(t * c_k)) where c = conv(f, f)
    Gradient: computed analytically.
    """
    f = softmax(x)
    c = fast_convolve(f)
    
    # Log-sum-exp of t*c
    tc = t * c
    tc_max = tc.max()
    lse_val = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    obj = 2.0 * n * lse_val / t
    
    # Weights (softmax of t*c)
    w = np.exp(tc - tc_max)
    w /= w.sum()
    
    # Gradient of obj w.r.t. f: 4n * correlate(w, f, 'valid')
    corr = np.correlate(w, f, 'valid')
    g = 4.0 * n * corr
    
    # Gradient w.r.t. x (softmax chain rule)
    g_dot_f = np.dot(g, f)
    grad_x = f * (g - g_dot_f)
    
    return obj, grad_x

def lbfgs_optimize(f0, t, maxiter=500, time_limit=60.0, ftol=1e-12, gtol=1e-8):
    """Run L-BFGS on the log-sum-exp smoothed objective."""
    n = len(f0)
    x0 = np.log(f0 + 1e-30)  # Inverse softmax (log of f)
    
    t0 = time.monotonic()
    
    def func_grad(x):
        if time.monotonic() - t0 > time_limit:
            # Return a gradient pointing to zero to signal stop
            return 0.0, np.zeros_like(x)
        return lse_objective_grad(x, t, n)
    
    result = minimize(func_grad, x0, jac=True, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol,
                               'maxfun': 10000})
    
    f_opt = softmax(result.x)
    return f_opt, result

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coordinate_descent(f0, time_limit, n_iter_max=100):
    """Coordinate descent with exact line search on the simplex."""
    f = f0.copy()
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            # Try to optimally rescale f[j] (with others scaled proportionally)
            best_j_val = f[j]
            best_j_obj = best_obj
            
            # Golden section or grid search for optimal f[j]
            # f[j] ranges from 0 to 1, others scale as (1-f[j])/(1-f[j]_old)
            old_val = f[j]
            for trial_frac in [0.3, 0.5, 0.7, 0.8, 1.1, 1.3, 1.5, 2.0]:
                new_val = old_val * trial_frac
                if new_val <= 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / (1.0 - old_val) if old_val < 1.0 else 1.0
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.arange(n) != j
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / (1.0 - old_val) if old_val < 1.0 else 1.0
                f[j] = best_j_val
                mask = np.arange(n) != j
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj
```

```python
"""

dual active-set ideas, diverse multi-start, length exploration, and coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_obj_grad(x, t, n):
    """
    Log-sum-exp smoothed objective and gradient in softmax parameterization.
    f = softmax(x), objective = 2n * (1/t) * log(sum_k exp(t*c_k))
    """
    f = softmax_fn(x)
    c = fast_convolve(f)

    tc = t * c
    tc_max = tc.max()
    lse_val = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    obj = 2.0 * n * lse_val / t

    # Softmax weights over convolution values
    w = np.exp(tc - tc_max)
    w /= w.sum()

    # Gradient of obj w.r.t. f: 4n * correlate(w, f, 'valid')
    corr = np.correlate(w, f, 'valid')
    g_f = 4.0 * n * corr

    # Chain rule through softmax: grad_x = f * (g_f - <g_f, f>)
    g_dot = np.dot(g_f, f)
    grad_x = f * (g_f - g_dot)

    return obj, grad_x


def lbfgs_optimize(f0, t, time_limit=60.0, maxiter=300, ftol=1e-14, gtol=1e-10):
    """Run L-BFGS on the log-sum-exp smoothed objective with softmax parameterization."""
    n = len(f0)
    x0 = np.log(np.maximum(f0, 1e-30))

    t0 = time.monotonic()

    def func_and_grad(x):
        if time.monotonic() - t0 > time_limit:
            return 0.0, np.zeros_like(x)
        return lse_obj_grad(x, t, n)

    try:
        result = minimize(func_and_grad, x0, jac=True, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol,
                                   'maxfun': 5000, 'maxls': 50})
    except Exception:
        return f0.copy(), result if 'result' in dir() else None

    f_opt = softmax_fn(result.x)
    f_opt = np.maximum(f_opt, 0.0)
    s = f_opt.sum()
    if s > 1e-15:
        f_opt /= s
    return f_opt, result


def gradient_flow(f0, time_limit=60.0, lr=0.01, lr_decay=0.999,
                  temp=5.0, temp_decay=0.9995, n_restarts=5):
    """Simple gradient flow with adaptive step size and periodic restarts."""
    n = len(f0)
    f = f0.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    cur_lr = lr
    cur_temp = temp
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        tc = cur_temp * c
        tc_max = tc.max()
        w = np.exp(tc - tc_max)
        w /= w.sum()

        sg = 2.0 * np.correlate(w, f, 'valid')
        sg_centered = sg - np.dot(sg, f)

        f_new = f - cur_lr * sg_centered
        f = project_simplex(f_new)

        cur_temp = max(cur_temp * temp_decay, 1.0)
        cur_lr = max(lr * (lr_decay ** it), lr * 0.01)

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            noise = 0.005 + 0.003 * min(no_improve // 200, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            cur_temp = max(temp * 0.7, 2.0)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_multipliers=6):
    """Coordinate descent with multiplicative rescaling of one coordinate at a time."""
    f = f0.copy()
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    passes = 0

    multipliers = [0.4, 0.6, 0.8, 1.25, 1.5, 2.0]
    if n_multipliers < len(multipliers):
        multipliers = multipliers[:n_multipliers]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                improved = True
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()

        if not improved and passes > 1:
            break

    return best_f, best_obj


def dual_initialize(n, n_iter=50):
    """
    Initialize using the dual (minimax) perspective.
    Alternates between:
    - Given w (distribution over conv positions), find f minimizing f^T W_w f on simplex
    - Given f, update w to concentrate on high-conv positions
    """
    f = np.ones(n) / n
    w = np.ones(2 * n - 1) / (2 * n - 1)

    for it in range(n_iter):
        # Solve min f^T W_w f s.t. f>=0, sum(f)=1 using projected gradient
        # W_w[j,m] = w[j+m]
        # Gradient: 2 * W_w @ f, where (W_w @ f)[j] = sum_m w[j+m]*f[m]
        f_proj = f.copy()
        for inner in range(20):
            grad = 2.0 * np.array([np.dot(w[j:j + n], f_proj) for j in range(n)])
            step = 2.0 / (it + inner + 3)
            f_new = f_proj - step * grad
            f_proj = project_simplex(f_new)
        f = f_proj

        # Update w
        c = fast_convolve(f)
        # Soft update with temperature
        t_w = 5.0 + 5.0 * it
        w_new = np.exp(t_w * (c - c.max()))
        w_new /= w_new.sum()
        w = 0.5 * w + 0.5 * w_new

    return f


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)

def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2

        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'flat':
        f = np.ones(n)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization as a sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_random_phase_init(n, rng):
    """
    Create an initialization inspired by random-phase sequences.
    In frequency domain, use random phases (restricted to real signals via conjugate symmetry).
    """
    fft_size = 1 << (2 * n - 2).bit_length()
    # Create random Fourier coefficients with conjugate symmetry
    freq = np.zeros(fft_size // 2 + 1)
    freq[0] = 1.0
    for k in range(1, fft_size // 2 + 1):
        mag = rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        freq[k] = mag * np.cos(phase) + 1j * mag * np.sin(phase)
    # Ensure conjugate symmetry for real signal
    spectrum = np.zeros(fft_size, dtype=complex)
    spectrum[:fft_size // 2 + 1] = freq
    for k in range(1, fft_size // 2):
        spectrum[fft_size - k] = np.conj(freq[k])

    signal = np.fft.irfft(spectrum)
    # Take first n samples, make non-negative
    f = signal[:n]
    f = np.maximum(f, 0.0)
    if f.sum() < 1e-10:
        f = np.ones(n) / n
    else:
        f /= f.sum()
    return f


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        # L-BFGS at new length
        t_temp = target_n
        a_opt, _ = lbfgs_optimize(f_interp, t=t_temp, time_limit=t_alloc, maxiter=400)
        try_seq(a_opt, f"p4_n{target_n}")

        # Quick gradient flow
        rem = t_phase4_end - time.monotonic()
        if rem > 15:
            a_opt2, _ = gradient_flow(f_interp, time_limit=min(rem * 0.3, 25),
                                      lr=0.02, temp=target_n, temp_decay=0.999)
            try_seq(a_opt2, f"p4_gf_n{target_n}")

    # === Phase 5: Coordinate descent polishing (150s) ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    t_phase5_end = time.monotonic() + 150
    if best_seq is not None:
        rem = t_phase5_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.7, 80)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p5_cd")

            # CD from perturbed start
            rem = t_phase5_end - time.monotonic()
            if rem > 15:
                n_b = len(best_seq)
                perturbed = project_simplex(best_seq + rng.standard_normal(n_b) * 0.003)
                a_opt2, _ = coordinate_descent(perturbed, time_limit=min(rem * 0.6, 40))
                try_seq(a_opt2, "p5_cd_pert")

    # === Phase 6: Final fine-tuning with very low temperature L-BFGS (100s) ===
    print(f"\n=== Phase 6: Final fine-tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            # Very high temperature (sharp approximation to max)
            t_sharp = 10 * n_best
            t_alloc = min(rem * 0.5, 50)
            print(f"  Sharp L-BFGS (t={t_sharp}), t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_optimize(best_seq, t=t_sharp, time_limit=t_alloc,
                                      maxiter=500, ftol=1e-16, gtol=1e-14)
            try_seq(a_opt, "p6_sharp")

            # Fine gradient flow
            rem = deadline - time.monotonic() - 8
            if rem > 15:
                t_alloc2 = min(rem * 0.5, 30)
                print(f"  Fine GF, t={t_alloc2:.0f}s")
                a_opt2, _ = gradient_flow(best_seq, time_limit=t_alloc2,
                                          lr=0.003, temp=5 * n_best, temp_decay=0.9998)
                try_seq(a_opt2, "p6_gf_fine")

            # Tiny perturbations
            rem = deadline - time.monotonic() - 5
            for pi in range(2):
                rem = deadline - time.monotonic() - 5
                if rem < 8:
                    break
                noise = rng.standard_normal(n_best) * 0.002
                f_pert = project_simplex(best_seq + noise)
                a_opt3, _ = lbfgs_optimize(f_pert, t=n_best, time_limit=min(rem * 0.4, 20),
                                           maxiter=200)
                try_seq(a_opt3, f"p6_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
