def lp_improve(a, M):
    n = len(a)

    constraints = [cvx.sum(g) >= 0]  # placeholder
    
    # (a * g)[k] <= M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_i a[i] * g[k-i] where 0 <= i < n and 0 <= k-i < n

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(w, a):
    """Cross-correlation: result[i] = sum_k w[k] * a[k-i] for valid indices.
    Actually: (w * a)[i] = sum_k w[k] * a[i-k]
    """
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix A where b = A @ g gives b = convolve(a, g).
    A has shape (2n-1, n). A[k, j] = a[k-j] if 0 <= k-j < n.
    """
    n = len(a)
    # Use scipy's toeplitz-like construction
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        end = j + n
        A[start:end, j] = a
    return A


def solve_lp(a, M):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(2*n-1)]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, abs_tol=1e-10, rel_tol=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            return np.ones(n) / n
    if g.value is None:
        return np.ones(n) / n
    return np.maximum(g.value, 0)

                if val < best_j_obj - 1e-14:
                    best_j_obj = val

def subgradient_method(a_init, time_limit=60.0, lr=0.005, momentum=0.9, top_k=50):
    """Subgradient method with softmax-weighted active peaks."""
    t0 = time.monotonic()
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = momentum * v - cur_lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        a = a_new
        
        b = fast_convolve(a)
        co = b.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.998
        cur_temp *= 1.02
        
        if no_improve > 200:
            rng = np.random.default_rng(hash(str(time.monotonic())) % (2**31))
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            cur_lr = max(cur_lr, 0.001)
            no_improve = 0
    
    return best_a


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix A where b = A @ g gives b = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_subproblem(a, M):
    """Solve: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    rhs = M * np.ones(2 * n - 1)
    constraints = [A @ g <= rhs]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, abs_tol=1e-10, rel_tol=1e-10, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, eps=1e-9)
        except Exception:
            try:

            except Exception:
                return np.ones(n) / n
    if g.value is None:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        avg_power = power.sum() / len(power)
        # Soft flatten: mix current power with flat
        alpha = 0.1
        target_power = (1-alpha) * power + alpha * avg_power

        # This is hard directly. Instead, use amplitude matching:
        # Keep the phase of A, change the amplitude to match target_power^(1/2)
        phase = A / np.abs(A + 1e-30)
        new_amp = np.sqrt(target_power)
        A_new = new_amp * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        val = fast_convolve(a_new).max()
        if val < best_obj:
            best_obj = val
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_subproblem(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    rhs = M * np.ones(2 * n - 1)
    constraints = [A @ g <= rhs]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
