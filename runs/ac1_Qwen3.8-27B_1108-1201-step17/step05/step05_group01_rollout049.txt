import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.12
            if m_exp > 15.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

            no_improve += 1

                    m_exp = 1.0

            m_exp = min(m_exp * 1.06, 5.0)
            continue

        best_alpha = 0.0

            if st < 1e-15:

                if st < 1e-15:

                val = fast_convolve(ft).max()

                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:

            m_exp = 1.0

def _smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # grad_a[i] = 2 * sum_m a[m] * w[m+i]
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

        try:

                         'maxfun': max_iter * 10}

            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:

def peak_shaving(f0, time_limit, top_k=50, verbose=False):
    """
    Novel Peak Shaving algorithm: directly targets autocorrelation peaks
    by moving mass away from peak-contributing positions.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    eta = 0.01
    no_improve = 0
    temp = 0.001

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()

        # Find top peaks above threshold
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        # Sort by height
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        # Weight peaks by their height (exponential)
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()

        # Compute "attractor" gradient: how much each position contributes to peaks
        attractor = np.zeros(n)
        for i in range(len(peaks)):
            if w_peaks[i] < 1e-15:
                continue
            k = int(peaks[i])
            # attractor[j] += w[k] * f[k-j] for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_idx = np.arange(js, je + 1)
            partner_idx = k - j_idx
            attractor[j_idx] += w_peaks[i] * f[partner_idx] / conv[k]

        # Normalize attractor to have zero mean
        attractor_mean = np.dot(f, attractor)
        direction = attractor - attractor_mean

        # Multiplicative update in log space
        log_f = np.log(f + 1e-30)
        log_f -= eta * direction
        f_new = np.exp(log_f - log_f.max())
        f_new /= f_new.sum()
        f_new = np.maximum(f_new, 0.0)

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and best_obj < 1.51:
                print(f"    PeakShave: obj={best_obj:.10f}, eta={eta:.5f}")
        else:
            no_improve += 1
            eta *= 0.96
            temp *= 1.02

        eta = max(eta, 0.001)

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            eta = 0.008
            temp = 0.001
            no_improve = 0

    return best_f, best_obj


def block_coord_desc(f0, time_limit, block_size=16):
    """Block coordinate descent with exact 1D line search per block element."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            # Try multiple multipliers
            for mult in [0.6, 0.8, 0.9, 1.1, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:

                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-16:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj

            eta = 0.03
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(old_x, f, bc_type='natural')
        f_interp = cs(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_restart(base, rng, n):
    """Generate a structured perturbation of the base sequence."""
    method = rng.integers(0, 5)
    f = base.copy()

    if method == 0:
        # Cosine bump
        k = rng.integers(1, max(2, n // 4))
        center = rng.uniform(0, 1) * n
        width = rng.uniform(0.1, 0.5) * n
        bump = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        bump = bump - bump.mean()
        f = f + 0.01 * bump
    elif method == 1:
        # Circular shift
        shift = rng.integers(1, n // 2)
        f = np.roll(f, shift)
        f = f + rng.standard_normal(n) * 0.003
    elif method == 2:
        # Frequency modulation
        k = rng.integers(1, max(2, n // 3))
        phase = rng.uniform(0, 2 * np.pi)
        mod = np.cos(2 * np.pi * k * np.arange(n) / n + phase)
        f = f * (1 + 0.02 * mod)
    elif method == 3:
        # Window function perturbation
        w = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
        f = f * (1 + 0.03 * (w - w.mean()))
    else:
        # Random Gaussian
        f = f + rng.standard_normal(n) * 0.005

    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f

            parent = p / p.sum()
