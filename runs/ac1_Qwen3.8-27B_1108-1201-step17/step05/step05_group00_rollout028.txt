import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_shaping(a0, time_limit, alpha=0.5, iterations_per_chunk=100):

    Fourier-domain iterative shaping:
    Flat spectrum -> Phase retrieval -> Non-negativity projection
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute FFT
        F = np.fft.rfft(a)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Flatten the spectrum: |F|^2 -> |F|^2^alpha
        # This makes the autocorrelation flatter
        mag_sq = mag ** 2
        # Normalize to preserve energy
        mag_sq_mean = mag_sq.mean()
        if mag_sq_mean > 1e-15:
            mag_sq_normed = mag_sq / mag_sq_mean
            mag_new = np.sqrt(mag_sq_normed ** alpha)
        else:
            mag_new = mag.copy()
        
        # Reconstruct with modified magnitude, same phase
        F_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, 2 * n - 1)[:n]

        # alpha starts at 0.5 (aggressive flattening) and goes to 0.8 (conservative)
        if it % 500 == 0 and alpha < 0.7:

    return best_a, best_obj

def subgradient_method(a0, time_limit, verbose=False):
    """
    Subgradient method with Polyak step size and averaging.
    Directly uses the true subgradient (not smoothed).
    """
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    # Polyak: need an upper bound F* and lower bound on gap
    # We'll use a running upper bound and track the best
    F_upper = best_obj * 1.1  # rough upper bound
    
    t0 = time.monotonic()
    it = 0
    a_avg = a.copy()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find maximizing (or near-maximizing) lags
        top_k = min(5, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        
        # Subgradient: for the active lags, grad[k] = 2 * sum of a[lag - k]
        # Use uniform weights over top peaks for a "averaged" subgradient
        grad = np.zeros(n)
        for i in peak_indices:
            i = int(i)
            wi = 1.0 / top_k
            k_lo = max(0, i - n + 1)
            k_hi = min(n - 1, i)
            if k_hi >= k_lo:
                a_slice = a[i - k_hi : i - k_lo + 1][::-1]
                grad[k_lo:k_hi+1] += 2.0 * wi * a_slice
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_norm = grad / gnorm
        
        # Step size: decreasing or Polyak
        step = 0.01 / np.sqrt(it + 1)
        
        a_new = a - step * grad_norm
        a_new = project_simplex(a_new)
        
        # Running average
        a_avg = (it * a_avg + a_new) / (it + 1)
        
        conv_avg = fast_convolve(a_avg)
        obj_avg = conv_avg.max()
        
        if obj_avg < best_obj - 1e-14:
            best_obj = obj_avg
            best_a = a_avg.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SubGrad iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        if no_improve > 1000:
            break
    
    return best_a, best_obj


def coordinate_descent_fine(a0, time_limit, verbose=False):
    """
    Fine coordinate descent: for each coordinate, find the optimal value
    via ternary search on the 1D function.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_epoch_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_epoch_improve < 3:
        epoch_improved = False
        
        # Sort indices by importance (contribution to peak)
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        
        # Indices that contribute to the peak
        k_lo = max(0, peak_k - n + 1)
        k_hi = min(n - 1, peak_k)
        
        # Process indices in order of their product contribution
        idx_range = np.arange(k_lo, k_hi + 1)
        if len(idx_range) > 0:
            products = a[idx_range] * a[peak_k - idx_range]
            order = idx_range[np.argsort(-products)]
        else:
            order = np.arange(n)
        
        # Also include other indices sorted by value
        other = np.setdiff1d(np.arange(n), idx_range)
        other_order = other[np.argsort(-a[other])] if len(other) > 0 else np.array([], dtype=int)
        
        full_order = np.concatenate([order, other_order])
        
        for j in full_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            j = int(j)
            old_val = a[j]
            
            # Binary/ternary search for optimal value of a[j]
            # Fix all other coordinates, vary a[j]
            # Objective as function of a[j]: 2*n*max_k conv_k(a) (with sum=1 constraint)
            # When a[j] changes by delta, sum changes by delta, so we need to renormalize
            # Or: fix sum of others = 1 - a[j], scale others accordingly
            
            if old_val >= 1.0 - 1e-10:
                continue
            
            # Try several candidate values
            candidates = [old_val * m for m in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]]
            candidates = [c for c in candidates if 0 < c < 1.0]
            if old_val < 0.01:
                candidates = [old_val * 2, old_val * 5, 0.005, 0.01, 0.02]
                candidates = [c for c in candidates if 0 < c < 1.0]
            
            best_c_val = old_val
            best_c_obj = fast_convolve(a).max()
            
            for cv in candidates:
                a_trial = a.copy()
                a_trial[j] = cv
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial = a_trial / ss
                obj = fast_convolve(a_trial).max()
                if obj < best_c_obj - 1e-14:
                    best_c_obj = obj
                    best_c_val = cv
            
            if best_c_val != old_val:
                a[j] = best_c_val
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_c_obj < best_obj - 1e-14:
                    best_obj = best_c_obj
                    best_a = a.copy()
                    epoch_improved = True
        
        if epoch_improved:
            no_epoch_improve = 0
        else:
            no_epoch_improve += 1
    
    return best_a, best_obj


def multiplicative_weight_refined(a0, time_limit, lr_init=0.003, verbose=False):
    """
    Refined multiplicative weight update with adaptive peak selection
    and periodic randomization.
    """
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    it = 0
    no_improve = 0
    temp = 0.01  # temperature for peak selection
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        
        # Select peaks with softmax weighting
        top_k = min(50, len(conv))
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peak_idx]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for ii in range(len(peak_idx)):
            if w[ii] < 1e-10:
                continue
            i = int(peak_idx[ii])
            wi = w[ii]
            k_lo = max(0, i - n + 1)
            k_hi = min(n - 1, i)
            if k_hi >= k_lo:
                a_slice = a[i - k_hi : i - k_lo + 1][::-1]
                grad[k_lo:k_hi+1] += 2.0 * wi * a_slice
        
        # Multiplicative update
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Adapt learning rate
        lr *= 0.9998
        temp *= 1.003
        
        # Random restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-12)
            a = a / a.sum()
            lr = lr_init * 0.5
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_worsey(n):
    """Worsey-like: sinc^2 shape (flat top, rapid decay)."""
    x = np.arange(n) - n / 2.0
    x_norm = x / (n * 0.15)
    a = np.sinc(x_norm) ** 2  # numpy sinc = sin(pi*x)/(pi*x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel multi-phase approach:
1. Fourier-domain spectral shaping (Gerchberg-Saxton style) for initialization
2. Softmax-smoothed gradient descent with Nesterov momentum
3. Refined multiplicative weight mirror descent with adaptive peak selection
4. Multi-length exploration via interpolation
5. Fine coordinate descent for final polish
Key innovations: spectral domain shaping for diverse initializations,

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, w, top_M=30):
    """
    Compute subgradient of max_k (a*a)[k] using weighted peaks.
    w: weights for each lag (length 2n-1)
    Returns gradient vector of length n.
    """
    n = len(a)
    top_M = min(top_M, len(w))
    # Get indices with significant weight
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    if len(active) == 0:
        return np.zeros(n)
    # Sort by weight descending, take top
    if len(active) > top_M:
        active = active[np.argsort(-w[active])[:top_M]]
    
    grad = np.zeros(n)
    for i in active:
        i = int(i)
        wi = w[i]
        k_lo = max(0, i - n + 1)
        k_hi = min(n - 1, i)
        if k_hi >= k_lo:
            # grad[k] += 2*wi*a[i-k] for k in [k_lo, k_hi]
            # a[i-k] for k=k_lo..k_hi means a goes from a[i-k_lo] down to a[i-k_hi]
            a_slice = a[i - k_hi:i - k_lo + 1][::-1]
            grad[k_lo:k_hi + 1] += 2.0 * wi * a_slice
    return grad


def fft_shaping(a0, time_limit, alpha=0.4, verbose=False):
    """
    Fourier-domain iterative shaping:
    Iteratively flatten the power spectrum and project back to non-negative.
    This is a Gerchberg-Saxton style algorithm for spectral flatness.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        F = np.fft.rfft(a)
        mag = np.abs(F)
        phase = F / np.where(mag > 1e-15, mag, 1.0)
        
        # Flatten spectrum: raise power to alpha < 1
        mag_sq = mag ** 2
        mag_sq_mean = mag_sq.mean()
        if mag_sq_mean > 1e-15:
            mag_sq_normed = mag_sq / mag_sq_mean
            mag_new = np.sqrt(mag_sq_normed ** alpha)
            # Ensure DC component is reasonable
            if mag_new[0] < 1e-10:
                mag_new[0] = 1.0
        else:
            mag_new = mag.copy()
        
        F_new = mag_new * phase
        a_new = np.fft.irfft(F_new, 2 * n - 1)[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            if verbose and it % 100 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.8f}")
        
        a = a_new
        
        # Gradually increase alpha (less aggressive flattening)
        if it % 200 == 0:
            alpha = min(alpha * 1.02, 0.7)
    
    return best_a, best_obj


def multiplicative_weight_refined(a0, time_limit, lr_init=0.004, temperature_init=0.01,
                                  verbose=False):
    """
    Refined multiplicative weight (mirror descent) with:
    - Adaptive peak selection via softmax
    - Learning rate decay
    - Periodic randomization to escape local minima
    - Symmetry encouragement
    """
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temperature_init
    it = 0
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        total_iters += 1
        
        conv = fast_convolve(a)
        
        # Select peaks with softmax weighting (top-K)
        top_k = min(40, len(conv))
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peak_idx].copy()
        # Only use peaks that are close to max
        pv_mask = pv > 0.8 * pv.max()
        if pv_mask.sum() < 1:
            pv_mask[np.argmax(pv)] = True
        peak_idx_active = peak_idx[pv_mask]
        pv_active = pv[pv_mask]
        
        scaled = pv_active / temp
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()
        
        # Build full weight vector
        w_full = np.zeros(len(conv))
        w_full[peak_idx_active] = w_peaks
        
        # Compute subgradient
        grad = compute_subgradient(a, w_full, top_M=min(len(peak_idx_active), 30))
        
        # Multiplicative update
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Optional: gentle symmetry enforcement
        if it % 50 == 0:
            sym_a = 0.5 * (a + a[::-1])
            sym_a = np.maximum(sym_a, 1e-15)
            sym_a = sym_a / sym_a.sum()
            sym_obj = fast_convolve(sym_a).max()
            if sym_obj < best_obj:
                a = sym_a
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MW iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        # Schedule updates
        lr *= 0.9997
        temp *= 1.004
        
        # Random restart if stuck
        if no_improve > 400:
            rng = np.random.default_rng(total_iters * 13 + 7)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-12)
            a = a / a.sum()
            lr = max(lr_init * 0.5, lr * 2)
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_a, best_obj


def softmax_gradient_descent(a0, time_limit, T_init=3.0, lr_init=0.005,
                             momentum=0.85, verbose=False):
    """
    Smoothed gradient descent using softmax approximation of max.
    With Nesterov momentum and adaptive step size.
    """
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    v = np.zeros(n)  # momentum buffer
    y = a.copy()  # Nesterov lookahead
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov: evaluate at lookahead point
        conv_y = fast_convolve(y)
        
        # Softmax weights
        conv_scaled = conv_y / T
        conv_scaled -= conv_scaled.max()
        w = np.exp(conv_scaled)
        w /= w.sum()
        
        # Compute gradient at y
        grad = compute_subgradient(y, w, top_M=30)
        
        # Nesterov momentum update
        v = momentum * v + grad
        a_new = y - lr * v
        a_new = project_simplex(a_new)
        
        # Update Nesterov lookahead
        y = a_new + momentum * (a_new - a)
        y = project_simplex(y)
        
        a = a_new
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f}, T={T:.3f}, lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.95
        
        T *= 0.9997
        T = max(T, 0.3)
        
        # Restart if stuck
        if no_improve > 600:
            rng = np.random.default_rng(it * 7 + 3)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-12)
            a = a / a.sum()
            v = np.zeros(n)
            y = a.copy()
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def coordinate_descent_fine(a0, time_limit, verbose=False):
    """

    if s < 1e-15:

        a = a / s

    best_obj = fast_convolve(a).max()

        epoch_improved = False
        
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        
        # Order: indices contributing to peak first (by product), then others
        k_lo = max(0, peak_k - n + 1)
        k_hi = min(n - 1, peak_k)
        idx_range = np.arange(k_lo, k_hi + 1)
        
        if len(idx_range) > 0:
            products = a[idx_range] * a[peak_k - idx_range]
            order_main = idx_range[np.argsort(-products)]
        else:
            order_main = np.array([], dtype=int)
        
        other = np.setdiff1d(np.arange(n), idx_range)
        if len(other) > 0:
            order_other = other[np.argsort(-a[other])]
        else:
            order_other = np.array([], dtype=int)
        
        full_order = np.concatenate([order_main, order_other])
        
        for j in full_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            j = int(j)
            old_val = a[j]
            if old_val < 1e-10:
                # Try small positive values
                candidates = [0.001, 0.002, 0.005, 0.01]
            else:
                candidates = [old_val * m for m in [0.4, 0.6, 0.8, 1.25, 1.5, 2.0]]
                candidates = [c for c in candidates if 0 < c < 0.9]
            
            best_c_val = old_val
            best_c_obj = best_obj
            
            for cv in candidates:
                a_trial = a.copy()
                a_trial[j] = cv
                ss = a_trial.sum()
                if ss < 1e-10 or ss > 2.0:
                    continue
                a_trial = a_trial / ss
                obj = fast_convolve(a_trial).max()
                if obj < best_c_obj - 1e-13:
                    best_c_obj = obj
                    best_c_val = cv
            
            if best_c_val != old_val:
                a[j] = best_c_val
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-10:
                    a = a / ss
                if best_c_obj < best_obj - 1e-14:
                    best_obj = best_c_obj
                    best_a = a.copy()
                    epoch_improved = True
        
        if epoch_improved:
            no_epoch_improve = 0
        else:
            no_epoch_improve += 1
    
    return best_a, best_obj


def peak_shaving_coordinate(a0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: specifically reduces contributions
    to the largest peaks by adjusting the responsible coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()

        k_lo = max(0, peak_k - n + 1)
        k_hi = min(n - 1, peak_k)
        idx = np.arange(k_lo, k_hi + 1)
        
        if len(idx) == 0:
            break
        
        # Find the coordinate with maximum product contribution
        products = a[idx] * a[peak_k - idx]
        j_max = idx[np.argmax(products)]
        
        # Try reducing j_max and redistributing
        old_val = a[j_max]
        if old_val < 1e-8:
            no_improve += 1
            continue
        
        best_trial_obj = M
        best_trial = None
        best_mult = 1.0
        
        for mult in [0.5, 0.7, 0.85, 0.9, 1.1, 1.3]:
            new_val = old_val * mult
            if new_val < 0 or new_val > 0.5:
                continue
            trial = a.copy()
            trial[j_max] = new_val
            # Redistribute difference to non-peak contributors
            diff = old_val - new_val
            if abs(diff) > 1e-15:
                mask = np.ones(n, dtype=bool)

                if mask.sum() > 0:
                    remaining = trial[mask].sum()
                    if remaining > 1e-10:

                else:
                    trial = trial / trial.sum()
                    continue
            trial = np.maximum(trial, 0)
            ss = trial.sum()
            if ss < 1e-10:
                continue
            trial = trial / ss
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-13:

                best_trial = trial
                best_mult = mult
        
        if best_trial is not None:

                best_a = a.copy()

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_sinc_sq(n, width_frac=0.2):
    """sinc^2 shaped sequence."""
    x = np.arange(n) - n / 2.0
    x_norm = x / (n * width_frac)
    a = np.sinc(x_norm) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine(n, power=2.0):
    """Cosine window with adjustable power."""
    x = np.arange(n) / (n - 1.0)
    a = np.cos(np.pi * (x - 0.5)) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print(f"\n=== Phase 2: Deep MW (n={len(best_seq) if best_seq is not None else '?'}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem * 0.28, 250)
        print(f"  MW: t={t_alloc:.0f}s")
        a_opt, obj = multiplicative_weight_refined(base, time_limit=t_alloc,
                                                    lr_init=0.005, temperature_init=0.008,
                                                    verbose=True)
        try_seq(a_opt, "mw_deep")

    # === Phase 3: Softmax Gradient Descent ===
    print(f"\n=== Phase 3: Softmax GD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 25:
            t_alloc = min(rem * 0.25, 180)
            print(f"  SGD: t={t_alloc:.0f}s")
            a_opt, obj = softmax_gradient_descent(best_seq, time_limit=t_alloc,
                                                   T_init=2.0, lr_init=0.004,
                                                   momentum=0.82, verbose=True)
            try_seq(a_opt, "sgd")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length ===")
    for target_n in [1024, 960, 832, 704, 640]:
        rem = deadline - time.monotonic() - 200
        if rem < 25:
            break
        base = best_seq if best_seq is not None else parent
        if len(base) == target_n:
            continue
        f_interp = interpolate_sequence(base, target_n)
        sc = try_seq(f_interp, f"interp_{target_n}")
        
        # Optimize on new length
        t_alloc = min(rem * 0.25, 50)
        if t_alloc > 5:
            a_opt, obj = multiplicative_weight_refined(f_interp, time_limit=t_alloc,
                                                       lr_init=0.003, temperature_init=0.01)
            try_seq(a_opt, f"mw_{target_n}")

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.35, 100)
            print(f"  PeakShave: t={t_alloc:.0f}s")
            a_opt, obj = peak_shaving_coordinate(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "peak_shave")

    # === Phase 6: Fine Coordinate Descent ===
    print(f"\n=== Phase 6: Fine CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            print(f"  FineCD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_fine(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "fine_cd")

    # === Phase 7: Final regime - different hyperparameters ===
    print(f"\n=== Phase 7: Final Optimizations ===")
    if best_seq is not None:
        # MW with slower decay
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.4, 60)
            a_opt, obj = multiplicative_weight_refined(best_seq, time_limit=t_alloc,
                                                       lr_init=0.002, temperature_init=0.005,
                                                       verbose=True)
            try_seq(a_opt, "mw_slow")
        
        # SGD with different params
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem * 0.5, 40)
            a_opt, obj = softmax_gradient_descent(best_seq, time_limit=t_alloc,
                                                   T_init=1.0, lr_init=0.002,
                                                   momentum=0.9)
            try_seq(a_opt, "sgd_lowT")
        
        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, obj = coordinate_descent_fine(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
