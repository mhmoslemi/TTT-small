import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_objective(z, n, T):
    """LogSumExp objective in log-domain."""
    # softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    return 2.0 * n * lse

def lse_gradient(z, n, T):
    """Gradient of LogSumExp objective w.r.t. z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    w = exp_terms / exp_terms.sum()
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return 2.0 * n * grad_z

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute cross-correlation: result[i] = sum_j w[i+j]*a[j] for i=0..len(a)-1."""
    n = len(a)
    m = len(w)
    size = m + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(W * np.conj(A), fft_size)
    return result[:n]


def softmax_stable(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_and_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z (log-domain)."""
    a = softmax_stable(z)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    lse = max_b + T * (shifted.max() + np.log(exp_shifted.sum()))
    obj = 2.0 * n * lse

    # Weights for active peaks
    w = exp_shifted / exp_shifted.sum()

    # Gradient w.r.t. a: cross-correlation of w and a
    grad_a = fast_cross_corr(w, a)

    # Project to tangent space of simplex
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def cg_logsumexp(z0, n, T, maxiter, ftol=1e-14, deadline=None):
    """Conjugate gradient on LogSumExp objective in log-domain."""
    z = z0.copy()
    obj_cur, g_cur = lse_obj_and_grad(z, n, T)

    d = -g_cur.copy()
    alpha = 1.0 / (np.dot(d, d) + 1e-30)
    best_obj = obj_cur
    best_z = z.copy()

    for it in range(maxiter):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Line search (backtracking)
        alpha = 1.0 / (np.dot(d, d) + 1e-30)
        for ls in range(30):
            z_new = z + alpha * d
            obj_try, _ = lse_obj_and_grad(z_new, n, T)
            if obj_try < obj_cur - 1e-15 or ls == 29:
                break
            alpha *= 0.5
            if alpha < 1e-20:
                break

        g_new = None
        if obj_try < obj_cur:
            z = z_new
            obj_cur = obj_try
        else:
            # Keep z unchanged, use old gradient
            pass

        obj_cur, g_cur = lse_obj_and_grad(z, n, T)

        if obj_cur < best_obj:
            best_obj = obj_cur
            best_z = z.copy()

        if np.linalg.norm(g_cur, np.inf) < 1e-12:
            break

        # Beta (Polak-Ribiere)
        beta = max(0.0, np.dot(g_cur, g_cur - g_new) / (np.dot(g_new, g_new) + 1e-30)) if g_new is not None else 0.0
        d = -g_cur + beta * d

        g_new = g_cur

        if it % 500 == 0 and it > 0:
            # Check convergence
            pass

    return best_z, best_obj


def lbfgs_logsumexp(z0, n, T, maxiter, deadline=None):
    """L-BFGS on LogSumExp objective in log-domain."""
    t0 = time.monotonic()

    def func(z):
        if deadline is not None and time.monotonic() > deadline:
            return 1e15
        obj, _ = lse_obj_and_grad(z, n, T)
        return obj

    def jac(z):
        if deadline is not None and time.monotonic() > deadline:
            return np.zeros_like(z)
        _, g = lse_obj_and_grad(z, n, T)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 200, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def active_set_newton(a0, n_iters=200, deadline=None):
    """Newton refinement: identify active peaks and minimize max over them."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: peaks within 0.1% of max
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])

        # Build the Hessian-like system for the active constraints
        # We want to minimize max_{k in active} (a*a)[k] subject to sum(a)=1, a>=0
        # Using Newton: solve (T_a^T T_a + lambda*I) da = -grad
        # where T_a maps a to the active convolution values

        # Gradient of max active constraint (average)
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(active)

        # Project gradient to simplex tangent
        grad = grad - np.mean(grad)

        # Simple steepest descent with line search in log-domain
        # (avoids the complexity of full Newton with constraints)
        z = np.log(np.maximum(a, 1e-300))
        step = 0.01 / (np.dot(grad, grad) + 1e-30)

        for ls in range(20):
            z_new = z - step * grad
            a_new = softmax_stable(z_new)
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                z = z_new
                break
            step *= 0.5

        if step < 1e-15:
            break

    return best_a


def multiplicative_refine(a0, time_limit, n_steps=100000):
    """Multiplicative (mirror) descent targeting active peaks with adaptive rate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.02
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set with adaptive threshold
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-5:
                break

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)

    import cvxpy as cp

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # LP: find b minimizing max conv(y, b) subject to b>=0, sum(b)=1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        # Build constraint matrix implicitly using cvxpy
        # conv(y, b)[k] = sum_i y[i] * b[k-i] for valid i
        # This is a linear function of b

        # For efficiency, build the constraint matrix
        T_mat = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_mat[j:j + n, j] = y

        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for sol in [solver, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if sol is None:
                continue
            try:
                prob.solve(solver=sol)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        sb = b_val.sum()
        if sb < 1e-15:
            continue
        b_val = b_val / sb

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_val) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Aggressive LogSumExp on parent ===
    print(f"\n=== Phase 1: Aggressive LogSumExp n={n_best} (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    t_alloc = min((overall_deadline - t0) * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ls, _ = full_logsumexp_optimize(best_seq, t_alloc, overall_deadline)
        local_update(a_ls, "logsumexp_main")

    # === Phase 2: Active set Newton ===
    print(f"\n=== Phase 2: Active Set Newton (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    t_alloc = min((overall_deadline - t0) * 0.08, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_newton = active_set_newton(best_seq, n_iters=5000, deadline=overall_deadline)
        local_update(a_newton, "newton")

    # === Phase 3: Nesterov LP ===
    print(f"\n=== Phase 3: Nesterov LP (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    t_alloc = min((overall_deadline - t0) * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=200000)
        local_update(a_lp, "nesterov_lp")

    # === Phase 4: Second LogSumExp (from updated best) ===
    print(f"\n=== Phase 4: LogSumExp Round 2 (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    t_alloc = min((overall_deadline - t0) * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ls2, _ = full_logsumexp_optimize(best_seq, t_alloc, overall_deadline)
        local_update(a_ls2, "logsumexp_2")

    # === Phase 5: Spectral init + optimization ===
    print(f"\n=== Phase 5: Spectral Init Exploration (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    t_alloc = min((overall_deadline - t0) * 0.12, 150.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        for trial in range(3):
            rem = overall_deadline - time.monotonic()
            if rem < 15:
                break
            a_spec = spectral_init(n_best, rng)
            # Quick LogSumExp
            z = np.log(np.maximum(a_spec, 1e-300))
            T_stages_quick = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
            for T in T_stages_quick:
                rem_sub = overall_deadline - time.monotonic()
                if rem_sub < 5:
                    break
                z = lbfgs_logsumexp(z, n_best, T, maxiter=100, deadline=overall_deadline)
            a_spec_opt = softmax_stable(z)
            local_update(a_spec_opt, f"spectral_{trial}")

    # === Phase 6: Multiplicative refinement ===
    print(f"\n=== Phase 6: Multiplicative Refine (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    t_alloc = min((overall_deadline - t0) * 0.08, 80.0)
    if t_alloc > 10:
        a_mult, _ = multiplicative_refine(best_seq, t_alloc, n_steps=100000)
        local_update(a_mult, "multiplicative")

    # === Phase 7: Final Newton + LogSumExp polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    t0 = time.monotonic()
    t_alloc = min((overall_deadline - t0) * 0.30, 200.0)
    if t_alloc > 20:
        # Newton
        t_sub = min(t_alloc * 0.3, 50.0)
        a_fin_newton = active_set_newton(best_seq, n_iters=3000, deadline=overall_deadline)
        local_update(a_fin_newton, "final_newton")

        # LogSumExp with very low temperatures
        t_sub = min((overall_deadline - time.monotonic()) * 0.4, 80.0)
        if t_sub > 10:
            z = np.log(np.maximum(best_seq, 1e-300))
            T_final = [0.1, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
            for T in T_final:
                rem_sub = overall_deadline - time.monotonic()
                if rem_sub < 5:
                    break
                z = lbfgs_logsumexp(z, n_best, T, maxiter=200, deadline=overall_deadline)
            a_fin = softmax_stable(z)
            local_update(a_fin, "final_lse")

        # Final multiplicative
        t_sub = min(overall_deadline - time.monotonic() - 5, 30.0)
        if t_sub > 5:
            a_fin_m, _ = multiplicative_refine(best_seq, t_sub, n_steps=50000)
            local_update(a_fin_m, "final_mult")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
