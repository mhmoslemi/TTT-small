- For the active constraints (where conv[k] = M), the gradient of conv[k] with respect to a must be a linear combination of the gradients of the active inequalities and the sum constraint.
- For inactive coordinates (a[i] = 0), the "reduced cost" must be non-positive.

The gradient of conv[k] with respect to a[i] is a[k-i] + a[i] (for valid k-i).

So at the optimum: for all active k, sum_{k active} λ_k * (a[k-i] + a[i]) = μ (for all i with a[i] > 0), where μ is the Lagrange multiplier for the sum constraint.

This means: sum_{k active} λ_k * a[k-i] + (sum_{k active} λ_k) * a[i] = μ

This is a linear system in a (for the support of a). If we know which k are active and which i are in the support, we can solve for a.

This suggests an algorithm:
1. Guess the active set A (which k values have conv[k] = M) and the support S (which i have a[i] > 0)
2. Solve the linear system: for i in S, sum_{k in A} λ_k * a[k-i] + (sum_{k in A} λ_k) * a[i] = μ
3. Also: sum_{i in S} a[i] = 1
4. Check: conv[k] = M for k in A, conv[k] < M for k not in A, a[i] > 0 for i in S
5. If not satisfied, adjust the active set / support and iterate

def fast_convolve(a):
    n = len(a)
    if n < 2:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def spectral_flattening(a0, time_limit, label=""):
    """Gerchberg-Saxton style spectral flattening for diversification."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    alpha = 0.5
    damping = 0.3

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        A = fft(a)
        mag = np.abs(A)
        ph = np.angle(A)

        mag_target = 1.0 / np.sqrt(n)
        mag_compressed = np.power(mag + 1e-30, alpha)
        mag_compressed /= np.mean(mag_compressed)

        A_new = mag_compressed * np.exp(1j * ph)
        a_new = np.real(ifft(A_new))

        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss

        a = (1 - damping) * a + damping * a_new
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a /= ss

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        if it % 20 == 0:
            alpha = max(0.1, alpha * 0.95)
            damping = max(0.05, damping * 0.97)

    return best_a, best_score

def active_set_polish(a0, time_limit, label=""):
    """
    KKT-based active set method:
    At optimum, for active lags K, lambda_k * a[k-i] + (sum lambda) * a[i] = mu
    for all i in support. We iterate: fix active set, solve linear system, check.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 200:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Find active lags (within 1% of max)
        threshold = M * 0.95
        active_lags = np.where(conv >= threshold)[0]
        if len(active_lags) < 2:
            active_lags = np.where(conv >= M * 0.8)[0]
        if len(active_lags) < 2:
            active_lags = np.arange(2 * n - 1)

        # Find support
        support = np.where(a > 1e-8)[0]
        if len(support) < 5:
            support = np.where(a > 1e-12)[0]
        if len(support) < 5:
            support = np.arange(n)

        # Build system: for i in support,
        # sum_{k in active} lambda_k * (a[k-i] + a[i]) = mu
        # We set mu = 1 (scale), so:
        # sum_{k in active} lambda_k * a[k-i] + (sum_{k in active} lambda_k) * a[i] = 1
        # And: sum_{i in support} a[i] = 1
        # Unknowns: a[i] for i in support, lambda_k for k in active_lags
        # But this is coupled. Instead, use iterative approach:
        # Fix lambdas, solve for a; fix a, solve for lambdas.

        # Solve for lambdas given a:
        # min sum lambda_k subject to: for all i in support,
        #   sum_k lambda_k * (a[k-i] + a[i]) >= 1
        #   lambda_k >= 0
        # This gives the dual of the LP best response!

        # Instead, let's just do a weighted pressure reduction:
        # Compute "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in active_lags:
            weight = conv[k] / M  # normalized weight
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                pressure[i] += weight * (a[j] + a[i])

        p_max = pressure.max()
        if p_max < 1e-20:
            break
        pressure = pressure / p_max

        # Reduce high-pressure coordinates, boost low-pressure ones
        eta = 0.05
        a_new = a * (1.0 - eta * pressure)
        a_new = np.maximum(a_new, 0)
        a_new += eta * 0.1 * (1 - pressure) * (a > 1e-10).astype(float)
        a_new = np.maximum(a_new, 0)

        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < score + 1e-12:
            a = a_new
        else:
            eta *= 0.5
            a_new = a * (1.0 - eta * pressure)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss > 1e-15:
                a_new = a_new / ss
                a = a_new

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(400.0, rem * 0.25)
    a3, s3, _ = lp_fictitious_play(best_seq, t3, solver=solver, label="p3")
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_lpfp")
    # Quick CD polish after LP
    t3b = min(60.0, rem * 0.04)
    a3b, s3b, _ = smart_coord_descent(best_seq, t3b, label="p3b")
    best_seq, best_score = check_and_update(a3b, best_seq, best_score, "p3b_cd")

    # === Phase 4: Active set polishing ===
    print(f"\n=== Phase 4: Active Set Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(120.0, rem * 0.08)
    a4, s4 = active_set_polish(best_seq, t4, label="p4")
    best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_as")
    # CD after active set
    t4b = min(60.0, rem * 0.04)
    a4b, s4b, _ = smart_coord_descent(best_seq, t4b, label="p4b")
    best_seq, best_score = check_and_update(a4b, best_seq, best_score, "p4b_cd")

    # === Phase 5: Perturbation + re-optimization loop ===
    print(f"\n=== Phase 5: Perturb + Optimize Loop (best={best_score:.6f}) ===")
    for rnd in range(5):
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        # Perturb
        if rnd % 3 == 0:
            perturbed = peak_targeted_perturb(best_seq, rng, strength=0.03 + 0.01 * rnd)
        elif rnd % 3 == 1:
            perturbed = random_restructure(best_seq, rng, frac=0.1 + 0.05 * rnd)
        else:
            # Spectral perturbation
            A = fft(perturbed if False else best_seq)
            mag = np.abs(A) * (1 + 0.1 * rng.standard_normal(len(A)))
            mag = np.maximum(mag, 0)
            A_new = mag * np.exp(1j * np.angle(A))
            perturbed = np.real(ifft(A_new))
            perturbed = np.maximum(perturbed, 0)
            ss = perturbed.sum()
            if ss > 1e-15:
                perturbed /= ss
            else:
                perturbed = best_seq.copy()

        # Optimize perturbed version
        t5 = min(80.0, rem * 0.1)
        a5, s5, _ = smart_coord_descent(perturbed, t5, label=f"p5r{rnd}")
        best_seq, best_score = check_and_update(a5, best_seq, best_score, f"p5_scd_r{rnd}")

        # LP on the perturbed result if it's competitive
        rem = deadline - time.monotonic()
        if rem > 60 and s5 < best_score + 0.01:
            t5lp = min(60.0, rem * 0.1)
            a5lp, s5lp, _ = lp_fictitious_play(a5, t5lp, solver=solver, label=f"p5lp_r{rnd}")
            best_seq, best_score = check_and_update(a5lp, best_seq, best_score, f"p5_lp_r{rnd}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for alt in [n_best, n_best // 2, n_best * 2, 768, 1280, 1536, 2048]:
        if 512 <= alt <= 3072 and alt not in alt_lengths:
            alt_lengths.append(alt)
    alt_lengths = alt_lengths[:4]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        if n_alt == n_best:
            init_alt = best_seq.copy()
        else:
            init_alt = interpolate_seq(best_seq, n_alt)

        t6 = min(80.0, rem * 0.15)
        a6, s6, _ = smart_coord_descent(init_alt, t6, label=f"p6_n{n_alt}")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, f"p6_n{n_alt}")

        # LP on best alternative
        rem = deadline - time.monotonic()
        if rem > 40 and s6 < best_score + 0.005:
            t6lp = min(60.0, rem * 0.1)
            a6lp, s6lp, _ = lp_fictitious_play(a6, t6lp, solver=solver, label=f"p6lp_n{n_alt}")
            best_seq, best_score = check_and_update(a6lp, best_seq, best_score, f"p6lp_n{n_alt}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(120.0, rem * 0.4)
        a7, s7 = fine_coord_descent(best_seq, t7, label="p7")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, "p7_fine")

    rem = deadline - time.monotonic()
    if rem > 20:
        t7b = min(80.0, rem * 0.4)
        a7b, s7b, _ = smart_coord_descent(best_seq, t7b, label="p7b")
        best_seq, best_score = check_and_update(a7b, best_seq, best_score, "p7b_scd")

    rem = deadline - time.monotonic()
    if rem > 15:
        a7c, s7c = fine_coord_descent(best_seq, min(rem * 0.7, 40.0), label="p7c")
        best_seq, best_score = check_and_update(a7c, best_seq, best_score, "p7c_fine")

    # === Phase 8: One final LP + CD cycle ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 8: Final LP+CD (best={best_score:.6f}) ===")
        t8 = min(rem * 0.4, 120.0)
        a8, s8, _ = lp_fictitious_play(best_seq, t8, solver=solver, label="p8")
        best_seq, best_score = check_and_update(a8, best_seq, best_score, "p8_lp")
        rem = deadline - time.monotonic()
        if rem > 20:
            a8b, s8b = fine_coord_descent(best_seq, min(rem * 0.6, 40.0), label="p8b")
            best_seq, best_score = check_and_update(a8b, best_seq, best_score, "p8b_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
