    m_hat = m / (1 - beta1 ** t)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LogSumExp
gradient descent with Adam optimizer, cyclic temperature annealing,
multi-start diversity, and peak-directed coordinate descent polish.
Key innovation: O(n log n) gradient computation via FFT-based cross-correlation,
enabling many more optimization steps within the time budget.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_cross_corr(w, a):
    """Compute C[i] = sum_k w[k]*a[k-i] for i=0..n-1 via FFT."""
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(W * np.conj(A), N)[:n]
    return c


def compute_lse_grad_and_score(z, T):
    """Compute gradient of 2n*LSE_T(conv(a,a)) w.r.t. log-params z.
    Returns (grad_z, true_score, a, b)."""
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

    # Softmax weights for LSE
    scaled = b / T - max_b / T
    exp_b = np.exp(scaled)
    w = exp_b / exp_b.sum()

    # Cross-correlation C[i] = sum_k w[k]*a[k-i]
    C = compute_cross_corr(w, a)

    # grad_a = 4n * C
    # grad_z = 4n * a * (C - dot(a, C))
    S = np.dot(a, C)
    grad_z = 4.0 * n * a * (C - S)

    true_score = 2.0 * n * max_b  # since s=1
    return grad_z, true_score, a, b


def adam_update(grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Single Adam step. Returns (m, v, z_update)."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v, update


def optimize_lse(a0, time_limit, rng, T_start=3.0, T_end=0.02, lr_init=0.08):
    """Main LogSumExp gradient descent with Adam and adaptive temperature."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-300)
    z = np.log(a0) - np.log(a0).mean()
    T = T_start
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    t0 = time.monotonic()
    stale = 0
    patience = 300

    while time.monotonic() - t0 < time_limit:
        progress = (time.monotonic() - t0) / time_limit
        T = max(T_end, T_start * (1.0 - progress) ** 2.5)

        grad_z, score, a, b = compute_lse_grad_and_score(z, T)
        t += 1
        m, v, update = adam_update(grad_z, m, v, t, lr)
        z = z - update
        z = z - z.mean()  # re-center for numerical stability

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        if stale > patience:
            # Reset and perturb
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(0.02, lr * 0.85)
            z = z + 0.05 * rng.standard_normal(n)
            z = z - z.mean()
            stale = 0
            patience = max(100, patience - 20)

        if t % 1000 == 0:
            lr = max(0.01, lr * 0.97)

    return best_a, best_score


def peak_cd(a0, time_limit, rng):
    """Peak-directed coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        max_k = int(np.argmax(b))

        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_k - idx]

        order = np.argsort(contribs)[::-1]
        improved = False
        for oi in order[:6]:
            pos = idx[oi]
            if a[pos] < 1e-10:
                continue
            for factor in [0.995, 0.99, 0.98, 0.965, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] * (1.0 - factor)
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-11:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if rng.random() < 0.4:
                # Random kick
                a_trial = a + 0.02 * rng.standard_normal(n)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
            if no_improve > 500:
                break

    return best_a, best_score


def multi_peak_subgrad(a0, time_limit, rng):
    """Subgradient method targeting multiple top peaks simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        # Top peaks within 1% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        active = active[:min(20, len(active))]  # cap active set

        # Average gradient over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[k - idx]
            # db_k/da_i = 2*a[k-i] for i in idx
            g[idx] += 2.0 * contrib
        g /= len(active)

        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        else:
            lr *= 0.95
            if lr < 1e-6:
                break
            # Occasional random restart
            if rng.random() < 0.1:
                a_trial = a + 0.01 * rng.standard_normal(n)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc2 = 2.0 * n * fast_convolve(a_trial).max()
                if sc2 < best_score:
                    a = a_trial
                    best_score = sc2
                    best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:

        a = np.sin(np.pi * x) ** 2
    elif choice == 2:

    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    elif choice == 7:
        a = (1 - x) ** 0.7 + 0.1
    elif choice == 8:
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif choice == 9:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2) + 0.1
    elif choice == 10:
        a = np.cos(np.pi * x) ** 2 + 0.2 * np.cos(2 * np.pi * x) ** 2
        a = np.maximum(a, 0)
    else:
        freqs = rng.integers(1, 8, size=4)
        amps = rng.standard_normal(4) * 0.1
        phases = rng.uniform(0, 2 * np.pi, 4)
        a = np.ones(n)
        for f, amp, ph in zip(freqs, amps, phases):
            a = a + amp * np.cos(2 * np.pi * f * x + ph)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def score_of(a):
    """Compute the objective for a given sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

            sa = score_of(trial)
            print(f"    Init score: {sa:.10f} (n={len(trial)})")

            a_s, ss = optimize_lse(trial, t_each, rng, T_start=2.0, T_end=0.03, lr_init=0.08)
            attempt_update(a_s, f"start_{si}")

    # === Phase 5: Second LogSumExp pass (colder) ===
    print(f"\n=== Phase 5: Cold refinement ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t5 = min(rem * 0.3, 300.0)
        print(f"  Budget: {t5:.0f}s, current best: {best_score:.10f}")
        a5, s5 = optimize_lse(best_seq, t5, rng, T_start=1.0, T_end=0.01, lr_init=0.05)
        attempt_update(a5, "p5_cold")
        print(f"  Phase 5 best: {s5:.10f}")

    # === Phase 6: Peak CD again ===
    rem = deadline - time.monotonic()
    if rem > 40:
        t6 = min(rem * 0.4, 100.0)
        a6, s6 = peak_cd(best_seq, t6, rng)
        attempt_update(a6, "p6_peakcd2")

    # === Phase 7: Final cold LogSumExp ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.6, 80.0)
        a7, s7 = optimize_lse(best_seq, t7, rng, T_start=0.3, T_end=0.005, lr_init=0.03)
        attempt_update(a7, "p7_final_cold")

    # Final verification
    if best_seq is not None:
        fs = score_of(best_seq)
        print(f"\nFINAL SCORE: {fs:.10f} (n={len(best_seq)})")
        best_score = fs

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
