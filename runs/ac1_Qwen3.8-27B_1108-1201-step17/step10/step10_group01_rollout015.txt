Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LogSumExp

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_gradient(a, beta, fft_size=None):
    """
    Compute the gradient of (1/beta) * log(sum_k exp(beta * (a*a)_k))
    using FFT for O(n log n) complexity.
    
    The gradient at position m is: sum_k w_k * a[k-m] (for valid k-m)
    where w_k = exp(beta * p_k) / sum_j exp(beta * p_j)
    and p = conv(a, a).
    """
    n = len(a)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1 << (size - 1).bit_length()
    
    # Compute p = conv(a, a)
    A = rfft(a, fft_size)
    p = irfft(A * A, fft_size)[:size]
    
    # Compute softmax weights
    max_p = p.max()
    exp_beta_p = np.exp(beta * (p - max_p))
    w = exp_beta_p / exp_beta_p.sum()
    
    # Compute gradient: grad_m = sum_k w_k * a[k-m] for valid indices
    # This is cross-correlation of w (padded to size) and a
    # grad_m = sum_j w[m+j] * a[j] for j=0..n-1 (where m+j < size)
    # = cross-correlation
    
    # Using FFT: grad = ifft(conj(fft(w_padded)) * fft(a_padded)) 
    # where w_padded is w with appropriate padding
    
    # Actually, let me think more carefully.
    # grad_m = sum_{j=0}^{n-1} w[m+j] * a[j]  (where m+j < size)
    # This is the cross-correlation of w and a, evaluated at lag m.
    
    # corr[m] = sum_j w[m+j] * a[j]
    # = sum_j w[m+j] * a[j]
    
    # If we define W = fft(w) and A_full = fft(a padded to size), then:
    # corr = ifft(conj(W) * A_full) ... hmm, let me think again.
    
    # Standard cross-correlation: corr[m] = sum_j x[j+m] * y[j]
    # In FFT domain: corr = ifft(conj(fft(x, N)) * fft(y, N)) where N >= len(x)+len(y)-1
    
    # Here x = w (length size), y = a (length n)
    # We need N >= size + n - 1
    # But we only need corr[0..n-1]
    
    # Actually, since w has length size = 2n-1 and a has length n:
    # corr[m] = sum_j w[m+j] * a[j] for j=0..n-1, m=0..n-1
    # We need w[m+j] to be valid, so m+j <= size-1, i.e., m <= size-1-j
    # For m up to n-1 and j up to n-1: m+j up to 2n-2 = size-1. OK.
    
    # FFT approach: pad both to length N >= size + n - 1 = 3n - 2
    N = 1 << (size + n - 2).bit_length()
    W = rfft(w, N)
    A_p = rfft(a, N)
    corr = irfft(np.conj(W) * A_p, N)
    grad = corr[:n]
    
    return grad, p, float((1.0/beta) * (max_p + np.log(exp_beta_p.sum())))


def lse_smooth_optimize(a0, time_limit, betas=None, lr_schedule='adaptive', 
                         momentum=0.9, print_interval=500):
    """
    Optimize using LogSumExp-smoothed objective with FFT-accelerated gradients.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    if betas is None:
        betas = [50, 30, 20, 15, 10, 8, 6, 5, 4, 3, 2.5, 2, 1.5, 1, 0.8, 0.5, 0.3, 0.2]
    
    # Precompute FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    corr_fft_size = 1 << (size + n - 2).bit_length()
    
    v = np.zeros(n)  # momentum velocity
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Determine number of iterations for this beta stage
        # Use more iterations for smaller beta (sharper)
        n_iters = int(min(3000, max(200, remaining * 500 / len(betas))))
        
        # Learning rate: scale inversely with beta for stability
        # The gradient magnitude is roughly 2*max_p/beta * n (normalized)
        # We want the step to be O(1/n) for stability
        base_lr = 0.5 / (beta * np.sqrt(n))
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient
            grad, p, lse_val = lse_gradient(a, beta, corr_fft_size)
            
            # The objective is 2n * LSE(beta, p) / (sum(a))^2
            # Gradient w.r.t. a: 2n * grad / (sum(a))^2 - 4n * LSE * a / (sum(a))^3
            # Since sum(a) = 1: gradient = 2n * grad - 4n * LSE * a
            # But we want to minimize 2n * max_p, so the gradient direction is grad
            # (the LSE is just a smooth approximation to max_p)
            # Actually the true gradient of (1/beta)log(sum exp(beta*p_k)) w.r.t. a_m
            # is sum_k w_k * dp_k/da_m = sum_k w_k * a[k-m] = what we computed as 'grad'
            # And the full objective gradient (with 2n factor and normalization) is:
            # 2n * grad (when sum(a)=1, the normalization constraint is handled by projection)
            
            # Nesterov momentum update
            v = momentum * v - base_lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            # Check improvement
            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new
            
            if (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE beta={beta}: iter={it+1}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=50000,
                         stall_patience=10, print_interval=500):
    """Nesterov LP alternation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(42)
    
    if solver is None:
        solver = _find_lp_solver()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = _solve_lp_best_response(y, solver, n)
        if b_new is None:
            stall += 1
            if stall > 20:
                break
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= stall_patience:
                noise = rng.standard_normal(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
        
        if (it + 1) % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_LP={M_b:.6f}")
    
    return best_a, best_score

def _solve_lp_best_response(a_fixed, solver, n, max_iters=3000):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iters, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def peak_coord_descent(a0, time_limit, n_steps=50000, top_m=6):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]
        
        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
            # Small random perturbation
            rng = np.random.default_rng(step)
            a = a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LogSumExp gradient
descent with adaptive temperature annealing, LP alternation, peak-targeted
coordinate descent, multi-length exploration, and diverse multi-start.
Key improvement over prior SOTA: O(n log n) gradient via FFT cross-correlation,
faster convergence with momentum + adaptive LR, and more aggressive peak
equalization through top-k weighted subgradients.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_lse_gradient(a, beta, corr_fft_size):
    """
    Compute (LSE value, gradient) of (1/beta)*log(sum_k exp(beta*p_k))
    where p = conv(a,a). Uses FFT for O(n log n) gradient.
    Gradient_m = 2 * sum_k w_k * a[k-m] = 2 * cross_corr(w, a)[m]
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Compute p = conv(a, a)
    A = rfft(a, fft_size)
    p = irfft(A * A, fft_size)[:size]

    # Softmax weights
    max_p = p.max()
    exp_beta_p = np.exp(beta * (p - max_p))
    w = exp_beta_p / exp_beta_p.sum()
    lse_val = (1.0 / beta) * (max_p + np.log(exp_beta_p.sum()))

    # Cross-correlation: grad_m = 2 * sum_j w[m+j] * a[j]
    W = rfft(w, corr_fft_size)
    A_full = rfft(a, corr_fft_size)
    corr = irfft(np.conj(W) * A_full, corr_fft_size)
    grad = 2.0 * corr[:n]

    return lse_val, grad, p


def lse_gradient_optimize(a0, time_limit, betas=None, momentum=0.9,
                          print_interval=200):
    """
    Main optimizer: LogSumExp-smoothed gradient descent with momentum.
    Adaptive learning rate per beta stage.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    if betas is None:
        betas = [40, 25, 15, 10, 7, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1, 0.8, 0.6, 0.45, 0.35, 0.25, 0.18, 0.12]

    size = 2 * n - 1
    corr_fft_size = 1 << (size + n - 2).bit_length()
    v = np.zeros(n)

    for si, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # More iterations for smaller beta (sharper peak approximation)
        if si < len(betas) // 2:
            n_iters = int(min(2000, max(100, remaining * 80 / len(betas))))
        else:
            n_iters = int(min(3000, max(200, remaining * 150 / len(betas))))

        # Learning rate: smaller beta means sharper gradient, use smaller LR
        base_lr = 0.8 / (beta * np.sqrt(n))
        base_lr = min(base_lr, 1.0 / n)

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            lse_val, grad, p = compute_lse_gradient(a, beta, corr_fft_size)

            # Momentum update
            v = momentum * v - base_lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new

            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new

            if (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE b={beta:.2f} it={it+1}/{n_iters} score={best_score:.8f} t={elapsed:.0f}s")

    return best_a, best_score


def topk_subgradient_optimize(a0, time_limit, n_steps=30000, top_k=30,
                              beta_w=3.0, momentum=0.85, print_interval=5000):
    """
    Subgradient method with top-k peak weighting and FFT-accelerated gradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.03
    no_improve = 0
    v = np.zeros(n)

    size = 2 * n - 1
    corr_fft_size = 1 << (size + n - 2).bit_length()
    fft_size = 1 << (size - 1).bit_length()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute auto-convolution
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:size]

        # Top-k softmax weighting
        top_idx = np.argsort(b)[::-1][:min(top_k, size)]
        weights = np.zeros(size)
        top_vals = b[top_idx]
        w_top = np.exp(beta_w * (top_vals - top_vals[0]))
        weights[top_idx] = w_top
        weights /= weights.sum()

        # FFT gradient
        W = rfft(weights, corr_fft_size)
        A_full = rfft(a, corr_fft_size)
        grad = 2.0 * irfft(np.conj(W) * A_full, corr_fft_size)[:n]

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            lr *= 0.92
            no_improve = 0
            v *= 0.7
        lr = max(1e-7, lr)

        if (step + 1) % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK-SG {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def _find_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def _solve_lp_best_response(a_fixed, solver, n, max_iters=5000):
    """Given fixed a (sum=1), find b>=0, sum(b)=1 minimizing max_k conv(a,b)_k."""
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iters, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            sv = b_val.sum()
            if sv < 1e-15:
                return None, None
            b_val = b_val / sv
            return b_val, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    return None, None
                b_val = b_val / sv
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None
