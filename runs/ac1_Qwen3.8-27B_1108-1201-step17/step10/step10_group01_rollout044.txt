import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def simplex_project(v, target=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    inds = np.arange(1, n + 1)
    cond = u * inds > (cv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a_fixed, timeout=5.0):
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_T(a_fixed)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s, float(res.x[-1])
    except Exception:
        pass
    return None, None

def minimax_iterate(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    y = a.copy()
    stall = 0
    prev_best = best_score
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, M_b = lp_best_response(y, timeout=3.0)
        if b_new is None:
            stall += 1
            if stall > 15:
                break
            continue
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        a = ((it) * a + alpha * (it + 2) * b_new) / (it + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        sc_a = evaluate_local(a)
        sc_y = evaluate_local(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if it % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 12:
                scale = 0.015 / np.sqrt(stall)
                y = np.maximum(y + rng.standard_normal(n) * scale, 0)
                y = y / y.sum()
                stall = 0
            if verbose and it % 500 == 0:
                print(f"    MM {time.monotonic()-t0:.0f}s it={it} score={best_score:.8f}")
    return best_a, best_score

def peak_subgrad(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    y = a.copy()
    t_k = 1.0
    lr = 0.008
    no_imp = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(y)
        max_c = c.max()
        # Active set
        thresh = max_c * 0.995
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Weights
        w = c[active] - thresh + 1e-10
        w = w / w.sum()
        # Gradient
        grad = np.zeros(n)
        for i, k in enumerate(active):
            if w[i] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je+1] += 2.0 * w[i] * y[k - np.arange(js, je+1)]
        # Step
        a_new = y - lr * grad
        a_new = simplex_project(a_new, 1.0)
        sc = evaluate_local(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = simplex_project(y, 1.0)
        a = a_new
        t_k = t_new
        if no_imp >= 60:
            lr *= 0.92
            no_imp = 0
        if lr < 1e-7:
            lr = 0.003
            # Reset with perturbation
            rng = np.random.default_rng(it)
            y = np.maximum(a + rng.standard_normal(n) * 0.005, 0)
            y = y / y.sum()
        if verbose and it % 3000 == 0:
            print(f"    PG {time.monotonic()-t0:.0f}s it={it} score={best_score:.8f} lr={lr:.6f}")
    return best_a, best_score
