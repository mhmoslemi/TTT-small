    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        r = np.nonzero(cond)[0][-1]
        theta = (cssv[r] - 1.0) / (r + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def gs_iterations(a0, time_limit, label="", target_flatness=0.0):
    """Gerchberg-Saxton style iterations to shape the spectrum for flat convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute FFT of zero-padded a
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: make |A|^2 such that IFFT(|A|^2) is as flat as possible in first 2n-1 positions
        # Strategy: damp the peaks of |A|^2 and boost the valleys
        A_sq = mag * mag
        # Normalize
        A_sq_norm = A_sq / A_sq.sum() if A_sq.sum() > 0 else A_sq

        # Apply a "flattening" pressure in frequency domain
        # Reduce high-magnitude frequencies, boost low ones
        median_mag = np.median(mag[mag > 1e-15]) if np.any(mag > 1e-15) else 1.0
        # Gentle flattening
        flatten_factor = 0.1 + 0.1 * (it % 10)
        new_mag = mag ** (1 - flatten_factor * 0.1)
        # Preserve DC
        new_mag[0] = mag[0]

        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}")

        # Check convergence
        if it > 100 and abs(a_new - a).max() < 1e-10:
            break

    return best_a, best_score

def smoothed_grad_descent(a0, time_limit, label="", beta_init=80.0, lr_init=0.03):
    """Nesterov-accelerated gradient descent on log-sum-exp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()

    beta = beta_init
    lr = lr_init
    gamma = 1.0
    y = a.copy()
    no_improve = 0
    patience = 200
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        # Compute log-sum-exp gradient
        conv_y = fast_convolve(y)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -200, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient computation: grad[j] = 2 * sum_{k} weights[k] * y[k-j] for valid k-j
        grad = np.zeros(n)
        active = np.where(weights > 1e-12)[0]
        for k in active:
            w = weights[k]
            if w < 1e-14:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j_vals = np.arange(i_start, i_end + 1)
            # grad[j] += 2 * w * y[k - j]
            grad[j_vals] += 2.0 * w * y[k - j_vals]

        # Mirror descent step (exponential family)
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        # Evaluate
        conv = fast_convolve(a)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve >= patience:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            patience = max(40, int(patience * 0.95))
            if lr < 1e-9:
                break
        elif it % 25 == 0:
            beta *= 1.03

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}, beta={beta:.1f}, lr={lr:.6f}")

    return best_a, best_score

def coord_descent(a0, time_limit, label="", num_top_peaks=8):
    """Coordinate descent targeting the top peaks of the convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_top + 1:].max()) if False else float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            # Generate candidate deltas targeting the top peaks
            candidates = [0.0, -ai * 0.9, -ai * 0.5, -ai * 0.99]

            # Find where affected lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.2 < d < ai * 1.2:
                        candidates.append(d)

            # Top-peak crossings
            top_k = min(num_top_peaks, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.2 < d < ai * 1.2:
                            candidates.append(d)

            # Fine fractional moves
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.3 * ai)

            # Evaluate candidates
            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Periodic recomputation to avoid drift
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps

def active_set_equalize(a0, time_limit, label="", K=10):
    """Try to equalize the top K peaks of the convolution using least-squares adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(a)
        M = conv.max()
        # Find top K peaks
        peak_threshold = M * 0.95
        peak_positions = np.where(conv >= peak_threshold)[0]
        if len(peak_positions) < 2:
            peak_positions = np.argsort(conv)[-K:]
        else:
            peak_positions = peak_positions[:max(K, len(peak_positions))]

        m_peaks = len(peak_positions)
        if m_peaks < 2:
            break

        # Build the linear system: for each peak k, we have
        # conv[k] + sum_i delta_i * 2 * a[k-i] = M_new
        # We want all peaks to be equal to M_new < M
        # Variables: delta_i (adjustments to a_i)
        # Constraint: a_i + delta_i >= 0, sum(delta_i) + sum(a_i) = 1 (preserve normalization)

        # To keep it tractable, only adjust a subset of coordinates
        # Focus on coordinates that contribute most to the peaks
        contrib = np.zeros(n)
        for k in peak_positions:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                contrib[i] += a[i] * a[k - i]

        # Select top contributing coordinates
        n_adjust = min(50, n)
        top_contrib = np.argsort(contrib)[-n_adjust:]

        # Build least squares system
        # For each peak k_j: sum_{i in top_contrib} 2*a[k_j - i]*delta_i = target_j - conv[k_j]
        # where target_j = M_new (unknown)
        # Additional constraint: sum(delta_i) = 0 (preserve sum)
        # Total unknowns: n_adjust + 1 (for M_new)
        # Total equations: m_peaks + 1 (sum constraint)

        A_mat = np.zeros((m_peaks + 1, n_adjust + 1))
        b_vec = np.zeros(m_peaks + 1)

        for j, k in enumerate(peak_positions):
            for idx, i in enumerate(top_contrib):
                kj = k - i
                if 0 <= kj < n:
                    A_mat[j, idx] = 2.0 * a[kj]
            A_mat[j, n_adjust] = -1.0  # coefficient for M_new
            b_vec[j] = 0.0  # we want conv[k_j] + ... = M_new

        # Sum constraint: sum(delta_i) = 0
        A_mat[m_peaks, :n_adjust] = 1.0
        b_vec[m_peaks] = 0.0

        # Solve least squares
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(A_mat, b_vec, rcond=None)
            deltas = sol[:n_adjust]
            M_new = sol[n_adjust]
        except Exception:
            break

        # Apply adjustments (with clipping to maintain non-negativity)
        scale = 0.3  # damping factor
        for idx, i in enumerate(top_contrib):
            a[i] = max(a[i] + scale * deltas[idx], 0.0)

        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Evaluate
        conv = fast_convolve(a)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        else:
            # Revert and stop if no improvement after several tries
            if M_new <= 0:
                break

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASE {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}, M={conv.max():.8f}")

        if it > 500:
            break

    return best_a, best_score

def fine_polish(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing with tiny steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            # Very fine steps
            for factor in [0.9995, 0.999, 0.998, 0.997, 0.995, 0.993, 0.99, 0.985, 0.98,
                          1.0005, 1.001, 1.002, 1.003, 1.005, 1.007, 1.01, 1.015, 1.02]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturb_sequence(a0, rng, method="gaussian", strength=0.05):
    """Perturb a sequence using various methods."""
    n = len(a0)
    a = a0.copy()

    if method == "gaussian":
        noise = rng.standard_normal(n) * strength
        a_new = a + noise
    elif method == "peak_targeted":
        conv = fast_convolve(a)
        M = conv.max()
        peak_thresh = M * 0.9
        peak_pos = np.where(conv >= peak_thresh)[0]
        pressure = np.zeros(n)
        for k in peak_pos:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                pressure[i] += a[i] * a[k - i]
        p_max = pressure.max()
        if p_max > 1e-20:
            pressure = pressure / p_max
        a_new = a * (1.0 - strength * 2 * pressure) + strength * 0.5 * a
    elif method == "restructure":
        frac = min(strength * 2, 0.3)
        mask = rng.random(n) > frac
        n_keep = max(n // 4, mask.sum())
        a_new = a.copy()
        a_new[~mask] = 0
        a_new[mask] = a_new[mask] * (1.0 + rng.standard_normal(int(mask.sum())) * 0.2)
        # Add some new energy at zero positions
        zeros = np.where(a_new < 1e-15)[0]
        if len(zeros) > 0:
            n_add = min(len(zeros), max(5, n // 20))
            add_idx = rng.choice(zeros, size=n_add, replace=False)
            a_new[add_idx] = rng.random(n_add) * a.max() * 0.3
    else:
        a_new = a.copy()

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s

def check_and_update(a, best_seq, best_score, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

            perturbed = perturb_sequence(best_seq, rng, methods[rnd], strengths[rnd])
            t5c = min(rem * 0.15, 80.0)
            a5c, s5c, _ = coord_descent(perturbed, t5c, label=f"p5p_r{rnd}")
            best_seq, best_score = check_and_update(a5c, best_seq, best_score, f"p5_pert_r{rnd}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    n_best = len(best_seq)
    alt_lengths = []
    for alt in [768, 1280, 1536, 2048, 3072]:
        if alt != n_best:
            alt_lengths.append(alt)
    alt_lengths = alt_lengths[:3]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t6 = min(rem * 0.2, 80.0)
        init_alt = interpolate_seq(best_seq, n_alt)
        a6, s6, _ = coord_descent(init_alt, t6, label=f"p6_n{n_alt}")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, f"p6_n{n_alt}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Intensive Polishing ===")
    # Multiple rounds of fine polish + active set
    for polish_rnd in range(3):
        rem = deadline - time.monotonic() - 30
        if rem < 15:
            break
        t7 = min(rem * 0.3, 80.0)
        a7, s7 = fine_polish(best_seq, t7, label=f"p7r{polish_rnd}")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, f"p7_fine_r{polish_rnd}")

        if rem > 30:
            t7b = min(rem * 0.1, 30.0)
            a7b, s7b = active_set_equalize(best_seq, t7b, label=f"p7a{polish_rnd}", K=8)
            best_seq, best_score = check_and_update(a7b, best_seq, best_score, f"p7_ase_r{polish_rnd}")

    # === Phase 8: Ultimate final polish ===
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        a_final, s_final = fine_polish(best_seq, rem, label="p8_final")
        best_seq, best_score = check_and_update(a_final, best_seq, best_score, "p8_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
