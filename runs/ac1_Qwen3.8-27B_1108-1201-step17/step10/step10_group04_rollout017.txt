        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b_resp.copy()
        else:
            y_new = y_new / sy

        a_new = (1.0 - alpha) * a + alpha * b_resp
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_resp.copy()
        else:
            a_new = a_new / sa

        y = y_new
        a = a_new

        score_a = subscore(a)
        score_y = subscore(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 5:
                rng = np.random.default_rng(it * 7919 + 3)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NG it={it} ({elapsed:.0f}s): best={best_score:.6f} M={M_val:.6f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, n_top=8, verbose=False):
    """Mirror descent (exponential weights) with top-K subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = subscore(a)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    lr = 0.4
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        max_b = b.max()

        top_idx = np.argsort(b)[::-1][:n_top]

        g = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(top_idx)

        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break

        z_new = z - lr * g
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()

        score = subscore(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stagnant = 0
            lr = min(lr * 1.03, 5.0)
        else:
            stagnant += 1
            lr *= 0.95

        a = a_new
        z = z_new

        if stagnant >= 40:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 10**8)
            a = a + rng.random(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            z = np.log(np.maximum(a, 1e-300))
            lr = 0.4
            stagnant = 0

        if verbose and (int((time.monotonic() - t0)) % 30 == 0):
            print(f"    MD t={time.monotonic()-t0:.0f}s: best={best_score:.6f}")

    return best_a, best_score


def lse_gradient_fft(z, T, n):
    """Compute gradient of LogSumExp objective w.r.t. z using FFT cross-correlation."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_conv(a)
    Tb = T * b
    max_Tb = min(Tb.max(), 500.0)
    w = np.exp(Tb - max_Tb)
    w = w / w.sum()

    # grad_a[i] = 2 * sum_j w[i+j] * a[j]  (cross-correlation of w and a)
    # Use FFT: cross_corr[i] = sum_j w[i+j]*a[j] = conv(w_rev, a)[2n-2-i]
    w_rev = w[::-1].copy()
    n_conv = 2 * n - 1
    fft_size = 1 << (n_conv - 1).bit_length()
    W_rev = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_full = irfft(W_rev * A, fft_size)[:n_conv]

    # grad_a[i] = 2 * conv_full[2n-2-i]
    grad_a = 2.0 * conv_full[2 * n - 2 - np.arange(n)]

    # Chain rule: grad_z = a * (grad_a - dot(a, grad_a))
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    return grad_z


def logsumexp_lbfgs(a0, time_limit, T_values=None, verbose=False):
    """LogSumExp smooth approximation + L-BFGS with FFT-accelerated gradient."""
    n = len(a0)
    if T_values is None:
        T_values = [100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1]

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    for si, T in enumerate(T_values):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def obj(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b = fast_conv(a_in)
            Tb = T * b
            max_Tb = min(Tb.max(), 500.0)
            val = (2.0 * n / T) * (max_Tb + np.log(np.exp(Tb - max_Tb).sum()))
            return val

        def grad(z_in):
            return lse_gradient_fft(z_in, T, n)

        max_iter = max(10, int(remaining / 0.5))
        try:
            res = minimize(fun=obj, x0=z, jac=grad, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-22,
                                    'gtol': 1e-14, 'maxfun': max_iter * 100,
                                    'disp': False})
            z = res.x
        except Exception:
            break

        if verbose:
            a_chk = np.exp(z - z.max())
            a_chk = a_chk / a_chk.sum()
            print(f"    LSE T={T}: score={subscore(a_chk):.6f} (t={time.monotonic()-t0:.0f}s)")

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def diverse_init(n, seed):
    """Generate diverse asymmetric initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)

    if choice == 0:
        a = np.exp(-4.0 * x)
    elif choice == 1:
        a = np.exp(4.0 * x)
    elif choice == 2:
        a = np.exp(-60.0 * x**2) + 0.3 * np.exp(-60.0 * (1.0 - x)**2)
    elif choice == 3:
        a = np.exp(-40.0 * (x - 0.25)**2) + 0.5 * np.exp(-40.0 * (x - 0.75)**2)
    elif choice == 4:
        # Random Fourier magnitude
        m = rng.integers(n // 4, n // 2)
        freqs = rng.choice(n, m, replace=False)
        phases = rng.uniform(0, 2 * np.pi, n)
        spec = np.ones(n, dtype=complex) * 0.1
        spec[freqs] = np.exp(1j * phases[freqs])
        spec[-freqs] = np.conj(spec[freqs])
        spec[0] = 1.0
        a = np.abs(np.fft.ifft(spec))**2
    elif choice == 5:
        # Geometric (strongly asymmetric)
        r = rng.uniform(0.6, 0.9)
        a = np.power(r, np.arange(n))
    elif choice == 6:
        # Two far-apart peaks
        p1 = rng.uniform(0.1, 0.3)
        p2 = rng.uniform(0.7, 0.9)
        sig = rng.uniform(0.02, 0.06)
        a = np.exp(-((x - p1) / sig)**2) + 0.4 * np.exp(-((x - p2) / sig)**2)
    elif choice == 7:
        # Cosine taper (asymmetric)
        a = np.cos(np.pi * (x - 0.15))**2
        a = np.maximum(a, 0)
    elif choice == 8:
        # Polynomially decaying
        p = rng.uniform(1.5, 4.0)
        a = np.power(1.0 - x + 0.05, p)
    elif choice == 9:
        # Random with smooth structure
        a = rng.random(max(5, n // 50))
        a = np.interp(x, np.linspace(0, 1, len(a)), a)
    elif choice == 10:
        # Sine-based asymmetric
        freq = rng.uniform(1, 5)
        a = 1.0 + 0.7 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely call the external evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_at_length(a_init, time_budget, solver, label=""):
    """Multi-phase optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0).astype(np.float64)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = subscore(best_a)
    t0 = time.monotonic()

    def try_update(a, src):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 1e-15:
            return
        a = a / ss
        sc = subscore(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"      [{src}] n={n}: {sc:.6f} (t={elapsed:.0f}s)")

    # Phase A: Mirror descent (fast, no LP)
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.15, 60.0)
    if t_a > 5:
        a_md, _ = mirror_descent(best_a, t_a, n_top=6)
        try_update(a_md, "MD")

    # Phase B: Nesterov LP game (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.45, 400.0)
    if t_b > 15:
        a_ng, _ = nesterov_game(best_a, t_b, solver=solver, verbose=True)
        try_update(a_ng, "NG")

    # Phase C: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.25, 200.0)
    if t_c > 10:
        a_lse = logsumexp_lbfgs(best_a, t_c, T_values=[20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1], verbose=False)
        try_update(a_lse, "LSE")

    # Phase D: Another NG pass from the improved point
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.15, 120.0)
    if t_d > 10:
        a_ng2, _ = nesterov_game(best_a, t_d, solver=solver, verbose=False)
        try_update(a_ng2, "NG2")

    # Phase E: Final LSE polish
    rem = time_budget - (time.monotonic() - t0)
    t_e = min(rem * 0.10, 60.0)
    if t_e > 5:
        a_polish = logsumexp_lbfgs(best_a, t_e, T_values=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05], verbose=False)
        try_update(a_polish, "POLISH")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: pan-length exploration with multi-phase optimization."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_deadline = t_start + max(1.0, float(budget_s) - 60.0)

    solver = detect_solver()
    print(f"LP solver: {solver}, total budget: {budget_s:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = subscore(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update_global(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = subscore(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine base length
    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 960
        best_seq = diverse_init(n_base, seed)
        best_score = subscore(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Optimize at base length
    print(f"\n=== Phase 1: Optimize at n={n_base} (score={best_score:.6f}) ===")
    rem = t_deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, sc_out = optimize_at_length(best_seq, t1, solver, f"n{n_base}")
        try_update_global(a_out, f"opt_n{n_base}")

    # Phase 2: Try longer lengths
    print(f"\n=== Phase 2: Longer lengths (best={best_score:.6f}) ===")
    longer_lengths = []
    for mult in [1.25, 1.5, 2.0, 2.5]:
        ln = int(n_base * mult)
        if n_base < ln <= 5000:
            longer_lengths.append(ln)
    longer_lengths = sorted(set(longer_lengths))

    for i, ln in enumerate(longer_lengths):
        rem = t_deadline - time.monotonic()
        if rem < 100:
            break
        t_alloc = min(rem * 0.22, 300.0)
        if t_alloc < 50:
            break
        print(f"\n  Trying n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_seq(best_seq, ln)
        sc_init = subscore(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"  Interpolation improved: {sc_init:.6f}")

        a_out, sc_out = optimize_at_length(init_long, t_alloc, solver, f"n{ln}")
        try_update_global(a_out, f"opt_n{ln}")

        # If better, do extra polish
        if a_out is not None and subscore(a_out) <= best_score + 1e-8:
            rem2 = t_deadline - time.monotonic()
            if rem2 > 40:
                t2 = min(rem2 * 0.15, 80.0)
                if t2 > 10:
                    a_pol = logsumexp_lbfgs(a_out, t2, T_values=[3.0, 1.5, 1.0, 0.5, 0.3, 0.1], verbose=False)
                    try_update_global(a_pol, f"polish_n{ln}")

    # Phase 3: Try shorter lengths
    print(f"\n=== Phase 3: Shorter lengths (best={best_score:.6f}) ===")
    shorter_lengths = []
    for mult in [0.75, 0.5, 0.4]:
        sn = int(n_base * mult)
        if 100 <= sn < n_base:
            shorter_lengths.append(sn)
    shorter_lengths = sorted(set(shorter_lengths))

    for sn in shorter_lengths:
        rem = t_deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.15, 120.0)
        if t_alloc < 20:
            break
        print(f"  Trying n={sn}, budget={t_alloc:.0f}s")
        init_short = interpolate_seq(best_seq, sn)
        a_out, sc_out = optimize_at_length(init_short, t_alloc, solver, f"n{sn}")
        try_update_global(a_out, f"opt_n{sn}")

    # Phase 4: Diverse restarts at the best length
    print(f"\n=== Phase 4: Diverse restarts at n={len(best_seq)} ===")
    n_best = len(best_seq)
    rem = t_deadline - time.monotonic()
    n_restarts = max(1, min(4, int(rem / 80)))
    for ri in range(n_restarts):
        rem = t_deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.25, 150.0)
        if t_alloc < 15:
            break
        init_r = diverse_init(n_best, seed + 1000 * (ri + 1))
        # Blend with best
        blend = 0.5 * best_seq + 0.5 * init_r
        blend = np.maximum(blend, 0)
        blend = blend / blend.sum()
        a_out, sc_out = optimize_at_length(blend, t_alloc, solver, f"restart{ri}_n{n_best}")
        try_update_global(a_out, f"restart{ri}")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = t_deadline - time.monotonic()
    if rem > 20:
        t_lp = min(rem * 0.4, 120.0)
        if t_lp > 10:
            a_fin, _ = nesterov_game(best_seq, t_lp, solver=solver, verbose=False)
            try_update_global(a_fin, "final_NG")

        rem = t_deadline - time.monotonic()
        if rem > 15:
            t_lse = min(rem * 0.35, 80.0)
            if t_lse > 5:
                a_fin2 = logsumexp_lbfgs(best_seq, t_lse,
                                         T_values=[3.0, 2.0, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05],
                                         verbose=True)
                try_update_global(a_fin2, "final_LSE")

        rem = t_deadline - time.monotonic()
        if rem > 10:
            t_md = min(rem * 0.3, 40.0)
            if t_md > 5:
                a_fin3, _ = mirror_descent(best_seq, t_md, n_top=10)
                try_update_global(a_fin3, "final_MD")

        rem = t_deadline - time.monotonic()
        if rem > 8:
            t_lp2 = min(rem * 0.5, 30.0)
            if t_lp2 > 5:
                a_fin4, _ = nesterov_game(best_seq, t_lp2, solver=solver, verbose=False)
                try_update_global(a_fin4, "final_NG2")

    # Final verification and return
    print(f"\n=== FINAL RESULT ===")
    if best_seq is not None:
        # Verify with actual evaluate_sequence
        final_score = evaluate_safe(best_seq)
        local_score = subscore(best_seq)
        print(f"  Local score: {local_score:.6f}")
        print(f"  Evaluated score: {final_score:.6f}")
        print(f"  Length: {len(best_seq)}")
        print(f"  Sum: {best_seq.sum():.6f}")
        print(f"  Max: {best_seq.max():.6f}")
        print(f"  Total time: {time.monotonic() - t_start:.0f}s")

        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score

        # Ensure valid output
        result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        return result

    # Fallback
    fallback = np.ones(100) / 100.0
    return [float(x) for x in fallback]
