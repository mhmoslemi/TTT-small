import numpy as np
from scipy.fft import rfft, irfft

import cvxpy as cp


def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autoconv(a)
    return float(2.0 * n * conv.max() / (s * s))


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)

        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):

    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. LP best-response alternating (direction finding)
2. Fine coordinate descent with optimal step selection (main polishing)
3. Peak-directed pair swaps (targeted redistribution)
4. Nesterov subgradient method with momentum (escape from local minima)
5. Multi-length exploration (n=512, 1024, 2048)
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autoconv(a)
    return float(2.0 * n * conv.max() / (s * s))


def build_conv_matrix(a):
    """Build T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconv(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Damped update
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.7
        else:
            alpha = max(0.2, 5.0 / (it + 5.0))
        
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
            a = a_new

        score_a = 2.0 * n * fast_autoconv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 20:
            # Random perturbation to escape
            if np.random.random() < 0.3:
                noise = np.random.standard_normal(n) * 0.005
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                s_pert = 2.0 * n * fast_autoconv(a_pert).max()
                if s_pert < best_score * 1.01:
                    a = a_pert
            stagnation = 0

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

        a = a / s

    conv = fast_autoconv(a)

    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for j in range(n):
            aj = a[j]

            k_end = j + n - 1  # Always valid since j+n-1 <= 2n-2

            unaff_max = 0.0

                    new_aff_max = q_val

                    best_new_M = new_M
                    best_delta = delta

                cur_score = 2.0 * n * M

                    best_a = a.copy()
