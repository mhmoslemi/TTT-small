                            eta_start=0.1, n_beta_steps=100, n_eta_per_beta=200):
    """Mirror descent (replicator dynamics) for the lse-smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    total_iters = n_beta_steps * n_eta_per_beta
    w_rev = None  # will be set each iteration
    
    for bi in range(n_beta_steps):
        if time.monotonic() - t0 > time_limit:
            break
        beta = beta_start * (beta_end / beta_start) ** (bi / max(1, n_beta_steps - 1))
        
        for ei in range(n_eta_per_beta):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Update eta (annealing within each beta stage)
            eta = eta_start * (0.95 ** ei)  # decay
            
            # Compute b = convolve(a, a)
            b = fast_convolve(a)
            
            # Compute w = softmax(beta * b)
            b_max = b.max()
            wb = np.exp(beta * (b - b_max))
            w = wb / wb.sum()
            
            # Compute grad = 2 * h where h[j] = convolve(a, w_rev)[2n-2-j]
            w_rev = w[::-1]
            L = n + len(w_rev) - 1  # = 3n-2
            fft_size = 1 << (L - 1).bit_length()
            A = rfft(a, fft_size)
            Wr = rfft(w_rev, fft_size)
            conv = irfft(A * Wr, fft_size)[:L]
            h = conv[n-1 : 2*n-1][::-1]
            grad = 2.0 * h
            
            # Mirror descent update
            log_a = np.log(a)
            log_Z = log_a - eta * grad
            log_Z_max = log_Z.max()
            Z = np.exp(log_Z - log_Z_max).sum() * np.exp(log_Z_max)
            # a_new = a * exp(-eta * grad) / Z
            log_a_new = log_a - eta * grad - np.log(Z)
            a = np.exp(log_a_new)
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = a0.copy() / a0.sum()
            else:
                a = a / s
            
            # Check improvement
            score = evaluate_local(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast mirror descent (replicator dynamics)
with adaptive LogSumExp temperature, Fourier-flat initializations, multi-length exploration,
LP warm-start refinement, and final L-BFGS polish. Key innovation: O(n log n) per-iteration
mirror descent replaces expensive LP solves for the bulk of optimization, enabling millions
of iterations within the time budget.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr_grad(w, a):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] (with a=0 outside [0,n-1]).
    Uses FFT: grad[j] = 2 * convolve(a, w_rev)[2n-2-j]
    Returns grad of shape (n,).
    """
    n = len(a)
    w_rev = w[::-1]
    L = n + len(w_rev) - 1
    fft_size = 1 << (L - 1).bit_length()
    A = rfft(a, fft_size)
    Wr = rfft(w_rev, fft_size)
    conv = irfft(A * Wr, fft_size)[:L]
    h = conv[n - 1: 2 * n - 1][::-1]
    return 2.0 * h


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def mirror_descent(a0, time_limit, beta_start=80.0, beta_end=2.0,
                   eta_base=0.15, n_total_iters=500000, eval_every=50):
    """
    Fast mirror descent (replicator dynamics) for lse-smoothed objective.
    Each iteration is O(n log n).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    log_a = np.log(a)

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    for it in range(n_total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature (geometric cooling)
        frac = it / max(1, n_total_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Step size decays with iteration and inversely with beta
        eta = eta_base / (1.0 + 0.01 * beta) * (1.0 / (1.0 + 0.001 * it))

        # Compute b = convolve(a, a)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:size]

        # Compute peak-weighted softmax
        b_max = b.max()
        # Clip to avoid overflow
        scaled = np.clip(beta * (b - b_max), -500, 500)
        wb = np.exp(scaled)
        wsum = wb.sum()
        if wsum < 1e-300:
            w = np.ones(size) / size
        else:
            w = wb / wsum

        # Compute gradient via FFT cross-correlation
        w_rev = w[::-1]
        L = n + size - 1
        fft_size2 = 1 << (L - 1).bit_length()
        A2 = rfft(a, fft_size2)
        Wr = rfft(w_rev, fft_size2)
        conv = irfft(A2 * Wr, fft_size2)[:L]
        h = conv[n - 1: 2 * n - 1][::-1]
        grad = 2.0 * h

        # Mirror descent (KL projection) update in log space
        log_a_new = log_a - eta * grad
        log_a_max = log_a_new.max()
        log_a_shifted = log_a_new - log_a_max
        Z_log = log_a_max + np.log(np.exp(log_a_shifted).sum())
        log_a = log_a_new - Z_log
        a = np.exp(log_a)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = a0.copy()
            a = a / a.sum()
            log_a = np.log(a)
        else:
            a = a / s
            log_a = np.log(np.maximum(a, 1e-300))

        # Periodic evaluation
        if (it + 1) % eval_every == 0:
            score = evaluate_local(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                if score < prev_best - 1e-10:
                    stall = 0
                else:
                    stall += 1
            prev_best = best_score

            # Restart with perturbation if stalled
            if stall >= 300:
                noise = rng.standard_normal(n) * 0.01
                a = a + noise
                a = np.maximum(a, 1e-300)
                a = a / a.sum()
                log_a = np.log(a)
                stall = 0

    # Final evaluation
    score = evaluate_local(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def fourier_flat_init(n, rng, spectral_power=1.0):
    """
    Generate initialization with approximately flat power spectrum.
    Take random phases in frequency domain, inverse FFT, take magnitude.
    This gives low autocorrelation by construction.
    """
    # Generate random phase spectrum
    phases = rng.uniform(0, 2 * np.pi, n)
    # Add a DC component for non-negativity
    spectrum = np.exp(1j * phases)
    # Apply spectral shaping (emphasize or de-emphasize certain bands)
    freqs = np.fft.fftfreq(n)
    shape = np.ones(n)
    # Slightly emphasize mid-frequencies for smoother result
    shape *= 1.0 + 0.3 * np.exp(-4.0 * np.abs(freqs))
    spectrum *= shape
    # Inverse FFT
    a = np.abs(np.fft.ifft(spectrum))
    # Ensure non-negative and positive sum
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng, method=0):
    """Diverse structured initializations."""
    x = np.linspace(0, 1, n)
    if method == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    elif method == 1:
        a = np.exp(-4.0 * (x - 0.5) ** 2 / (0.1 * (x - 0.5) ** 2 + 0.05))
    elif method == 2:
        a = np.sin(np.pi * x) ** 2
    elif method == 3:
        a = (1 - x) ** 1.5 + 0.1
    elif method == 4:
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif method == 5:
        # Slightly asymmetric
        a = np.exp(-3.5 * (x - 0.45) ** 2) + 0.3
    else:
        a = fourier_flat_init(n, rng)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def solve_lp_single(a_fixed, solver=None):
    """Solve one LP: given a (sum=1), find b minimizing max_k (a*b)[k]."""
    if solver is None:
        solver = _find_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build T using structured approach (avoid explicit dense matrix for large n)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # For n up to ~2000, build it directly
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    return None, None


def nesterov_lp_short(a0, time_limit, solver=None, max_iters=5000):
    """Short Nesterov LP alternation (used as refinement, not main workhorse)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, evaluate_local(a0)
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_single(y, solver)
        if b_new is None:
            break

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T_stages=None):
    """LogSumExp-smoothed L-BFGS for final polish."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02]

    n_stages = len(T_stages)
    iters_per = max(10, 3000 // n_stages)

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_loc(zz):
            aa = np.exp(zz - zz.max())
            aa = aa / aa.sum()
            bb = fast_convolve(aa)
            max_b = bb.max()
            exp_b = np.exp(bb - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse

        def jac_loc(zz):
            aa = np.exp(zz - zz.max())
            aa = aa / aa.sum()
            bb = fast_convolve(aa)
            max_b = bb.max()
            exp_b = np.exp(bb - max_b)
            ww = exp_b / exp_b.sum()
            # grad = 2 * cross_corr(ww, aa)
            w_rev = ww[::-1]
            L = n + len(w_rev) - 1
            fs = 1 << (L - 1).bit_length()
            AA = rfft(aa, fs)
            Wr = rfft(w_rev, fs)
            conv = irfft(AA * Wr, fs)[:L]
            h = conv[n - 1: 2 * n - 1][::-1]
            grad_a = 2.0 * h / T_stage
            S = np.dot(grad_a, aa)
            return aa * (grad_a - S)

        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def _find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
