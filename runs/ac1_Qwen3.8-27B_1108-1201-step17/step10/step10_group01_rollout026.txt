    a = a / s

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def compute_gradient_fft(a, w):
    """Compute grad[j] = sum_l w[j+l]*a[l] using FFT cross-correlation."""
    n = len(a)
    N = 1 << (2 * n - 1 - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    return corr[:n]

            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            lse = max_b + np.log(np.exp(b - max_b).sum())
            return (2.0 * n / T) * lse
        
        def jac(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(T * (b - max_b))
            w = exp_b / exp_b.sum()
            # Gradient via FFT
            grad = compute_gradient_fft(a, w)
            grad *= 2.0 / T
            # Adjust for softmax parameterization

            S = np.dot(grad, a)
            return a * (grad - S)
        
        try:
            maxiter = max(50, int(remaining * 2))
            res = minimize(
                fun=func, x0=z0, jac=jac, method='L-BFGS-B',

                best_a = a_cur.copy()
        except Exception as e:

    return best_a, best_score

def make_initialization(n, rng, method=0):

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
