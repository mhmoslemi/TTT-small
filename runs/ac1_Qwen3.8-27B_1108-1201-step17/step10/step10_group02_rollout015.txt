import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

# LP for best response
def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp(a_fixed):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except:
            continue
    return None

# Adam optimizer in log-space
def adam_optimize(a0, time_limit, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10).copy()
    a = a / a.sum()
    z = np.log(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        # Subgradient: average over active peaks
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Adam update in log-space
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= np.mean(z)  # Keep on simplex (in log-space)
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()
        
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        m = m  # keep momentum
        v = v
    
    return best_a, best_score

# LP alternating
def lp_alternating(a0, time_limit, n_iters=10000):

        b = b_new

        a = a_new

            best_a = a.copy()
        
        if (it+1) % 100 == 0:

    return best_a, best_score

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    last_improve = 0
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_response(a)
        a_new = solve_lp_response(b_new)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            last_improve = it
            stall_count = 0
        a = a_new
        
        if it - last_improve > 100:
            stall_count += 1
            if stall_count > 8:
                rng = np.random.default_rng(hash((it, n)) % 2**31)
                a = a * (1.0 + rng.standard_normal(n) * 0.005)
                a = np.maximum(a, 0)
                a = a / a.sum()
                last_improve = it
                stall_count = 0
        
        if verbose and (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def adam_logspace(a0, time_limit, lr=0.05, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    log_a = np.log(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    no_improve = 0
    current_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))
        
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        log_a = log_a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 300:
            current_lr *= 0.85
            no_improve = 0
        
        if no_improve > 2000:
            rng = np.random.default_rng(t * 31 + 7)
            a = a * (1 + rng.standard_normal(n) * 0.008)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            log_a = np.log(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            current_lr = lr
        
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): best={best_score:.10f} lr={current_lr:.6f}")
    
    return best_a, best_score

def mirror_descent(a0, time_limit, eta0=0.05, verbose=True):
    """Mirror descent with adaptive step size in the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))
        
        log_a = np.log(a)
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 200:
            eta *= 0.95
            no_improve = 0
        
        if no_improve > 3000:
            rng = np.random.default_rng(it * 13 + 1)
            a = a * (1 + rng.standard_normal(n) * 0.005)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            eta = eta0
        
        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc it={it+1} ({elapsed:.0f}s): best={best_score:.10f} eta={eta:.6f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, max_steps=100000, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-15:
                continue
            for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.985, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-18:
                    break
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            a_trial = a.copy()
            for pos in top_positions[:5]:
                a_trial[pos] *= 0.999
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = evaluate_local(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 800:
                    break
        else:
            stall = 0
        
        if verbose and (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.10f}")
    
    return best_a, best_score

def structured_init(n, rng, method=None):
    x = np.linspace(0, 1, n)
    if method is None:
        method = ['cosine', 'u_shape', 'gaussian', 'sine', 'poly', 'bimodal'][rng.integers(0, 6)]
    
    if method == 'cosine':
        a = np.cos(np.pi * x / 2) ** 2
    elif method == 'u_shape':
        a = 0.2 + 0.8 * (2 * x - 1) ** 2
    elif method == 'gaussian':
        a = np.exp(-4 * (x - 0.5) ** 2 / 0.15)
    elif method == 'sine':
        a = np.sin(np.pi * x) ** 0.7
    elif method == 'poly':
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0.1)
    else:  # bimodal
        a = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)
    
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def propose_candidate():
    """
    Multi-phase search: LP alternating + mirror descent + Adam + coordinate descent.
    Explores multiple n values and uses restarts to escape local minima.
    """
    rng = np.random.default_rng(777)
    deadline = time.monotonic() + 1780.0
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")
    
    # === Phase 1: Quick LP on parent to ensure it's locally optimal ===
    print(f"\n=== Phase 1: LP warm-up on parent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        a_lp, _ = lp_alternating(best_seq, t_alloc, max_iters=20000, verbose=True)
        try_update(a_lp, "lp_warmup")
    
    # === Phase 2: Mirror descent for exploration ===
    print(f"\n=== Phase 2: Mirror descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        a_md, _ = mirror_descent(best_seq, t_alloc, eta0=0.04, verbose=True)
        try_update(a_md, "mirror_descent")
    
    # === Phase 3: Adam refinement ===
    print(f"\n=== Phase 3: Adam (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 15:
        a_adam, _ = adam_logspace(best_seq, t_alloc, lr=0.03, verbose=True)
        try_update(a_adam, "adam")
    
    # === Phase 4: Main LP alternating (longest phase) ===
    print(f"\n=== Phase 4: Main LP alternating (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 20:
        a_lp2, _ = lp_alternating(best_seq, t_alloc, max_iters=100000, verbose=True)
        try_update(a_lp2, "lp_main")
    
    # === Phase 5: Try different n values ===
    print(f"\n=== Phase 5: Multi-n exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    n_options = [1500, 1800, 2000]
    t_per_n = min(80.0, rem * 0.10 / len(n_options))
    for n_try in n_options:
        if time.monotonic() > deadline:
            break
        if t_per_n < 10:
            break
        a_try = structured_init(n_try, rng)
        a_try, _ = lp_alternating(a_try, t_per_n, max_iters=5000, verbose=False)
        try_update(a_try, f"lp_n{n_try}")
    
    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord descent polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_cd, _ = coordinate_descent(best_seq, t_alloc, max_steps=200000, verbose=True)
        try_update(a_cd, "coord_descent")
    
    # === Phase 7: Final short LP ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 90.0)
    if t_alloc > 5:
        a_fin, _ = lp_alternating(best_seq, t_alloc, max_iters=30000, verbose=True)
        try_update(a_fin, "final_lp")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
