def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        T[k, m_lo:m_hi + 1] = a[k - m_hi:k - m_lo + 1][::-1]
    return T


def solve_lp_best_response(a_fixed, solver=None):

    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def exact_coord_descent(a0, time_limit, verbose=False, label=""):

    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:

            quad_idx = -1

            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,

                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

                    changes[quad_idx] = delta * (2.0 * v + delta)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def peak_shaving(a0, time_limit, rng, verbose=False):

def spectral_perturb(a, rng, strength=0.1):
    """Apply spectral smoothing then add noise."""
    n = len(a)
    ft = rfft(a)
    # Low-pass filter
    n_freq = len(ft)
    cutoff = max(2, int(n_freq * 0.7))
    ft_smooth = ft.copy()
    ft_smooth[cutoff:] *= 0.3
    a_smooth = irfft(ft_smooth, n)
    a_out = (1 - strength) * a + strength * np.abs(a_smooth)
    a_out = a_out + rng.standard_normal(n) * 0.01
    a_out = np.maximum(a_out, 0)
    s = a_out.sum()
    if s > 1e-10:
        a_out = a_out / s
    else:
        a_out = a.copy()
    return a_out


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

            phase1_best = (s_opt, a_opt)
            print(f"  n={n_test}: {s_opt:.8f}")

    if phase1_best[0] < best_score:
        best_score = phase1_best[0]
        best_seq = phase1_best[1].copy()
        print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # Phase 2: Main LP Fictitious Play
    n_work = len(best_seq)
    print(f"\n=== Phase 2: LP Fictitious Play (n={n_work}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 30:
        t_fp = min(rem * 0.55, 500.0)
        print(f"  FP budget={t_fp:.0f}s")
        a_fp, s_fp = lp_fictitious_play(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            print(f"  NEW BEST FP: {sc:.8f}")

        # Try at a different length
        for alt_n in [n_work + 128, n_work - 128]:
            if alt_n < 256 or alt_n > 2560:
                continue
            rem2 = deadline - time.monotonic() - 550
            if rem2 < 30:
                break
            t_fp2 = min(rem2 * 0.3, 150.0)
            a_init2 = interpolate_seq(best_seq, alt_n)
            a_fp2, s_fp2 = lp_fictitious_play(a_init2, t_fp2, solver=solver)
            sc2 = float(evaluate_sequence(a_fp2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_fp2.copy()
                n_work = alt_n
                print(f"  NEW BEST FP n={alt_n}: {sc2:.8f}")

    # Phase 3: Exact Coordinate Descent Polishing
    n_work = len(best_seq)
    print(f"\n=== Phase 3: ECD Polishing (n={n_work}, best={best_score:.8f}) ===")
    for rnd in range(4):
        rem = deadline - time.monotonic() - 500
        if rem < 20:
            break
        t_cd = min(rem * 0.25, 150.0)
        print(f"  ECD round {rnd + 1}, budget={t_cd:.0f}s")
        a_cd, s_cd, sw = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph3_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD: {sc:.8f}")
        elif rnd > 0:
            # Perturb and retry
            perturbed = best_seq + rng.standard_normal(n_work) * 0.015
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_cd2, s_cd2, _ = exact_coord_descent(perturbed, min(t_cd * 0.5, 60.0))
            sc2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_cd2.copy()
                print(f"  NEW BEST ECD-pert: {sc2:.8f}")

    # Phase 4: Peak Shaving
    n_work = len(best_seq)
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t_ps = min(rem * 0.4, 120.0)
        print(f"  PS budget={t_ps:.0f}s")
        a_ps, s_ps = peak_shaving(best_seq, t_ps, rng, verbose=True)
        sc = float(evaluate_sequence(a_ps.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_ps.copy()
            print(f"  NEW BEST PS: {sc:.8f}")
            # Follow up with ECD
            t_cd_ps = min(60.0, (deadline - time.monotonic() - 200))
            if t_cd_ps > 10:
                a_ps2, s_ps2, _ = exact_coord_descent(best_seq, t_cd_ps, label="after_ps")
                sc2 = float(evaluate_sequence(a_ps2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_ps2.copy()
                    print(f"  NEW BEST PS+ECD: {sc2:.8f}")

    # Phase 5: Basin hopping with spectral perturbations
    n_work = len(best_seq)
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_bh = min(rem * 0.6, 250.0)
        print(f"  BH budget={t_bh:.0f}s")
        t0_bh = time.monotonic()
        bh_iter = 0
        while time.monotonic() - t0_bh < t_bh:
            bh_iter += 1
            rem_inner = t_bh - (time.monotonic() - t0_bh)
            if rem_inner < 30:
                break
            t_alloc = min(rem_inner * 0.4, 80.0)

            # Different perturbation strategies
            pert_type = bh_iter % 5
            if pert_type == 0:
                perturbed = spectral_perturb(best_seq, rng, strength=0.15)
            elif pert_type == 1:
                # Gaussian noise
                perturbed = best_seq + rng.standard_normal(n_work) * 0.02
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
            elif pert_type == 2:
                # Asymmetric shift
                shift_amt = rng.integers(2, max(3, n_work // 20))
                shifted = np.roll(best_seq, shift_amt)
                perturbed = 0.7 * best_seq + 0.3 * shifted
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
            elif pert_type == 3:
                # Local mass transfer
                perturbed = best_seq.copy()
                n_transfer = max(3, n_work // 50)
                for _ in range(n_transfer):
                    src = rng.integers(0, n_work)
                    dst = rng.integers(0, n_work)
                    amt = perturbed[src] * 0.1 * rng.random()
                    perturbed[src] -= amt
                    perturbed[dst] += amt
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
            else:
                # Interpolation with random smooth function
                t_arr = np.linspace(0, 1, n_work)
                sm = np.exp(-((t_arr - rng.uniform(0.2, 0.8)) ** 2) / (2 * (0.1 + 0.2 * rng.random()) ** 2))
                sm = sm / sm.sum()
                perturbed = 0.8 * best_seq + 0.2 * sm
                perturbed = perturbed / perturbed.sum()

            a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh{bh_iter}")
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"    BH {bh_iter} (type={pert_type}): NEW BEST {sc:.8f}")

    # Phase 6: Final polishing
    n_work = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_fine = min(rem * 0.5, 80.0)
        a_fine, s_fine, _ = exact_coord_descent(best_seq, t_fine, verbose=True, label="final1")
        sc = float(evaluate_sequence(a_fine.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fine.copy()
            print(f"  NEW BEST final1: {sc:.8f}")

    rem = deadline - time.monotonic() - 15
    if rem > 10:
        # One more ECD pass
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.6, 40.0), label="last")
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()
            print(f"  NEW BEST last: {sc:.8f}")

    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a_fin, s_fin, _ = exact_coord_descent(best_seq, min(rem * 0.7, 20.0), label="fin")
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
