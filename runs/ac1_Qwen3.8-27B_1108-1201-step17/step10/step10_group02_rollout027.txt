def get_lp_solver():
    global _lp_solver_cached
    if '_lp_solver_cached' not in globals() or globals().get('_lp_solver_cached') is None:
        for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    globals().update(_lp_solver_cached=solver_name)
                    return solver_name
            except Exception:
                continue
        globals().update(_lp_solver_cached=cp.ECOS)
    return globals().get('_lp_solver_cached')


def build_cross_conv_matrix(a_fixed):
    """Build matrix T (size x n) such that T @ a = convolve(a_fixed, a)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            T[k, j] = a_fixed[k - j]
    return T


def successive_linearization(a0, time_limit, solver=None, max_iters=5000):
    """
    Successive linearization: iteratively solve LP relaxation.
    At each step, linearize conv(a,a) at current point and solve the LP.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    a_cv = cp.Variable(n, nonneg=True)
    t_cv = cp.Variable()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current convolution and build linearization
        conv_aa = fast_convolve(a)
        T = build_cross_conv_matrix(a)
        
        # LP: min t s.t. sum(a)=1, a>=0, 2*T@a - conv_aa <= t
        constraints = [cp.sum(a_cv) == 1,
                       2.0 * T @ a_cv - conv_aa <= t_cv]
        prob = cp.Problem(cp.Minimize(t_cv), constraints)
        
        solved = False
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
        except Exception:
            pass
        
        if not solved:
            for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        solved = True
                        break
                except Exception:
                    continue
        
        if not solved:
            break
        
        a_new = np.asarray(a_cv.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        # Damped update to avoid cycling
        alpha = 1.0 / (1.0 + 0.1 * np.sqrt(it + 1))
        a_upd = (1 - alpha) * a + alpha * a_new
        a_upd = np.maximum(a_upd, 0)
        su = a_upd.sum()
        if su < 1e-15:
            a_upd = a_new
        else:
            a_upd = a_upd / su
        
        score = evaluate_local(a_upd)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_upd.copy()
        
        a = a_upd
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SL iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def kkt_deflation(a0, time_limit, n_iters=10000):
    """
    Uses KKT flatness condition: at optimum, sum_{k in K} lambda_k * a[k-i] = C
    for all i in supp(a). We measure violation and use it as descent direction.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: peaks within 0.1% of max
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute weighted "effective convolution" w_i = sum_{k in K} (1/|K|) * a[k-i]
        w = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            w[idx] += a[k - idx]
        w /= len(active)
        
        # Compute violation: w should be constant on support
        support = a > 1e-12
        if support.sum() < 2:
            break
        w_supp = w[support]
        w_mean = w_supp.mean()
        violation = np.std(w_supp)
        
        if violation < 1e-10:
            break
        
        # Descent direction: move a to make w flatter
        # If w_i > w_mean, decrease a_i; if w_i < w_mean, increase a_i
        # Use the gradient: d(w_i)/d(a_i) relates to the structure
        # Simple approach: proportional adjustment
        direction = w_mean - w  # positive where we want to increase
        
        # Project to zero-sum (to preserve normalization)
        direction -= direction.mean()
        
        # Mirror descent step
        eta = min(0.02, 0.5 * violation / (np.std(direction) + 1e-15))
        log_a = np.log(np.maximum(a, 1e-300))
        d_shifted = direction - direction.min()
        log_a_new = log_a - eta * d_shifted + eta * d_shifted.max()
        # Actually we want to increase where direction is positive
        log_a_new = log_a + eta * direction
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        sn = a_new.sum()
        if sn < 1e-300:
            break
        a_new = a_new / sn
        
        score = evaluate_local(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                log_a_try = log_a + eta * frac * direction
                log_a_try -= log_a_try.max()
                a_try = np.exp(log_a_try)
                st = a_try.sum()
                if st < 1e-300:
                    break
                a_try = a_try / st
                score_try = evaluate_local(a_try)
                if score_try < best_score - 1e-14:
                    best_score = score_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, add noise
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.standard_normal(n) * 0.002
                a_try = a + noise
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st > 1e-15:
                    a_try = a_try / st
                    score_try = evaluate_local(a_try)
                    if score_try < best_score:
                        best_score = score_try
                        best_a = a_try.copy()
                        a = a_try
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} viol={violation:.2e}")
    
    return best_a, best_score


def two_point_descent(a0, time_limit, n_rounds=200):
    """
    Exact 2-point coordinate descent. For each pair (i,j), find the optimal
    transfer delta that minimizes max_k conv(a',a')[k].
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Precompute: for each position k, which pairs (i,j) affect it
    # We'll focus on pairs where i is at a peak-contributing position
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find peak-contributing indices
        peak_k = int(np.argmax(b))
        # Indices that contribute to the peak
        pjs = max(0, peak_k - n + 1)
        pje = min(n - 1, peak_k)
        peak_idx = np.arange(pjs, pje + 1)
        contribs = a[peak_idx] * a[peak_k - peak_idx]
        
        # Get top contributing indices
        top_k = min(20, len(peak_idx))
        top_pos = peak_idx[np.argsort(contribs)[::-1][:top_k]]
        
        improved = False
        for i in top_pos:
            if a[i] < 1e-8:
                continue
            
            # Try transferring mass from i to various other positions
            # The best targets are positions where a is small (to not create new peaks)
            # and where the transfer reduces the peak
            candidates = np.argsort(a)[:min(30, n // 10)]
            # Also try neighbors of i
            neighbors = [i - 1, i + 1, i - 2, i + 2]
            neighbors = [x for x in neighbors if 0 <= x < n and x != i]
            
            for j in list(candidates) + neighbors:
                if j == i or a[j] < -1e-12:
                    continue
                
                # Solve 1D problem: find delta that minimizes max_k conv(a',a')[k]
                # a'_i = a_i + delta, a'_j = a_j - delta
                # Valid range: delta in (-a_i, a_j)
                # conv(a',a')[k] = conv(a,a)[k] + 2*delta*(a_{k-i} - a_{k-j}) 
                #                  + delta^2*(1[k=2i] - 2*1[k=i+j] + 1[k=2j])
                
                # The critical k values are: 2i, i+j, 2j, and positions near current peaks
                critical_k = {2 * i, i + j, 2 * j}
                # Add current peak positions
                top_peaks = np.argsort(b)[::-1][:10]
                critical_k.update(top_peaks.tolist())
                critical_k = [k for k in critical_k if 0 <= k <= 2 * n - 2]
                
                if len(critical_k) == 0:
                    continue
                
                ck = np.array(critical_k)
                c_vals = b[ck]
                # d_k = a_{k-i} - a_{k-j}
                d_vals = np.zeros(len(ck))
                e_vals = np.zeros(len(ck))
                for idx_k, k in enumerate(ck):
                    ki = k - i
                    kj = k - j
                    d_vals[idx_k] = 0
                    if 0 <= ki < n:
                        d_vals[idx_k] += a[ki]
                    if 0 <= kj < n:
                        d_vals[idx_k] -= a[kj]
                    if k == 2 * i:
                        e_vals[idx_k] += 1
                    if k == i + j:
                        e_vals[idx_k] -= 2
                    if k == 2 * j:
                        e_vals[idx_k] += 1
                
                # Minimize max_k [c_k + 2*delta*d_k + delta^2*e_k]
                # over delta in (-a_i + eps, a_j - eps)
                lo = -a[i] + 1e-12
                hi = a[j] - 1e-12
                if lo >= hi:
                    continue
                
                # Use ternary search on the convex function f(delta) = max_k [...]
                # Actually f is not necessarily convex, but the max of quadratics
                # with e_k >= 0 for most terms should be well-behaved
                # Use golden section search
                
                for sub in range(40):
                    m1 = lo + 0.382 * (hi - lo)
                    m2 = lo + 0.618 * (hi - lo)
                    f1 = np.max(c_vals + 2 * m1 * d_vals + m1 ** 2 * e_vals)
                    f2 = np.max(c_vals + 2 * m2 * d_vals + m2 ** 2 * e_vals)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                
                delta = (lo + hi) / 2.0
                
                # Evaluate
                a_trial = a.copy()
                a_trial[i] += delta
                a_trial[j] -= delta
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
            
            if improved:
                continue
        
        if (rnd + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    2Pt round={rnd+1} ({elapsed:.0f}s): score={best_score:.8f}")
        
        if not improved and rnd > 50:
            # Try random pairs
            for _ in range(5):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                # Quick try with small delta
                for delta_try in [0.001, -0.001, 0.005, -0.005]:
                    if a[i] + delta_try < 0 or a[j] - delta_try < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] += delta_try
                    a_trial[j] -= delta_try
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a_trial.copy()
                        a = a_trial
                        break
    
    return best_a, best_score


def softmax_safe(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad(z, T_temp):
    """LogSumExp smoothed objective and gradient in softmax parameterization."""
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T_temp)
    w = exp_b / exp_b.sum()
    
    # Gradient of sum_k w_k * b_k with respect to a
    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Objective: (2n) * T * logsumexp(b/T) / 1 (since sum(a)=1)
    logsumexp = max_b + T_temp * np.log(exp_b.sum())
    obj = 2.0 * n * logsumexp
    
    # Convert to z-gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on log-sum-exp with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    n_stages = len(T_stages)
    time_per_stage = time_limit / n_stages
    
    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        def func(zi):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = lse_objective_and_grad(zi, T)
            return obj
        
        def jac(zi):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zi)
            _, grad = lse_objective_and_grad(zi, T)
            return grad
        
        maxiter = max(10, int(time_per_stage * 50))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': maxiter * 200,
                                    'disp': False})
            z = res.x
            a_check = softmax_safe(z)
            sc = evaluate_local(a_check)
            if si % 3 == 2:
                elapsed = time.monotonic() - t0
                print(f"    LSE T={T}: score={sc:.8f} ({elapsed:.0f}s)")
        except Exception:
            pass
    
    return softmax_safe(z)


def mirror_descent_fast(a0, time_limit, eta0=0.05, n_iters=200000):
    """Fast mirror descent with active set tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g *= 2.0 / len(active)
        
        # Update
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.mean()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        sn = a_new.sum()
        if sn < 1e-300:
            eta *= 0.9
            continue
        a_new = a_new / sn
        
        score = evaluate_local(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                eta *= 0.8
                no_improve = 0
        
        # Adaptive eta
        if (it + 1) % 500 == 0:
            eta = max(0.001, eta * 0.98)
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def nuttall_window(n):
    """Nuttall window: designed for flat autocorrelation."""
    a0 = 0.3635819
    a1 = 0.4891775
    a2 = 0.1365995
    a3 = 0.0106411
    x = np.arange(n)
    w = (a0 - a1 * np.cos(2 * np.pi * x / n) +
         a2 * np.cos(4 * np.pi * x / n) - a3 * np.cos(6 * np.pi * x / n))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def blackman_harris(n):
    """Blackman-Harris 4-term window."""
    a0 = 0.35875
    a1 = 0.48829
    a2 = 0.14128
    a3 = 0.01168
    x = np.arange(n)
    w = (a0 - a1 * np.cos(2 * np.pi * x / n) +
         a2 * np.cos(4 * np.pi * x / n) - a3 * np.cos(6 * np.pi * x / n))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def raised_cosine(n):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    w = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
    w = np.maximum(w, 1e-10)
    s = w.sum()
    return w / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining successive linearization, KKT deflation,
    log-sum-exp L-BFGS, mirror descent, and 2-point exact descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = nuttall_window(1200)
        best_seq = parent.copy()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_local(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")
    
    # === Phase 1: Successive Linearization (main engine) ===
    print(f"\n=== Phase 1: Successive Linearization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sl, _ = successive_linearization(best_seq, t_alloc, solver=solver)
        try_update(a_sl, "SL")
    
    # === Phase 2: LogSumExp L-BFGS (smooth refinement) ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 20:
