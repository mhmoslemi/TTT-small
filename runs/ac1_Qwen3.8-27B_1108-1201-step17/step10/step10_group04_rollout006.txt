import numpy as np

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Fast cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that (T @ b)[k] = conv(a_fixed, b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))

def compute_score(a):
    """Compute the objective score directly (faster than evaluate_sequence for internal use)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


_lp_solver = None

    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:

def compute_subgradient(a, n_active_frac=0.1):
    """Compute subgradient of max_k (a*a)_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def flat_spectrum_init(n, rng, method='zadoff'):
    """Generate initializations inspired by flat-spectrum sequences."""
    if method == 'zadoff':
        # Zadoff-Chu inspired: use chirp-like structure
        # For real non-negative, we use magnitude of complex Zadoff-Chu
        root = rng.integers(1, n)
        if root % 2 == 0:
            root += 1
        k = np.arange(n)
        if n % 2 == 0:
            phase = -1j * np.pi * root * k * (k + 1) / n
        else:
            phase = -1j * np.pi * root * k * k / n
        zc = np.exp(phase)
        # Take magnitude (which is 1 for ZC) - not useful directly
        # Instead, use as a mask for a smooth window
        window = 0.5 + 0.5 * np.cos(np.pi * k / (n - 1))
        a = window * (1 + 0.3 * np.abs(zc).max() * np.cos(2 * np.pi * root * k / n))
        a = np.maximum(a, 0)
        
    elif method == 'chirp':
        # Chirp-like: linear frequency sweep
        k = np.arange(n)
        beta = rng.uniform(0.5, 2.0)
        phase = np.pi * beta * (k / n) ** 2
        a = 0.5 + 0.5 * np.cos(phase)
        # Smooth window
        w = np.sin(np.pi * (k + 0.5) / n)
        a = a * w
        a = np.maximum(a, 1e-6)
        
    elif method == 'raised_cosine':
        k = np.arange(n)
        a = np.cos(np.pi * (k / (n - 1) - 0.5)) ** 2
        a = np.maximum(a, 1e-6)
        
    elif method == 'tapered':
        # Strongly tapered edges
        k = np.arange(n)
        t = k / (n - 1)
        a = np.sin(np.pi * t) ** rng.uniform(1, 3)
        a = np.maximum(a, 1e-6)
        
    elif method == 'oscillatory':
        # Multiple frequency components
        k = np.arange(n)
        a = np.ones(n)
        n_harms = rng.integers(2, 5)
        for h in range(n_harms):
            freq = rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * freq * k / n + phase)
        # Apply window
        w = 0.3 + 0.7 * np.sin(np.pi * k / n)
        a = a * w
        a = np.maximum(a, 1e-6)
        
    elif method == 'bump':
        # Gaussian bump(s)
        k = np.arange(n)
        t = k / n
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((t - center) / width) ** 2)
        # Optionally add a second bump
        if rng.random() < 0.5:
            center2 = 1.0 - center
            a += np.exp(-((t - center2) / width) ** 2)
        a = np.maximum(a, 1e-6)
        
    elif method == 'polynomial':
        # Polynomial shape
        k = np.arange(n)
        t = k / (n - 1)
        poly_order = rng.integers(2, 6)
        coeffs = rng.standard_normal(poly_order + 1)
        a = np.polyval(coeffs, 2 * t - 1)
        a = np.maximum(a, 1e-6)
        
    else:
        # Default: smooth random
        k = np.arange(n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * rng.uniform(0.5, 2) * k / n)
        a = np.maximum(a, 1e-6)
    
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=500000):
    """Nesterov-accelerated LP best-response."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step
        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        score_y = compute_score(y)

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
                prev_best = best_score
            else:
                stall_count += 1
            if stall_count >= 10:
                rng = np.random.default_rng((it * 13 + 37) % 2**31)
                noise = rng.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall_count = 0

    return best_a, best_score


def adam_logsumexp(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=15, lr_start=0.5, beta1=0.9, beta2=0.999, eps=1e-8):

            break

        t_stage0 = time.monotonic()

        for it in range(n_iters):

                break
            
            a = softmax_from_z(z)
            b = fast_convolve(a)
            
            # LogSumExp
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(exp_Tb.sum())
            
            # Gradient of LSE w.r.t. z
            w = exp_Tb / exp_Tb.sum()  # Softmax weights over conv bins
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (it + 1))

        # Anneal: reduce learning rate per stage
        lr_start *= 0.7
    
    return softmax_from_z(z), best_a, best_score
