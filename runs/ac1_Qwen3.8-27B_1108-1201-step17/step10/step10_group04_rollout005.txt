# ============ FAST CONVLUTION ============

def fast_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_cross(a, b):
    """Compute full cross-convolution of a and b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

# ============ SOFTMAX UTILITIES ============

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

# ============ LP SOLVER SETUP ============

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    # Try solvers in order of expected speed for our problem
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(100, nonneg=True)
            M_var = cp.Variable()
            prob = cp.Problem(cp.Minimize(M_var), [cp.sum(x) == 1, cp.sum(x) <= M_var])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate') and elapsed < 5.0:
                _lp_solver = solver_name
                print(f"  LP solver test: {solver_name} took {elapsed:.3f}s")
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def solve_lp_best_response(a_fixed, solver=None, verbose=False):
    """Solve LP: min M s.t. (a*b)_k <= M for all k, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try alternatives
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

# ============ NESTEROV LP (CORRECTED) ============

def nesterov_lp(a0, time_limit, solver=None, n_max_iters=500000, verbose=False):
    """
    Nesterov-accelerated LP best-response.
    
    We're solving the saddle point problem:
    min_a max_b f(a,b) where f(a,b) = max_k (a*b)_k
    over a, b in simplex.
    
    The LP gives the inner minimization (best response in b given a).
    Nesterov acceleration speeds up convergence to the saddle point.
    """
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Nesterov state
    y = a.copy()  # lookahead point
    a_prev = a.copy()
    t_k = 1.0  # Nesterov parameter
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Adaptive step
    step_scale = 1.0
    stall_count = 0
    prev_best = best_score
    best_M = None
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP at lookahead point y
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Update Nesterov parameter
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        
        # Compute new a: move toward b_new
        alpha = 1.0 / t_next
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov extrapolation for y
        y_new = a_new + ((t_k - 1.0) / t_next) * (a_new - a)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / s_y
        
        a_prev = a.copy()
        a = a_new
        y = y_new
        t_k = t_next
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        # Adaptive stall detection
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
            else:
                stall_count += 1
            
            if stall_count >= 10:
                # Perturb to escape
                rng = np.random.default_rng(it * 7 + 13)
                noise_scale = 0.01 + 0.02 * min(stall_count / 20.0, 1.0)
                noise = rng.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise * 0.5
                a = np.maximum(a, 0)
                a = a / a.sum()
                t_k = 1.0  # Reset Nesterov
                stall_count = 0
                prev_best = best_score
            
            if verbose and (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, t_k={t_k:.1f}")
    
    return best_a, best_score

# ============ LOG-SUM-EXP L-BFGS ============

def lse_objective_and_grad(z, T_val, n, cache=None):
    """Compute log-sum-exp objective and gradient."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    # Gradient
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def lbgfs_logsumexp(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=15, verbose=False):
    """L-BFGS with LogSumExp, multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        # More iterations for lower temperatures
        iters = max(30, int(500 / (si + 1)))
        
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = min(Tb.max(), 700)
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = min(Tb.max(), 700)
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                wk = w[k]
                if wk < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        T_val = stage_T
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        if verbose:
            elapsed = time.monotonic() - t0
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            print(f"    LSE stage {si+1}/{n_stages} T={stage_T:.3f} ({elapsed:.0f}s): score={sc:.6f}")
    
    return softmax_from_z(z)

# ============ SPECTRAL-DOMAIN OPTIMIZATION (NOVEL) ============

def spectral_optimize(a0, time_limit, n_iters=5000, verbose=False):
    """
    Novel: Optimize in a mixed domain.
    Parameterize a via its DFT magnitudes (with fixed phases from current a),
    then optimize the magnitudes to flatten the autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Get initial phase
    A = rfft(a)
    phase = np.angle(A)
    mag = np.abs(A)
    
    # Work with log-magnitudes for positivity
    log_mag = np.log(np.maximum(mag, 1e-10))
    
    lr = 0.05
    momentum_buf = np.zeros_like(log_mag)
    momentum_rate = 0.9
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current a from log_mag
        A_cur = np.exp(log_mag) * np.exp(1j * phase)
        a_cur = irfft(A_cur, 2*n-1)[:n]
        a_cur = np.maximum(a_cur, 0)
        s_cur = a_cur.sum()
        if s_cur < 1e-10:
            break
        a_cur = a_cur / s_cur
        
        # Compute autocorrelation
        b = fast_convolve(a_cur)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_cur.copy()
        
        # Gradient of max_b w.r.t. log_mag
        # Use active-set approximation
        threshold = max_b - 0.1 * max_b
        active = np.where(b >= threshold)[0]
        
        # Gradient of b_k w.r.t. a_j: a_{k-j} (for valid j)
        grad_a = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += a_cur[k - idx]
        grad_a /= len(active)
        
        # Chain rule: d(b)/d(log_mag) = d(b)/d(a) * d(a)/d(log_mag)
        # This is complicated, so use finite difference approximation
        # Instead, just normalize the gradient and use it as a direction
        # for log_mag updates via projection
        
        # Alternative: use the gradient in a-space to update a directly,
        # then re-extract log_mag
        a_trial = a_cur - lr * grad_a
        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial < 1e-10:
            a_trial = a_cur
            s_trial = 1.0
        a_trial = a_trial / s_trial
        
        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        
        if score_trial < score:
            # Accept: update log_mag from a_trial
            A_trial = rfft(a_trial)
            new_mag = np.abs(A_trial)
            log_mag = np.log(np.maximum(new_mag, 1e-10))
            phase = np.angle(A_trial)  # Update phase too
            lr = min(lr * 1.05, 0.5)
        else:
            lr = max(lr * 0.8, 0.001)
        
        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

# ============ PEAK DEFLATION ============

def peak_deflation(a0, time_limit, n_steps=50000, verbose=False):
    """
    Novel: Directly deflate the top peaks of the autocorrelation by
    redistributing mass from elements that contribute to those peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        score = 2.0 * n * max_b
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Move mass from highest contributor to a "safe" position
        src = idx[np.argmax(contrib)]
        
        # Find safe positions (those not contributing to top peaks)
        top_peaks = np.argsort(b)[::-1][:3]
        unsafe = set()
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            unsafe.update(range(js2, je2 + 1))
        
        safe_positions = [j for j in range(n) if j not in unsafe]
        if not safe_positions:
            safe_positions = [j for j in range(n) if j != src]
        
        dst = rng.choice(safe_positions)
        
        # Try different fractions
        improved = False
        for frac in [0.02, 0.05, 0.01, 0.08]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score_trial = 2.0 * n * fast_convolve(a_trial).max()
            if score_trial < best_score - 1e-14:
                a = a_trial
                best_score = score_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Random exploration
            if step % 10 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    a_trial = a.copy()
                    amt = a[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score_trial = 2.0 * n * fast_convolve(a_trial).max()
                        if score_trial < best_score - 1e-14:
                            a = a_trial
                            best_score = score_trial
                            best_a = a.copy()
    
    return best_a, best_score

# ============ INITIALIZATION ============

def diverse_init(n, rng, method=None):
    """Generate diverse initializations."""
    if method is None:
        method = rng.integers(0, 10)
    
    x = np.linspace(0, 1, n)
    
    if method == 0:
        a = np.ones(n)
    elif method == 1:
        a = np.cos(np.pi * x)**2
    elif method == 2:
        a = np.exp(-4 * x * (1-x))
    elif method == 3:
        # Smooth random
        a = np.abs(np.fft.irfft(np.random.default_rng(rng.integers(1000)).random(n//2+1) * np.exp(-2*np.linspace(0,1,n//2+1)**2), n))
    elif method == 4:
        # Piecewise linear
        k = rng.integers(3, 10)
        a = np.zeros(n)
        for i in range(k):
            a[int(i*n/k):int((i+1)*n/k)] = rng.uniform(0.5, 1.5)
    elif method == 5:
        # Cosine modulated
        freq = rng.uniform(1, 5)
        a = 0.5 + 0.4 * np.cos(2*np.pi*freq*x + rng.uniform(0, 2*np.pi))
    elif method == 6:
        # Gaussian bump + noise
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-(x-center)**2 / (2*width**2))
        a += 0.1 * rng.random(n)
    elif method == 7:
        # Sinc-like
        a = np.sinc(4 * (x - 0.5))
        a = np.abs(a)
    elif method == 8:
        # Two bumps
        c1, c2 = 0.3, 0.7
        a = np.exp(-(x-c1)**2/0.02) + np.exp(-(x-c2)**2/0.02)
    else:
        # Random with smoothing
        a = rng.random(n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

# ============ HELPERS ============

def compute_score(a):
    """Compute the objective value."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def evaluate_safe(a):
    """Safely evaluate via the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
