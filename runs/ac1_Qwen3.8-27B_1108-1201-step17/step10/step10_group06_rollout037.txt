"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative Weights (dual-guided) iterative method
2. LP Fictitious Play with adaptive steps
3. Exact coordinate descent with incremental convolution updates
4. Structured basin hopping for escaping local minima
5. Multi-resolution continuation with varying sequence lengths

import numpy as np
from scipy.fft import rfft, irfft
import time

_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return _LP_SOLVER

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'PDLP', 'SCS', 'GLPK']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def compute_row_sums_fft(a, lam):
    """Compute row_sums[i] = sum_m a[m] * lam[m+i] using FFT."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    L = rfft(lam, N)
    corr = irfft(np.conj(A) * L, N)
    return corr[:n]

def multiplicative_weights(a0, n_iter=500, p=6, eta=0.5, verbose=False):
    """
    Multiplicative weights algorithm for the self-convolution minimization game.
    Uses the dual variables from the LP to guide updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for t in range(n_iter):
        conv = fast_convolve(a)
        M = float(conv.max())
        if M < 1e-15:
            break

        # Soft active set: lambda_k proportional to (conv_k/M)^p
        ratios = conv / M
        lam = ratios ** p
        lam_sum = lam.sum()
        if lam_sum < 1e-15:
            lam = np.ones(2*n-1) / (2*n-1)
        else:
            lam = lam / lam_sum

        # Compute row_sums[i] = sum_k lam[k] * a[k-i]
        row_sums = compute_row_sums_fft(a, lam)
        row_sums = np.maximum(row_sums, 1e-15)

        # Multiplicative weights update
        # a_new[i] proportional to a[i] * (M / row_sums[i])^eta
        ratio = M / row_sums
        a_new = a * (ratio ** eta)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        # Damped update to maintain stability
        alpha = min(1.0, 0.3 + 0.7 * t / max(1, n_iter // 4))
        a = (1 - alpha) * a + alpha * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()

        # Check score
        sc = 2.0 * n * float(fast_convolve(a).max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Also check a_new directly
        sc_n = 2.0 * n * float(fast_convolve(a_new).max())
        if sc_n < best_score:
            best_score = sc_n
            best_a = a_new.copy()

        if verbose and t % 200 == 0:
            print(f"    MW iter {t}/{n_iter}: best={best_score:.8f}, M={M:.6f}")

    return best_a, best_score

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-16:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with incremental convolution tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = -1
            if has_quad:
                quad_idx = 2 * i - k_lo

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                       _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                       _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v + 1e-6, 5.0 / n)
            best_x = golden_section_search(_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    FCD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.25, 0.45)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_ramp(n, rng, direction=1):
    """Initialize with a ramp (linear) profile."""
    t = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = t + 1.0
    else:
        a = (n - t) + 1.0
    a = a ** rng.uniform(1.5, 3.0)
    s = a.sum()
    return a / s

def init_bimodal(n, rng):
    """Initialize with two peaks."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w1 = n * rng.uniform(0.08, 0.15)
    w2 = n * rng.uniform(0.08, 0.15)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    return a / s

def basin_hopping(a0, time_limit, rng, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 40:
            break
        t_alloc = min(rem * 0.3, 100.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Gaussian noise
            sigma = 0.02 + 0.03 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Block permutation
            blen = max(4, n // rng.integers(6, 20))
            i1 = rng.integers(0, n - 2 * blen)
            i2 = i1 + rng.integers(blen, n - blen - i1)
            temp = perturbed[i1:i1+blen].copy()
            perturbed[i1:i1+blen] = perturbed[i2:i2+blen]
            perturbed[i2:i2+blen] = temp
        elif pert_type == 2:
            # Shift mixing
            shift = rng.integers(1, max(2, n // 8))
            shifted = np.roll(a, shift)
            perturbed = 0.7 * a + 0.3 * shifted
        elif pert_type == 3:
            # Peak suppression
            conv_p = fast_convolve(a)
            peaks = np.argpartition(conv_p, -5)[-5:]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                picks = rng.choice(idx, size=min(3, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.08 * rng.random())
        else:
            # Random coordinate scaling
            n_change = max(5, n // 10)
            idx = rng.choice(n, size=n_change, replace=False)
            for ii in idx:
                perturbed[ii] *= rng.uniform(0.7, 1.3)

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh_{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: improved to {best_score:.8f}")

    return best_a, best_score

def propose_candidate():
    """
    Multi-phase optimization combining:
    1. Multiplicative weights (dual-guided) at multiple lengths
    2. LP Fictitious Play
    3. Coordinate descent polishing
    4. Basin hopping for escaping local minima
    5. Final fine polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        # Fallback: create initial candidates
        n0 = 1024
        parent = init_asymmetric(n0, rng, 0.3)
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Generated parent: n={n0}, score={best_score:.8f}")

    n_parent = len(parent)

    # ===== Phase 1: Multiplicative Weights at multiple lengths =====
    print(f"\n=== Phase 1: Multiplicative Weights (best={best_score:.8f}) ===")
    t1_start = time.monotonic()
    n_values = sorted(set([n_parent, 1024, 1536, 2048, 3072]))
    n_values = [n for n in n_values if 512 <= n <= 4096]

    for n in n_values:
        if time.monotonic() - t1_start > 300:
            break
        rem = min(60.0, deadline - time.monotonic() - 1200)
        if rem < 10:
            break

        # Multiple initializations for this length
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_asymmetric(n, rng, 0.25))
        inits.append(init_asymmetric(n, rng, 0.35))
        inits.append(init_ramp(n, rng, 1))
        inits.append(init_ramp(n, rng, -1))
        inits.append(init_bimodal(n, rng))

        best_n_score = float('inf')
        best_n_seq = None

        for j, init in enumerate(inits[:6]):
            t_left = 60.0 - (time.monotonic() - t1_start % 60)
            if time.monotonic() - t1_start > 300:
                break
            n_iter = max(200, min(1000, int(10.0 / max(0.01, (time.monotonic() - t1_start) / 300.0) * 10)))
            n_iter = max(300, min(2000, n_iter))

            a_mw, s_mw = multiplicative_weights(init, n_iter=n_iter, p=8, eta=0.4)
            sc = compute_score(a_mw)
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = a_mw.copy()

            if j == 0:
                print(f"    MW n={n} init={j}: score={sc:.8f}")

        if best_n_seq is not None:
            sc = float(evaluate_sequence(best_n_seq.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = best_n_seq.copy()
                print(f"  NEW BEST MW: {sc:.8f} (n={n})")
            elif best_n_score < best_score * 1.02:
                # Keep as alternative for later phases
                pass

    print(f"  Phase 1 done: best={best_score:.8f}")

    # ===== Phase 2: LP Fictitious Play =====
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_fp = min(rem * 0.5, 350.0)
        print(f"  FP budget={t_fp:.0f}s, n={n_work}")
        a_fp, s_fp = lp_fictitious_play(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            print(f"  NEW BEST FP: {sc:.8f}")

        # Try MW on the FP result
        if time.monotonic() < deadline - 700:
            a_mw2, s_mw2 = multiplicative_weights(a_fp, n_iter=1000, p=10, eta=0.3)
            sc2 = float(evaluate_sequence(a_mw2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_mw2.copy()
                print(f"  NEW BEST MW-after-FP: {sc2:.8f}")

        # Also try at a different length
        for alt_n in [n_work + 256, n_work - 256, n_work + 512]:
            if alt_n < 512 or alt_n > 4096:
                continue
            rem2 = deadline - time.monotonic() - 600
            if rem2 < 60:
                break
            t_fp2 = min(rem2 * 0.3, 120.0)
            a_init = interpolate_seq(best_seq, alt_n)
            a_fp2, s_fp2 = lp_fictitious_play(a_init, t_fp2, solver=solver)
            sc2 = float(evaluate_sequence(a_fp2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_fp2.copy()
                print(f"  NEW BEST FP n={alt_n}: {sc2:.8f}")

    # ===== Phase 3: Exact Coordinate Descent Polishing =====
    print(f"\n=== Phase 3: ECD Polishing (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(4):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_cd = min(rem * 0.2, 150.0)
        print(f"  ECD round {rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd, _ = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"p3_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD: {sc:.8f}")
        else:
            # Try perturbed version
            if time.monotonic() < deadline - 350:
                perturbed = best_seq + rng.standard_normal(n_work) * 0.015
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                a_cd2, s_cd2, _ = exact_coord_descent(perturbed, min(t_cd * 0.5, 60.0))
                sc2 = float(evaluate_sequence(a_cd2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_cd2.copy()
                    print(f"  NEW BEST ECD-pert: {sc2:.8f}")

    # ===== Phase 4: Basin Hopping =====
    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 300
    if rem > 60:
        t_bh = min(rem * 0.6, 250.0)
        print(f"  BH budget={t_bh:.0f}s")
        a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, verbose=True)
        sc = float(evaluate_sequence(a_bh.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bh.copy()
            print(f"  NEW BEST BH: {sc:.8f}")

    # ===== Phase 5: More MW + ECD cycles =====
    print(f"\n=== Phase 5: MW+ECD refinement (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for cycle in range(3):
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        # MW step
        t_mw = min(rem * 0.2, 60.0)
        n_iter_mw = int(t_mw * 50)
        n_iter_mw = max(200, min(2000, n_iter_mw))
        a_mw, s_mw = multiplicative_weights(best_seq, n_iter=n_iter_mw, p=12, eta=0.2)
        sc = compute_score(a_mw)
        if sc < best_score:
            best_score = sc
            best_seq = a_mw.copy()

        # ECD step
        t_cd = min(rem * 0.3, 80.0)
        a_cd, s_cd, _ = exact_coord_descent(best_seq, t_cd)
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  Cycle {cycle+1}: NEW BEST {sc:.8f}")

    # ===== Phase 6: Final Fine Polishing =====
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t_fine = min(rem * 0.5, 60.0)
        a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, verbose=True, label="final")
        sc = float(evaluate_sequence(a_fine.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fine.copy()
            print(f"  NEW BEST Fine: {sc:.8f}")

    # Last second ECD
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.7, 30.0))
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()
            print(f"  NEW BEST last: {sc:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
