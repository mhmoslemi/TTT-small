import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_gradient_k(a, k):
    """Compute d/d(a_j) conv(a,a)[k] for a single k.
    Result[j] = a[k-j] + a[j-k] (with out-of-bounds treated as 0)."""
    n = len(a)
    g = np.zeros(n)
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-(n-1) <= j <= k
    js1 = max(0, k - n + 1)
    je1 = min(n - 1, k)
    if js1 <= je1:
        g[js1:je1 + 1] += a[k - np.arange(js1, je1 + 1)]
    # a[j-k] is valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
    js2 = max(0, k)
    je2 = min(n - 1, k + n - 1)
    if js2 <= je2:
        g[js2:je2 + 1] += a[np.arange(js2, je2 + 1) - k]
    return g


def conv_gradient_all(a):
    """Compute gradients for all k values. Returns (2n-1, n) array (sparse-like)."""
    n = len(a)
    size = 2 * n - 1
    result = np.zeros((size, n))
    for k in range(size):
        result[k] = conv_gradient_k(a, k)
    return result


def find_active_set(b, top_frac=0.05, min_count=5, max_count=200):
    """Find the set of active lags (near the max of the convolution)."""
    max_b = b.max()
    threshold = max_b * (1.0 - top_frac)
    active = np.where(b >= threshold)[0]
    if len(active) < min_count:
        # Take top min_count
        k = min_count
        active = np.argsort(b)[-k:]
    if len(active) > max_count:
        k = max_count
        active = np.argsort(b)[-k:]
    return active


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, active_set, solver):
    """Given fixed a and active set, find b minimizing max_{k in active} conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix for active set only
    # T[k_idx, j] = a_fixed[k - j] if valid (for conv(a, b)[k])
    # conv(a,b)[k] = sum_j a[k-j] * b[j] where 0<=k-j<n
    T_rows = []
    for k in active_set:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = a_fixed[k - j]
        T_rows.append(row)
    T_mat = cp.Constant(np.array(T_rows))

    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                b_val = b_val / s_sum
                return b_val
        except Exception:
            continue
    return None


def solve_lp_self_update(a_current, active_set, solver):
    """Linearized self-update: minimize max of linear approximations of a^T Q_k a."""
    n = len(a_current)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Linearize: a^T Q_k a ≈ 2(Q_k a_cur)^T a - a_cur^T Q_k a_cur
    # = 2 * grad_k(a_cur) . a - conv(a_cur)[k]
    b_conv = fast_convolve(a_current)
    T_rows = []
    offset = np.zeros(len(active_set))
    for idx, k in enumerate(active_set):
        g = conv_gradient_k(a_current, k)
        T_rows.append(2.0 * g)
        offset[idx] = b_conv[k]
    T_mat = cp.Constant(np.array(T_rows))
    offset_const = cp.Constant(offset)

    constraints = [T_mat @ b_var - offset_const <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                b_val = b_val / s_sum
                return b_val
        except Exception:
            continue
    return None


def nesterov_cutting_plane(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov-accelerated cutting-plane LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    active_set = find_active_set(fast_convolve(a), top_frac=0.1, min_count=10, max_count=100)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Best response LP (fixed a, optimize b)
        b_br = solve_lp_best_response(y, active_set, solver)

        # Step 2: Self-update LP (linearize at current a, optimize a_new)
        a_su = solve_lp_self_update(a, active_set, solver)

        # Combine directions
        direction = np.zeros(n)
        count = 0
        if b_br is not None:
            direction += b_br
            count += 1
        if a_su is not None:
            direction += a_su
            count += 1
        if count == 0:
            # Fallback: use current a
            direction = a.copy()

        direction = direction / count
        direction = np.maximum(direction, 0)
        s_dir = direction.sum()
        if s_dir < 1e-15:
            direction = a.copy()
            s_dir = 1.0
        direction = direction / s_dir

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * direction
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * direction) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate
        b_a = fast_convolve(a)
        score_a = 2.0 * n * b_a.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        b_y = fast_convolve(y)
        score_y = 2.0 * n * b_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Update active set every 50 iterations
        if (it + 1) % 50 == 0:
            ref_conv = fast_convolve(a)
            active_set = find_active_set(ref_conv, top_frac=0.08, min_count=8, max_count=150)

            # Stall detection
            if (it + 1) % 200 == 0:
                if best_score < prev_best - 1e-10:
                    stall_count = 0
                else:
                    stall_count += 1
                prev_best = best_score

                if stall_count >= 8:
                    # Structured perturbation
                    x = np.linspace(0, 2 * np.pi, n)
                    freq = rng.integers(1, 8)
                    noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                    y = y + noise
                    y = np.maximum(y, 0)
                    y = y / y.sum()
                    stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    CP-iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} |A|={len(active_set)}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T, deadline):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    if time.monotonic() > deadline:
        return 1e15, np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_lbgfs_polish(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_at_T(zz):
            obj, _ = lse_obj_grad(zz, n, stage_T, deadline)
            return obj

        def jac_at_T(zz):
            _, g = lse_obj_grad(zz, n, stage_T, deadline)
            return g

        try:
            res = minimize(
                fun=func_at_T, x0=z, jac=jac_at_T, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def mirror_descent_polish(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: top peaks
        threshold = max_b * (1.0 - 0.002)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])
        if len(active) > 50:
            active = np.argsort(b)[-50:]

        # Subgradient (average over active)
        g = np.zeros(n)
        for k in active:
            g += conv_gradient_k(a, k)
        g /= len(active)

        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            eta *= 0.8
            no_improve = 0
            if eta < 0.001:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Exponential decay
        a = np.exp(-3.0 * x)
    elif choice == 1:
        # Linear ramp
        a = 1.0 - x
    elif choice == 2:
        # Quadratic
        a = (1.0 - x) ** 2
    elif choice == 3:
        # Sine window (asymmetric)
        a = np.sin(np.pi * x ** 0.7)
    elif choice == 4:
        # Gaussian (off-center)
        a = np.exp(-4.0 * (x - 0.4) ** 2)
    elif choice == 5:
        # Two bumps (asymmetric)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 6:
        # Power law
        a = (1.0 - x) ** 1.5
        a = np.maximum(a, 0.01)
    else:
        # Cosine (half)
        a = np.cos(np.pi * x / 2.0) ** 2

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses cutting-plane LP alternating with Nesterov, LogSumExp L-BFGS,
    mirror descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    print(f"Initial best: score={best_score:.8f}, n={len(best_seq)}")

    # Helper: try to update best
    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Cutting-plane LP alternating on parent ===
    print(f"\n=== Phase 1: CP-LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 540.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cp, _ = nesterov_cutting_plane(best_seq, t_alloc, solver, n_max_iters=1000000)
        local_update(a_cp, "cp_lp_1")

    # === Phase 2: LogSumExp L-BFGS polish ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 360.0)
    if t_alloc > 30:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        print(f"  {len(T_stages)} stages, budget={t_alloc:.0f}s")
        a_lse = lse_lbgfs_polish(best_seq, t_alloc, T_stages, maxiter_per_stage=8000)
        local_update(a_lse, "lse_lbgfs")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_md, _ = mirror_descent_polish(best_seq, t_alloc, n_steps=200000)
        local_update(a_md, "mirror_descent")

    # === Phase 4: Second CP-LP round ===
    print(f"\n=== Phase 4: CP-LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 270.0)
    if t_alloc > 30:
        a_cp2, _ = nesterov_cutting_plane(best_seq, t_alloc, solver, n_max_iters=500000)
        local_update(a_cp2, "cp_lp_2")

    # === Phase 5: Try n=1500 ===
    print(f"\n=== Phase 5: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n1500 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = nesterov_cutting_plane(init_1500, t_n1500 * 0.6, solver, n_max_iters=200000)
        local_update(a_1500, "cp_lp_n1500")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1500 and rem > 40:
            T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_1500b = lse_lbgfs_polish(best_seq, min(rem, 100.0), T_stages2, maxiter_per_stage=4000)
            local_update(a_1500b, "lse_n1500")

    # === Phase 6: Try n=2000 ===
    print(f"\n=== Phase 6: Try n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_n2000 = min(rem * 0.10, 150.0)
        print(f"  Budget: {t_n2000:.0f}s")
        init_2000 = interpolate_sequence(best_seq, 2000)
        a_2000, _ = nesterov_cutting_plane(init_2000, t_n2000 * 0.6, solver, n_max_iters=100000)
        local_update(a_2000, "cp_lp_n2000")

    # === Phase 7: Perturbation escape ===
    print(f"\n=== Phase 7: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.015 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.25, 50.0)
            a_opt, _ = nesterov_cutting_plane(a_pert, t_sub, solver, n_max_iters=100000)
            local_update(a_opt, f"pert_{trial}")

    # === Phase 8: Final LSE polish ===
    print(f"\n=== Phase 8: Final LSE Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        T_stages3 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_fin = lse_lbgfs_polish(best_seq, min(rem * 0.4, 40.0), T_stages3, maxiter_per_stage=5000)
        local_update(a_fin, "final_lse")

    # === Phase 9: Final mirror descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        a_final_md, _ = mirror_descent_polish(best_seq, min(rem * 0.4, 20.0), n_steps=50000)
        local_update(a_final_md, "final_md")

    # === Phase 10: Final CP-LP ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_final_cp, _ = nesterov_cutting_plane(best_seq, min(rem * 0.5, 15.0), solver, n_max_iters=30000)
        local_update(a_final_cp, "final_cp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
