def project_simplex(v):
    n = len(v)

        return np.ones(n) / n

    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, top_frac=0.05):
    n = len(a)
    b = fast_convolve_self(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - top_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


def subgradient_nesterov(a0, time_limit, n_active_frac=0.05):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.2
    no_improve = 0
    y = a.copy()
    momentum = 0.9

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        g, max_b = compute_subgradient(a, n_active_frac)
        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break

        alpha = lr
        improved = False
        for _ in range(8):
            y_new = y - alpha * g / max(g_norm, 1e-15)
            a_new = project_simplex(y_new)
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                y = y_new
                a = a_new
                no_improve = 0
                lr = min(alpha * 1.05, 2.0)
                improved = True
                break
            alpha *= 0.5
        if not improved:
            no_improve += 1
            lr = max(lr * 0.85, 0.01)
            if no_improve >= 20:
                rng = np.random.default_rng(step * 7 + 3)
                noise = rng.random(n) * 0.02
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                y = a.copy()
                no_improve = 0
                lr = 0.2
        if step % 50000 == 0 and step > 0:
            print(f"    SG: {time.monotonic()-t0:.0f}s score={best_score:.6f}")

    return best_a, best_score


def spectral_shaping_init(n, rng, n_iters=50):
    """Initialize using frequency-domain error reduction."""
    a = np.ones(n) / n
    N = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        B = fft(a_pad)
        B_mag = np.abs(B)
        B_phase = np.angle(B)

        # Target: make |B|^2 more concentrated near f=0 (to flatten autocorrelation)
        # Use a soft "tapering" of the high-frequency components
        freqs = np.arange(N)
        max_freq = n
        taper = np.ones(N)
        for f in range(1, N // 2 + 1):
            if f > max_freq // 2:
                taper[f] = 0.5
                taper[N - f] = 0.5

        B_new_mag = B_mag * taper
        # Preserve DC (sum constraint)
        B_new_mag[0] = B_mag[0]
        B_new_mag[N - 0] = B_mag[0]

        B_new = B_new_mag * np.exp(1j * B_phase)
        a_new = np.real(ifft(B_new))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s
        # Add small random perturbation to avoid stagnation
        a += rng.normal(0, 0.001, n)
        a = np.maximum(a, 0)
        a = a / a.sum()

    return a


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2 * (1 + 0.05 * np.cos(30 * np.pi * x))
    elif choice == 4:
        # m-sequence-like: use a pseudo-random binary pattern smoothed
        bits = (rng.integers(0, 2, n) == 1).astype(float)
        # Smooth with a small kernel
        kernel = np.ones(5) / 5
        a = np.convolve(bits, kernel, mode='same')
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x * 1.5)
        a = a ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_best_response_loop(a0, time_limit, solver, max_iters=50000):
    """LP-based best response with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    y = a.copy()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_fast(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        sc_a = compute_score(a)
        sc_y = compute_score(y)

        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        if it % 300 == 0 and it > 0:
            if best_score < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 5:
                rng = np.random.default_rng(it * 11 + 41)
                noise = rng.random(n) * 0.03
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            elapsed = time.monotonic() - t0
            if it % 1500 == 0:
                print(f"    LP it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def local_peak_search(a0, time_limit, n_steps=20000):
    """Local search: move mass from peak-contributing elements to others."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve_self(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        if contrib.max() < 1e-15:
            break
        src_idx = np.argmax(contrib)
        src = idx[src_idx]

        # Find destination: low-contribution position
        peak_areas = []
        for d in range(-2, 3):
            k2 = peak_k + d
            if 0 <= k2 < len(b) and b[k2] > 0.8 * b.max():
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.extend(range(js2, je2 + 1))
        peak_area_set = set(peak_areas)

        candidates = np.array([j for j in range(n) if j not in peak_area_set and a[j] < a.mean()])
        if len(candidates) == 0:
            candidates = np.array([j for j in range(n) if j != src])
        if len(candidates) == 0:
            break
        dst = rng.integers(0, len(candidates))
        dst = candidates[dst]

        improved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break

        if not improved and step % 500 == 499:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()

    return best_a, best_score


def optimize_at_length(a_init, n, time_budget, solver, label=""):
    """Multi-phase optimization at fixed length."""
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)
    t0 = time.monotonic()

    def try_update(a, src):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            print(f"    NEW BEST [{src}]: {sc:.6f} (n={n}, t={time.monotonic()-t0:.0f}s)")

    # Phase 1: Quick subgradient warm-up (15% of budget)
    rem = time_budget - (time.monotonic() - t0)
    t1 = min(rem * 0.15, 60.0)
    if t1 > 5:
        a_out, _ = subgradient_nesterov(best_a, t1, n_active_frac=0.05)
        try_update(a_out, "sg")

    # Phase 2: LP best-response (50% of budget)
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.50, 600.0)
    if t2 > 10:
        a_out, _ = lp_best_response_loop(best_a, t2, solver, max_iters=100000)
        try_update(a_out, "lp")

    # Phase 3: Another LP round with perturbation (20% of budget)
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.20, 200.0)
    if t3 > 5:
        rng = np.random.default_rng(int(time.monotonic() * 100))
        perturbed = best_a + rng.normal(0, 0.01, n)
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        a_out, _ = lp_best_response_loop(perturbed, t3, solver, max_iters=50000)
        try_update(a_out, "lp2")

    # Phase 4: Local peak search (10% of budget)
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.10, 60.0)
    if t4 > 3:
        a_out, _ = local_peak_search(best_a, t4, n_steps=15000)
        try_update(a_out, "local")

    # Phase 5: Final subgradient polish (5% of budget)
    rem = time_budget - (time.monotonic() - t0)
    t5 = min(rem * 0.05, 30.0)
    if t5 > 2:
        a_out, _ = subgradient_nesterov(best_a, t5, n_active_frac=0.02)
        try_update(a_out, "sg_final")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-n exploration with LP-based optimization."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_limit = t_start + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    solver = get_solver()
    print(f"LP solver: {solver}")

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1000
        best_seq = structured_init(n_base, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def try_update_global(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s) ***")

    # Determine n values to try
    n_candidates = []
    base = n_base
    for mult in [0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0]:
        nn = int(base * mult)
        if 100 <= nn <= 5000:
            n_candidates.append(nn)
    # Add some fixed values
    for nn in [400, 600, 800, 1000, 1200, 1600, 2000]:
        if nn not in n_candidates:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))

    # Sort by expected promise: start near base, then expand
    n_candidates.sort(key=lambda x: abs(x - base))

    print(f"\nN candidates: {n_candidates}")
    print(f"Starting with n={base}, best={best_score:.6f}")

    # Phase 1: Optimize at the base length (most important)
    print(f"\n=== Optimizing n={base} ===")
    rem = time_limit - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        init = best_seq if len(best_seq) == base else interpolate_to_length(best_seq, base)
        a_out, _ = optimize_at_length(init, base, t_alloc, solver, f"n{base}")
        try_update_global(a_out, f"opt_n{base}")

    # Phase 2: Try other n values in order of promise
    print(f"\n=== Exploring other n values (best={best_score:.6f}) ===")
    tried = {base}
    for nn in n_candidates:
        if nn in tried:
            continue
        tried.add(nn)
        rem = time_limit - time.monotonic()
        if rem < 90:
            break
        t_alloc = min(rem * 0.15, 200.0)
        if t_alloc < 30:
            break
        print(f"\n  Trying n={nn}, budget={t_alloc:.0f}s (remaining={rem:.0f}s)")

        # Initialize: interpolate from best or use structured
        if len(best_seq) == nn:
            init = best_seq.copy()
        else:
            init = interpolate_to_length(best_seq, nn)
            # Also try spectral init
            spec_init = spectral_shaping_init(nn, rng, n_iters=20)
            sc_init = compute_score(init)
            sc_spec = compute_score(spec_init)
            if sc_spec < sc_init:
                init = spec_init
                print(f"    Using spectral init: {sc_spec:.6f}")

        sc_init = compute_score(init)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init.copy()

        a_out, _ = optimize_at_length(init, nn, t_alloc, solver, f"n{nn}")
        try_update_global(a_out, f"n{nn}")

    # Phase 3: Final polish on the best sequence
    print(f"\n=== Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit - time.monotonic()
    if rem > 20:
        t_lp = min(rem * 0.4, 120.0)
        if t_lp > 10:
            a_fin, _ = lp_best_response_loop(best_seq, t_lp, solver, max_iters=50000)
            try_update_global(a_fin, "final_lp")

        rem = time_limit - time.monotonic()
        if rem > 15:
            t_loc = min(rem * 0.3, 60.0)
            if t_loc > 5:
                a_fin2, _ = local_peak_search(best_seq, t_loc, n_steps=20000)
                try_update_global(a_fin2, "final_local")

        rem = time_limit - time.monotonic()
        if rem > 10:
            t_sg = min(rem * 0.3, 30.0)
            if t_sg > 3:
                a_fin3, _ = subgradient_nesterov(best_seq, t_sg, n_active_frac=0.02)
                try_update_global(a_fin3, "final_sg")

        # One more LP with perturbation
        rem = time_limit - time.monotonic()
        if rem > 10:
            rng2 = np.random.default_rng(int(time.monotonic() * 1000) % 99999)
            pert = best_seq + rng2.normal(0, 0.005, len(best_seq))
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            t_lp2 = min(rem * 0.5, 30.0)
            if t_lp2 > 5:
                a_fin4, _ = lp_best_response_loop(pert, t_lp2, solver, max_iters=20000)
                try_update_global(a_fin4, "final_lp_pert")

    # Final verification
    if best_seq is not None:
        final_sc = compute_score(best_seq)
        print(f"\nFINAL: score={final_sc:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
