import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a, b=None):
    """Full linear convolution. If b is None, computes a*a."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def conv_atr(a, y):
    """Compute T(a)^T @ y where T(a) is the convolution matrix.
    (T(a)^T y)_i = sum_{k} y_k * a[k-i] (with a[j]=0 for j out of [0,n))
    Equivalently: (T(a)^T y)_i = sum_{j=0}^{n-1} a[j] * y[i+j]
    """
    n = len(a)
    ly = len(y)
    result = np.zeros(n)
    # (T(a)^T y)_i = sum_{j=0}^{n-1} a[j] * y[i+j] (y[k]=0 for k<0 or k>=2n-1)
    # Use FFT: this is a cross-correlation
    # flip y, convolve with a
    y_flip = y[::-1]
    fft_size = 1 << (n + ly - 1 - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    Y = rfft(y_flip, fft_size)
    conv = irfft(A * Y, fft_size)
    # conv[m] = sum_j a[j] * y_flip[m-j] = sum_j a[j] * y[ly-1-(m-j)]
    # = sum_j a[j] * y[ly-1-m+j]
    # We want result[i] = sum_j a[j] * y[i+j]
    # Set ly-1-m+j = i+j => ly-1-m = i => m = ly-1-i
    # So result[i] = conv[ly-1-i]
    for i in range(n):
        idx = ly - 1 - i
        if 0 <= idx < len(conv):
            result[i] = conv[idx]
    return result


def build_conv_matrix(a):
    """Build full convolution matrix T(a) of shape (2n-1, n)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None, verbose=False):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    # Get dual values for the inequality constraints (y*)
    y_dual = None
    try:
        dual_vals = []
        for i, con in enumerate(constraints):
            if hasattr(con, 'dual_value') and con.dual_value is not None:
                dual_vals.append(con.dual_value)
            else:
                dual_vals.append(None)
        # The first 2n-1 constraints are T@b <= M
        if len(dual_vals) >= 2 * n - 1:
            y_dual = np.array([max(0.0, float(d)) if d is not None else 0.0
                              for d in dual_vals[:2 * n - 1]])
            sy = y_dual.sum()
            if sy > 1e-15:
                y_dual = y_dual / sy
    except Exception:
        y_dual = None
    return b_val, float(M_var.value), y_dual


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=100000):
    """Nesterov-accelerated LP best-response with proper momentum."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    # Initialize both iterates
    x = a.copy()  # current iterate
    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_dual = None
    t0 = time.monotonic()
    prev_x = x.copy()
    stall = 0
    prev_best = best_score
    it = 0

    while it < max_iters:
        if time.monotonic() - t0 > time_limit:
            break

        # Compute Nesterov momentum coefficient
        t_cur = it + 1
        beta = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * t_cur * t_cur))

        # Solve LP: best response to y
        b_new, M_val, y_dual = solve_lp_best_response(y, solver)
        if b_new is None:
            it += 1
            continue

        # Update: x_new = b_new (the "primal" iterate jumps to best response)
        x_new = b_new.copy()
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx > 1e-15:
            x_new = x_new / sx

        # Nesterov extrapolation for y
        y_new = x_new + ((beta * t_cur - 1.0) / (beta * (t_cur + 1.0))) * (x_new - prev_x)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        else:
            y_new = x_new.copy()

        # Track best
        score_x = 2.0 * n * fast_convolve(x_new).max()
        score_y = 2.0 * n * fast_convolve(y_new).max()

        if score_x < best_score - 1e-12:
            best_score = score_x
            best_a = x_new.copy()
            if y_dual is not None:
                best_dual = y_dual
            stall = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y_new.copy()
            if y_dual is not None:
                best_dual = y_dual
            stall = 0
        else:
            stall += 1

        # Periodic progress check
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            prev_best = best_score
            elapsed = time.monotonic() - t0
            print(f"    NLP it={it+1} ({elapsed:.0f}s): best={best_score:.6f}, "
                  f"score_x={score_x:.6f}, M_lp={M_val:.6f}")

        # Stall handling: add perturbation
        if stall >= 40:
            rng = np.random.default_rng(it * 31 + 17)
            noise = rng.random(n) * 0.02
            y_new = y_new + noise
            y_new = np.maximum(y_new, 0)
            y_new = y_new / y_new.sum()
            stall = 0

        prev_x = x_new
        x = x_new
        y = y_new
        it += 1

    return best_a, best_score, best_dual


def subgradient_mirror_descent(a0, time_limit, max_iters=200000):
    """Mirror descent with subgradient from top active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for step in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient from top peaks
        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        # Use top K active set
        sorted_idx = np.argsort(b)[::-1]
        K = min(max(3, n // 10), 50)
        active = sorted_idx[:K]
        # Weight by how close to max
        weights = np.maximum(b[active] - 0.8 * max_b, 0.01)
        weights = weights / weights.sum()

        # Subgradient: g_i = sum_k w_k * 2*a[k-i] (with bounds)
        g = np.zeros(n)
        for k, wk in zip(active, weights):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * wk * a[k - idx]

        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break

        # Projected mirror descent step
        for bt in range(12):
            z_new = z - lr * g
            a_new = softmax_from_z(z_new)
            b_new = fast_convolve(a_new)
            score_new = 2.0 * n * b_new.max()

            if score_new <= best_score + 1e-10:
                if score_new < best_score - 1e-12:
                    best_score = score_new
                    best_a = a_new.copy()
                    no_improve = 0
                a = a_new
                z = z_new
                lr = min(lr * 1.05, 5.0)
                break
            lr *= 0.5
        else:
            no_improve += 1
            lr = max(lr * 0.7, 0.01)
            if no_improve >= 50:
                rng = np.random.default_rng(step * 7 + 3)
                noise = rng.random(n) * 0.01
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                no_improve = 0
                lr = 0.3

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MIR it={step+1} ({elapsed:.0f}s): best={best_score:.6f}, lr={lr:.4f}")

    return best_a, best_score


def peak_shaving_local_search(a0, time_limit, max_iters=50000):
    """Local search that specifically targets reducing the top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_b = fast_convolve(a)
    best_score = 2.0 * n * best_b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    for step in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Get top peaks
        sorted_idx = np.argsort(b)[::-1]
        top_k = sorted_idx[:min(8, n)]
        top_val = b[top_k[0]]
        second_val = b[top_k[1]] if len(top_k) > 1 else top_val * 0.9
        gap = top_val - second_val

        improved = False

        # Strategy 1: Reduce the dominant peak by moving mass from its contributors
        for k in top_k[:3]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the biggest contributor
            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find a good destination: position that doesn't contribute to top peaks
            peak_coverage = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_coverage.update(range(js2, je2 + 1))

            # Try moving to uncovered positions
            candidates = [j for j in range(n) if j not in peak_coverage]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            for frac in [0.005, 0.01, 0.02, 0.05]:
                dst = rng.choice(candidates)
                amt = a[src] * frac
                if amt < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_tr = a_trial.sum()
                if s_tr < 0.01:
                    continue
                a_trial = a_trial / s_tr
                b_trial = fast_convolve(a_trial)
                score = 2.0 * n * b_trial.max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_a = a.copy()
                    best_score = score
                    best_b = b_trial
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Symmetrize if beneficial
        if not improved and step % 50 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 0)
            a_sym = a_sym / a_sym.sum()
            b_sym = fast_convolve(a_sym)
            score_sym = 2.0 * n * b_sym.max()
            if score_sym < best_score - 1e-8:
                a = a_sym
                best_a = a.copy()
                best_score = score_sym
                best_b = b_sym
                no_improve = 0
                improved = True

        if not improved:
            no_improve += 1
            # Random exploration
            if no_improve % 100 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_tr = a_trial.sum()
                    if s_tr > 0.01:
                        a_trial = a_trial / s_tr
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_a = a.copy()
                            best_score = score
                            no_improve = 0

            if no_improve > 5000:
                break

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PEA it={step+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=10):
    """L-BFGS with LogSumExp approximation of the max."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_obj(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 500:
            max_Tb = 500
        exp_Tb = np.exp(Tb - max_Tb)
        lse = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * lse

    def lse_jac(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 500:
            max_Tb = 500
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # grad of lse wrt a: sum_k w_k * grad_a(b_k) = sum_k w_k * 2*a[k-i]
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        # Chain rule through softmax: grad_z = a * (grad_a - <grad_a, a>)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(zz):
            return lse_obj(zz, stage_T)

        def jac(zz):
            return lse_jac(zz, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': 100, 'ftol': 1e-16,
                         'gtol': 1e-10, 'maxfun': 5000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 1:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif choice == 4:
        # Triangular (peak in middle)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif choice == 5:
        # Two peaks
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    elif choice == 6:
        # Edge concentrated
        a = np.exp(-10 * x) + np.exp(-10 * (1 - x))
        a = a ** 1.5
    elif choice == 7:
        # Oscillating
        a = 0.5 + 0.3 * np.cos(20 * np.pi * x) + 0.2 * np.cos(40 * np.pi * x)
    elif choice == 8:
        # Broad Gaussian
        a = np.exp(-5 * (x - 0.5) ** 2)
    else:
        # Uniform with noise
        a = 1.0 + 0.3 * np.sin(10 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def optimize_at_length(a_init, n, time_budget, solver, label=""):
    """Run multi-phase optimization at a fixed length."""
    best_a = np.maximum(a_init.copy(), 0).astype(np.float64)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    def try_update(a, src_label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = 2.0 * n * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  UPDATE [{src_label}]: {sc:.6f} (n={n}, t={elapsed:.0f}s)")

    # Phase A: Nesterov LP (main workhorse) - 50% of budget
    rem = time_budget - (time.monotonic() - t0)
    t_a = rem * 0.50
    if t_a > 15:
        print(f"  Phase A: NLP, budget={t_a:.0f}s")
        a_out, _, _ = nesterov_lp_optimize(best_a, t_a, solver=solver)
        try_update(a_out, f"NLP_{label}")

    # Phase B: Subgradient mirror descent - 20%
    rem = time_budget - (time.monotonic() - t0)
    t_b = rem * 0.20
    if t_b > 10:
        print(f"  Phase B: MDM, budget={t_b:.0f}s")
        a_out, _ = subgradient_mirror_descent(best_a, t_b)
        try_update(a_out, f"MDM_{label}")

    # Phase C: Peak shaving - 15%
    rem = time_budget - (time.monotonic() - t0)
    t_c = rem * 0.15
    if t_c > 5:
        print(f"  Phase C: PeakShave, budget={t_c:.0f}s")
        a_out, _ = peak_shaving_local_search(best_a, t_c)
        try_update(a_out, f"PSC_{label}")

    # Phase D: L-BFGS LogSumExp - 10%
    rem = time_budget - (time.monotonic() - t0)
    t_d = rem * 0.10
    if t_d > 5:
        print(f"  Phase D: LSE, budget={t_d:.0f}s")
        a_out = lbgfs_logsumexp(best_a, t_d, T_start=20.0, T_end=0.1, n_stages=8)
        try_update(a_out, f"LSE_{label}")

    # Phase E: Final NLP polish - 5%
    rem = time_budget - (time.monotonic() - t0)
    t_e = rem * 0.05
    if t_e > 5:
        print(f"  Phase E: NLP polish, budget={t_e:.0f}s")
        a_out, _, _ = nesterov_lp_optimize(best_a, t_e, solver=solver)
        try_update(a_out, f"NLPfin_{label}")

    return best_a, best_score

        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score

    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
