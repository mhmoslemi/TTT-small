import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def smooth_objective(z, n, T, fft_cache):
    """Correct LogSumExp smoothed objective: 2n * (1/T) * log(sum exp(T*conv(a,a)))."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = T * (b - max_b)
    log_sum = np.log(np.exp(scaled).sum())
    smooth_max = max_b + log_sum / T
    return 2.0 * n * smooth_max

def smooth_gradient(z, n, T):
    """Gradient of the smoothed objective with respect to z (log-space)."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = T * (b - max_b)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # Compute G[i] = sum_k w[k] * a[k-i] using FFT
    N = 1 << (3 * n - 2 - 1).bit_length()
    w_fft = np.zeros(N, dtype=np.complex128)
    w_fft[:len(w)] = w
    a_fft = np.zeros(N, dtype=np.complex128)
    a_fft[:n] = a
    G_circ = np.real(np.fft.ifft(w_fft * np.conj(a_fft)))[:n]
    
    # Gradient w.r.t. z using softmax Jacobian
    S = np.dot(G_circ, a)
    grad_z = 4.0 * n * a * (G_circ - S)
    return grad_z
