L-BFGS with multi-stage cooling, parametric initialization sweeps,

Key innovations: FFT-based O(n log n) gradient computation, adaptive
temperature schedule, exponential-family parameter sweep for init,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fft_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, auto-convolution."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fft_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_objective_and_grad_z(z, n, T):
    """Compute LSE-smoothed objective and gradient w.r.t. log-parameters z."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fft_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    lse = max_b + T * np.log(exp_b.sum())
    obj = 2.0 * n * lse

    # Gradient computation via FFT
    # grad_a[i] = sum_k w[k] * (2*a[k-i] - a[i]*[k==2i])
    # First term: cross-correlation corr[i] = sum_k w[k]*a[k-i] = sum_j a[j]*w[j+i]
    a_rev = a[::-1]
    w_len = len(w)
    conv_size = n + w_len - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)[:conv_size]
    corr = conv[n - 1:2 * n - 1]

    grad_a = 2.0 * corr
    diag_idx = 2 * np.arange(n)
    valid_diag = diag_idx < w_len
    grad_a[valid_diag] -= a[valid_diag] * w[diag_idx[valid_diag]]

    # Scale by 2n*T for the full objective gradient w.r.t. a
    grad_obj = 2.0 * n * T * grad_a

    # Transform from a-space to z-space (softmax parameterization)
    S = np.dot(grad_obj, a)
    grad_z = a * (grad_obj - S)

    return obj, grad_z

def peak_coord_descent(a0, time_limit, verbose=True):

def nesterov_subgradient(a0, time_limit, verbose=True):
    """Nesterov-accelerated subgradient on active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fft_convolve(y)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Average subgradient over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Nesterov update
        t_val = (1 + np.sqrt(1 + 4 * step * step)) / 2
        beta = (1 - t_val) / (step + 1)

        a_new = y - lr * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = a_new
        score = evaluate_local(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-8, lr)

        if verbose and step % 10000 == 0:
            print(f"    SubGrad ({time.monotonic()-t0:.0f}s): score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def exponential_init(n, gamma, rng=None):
    """Create exponential decay initialization: a[i] = exp(-gamma*i/n) normalized."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * i / n)
    if rng is not None:
        a *= (1 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def power_law_init(n, p, rng=None):

    x = i / max(n - 1, 1)
    a = (1 - x) ** p
    if rng is not None:
        a *= (1 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def two_hump_init(n, center, width, ratio, rng=None):

    candidates = []

    # Exponential decays
    for gamma in [0.5, 0.8, 1.0, 1.2, 1.4, 1.5, 1.8, 2.0, 2.5, 3.0]:
        candidates.append(exponential_init(n, gamma, rng))

    # Power laws
    for p in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(power_law_init(n, p, rng))

    # Stretched exponentials
    for gamma_val in [1.0, 2.0, 3.0, 4.0]:
        for p_val in [1.5, 2.0, 2.5, 3.0]:
            candidates.append(stretched_exp_init(n, gamma_val, p_val, rng))

    # Two-hump
    for center in [0.3, 0.5, 0.7]:
        for ratio in [0.05, 0.1, 0.2]:
            candidates.append(two_hump_init(n, center, 0.15, ratio, rng))

    # Random
    for _ in range(5):
        a = np.abs(rng.standard_normal(n))
        a = a / a.sum()
        candidates.append(a)

    if verbose:
        print(f"  Parametric sweep: {len(candidates)} candidates")

    for ci, cand in enumerate(candidates):
        if time.monotonic() - t0 > budget_s:
            break
        score = evaluate_local(cand)
        if score < best_score:
            best_score = score
            best_a = cand.copy()

    if verbose:
        print(f"  Best parametric: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multistart_lbfgs(a0, time_limit, n_starts=3, verbose=True):
    """Run LSE L-BFGS from multiple perturbed starts."""
    n = len(a0)
    rng = np.random.default_rng(777)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = evaluate_local(a0)

    time_per_start = time_limit / n_starts

    for si in range(n_starts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        # Create perturbation
        if si == 0:
            a_start = a0.copy()
        elif si == 1:
            x = np.linspace(0, 2 * np.pi, n)
            pert = 0.02 * np.cos(3 * x) + 0.01 * np.cos(7 * x)
            a_start = a0 + pert
        else:
            pert = 0.015 * rng.standard_normal(n)
            a_start = a0 + pert

        a_start = np.maximum(a_start, 0)
        ss = a_start.sum()
        if ss < 1e-15:
            continue
        a_start = a_start / ss

        T_stages = [5.0, 3.0, 2.0, 1.4, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]
        a_res, score = lse_lbfgs_optimize(a_start, time_per_start - (time.monotonic() - t0),
                                           T_stages=T_stages, verbose=False)
        if score < best_score:
            best_score = score
            best_a = a_res.copy()
            if verbose:
                print(f"    MultiStart {si} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Parametric sweep for good initialization
    2. LSE L-BFGS main optimization
    3. Peak-directed coordinate descent
    4. Nesterov subgradient
    5. Multi-start LSE L-BFGS
    6. Try different lengths
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01 or not np.isfinite(s):
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_best = len(best_seq) if best_seq is not None else 1200

    # === Phase 0: Parametric sweep (quick) ===
    print(f"\n=== Phase 0: Parametric sweep, n={n_best} ===")
    rem = deadline - time.monotonic()
    t0_sweep = min(rem * 0.05, 30.0)
    if t0_sweep > 5:
        a_param, s_param = parametric_sweep(n_best, t0_sweep, verbose=True)
        if a_param is not None and s_param < best_score:
            try_update(a_param, "parametric")
        elif a_param is not None:
            # Even if not better, use as alternative start later
            pass
    else:
        a_param = None

    # === Phase 1: Main LSE L-BFGS ===
    print(f"\n=== Phase 1: LSE L-BFGS main, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        T_stages = [10.0, 6.0, 4.0, 3.0, 2.2, 1.6, 1.2, 0.9, 0.65, 0.45,
                    0.32, 0.22, 0.15, 0.1, 0.07, 0.045, 0.03, 0.02, 0.012, 0.007]
        a_res, _ = lse_lbfgs_optimize(best_seq, t1, T_stages=T_stages, verbose=True)
        try_update(a_res, "lse_main")

    # === Phase 2: Peak-directed coordinate descent ===
    print(f"\n=== Phase 2: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 15:
        a_cd, _ = peak_coord_descent(best_seq, t2, verbose=True)
        try_update(a_cd, "peak_cd")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov SubGrad, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 150.0)
    if t3 > 15:
        a_sg, _ = nesterov_subgradient(best_seq, t3, verbose=True)
        try_update(a_sg, "subgrad")

    # === Phase 4: Multi-start LSE L-BFGS ===
    print(f"\n=== Phase 4: Multi-start, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 280.0)
    if t4 > 30:
        a_ms, _ = multistart_lbfgs(best_seq, t4, n_starts=3, verbose=True)
        try_update(a_ms, "multistart")

    # === Phase 5: Try different length ===
    print(f"\n=== Phase 5: Alt length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        # Try slightly larger length
        alt_n = int(n_cur * 1.15)
        t_alt = min(rem * 0.4, 200.0)
        if t_alt > 30:
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            T_stages_alt = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02]
            a_alt, _ = lse_lbfgs_optimize(init_alt, t_alt * 0.7, T_stages=T_stages_alt, verbose=True)
            try_update(a_alt, f"alt_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 20:
                a_alt_cd, _ = peak_coord_descent(a_alt, min(rem * 0.5, 50.0), verbose=False)
                try_update(a_alt_cd, f"alt_n{alt_n}_cd")

        # Also try slightly smaller length
        rem = deadline - time.monotonic()
        if rem > 80:
            alt_n2 = int(n_cur * 0.85)
            t_alt2 = min(rem * 0.35, 150.0)
            if t_alt2 > 20:
                print(f"  Trying n={alt_n2}, budget={t_alt2:.0f}s")
                init_alt2 = interpolate_sequence(best_seq, alt_n2)
                T_stages_alt2 = [4.0, 2.5, 1.5, 1.0, 0.7, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]
                a_alt2, _ = lse_lbfgs_optimize(init_alt2, t_alt2 * 0.7, T_stages=T_stages_alt2, verbose=True)
                try_update(a_alt2, f"alt_n{alt_n2}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.4, 80.0)
        if t_fin > 10:
            T_stages_fin = [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]
            a_fin, _ = lse_lbfgs_optimize(best_seq, t_fin, T_stages=T_stages_fin, verbose=True)
            try_update(a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_cd2, _ = peak_coord_descent(best_seq, min(rem * 0.5, 60.0), verbose=True)
        try_update(a_cd2, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_sg2, _ = nesterov_subgradient(best_seq, min(rem * 0.6, 40.0), verbose=False)
        try_update(a_sg2, "final_sg")

    # === Phase 7: One more LSE pass if time remains ===
    rem = deadline - time.monotonic()
    if rem > 15:
        T_stages_last = [0.5, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01]
        a_last, _ = lse_lbfgs_optimize(best_seq, min(rem * 0.8, 40.0),
                                        T_stages=T_stages_last, verbose=True)
        try_update(a_last, "last_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
