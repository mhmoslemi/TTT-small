def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    #---Security Checks---#
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    
    # Convert all elements to float and apply defensive clipping
    safe_sequence = [float(x) for x in sequence]
    for x in safe_sequence:
        if isinstance(x, bool):
            return np.inf
        if not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    
    safe_sequence = [max(0, x) for x in safe_sequence]
    safe_sequence = [min(1000.0, x) for x in safe_sequence]
    
    n = len(safe_sequence)
    b_sequence = np.convolve(safe_sequence, safe_sequence, mode="full")
    max_b = max(b_sequence) if n > 0 else np.inf
    
    sum_a = sum(safe_sequence)
    if sum_a < 0.01:
        return np.inf
    
    return float(2 * n * max_b / (sum_a ** 2))

def structured_init(n: int, rng=None) -> np.ndarray:
    """
    Generates diverse structured initializations.
    n: target sequence length
    rng: random number generator for reproducibility
    Returns: numpy array of coefficients summing to 1.0
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    # Generate one of seven standard initialization profiles
    choice = rng.integers(0, 7)
    
    if choice == 0:  # Cosine decay profile
        x = np.linspace(0, np.pi * 2, n)
        return 0.2 * np.cos(x) ** 2 + 0.8
    elif choice == 1:  # Gaussian profile
