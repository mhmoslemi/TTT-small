```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast cyclic coordinate descent
with exact 1D subproblem solves, multi-peak awareness, adaptive steps, and
random restarts. Key innovation: O(n) per-coordinate update via incremental
convolution modification enables 100x more iterations than FFT-based methods.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def update_b_for_coord(b, a, j, new_val, old_val, n):
    """Compute new b after changing a[j] from old_val to new_val. O(n) operation."""
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return b
    delta_sq = new_val * new_val - old_val * old_val

    new_b = b.copy()

    # G_k = 2*a[k-j] for k in range where 0 <= k-j < n, k != 2j
    lo = max(0, j)
    hi = min(2 * n - 2, j + n - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        G = 2.0 * a[ks - j]
        # Zero out G at k=2j
        if lo <= 2 * j <= hi:
            G[2 * j - lo] = 0.0
        new_b[lo:hi + 1] += delta * G

    # H_{2j} = 1: add delta_sq at position 2j
    if 0 <= 2 * j <= 2 * n - 2:
        new_b[2 * j] += delta_sq

    return new_b


def coord_descent_fast(a, b, S, n, time_limit, rng, verbose=False):
    """
    Fast cyclic coordinate descent with incremental convolution updates.
    Tries multiple candidate values per coordinate and accepts improvements.
    """
    t0 = time.monotonic()
    best_score = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()
    best_b = b.copy()
    best_S = S

    stall = 0
    sweep_count = 0
    order = rng.permutation(n)

    while time.monotonic() - t0 < time_limit:
        sweep_count += 1
        improved_this_sweep = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            # Generate candidate values
            # Focus on reducing coordinates that contribute to peaks
            candidates = [old_val]
            # Reduction candidates
            for frac in [0.1, 0.3, 0.5, 0.7, 0.85, 0.95]:
                candidates.append(old_val * frac)
            # Small increase candidates
            for frac in [1.1, 1.3, 1.5]:
                candidates.append(old_val * frac)

            best_x = old_val
            best_local_score = best_score
            best_local_b = b

            for x in candidates:
                if x <= 0 and old_val > 0:
                    x = 0.0
                elif x <= 1e-15:
                    continue
                if x == old_val:
                    continue
                new_S = S + (x - old_val)
                if new_S < 0.01:
                    continue
                new_b = update_b_for_coord(b, a, j, x, old_val, n)
                score = 2.0 * n * new_b.max() / (new_S * new_S)
                if score < best_local_score - 1e-12:
                    best_local_score = score
                    best_x = x
                    best_local_b = new_b

            if best_x != old_val:
                S = S + (best_x - old_val)
                a[j] = best_x
                b = best_local_b
                improved_this_sweep = True
                if best_local_score < best_score - 1e-12:
                    best_score = best_local_score
                    best_a = a.copy()
                    best_b = b.copy()
                    best_S = S

        if improved_this_sweep:
            stall = 0
        else:
            stall += 1

        # Reshuffle order periodically
        if sweep_count % 10 == 0:
            order = rng.permutation(n)

        # If stuck, add noise and try again
        if stall > 50:
            noise_scale = min(0.05, 0.01 * np.sqrt(stall / 50))
            noise = rng.standard_normal(n) * noise_scale
            a_perturbed = a + noise
            a_perturbed = np.maximum(a_perturbed, 0)
            ps = a_perturbed.sum()
            if ps > 0.01:
                a_perturbed = a_perturbed / (ps / S)
                b_new = fast_convolve(a_perturbed)
                score_new = 2.0 * n * b_new.max() / S**2
                if score_new < best_score * 1.02:  # Accept if not too bad
                    a = a_perturbed
                    b = b_new
                else:
                    a = best_a.copy()
                    b = best_b.copy()
                    S = best_S
            stall = 0

        if verbose and sweep_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep_count} ({elapsed:.1f}s): score={best_score:.8f} stall={stall}")

    return best_a, best_score, best_b, best_S

def peak_directed_cd(a, b, S, n, time_limit, rng, verbose=False):

    Coordinate descent that specifically targets peak-contributing coordinates.
    Focuses optimization effort on the coordinates that contribute most to
    the maximum autocorrelation value.
    """
    t0 = time.monotonic()
    best_score = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()
    best_b = b.copy()
    best_S = S

    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Recompute b if needed
        max_idx = np.argmax(b)
        max_val = b[max_idx]

        # Find coordinates contributing to the peak
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        peak_coords = np.arange(js, je + 1)

        # Concentration: a[j] * a[max_idx - j]
        contribs = a[peak_coords] * a[max_idx - peak_coords]
        total_contrib = contribs.sum()

        if total_contrib < 1e-15:
            break

        # Sort by contribution (most important first)
        sorted_idx = peak_coords[np.argsort(contribs)[::-1]]

        # Try to reduce top contributors
        improved = False
        for j in sorted_idx[:20]:  # Top 20 contributors
            old_val = a[j]
            if old_val < 1e-12:
                continue

            # Try reducing this coordinate
            for frac in [0.3, 0.5, 0.7, 0.85, 0.95]:
                x = old_val * frac
                if x <= 1e-15:
                    continue
                new_S = S + (x - old_val)
                if new_S < 0.01:
                    continue
                new_b = update_b_for_coord(b, a, j, x, old_val, n)
                score = 2.0 * n * new_b.max() / (new_S * new_S)
                if score < best_score - 1e-12:
                    S = new_S
                    a[j] = x
                    b = new_b
                    best_score = score
                    best_a = a.copy()
                    best_b = b.copy()
                    best_S = S
                    improved = True
                    break

        if not improved:
            stal1 = stall + 1
            stall = stal1
            # Try a random coordinate
            j = rng.integers(0, n)
            old_val = a[j]
            if old_val > 1e-12:
                for frac in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
                    x = old_val * frac
                    if x <= 1e-15:
                        continue
                    new_S = S + (x - old_val)
                    if new_S < 0.01:
                        continue
                    new_b = update_b_for_coord(b, a, j, x, old_val, n)
                    score = 2.0 * n * new_b.max() / (new_S * new_S)
                    if score < best_score - 1e-12:
                        S = new_S
                        a[j] = x
                        b = new_b
                        best_score = score
                        best_a = a.copy()
                        best_b = b.copy()
                        best_S = S
                        improved = True
                        break

        if improved:
            stall = 0

        if stall > 300:
            # Perturbation to escape
            noise_scale = min(0.03, 0.005 * np.sqrt(stall / 100))
            a_pert = a + rng.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps > 0.01:
                a_pert = a_pert * (S / ps)
                b_pert = fast_convolve(a_pert)
                score_pert = 2.0 * n * b_pert.max() / S**2
                if score_pert < best_score * 1.01:
                    a = a_pert
                    b = b_pert
                else:
                    a = best_a.copy()
                    b = best_b.copy()
                    S = best_S
            stall = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD it={it} ({elapsed:.1f}s): score={best_score:.10f} stall={stall}")

    return best_a, best_score, best_b, best_S


def block_cd(a, b, S, n, time_limit, rng, block_size=3, verbose=False):
    """
    Block coordinate descent: update small blocks of coordinates simultaneously.
    Uses a local quadratic approximation for the block subproblem.
    """
    t0 = time.monotonic()
    best_score = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()
    best_b = b.copy()
    best_S = S

    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Select a random block
        start = rng.integers(0, n - block_size + 1)
        block = np.arange(start, start + block_size)

        # Try perturbation of the block
        old_block = a[block].copy()
        old_S = S

        # Generate random perturbation
        for trial in range(10):
            scale = 0.1 * (0.5 ** trial)
            pert = rng.standard_normal(block_size) * scale
            new_block = old_block + pert
            new_block = np.maximum(new_block, 0)

            delta = new_block - old_block
            new_S = S + delta.sum()
            if new_S < 0.01:
                continue

            # Update b for each coordinate in the block
            new_b = b.copy()
            for idx, j in enumerate(block):
                if abs(delta[idx]) > 1e-15:
                    new_b = update_b_for_coord(new_b, a, j, new_block[idx], old_block[idx], n)

            score = 2.0 * n * new_b.max() / (new_S * new_S)
            if score < best_score - 1e-12:
                a[block] = new_block
                S = new_S
                b = new_b
                best_score = score
                best_a = a.copy()
                best_b = b.copy()
                best_S = S
                stall = 0
                break
        else:
            stall += 1

        if stall > 200:
            break

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BlockCD it={it} ({elapsed:.1f}s): score={best_score:.10f}")

    return best_a, best_score, best_b, best_S


def try_multiple_lengths(a_base, scores_by_n, time_per_n, rng):
    """Try different sequence lengths by interpolation and quick optimization."""
    base_n = len(a_base)
    candidate_ns = [base_n, base_n - 200, base_n + 200, base_n + 400, base_n - 400]
    candidate_ns = sorted(set([n for n in candidate_ns if 400 <= n <= 3000]))

    t0 = time.monotonic()
    best_overall = (a_base, scores_by_n.get(base_n, evaluate_local(a_base)))

    for n_try in candidate_ns:
        if time.monotonic() - t0 > time_per_n * len(candidate_ns):
            break
        if n_try in scores_by_n:
            continue

        a_try = interpolate_sequence(a_base, n_try)
        score_try = evaluate_local(a_try)
        scores_by_n[n_try] = score_try

        # Quick coordinate descent on this length
        b_try = fast_convolve(a_try)
        S_try = a_try.sum()
        if S_try < 0.01:
            continue

        t_remaining = max(5.0, (time_per_n * len(candidate_ns)) - (time.monotonic() - t0))
        a_opt, sc_opt, _, _ = coord_descent_fast(
            a_try, b_try, S_try, n_try, min(t_remaining * 0.4, 120.0), rng
        )

        if sc_opt < best_overall[1]:
            best_overall = (a_opt, sc_opt)
            a_base = a_opt
            base_n = n_try

    return best_overall[0], best_overall[1], scores_by_n


def structured_init(n, rng, style=None):
    """Create a structured initial sequence."""
    if style is None:
        style = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if style == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 2
    elif style == 2:
        a = np.exp(-4 * x * (1 - x))
    elif style == 3:
        # Asymmetric: mass shifted
        a = np.exp(-3 * (x - 0.3) ** 2 / 0.2)
    elif style == 4:
        # Bimodal
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.7 * np.exp(-8 * (x - 0.7) ** 2)
    elif style == 5:
        # Power law decay
        a = (1.0 - x) ** 1.5
    else:
        a = np.exp(-3 * x * (1 - x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining fast coordinate descent, peak-directed
    search, block updates, and random restarts at multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    scores_by_n = {}

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_local(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
            scores_by_n[len(parent)] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, style=0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")
        scores_by_n[len(best_seq)] = best_score

    n_best = len(best_seq)
    a_curr = best_seq.copy()
    S_curr = a_curr.sum()
    b_curr = fast_convolve(a_curr)

    # === Phase 1: Fast coordinate descent (main workhorse) ===
    print(f"\n=== Phase 1: Fast CD (score={best_score:.10f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_curr, sc_cd, b_curr, S_curr = coord_descent_fast(
            a_curr, b_curr, S_curr, n_best, t_alloc, rng, verbose=True
        )
        update_best(a_curr, "fast_cd_p1")

    # === Phase 2: Peak-directed coordinate descent ===
    print(f"\n=== Phase 2: Peak-Directed CD (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_p, sc_pk, b_p, S_p = peak_directed_cd(
            a_curr, b_curr, S_curr, n_best, t_alloc, rng, verbose=True
        )
        update_best(a_p, "peak_cd")
        if sc_pk < 2.0 * n_best * b_curr.max() / S_curr**2:
            a_curr, b_curr, S_curr = a_p, b_p, S_p

    # === Phase 3: Fine coordinate descent ===
    print(f"\n=== Phase 3: Fine CD (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_f, sc_f, b_f, S_f = coord_descent_fine(
            a_curr, b_curr, S_curr, n_best, t_alloc, rng, verbose=True
        )
        update_best(a_f, "fine_cd")
        if sc_f < 2.0 * n_best * b_curr.max() / S_curr**2:
            a_curr, b_curr, S_curr = a_f, b_f, S_f

    # === Phase 4: Block coordinate descent ===
    print(f"\n=== Phase 4: Block CD (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_blk, sc_blk, b_blk, S_blk = block_cd(
            a_curr, b_curr, S_curr, n_best, t_alloc, rng, verbose=True
        )
        update_best(a_blk, "block_cd")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ml, sc_ml, scores_by_n = try_multiple_lengths(
            a_curr, scores_by_n, t_alloc / 5.0, rng
        )
        update_best(a_ml, "multi_length")
        if sc_ml < best_score:
            # Switch to the better length
            n_best = len(a_ml)
            a_curr = a_ml.copy()
            S_curr = a_curr.sum()
            b_curr = fast_convolve(a_curr)

    # === Phase 6: Random restarts from best ===
    print(f"\n=== Phase 6: Random Restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_restarts = min(5, int(rem / 40))
        t_per_restart = rem / (n_restarts + 1)
        print(f"  Restarts: {n_restarts}, budget each: {t_per_restart:.0f}s")

        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            print(f"  Restart {r+1}/{n_restarts}...")

            # Different perturbation strategies
            if r == 0:
                a_r = a_curr + rng.standard_normal(n_best) * 0.02
            elif r == 1:
                a_r = structured_init(n_best, rng, style=r) * 0.3 + a_curr * 0.7
            elif r == 2:
                # Swap random chunks
                a_r = a_curr.copy()
                for _ in range(3):
                    i = rng.integers(0, n_best - 50)
                    j = rng.integers(0, n_best - 50)
                    a_r[i:i+20], a_r[j:j+20] = a_r[j:j+20].copy(), a_r[i:i+20].copy()
            else:
                # Scale certain regions
                a_r = a_curr.copy()
                center = rng.integers(0, n_best)
                width = rng.integers(20, 100)
                l = max(0, center - width)
                rr = min(n_best, center + width)
                a_r[l:rr] *= 0.5

            a_r = np.maximum(a_r, 0)
            sr = a_r.sum()
            if sr < 0.01:
                continue
            a_r = a_r / sr

            b_r = fast_convolve(a_r)
            S_r = a_r.sum()

            a_opt, sc_opt, b_opt, S_opt = coord_descent_fast(
                a_r, b_r, S_r, n_best, t_per_restart, rng
            )
            update_best(a_opt, f"restart_{r}")

    # === Phase 7: Final fine polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin, sc_fin, b_fin, S_fin = coord_descent_fine(
            best_seq, fast_convolve(best_seq), best_seq.sum(), len(best_seq),
            min(rem, 60.0), rng
        )
        update_best(a_fin, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
