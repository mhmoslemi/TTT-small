"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale LP alternation
with spectral initialization, adaptive constraint selection, and direct
gradient polish. Key improvements: spectral design for initialization,
progressive resolution ramping for LP warm-starting, and active-set

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_fast_response(a_fixed, solver, n_constraints=None, tol=0.01):
    """Solve LP: min M s.t. (T_a @ b)_k <= M, sum(b)=1, b>=0.
    Use constraint selection for speed."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    
    if n_constraints is not None and n_constraints < T.shape[0]:
        # Select most important constraints (those with largest T row sums)
        row_sums = T.sum(axis=1)
        idx = np.argsort(row_sums)[::-1][:n_constraints]
        T = T[idx]
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    return None, None

def spectral_init(n, n_iters=200):
    """Design a non-negative sequence with approximately flat auto-correlation."""
    # Target: flat auto-correlation means |FFT(a)|^2 is constant
    # Start with random phase, iterate
    rng = np.random.default_rng(42)
    
    # Desired magnitude spectrum (flat)
    target_mag = np.ones(n // 2 + 1)
    
    # Initialize
    a = np.ones(n) / n
    for it in range(n_iters):
        # Compute current spectrum
        A = rfft(a, 2 * n)  # Use zero-padded FFT for linear conv
        # Actually, for auto-correlation, we need |A|^2 where A is FFT of a
        A = rfft(a)
        # Project to desired magnitude
        phase = A / np.maximum(np.abs(A), 1e-15)
        A_new = target_mag * phase
        a = irfft(A_new, n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n

    n = len(a0)

    s = a.sum()

    a = a / s
    
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        b_new, M_b = solve_lp_fast_response(y, solver, n_constraints)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Primal iterated average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 8:

                y = y / y.sum()
                stall = 0

                noise = rng.standard_normal(n) * 0.005
                y = y + noise

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

    """Solve LP: min M s.t. (T_a @ b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)

    M_var = cp.Variable()

        T_full = build_conv_matrix(a_fixed)
        T = T_full[active_idx]
    else:
        T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    return None, None

def lp_alternation_nesterov(a0, time_limit, solver, max_iters=100000,
                             use_constraint_selection=True, cs_threshold=0.95):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)
    active_idx = None
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_response(y, solver, active_idx)
        if b_new is None:
            active_idx = None  # Reset to full constraints
            stall += 1
            if stall > 50:
                break
            continue
        
        # Check if we need to restart with full constraints
        if use_constraint_selection and active_idx is not None and (it + 1) % 200 == 0:
            # Verify constraint satisfaction
            T_full = build_conv_matrix(y)
            conv = T_full @ b_new
            max_conv = conv.max()
            if max_conv > M_b * 1.01:
                active_idx = None  # Need full constraints
        
        # Update active constraints
        if use_constraint_selection and M_b > 1e-10:
            T_y = build_conv_matrix(y)
            conv_with_b = T_y @ b_new
            ratio = conv_with_b / max(M_b, 1e-15)
            active_idx = np.where(ratio >= cs_threshold)[0]
            if len(active_idx) < 10:
                active_idx = np.argsort(ratio)[::-1][:50]
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Primal iterated average
        weight_old = it + 1
        weight_new = alpha * (it + 2)
        a = (weight_old * a + weight_new * b_new) / (weight_old + weight_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection
        if (it + 1) % 1000 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 6:
                noise_scale = 0.05 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                if y.sum() > 1e-15:
                    y = y / y.sum()
                active_idx = None
                stall = 0
            elif stall >= 3:
                noise = rng.standard_normal(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                if y.sum() > 1e-15:
                    y = y / y.sum()
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def lse_gradient(a, beta):
    """Compute gradient of (2n/beta)*log(sum exp(beta*(a*a)[k])) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(beta * (b - max_b))
    w = exp_b / exp_b.sum()
    
    # grad[m] = (2n*2/beta) * sum_k w[k] * a[k-m] (if 0<=k-m<n)
    # This is 4n/beta * cross_correlation(w, a)[m]
    
    # cross_correlation(w, a)[m] = sum_k w[k] * a[k-m] (with bounds)
    # = sum_i w[m+i] * a[i] (where i = k-m, 0<=i<n)
    # This is the "flip and convolve" of w with a
    # = (reverse(w) * a)[2n-2-m] ... let me just compute it directly via FFT
    
    # Actually: sum_k w[k] * a[k-m] where 0<=k<2n-1 and 0<=k-m<n
    # Let's pad w to length 4n and compute linear cross-correlation
    w_padded = np.zeros(4 * n)
    w_padded[:2*n-1] = w
    a_padded = np.zeros(4 * n)
    a_padded[:n] = a
    
    # Cross-correlation: (w ⊛ a)[m] = sum_k w[k] * a[k-m]
    # = sum_k w[k+m] * a[k] (substituting k -> k+m)
    # This is the standard cross-correlation, computable as:
    # FFT^{-1}(conj(FFT(w)) * FFT(a))
    W = rfft(w_padded, 4*n)
    A = rfft(a_padded, 4*n)
    cc = irfft(np.conj(W) * A, 4*n)
    # cc[m] = sum_k w[k] * a[k-m] (for m = 0, 1, ..., 4n-1)
    # We want m in [0, n-1]
    grad = (4.0 * n / beta) * cc[:n]
    
    # Subtract the mean to project onto sum=0 subspace (for simplex constraint)
    # Actually, we'll handle the simplex constraint via projection
    return grad

def lse_lbfgs(a0, time_limit, betas=None, max_iter_per_stage=2000):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    if betas is None:
        betas = [50.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        def func(z_val):
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b_val = fast_convolve(a_val)
            max_b = b_val.max()
            exp_b = np.exp(beta * (b_val - max_b))
            lse = max_b + np.log(exp_b.sum()) / beta
            return 2.0 * n * lse
        
        def jac(z_val):
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b_val = fast_convolve(a_val)
            max_b = b_val.max()
            exp_b = np.exp(beta * (b_val - max_b))
            w = exp_b / exp_b.sum()
            
            # grad of 2n * lse w.r.t. a
            # d lse / d a_m = sum_k w[k] * 2 * a[k-m] (if valid) / 1 (since lse = max_b + log(sum exp)/beta)
            # Wait: lse = max_b + (1/beta) * log(sum_k exp(beta * (b - max_b)))
            # d lse / d a_m = (1/beta) * sum_k exp(beta*(b_k - max_b)) / sum_k exp(beta*(b_k - max_b)) * beta * d b_k / d a_m
            # = sum_k w[k] * d b_k / d a_m
            # = sum_k w[k] * 2 * a[k-m] (if valid)
            
            # Compute via FFT
            w_padded = np.zeros(4 * n)
            w_padded[:2*n-1] = w
            a_padded = np.zeros(4 * n)
            a_padded[:n] = a_val
            W = rfft(w_padded, 4*n)
            A = rfft(a_padded, 4*n)
            cc = irfft(np.conj(W) * A, 4*n)
            grad_a = 2.0 * cc[:n]
            
            # Now convert from grad_a to grad_z (chain rule for softmax)
            # a = softmax(z), so a_i = exp(z_i) / sum(exp(z))
            # d a_i / d z_j = a_i * (δ_ij - a_j)
            # grad_z_j = sum_i grad_a_i * d a_i / d z_j = sum_i grad_a_i * a_i * (δ_ij - a_j)
            # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
            S = np.dot(grad_a, a_val)
            grad_z = a_val * (grad_a - S)
            grad_z *= 2.0 * n
            
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-10, 'maxfun': max_iter_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

def active_set_subgradient(a0, time_limit, max_steps=30000, initial_lr=0.02):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = initial_lr
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: top 3 peaks
        top_k = min(5, len(b))
        active = np.argsort(b)[::-1][:top_k]
        
        # Compute gradient as weighted average over active peaks
        g = np.zeros(n)
        for k in active:
            # grad of (a*a)[k] w.r.t. a[m] = 2*a[k-m] (if 0<=k-m<n)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                j = k - m
                if 0 <= j < n:
                    g[m] += 2.0 * a[j]
        g /= len(active)
        
        # Project gradient step onto feasible set
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.85
                no_improve = 0
            lr = max(1e-9, lr)
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score

def spectral_init(n, n_iters=300, seed=42):
    """Generate initial sequence with approximately flat auto-correlation."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    
    for it in range(n_iters):
        A = rfft(a, 2 * n)
        # Target: |A|^2 should be as flat as possible
        # Project: take current phase, set magnitude to target
        target_mag = np.sqrt(np.ones(n // 2 + 1 + n - 1).sum() / (n // 2 + 1 + n - 1))
        # Actually, for flat auto-correlation, we want |FFT(a)|^2 = constant
        # The DC component is fixed (sum(a) = 1)
        phase = A / np.maximum(np.abs(A), 1e-15)
        new_mag = np.ones(len(A))
        new_mag[0] = np.abs(A[0])  # Keep DC
        A_new = new_mag * phase
        a_new = irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            break
        a = a_new
    
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
