"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced bidirectional LP alternation
with Nesterov acceleration, multi-n exploration, random restarts to escape local minima,
peak-equalization via SLSQP, and Fourier-domain refinement.
Key improvements: (1) aggressive multi-n sweep, (2) LP alternation with adaptive
restart and convergence detection, (3) multiple independent runs from different
start points, (4) SLSQP-based peak equalization as a new optimization angle.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_given_fixed(fixed, solver):
    n = len(fixed)
    x_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed)
    constraints = [T @ x_var <= M_var, cp.sum(x_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver]
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.CBC]:
        if alt not in solvers_to_try:
            solvers_to_try.append(alt)
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.asarray(x_var.value, dtype=np.float64)
                x_val = np.maximum(x_val, 0)
                ss = x_val.sum()
                if ss > 1e-15:
                    return x_val / ss, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_alternation(a0, time_limit, solver, max_iters=1000000, verbose=True, seed=42):
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(seed)
    last_improve_iter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M1 = solve_lp_given_fixed(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 3)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)

        a_new, M2 = solve_lp_given_fixed(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 4)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)

        # Nesterov combined points
        a_comb = ((it + 1) * a + (2.0 / (it + 3)) * (it + 3) * y_a) / (it + 1 + 2.0)
        a_comb = normalize(a_comb)
        b_comb = ((it + 1) * b + (2.0 / (it + 4)) * (it + 4) * y_b) / (it + 1 + 2.0)
        b_comb = normalize(b_comb)
        a = a_comb
        b = b_comb

        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            last_improve_iter = it

        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()
            last_improve_iter = it

        if (it + 1) % 1000 == 0:
            if verbose:
                print(f"      LP it={it+1} ({time.monotonic()-t0:.0f}s) score={best_score:.10f}")
            if best_score < prev_best - 1e-11:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                noise_scale = min(0.05, 0.02 * math.sqrt(stall_count))
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall_count = 0

    return best_a, best_score


def quick_subgradient(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.08)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.92
                no_improve = 0
        lr = max(1e-8, lr)

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    z = z - z.max()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 50))

        def obj_and_grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            obj_val = 2.0 * n * lse

            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-13:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - S) / T
            return obj_val, grad_z

        def obj_only(z):
            return obj_and_grad(z)[0]

        def grad_only(z):
            return obj_and_grad(z)[1]

        try:
            res = minimize(
                obj_only, z, jac=grad_only, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_equalization_slsqp(a0, time_limit, n_peaks=10):
    """Use SLSQP to directly minimize the max of top peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    b = fast_convolve(a)
    M = b.max()
    # Find the n_peaks positions with highest values
    top_indices = np.argsort(b)[-n_peaks:]

    def objective(x):
        a_cur = np.maximum(x, 0)
        s = a_cur.sum()
        if s < 1e-10:
            return 1e15
        a_cur = a_cur / s
        b_cur = fast_convolve(a_cur)
        # Use LogSumExp as smooth max
        top_vals = b_cur[top_indices]
        T = 0.01
        shifted = top_vals / T
        shifted -= shifted.max()
        lse = T * (shifted.max() + np.log(np.exp(shifted).sum()))
        return 2.0 * n * lse

    def gradient(x):
        a_cur = np.maximum(x, 0)
        s = a_cur.sum()
        if s < 1e-10:
            return np.zeros(n)
        a_cur = a_cur / s
        b_cur = fast_convolve(a_cur)
        top_vals = b_cur[top_indices]
        T = 0.01
        shifted = top_vals / T
        shifted -= shifted.max()
        w = np.exp(shifted) / np.exp(shifted).sum()

        grad_a = np.zeros(n)
        for i, k in enumerate(top_indices):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[i] * a_cur[k - idx]

        S = np.dot(grad_a, a_cur)
        return grad_a / T - grad_a * S / (a_cur + 1e-30)

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n

    try:
        res = minimize(objective, a, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-16, 'disp': False})
        a_new = np.maximum(res.x, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):

    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))

    else:

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fourier_refine(a0, n_freq, time_limit):

    a = normalize(a0)

    x0[0] = A[0].real
    for i in range(1, n_freq + 1):
        x0[2 * i - 1] = A[i].real
        x0[2 * i] = A[i].imag

    A_rest = A[n_freq + 1:].copy()

    def fourier_obj(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()

    try:
        res = minimize(fourier_obj, x0, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-12,
                                'adaptive': True})
        x_opt = res.x
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]

        a_pert = best_seq + rng.standard_normal(n_cur) * 0.005
        a_pert = normalize(a_pert)
        a_lp2, _ = lp_alternation(a_pert, t3, solver, verbose=False, seed=9999)
        try_update(a_lp2, "lp_restart")

    # === Phase 4: Peak equalization via SLSQP ===
    print(f"\n=== Phase 4: Peak Equalization (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.1, 120.0)
    if t4 > 15:
        a_pe, _ = peak_equalization_slsqp(best_seq, t4, n_peaks=8)
        try_update(a_pe, "peak_eq")

    # === Phase 5: L-BFGS LogSumExp polish ===
    print(f"\n=== Phase 5: L-BFGS Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if t5 > 20:
        a_lb, _ = lbgfs_logsumexp(best_seq, t5,
                                  T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005])
        try_update(a_lb, "lbgfs_polish")

    # === Phase 6: Subgradient fine-tuning ===
    print(f"\n=== Phase 6: Subgradient Fine-tune (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 80.0)
    if t6 > 10:
        a_sg, _ = quick_subgradient(best_seq, t6, n_steps=30000)
        try_update(a_sg, "subgrad_final")

    # === Phase 7: Final short LP on best ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.1, 100.0)
    if t7 > 15:
        a_fin, _ = lp_alternation(best_seq, t7, solver, verbose=False, seed=31337)
        try_update(a_fin, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
