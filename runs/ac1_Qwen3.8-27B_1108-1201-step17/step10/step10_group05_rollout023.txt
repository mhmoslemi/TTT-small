"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient with adaptive step sizes
2. LP fictitious play with fast convergence
3. Enhanced coordinate descent with spectral awareness
4. Multi-length exploration with structured seeds
5. Iterative refinement loops
"""
import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_subgradient(a, top_k_frac=0.05):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    # Use all indices within top_k_frac of M for a more stable gradient
    threshold = M - top_k_frac * M
    near_peak = conv >= threshold
    if near_peak.sum() == 0:
        near_peak = np.zeros_like(conv, dtype=bool)
        near_peak[conv.argmax()] = True
    
    rho = 1.0 / near_peak.sum()
    weights = np.where(near_peak, rho, 0.0)
    
    # grad[i] = 2 * sum_k weights[k] * a[k-i] (where k-i is valid)
    grad = np.zeros(n)
    active_k = np.where(near_peak)[0]
    for k in active_k:
        w = weights[k]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w * a[k - j]
    
    return grad, M

def nesterov_subgradient(a0, time_limit, label=""):
    """Nesterov-accelerated subgradient method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    no_improve = 0
    top_k_frac = 0.05
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        step = min(1.0 / np.sqrt(it + 1), 0.5 / (it + 1))
        
        # Nesterov lookahead
        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))
        
        # Subgradient
        grad, M = compute_subgradient(y, top_k_frac)
        
        # Step
        a_new = a - step * grad
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

def nesterov_subgradient(a0, time_limit, label=""):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

            grad[j] += 2.0 * w * y[k - j]

        a_new = simplex_project(a - step * grad)
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if no_improve > 100 and it % 100 == 0:
            top_k_frac = min(0.3, top_k_frac * 1.3)
        if no_improve > 800:
            break
        if it % 200 == 0:
            print(f"    NSub {label} it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def peak_targeted_perturb(a0, rng, strength=0.05):
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.5 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.15):
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 4):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.4)
    a = np.maximum(a, 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.3))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.5
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def structured_init(n, rng, method="bump"):
    """Generate structured initial sequences."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "bump":
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.5) ** 2)
        a += 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif method == "dual_bump":
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * (x - 0.3) ** 2) + np.exp(-6 * (x - 0.7) ** 2)
        a += 0.05 * rng.standard_normal(n)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif method == "periodic":
        p = max(2, n // 256)
        base = 1.0 + 0.3 * np.sin(2 * np.pi * np.arange(p) / p)
        a = np.tile(base, n // p + 1)[:n]
        a += 0.02 * rng.standard_normal(n)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif method == "fractal":
        a = np.ones(n) / n
        for scale in [n // 2, n // 4, n // 8, n // 16]:
            if scale < 2:
                break
            pattern = 1.0 + 0.5 * rng.standard_normal(scale)
            pattern = np.maximum(pattern, 0.1)
            a_ext = np.tile(pattern, n // scale + 1)[:n]
            a = a + 0.3 * a_ext
            a = np.maximum(a, 0)
            a = a / a.sum()
        return a
    else:
        a = np.ones(n) + 0.5 * rng.standard_normal(n)
        a = np.maximum(a, 0)
        return a / a.sum()
