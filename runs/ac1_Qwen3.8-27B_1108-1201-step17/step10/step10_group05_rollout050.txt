import numpy as np

_lp_solver_cached = None

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCS, cp.GLOP, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:

                break

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

            coeffs = 2.0 * a[idx_range]
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            # Fine fractional steps
            for factor in np.concatenate([
                np.linspace(0.9999, 0.999, 3),
                np.linspace(0.999, 0.99, 5),
                np.linspace(0.99, 0.95, 8),
                np.linspace(0.95, 0.8, 10),
                np.linspace(1.0001, 1.01, 3),
                np.linspace(1.01, 1.1, 8),
                np.linspace(1.1, 1.5, 5),
            ]):
                delta = ai * (factor - 1.0)
                if delta < -ai - 1e-14:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping and dual-guided updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    b = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            # Fallback: use softmax over convolution peaks
            conv_a = fast_convolve(a)
            beta = 50.0
            weights = np.exp(beta * (conv_a - conv_a.max()))
            b_new = weights / weights.sum()
            # Map from convolution indices to sequence indices
            # conv_a[k] = sum_j a[j]*a[k-j], so the "pressure" on a[j] is sum_k w[k]*a[k-j]
            # Approximate: b_new[j] proportional to sum_k w[k] * a[k-j]
            pressure = np.zeros(n)
            active_k = np.where(weights > 1e-10)[0]
            for k in active_k:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                pressure[j_start:j_end+1] += weights[k] * a[k - np.arange(j_start, j_end+1)]
            if pressure.max() > 1e-20:
                b_new = pressure / pressure.sum()

        # Adaptive damping
        alpha = min(1.0, 15.0 / (iters + 10))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Also try the LP solution directly
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        if iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters

def soft_averaged_p(a0, K=5, temperature=0.01):
    """Generate a soft dual distribution and use it to compute a pressure-based update."""
    n = len(a0)
    conv_a = fast_convolve(a0)
    # Softmax over top K
    sorted_idx = np.argsort(conv_a)[::-1][:K]
    top_vals = conv_a[sorted_idx]
    weights = np.exp(-(top_vals - top_vals.min()) / (temperature * top_vals.max() + 1e-10))
    weights = weights / weights.sum()
    
    # Pressure on each coordinate
    pressure = np.zeros(n)
    for w, k in zip(weights, sorted_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        pressure[j_start:j_end+1] += w * a0[k - np.arange(j_start, j_end+1)]
    
    if pressure.max() > 1e-20:
        return pressure / pressure.sum()
    return np.ones(n) / n

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.15):
    """Randomly restructure to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(20, n // 4):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.4)
    a = np.maximum(a, 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.3))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.3
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score

def mirror_descent_polish(a0, time_limit, label=""):
    """Mirror descent with log-sum-exp smoothing for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = 50.0
    lr = 0.005
    no_improve = 0
    gamma = 1.0
    y = a.copy()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y_proj)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -30, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        grad = np.zeros(n)
        active_k = np.where(weights > 1e-8)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.85
            beta *= 1.2
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 20 == 0:
            beta *= 1.02

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
