**SQP for the non-convex max-quadratic problem:**

At iteration $t$ with current point $a_t$:
1. Compute $b_t = a_t * a_t$, $M_t = \max b_t$
2. Active set: $S_t = \{k : b_t[k] \ge M_t - \epsilon\}$ (use adaptive $\epsilon$)
3. Solve the QP subproblem:
   $$\min_{d} \quad \frac{1}{2} d^T H d + g^T d$$
   subject to: $c_k^T d \le u_k, \quad k \in S_t$
   $$\sum d_j = 0, \quad a_t + d \ge 0$$
   
   Where:
   - $g = 2 \cdot \text{mean}_{k \in S_t} [(A_k)^T a_t]$ (gradient of mean of active quadratics)
   - $H = \sum_{k \in S_t} A_k$ (sum of Hessians of active quadratics, or use a BFGS approximation)
   - $c_k^T d = 2 (a_t)^T A_k d$ (linearized constraint) -- wait, this isn't right.
   
   Actually, let me reconsider. The subproblem in SQP for $\min \max_k q_k(a)$ is:
   $$\min_d \quad \max_{k \in S} (q_k(a) + \nabla q_k(a)^T d + \frac{1}{2} d^T \nabla^2 q_k(a) d)$$
   subject to non-active constraints.
   
   This is still non-linear in $d$ (due to the quadratic terms). For SQP, we typically linearize the constraints and keep the quadratic objective:
   $$\min_d \quad q_{k_0}(a) + \nabla q_{k_0}(a)^T d + \frac{1}{2} d^T (LBB + \mu I) d$$
   subject to: $q_k(a) + \nabla q_k(a)^T d \le M', \quad k \in S \setminus \{k_0\}$
   (where $M'$ is an additional variable to minimize)
   
   This is getting complicated. Let me simplify.

**Simplified SQP:**

At each step:
1. Compute $b = a*a$, $M = \max b$, active set $S$
2. Solve LP: $\min_d M'$ subject to $b[k] + 2(a*d)[k] \le M'$ for all $k$ in "active + near-active" set, $\sum d = 0$, $a + d \ge 0$
   - This gives direction $d_{LP}$
3. Compute second-order correction: $q[k] = (d_{LP} * d_{LP})[k]$
4. Solve modified LP: $\min_d M'$ subject to $b[k] + 2(a*d)[k] + q[k] \le M'$ for all $k$ in active set, $\sum d = 0$, $a + d \ge 0$
   - This gives direction $d_{SQP}$ (the $q[k]$ term acts as a constant offset, penalizing directions that would create high second-order terms)
5. Line search: find best $t \in (0, 1]$ such that $(a + t d_{SQP})*(a + t d_{SQP})$ has lower max than $b$
6. Update $a \leftarrow a + t \cdot d_{SQP}$

The key insight is that step 4 accounts for the curvature of the objective, which the standard LP (step 2) ignores.

Let me also think about a **"direct max reduction"** approach:

At each step, identify the single worst position $k^* = \arg\max_k (a*a)[k]$. Then:
- The gradient of $(a*a)[k^*]$ with respect to $a$ is $2 \cdot (a)[k^* - j]$ (reversed, appropriately bounded)
- Take a step in the negative gradient direction, projected onto the simplex

This is essentially a subgradient method focused on the single worst constraint. It's simple but might be slow. However, combined with the LP approach, it could help in the final stages.

4. Redistribute the mass to positions that contribute less

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Bidirectional LP alternation with Nesterov averaging
2. SQP-style second-order LP correction (novel)
3. Adaptive active-set management
4. Multi-n exploration with interpolation
5. Fourier-parametric low-dimensional refinement
6. Direct subgradient and exact line-search polishing
7. Random perturbation restarts to escape local minima

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective (assumes sum(a)=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.GLPK, cp.GUROBI]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s != solver]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                ss = b_val.sum()
                if ss > 1e-15:
                    return b_val / ss, float(M_var.value)
        except Exception:
            continue
    return None, None


def bidirectional_lp_alternation(a0, time_limit, solver, max_iters=100000):
    """Alternating LP with Nesterov averaging and adaptive noise."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, _ = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = normalize(b + alpha * (b_new - b))

        a_new, _ = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = normalize(a + alpha * (a_new - a))

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 15:
                noise_scale = 0.03 / math.sqrt(stall)
                y_a = normalize(y_a + rng.standard_normal(n) * noise_scale)
                y_b = normalize(y_b + rng.standard_normal(n) * noise_scale)
                stall = 0
            if (it + 1) % 1000 == 0:
                print(f"    BLA iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def sqp_refinement(a0, time_limit, solver, n_steps=500):
    """SQP-style refinement: LP direction + second-order correction + line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    overall_best = best_score

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        size = 2 * n - 1

        # Adaptive active set threshold
        threshold = M * 0.95
        active_k = np.where(b >= threshold)[0]
        if len(active_k) == 0:
            active_k = np.array([int(np.argmax(b))])
        m_act = len(active_k)

        # Step 1: Standard LP direction
        G = np.zeros((m_act, n))
        for i, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            G[i, js:je+1] = 2.0 * a[k - np.arange(js, je + 1)]

        # Also add constraints for near-inactive positions
        near_thresh = M * 0.8
        near_k = np.where((b < threshold) & (b >= near_thresh))[0]
        if len(near_k) > 0 and len(near_k) < 300:
            G2 = np.zeros((len(near_k), n))
            for i, k in enumerate(near_k):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                G2[i, js:je+1] = 2.0 * a[k - np.arange(js, je + 1)]
            G_full = np.vstack([G, G2])
            rhs = np.concatenate([np.zeros(m_act), np.zeros(len(near_k))])
        else:
            G_full = G
            rhs = np.zeros(m_act)

        # LP: minimize M' subject to G_full @ d <= M' - b[active_k_near_k], sum(d)=0, a+d>=0
        d_var = cp.Variable(n)
        Mp_var = cp.Variable()
        b_all = np.concatenate([b[active_k], b[near_k]]) if len(near_k) > 0 else b[active_k]
        constraints = [
            G_full @ d_var <= Mp_var - b_all,
            cp.sum(d_var) == 0,
            a + d_var >= 0,
        ]
        prob = cp.Problem(cp.Minimize(Mp_var), constraints)
        d_lp = None
        for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    d_lp = np.asarray(d_var.value, dtype=np.float64)
                    break
            except Exception:
                continue

        if d_lp is None:
            continue

        # Step 2: Second-order correction
        q = fast_convolve(d_lp)  # (d_lp * d_lp)[k]
        # Rebuild LP with q as constant offset
        if len(near_k) > 0 and len(near_k) < 300:
            b_all2 = np.concatenate([b[active_k] + q[active_k], b[near_k] + q[near_k]])
        else:
            b_all2 = b[active_k] + q[active_k]

        d_var2 = cp.Variable(n)
        Mp_var2 = cp.Variable()
        constraints2 = [
            G_full @ d_var2 <= Mp_var2 - b_all2,
            cp.sum(d_var2) == 0,
            a + d_var2 >= 0,
        ]
        prob2 = cp.Problem(cp.Minimize(Mp_var2), constraints2)
        d_sqp = None
        for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob2.solve(solver=s)
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    d_sqp = np.asarray(d_var2.value, dtype=np.float64)
                    break
            except Exception:
                continue

        d = d_sqp if d_sqp is not None else d_lp

        # Step 3: Line search (exact, on the true non-smooth objective)
        best_t = 0.0
        best_f = best_score
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_trial = a + t * d
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            f_trial = 2.0 * n * fast_convolve(a_trial).max()
            if f_trial < best_f:
                best_f = f_trial
                best_t = t

        if best_t > 0 and best_f < best_score - 1e-14:
            a = a + best_t * d
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = best_f
            best_a = a.copy()
            if best_score < overall_best:
                overall_best = best_score
        elif (step + 1) % 50 == 0:
            # Add small noise to escape
            a = normalize(a + np.random.default_rng(step).standard_normal(n) * 0.005)

        if (step + 1) % 100 == 0:
            print(f"    SQP step={step+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}, overall={overall_best:.10f}")

    return best_a, best_score


def fourier_lowdim_refine(a0, time_limit, n_modes=30):
    """Optimize a low-dimensional Fourier parameterization of the sequence."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)

    # Parameterize: optimize first n_modes complex coefficients + DC
    # x = [A[0].real, A[1].real, A[1].imag, ..., A[n_modes].real, A[n_modes].imag]
    n_params = 2 * n_modes + 1
    x0 = np.zeros(n_params)
    x0[0] = A[0].real
    for i in range(1, n_modes + 1):
        x0[2*i - 1] = A[i].real
        x0[2*i] = A[i].imag

    A_rest = A[n_modes + 1:].copy()

    def make_fourier_obj(x, A_rest, n, n_modes, fft_size):
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_modes + 1):
            A_new[i] = x[2*i - 1] + 1j * x[2*i]
        A_new[n_modes + 1:] = A_rest
        a_cand = irfft(A_new, fft_size)[:n]
        a_cand = np.maximum(a_cand, 0)
        s = a_cand.sum()
        if s < 1e-10:
            return 1e15
        a_cand = a_cand / s
        b = fast_convolve(a_cand)
        return 2.0 * n * b.max()

    def fourier_obj(x):
        return make_fourier_obj(x, A_rest, n, n_modes, fft_size)

    # Nelder-Mead (derivative-free, robust for low-dim)
    try:
        res = minimize(
            fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': 10000, 'xatol': 1e-15, 'fatol': 1e-15, 'adaptive': True}
        )
        x_opt = res.x
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_modes + 1):
            A_new[i] = x_opt[2*i - 1] + 1j * x_opt[2*i]
        A_new[n_modes + 1:] = A_rest
        a_cand = irfft(A_new, fft_size)[:n]
        a_cand = np.maximum(a_cand, 0)
        s = a_cand.sum()
        if s > 1e-10:
            a_cand = a_cand / s
            sc = 2.0 * n * fast_convolve(a_cand).max()
            if sc < best_score:
                best_score = sc
                best_a = a_cand.copy()
    except Exception:
        pass

    # L-BFGS-B with finite differences for further refinement
    if time.monotonic() - t0 < time_limit * 0.6:
        try:
            res2 = minimize(
                fourier_obj, x_opt, method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 50000}
            )
            A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
            A_new[0] = res2.x[0]
            for i in range(1, n_modes + 1):
                A_new[i] = res2.x[2*i - 1] + 1j * res2.x[2*i]
            A_new[n_modes + 1:] = A_rest
            a_cand = irfft(A_new, fft_size)[:n]
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s > 1e-10:
                a_cand = a_cand / s
                sc = 2.0 * n * fast_convolve(a_cand).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_cand.copy()
        except Exception:
            pass

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        max_iter = max(30, int(min(remaining * 30, 3000)))

        def obj_and_grad(z_cur):
            z_max = z_cur.max()
            exp_z = np.exp(z_cur - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            lse = b_max + np.log(exp_ws.sum())
            obj_val = 2.0 * n * lse

            # Gradient
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                grad_a[js:je+1] += 2.0 * w[k] * a_cur[k - np.arange(js, je + 1)]
            S = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - S)
            return obj_val, grad_z

        def obj_fn(z_cur):
            return obj_and_grad(z_cur)[0]

        def grad_fn(z_cur):
            return obj_and_grad(z_cur)[1]

        try:
            res = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': max_iter * 100}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def subgradient_polish(a0, time_limit):
    """Subgradient method focused on the worst constraint, with exact line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    rng = np.random.default_rng(777)

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Active set: positions within 0.1% of max
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient (average over active)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je+1] += 2.0 * a[k - np.arange(js, je + 1)]
        g /= len(active)

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g /= g_norm

        # Step with line search
        best_t = 0.0
        best_f = best_score
        for t in [lr, lr*0.5, lr*0.25, lr*0.1]:
            a_trial = a - t * g
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            f_trial = 2.0 * n * fast_convolve(a_trial).max()
            if f_trial < best_f:
                best_f = f_trial
                best_t = t

        if best_t > 0 and best_f < best_score - 1e-14:
            a = a - best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = best_f
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.9
                no_improve = 0
            if no_improve >= 500:
                # Add noise
                a = normalize(a + rng.standard_normal(n) * 0.002)
                no_improve = 0

        lr = max(1e-8, lr)
        if (step + 1) % 10000 == 0:
            print(f"    SubGrad step={step+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def active_set_exact_optimize(a0, time_limit, solver, n_rounds=100):
    """Iteratively solve LP with shrinking active set for precise convergence."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Tight active set
        for frac in [0.999, 0.998, 0.995, 0.99, 0.98]:
            threshold = M * frac
            active_k = np.where(b >= threshold)[0]
            if len(active_k) == 0:
                continue
            m_act = len(active_k)

            G = np.zeros((m_act, n))
            for i, k in enumerate(active_k):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                G[i, js:je+1] = 2.0 * a[k - np.arange(js, je + 1)]

            # Constraint for all others: don't increase
            other_k = np.where(b < threshold)[0]
            # Only add constraints for those within 5% of M
            constr_k = other_k[b[other_k] >= M * 0.95]
            if len(constr_k) > 50:
                constr_k = constr_k[:50]  # Limit for efficiency

            d_var = cp.Variable(n)
            Mp_var = cp.Variable()
            constraints = [
                G @ d_var <= Mp_var - b[active_k],
                cp.sum(d_var) == 0,
                a + d_var >= 0,
            ]
            if len(constr_k) > 0:
                G_c = np.zeros((len(constr_k), n))
                for i, k in enumerate(constr_k):
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    G_c[i, js:je+1] = 2.0 * a[k - np.arange(js, je + 1)]
                constraints.append(G_c @ d_var <= b[constr_k] - b[constr_k])  # i.e., <= 0

            prob = cp.Problem(cp.Minimize(Mp_var), constraints)
            d_val = None
            for s in [solver, cp.ECOS, cp.SCS]:
                try:
                    prob.solve(solver=s)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        d_val = np.asarray(d_var.value, dtype=np.float64)
                        break
                except Exception:
                    continue

            if d_val is None:
                continue

            # Line search
            best_t = 0.0
            best_f = best_score
            for t in [1.0, 0.5, 0.3, 0.1, 0.05]:
                a_trial = a + t * d_val
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                f_trial = 2.0 * n * fast_convolve(a_trial).max()
                if f_trial < best_f:
                    best_f = f_trial
                    best_t = t

            if best_t > 0 and best_f < best_score - 1e-14:
                a = a + best_t * d_val
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_f
                best_a = a.copy()
                break  # Move to next round with tighter threshold
            else:
                continue

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP alternation for coarse optimization
    2. SQP refinement with second-order correction
    3. Multi-n exploration
    4. Active-set exact optimization
    5. Fourier low-dim refinement
    6. LogSumExp L-BFGS polishing
    7. Subgradient final polish
    8. Random restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = normalize(best_seq)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Bidirectional LP alternation ===
    print(f"\n=== Phase 1: LP Alternation (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = bidirectional_lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "lp_alt")

    # === Phase 2: SQP Refinement ===
    print(f"\n=== Phase 2: SQP Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_sqp, _ = sqp_refinement(best_seq, t2, solver, n_steps=300)
        try_update(a_sqp, "sqp")

    # === Phase 3: Active-Set Exact Optimization ===
    print(f"\n=== Phase 3: Active-Set Exact (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 200.0)
    if t3 > 15:
        print(f"  Budget: {t3:.0f}s")
        a_as, _ = active_set_exact_optimize(best_seq, t3, solver, n_rounds=200)
        try_update(a_as, "active_set")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        # Try several n values
        n_targets = [800, 1000, 1200, 1500, 1800, 2000, 2400, 3000, 3600, 4800]
        # Sort by proximity to current best
        n_targets = [nt for nt in n_targets if nt != n_best]
        n_targets.sort(key=lambda x: abs(x - n_best))

        for cand_n in n_targets[:6]:  # Try up to 6 different n
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.15, 100.0)
            if t_sub < 20:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)

            # Quick LP
            a_cand, _ = bidirectional_lp_alternation(init_cand, t_sub * 0.5, solver)
            try_update(a_cand, f"lp_n{cand_n}")

            # Quick SQP
            rem = deadline - time.monotonic()
            if rem > 15:
                a_cand2, _ = sqp_refinement(a_cand if a_cand is not None else init_cand,
                                            min(rem * 0.3, 40.0), solver, n_steps=50)
                try_update(a_cand2, f"sqp_n{cand_n}")

    # === Phase 5: Fourier Low-dim Refinement ===
    print(f"\n=== Phase 5: Fourier Refinement (score={best_score:.10f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        for n_modes in [20, 40, 60]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_sub = min(rem * 0.15, 60.0)
            if t_sub < 10:
                break
            a_four, _ = fourier_lowdim_refine(best_seq, t_sub, n_modes=n_modes)
            try_update(a_four, f"fourier_K{n_modes}")

    # === Phase 6: LogSumExp L-BFGS ===
    print(f"\n=== Phase 6: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 200.0)
    if t6 > 20:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        print(f"  T_stages: {T_stages}, budget={t6:.0f}s")
        a_lb, _ = logsumexp_lbgfs(best_seq, t6, T_stages=T_stages)
        try_update(a_lb, "lbgfs_lowT")

    # === Phase 7: Subgradient Polish ===
    print(f"\n=== Phase 7: Subgradient Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.10, 100.0)
    if t7 > 10:
        a_sg, _ = subgradient_polish(best_seq, t7)
        try_update(a_sg, "subgrad")

    # === Phase 8: Random Restarts ===
    print(f"\n=== Phase 8: Random Restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_restarts = min(int(rem / 20), 8)
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_sub = min(rem * 0.1, 40.0)
            # Perturb the current best
            noise_scale = 0.01 + 0.005 * r
            perturbed = normalize(best_seq + rng.standard_normal(len(best_seq)) * noise_scale)
            # Quick optimize
            a_r, _ = sqp_refinement(perturbed, t_sub * 0.6, solver, n_steps=30)
            try_update(a_r, f"restart_{r}")
            rem = deadline - time.monotonic()
            if rem > 10:
                a_r2, _ = logsumexp_lbgfs(a_r, min(rem * 0.3, 20.0),
                                         T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.01])
                try_update(a_r2, f"restart_{r}_lbgfs")

    # === Phase 9: Final Intensive Polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # SQP again
        t_f1 = min(rem * 0.3, 60.0)
        if t_f1 > 10:
            a_f1, _ = sqp_refinement(best_seq, t_f1, solver, n_steps=100)
            try_update(a_f1, "final_sqp")

        # Active set
        rem = deadline - time.monotonic()
        t_f2 = min(rem * 0.3, 50.0)
        if t_f2 > 10:
            a_f2, _ = active_set_exact_optimize(best_seq, t_f2, solver, n_rounds=100)
            try_update(a_f2, "final_active")

        # Very low T L-BFGS
        rem = deadline - time.monotonic()
        t_f3 = min(rem * 0.3, 50.0)
        if t_f3 > 10:
            a_f3, _ = logsumexp_lbgfs(best_seq, t_f3,
                                     T_stages=[0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
            try_update(a_f3, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
