"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

Key theorem: For any non-negative sequence a of length n,
  score = 2*n*max(conv(a,a))/(sum(a))^2 >= 2
with equality iff a is uniform (all elements equal).

Proof: max(conv) >= conv[0] = sum(a_i^2) >= (sum(a_i))^2/n  (Cauchy-Schwarz)
Therefore: score >= 2*n*(sum)^2/n/(sum)^2 = 2.

This solution returns a uniform sequence (the optimal solution) after

import numpy as np

def fast_convolve(a):
    from scipy.fft import rfft, irfft
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_local(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

            lo = 0.0

            c = hi - gr * (hi - lo)

                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = score_at_x(d)
