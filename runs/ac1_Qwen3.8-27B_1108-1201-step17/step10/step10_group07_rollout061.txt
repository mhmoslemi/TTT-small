import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, solver):

    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)

        cp.sum(b_var) == 1

    try:

If j = k-j (i.e., k = 2j): d(c[k])/d(a[j]) = 2*a[j]

So the subgradient of max_k c[k] with respect to a is:
g[j] = sum_{k in argmax} 2*a[k-j] (for valid k-j)

This is similar to what the subgradient descent in the previous code does.

Let me think about a **hybrid approach**:
1. Use LP for the main optimization (it's very effective)
2. Add a "local search" step that specifically targets the top few peaks
3. Use multiple starting points with different n values
4. Add a "mutation" step to escape local minima

Let me also consider: what if I try to directly minimize the objective using a **trust-region method** or **L-BFGS-B** from scipy?

The objective is f(a) = 2*n*max_k c[k] / (sum a)^2, which is non-smooth (due to the max). But we can smooth it with a softmax:
f_smooth(a) = 2*n * logsumexp(c / tau) / (sum a)^2

for small tau. This is smooth and can be optimized with L-BFGS-B.

Actually, let me think about this more carefully. The softmax smoothing:
logsumexp(c/tau) ≈ max(c) + tau * log(K) for small tau (where K is the number of near-max elements).

So f_smooth ≈ 2*n*(max(c) + tau*log(K)) / S^2, which is close to f for small tau.

Using L-BFGS-B with this smoothed objective could be a good complement to the LP approach.

Let me also think about the **structure of the optimal solution**. 

For the "anti-symmetric" insight: if a[i] = f(i) for i < n/2 and a[i] = g(i) for i >= n/2, where f is large and g is small (or vice versa), then c[n-1] = sum f(i)*g(n-1-i) is small. But other lags might be large.

The optimal balance is to make all c[k] as equal as possible (like a "flat" autocorrelation). This is similar to the "constant amplitude zero autocorrelation" (CAZAC) sequences in radar/communications, but for non-negative real sequences.

For non-negative sequences, perfect flat autocorrelation is impossible (since c[k] >= 0 and sum of all c[k] = (sum a)^2 = S^2, so the average c[k] = S^2/(2n-1)). The best we can do is make c[k] as close to S^2/(2n-1) as possible for all k.

If we could achieve c[k] = S^2/(2n-1) for all k, then M = S^2/(2n-1) and score = 2*n*S^2/((2n-1)*S^2) = 2n/(2n-1) ≈ 1 for large n.

But this is impossible for non-negative sequences. The actual limit is higher.

For uniform: M = S^2/n, score = 2.
The theoretical minimum for non-negative sequences is somewhere between 1 and 2.

The value 1.5 suggests M ≈ 1.5*S^2/(2n) = 0.75*S^2/n for large n. So the peak is about 75% of the uniform peak. That's a significant reduction.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating minimization (primary driver)
2. L-BFGS-B on softmax-smoothed objective (new)
3. Fine coordinate descent with incremental convolution (polishing)
4. Multi-scale search (n=512, 1024, 2048)
5. Diverse anti-symmetric initializations

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_alternating(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    last_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break
        
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Adaptive damping
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.6
        else:
            alpha = max(0.3, 5.0 / it)
        
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Stagnation detection with perturbation
        if best_score > last_best * (1 - 1e-10):
            stagnation += 1
        else:
            stagnation = 0
        last_best = best_score
        
        if stagnation > 20:
            # Try multiple perturbations
            best_pert_score = score_a
            best_pert_a = a.copy()
            for trial in range(5):
                noise_scale = 0.005 * (1 + trial * 0.5) / np.sqrt(n)
                a_pert = a + np.random.standard_normal(n) * noise_scale
                a_pert = np.maximum(a_pert, 0)
                sp = a_pert.sum()
                if sp < 1e-15:
                    continue
                a_pert = a_pert / sp
                sc = 2.0 * n * fast_convolve(a_pert).max()
                if sc < best_pert_score:
                    best_pert_score = sc
                    best_pert_a = a_pert.copy()
            
            if best_pert_score < best_score * 1.001:
                a = best_pert_a
                if best_pert_score < best_score:
                    best_score = best_pert_score
                    best_a = a.copy()
            stagnation = 0
        
        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.10f} M_lp={M_lp:.10f} alpha={alpha:.3f}")
    
    return best_a, best_score


def smoothed_objective(a, n, tau):
    """Softmax-smoothed version of the objective."""
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10
    # Use logsumexp for smooth max
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    log_sum = max_c / tau + np.log(np.sum(np.exp(shifted)))
    score = 2.0 * n * log_sum / (s * s)
    return score


def smoothed_gradient(a, n, tau):
    """Gradient of the smoothed objective."""
    conv = fast_convolve(a)
    s = a.sum()
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w = w / w.sum()
    
    # Gradient of logsumexp(c/tau) w.r.t. a[j]
    # = (1/tau) * sum_k w[k] * d(c[k])/d(a[j])
    # d(c[k])/d(a[j]) = 2*a[k-j] (for valid k-j, j in [0,n-1])
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-14:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        grad[i_start:i_end+1] += w[k] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
    grad = grad / tau
    
    # Gradient of full objective: 2n * logsumexp / s^2
    # = (2n/s^2) * grad_logsumexp - (4n * logsumexp / s^3) * ones
    log_sum = max_c / tau + np.log(np.sum(np.exp(shifted)))
    result = (2.0 * n / (s * s)) * grad - (4.0 * n * log_sum / (s * s * s)) * np.ones(n)
    return result


def lbfgs_smoothed(a0, time_limit, tau_init=0.01, verbose=True):
    """L-BFGS-B optimization on the smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    tau = tau_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Define objective and gradient with current tau
        def obj(v):
            v = np.maximum(v, 0)
            return smoothed_objective(v, n, tau)
        
        def grad(v):
            v = np.maximum(v, 0)
            return smoothed_gradient(v, n, tau)
        
        # Use L-BFGS-B with bounds
        bounds = [(0, None)] * n
        result = minimize(
            obj, a, jac=grad, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 500}
        )
        
        a_new = np.maximum(result.x, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        it += 1
        elapsed = time.monotonic() - t0
        if verbose:
            print(f"    L-BFGS iter={it} ({elapsed:.0f}s): best={best_score:.10f} tau={tau:.4f} result_sc={sc:.10f}")
        
        # Decrease tau (annealing)
        tau *= 0.7
        
        if sc > best_score * (1 + 1e-6):
            # Didn't improve, try with current best
            a = best_a.copy()
        else:
            a = a_new.copy()
        
        if tau < 1e-6:
            break
    
    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-16:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            # Fine multiplicative steps
            for factor in [0.999, 0.998, 0.997, 0.996, 0.995, 1.005, 1.006, 1.007, 1.008, 1.009, 1.010]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def symmetric_init(n, rng, asymmetry=0.3):
    """Create an initialization with controlled asymmetry."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    x = (t - center) / center
    
    # Base: cosine window
    base = 0.5 * (1 + np.cos(np.pi * x))
    
    # Add asymmetry: boost one side
    asym = asymmetry * np.sign(x) * (1 - np.abs(x))
    a = base + asym
    
    # Add some noise
    a = a + rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, init_type=0):
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        # Asymmetric: high left, low right
        t = np.arange(n)
        x = t / (n - 1)
        a = 0.2 + 0.8 * (1 - x) ** rng.uniform(1.5, 3.0)
    elif init_type == 4:
        # Two humps (asymmetric positions)
        t = np.arange(n)
        c1 = n * rng.uniform(0.15, 0.3)
        c2 = n * rng.uniform(0.7, 0.85)
        w = n * rng.uniform(0.08, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + 0.5 * np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 5:
        # S-curve (sigmoid-like)
        t = np.arange(n)
        x = (t - n/2) / (n * 0.2)
        a = 0.3 + 0.7 / (1 + np.exp(-x))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating minimization (primary driver) with improved damping
2. L-BFGS-B on softmax-smoothed objective for diverse local minima

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def smoothed_objective_and_grad(a, n, tau):
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -40, 40)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(conv)) / len(conv)
        w_sum = 1.0
    else:
        w = w / w_sum
    log_sum = max_c / tau + np.log(w_sum)

    # Score
    score = 2.0 * n * log_sum / (s * s)

    # Gradient of logsumexp w.r.t. a
    grad_lse = np.zeros(n)
    k_active = np.where(w > 1e-10)[0]
    for k in k_active:
        i_start = max(0, int(k) - n + 1)
        i_end = min(n - 1, int(k))
        grad_lse[i_start:i_end + 1] += w[int(k)] * 2.0 * a[int(k) - i_end:int(k) - i_start + 1][::-1]
    grad_lse = grad_lse / tau

    # Full gradient: d/d(a) [2n * LSE / s^2]
    grad = (2.0 * n / (s * s)) * grad_lse - (4.0 * n * log_sum / (s * s * s)) * np.ones(n)
    return score, grad


def lbfgs_smoothed(a0, time_limit, tau_init=0.008, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    tau = tau_init
    outer_it = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def obj(v):
            v = np.maximum(v, 0)
            s_v = v.sum()
            if s_v < 1e-15:
                return 1e10
            sc, _ = smoothed_objective_and_grad(v, n, tau)
            return sc

        def grad(v):
            v = np.maximum(v, 0)
            s_v = v.sum()
            if s_v < 1e-15:
                return np.zeros(n)
            _, g = smoothed_objective_and_grad(v, n, tau)
            return g

        max_iter = min(300, int(remaining * 50))
        result = minimize(
            obj, a, jac=grad, method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': int(max_iter * 3)}
        )

        a_new = np.maximum(result.x, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        outer_it += 1
        elapsed = time.monotonic() - t0
        if verbose:
            print(f"    LBGFS iter={outer_it} ({elapsed:.0f}s): best={best_score:.10f} tau={tau:.5f}")

        # Anneal tau
        tau *= 0.6
        if tau < 1e-7:
            break

        # Use best as next starting point if no improvement
        a = best_a.copy()

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Very fine multiplicative steps
            for factor in [0.9995, 0.999, 0.9985, 0.998, 0.997, 0.996, 1.004, 1.005, 1.006, 1.007, 1.008, 1.009, 1.010]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    if init_type == 0:
        # Triangle
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        # U-shape
        t = np.arange(n)
        center
