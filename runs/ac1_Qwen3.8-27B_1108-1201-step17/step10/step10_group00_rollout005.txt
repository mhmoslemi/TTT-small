import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        js = max(0, 0)
        je = min(n, size - j)
        T[j:min(size, j + n), j] = a[:min(n, size - j)]
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CBC, cp.CPLEX, cp.XPRESS, cp.PDLP, cp.CLARABEL, cp.GLOP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, warm_start=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            if warm_start is not None:
                b_var.value = warm_start
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp(np.clip((b - max_b) / T_temp, -500, 0))
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        idx = np.arange(js, je)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def nesterov_subgradient(a0, time_limit, active_frac=0.002):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Track running average for Polyak step size
    f_avg = best_score
    f_count = 0
    f_sum = 0.0

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        current_score = 2.0 * n * max_b

        # Update running average
        f_sum += current_score
        f_count += 1
        f_avg = f_sum / f_count

        # Active set: peaks within active_frac of max
        threshold = max_b * (1.0 - active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient (uniform average over active peaks)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n, k + 1)
            idx = np.arange(js, je)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)

        # Step size: Polyak-like
        f_lower = 2.0 * n * (1.0 / (2.0 * n - 1.0))  # theoretical lower bound
        gap = f_avg - f_lower
        if gap < 1e-15:
            gap = 1e-15
        eta = gap / (np.dot(g, g) + 1e-30)
        eta = min(eta, 1.0)

        # Nesterov momentum
        alpha = 2.0 / (it + 2)
        y_new = y - eta * g
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
            sy = 1.0
        y_new = y_new / sy

        # Nesterov extrapolation
        t_curr = (it + 1) * alpha + 1.0
        t_prev = it * (2.0 / (it + 1)) + 1.0 if it > 0 else 1.0
        beta = (t_curr - 1.0) / t_prev if t_prev > 0 else 1.0
        beta = min(beta, 1.5)  # cap for stability

        y = (1.0 - beta) * y + beta * y_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_new.copy()
        else:
            y = y / sy

        # Compute score
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it+1} ({elapsed:.0f}s): score={best_score:.10f} eta={eta:.6f}")

    return best_a, best_score

        lp_times.append(time.monotonic() - lp_t0)

        if b_new is not None:
            warm_b = b_new.copy()

        if b_new is None:
            continue

        # Nesterov step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (it + 2) * b_new) / (it + 1 + (it + 2))
        # Wait, that's not right. Let me use the standard Nesterov formula.

        # x_k = ((k-1) * x_{k-1} + k * best_response(y_k)) / k  ... hmm
        # Actually the standard is:
        # x_k = (sum_{i=1}^{k} λ_i z_i) / (sum λ_i) where λ_i = i(i+1)/2
        # Simpler: a = convex combination with increasing weight on new points
        a = np.maximum(a, 0)
        # Use running average with Nesterov weights

        lambda_prev = it * (it + 1) / 2.0 if it > 0 else 0.0
        if lambda_prev + lambda_k > 0:
            a = (lambda_prev * a + lambda_k * b_new) / (lambda_prev + lambda_k)
        
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stall detection
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:

            else:

            if stall_count >= 20:

                freq = rng.integers(1, 8)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) if len(lp_times) >= 10 else 0
                print(f"    NLP it={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp*1000:.0f}ms")

        # Speed check
        if len(lp_times) > 50 and np.mean(lp_times[-50:]) > 5.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=2000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(10, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_):
            obj, _ = logsumexp_obj_grad(z_, n, stage_T)
            return obj

        def jac(z_):
            _, g = logsumexp_obj_grad(z_, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def coordinate_descent(a0, time_limit, max_iter=1000000):
    """2D coordinate descent: change pairs (i,j) to maintain sum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum = 1

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    # Precompute: for each pair (i,j), the effect of transferring δ from j to i
    # conv changes: for lag k, Δc_k = 2*(a[i]-a[j])*a[k-i_or_j] approximately
    # Better: just do a local search

    # Simple approach: at each step, pick random pair (i,j) and try to transfer mass
    # Use golden section or ternary search for the optimal transfer amount

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[i] < 1e-10 or a[j] < 1e-10:
            continue

        # Try transferring δ from j to i: a[i] += δ, a[j] -= δ
        # Optimal δ is in [-a[j], a[i] + something]... actually δ ∈ [-a[j], ∞) but a[j]-δ ≥ 0 so δ ≤ a[j]
        # And a[i]+δ ≥ 0 is always satisfied for δ ≥ -a[i]
        # So δ ∈ [-a[i], a[j]]

        lo = -a[i] * 0.9
        hi = a[j] * 0.9

        # Quick ternary search
        best_d = 0.0
        best_val = 2.0 * n * fast_convolve(a).max()
        
        # Sample a few values first
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9, -0.1, -0.3, -0.5]:
            d = frac * (hi - lo) + lo
            a_test = a.copy()
            a_test[i] += d
            a_test[j] -= d
            if a_test[i] < 0 or a_test[j] < 0:
                continue
            st = 2.0 * n * fast_convolve(a_test).max()
            if st < best_val:
                best_val = st
                best_d = d

        if best_d != 0 and best_val < best_score - 1e-12:
            a[i] += best_d
            a[j] -= best_d
            st = 2.0 * n * fast_convolve(a).max()
            if st < best_score:
                best_score = st
                best_a = a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent it={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def evaluate_local(a):
    """Evaluate sequence score locally (fast, assumes sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Blackman window
        a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
