import numpy as np
from scipy.fft import rfft, irfft

_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return 'ECOS'

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if cond.any():
        idx = int(np.max(np.where(cond)[0]))
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def solve_first_order_lp(a, top_k=30, solver=None):
    n = len(a)
    if solver is None:
        solver = get_lp_solver()
    conv = fast_convolve(a)
    k_active = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_active)[-k_active:]
    peaks = np.sort(peaks)

    import cvxpy as cp
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(delta) == 0, delta >= -a - 1e-12]

    for k in peaks:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            constraints.append(t >= conv[k])
        else:
            indices = np.arange(i_start, i_end + 1)
            coeff = a[k - indices]
            constraints.append(t >= conv[k] + 2.0 * (coeff @ delta))

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'SCS', 'PDLP', 'GLPK']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if delta.value is None or t.value is None:
        return None, None
    d = np.asarray(delta.value, dtype=np.float64)
    t_val = float(t.value)
    return d, t_val

def line_search_alpha(a, direction, current_score):
    n = len(a)
    max_alpha = 0.5
    neg_idx = direction < 0
    if neg_idx.any():
        ratios = a[neg_idx] / (-direction[neg_idx])
        max_alpha = min(max_alpha, 0.95 * float(np.min(ratios)))
    max_alpha = max(max_alpha, 1e-8)

    def score_at_alpha(al):
        a_new = a + al * direction
        if np.any(a_new < -1e-12):
            return float('inf')
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 0.01:
            return float('inf')
        conv = fast_convolve(a_new)
        return 2.0 * n * float(conv.max()) / (cs * cs)

    best_alpha = golden_section_search(score_at_alpha, 0.0, max_alpha, max_iter=30)
    return best_alpha

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            def _score(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(_score, 0.0, upper, max_iter=25)

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_a, best_score

        a_new = np.maximum(a_new, 0.0)

        else:
            a_new = a_new / cs

        if score_new < best_score - 1e-12:

def kkt_polish(a0, time_limit, solver=None, verbose=False):
    """Polish using KKT conditions: identify active set, solve linear system."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for outer in range(20):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        max_val = float(conv.max())
        tol = 1e-6 * max_val
        active = np.where(conv > max_val - tol)[0]

        if len(active) == 0:
            break
        if len(active) > 50:
            # Keep only top 50
            top_idx = np.argsort(conv)[-50:]
            active = top_idx

        # Solve: find a' such that:
        # 1. conv_{a'}[k] = max_val for k in active (approximately)
        # 2. sum(a') = 1
        # 3. a' >= 0
        # 4. Gradient condition: for i in support, sum_k lambda_k * 2*a'[k-i] = const

        # Simplified: just enforce equal peaks + sum constraint via coordinates
        # Use weighted least squares to flatten active peaks
        m = len(active)
        if m < 2:
            break

        # For small perturbations: conv_new[k] ≈ conv[k] + 2*sum_i a[k-i]*delta[i]
        # We want: conv[k] + 2*(A_k @ delta) = max_val for k in active
        # And: sum(delta) = 0

        # This gives m+1 equations, n unknowns. Use least squares.
        A_mat = np.zeros((m + 1, n))
        b_vec = np.zeros(m + 1)

        for idx, k in enumerate(active):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                indices = np.arange(i_start, i_end + 1)
                A_mat[idx, indices] = 2.0 * a[k - indices]
            b_vec[idx] = max_val - conv[k]

        A_mat[m, :] = 1.0
        b_vec[m] = 0.0

        # Solve least squares with nonnegativity: min ||A@delta - b||^2 s.t. a+delta >= 0
        # Use simple approach: unregularized LS, then clip
        try:
            delta, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        except Exception:
            break

        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        cs = a_new.sum()
        if cs < 1e-10:
            break
        a_new = a_new / cs

        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            if verbose:
                print(f"    KKT {outer}: {best_score:.10f}")
        else:
            # Try damped version
            for damp in [0.5, 0.25, 0.1]:
                a_try = a + damp * delta
                a_try = np.maximum(a_try, 0.0)
                cs2 = a_try.sum()
                if cs2 < 1e-10:
                    continue
                a_try = a_try / cs2
                sc2 = compute_score(a_try)
                if sc2 < best_score - 1e-13:
                    best_score = sc2
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                break

    return best_a, best_score

def init_asymmetric(n, rng, shift=0.3):

    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
   amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_from_parent(parent, n, rng):
    if n == len(parent):
        return parent.copy()
    old_n = len(parent)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    p_norm = parent / parent.sum()
    f_interp = np.interp(new_x, old_x, p_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-10:
        f_interp = f_interp / ss
    # Add small noise to break symmetry
    noise = rng.standard_normal(n) * 0.005
    f_interp = f_interp + noise
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-10:
        f_interp = f_interp / ss
    return f_interp

def basin_hopping(a0, time_limit, rng, solver=None, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        # Generate perturbation
        pert_type = count % 3
        perturbed = a.copy()

        if pert_type == 0:
            # Gaussian noise
            sigma = 0.01 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Peak-directed: reduce mass near high-convolution lags
            conv_p = fast_convolve(a)
            top_k = min(10, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 5)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                perturbed[picks] *= (1.0 - 0.03 * rng.random())
        else:
            # Smooth perturbation: mix with shifted version
            shift = rng.integers(1, max(2, n // 10))
            shifted = np.roll(a, shift)
            perturbed = 0.8 * a + 0.2 * shifted

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Optimize the perturbation
        t_alloc = min(rem * 0.3, 60.0)
        a_opt, s_opt = exact_coord_descent(perturbed, t_alloc)
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: {best_score:.8f}")

    return best_a, best_score

def propose_candidate():

    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses first-order LP directions, coordinate descent, KKT polishing,
    basin hopping, and multi-length exploration.

    deadline = time.monotonic() + 1785.0

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Phase 1: Diverse initialization and quick screening
    print("\n=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    n_candidates = set()
    if parent is not None:
        n_candidates.add(len(parent))
    for nn in [512, 768, 1024, 1536, 2048]:
        n_candidates.add(nn)
    n_candidates = sorted([nn for nn in n_candidates if 256 <= nn <= 4096])

    candidates = []
    for n in n_candidates[:6]:
        if time.monotonic() - t_start > 100:
            break

            inits.append(init_from_parent(parent, n, rng))
        inits.append(init_asymmetric(n, rng, 0.3))
        inits.append(init_asymmetric(n, rng, 0.7))

        for j, init in enumerate(inits[:4]):
            t_left = 100.0 - (time.monotonic() - t_start)
            if t_left < 3:

            t_alloc = max(2.0, min(8.0, t_left / 5.0))
            a_opt, s_opt = exact_coord_descent(init, t_alloc)
            candidates.append((s_opt, a_opt, n))
            print(f"  n={n} init={j}: score={s_opt:.8f}")

    candidates.sort(key=lambda x: x[0])
    for s_c, a_c, n_c in candidates[:3]:
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST p1: {sc:.8f} (n={n_c})")

    if best_seq is None:
        best_seq = np.ones(1024) / 1024.0

    # Phase 2: LP-based optimization on best sequence

    rem = deadline - time.monotonic() - 600
