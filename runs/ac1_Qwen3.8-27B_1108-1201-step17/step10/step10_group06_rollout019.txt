import numpy as np

_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue

    return 'ECOS'

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, max_time=10)
    except Exception:

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            # Try to find best x using quadratic approximation + refinement
            # First try: x=0, x=v*0.5, x=v*1.5, and current best from quick scan
            candidates = [0.0, v * 0.5, v * 1.5, v]
            # Also try based on gradient direction
            grad_est = 0.0
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    grad_est += 2.0 * v * (1.0 if conv[k] == float(conv[k_lo:k_hi+1].max()) else 0.1)
                else:
                    grad_est += 2.0 * a_partners[idx] * (1.0 if conv[k] == float(conv[k_lo:k_hi+1].max()) else 0.1)
            candidates.append(max(0.0, v - 0.1 * v * np.sign(grad_est)))

            best_x = v
            best_sc = best_score

            for cx in candidates:
                cx = max(0.0, min(cx, 5.0 / n))
                nv = c_vals + 2.0 * cx * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + cx * cx
                ns = current_sum + (cx - v)
                if ns < 0.01:
                    continue
                nm = max(unaff_max, float(nv.max()))
                sc = 2.0 * n * nm / (ns * ns)
                if sc < best_sc:
                    best_sc = sc
                    best_x = cx

            # Refine with golden section around best candidate
            lo = max(0.0, best_x - 0.3 * v)
            hi = min(5.0 / n, best_x + 0.3 * v)
            if hi - lo < 1e-15:
                hi = best_x + 0.1 * v + 1e-10
                lo = max(0.0, best_x - 0.1 * v)

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)

            def _eval_x(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            fc = _eval_x(c_pt)
            fd = _eval_x(d_pt)
            for _ in range(25):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = _eval_x(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = _eval_x(d_pt)

            best_x = (lo + hi) * 0.5
            best_sc2 = _eval_x(best_x)
            if best_sc2 < best_sc:
                best_sc = best_sc2

            if abs(best_x - v) > 1e-13 and best_sc < best_score:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 20 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def subgradient_descent(a0, time_limit, verbose=False):
    n = len(a0)

        k_act = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.005 * max_val
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            partners = k - j
            contrib = 2.0 * a[partners]
            # Handle the case where j == k-j (diagonal)
            diag_mask = (j == partners)
            if diag_mask.any():
                contrib[diag_mask] = a[partners[diag_mask]]
            grad[j] += w[idx] * contrib

        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        sc = 2.0 * n * float(fast_convolve(a_new).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0

            no_improve += 1

        sc_avg = 2.0 * n * float(fast_convolve(a_avg).max())
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        a = a_new
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
        if no_improve > 300:
            break
        lr = max(lr * 0.9995, 0.0001)

    return best_a, best_score

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_chirp(n, rng, offset=0.0):
    """Initialize with a chirp-like structure."""
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.3, 0.7)
    phase = offset + rng.uniform(0, 2 * np.pi)
    a = 1.0 + 0.8 * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 0.1)
    s = a.sum()
    return a / s

def init_two_bump(n, rng, p1=0.3, p2=0.7, w1=0.15, w2=0.15):
    """Two asymmetric Gaussian bumps."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-0.5 * ((t - p1) / w1) ** 2) + np.exp(-0.5 * ((t - p2) / w2) ** 2)
    a = np.maximum(a, 1e-6)
    s = a.sum()
    return a / s

def init_flat_spectrum(n, rng):
    """Try to create a sequence with flat DFT magnitude."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    a = irfft(np.exp(1j * phases), n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_power_law(n, rng, center_ratio=0.5):
    """Power-law shaped initialization."""
    t = np.arange(n, dtype=np.float64) / n
    center = center_ratio
    alpha = rng.uniform(0.3, 1.5)
    a = 1.0 / (1.0 + alpha * np.abs(t - center) ** 2)
    s = a.sum()
    return a / s

def peak_deflation_basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping guided by peak deflation: perturb positions contributing to peaks."""
    n = len(a0)

            idx = np.arange(i_start, i_end + 1)

            # Pick positions that contribute most to this peak

            top_idx = idx[np.argmax(contrib)]
            # Perturb: reduce this position, add to a random other position
            reduction = rng.uniform(0.02, 0.08) * a[top_idx]
            perturbed[top_idx] -= reduction
            # Add to a position that contributes to a low-convolution area
            low_k = int(rng.choice(np.argpartition(conv, k_act)[:k_act]))
            low_i_start = max(0, low_k - n + 1)
            low_i_end = min(n - 1, low_k)
            if low_i_end >= low_i_start:
                add_idx = int(rng.randint(low_i_start, low_i_end + 1))
                perturbed[add_idx] += reduction

        # Also add some noise
        sigma = rng.uniform(0.005, 0.02)
        perturbed += rng.standard_normal(n) * sigma
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc)
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    PH {count}: improved to {best_score:.8f}")

    return best_a, best_score

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Phase 1: Diverse initialization across multiple n
    print("\n=== Phase 1: Diverse Init ===")
    t1_start = time.monotonic()
    n_parent = len(parent) if parent is not None else 1152

    n_values = [n_parent]
    for alt in [1024, 1152, 1280, 1400, 1536, 1600, 2048]:
        if alt != n_parent and 512 <= alt <= 2560:
            n_values.append(alt)
    n_values = list(set(n_values))[:6]

    candidates = []
    for n in n_values:
        if time.monotonic() - t1_start > 150:
            break
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
            inits.append(interpolate_seq(parent, n) + rng.standard_normal(n) * 0.05)
        inits.append(init_two_bump(n, rng, 0.25, 0.75, 0.12, 0.18))
        inits.append(init_two_bump(n, rng, 0.35, 0.65, 0.1, 0.25))
        inits.append(init_chirp(n, rng))
        inits.append(init_flat_spectrum(n, rng))
        inits.append(init_power_law(n, rng, 0.3))
        inits.append(np.ones(n) / n)

        for j, cand in enumerate(inits):
            t_left = 150.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(2.0, min(8.0, t_left / 8.0))
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc)
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates")
    for s_c, a_c, n_c in candidates[:5]:
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST p1: {sc:.8f} (n={n_c})")

    n_work = len(best_seq) if best_seq is not None else 1152
    print(f"  Working with n={n_work}, current best={best_score:.8f}")

    # Phase 2: Subgradient pre-optimization
    print(f"\n=== Phase 2: Subgradient Pre-opt ===")
    rem = deadline - time.monotonic() - 900
    if rem > 20:
        t_sg = min(rem * 0.3, 120.0)
        a_sg, s_sg = subgradient_descent(best_seq, t_sg)
        sc = float(evaluate_sequence(a_sg.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_sg.copy()
            print(f"  NEW BEST SG: {sc:.8f}")

    # Phase 3: LP Fictitious Play (main driver)
    print(f"\n=== Phase 3: LP Fictitious Play ===")
    rem = deadline - time.monotonic() - 700
    if rem > 30:
        t_fp = min(rem * 0.6, 450.0)
        print(f"  FP budget={t_fp:.0f}s")
        a_fp, s_fp, iters = lp_fictitious_play(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            print(f"  NEW BEST FP: {sc:.8f}")

        # Try at different lengths
        for alt_n in [n_work + 96, n_work - 96, n_work + 192]:
            if alt_n < 512 or alt_n > 2560:
                continue
            rem2 = deadline - time.monotonic() - 500
            if rem2 < 30:
                break
            t_fp2 = min(rem2 * 0.4, 150.0)
            a_init = interpolate_seq(best_seq, alt_n)
            a_fp2, s_fp2, _ = lp_fictitious_play(a_init, t_fp2, solver=solver)
            sc2 = float(evaluate_sequence(a_fp2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_fp2.copy()
                n_work = alt_n
                print(f"  NEW BEST FP n={alt_n}: {sc2:.8f}")

    # Phase 4: Exact Coordinate Descent Polishing
    print(f"\n=== Phase 4: ECD Polishing ===")
    n_work = len(best_seq)
    for rnd in range(4):
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_cd = min(rem * 0.25, 150.0)
        print(f"  ECD round {rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd, sw = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph4_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD: {sc:.8f}")
        else:
            if rnd == 0:
                # Perturb to escape local minimum
                perturbed = best_seq + rng.standard_normal(n_work) * 0.015
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                a_cd2, s_cd2, _ = exact_coord_descent(perturbed, min(t_cd * 0.5, 60.0))
                sc2 = float(evaluate_sequence(a_cd2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_cd2.copy()
                    print(f"  NEW BEST ECD-pert: {sc2:.8f}")

    # Phase 5: Peak-deflation Basin Hopping
    print(f"\n=== Phase 5: Peak Deflation BH ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_bh = min(rem * 0.7, 300.0)
        print(f"  BH budget={t_bh:.0f}s")
        a_bh, s_bh = peak_deflation_basin_hopping(best_seq, t_bh, rng, verbose=True)
        sc = float(evaluate_sequence(a_bh.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bh.copy()
            print(f"  NEW BEST BH: {sc:.8f}")

    # Phase 6: Final polishing
    print(f"\n=== Phase 6: Final Polish ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_fine = min(rem * 0.5, 100.0)
        print(f"  Final ECD budget={t_fine:.0f}s")
        a_fine, s_fine, _ = exact_coord_descent(best_seq, t_fine, verbose=True, label="final")
        sc = float(evaluate_sequence(a_fine.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fine.copy()
            print(f"  NEW BEST final: {sc:.8f}")

    rem = deadline - time.monotonic() - 10
    if rem > 10:
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.7, 40.0), label="last")
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()
            print(f"  NEW BEST last: {sc:.8f}")

    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
