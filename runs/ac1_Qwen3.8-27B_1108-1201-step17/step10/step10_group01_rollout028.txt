    # = (a * w)[i] in the "cross-correlation" sense
    # Using FFT: if we define W_full of length 2n-1 = w, and pad a to length 2n-1:
    # Actually, the simplest: grad_a = np.correlate(w, a, mode='valid')[:n]... no
    # Let me just use the formula directly with FFT:
    # grad_a[i] = sum_{j=0}^{n-1} a[j] w[i+j]
    # Make w_padded of length 2n-1, a of length n (padded to 2n-1)
    # This is the cross-correlation at lag i
    # corr[i] = sum_j a[mapped_j] w[i + j]
    # Hmm, let me just do it the straightforward way for correctness:
    # Actually the cleanest FFT approach:
    # Define a_rev[i] = a[n-1-i] (reverse a)
    # Then grad_a[i] = sum_{j=0}^{n-1} a[j] w[i+j]
    # = sum_{j=0}^{n-1} a_rev[n-1-j] w[i+j]
    # Let k = n-1-j, j = n-1-k: sum_{k=0}^{n-1} a_rev[k] w[i+n-1-k]
    # = (a_rev * w)[i+n-1] (linear convolution)
    a_rev = a[::-1]
    conv_aw = fast_convolve_full(a_rev, w)
    # conv_aw has length n + (2n-1) - 1 = 3n - 2
    # We want positions i+n-1 for i=0..n-1
    return conv_aw[n - 1: 2 * n - 1]


def fast_convolve_full(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def lse_objective_grad(z, n, T):
    """Compute objective and gradient of LogSumExp-smoothed problem in log space.
    a = softmax(z), objective = 2n * (1/T) * log(sum exp(T * conv(a,a)_k))
    """
    # Compute a from z
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    T_b = T * (b - max_b)
    exp_Tb = np.exp(T_b)
    lse = max_b + np.log(exp_Tb.sum()) / T
    obj = 2.0 * n * lse
    
    # Weights
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient wrt a: grad_a[i] = 2n * sum_k w[k] * a[k-i] (for valid k-i)
    # Using FFT: grad_a = 2n * cross_corr(w, a)
    a_rev = a[::-1]
    conv_result = fast_convolve_full(a_rev, w)
    grad_a = 2.0 * n * conv_result[n - 1: 2 * n - 1]
    
    # Gradient wrt z: a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_objective_only(z, n, T):
    """Objective only (for L-BFGS which needs separate func and jac)."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    T_b = T * (b - max_b)
    exp_Tb = np.exp(T_b)
    lse = max_b + np.log(exp_Tb.sum()) / T
    return 2.0 * n * lse


def lse_grad_only(z, n, T):
    """Gradient only."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    T_b = T * (b - max_b)
    exp_Tb = np.exp(T_b)
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1]
    conv_result = fast_convolve_full(a_rev, w)
    grad_a = 2.0 * n * conv_result[n - 1: 2 * n - 1]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def optimize_lse_continuation(a0, n, time_limit, print_tag=""):
    """Optimize using LogSumExp continuation with L-BFGS."""
    t0 = time.monotonic()
    z = np.log(np.maximum(a0, 1e-300))
    best_a = a0.copy()
    best_score = evaluate_local(a0)
    
    # Temperature schedule: adaptive
    T = 5.0
    max_T = 5000.0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Determine max iterations for this stage
        max_iters = min(int(remaining * 20), 3000)
        
        def func(z_local):
            return lse_objective_only(z_local, n, T)
        
        def jac(z_local):
            return lse_grad_only(z_local, n, T)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': max_iters * 50, 'disp': False}
            )
            z = res.x
        except Exception:
            break
        
        # Evaluate actual objective
        a = np.exp(z - z.max())
        a = a / a.sum()
        score = evaluate_local(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        a = best_a  # Always continue from best
        
        # Check convergence of LSE to true max
        b = fast_convolve(a)
        max_b = b.max()
        sorted_b = np.sort(b)[::-1]
        # Gap between LSE and true max
        T_b = T * (b - max_b)
        exp_Tb = np.exp(T_b)
        lse_val = max_b + np.log(exp_Tb.sum()) / T
        gap = lse_val - max_b
        
        # If gap is small enough, we're converged
        if gap < 1e-10 * max(1, max_b):
            break
        
        # Adaptive temperature increase
        # If we made progress, increase T more aggressively
        if score < best_score + 1e-12:
            T *= 2.0
        else:
            T *= 1.5
        
        if T > max_T:
            break
        
        # Check time
        if time.monotonic() - t0 > time_limit * 0.85:
            break
    
    # Final evaluation
    a = np.exp(z - z.max())
    a = a / a.sum()
    score = evaluate_local(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def peak_coord_descent(a0, time_limit, top_k=5, print_tag=""):
    """Coordinate descent targeting top convolution peaks for final refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall_count = 0
    rng = np.random.default_rng(123)
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        max_b = b.max()
        
        # Get top peaks
        peak_indices = np.argsort(b)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            # Contribution of a[pos] to this peak: a[pos] * a[pk - pos]
            contribs = a[idx] * a[pk - idx]
            # Try reducing top contributors
            top_contrib = np.argsort(contribs)[::-1][:min(8, len(idx))]
            
            for ci in top_contrib:
                pos = idx[ci]
                if a[pos] < 1e-10:
                    continue
                # Try several reduction factors
                for factor in [0.995, 0.99, 0.985, 0.98, 0.97, 0.95]:
                    reduction = a[pos] * (1.0 - factor)
                    if reduction < 1e-15:
                        continue
                    
                    # Redistribute reduction proportionally to other elements
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    
                    a_trial = a.copy()
                    a_trial[pos] -= reduction
                    a_trial[mask] += reduction * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    
                    new_score = evaluate_local(a_trial)
                    if new_score < best_score - 1e-13:
                        a = a_trial
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                        stall_count = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try simultaneous small perturbations
            peak = peak_indices[0]
            js = max(0, peak - n + 1)
            je = min(n - 1, peak)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak - idx]
            top5 = idx[np.argsort(contribs)[::-1][:5]]
            
            for attempt in range(3):
                a_trial = a.copy()
                for pos in top5:
                    a_trial[pos] *= (0.99 - 0.005 * rng.random())
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                new_score = evaluate_local(a_trial)
                if new_score < best_score - 1e-14:
                    a = a_trial
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    stall_count = 0
                    break
            
            if not improved:
                # Try expanding non-contributing elements
                non_contrib = np.setdiff1d(np.arange(n), top5)
                if len(non_contrib) > 0:
                    expand_pos = non_contrib[rng.integers(len(non_contrib))]
                    if a[expand_pos] < 0.5 / n:  # Don't make it too large
                        a_trial = a.copy()
                        a_trial[expand_pos] *= 1.01
                        a_trial = a_trial / a_trial.sum()
                        new_score = evaluate_local(a_trial)
                        if new_score < best_score - 1e-14:
                            a = a_trial
                            best_score = new_score
                            best_a = a.copy()
                            improved = True
                            stall_count = 0
        
        if not improved:
            stall_count += 1
            if stall_count > 200:
                break
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {print_tag} CD {elapsed:.0f}s: score={best_score:.8f} stall={stall_count}")
    
    return best_a, best_score


def projected_gradient_optimize(a0, n, time_limit, print_tag=""):
    """Projected gradient with Barzilai-Borwein step size on the smoothed objective."""
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    T = 20.0
    max_T = 3000.0
    prev_a = a.copy()
    prev_grad = None
    bb_alpha = 0.1
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Compute gradient at current a
        b = fast_convolve(a)
        max_b = b.max()
        T_b = T * (b - max_b)
        exp_Tb = np.exp(T_b)
        w = exp_Tb / exp_Tb.sum()
        
        # grad_a[i] = 2n * sum_k w[k] * a[k-i]
        a_rev = a[::-1]
        conv_result = fast_convolve_full(a_rev, w)
        grad_a = 2.0 * n * conv_result[n - 1: 2 * n - 1]
        
        # Projected gradient step
        if prev_grad is not None:
            # BB step size
            s_vec = a - prev_a
            y_vec = grad_a - prev_grad
            sS = np.dot(s_vec, s_vec)
            sY = np.dot(s_vec, y_vec)
            if sS > 1e-20 and abs(sY) > 1e-15:
                bb_alpha = sS / sY
                bb_alpha = np.clip(bb_alpha, 0.01, 10.0)
        
        # Gradient step with damping
        damping = min(1.0, 1.0 / (np.linalg.norm(grad_a) + 1e-10))
        step_size = bb_alpha * damping
        a_trial = a - step_size * grad_a
        a_trial = project_simplex(a_trial)
        
        score = evaluate_local(a_trial)
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
        
        # Backtracking if needed
        if score > best_score:
            # Try smaller step
            for backtrack in [0.5, 0.1, 0.01]:
                a_trial2 = a - step_size * backtrack * grad_a
                a_trial2 = project_simplex(a_trial2)
                score2 = evaluate_local(a_trial2)
                if score2 < best_score:
                    a = a_trial2
                    best_score = score2
                    best_a = a.copy()
                    break
            else:
                a = best_a.copy()
        else:
            a = a_trial
        
        prev_a = a.copy()
        prev_grad = grad_a.copy()
        
        # Adaptive temperature
        if it % 100 == 0:
            lse_val = max_b + np.log(exp_Tb.sum()) / T
            gap = lse_val - max_b
            if gap < 1e-9 * max(1, max_b):
                break
            if T < 100:
                T *= 2.0
            elif T < 1000:
                T *= 1.5
            else:
                T *= 1.2
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {print_tag} PG {elapsed:.0f}s: score={best_score:.8f} T={T:.1f}")
    
    return best_a, best_score


def generate_initialization(n, rng, style=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if style == 0:
        # Linear increasing
        a = x + 0.1
    elif style == 1:
        # Quadratic increasing
        a = x**2 + 0.1
    elif style == 2:
        # Exponential
        a = np.exp(2 * x)
    elif style == 3:
        # Skewed bell (peak at 0.3)
        a = np.exp(-5 * (x - 0.3)**2) + 0.1
    elif style == 4:
        # Skewed bell (peak at 0.7)
        a = np.exp(-5 * (x - 0.7)**2) + 0.1
    elif style == 5:
        # Cosine half-wave (symmetric)
        a = np.cos(np.pi * x / 2)**2 + 0.05
    elif style == 6:
        # Power law
        a = (1 - x)**0.5 + 0.1
    elif style == 7:
        # Sine weighted
        a = np.sin(np.pi * x)**0.5 + 0.1
    elif style == 8:
        # Very skewed
        a = x**3 + 0.05
    elif style == 9:
        # Random smooth
        k = rng.integers(2, 6)
        freqs = rng.uniform(0.5, 3.0, k)
        phases = rng.uniform(0, 2*np.pi, k)
        amps = rng.uniform(0.3, 1.0, k)
        a = 0.5
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2*np.pi*f*x + p)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def nesterov_projected_gradient(a0, n, time_limit, print_tag=""):
    """Nesterov-accelerated projected gradient on the LSE-smoothed objective."""
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    T = 10.0
    max_T = 3000.0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Compute gradient at y
        b = fast_convolve(y)
        max_b = b.max()
        T_b = T * (b - max_b)
        exp_Tb = np.exp(T_b)
        w = exp_Tb / exp_Tb.sum()
        
        a_rev = y[::-1]
        conv_result = fast_convolve_full(a_rev, w)
        grad_y = 2.0 * n * conv_result[n - 1: 2 * n - 1]
        
        # Step size (Polyak-like)
        lr = 1.0 / (it + 1)
        lr = min(lr, 0.5 / (np.linalg.norm(grad_y) + 1e-10))
        
        # Gradient step from y
        a_new = y - lr * grad_y
        a_new = project_simplex(a_new)
        
        # Nesterov update for y
        theta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta = (1.0 - 1.0 / theta) / (1.0 + beta_prev) if it > 1 else 0.0
        # Simple Nesterov: y = a_new + (it/(it+2)) * (a_new - a)
        momentum = it / (it + 2)
        y = a_new + momentum * (a_new - a)
        y = project_simplex(y)
        
        a = a_new
        
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adaptive temperature
        if it % 50 == 0:
            lse_val = max_b + np.log(exp_Tb.sum()) / T
            gap = lse_val - max_b
            if gap < 1e-10 * max(1, max_b):
                break
            T = min(T * 1.8, max_T)
        
        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {print_tag} NG {elapsed:.0f}s: score={best_score:.8f} T={T:.1f}")
    
    return best_a, best_score


beta_prev = 1.0


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LogSumExp continuation, projected gradient, and peak-directed CD.
    """
    global beta_prev
    beta_prev = 1.0
    
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = generate_initialization(n_init, rng, style=0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: LSE continuation on parent ===
    print(f"\n=== Phase 1: LSE Continuation, n={n_parent}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_res, _ = optimize_lse_continuation(best_seq, n_parent, t1, "P1")
        try_update(a_res, "lse_p1")
    
    # === Phase 2: Projected gradient refinement ===
    print(f"\n=== Phase 2: Projected Gradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        a_res, _ = projected_gradient_optimize(best_seq, n_parent, t2, "P2")
        try_update(a_res, "pg_p2")
    
    # === Phase 3: LSE again (from improved point) ===
    print(f"\n=== Phase 3: LSE Pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        a_res, _ = optimize_lse_continuation(best_seq, n_parent, t3, "P3")
        try_update(a_res, "lse_p3")
    
    # === Phase 4: Peak coordinate descent ===
    print(f"\n=== Phase 4: Peak CD, score={best
