                #   c_k = sum_j a_j * a_{k-j} where both a_j and a_{k-j} are from the same vector a.
                #   dc_k/da_i = sum_j [da_j/da_i * a_{k-j} + a_j * da_{k-j}/da_i]
                #   = a_{k-i} + a_{k-i} = 2*a_{k-i} (when 0<=k-i<n)
                #   But if k-i = i (i.e., k=2i), then it's just a_i (one term).
                #   Actually no: dc_k/da_i = a_{k-i} (from j=i) + a_{k-i} (from j=k-i when k-i != i)
                #   If k-i = i, both terms give a_i, so total is 2*a_i still.
                #   Wait no. Let me index more carefully.
                #   c_k = sum_j a_j * a_{k-j}
                #   For a fixed i, a_i appears in c_k when:
                #     j = i: term is a_i * a_{k-i}
                #     k-j = i, i.e., j = k-i: term is a_{k-i} * a_i
                #   So if i != k-i (i.e., k != 2i), a_i appears twice, giving 2*a_i * ... 
                #   No wait: dc_k/da_i = [d/da_i](a_i * a_{k-i}) + [d/da_i](a_{k-i} * a_i)
                #   = (a_{k-i} + a_i * 0) + (a_i * 0 + a_{k-i})
                #   Hmm no. a_i and a_{k-i} are different components (unless i = k-i).
                #   dc_k/da_i = (d a_i/da_i) * a_{k-i} + a_i * (d a_{k-i}/da_i)
                #              + (d a_{k-i}/da_i) * a_i + a_{k-i} * (d a_i/da_i)  [if k-i != i]
                #   This is getting confusing. Let me just use:
                #   c_k = sum_j a_j * a_{k-j}
                #   dc_k/da_i: the only terms that depend on a_i are:
                #     j=i: a_i * a_{k-i} → derivative = a_{k-i}
                #     j=k-i: a_{k-i} * a_i → derivative = a_{k-i} (if k-i != i)
                #     if k-i == i: a_i * a_i → derivative = 2*a_i
                #   So: dc_k/da_i = a_{k-i} + (a_{k-i} if k-i != i and 0<=k-i<n else 0)
                #   = 2*a_{k-i} if k != 2i and 0<=k-i<n
                #   = 2*a_i if k = 2i
                #   In all cases where 0 <= k-i < n: dc_k/da_i = 2*a_{k-i}
                #   (because when k=2i, a_{k-i} = a_i, so 2*a_{k-i} = 2*a_i)
                #   
                #   So the gradient of c_k w.r.t. a is: (grad)_i = 2*a_{k-i} if 0<=k-i<n, else 0.
                #   Which is 2 * reverse(a) shifted by (k - (n-1)).
                pass  # This is handled below in the general case
        
        return objective, grad
    
    # For finite p
    c_p = c ** p
    sum_c_p = c_p.sum()
    if sum_c_p < 1e-300:
        return 0.0, np.zeros(len(a))
    
    objective = sum_c_p ** (1.0 / p)
    
    # Weights: w_k = c_k^(p-1) / (sum c^p)^(1/p - 1) / p
    # Actually: d f_p / da_i = (1/p) * (sum c^p)^(1/p - 1) * p * sum_k c_k^(p-1) * dc_k/da_i
    # = (sum c^p)^(1/p - 1) * sum_k c_k^(p-1) * 2 * a_{k-i}
    # = 2 * (sum c^p)^(1/p - 1) * (c^(p-1) * a)[i]  [convolution]
    
    c_pw1 = c ** (p - 1)
    # convolve c_pw1 with a
    # c_pw1 has length 2n-1, a has length n
    # (c_pw1 * a)[i] = sum_k c_pw1[k] * a[i-k]... 
    # Actually we need sum_k c_k^(p-1) * a_{k-i}
    # This is the cross-correlation of c_pw1 and a
    # = convolve(c_pw1, reverse(a))[n-1 + ...]
    
    # Let's compute it differently:
    # sum_k c_k^(p-1) * a_{k-i} for i = 0, ..., n-1
    # This is the convolution of c_pw1 (length 2n-1) with a (length n), 
    # evaluated at specific indices.
    # conv(c_pw1, a)[m] = sum_j c_pw1[j] * a[m-j]
    # We want sum_k c_pw1[k] * a[k-i] = conv(c_pw1, a)[k] with k-i = j... 
    # Let j = k-i, then k = i+j:
    # sum_j c_pw1[i+j] * a[j]
    # This is a cross-correlation.
    
    # Easier: reverse c_pw1 and convolve with a
    c_pw1_rev = c_pw1[::-1]  # length 2n-1
    # conv(c_pw1_rev, a) has length 2n-1 + n - 1 = 3n-1
    # We want the values at indices [n-1, n-1+n-1] = [n-1, 2n-2]
    conv_result = np.convolve(c_pw1_rev, a)
    # The i-th element of the gradient (before scaling) is conv_result[n-1+i]
    grad_unscaled = conv_result[n - 1: n - 1 + n]
    
    scale = 2.0 * sum_c_p ** (1.0 / p - 1.0)
    grad = scale * grad_unscaled
    
    return objective, grad


def fista_pnorm(a0, p, time_limit, max_iters=100000):
    """
    FISTA for minimizing the p-norm of conv(a,a) subject to a>=0, sum(a)=1.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Estimate Lipschitz constant
    # The gradient of f_p is Lipschitz with constant L ~ O(n * max(c)^(p-1))
    # Use adaptive step size
    gamma = 0.5  # initial step size
    c = fast_convolve(a)
    max_c = c.max()
    if max_c > 1e-15:
        gamma = 0.01 / (n * max_c ** (p - 1)) if p > 1 else 0.01
        gamma = min(gamma, 0.5)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        _, grad_y = pnorm_objective_and_grad(y, p)
        
        # Gradient step
        z = y - gamma * grad_y
        
        # Project onto simplex
        x_new = project_simplex(z)
        
        # Evaluate
        score = evaluate_local(x_new)
        if score < best_score:
            best_score = score
            best_a = x_new.copy()
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0)
        # Renormalize y to sum=1 (it's just a momentum point, should be on simplex)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        a = x_new
        t_k = t_new
        
        # Adaptive step size (backtracking)
        if it % 50 == 0 and it > 0:
            # Check if we're making progress
            pass
        
        if (it + 1) % 10000 == 0:
            print(f"    FISTA p={p} iter={it+1}: obj={pnorm_objective_and_grad(a, p)[0]:.6f} score={best_score:.8f}")
    
    return best_a, best_score


def mirror_descent_v2(a0, time_limit, eta0=0.05, active_frac=0.999):
    """
    Improved mirror descent with better step size scheduling.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    prev_a = a.copy()
    prev_g = None
    no_improve = 0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution at Nesterov point
        b = fast_convolve(y)
        max_b = b.max()
        
        # Find active peaks (multiple levels)
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient with weighting by proximity to max
        weights = b[active] - max_b * (1 - active_frac)  # proportional to how close to max
        weights = np.maximum(weights, 1e-10)
        weights = weights / weights.sum()
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[k - active[0] if k in active else 0] * a[k - idx]
        # Actually let me redo this more carefully
        g = np.zeros(n)
        for w, k in zip(weights, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        
        # Barzilai-Borwein step size
        if it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-15 and y_dot_y > 1e-15:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.7 * eta + 0.3 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-6, min(eta, 2.0))
        
        prev_a = a.copy()
        prev_g = g.copy()
        
        # Mirror descent update (exponential family)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        a_new = a_new / a_new_sum
        
        # Nesterov acceleration
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
            ys = 1.0
        y = y / ys
        
        a = a_new
        
        # Evaluate
        score = evaluate_local(a)
        score_y = 2.0 * n * fast_convolve(y).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic restart with noise if stuck
        if no_improve >= 300:
            rng = np.random.default_rng(it * 7 + 13)
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)
        
        if (it + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver_name=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [solver_name, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    if solver_name is None:
        solvers = solvers[1:]
    
    for solver in solvers:
        if solver is None:
            continue
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def alternating_lp(a0, time_limit, n_iters=100, solver_name=None):
    """
    Alternating LP optimization with Nesterov acceleration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    prev_best = best_score
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # LP step: find best response b given current y
        b_new, M_b = solve_lp_best_response(y, solver_name)
        if b_new is None:
            stall += 1
            if stall > 10:
                break
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Averaged iterate
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection
        if (it + 1) % 10 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                noise = rng.standard_normal(n) * 0.01
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                stall = 0
        
        if (it + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ALT-LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=50000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for peak_k in peaks[:3]:  # Only check top 3 peaks
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.997, 0.995, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                # Try simultaneous adjustments
                a_trial = a.copy()
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:5]]
                for pos in top_positions:
                    if a[pos] > 1e-12:
                        a_trial[pos] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    if stall > 2000:
                        break
        else:
            stall = 0
        
        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def pnorm_continuation(a0, time_limit, p_values=None):
    """
    p-norm continuation method: optimize p-norm objective for increasing p.
    Each stage uses FISTA.
    """
    n = len(a0)
    if p_values is None:
        p_values = [2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64]
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    n_stages = len(p_values)
    time_per_stage = time_limit / n_stages
    
    for i, p in enumerate(p_values):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_alloc = min(time_per_stage * 1.5, remaining * 0.9)
        if t_alloc < 2:
            break
        
        print(f"  p-norm stage p={p}, time_alloc={t_alloc:.1f}s")
        a_stage, _ = fista_pnorm(a, p, t_alloc, max_iters=50000)
        
        # Check if we improved
        sc = evaluate_local(a_stage)
        if sc < best_score:
            best_score = sc
            best_a = a_stage.copy()
            print(f"    NEW BEST at p={p}: {sc:.8f}")
        
        a = a_stage
    
    return best_a, best_score


def symmetric_refine(a0, time_limit):
    """
    Refine assuming symmetry: a[i] = a[n-1-i].
    Only optimize the first half.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Enforce symmetry as starting point
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / a_sym.sum()
    
    # Work with half the variables
    half = a_sym[:n // 2]
    
    best_a = a_sym.copy()
    best_score = evaluate_local(a_sym)
    t0 = time.monotonic()
    
    # Use mirror descent on the half-vector
    h = half.copy()
    sh = h.sum()
    if sh < 1e-15:
        h = np.ones(n // 2) / (n // 2)
        sh = 1.0
    h = h / sh
    
    eta = 0.05
    no_improve = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Reconstruct full vector
        if n % 2 == 0:
            a_full = np.concatenate([h, h[::-1]])
        else:
            a_full = np.concatenate([h, h[::-1]])
            a_full = a_full[:n]
        
        a_full = np.maximum(a_full, 0)
        sf = a_full.sum()
        if sf < 1e-15:
            continue
        a_full = a_full / sf
        
        # Compute gradient
        b = fast_convolve(a_full)
        max_b = b.max()
        threshold = max_b * 0.9999
        active = np.where(b >= threshold)[0]
        
        g_full = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_full[idx] += a_full[k - idx]
        
        # Project gradient onto half-space
        g_half = 0.5 * (g_full[:n//2] + g_full[n//2:][::-1])
        g_half = g_half / max(1, len(active))
        
        # Mirror descent update
        log_h = np.log(np.maximum(h, 1e-300))
        g_shifted = g_half - g_half.min()
        log_h_new = log_h - eta * g_shifted
        log_h_new -= log_h_new.max()
        h_new = np.exp(log_h_new)
        h_new = h_new / h_new.sum()
        
        h = h_new
        
        # Evaluate
        if n % 2 == 0:
            a_check = np.concatenate([h, h[::-1]])
        else:
            a_check = np.concatenate([h, h[::-1]])[:n]
        a_check = np.maximum(a_check, 0)
        a_check = a_check / a_check.sum()
        score = evaluate_local(a_check)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_check.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 200:
            rng = np.random.default_rng(it * 3 + 7)
            h = h + rng.standard_normal(len(h)) * 0.01
            h = np.maximum(h, 0)
            h = h / h.sum()
            no_improve = 0
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SymRefine iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def multi_restart_mirror(a0, time_limit, n_restarts=5):
    """
    Multiple restarts of mirror descent with different initial perturbations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    time_per_restart = time_limit / n_restarts
    
    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        t_alloc = min(time_per_restart, remaining * 0.95)
        
        # Perturb the starting point
        rng = np.random.default_rng(1000 + r * 7)
        if r == 0:
            a_start = a.copy()
        else:
            noise = rng.standard_normal(n) * (0.05 + 0.02 * r)
            a_start = a + noise
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        
        a_res, sc = mirror_descent_v2(a_start, t_alloc, 
                                       eta0=0.04 + 0.01 * r,
                                       active_frac=0.9995)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
            print(f"  Restart {r}: NEW BEST {sc:.8f}")
    
    return best_a, best_score

            parent = p / p.sum()
            p_score = evaluate_local(parent)
