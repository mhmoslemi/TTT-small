LogSumExp smoothing, multi-scale temperature cooling, LP-based refinement,
and multi-length exploration.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_gradients(a, t):
    """Compute gradient of smoothed objective using FFT cross-correlation."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    p = np.exp((b - max_b) / t)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(b))
        p[int(np.argmax(b))] = 1.0
        p_sum = 1.0
    p /= p_sum
    
    # Cross-correlation: c[i] = sum_k p[k] * a[k-i]
    # Using FFT: c = IFFT(conj(FFT(p)) * FFT(a))
    p_len = len(p)  # 2n-1
    a_len = n
    fft_size = 1 << (p_len + a_len - 2).bit_length()
    
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    C = irfft(np.conj(P) * A, fft_size)
    
    c = C[:n]
    grad = (4.0 * n / t) * c
    return grad, b, max_b


def adam_logsumexp(a0, time_limit, T_stages=None, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with LogSumExp smoothing and multi-stage temperature cooling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    if T_stages is None:
        T_stages = np.geomspace(20, 0.02, 25)
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adapt learning rate based on temperature
        current_lr = lr * min(1.0, T / 5.0 + 0.1)
        
        while time.monotonic() - t0 < time_limit:
            step += 1
            
            grad, b, max_b = compute_gradients(a, T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            
            a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            a = a_new
            
            # Check score periodically
            if step % 100 == 0:
                score = evaluate_local(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            
            # Convergence check
            if np.max(np.abs(a_new - a)) < 1e-12:
                break
        
        # Evaluate at end of stage
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Print progress
        elapsed = time.monotonic() - t0
        print(f"    Adam T={T:.4f} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def lp_best_response(a_fixed, solver=None, max_iter=10000):
    """Solve LP: minimize M s.t. T@b <= M, sum(b)=1, b>=0."""
    if not HAS_CVXPY:
        return None, None
    
    if solver is None:
        solver = _get_lp_solver()
    
    n = len(a_fixed)

    M_var = cp.Variable()
    
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, max_iter=max_iter, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)

                if s < 1e-15:

def nesterov_lp(a0, time_limit, solver=None, max_iters=100000):
    """Nesterov-accelerated LP alternation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 20:
                break
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_coord_descent(a0, time_limit, n_steps=50000):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        # Try reducing top contributors
        top_idx = idx[np.argsort(contribs)[::-1][:5]]
        for pos in top_idx:
            if a[pos] < 1e-12:
                continue
            for factor in [0.99, 0.98, 0.97]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] * (1.0 - factor)
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated Adam with
LogSumExp smoothing, multi-scale temperature cooling, LP-based refinement,
beta-distribution initializations, and multi-length exploration.
Key innovation: O(n log n) gradient via FFT cross-correlation enables
thousands more optimization steps than the previous O(n^2) approach.

import numpy as np
import time
from scipy.fft import rfft, irfft

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False

_lp_solver_name = None


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    global _lp_solver_name
    if not HAS_CVXPY:
        return None
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return cp.ECOS


def compute_grad_fft(a, t):
    """Compute gradient of (2n/t)*LogSumExp(conv(a,a)/t) using FFT cross-correlation.
    Returns (gradient, b, max_b)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    p = np.exp((b - max_b) / t)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(b))
        p[int(np.argmax(b))] = 1.0
        p_sum = 1.0
    p /= p_sum

    # Cross-correlation: c[i] = sum_k p[k] * a[k-i] for i=0..n-1
    # c = IFFT(conj(FFT(p)) * FFT(a)) with zero-padding
    p_len = len(p)
    fft_size = 1 << (p_len + n - 2).bit_length()

    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    C = irfft(np.conj(P) * A, fft_size)

    c = C[:n]
    grad = (4.0 * n / t) * c
    return grad, b, max_b


def adam_logsumexp(a0, time_limit, T_stages=None, lr=0.004, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with LogSumExp smoothing and multi-stage temperature cooling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

        a = np.ones(n) / n
    a = a / s

        T_stages = np.geomspace(20.0, 0.02, 30)

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = evaluate_local(a)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break

        current_lr = lr * (0.3 + 0.7 * min(1.0, T / 5.0))
        steps_this_stage = 0
        last_improve_step = 0

        while time.monotonic() - t0 < time_limit:
            step += 1
            steps_this_stage += 1

            grad, b, max_b = compute_grad_fft(a, T)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)

            a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)

        elapsed = time.monotonic() - t0

def lp_best_response(a_fixed, solver=None, max_iter=5000):
    """Solve LP: minimize M s.t. T@b <= M, sum(b)=1, b>=0."""
    if not HAS_CVXPY:
        return None, None
    if solver is None:
        solver = get_lp_solver()
    if solver is None:
        return None, None

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for s_name in [solver, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=s_name, max_iter=max_iter, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def nesterov_lp(a0, time_limit, solver=None, max_iters=100000, stall_patience=12):
    """Nesterov-accelerated LP alternation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(77)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 25:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 400 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise_scale = 0.015 / np.sqrt(float(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall = 0

    return best_a, best_score


def peak_coord_descent(a0, time_limit, n_steps=50000):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        improved = False
        top_count = min(6, len(idx))
        top_idx = idx[np.argsort(contribs)[::-1][:top_count]]

        for pos in top_idx:
            if a[pos] < 1e-12:
                continue
            for factor in [0.995, 0.99, 0.98, 0.97]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] * (1.0 - factor)
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction of top 3
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.997
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                score = evaluate_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 400:
                        break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_beta_init(n, alpha=0.5, beta=0.5):
    """Create a beta-distribution-based initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_arcosine_init(n):
    """Arcsine distribution initialization (alpha=beta=0.5)."""
    return make_beta_init(n, 0.5, 0.5)

            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = evaluate_local(a_trial)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-7, lr)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: FFT-accelerated Adam with LogSumExp smoothing,
    LP-based refinement, peak coordinate descent, subgradient polishing,
    and multi-length exploration with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = make_arcosine_init(n_init)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Adam with LogSumExp on parent ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: Adam+LSE n={n_cur}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 420.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_res, _ = adam_logsumexp(
            best_seq, t1,
            T_stages=np.geomspace(15.0, 0.02, 35),
            lr=0.004, beta1=0.9, beta2=0.999
        )
        try_update(a_res, "adam_p1")

    # === Phase 2: LP refinement ===
    print(f"\n=== Phase 2: LP n={n_cur}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 380.0)
    if t2 > 30:
        solver = get_lp_solver()
        if solver is not None:
            print(f"  LP solver: {solver}, budget: {t2:.0f}s")
            a_res, _ = nesterov_lp(best_seq, t2, solver=solver,
                                    max_iters=100000, stall_patience=15)
            try_update(a_res, "lp_p2")

    # === Phase 3: Fine Adam (lower T) ===
    print(f"\n=== Phase 3: Fine Adam n={n_cur}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 150.0)
    if t3 > 15:
        a_res, _ = adam_logsumexp(
            best_seq, t3,
            T_stages=np.geomspace(5.0, 0.01, 25),
            lr=0.002, beta1=0.9, beta2=0.999
        )
        try_update(a_res, "adam_p3")

    # === Phase 4: Peak CD ===
    print(f"\n=== Phase 4: PeakCD n={n_cur}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 100.0)
    if t4 > 10:
        a_res, _ = peak_coord_descent(best_seq, t4, n_steps=60000)
        try_update(a_res, "peakcd_p4")

    # === Phase 5: Try longer sequence (1.5x) ===
    print(f"\n=== Phase 5: Longer n, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_long = int(n_cur * 1.5)
        t5 = min(rem * 0.30, 300.0)
        if t5 > 30:
            print(f"  Trying n={n_long}, budget={t5:.0f}s")
            init_long = interpolate_sequence(best_seq, n_long)
            a_res, _ = adam_logsumexp(
                init_long, t5,
                T_stages=np.geomspace(15.0, 0.02, 30),
                lr=0.004
            )
            try_update(a_res, f"adam_long_{n_long}")

            # LP polish on longer if time permits
            rem = deadline - time.monotonic()
            if rem > 30:
                solver = get_lp_solver()
                if solver is not None:
                    t5b = min(rem * 0.5, 120.0)
                    a_res2, _ = nesterov_lp(a_res if a_res is not None else init_long,
                                             t5b, solver=solver, max_iters=50000)
                    try_update(a_res2, f"lp_long_{n_long}")

    # === Phase 6: Try much longer sequence (2.5x) ===
    print(f"\n=== Phase 6: Very long n, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_vlong = int(n_cur * 2.5)
        t6 = min(rem * 0.35, 250.0)
        if t6 > 30:
            print(f"  Trying n={n_vlong}, budget={t6:.0f}s")
            init_vlong = interpolate_sequence(best_seq, n_vlong)
            a_res, _ = adam_logsumexp(
                init_vlong, t6,
                T_stages=np.geomspace(20.0, 0.05, 25),
                lr=0.005
            )
            try_update(a_res, f"adam_vlong_{n_vlong}")

    # === Phase 7: Try different initialization at current length ===
    print(f"\n=== Phase 7: Alt init, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t7 = min(rem * 0.3, 120.0)
        if t7 > 20:
            # Arcsine init
            init_alt = make_arcosine_init(n_cur)
            a_res, _ = adam_logsumexp(
                init_alt, t7,
                T_stages=np.geomspace(10.0, 0.02, 25),
                lr=0.004
            )
            try_update(a_res, "alt_arcosine")

    # === Phase 8: Subgradient polish ===
    print(f"\n=== Phase 8: SubGrad, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.08, 60.0)
    if t8 > 8:
        a_res, _ = subgradient_refine(best_seq, t8, n_steps=30000)
        try_update(a_res, "subgrad_p8")

    # === Phase 9: Final fine Adam ===
    print(f"\n=== Phase 9: Final Adam, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.5, 50.0)
        if t9 > 8:
            a_res, _ = adam_logsumexp(
                best_seq, t9,
                T_stages=np.geomspace(2.0, 0.01, 20),
                lr=0.001
            )
            try_update(a_res, "final_adam")

    # === Phase 10: Final peak CD ===
    rem = deadline - time.monotonic()
    if rem > 8:
        t10 = min(rem * 0.6, 30.0)
        a_res, _ = peak_coord_descent(best_seq, t10, n_steps=25000)
        try_update(a_res, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
