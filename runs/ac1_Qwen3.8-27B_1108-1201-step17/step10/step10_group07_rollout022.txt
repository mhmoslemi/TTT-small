def normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * length)
        vals.extend(a[:length])
    T = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T


def lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_conv_matrix_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones_vec = cp.Constant(np.ones(2 * n - 1))
    constraints = [
        T @ b_var <= M_var * ones_vec,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def line_search_lp(a, b, max_alpha, n_evals=20):
    """
    Line search: find best alpha in [0, max_alpha] minimizing score((1-alpha)*a + alpha*b).
    Uses golden section search.
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    
    def f(alpha):
        x = (1.0 - alpha) * a + alpha * b
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return float('inf')
        x = x / s
        conv = fast_convolve(x)
        return 2.0 * len(a) * conv.max()
    
    lo = 0.0
    hi = max_alpha
    
    # Initial evaluations
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    best_alpha = (lo + hi) / 2.0
    return best_alpha

def diverse_init(n, seed, init_type=0):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    if init_type == 0:  # Triangle
        a = np.maximum(1.0 - np.abs(t - center) / center, 0.0)
    elif init_type == 1:  # U-shape
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:  # Cosine
        x = (t - center) / center
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * x)), 0.0)
    elif init_type == 3:  # Two humps
        c1 = n * rng.uniform(0.25, 0.35)
        c2 = n * rng.uniform(0.65, 0.75)
        w = n * 0.12
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:  # Nearly uniform with small perturbation
        a = np.ones(n) + 0.01 * rng.standard_normal(n)
        a = np.maximum(a, 0.0)
    else:  # Power cosine
        x = (t - center) / center
        p = rng.uniform(0.5, 2.0)
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * x)), 0.0) ** p
    
    a = np.maximum(a, 0.0)
    return normalize(a)

from scipy.fft import rfft, irfft
from scipy import sparse
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


_lp_solver_cache = None


def find_lp_solver():
    global _lp_solver_cache
    if _lp_solver_cache is not None:
        return _lp_solver_cache
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cache = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cache = cp.ECOS
    return _lp_solver_cache


def build_conv_matrix_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * length)
        vals.extend(a[:length].tolist())
    T = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones_vec = cp.Constant(np.ones(2 * n - 1))
    constraints = [
        T @ b_var <= M_var * ones_vec,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def diverse_init(n, seed, init_type=0):
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if init_type == 0:
        a = np.maximum(1.0 - np.abs(t - center) / center, 0.0)
    elif init_type == 1:
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        x = (t - center) / center
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * x)), 0.0)
    elif init_type == 3:
        c1 = n * rng.uniform(0.25, 0.35)
        c2 = n * rng.uniform(0.65, 0.75)
        w = n * 0.12
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        a = np.ones(n) + 0.005 * rng.standard_normal(n)
        a = np.maximum(a, 0.0)
    else:
        x = (t - center) / center
        p = rng.uniform(0.3, 1.5)
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * x)), 0.0) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
