import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective (assumes sum(a)=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver] + [x for x in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if x != solver]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv > 1e-15:
                    return b_val / sv
        except Exception:
            continue
    return None


def lp_alternation(a0, time_limit, solver, max_iters=100000):

    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        a_new = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 15:

                noise = 0.01 / math.sqrt(stall)
                y_a = normalize(y_a + rng.standard_normal(n) * noise)

                stall = 0

        if (it + 1) % 500 == 0:
            print(f"    LP alt iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Warm-started bidirectional LP alternation for initialization.
2. Fast coordinate descent with exact 1D golden-section optimization
   (leveraging the closed-form update of the convolution, avoiding FFTs).
3. Nesterov-accelerated exponential-weights (mirror) descent with
   temperature-annealed subgradients.
4. Final LP polish and multi-n exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s != solver]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv > 1e-15:
                    return b_val / sv
        except Exception:
            continue
    return None


def lp_alternation(a0, time_limit, solver, max_iters=50000):
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        a_new = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 20:
                noise = 0.02 / math.sqrt(stall)
                y_a = normalize(y_a + rng.standard_normal(n) * noise)
                y_b = normalize(y_b + rng.standard_normal(n) * noise)
                stall = 0

        if (it + 1) % 200 == 0:
            print(f"    LP alt iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def coord_descent_fast(a0, time_limit, max_sweeps=5000):
    """Fast coordinate descent using closed-form convolution updates."""
    n = len(a0)
    a = normalize(a0).copy()
    b = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    gr = (math.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n) if sweep > 0 else np.arange(n)
        sweep_improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            a_j = a[j]
            if a_j < 1e-15:
                # Try increasing from zero
                delta_candidates = [1e-6 / n, 1e-5 / n, 1e-4 / n]
            else:
                # Gradient-based candidate range
                delta_candidates = [
                    -0.1 * a_j, -0.05 * a_j, -0.02 * a_j,
                    0.02 * a_j, 0.05 * a_j, 0.1 * a_j
                ]
                # Filter valid candidates
                delta_candidates = [d for d in delta_candidates if d > -a_j + 1e-15 and d < 1.0]
                if not delta_candidates:
                    continue

            # Compute c and d vectors
            # c[k] = a[k-j] * I(0 <= k-j < n) + a[j]
            c = np.full(2 * n - 1, a_j)
            k_lo = j
            k_hi = min(2 * n - 1, j + n)
            if k_lo < k_hi:
                c[k_lo:k_hi] += a[0:k_hi - k_lo]
            # d[k] = 1 if 2j == k
            d = np.zeros(2 * n - 1)
            two_j = 2 * j
            if two_j < 2 * n - 1:
                d[two_j] = 1.0

            # Golden section search for optimal delta
            lo = -a_j * 0.9999
            hi = 0.5 * a_j  # don't increase too much in one step
            if a_j < 1e-10:
                lo = -1e-15
                hi = 1e-3 / n

            # Evaluate at golden section points
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            def eval_delta(delta):
                if delta <= -0.999:
                    return 1e10
                denom = (1.0 + delta) ** 2
                b_trial = (b + delta * c + (delta * delta) * d) / denom
                return b_trial.max()

            f1 = eval_delta(c1)
            f2 = eval_delta(c2)

            for _ in range(40):
                if hi - lo < 1e-14 * max(1.0, abs(lo), abs(hi)):
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_delta(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_delta(c2)

            delta_opt = (lo + hi) / 2.0
            f_opt = eval_delta(delta_opt)
            f_current = b.max()

            if f_opt < f_current - 1e-15 and delta_opt > -a_j + 1e-15:
                # Apply update
                a[j] = (a_j + delta_opt)
                # Normalize: divide all by (1 + delta_opt)
                inv = 1.0 / (1.0 + delta_opt)
                a *= inv
                # Update b using the formula
                b = (b + delta_opt * c + (delta_opt * delta_opt) * d) * (inv * inv)
                # Actually, b_new = (b + delta*c + delta^2*d) / (1+delta)^2 = (b + delta*c + delta^2*d) * inv^2
                # which is what I computed above
                
                # Clip tinynegative values
                a = np.maximum(a, 0.0)
                b = np.maximum(b, 0.0)
                
                cur_score = 2.0 * n * b.max()
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                    sweep_improved = True
                    no_improve = 0
                else:
                    no_improve += 1

        if not sweep_improved:
            no_improve += 1
            if no_improve >= 50:
                # Add small noise to escape
                a = normalize(a + rng.standard_normal(n) * 1e-5)
                b = fast_convolve(a)
                no_improve = 0

        if (sweep + 1) % 100 == 0:
            print(f"    CD sweep={sweep+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def nesterov_mirror_descent(a0, time_limit, T_stages=None):
    """Nesterov-accelerated exponential weights (mirror descent on simplex)."""
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300))
    z -= z.max()  # for numerical stability
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    if T_stages is None:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    beta = 0.9  # momentum
    v = np.zeros(n)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        max_iters = int(remaining * 50)  # ~50 iterations per second
        eta = 1.0 / math.sqrt(max(n, 10))

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            b_max = b.max()
            score = 2.0 * n * b_max

            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()

            # Compute subgradient with temperature smoothing
            w = np.exp(b / T - b_max / T)
            w = w / w.sum()

            # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (with proper indexing)
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-12:
                    continue
                # contribution to g[j] from position k: 2 * w[k] * a[k-j] for j in [k-n+1, k]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += 2.0 * w[k] * a[k - idx]

            # Momentum update
            v = beta * v + (1.0 - beta) * g

            # Nesterov step
            z = z - eta * v
            z -= z.max()  # numerical stability

            # Adaptive step size
            if it % 100 == 0 and it > 0:
                eta *= 0.95  # gradual decay

    return best_a, best_score
