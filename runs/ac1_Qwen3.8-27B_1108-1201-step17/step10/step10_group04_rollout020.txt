import numpy as np

def fast_convolve(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_rows(a, active_set):
    """Build rows of convolution matrix for given active set.
    Returns (2n-1, n) matrix where row k has a[j] at position j if valid."""
    n = len(a)
    rows = []
    for k in active_set:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            row[js:je+1] = a[js:je+1]
        rows.append(row)
    return np.array(rows)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def solve_lp_restricted(a, active_set, solver=None, time_check=None, t0=None):
    """Solve restricted LP: min M s.t. conv(a,b)[k] <= M for k in active_set, sum(b)=1, b>=0.
    Returns (b, M) or (None, None)."""
    if solver is None:
        solver = get_solver()
    if time_check is not None and time.monotonic() - t0 > time_check:
        return None, None

    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    T = build_conv_matrix_rows(a, active_set)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, max(float(M_var.value), 1e-15)


def accelerated_lp(a0, time_limit, solver=None, active_frac=0.3, expand_every=50):
    """Nesterov-accelerated LP best-response with adaptive active set."""
    if solver is None:
        solver = get_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Initial active set
    b_conv = fast_convolve(a)
    frac = active_frac
    threshold = b_conv.max() - frac * b_conv.max()
    active_set = set(np.where(b_conv >= threshold)[0].tolist())

    prev_a = a.copy()
    momentum_coeff = 0.0

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP at extrapolated point y
        b_new, M_val = solve_lp_restricted(y, active_set, solver, time_limit - (time.monotonic() - t0), t0)
        if b_new is None:
            # Fallback: use full active set
            b_new, M_val = solve_lp_restricted(y, set(range(2*n-1)), solver, time_limit - (time.monotonic() - t0), t0)
            if b_new is None:
                break

        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        a_new = (1.0 - t_k) * a + t_k * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        # Nesterov extrapolation
        momentum_coeff = it / (it + 2.0)
        y_new = a_new + momentum_coeff * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        # Check score
        sc = score_normalized(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        sc_y = score_normalized(y)
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()

        # Update active set periodically
        if (it + 1) % expand_every == 0:
            b_conv = fast_convolve(a)
            # Gradually expand active set
            frac = max(0.1, active_frac * (0.95 ** (it // expand_every)))
            threshold = b_conv.max() - frac * b_conv.max()
            active_set = set(np.where(b_conv >= threshold)[0].tolist())

            if it % (expand_every * 10) == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, |active|={len(active_set)}")

    return best_a, best_score


def kkt_polish(a0, time_limit, tol=1e-8, max_iters=2000):
    """Polish using KKT conditions. At optimum, exists p on active set such that
    sum_k p_k * 2*shift_k(a) = lambda on support, <= lambda where a=0."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_conv = fast_convolve(a)
        max_b = b_conv.max()
        # Active set: within tol of max
        active = np.where(b_conv >= max_b - max(tol, 1e-10))[0]
        m = len(active)

        if m == 0:
            break

        # Compute gradient: for each active k, grad_k[i] = 2*a[k-i] (if valid)
        # KKT: sum_k p_k * grad_k = lambda * 1 (on support)
        # We solve for p and lambda
        # grad[i] = sum_k p_k * 2 * a[k-i] (with bounds)
        # We want grad[i] = lambda for all i with a[i] > 0
        # And grad[i] <= lambda for all i with a[i] = 0
        # Plus sum(p) = 1, p >= 0

        # Build: for each i in support, sum_k p_k * 2*a[k-i] = lambda
        support = np.where(a > 1e-15)[0]
        ns = len(support)

        if ns < 2:
            break

        # Build linear system: A @ [p_0, ..., p_{m-1}, lambda]^T = rhs
        # For each i in support: sum_k 2*a[k-i]*p_k - lambda = 0
        # Plus: sum_k p_k = 1
        # We have m+1 unknowns and ns+1 equations. If m >= ns, system is overdetermined.
        # Use least squares.

        A_mat = np.zeros((ns + 1, m + 1))
        rhs = np.zeros(ns + 1)

        for idx, i in enumerate(support):
            for kid, k in enumerate(active):
                j = k - i
                if 0 <= j < n:
                    A_mat[idx, kid] = 2.0 * a[j]
            A_mat[idx, m] = -1.0
            rhs[idx] = 0.0

        # Normalization
        for kid in range(m):
            A_mat[ns, kid] = 1.0
        rhs[ns] = 1.0

        # Solve least squares
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(A_mat, rhs, rcond=None)
        except Exception:
            break

        p_val = np.maximum(sol[:m], 0)
        p_sum = p_val.sum()
        if p_sum < 1e-15:
            p_val = np.ones(m) / m
        else:
            p_val = p_val / p_sum
        lam = sol[m]

        if lam <= 0:
            # Try different approach: use the LP dual
            break

        # Compute the gradient direction
        grad = np.zeros(n)
        for kid, k in enumerate(active):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += p_val[kid] * 2.0 * a[j]

        # Check KKT satisfaction
        grad_support = grad[support]
        kkt_residual = np.abs(grad_support - lam).max()

        if kkt_residual < tol:
            break

        # Take a step: move a in direction (1 - grad/lam) to make grad more uniform
        # Use mirror descent step
        direction = a * (1.0 - grad / max(lam, 1e-15))
        direction = np.maximum(direction, 0)
        ds = direction.sum()
        if ds < 1e-15:
            break
        direction = direction / ds

        # Line search
        best_alpha = None
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = (1 - alpha) * a + alpha * direction
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-12:
                best_alpha = alpha
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                break

        if best_alpha is None:
            # Try smaller steps
            improved = False
            for alpha in [0.001, 0.003, 0.01]:
                a_trial = (1 - alpha) * a + alpha * direction
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
            if not improved:
                if it > 50:
                    break
                # Perturb
                rng = np.random.default_rng(it * 7 + 3)
                a = a + rng.normal(0, 0.005, n)
                a = np.maximum(a, 0)
                a = a / a.sum()

        if it % 200 == 0 and it > 0:
            print(f"      KKT iter={it}: score={best_score:.6f}, kkt_res={kkt_residual:.2e}")

    return best_a


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=8):
    """L-BFGS with LogSumExp smooth approximation."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = softmax_from_z(z).copy()
    best_score = score_normalized(best_a)

    def lse_obj(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        mx = Tb.max()
        if mx > 700:
            return 1e15
        e = np.exp(Tb - mx)
        lse = mx + np.log(e.sum())
        return (2.0 * n / T_val) * lse

    def lse_grad(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        mx = Tb.max()
        if mx > 700:
            return np.zeros_like(z_in)
        e = np.exp(Tb - mx)
        w = e / e.sum()
        # grad of (2n/T) * lse w.r.t. a_i
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iter_budget = max(10, int(remaining * 0.7 / (len(temps) - si)))

        try:
            res = minimize(
                fun=lambda z_in: lse_obj(z_in, stage_T),
                x0=z,
                jac=lambda z_in: lse_grad(z_in, stage_T),
                method='L-BFGS-B',
                options={'maxiter': iter_budget, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iter_budget * 200,
                         'disp': False}
            )
            z = res.x
            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = z_from_a(best_a)
        except Exception:
            pass

    return best_a

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                no_improve = 0
                lr = 0.5

        if step % 50000 == 0 and step > 0:
            print(f"      MirrorDesc {time.monotonic()-t0:.0f}s: score={best_score:.6f}")

    return best_a


def optimize_at_length(a_init, time_budget, solver, label=""):
    """Multi-phase optimization at fixed length."""
    best_a = np.maximum(a_init.copy(), 0).astype(np.float64)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(len(a_init)) / len(a_init)
    best_a = best_a / best_a.sum()
    best_score = score_normalized(best_a)
    t0 = time.monotonic()

    def try_update(a, src):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a.astype(np.float64), 0)
        ss = a.sum()
        if ss < 1e-15:
            return
        a = a / ss
        sc = score_normalized(a)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            print(f"    NEW BEST [{src}]: {sc:.6f} (t={time.monotonic()-t0:.0f}s)")

    # Phase 1: Mirror descent (quick, cheap)
    rem = time_budget - (time.monotonic() - t0)
    t1 = min(rem * 0.10, 60.0)
    if t1 > 5:
        a_out = mirror_descent(best_a, t1)
        try_update(a_out, f"md_{label}")

    # Phase 2: Accelerated LP (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.50, 500.0)
    if t2 > 20:
        a_out, _ = accelerated_lp(best_a, t2, solver=solver, active_frac=0.3, expand_every=30)
        try_update(a_out, f"lp_{label}")

    # Phase 3: KKT polish
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.20, 120.0)
    if t3 > 10:
        a_out = kkt_polish(best_a, t3)
        try_update(a_out, f"kkt_{label}")

    # Phase 4: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.20, 100.0)
    if t4 > 10:
        a_out = lbgfs_logsumexp(best_a, t4, T_start=20.0, T_end=0.05, n_stages=8)
        try_update(a_out, f"lse_{label}")

    # Phase 5: More LP
    rem = time_budget - (time.monotonic() - t0)
    t5 = min(rem * 0.50, 100.0)
    if t5 > 10:
        a_out, _ = accelerated_lp(best_a, t5, solver=solver, active_frac=0.15, expand_every=20)
        try_update(a_out, f"lp2_{label}")

    # Phase 6: Final KKT
    rem = time_budget - (time.monotonic() - t0)
    t6 = min(rem * 0.50, 50.0)
    if t6 > 5:
        a_out = kkt_polish(best_a, t6, tol=1e-10)
        try_update(a_out, f"kkt2_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_limit = t_global + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_solver()
    print(f"LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 960
        best_seq = structured_init(n_base, rng)
        best_score = score_normalized(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def global_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        sc = score_normalized(a)
        if sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # Phase 1: Optimize at parent length
    print(f"\n=== Phase 1: n={n_base} (best={best_score:.6f}) ===")
    rem = time_limit - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(best_seq, t1, solver, label=f"n{n_base}")
        global_try_update(a_out, f"opt_n{n_base}")

    # Phase 2: Try nearby lengths
    print(f"\n=== Phase 2: Length exploration (best={best_score:.6f}) ===")
    candidate_lengths = []
    for mult in [1.2, 1.5, 0.8, 1.3, 2.0]:
        cn = int(n_base * mult)
        if 200 <= cn <= 5000 and cn != n_base:
            candidate_lengths.append(cn)
    candidate_lengths = sorted(set(candidate_lengths))[:4]

    for cn in candidate_lengths:
        rem = time_limit - time.monotonic()
        if rem < 100:
            break
        t_alloc = min(rem * 0.22, 300.0)
        if t_alloc < 50:
            break
        print(f"  Trying n={cn}, budget={t_alloc:.0f}s")

        init = interpolate_sequence(best_seq, cn)
        sc_init = score_normalized(init)
        cur = init
        if sc_init < best_score - 1e-8:
            global_try_update(cur, f"interp_n{cn}")

        a_out, _ = optimize_at_length(cur, t_alloc, solver, label=f"n{cn}")
        global_try_update(a_out, f"opt_n{cn}")

    # Phase 3: Final intensive polish on best
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit - time.monotonic()
    if rem > 30:
        # More LP with tighter active set
        t_lp = min(rem * 0.35, 200.0)
        if t_lp > 15:
            a_fin, _ = accelerated_lp(best_seq, t_lp, solver=solver, active_frac=0.1, expand_every=20)
            global_try_update(a_fin, "final_lp")

        # KKT with tight tolerance
        rem = time_limit - time.monotonic()
        t_kkt = min(rem * 0.30, 100.0)
        if t_kkt > 10:
            a_fin2 = kkt_polish(best_seq, t_kkt, tol=1e-12, max_iters=5000)
            global_try_update(a_fin2, "final_kkt")

        # L-BFGS fine-tune
        rem = time_limit - time.monotonic()
        t_lse = min(rem * 0.35, 80.0)
        if t_lse > 10:
            a_fin3 = lbgfs_logsumexp(best_seq, t_lse, T_start=5.0, T_end=0.02, n_stages=10)
            global_try_update(a_fin3, "final_lse")

    # Final verification with the actual evaluate_sequence
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            print(f"\nFINAL (evaluate_sequence): {fs:.6f} (n={len(best_seq)})")
        else:
            print(f"\nFINAL (normalized score): {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
