Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smoothed gradient descent
with FFT-accelerated gradients, Nesterov momentum, and multi-phase optimization.

import numpy as np
from scipy.fft import rfft, irfft
import time

def next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fft_grad_from_weights(w, a):
    """Compute grad[i] = sum_k w[k] * a[k-i] using FFT.
    w has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    a_rev = a[::-1]
    total_len = len(w) + n - 1  # 3n-2
    fft_size = next_pow2(total_len)
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    result = irfft(W * AR, fft_size)[:total_len]
    return result[n - 1: 2 * n - 1]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

def nesterov_smoothed_descent(a0, time_limit, label="", beta_start=15.0, beta_end=200.0):
    """Nesterov-accelerated smoothed gradient descent with adaptive beta."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = exact_score(a)
    t0 = time.monotonic()
    
    gamma = 1.0
    y = a.copy()
    it = 0
    no_improve = 0
    lr = 0.01
    
    # Total planned iterations (estimate based on time)
    # Each iteration takes ~O(n log n) time, for n=1024 that's ~0.1ms
    # So we can do ~10M iterations in 1800s (but with overhead, maybe 1M)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov momentum
        if it > 1:
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
            y = ((it + t_k - 1) / t_k) * a - ((it - 1) / t_k) * y
            y = simplex_project(np.maximum(y, 0.0))
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        
        # Flying start schedule for beta
        progress = min(1.0, it / max(it, 10000))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient
        grad = smooth_grad(y, beta)
        
        # Adaptive step size
        # The gradient of the smoothed function has Lipschitz constant ~ 2*beta*max_w*2*max_a
        # For safety, use a small step
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        step = lr / (1.0 + 0.01 * it)
        
        a_new = a - step * grad
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = exact_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.1f}")

    return best_a, best_score


def exact_subgrad_step(a, step_size):
    """One step of exact subgradient method."""
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()
    
    # Use all indices within 0.1% of max for a more stable subgradient
    threshold = M * 0.999
    near_max = np.where(conv_a >= threshold)[0]
    rho = 1.0 / len(near_max)
    
    grad = np.zeros(n)
    for k in near_max:
        wk = rho
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * wk * a[k - j]
    
    a_new = a - step_size * grad
    a_new = simplex_project(a_new)
    return a_new

def structured_perturb(a0, rng, strength=0.05):
    """Perturb by reducing peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    
    # Pressure: which coordinates contribute most to peaks
    pressure = np.zeros(n)
    peak_thresh = M * 0.85
    peaks = np.where(conv >= peak_thresh)[0]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            pressure[i] += a[i] * a[k - i]
    
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    
    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure + strength * 0.5 * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

import numpy as np
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fft_cross_grad(w, a):
    """Compute S[i] = sum_k w[k] * a[k-i] for i in [0, n-1] via FFT.
    Identity: S[i] = full_conv(w, a_rev)[n-1+i] where a_rev = a[::-1].
    """
    n = len(a)
    a_rev = a[::-1]
    total_len = len(w) + n - 1
    fft_size = _next_pow2(total_len)
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    result = irfft(W * AR, fft_size)[:total_len]
    return result[n - 1: 2 * n - 1]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def smooth_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv_k)) at point a."""
    n = len(a)
    conv_a = fast_convolve(a)
    max_c = conv_a.max()
    scaled = beta * (conv_a - max_c)
    scaled = np.clip(scaled, -700, 0)
    w = np.exp(scaled)
    Z = w.sum()
    if Z < 1e-300:
        w = np.zeros(len(conv_a))
        w[int(np.argmax(conv_a))] = 1.0
    else:
        w = w / Z
    grad = 2.0 * fft_cross_grad(w, a)
    return grad

        a = a / s

    best_a = a.copy()
    best_score = exact_score(a)
    t0 = time.monotonic()

    gamma = 1.0
    y = a.copy()
    prev_a = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov momentum
        if it > 1:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
            beta_mom = (gamma - 1.0) / t_new
            y = a + beta_mom * (a - prev_a)
            y = simplex_project(np.maximum(y, 0.0))
            gamma = t_new
        prev_a = a.copy()

        # Adaptive temperature (geometric schedule)
        total_est = max(it, 5000)
        progress = min(1.0, float(it) / 50000.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute gradient at y
        grad = smooth_grad(y, beta)

        # Step size: inverse of Lipschitz constant estimate
        # L ~ 4*beta for our problem (conservative)
        lr = lr_base / (beta * (1.0 + 0.0005 * it))

        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        # Periodic exact evaluation
        if it % 30 == 0 or it <= 5:
            score = exact_score(a_new)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

        a = a_new

        # Adaptive restart
        if no_improve > 500:
            lr = max(lr * 0.7, 1e-10)
            no_improve = 0
            if lr < 1e-12:
                break

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSGD {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}, beta={beta:.1f}, lr={lr:.2e}")

    # Final check
    score = exact_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def exact_subgrad_nesterov(a0, time_limit, label=""):
    """Exact subgradient method with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = exact_score(a)
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    prev_a = a.copy()
    avg_a = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        if it > 1:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
            beta_mom = (gamma - 1.0) / t_new
            y = a + beta_mom * (a - prev_a)
            y = simplex_project(np.maximum(y, 0.0))
            gamma = t_new
        prev_a = a.copy()

        # Exact subgradient at y
        conv_y = fast_convolve(y)
        M = conv_y.max()
        threshold = M * 0.995
        near_max = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_max)

        grad = np.zeros(n)
        for k in near_max:
            wk = rho
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * wk * y[k - j]

        # Step
        step = 0.3 / np.sqrt(1.0 + it)
        a_new = a - step * grad
        a_new = simplex_project(a_new)

        # Nesterov average
        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new

        # Evaluate
        for cand in [a_new, avg_a]:
            score = exact_score(cand)
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with exact 1D line search per coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            # Where lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.1 < d < ai * 0.5:
                        candidates.append(d)

            # Top crossings
            top_k = min(4, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.1 < d < ai * 0.5:
                            candidates.append(d)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 30 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 80:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            print(f"    CD {label} sweep={sweeps} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    conv = fast_convolve(a)
    M = float(conv.max())
    final_score = 2.0 * n * M
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score

def peak_pressure_perturb(a0, rng, strength=0.04):
    """Perturb by reducing pressure on peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()

    pressure = np.zeros(n)
    peak_thresh = M * 0.8
    peaks = np.where(conv >= peak_thresh)[0]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            pressure[i] += a[i] * a[k - i]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n) * 0.2
    a_new = a * (1.0 - strength * pressure + 0.5 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def symmetry_break_perturb(a0, rng, strength=0.1):
    """Break symmetry by adding asymmetric noise."""
    n = len(a0)
    a = a0.copy()
    # Asymmetric: shift mass from left to right (or vice versa)
    shift = np.linspace(-1, 1, n) * strength
    noise = rng.standard_normal(n) * strength * 0.3
    a_new = a + shift * a + noise * a
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
