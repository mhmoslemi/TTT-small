import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def evaluate_safe(a):

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lse_objective_and_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z using FFT."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w_sum = exp_Tb.sum()
    if w_sum < 1e-300:
        return 2.0 * n * max_Tb / T_val, np.zeros(n)
    w = exp_Tb / w_sum
    obj = (2.0 * n / T_val) * (max_Tb + np.log(w_sum))

    # Gradient: grad_i = (2/T) * sum_k w_k * a_{k-i}
    # This is a cross-correlation of w with a
    # w has length 2n-1, a has length n
    # result_i = sum_k w_k * a_{k-i} for valid indices
    # = sum_j w_{i+j} * a_j
    grad_a = np.zeros(n)
    size = 2 * n - 1
    fft_size = 1 << (2 * size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: (w star a)_i = sum_j w_{i+j} a_j
    # = ifft(conj(fft(w)) * fft(a)) shifted
    # Actually: cross-correlation = ifft(conj(W) * A) gives sum_j w_j * a_{j-i}
    # We want sum_k w_k * a_{k-i} = sum_j w_{i+j} * a_j
    # Let's compute directly using the convolution structure
    # sum_k w_k * a_{k-i}: let m = k-i, k = m+i
    # = sum_m w_{m+i} * a_m (where w_{m+i} is 0 if m+i >= 2n-1)
    # This is (w shifted by -i) dot a = cross-correlation
    # cross_corr[m] = sum_k w[k] * a[k-m] (standard definition)
    # We want: for each i, grad_i = (2/T) * sum_k w[k] * a[k-i]
    # = (2/T) * cross_corr[i] where cross_corr[i] = sum_k w[k] * a[k-i]
    # Using FFT: cross_corr = ifft(conj(fft(w_padded)) * fft(a_padded))
    w_pad = np.zeros(fft_size)
    w_pad[:size] = w
    W_full = rfft(w_pad, fft_size)
    A_full = rfft(np.zeros(fft_size))
    A_full[:n] = a  # This won't work with rfft directly
    # Let me use a different approach
    # grad_i = (2/T) * sum_k w_k * a_{k-i}
    # For i in [0, n-1], k ranges over [0, 2n-2], and k-i must be in [0, n-1]
    # So k ranges over [i, i+n-1] (intersected with [0, 2n-2])
    # grad_i = (2/T) * sum_{k=i}^{min(i+n-1, 2n-2)} w_k * a_{k-i}
    
    # Compute via direct method for correctness (n is at most ~2000)
    # For each i in range(n):
    #   k_start = i, k_end = min(i + n, 2*n - 1)
    #   grad_a[i] = (2/T) * dot(w[k_start:k_end], a[:k_end-k_start])
    
    for i in range(n):
        k_end = min(i + n, 2 * n - 1)
        k_start = i
        if k_start >= k_end:
            continue
        grad_a[i] = np.dot(w[k_start:k_end], a[:k_end - k_start])
    grad_a *= 2.0 / T_val

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=500.0, T_end=0.01, n_stages=25):
    """Multi-stage LogSumExp L-BFGS with FFT-based gradients."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(30, int(200 / n_stages * 3))
        if si < n_stages // 3:
            iters = max(50, iters * 2)

        def func(z_in):
            val, _ = lse_objective_and_grad(z_in, stage_T, n)
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return val

        def jac(z_in):
            _, g = lse_objective_and_grad(z_in, stage_T, n)
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-12, 'maxfun': iters * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        if si % 5 == 0:
            a_cur = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_cur).max()
            print(f"    LSE stage {si+1}/{n_stages} T={stage_T:.2f} ({elapsed:.0f}s): score={sc:.6f}")

    return softmax_from_z(z)


def subgradient_simplex(a0, time_limit, n_steps=100000):
    """Subgradient method with projection onto simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set
        thresh = max_b - 0.05 * max_b
        active = np.where(b >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Projected gradient step
        a_new = a - lr * g
        # Project onto simplex {x: x >= 0, sum(x) = 1}
        # Using the standard algorithm
        v = np.sort(a_new)[::-1]
        cssv = np.cumsum(v)
        rho_candidates = np.where(v * np.arange(1, n + 1) > (cssv - 1))[0]
        if len(rho_candidates) == 0:
            a_new = np.ones(n) / n
        else:
            rho = rho_candidates[-1]
            theta = (cssv[rho] - 1) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0)
            a_new = a_new / a_new.sum()

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr *= 0.9999

        if stall >= 500:
            rng = np.random.default_rng(step * 7 + 3)
            noise = rng.random(n) * 0.02
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score

def pairwise_coordinate_descent(a0, time_limit, n_rounds=200):
    """Pairwise coordinate descent: transfer mass between pairs."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute b to identify good transfer candidates
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        # Try several random pairs
        n_pairs = min(50, n // 5)
        improved = False
        for _ in range(n_pairs):
            # Source: high contributor to peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[peak_k - idx]
            src = idx[np.argmax(contrib)]
            dst = rng.integers(n)
            if dst == src:
                dst = (dst + 1) % n

            for frac in [0.02, 0.05, 0.1, 0.15]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 0.01:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and rnd > 50:
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.2)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 0.01:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-10:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

        if rnd % 50 == 0 and rnd > 0:
            elapsed = time.monotonic() - t0
            print(f"    PairDesc round={rnd} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.exp(-4 * x ** 2) * (1 + 0.15 * np.cos(6 * x))
    elif choice == 8:
        # U-shaped (high at edges)
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
    else:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-phase optimization with emphasis on
    LogSumExp L-BFGS with tight annealing schedule and coordinate descent."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Long LogSumExp L-BFGS at base length ===
    print(f"\n=== Phase 1: LSE L-BFGS at n={n_base} (best={best_score:.6f}) ===")
    rem = t_deadline - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out = lse_lbfgs(best_seq, t1, T_start=1000.0, T_end=0.005, n_stages=30)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: Subgradient at base length ===
    rem = t_deadline - time.monotonic()
    t2 = min(rem * 0.12, 120.0)
    if t2 > 30 and best_seq is not None:
        print(f"\n=== Phase 2: Subgradient at n={len(best_seq)} ===")
        a_out, _ = subgradient_simplex(best_seq, t2, n_steps=50000)
        try_update(a_out, f"subgrad_n{len(best_seq)}")

    # === Phase 3: Try nearby lengths ===
    print(f"\n=== Phase 3: Nearby lengths (best={best_score:.6f}) ===")
    nearby_lengths = []
    for delta in [-200, -100, 100, 200, 300, 400]:
        ln = n_base + delta
        if 400 <= ln <= 4000 and ln != n_base:
            nearby_lengths.append(ln)
    nearby_lengths = sorted(set(nearby_lengths))[:4]

    for ln in nearby_lengths:
        rem = t_deadline - time.monotonic()
        if rem < 100:
            break
        t_alloc = min(rem * 0.15, 180.0)
        if t_alloc < 40:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()

        a_out = lse_lbfgs(init_long, t_alloc, T_start=200.0, T_end=0.01, n_stages=20)
        try_update(a_out, f"lse_n{ln}")

    # === Phase 4: LP-based best response on best found ===
    rem = t_deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 30 and best_seq is not None:
        print(f"\n=== Phase 4: LP Nesterov at n={len(best_seq)} (best={best_score:.6f}) ===")
        a_out, _ = nesterov_lp(best_seq, t4, solver=solver, n_max_iters=50000)
        try_update(a_out, f"lp_n{len(best_seq)}")

    # === Phase 5: Coordinate descent refinement ===
    rem = t_deadline - time.monotonic()
    t5 = min(rem * 0.12, 100.0)
    if t5 > 20 and best_seq is not None:
        print(f"\n=== Phase 5: Coordinate descent at n={len(best_seq)} ===")
        a_out, _ = pairwise_coordinate_descent(best_seq, t5, n_rounds=100)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Final LogSumExp polish ===
    rem = t_deadline - time.monotonic()
    t6 = min(rem * 0.25, 250.0)
    if t6 > 30 and best_seq is not None:
        print(f"\n=== Phase 6: Final LSE polish at n={len(best_seq)} (best={best_score:.6f}) ===")
        # Use very tight temperature range for final polish
        a_out = lse_lbfgs(best_seq, t6, T_start=10.0, T_end=0.001, n_stages=20)
        try_update(a_out, f"final_lse_n{len(best_seq)}")

    # === Phase 7: One more LP pass ===
    rem = t_deadline - time.monotonic()
    t7 = min(rem * 0.15, 80.0)
    if t7 > 20 and best_seq is not None:
        print(f"\n=== Phase 7: Final LP at n={len(best_seq)} ===")
        a_out, _ = nesterov_lp(best_seq, t7, solver=solver, n_max_iters=30000)
        try_update(a_out, f"final_lp_n{len(best_seq)}")

    # === Phase 8: Final coordinate descent ===
    rem = t_deadline - time.monotonic()
    t8 = min(rem * 0.10, 60.0)
    if t8 > 15 and best_seq is not None:
        print(f"\n=== Phase 8: Final coord descent at n={len(best_seq)} ===")
        a_out, _ = pairwise_coordinate_descent(best_seq, t8, n_rounds=80)
        try_update(a_out, f"final_coord_n{len(best_seq)}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed_total = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, total time={elapsed_total:.0f}s)")
    return [float(x) for x in best_seq]
