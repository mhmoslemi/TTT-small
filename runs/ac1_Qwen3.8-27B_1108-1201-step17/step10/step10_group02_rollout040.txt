Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent with 
soft-weighted subgradients, LP best response, and log-sum-exp L-BFGS polishing.
Key innovations: exponential peak weighting for smoother subgradient directions,
adaptive step size with duality-gap-based tuning, and multi-scale search.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def compute_soft_subgradient(a, b, temperature):
    """Compute subgradient with exponential peak weighting."""
    n = len(a)
    size = 2 * n - 1
    max_b = b.max()
    # Soft weights: emphasize peaks
    w = np.exp((b - max_b) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= w_sum
    g = np.zeros(n)
    for k in range(size):
        if w[k] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w[k] * a[k - idx]
    return g

def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (prob dist), find b (prob dist) minimizing max_k convolve(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    seen = set()
    for s in solvers_to_try:
        if s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                return b_val / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def compute_soft_subgradient(a, b, temperature):
    """Compute subgradient with exponential peak weighting."""
    n = len(a)
    size = 2 * n - 1
    max_b = b.max()
    w = np.exp((b - max_b) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= w_sum
    g = np.zeros(n)
    for k in range(size):
        if w[k] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w[k] * a[k - idx]
    return g


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov-accelerated LP alternating for the bilinear game."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42424)
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                noise_scale = 0.015 / np.sqrt(stall)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall = 0
        if (it + 1) % 200 == 0:
            print(f"    NLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def mirror_descent_soft(a0, time_limit, eta0=0.04, temp0=0.005, nesterov=True):
    """Mirror descent with soft-weighted subgradient and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    temp = temp0
    no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(20000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        g = compute_soft_subgradient(y, b, temp)

        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum

        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new
        score = evaluate_local(a)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            temp *= 0.99995
        else:
            no_improve += 1

        if no_improve == 300:
            eta *= 0.92
            temp *= 0.9
            no_improve = 0
        elif no_improve >= 1500:
            noise = rng.standard_normal(n) * 0.008
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = eta0
            temp = max(temp0, temp * 1.5)

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-soft iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f} temp={temp:.6f}")

    return best_a, best_score


class _SmoothedObjFunc:
    """Callable for log-sum-exp smoothed objective (avoids closures)."""
    def __init__(self, T, n):
        self.T = T
        self.n = n

    def __call__(self, z):
        a_in = np.exp(z - z.max())
        a_in = a_in / a_in.sum()
        b_in = fast_convolve(a_in)
        max_b = b_in.max()
        exp_b = np.exp(b_in / self.T - max_b / self.T)
        return self.T * (max_b / self.T + np.log(exp_b.sum()))


class _SmoothedJacFunc:
    """Callable for log-sum-exp smoothed gradient (avoids closures)."""
    def __init__(self, T, n):
        self.T = T
        self.n = n

    def __call__(self, z):
        a_in = np.exp(z - z.max())
        a_in = a_in / a_in.sum()
        b_in = fast_convolve(a_in)
        max_b = b_in.max()
        exp_b = np.exp(b_in / self.T - max_b / self.T)
        w = exp_b / exp_b.sum()
        grad_a = np.zeros(self.n)
        size = 2 * self.n - 1
        for k in range(size):
            if w[k] < 1e-12:
                continue
            js = max(0, k - self.n + 1)
            je = min(self.n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a_in[k - idx]
        S = np.dot(grad_a, a_in)
        return a_in * (grad_a - S)


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break

        obj_fn = _SmoothedObjFunc(T, n)
        jac_fn = _SmoothedJacFunc(T, n)

        try:
            res = minimize(fun=obj_fn, x0=z, jac=jac_fn, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def coordinate_descent_refine(a0, time_limit, n_steps=80000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:4]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.997, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 800:
                break
        else:
            stall = 0

        if (step + 1) % 20000 == 0:
            print(f"    CoordDesc ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-2.5 * (x - 0.5) ** 2 / 0.25)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP best-response Nesterov, soft mirror descent, L-BFGS polish,
    coordinate descent, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
            parent_loaded = True
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    # Helper: try updating best
    def _try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LP Nesterov (most accurate per iteration) ===
    print(f"\n=== Phase 1: LP Nesterov (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.32, 550.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver)
        _try_update(a_lp, "lp_nesterov")

    # === Phase 2: Soft Mirror Descent (many fast iterations) ===
    print(f"\n=== Phase 2: Soft Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 420.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_soft(best_seq, t_alloc, eta0=0.03, temp0=0.003)
        _try_update(a_md, "mirror_soft")

    # === Phase 3: Second LP pass with smaller step ===
    print(f"\n=== Phase 3: LP Nesterov Pass 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver)
        _try_update(a_lp2, "lp_nesterov_2")

    # === Phase 4: Log-sum-exp L-BFGS (smooth optimization) ===
    print(f"\n=== Phase 4: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        a_lbf = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=4000)
        _try_update(a_lbf, "lbgfs")

    # === Phase 5: Coordinate Descent (targeted polish) ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=100000)
        _try_update(a_cd, "coord_desc")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    target_ns = [800, 1000, 1500, 2000]
    target_ns = [n for n in target_ns if n != n_best]
    per_n_budget = max(5.0, min(rem * 0.03, 50.0))

    for n_try in target_ns:
        if time.monotonic() > deadline - 15:
            break
        a_interp = interpolate_sequence(best_seq, n_try)
        # Quick LP refinement at new n
        a_try, _ = nesterov_lp_refine(a_interp, per_n_budget, solver=solver)
        _try_update(a_try, f"n_{n_try}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin = logsumexp_lbgfs(best_seq, min(rem * 0.5, 60.0),
                                T_stages=[3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                maxiter_per_stage=3000)
        _try_update(a_fin, "final_lbgfs")

        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_cd2, _ = coordinate_descent_refine(best_seq, min(rem2 * 0.8, 50.0), n_steps=50000)
            _try_update(a_cd2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
