The idea:

2. $|\hat{a}_{new}(\omega)| = \text{flat\_target}$ (or a smoothed version of the current amplitude)
3. $\hat{a}_{new}(\omega) = |\hat{a}_{new}(\omega)| \cdot e^{i\arg(\hat{a}(\omega))}$
4. $a_{new} = \text{IDFT}(\hat{a}_{new})$
5. $a_{new} = \max(a_{new}, 0)$
6. Normalize: $a_{new} = a_{new} / \sum a_{new}$

- A constant (ideal, but might not be achievable)
- The current amplitude smoothed (e.g., $|\hat{a}_{new}| = \alpha \cdot |\hat{a}| + (1-\alpha) \cdot \text{mean}(|\hat{a}|)$)

        amp = np.abs(A)

        mean_amp = np.mean(amp)

        sc = local_eval(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_response(fixed, solver):
    """Given fixed (sum=1), find x >= 0, sum(x)=1 minimizing max_k conv(fixed, x)[k]."""
    n = len(fixed)
    x_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed)
    constraints = [T @ x_var <= M_var, cp.sum(x_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver]
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt not in solvers_to_try:
            solvers_to_try.append(alt)
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.asarray(x_var.value, dtype=np.float64)
                x_val = np.maximum(x_val, 0)
                ss = x_val.sum()
                if ss > 1e-15:
                    return x_val / ss, float(M_var.value)
        except Exception:
            continue
    return None, None

def logsumexp_lbgfs(a0, time_limit, T_stages=None):

    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())

    def lse_obj(z, T):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        A = rfft(a_cur, fft_size)
        b = irfft(A * A, fft_size)[:2 * n - 1]
        b_shift = b / T
        b_max = b_shift.max()
        lse = b_max + np.log(np.maximum(np.exp(b_shift - b_max).sum(), 1e-300))
        return (2.0 * n) * lse

    def lse_grad(z, T):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()

        # Gradient w.r.t. a
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
        S = np.dot(grad_a, a_cur)
        return a_cur * (grad_a - S)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        max_iter = max(30, int(remaining * 15))

        def obj(z):
            return lse_obj(z, T)

        def grad(z):
            return lse_grad(z, T)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def spectral_flattening(a0, time_limit, n_iters=50000):
    """Novel: Iteratively flatten the amplitude of the DFT while maintaining non-negativity."""
    n = len(a0)
    a = normalize(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Adaptive alpha: start aggressive, become more careful
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        amp = np.abs(A)
        phase = np.angle(A)

        # Progressive flattening
        progress = min(1.0, it / (n_iters * 0.5))
        alpha = 1.0 - 0.7 * progress  # from 1.0 (no change) to 0.3 (aggressive)

        # Target: blend current amplitude with mean
        mean_amp = np.mean(amp[1:])  # exclude DC
        # For DC, keep it (it's related to the sum)
        amp_new = amp.copy()
        amp_new[1:] = alpha * amp[1:] + (1.0 - alpha) * mean_amp

        A_new = amp_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 5000 == 0:
            print(f"    Spectral iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def peak_targeting(a0, time_limit, n_steps=50000):
    """Subgradient method targeting the single worst peak with adaptive step."""
    n = len(a0)
    a = normalize(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        peak_idx = int(np.argmax(b))

        # Gradient at the peak
        g = np.zeros(n)
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[peak_idx - idx]

        # Also consider near-peaks (within 1% of max)
        threshold = M * 0.995
        near_peaks = np.where(b >= threshold)[0]
        if len(near_peaks) > 1:
            g_total = np.zeros(n)
            for k in near_peaks:
                js2 = max(0, k - n + 1)
                je2 = min(n - 1, k)
                idx2 = np.arange(js2, je2 + 1)
                g_total[idx2] += 2.0 * a[k - idx2]
            g = g_total / len(near_peaks)

        # Projected gradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.02, 0.08)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)

    return best_a, best_score


def weighted_subgradient(a0, time_limit, T=0.5):
    """Weighted subgradient using exponential weights on peaks (smooth max approximation)."""
    n = len(a0)
    a = normalize(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03

    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_max = b.max()
        b_shift = (b - b_max) / T
        w = np.exp(b_shift)
        w /= w.sum()

        # Weighted gradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[k] * a[k - idx]

        # Projected step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial = a_trial / st
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                lr = min(lr * 1.01, 0.08)
            else:
                lr *= 0.995

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    elif choice == 8:
        # "U" shape - mass at edges
        a = (x - 0.5) ** 2 + 0.05
    else:
        # "W" shape
        a = 0.5 + 0.4 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def two_scale_refinement(a0, solver, time_limit):
    """Refine at two scales: coarse (low n) then fine (high n)."""
    n = len(a0)
    # Coarse scale
    n_coarse = max(100, n // 4)
    a_coarse = interpolate_sequence(a0, n_coarse)
    a_c, sc_c = lp_alternation(a_coarse, time_limit * 0.15, solver)

    # Interpolate back to fine scale
    a_fine = interpolate_sequence(a_c, n)
    a_f, sc_f = lp_alternation(a_fine, time_limit * 0.4, solver)

    return a_f, sc_f


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses improved LP alternation, aggressive LogSumExp, spectral flattening,
    peak targeting, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
        elif np.isfinite(sc) and abs(sc - best_score) < 1e-6:
            print(f"  [~] {label}: {sc:.10f} (n={len(a)}) - no improvement")

    # === Phase 1: Main LP Alternation ===
    print(f"\n=== Phase 1: LP Alternation (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "lp_main")

    # === Phase 2: LogSumExp L-BFGS (very aggressive) ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 350.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lb, sc_lb = logsumexp_lbgfs(best_seq, t2, T_stages=T_stages)
        try_update(a_lb, "lse_aggressive")

    # === Phase 3: Spectral Flattening (novel) ===
    print(f"\n=== Phase 3: Spectral Flattening (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 120.0)
    if t3 > 15:
        print(f"  Budget: {t3:.0f}s")
        a_spec, sc_spec = spectral_flattening(best_seq, t3, n_iters=80000)
        try_update(a_spec, "spectral")

    # === Phase 4: Peak Targeting ===
    print(f"\n=== Phase 4: Peak Targeting (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 80.0)
    if t4 > 10:
        a_pt, sc_pt = peak_targeting(best_seq, t4, n_steps=80000)
        try_update(a_pt, "peak_target")

    # === Phase 5: Weighted Subgradient ===
    print(f"\n=== Phase 5: Weighted Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.07, 70.0)
    if t5 > 10:
        a_ws, sc_ws = weighted_subgradient(best_seq, t5, T=0.3)
        try_update(a_ws, "weighted_sg")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_candidates = [800, 1000, 1100, 1300, 1400, 1600, 1800, 2000]
        # Also try near the current best
        n_candidates.append(int(n_best * 0.9))
        n_candidates.append(int(n_best * 1.1))
        n_candidates = sorted(set(n for n in n_candidates if 400 <= n <= 3000))

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            if cand_n == n_best:
                continue
            t_sub = min(rem * 0.15, 80.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick LP
            a_cand, _ = lp_alternation(init_cand, t_sub * 0.5, solver)
            try_update(a_cand, f"lp_n{cand_n}")
            # Quick LSE
            rem = deadline - time.monotonic()
            if rem > 25 and a_cand is not None:
                T_sub = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
                a_cand2, _ = logsumexp_lbgfs(a_cand, min(rem * 0.3, 40.0), T_stages=T_sub)
                try_update(a_cand2, f"lse_n{cand_n}")

    # === Phase 7: Second LP pass (with current best) ===
    print(f"\n=== Phase 7: Second LP Pass (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t7 = min(rem * 0.20, 200.0)
        a_lp2, sc_lp2 = lp_alternation(best_seq, t7, solver)
        try_update(a_lp2, "lp_second")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Very low T LSE
        t_fin = min(rem * 0.4, 80.0)
        if t_fin > 10:
            a_fin, _ = logsumexp_lbgfs(best_seq, t_fin,
                                       T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
            try_update(a_fin, "final_lse")

        # Peak targeting
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = peak_targeting(best_seq, min(rem * 0.3, 30.0), n_steps=50000)
            try_update(a_fin2, "final_pt")

        # Weighted subgradient with very low T
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin3, _ = weighted_subgradient(best_seq, min(rem * 0.3, 20.0), T=0.1)
            try_update(a_fin3, "final_ws")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
