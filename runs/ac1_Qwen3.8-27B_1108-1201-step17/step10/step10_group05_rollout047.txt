def simplex_project(v):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

        return np.ones(n) / n

def exact_1d_minimize(a, i, conv, unaffected_max):

    """

    ai = a[i]

    k_end = min(i + n - 1, 2 * n - 2)
    
    affected = conv[k_start:k_end + 1].copy()
    coeffs = a[k_start - i:k_end - i + 1] + ai  # a[k-i] + a[i] for each affected k
    has_quad = (2 * i >= k_start) and (2 * i <= k_end)

def ipf_best_response(a, max_iter=200, tol=1e-8):
    """
    Approximate the LP best response using Iterative Proportional Fitting (IPF).
    The LP is: minimize M s.t. T(a) @ b <= M * 1, sum(b) = 1, b >= 0
    Equivalently, for a given M, find b with sum(b)=1, b>=0, T(a)@b <= M*1
    We can use a multiplicative weights / IPF approach.
    """
    n = len(a)
    T_rows = 2 * n - 1
    b = np.ones(n) / n
    
    # Compute the "tightness" of each constraint
    for it in range(max_iter):
        conv_ab = np.convolve(a, b)  # T(a) @ b
        M = conv_ab.max()
        
        if M < tol:
            break
        
        # Multiplicative weights update: reduce b where the constraint is tight
        # For constraint k: sum_i a_{k-i} * b_i <= M
        # Update: b_i *= exp(-eta * sum_k 1[constraint k tight] * a_{k-i} / M)
        
        # Use a smoothed version for stability
        eta = 1.0 / (M + 1e-10)
        log_update = np.zeros(n)
        for k in range(T_rows):
            weight = np.exp(-eta * conv_ab[k] * 10)  # soft version
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                log_update[i] -= weight * a[k - i]
        
        b *= np.exp(log_update)
        b = np.maximum(b, 1e-30)
        b /= b.sum()
        
        if it % 50 == 0:
            pass  # verbose
    
    return b, float(conv_ab.max()) if 'conv_ab' in dir() else 0.0

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_conv(a, b):
    """Compute cross-convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project onto the probability simplex {x >= 0, sum(x) = 1} in O(n) time."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T where (T @ b)[k] = sum_i a[k-i]*b[i]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():

def solve_lp_best_response_fast(a_fixed, T, solver, warm_b=None):
    """
    Solve the LP best response: min M s.t. T @ b <= M*1, sum(b)=1, b>=0.
    Uses pre-built matrix T for speed.
    """
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    if warm_b is not None:
        b_var.value = warm_b
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    
    # Get dual variables for the convolution constraints
    duals = None
    try:
        # The first (2n-1) constraints are T @ b <= M
        # In cvxpy, constraint dual values are accessible
        dual_list = prob.constraints[0].dual_value
        if dual_list is not None:
            duals = np.asarray(dual_list, dtype=np.float64)
    except Exception:
        pass
    
    return b_val, float(M_var.value), duals

def exact_delta_minimize(a, i, conv, unaffected_max):
    """
    Find optimal delta for coordinate i using bisection on the convex function.
    f(delta) = max(unaff_max, max over affected k of (conv[k] + delta*coeff[k] + delta^2*q[k]))
    This is convex in delta >= -a[i].
    """
    n = len(a)
    ai = a[i]
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    
    affected = conv[k_start:k_end + 1]
    coeffs = a[k_start - i:k_end - i + 1] + ai
    has_quad = (2 * i >= k_start) and (2 * i <= k_end)
    quad_idx = 2 * i - k_start if has_quad else -1
    
    def obj(delta):
        """Compute max after applying delta."""
        new_vals = affected + delta * coeffs
        if has_quad:
            new_vals = new_vals.copy()
            new_vals[quad_idx] += delta * delta
        return max(unaffected_max, float(new_vals.max()))
    
    # Find bounds for bisection
    delta_lo = -ai
    # Upper bound: try ai (doubling) and expand if needed
    delta_hi = ai * 1.5
    if delta_hi <= delta_lo:
        delta_hi = delta_lo + 1e-10
    
    # Check if the minimum is at the boundary
    f_lo = obj(delta_lo)
    f_hi = obj(delta_hi)
    
    # Expand upper bound if f is still decreasing
    for _ in range(10):
        if f_hi > f_lo:
            break
        delta_hi *= 2.0
        if delta_hi > ai * 100:
            break
        f_hi = obj(delta_hi)
    
    # If function is decreasing at delta_lo, minimum is at delta_lo
    # Check subgradient at delta_lo
    if f_lo <= f_hi:
        # Minimum is near delta_lo or at delta_lo
        # Do a small bisection to check
        test = delta_lo + (delta_hi - delta_lo) * 0.01
        if obj(test) >= f_lo:
            return delta_lo
    
    # Bisection for minimum of convex function
    # Use the subgradient: f'(delta) = coeff[argmax_k] (approximately)
    # More precisely, use the fact that f is convex and find where subgradient crosses 0
    
    for _ in range(50):
        mid = (delta_lo + delta_hi) / 2.0
        f_mid = obj(mid)
        
        # Compute subgradient at mid
        new_vals_mid = affected + mid * coeffs
        if has_quad:
            new_vals_mid = new_vals_mid.copy()
            new_vals_mid[quad_idx] += mid * mid
        peak_idx = np.argmax(new_vals_mid)
        if new_vals_mid[peak_idx] > unaffected_max:
            # Subgradient is coeffs[peak_idx] + 2*mid*q[peak_idx]
            if has_quad and peak_idx == quad_idx:
                subgrad = coeffs[peak_idx] + 2 * mid
            else:
                subgrad = coeffs[peak_idx]
        else:
            subgrad = 0  # flat region
        
        if subgrad < 0:
            delta_lo = mid
        else:
            delta_hi = mid
        
        if delta_hi - delta_lo < 1e-12 * max(1, abs(ai)):
            break
    
    return (delta_lo + delta_hi) / 2.0

def coordinate_descent_exact(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent with exact 1D minimization using bisection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            ai = a[i]
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaffected_max = max(unaff_left, unaff_right)
            
            # Find optimal delta
            delta = exact_delta_minimize(a, i, conv, unaffected_max)
            delta = max(delta, -ai)
            
            if abs(delta) > 1e-14:
                # Compute new max to verify improvement
                coeffs = a[k_start - i:k_end - i + 1] + ai
                new_affected = conv[k_start:k_end + 1] + delta * coeffs
                if 2 * i >= k_start and 2 * i <= k_end:
                    new_affected = new_affected.copy()
                    new_affected[2 * i - k_start] += delta * delta
                new_max = max(unaffected_max, float(new_affected.max()))
                
                if new_max < M - 1e-14:
                    # Apply update
                    a[i] = ai + delta
                    conv[k_start:k_end + 1] = new_affected
                    M = new_max
                    improved = True
                    
                    cur_score = 2.0 * n * M
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
        
        # Periodically recompute to avoid numerical drift
        if sweeps % 30 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
        
        if not improved:
            no_improve += 1
            if no_improve > 300:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score, sweeps

def nesterov_subgradient(a0, time_limit, label="", beta_schedule='linear', verbose=True):
    """
    Nesterov-accelerated subgradient method on the exact (non-smooth) objective.
    Uses averaged near-peak subgradients for stability.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    avg_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Step size with Nesterov schedule
        step = 1.0 / np.sqrt(it + 1)
        
        # Nesterov lookahead
        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M_y = float(conv_y.max())
        
        # Average over near-peak positions for stability
        frac = 0.01 + 0.04 * min(1, it / 100)
        threshold = M_y * (1 - frac)
        near_peak = np.where(conv_y >= threshold)[0]
        
        if len(near_peak) == 0:
            near_peak = [conv_y.argmax()]
        
        # Compute weighted subgradient
        weights_peak = conv_y[near_peak]
        weights_peak = weights_peak / weights_peak.sum()
        
        # Efficient gradient using cross-correlation
        grad = np.zeros(n)
        for idx, k in enumerate(near_peak):
            w = weights_peak[idx]
            if w < 1e-12:
                continue
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            j = np.arange(i_min, i_max + 1)
            grad[j] += 2.0 * w * y[k - j]
        
        # Gradient step
        a_new = a - step * grad
        a_new = simplex_project(a_new)
        
        # Nesterov average
        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new
        
        # Evaluate both candidates
        for cand in [a_new, avg_a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * float(fast_convolve(cand).max())
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()
        
        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        
        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSub {label} it={it} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))
        
        # Compute smoothed gradient
        conv_y = fast_convolve(y)
        M_y = float(conv_y.max())
        scaled = beta * (conv_y - M_y)
        scaled = np.clip(scaled, -60, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z
        
        # Gradient computation
        grad = np.zeros(n)
        active_k = np.where(weights > 1e-10)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-12:
                continue
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            j = np.arange(i_min, i_max + 1)
            grad[j] += 2.0 * w * y[k - j]
        
        # Mirror descent step (multiplicative update)
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.85
            beta *= 1.2
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 20 == 0:
            beta *= 1.005
        
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD {label} it={it} ({elapsed:.0f}s): score={best_score:.8f}, beta={beta:.1f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spectral_init(n, sparsity=0.0):
    """
    Initialize using a spectral approach: try to make the power spectrum flat.
    Start with a random phase sequence (as amplitudes) and refine.
    """
    rng = np.random.default_rng(42)
    if sparsity > 0:
        a = np.zeros(n)
        n_nonzero = max(10, int(n * (1 - sparsity)))
        idx = rng.choice(n, n_nonzero, replace=False)
        a[idx] = 1.0 / n_nonzero
    else:
        # Try different structured initializations
        # Arc-shaped (Beta(0.5, 0.5))
        x = np.linspace(1e-6, 1 - 1e-6, n)
        a = 1.0 / np.sqrt(x * (1 - x))
    
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a)}) ***")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization combining:
    1. Exact 1D coordinate descent (bisection-based)
    2. LP Fictitious Play with pre-built matrices
    3. Active-set focused coordinate descent
    4. Nesterov subgradient and smoothed mirror descent
    5. Multi-length exploration with interpolation
    6. Structured perturbations to escape local optima
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)
    
    # Get LP solver
    try:
        solver = get_lp_solver()
        print(f"Using LP solver: {solver}")
    except Exception as e:
        print(f"LP solver detection failed: {e}")
        solver = None
    
    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    def time_remaining():
        return deadline - time.monotonic()
    
    # === Phase 1: Exact coordinate descent on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Exact Coordinate Descent (initial best={best_score:.8f})")
    print(f"{'='*60}")
    rem = time_remaining()

    if t1 > 10:
        print(f"  Budget: {t1:.0f}s, n={len(best_seq)}")
        a1, s1, sw1 = coordinate_descent_exact(best_seq, t1, verbose=True, label="p1")
        best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_CD")

    print(f"\n{'='*60}")
    print(f"Phase 2: LP Fictitious Play (best={best_score:.8f})")
    print(f"{'='*60}")
    if solver is not None and time_remaining() > 200:
        rem = time_remaining() - 300
        t2 = min(max(rem * 0.25, 60), 350.0)
        if t2 > 20:
            print(f"  Budget: {t2:.0f}s, n={len(best_seq)}")
            a2, s2, it2 = lp_fictitious_play_v2(best_seq, t2, solver, label="p2", verbose=True)
            best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_LPFP")
    
    # === Phase 3: Another round of exact CD ===
    print(f"\n{'='*60}")
    print(f"Phase 3: CD Round 2 (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = time_remaining()
    t3 = min(150.0, rem * 0.1)
    if t3 > 10:
        print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
        a3, s3, sw3 = coordinate_descent_exact(best_seq, t3, verbose=True, label="p3")
        best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_CD2")
    
    # === Phase 4: Active-set CD (faster per iteration) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Active-Set CD (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = time_remaining()
    t4 = min(120.0, rem * 0.08)
    if t4 > 10:
        print(f"  Budget: {t4:.0f}s, n={len(best_seq)}")
        a4, s4, it4 = active_set_cd(best_seq, t4, verbose=True, label="p4", tol_frac=0.003)
        best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_ASCD")
    
    # === Phase 5: Nesterov subgradient ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Nesterov Subgradient (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = time_remaining()
    t5 = min(120.0, rem * 0.08)
    if t5 > 20:
        print(f"  Budget: {t5:.0f}s, n={len(best_seq)}")

        best_seq, best_score = check_and_update(a5, best_seq, best_score, "p5_NSub")
    
    # === Phase 6: Smoothed mirror descent ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Smoothed Mirror Descent (best={best_score:.8f})")

    rem = time_remaining()
    t6 = min(120.0, rem * 0.08)
    if t6 > 20:
        print(f"  Budget: {t6:.0f}s, n={len(best_seq)}")

        best_seq, best_score = check_and_update(a6, best_seq, best_score, "p6_SMD")
    
    # === Phase 7: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Multi-Length Exploration (best={best_score:.8f})")
    print(f"{'='*60}")
    n_base = len(best_seq)
    alt_lengths = []
    for alt in [768, 1024, 1280, 1536, 2048, 2560]:
        if alt != n_base and 512 <= alt <= 3072:
            alt_lengths.append(alt)
    alt_lengths = alt_lengths[:4]
    
    for n_alt in alt_lengths:
        rem = time_remaining() - 100
        if rem < 30:
            break
        t_alt = min(rem * 0.25, 100.0)
        if t_alt < 15:
            continue
        
        print(f"\n  Trying n={n_alt}, budget={t_alt:.0f}s")
        init_alt = interpolate_seq(best_seq, n_alt)
        
        # Run CD on this length
        a_alt, s_alt, _ = coordinate_descent_exact(init_alt, t_alt, verbose=False, label=f"p7_n{n_alt}")
        best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"p7_n{n_alt}")
    
    # === Phase 8: Perturbation + re-optimization loop ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Perturb & Re-optimize (best={best_score:.8f})")
    print(f"{'='*60}")
    for rnd in range(3):
        rem = time_remaining() - 80
        if rem < 40:
            break
        
        # Create perturbation
        conv = fast_convolve(best_seq)
        M = conv.max()
        peak_threshold = M * 0.95
        peak_positions = np.where(conv >= peak_threshold)[0]
        
        # Pressure: how much each coordinate contributes to peaks
        n = len(best_seq)
        pressure = np.zeros(n)
        for k in peak_positions:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                pressure[i] += best_seq[i] * best_seq[j]
        
        p_max = pressure.max()
        if p_max > 1e-20:
            pressure = pressure / p_max
        
        strength = 0.02 + 0.01 * rnd
        noise = rng.standard_normal(n) * 0.3
        perturbed = best_seq * (1.0 - strength * pressure + strength * 0.5 + noise * strength * 0.5)
        perturbed = np.maximum(perturbed, 0)
        s_p = perturbed.sum()
        if s_p > 1e-15:
            perturbed = perturbed / s_p
        
        t_pert = min(rem * 0.3, 60.0)
        if t_pert > 10:
            a_p, s_p_score, _ = coordinate_descent_exact(perturbed, t_pert, verbose=False, label=f"p8r{rnd}")
            best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"p8_pert_r{rnd}")
    
    # === Phase 9: Final intensive polishing ===
    print(f"\n{'='*60}")
    print(f"Phase 9: Final Polish (best={best_score:.8f})")
    print(f"{'='*60}")
    
    # CD pass 1
    rem = time_remaining() - 30
    if rem > 20:
        t_f1 = min(rem * 0.4, 80.0)
        a_f1, s_f1, _ = coordinate_descent_exact(best_seq, t_f1, verbose=False, label="p9a")
        best_seq, best_score = check_and_update(a_f1, best_seq, best_score, "p9a")
    
    # Active-set CD
    rem = time_remaining() - 20
    if rem > 15:
        t_f2 = min(rem * 0.4, 50.0)
        a_f2, s_f2, _ = active_set_cd(best_seq, t_f2, verbose=False, label="p9b", tol_frac=0.001)
        best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "p9b")
    
    # Final CD
    rem = time_remaining() - 10
    if rem > 10:
        a_f3, s_f3, _ = coordinate_descent_exact(best_seq, rem * 0.7, verbose=False, label="p9c")
        best_seq, best_score = check_and_update(a_f3, best_seq, best_score, "p9c")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification with the actual evaluator
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Something went wrong, use a safe fallback
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
