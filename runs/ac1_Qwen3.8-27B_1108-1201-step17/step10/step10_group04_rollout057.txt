import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Compute linear convolution a*b using FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-weights to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Get log-weights from a probability vector."""
    return np.log(np.maximum(a, 1e-300))


def evaluate_safe(a):

def lse_objective_and_grad(z, T, n):
    """
    Compute LogSumExp-smoothed objective and its gradient w.r.t. z.
    f_T(a) = (2n/T) * log(sum_k exp(T * conv(a,a)[k]))
    Since a = softmax(z), we compute df_T/dz.
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    w = exp_Tb / sum_exp  # Weights for the active peaks

    # f_T = (2n/T) * (max_Tb + log(sum_exp))
    obj = (2.0 * n / T) * (max_Tb + np.log(sum_exp))

    # Gradient of f_T w.r.t. a:
    # df_T/da[i] = (2n) * sum_k w[k] * d(conv(a,a)[k])/da[i]
    # d(conv(a,a)[k])/da[i] = 2*a[k-i] (with a[x]=0 for x outside [0,n-1])
    # So df_T/da[i] = (2n) * sum_k w[k] * 2*a[k-i] = 4n * (cross_corr of w and a at lag i)

    # Compute sum_k w[k] * a[k-i] for all i using cross-correlation via FFT
    # cross_corr[i] = sum_k w[k] * a[k-i] = (w * a_flipped)[offset + i]
    # Let's compute it directly using the convolution structure
    # sum_k w[k] * a[k-i] where valid k and k-i:
    # = sum_j w[i+j] * a[j] (substituting j = k-i, so k = i+j)
    # This is the cross-correlation of w and a.

    # Using FFT: xcorr = ifft(fft(w) * conj(fft(a))) gives circular cross-correlation
    # We need linear cross-correlation, so pad
    N = len(w) + len(a) - 1  # 2n-1 + n - 1 = 3n-2
    fft_size = 1 << (N - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    xcorr = irfft(W * np.conj(A), fft_size)
    # xcorr[m] = sum_k w[k] * conj(a[m-k]) for circular
    # For real signals: xcorr[m] = sum_k w[k] * a[m-k]
    # We want sum_k w[k] * a[k-i] = xcorr[-i + offset]... 
    # Actually, let me just compute it differently.
    # sum_k w[k] * a[k-i] for i=0,...,n-1
    # Let c = conv(w, a) (linear, length 2n-1+n-1 = 3n-2)... no.
    # conv(w, a)[m] = sum_k w[k] * a[m-k]
    # We want sum_k w[k] * a[k-i] = sum_j w[i+j] * a[j] (j = k-i)
    # = conv(w_flipped, a)[n-1-i + n-1]... this is getting confusing.

    # Simpler: just compute grad_a directly
    grad_a = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 4.0 * n * w[k] * a[k - idx]

    # Convert to gradient w.r.t. z:
    # a = softmax(z), so da/dz is the softmax Jacobian
    # dz_to_da: da_i/dz_j = a_i * (delta_ij - a_j)
    # So grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=500.0, T_end=0.01, n_stages=20):
    """
    L-BFGS with LogSumExp approximation and multi-stage annealing.
    Optimizes over z (log-weights) for efficiency.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_score = 2.0 * n * fast_convolve(a0).max()

    temps = np.geomspace(T_start, T_end, n_stages)
    z_prev = z

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Allocate iterations based on remaining time
        iters_per_stage = max(50, int(20000 / n_stages))

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15, None
            obj, _ = lse_objective_and_grad(z_in, stage_T, n)
            return obj

        def func_jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15, np.zeros_like(z_in)
            return lse_objective_and_grad(z_in, stage_T, n)

        try:
            res = minimize(
                fun=func_jac, x0=z, jac=True, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Check improvement
        a_check = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_check).max()
        if score < best_score:
            best_score = score

    a_final = softmax_from_z(z)
    return a_final


def nesterov_subgrad(a0, time_limit, n_active_frac=0.05):
    """
    Nesterov-accelerated subgradient method for max_k conv(a,a)[k].
    Uses the simplex constraint via softmax parameterization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    y = a.copy()  # Nesterov extrapolation point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient at y (extrapolated point)
        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: for each active k, add a_reversed contribution
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += y[k - idx]
        g /= len(active)

        # Nesterov step
        alpha = 2.0 / (step + 2.0)
        z_new = z - alpha * g * (max_b + 1e-10)
        a_new = softmax_from_z(z_new)

        # Extrapolate
        beta = (1.0 + np.sqrt(1.0 + 4.0 * alpha * alpha)) / 2.0
        y = a_new + (beta - 1.0) / beta * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a_new

        a = a_new
        z = z_new

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            # Perturb to escape
            rng = np.random.default_rng(step * 7 + 3)
            noise = rng.random(n) * 0.02
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            z = z_from_a(a)
            y = a.copy()
            no_improve = 0
            lr = max(lr * 0.8, 0.05)

    return best_a, best_score


def iterative_flattening(a0, time_limit, n_steps=50000):
    """
    New heuristic: iteratively flatten the autocorrelation.
    
    Idea: At each step, identify lags where conv(a,a)[k] is above the target
    (mean = S^2/(2n-1)) and reduce contributions from those lags by adjusting
    a. Specifically, for each "hot" lag k, the products a[j]*a[k-j] that
    contribute to the excess are reduced by transferring mass from high
    contributors to low ones.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        target = 1.0 / (2.0 * n - 1.0)  # Ideal flat value (with sum=1)
        excess = b - target

        # Find the worst excess
        worst_k = int(np.argmax(excess))
        worst_excess = excess[worst_k]

        if worst_excess < 1e-10:
            break  # Already flat enough

        # Find which pairs (j, k-j) contribute to this lag
        js = max(0, worst_k - n + 1)
        je = min(n - 1, worst_k)
        idx = np.arange(js, je + 1)
        partners = worst_k - idx
        contrib = a[idx] * a[partners]

        if contrib.max() < 1e-15:
            continue

        # Move mass from high contributors to a "cold" position
        src_local = int(np.argmax(contrib))
        src = idx[src_local]

        # Find a "cold" position: one that doesn't contribute to the top excess lags
        top_excess_lags = np.argsort(excess)[-3:]
        hot_positions = set()
        for k in top_excess_lags:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            hot_positions.update(range(js2, je2 + 1))

        # Candidates: positions not in hot set, or all positions minus src
        cold_candidates = [j for j in range(n) if j not in hot_positions and j != src]
        if not cold_candidates:
            cold_candidates = [j for j in range(n) if j != src]

        # Pick the coldest candidate (lowest value of a)
        dst = min(cold_candidates, key=lambda j: a[j])

        # Try several transfer amounts
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                break

        # Every 1000 steps, try a random perturbation if stuck
        if step % 1000 == 999:
            # Check if stuck
            b_now = fast_convolve(a)
            if b_now.max() > best_score / (2.0 * n) * 1.001 + 1e-10:
                # We've been stuck, try random restart partial
                pass  # Continue with current approach

    return best_a, best_score


def block_coordinate_descent(a0, time_limit, block_size=20, n_sweeps=10000):

    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)

    n_blocks = n // block_size
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        for bi in range(n_blocks):
            start = bi * block_size
            end = min(start + block_size, n)
            if end - start < 2:
                break

            # For this block, try several perturbations
            a_block = a[start:end].copy()
            for trial in range(5):
                a_trial = a.copy()
                if trial == 0:
                    # Scale up
                    a_trial[start:end] *= 1.1
                elif trial == 1:
                    # Scale down
                    a_trial[start:end] *= 0.9
                elif trial == 2:
                    # Add noise
                    a_trial[start:end] += rng.normal(0, 0.01, end - start)
                elif trial == 3:
                    # Move mass from center of block to edges
                    mid = (start + end) // 2
                    amt = a[mid] * 0.1
                    a_trial[mid] -= amt
                    a_trial[start + 1] += amt
                else:
                    # Random redistribution within block
                    total = a_trial[start:end].sum()
                    a_trial[start:end] = rng.dirichlet(np.ones(end - start)) * total

                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    break

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * np.linspace(0, 1, n))
        a = np.maximum(a, 0.01)
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(7 * x)
        a = np.maximum(a, 0)
    elif choice == 6:
        # Gaussian with oscillation
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2) * (1 + 0.3 * np.cos(10 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 7:
        # Sinc-like
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0)
    elif choice == 8:
        # Multi-bump
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.3) ** 2) + 0.5 * np.exp(-10 * (x - 0.7) ** 2)
    else:
        # Smooth random
        a = np.abs(np.cumsum(rng.normal(0, 1, n)))
        a = a / a.max()

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def optimize_at_length(a_init, time_budget, label=""):
    """
    Run multi-phase optimization at a fixed length.
    Phases: LogSumExp L-BFGS, Nesterov subgrad, flattening, block CD.
    """
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    def try_update(a, src):
        nonlocal best_a, best_score
        if a is None:
            return False
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 1e-15:
            return False
        a = a / ss
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  [{src}] NEW BEST: {2*n*sc/n:.6f} obj={2*n*sc/(ss*ss):.6f} (n={n}, t={elapsed:.0f}s)")
            return True
        return False

    # Phase 1: LogSumExp L-BFGS (most effective for smooth optimization)
    rem = time_budget - (time.monotonic() - t0)
    t1 = min(rem * 0.40, 400.0)
    if t1 > 20:
        print(f"  Phase 1: LSE L-BFGS, budget={t1:.0f}s")
        a_out = lbgfs_logsumexp(best_a, t1, T_start=300.0, T_end=0.01, n_stages=20)
        try_update(a_out, "lse")

    # Phase 2: Nesterov subgradient
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.20, 150.0)
    if t2 > 10:
        print(f"  Phase 2: Nesterov subgrad, budget={t2:.0f}s")
        a_out, _ = nesterov_subgrad(best_a, t2, n_active_frac=0.03)
        try_update(a_out, "nesterov")

    # Phase 3: Iterative flattening
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.20, 150.0)
    if t3 > 10:
        print(f"  Phase 3: Flattening, budget={t3:.0f}s")
        a_out, _ = iterative_flattening(best_a, t3, n_steps=200000)
        try_update(a_out, "flatten")

    # Phase 4: Block coordinate descent
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.10, 80.0)
    if t4 > 5:
        print(f"  Phase 4: Block CD, budget={t4:.0f}s")
        a_out, _ = block_coordinate_descent(best_a, t4, block_size=10, n_sweeps=50000)
        try_update(a_out, "blockcd")

    # Phase 5: More LSE with lower temperature (fine-tuning)
    rem = time_budget - (time.monotonic() - t0)
    t5 = min(rem * 0.10, 60.0)
    if t5 > 5:
        print(f"  Phase 5: LSE fine-tune, budget={t5:.0f}s")
        a_out = lbgfs_logsumexp(best_a, t5, T_start=20.0, T_end=0.005, n_stages=10)
        try_update(a_out, "lse_fine")

    return best_a, best_score


def peak_equalization(a0, time_limit):
    """
    Final polish: directly equalize the top autocorrelation peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 5000:
            break

        b = fast_convolve(a)
        sorted_idx = np.argsort(b)[::-1]
        top_k_list = sorted_idx[:min(5, len(b))]
        top_vals = b[top_k_list]
        peak_gap = top_vals[0] - top_vals[-1]

        improved = False
        for k in top_k_list:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            partners = k - idx
            contrib = a[idx] * a[partners]

            src = idx[int(np.argmax(contrib))]

            # Find destination: position with lowest contribution to top peaks
            peak_areas = set()
            for k2 in top_k_list:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            dst = min(candidates, key=lambda j: a[j])

            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Random perturbation every few hundred steps
            if no_improve % 200 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.01, 0.05)
                    a_trial = a.copy()
                    amt = a[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            no_improve = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Explores multiple sequence lengths using
    LogSumExp L-BFGS, Nesterov subgradient, iterative flattening,
    block coordinate descent, and peak equalization.
    """
    global time_limit_global
    rng = np.random.default_rng(seed)
    time_limit_global = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = 2.0 * n_base * fast_convolve(best_seq).max()
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    print(f"Starting: n={n_base}, best_score={best_score:.6f}")
    print(f"Time budget: {time_limit_global - time.monotonic():.0f}s")

    # Phase 1: Intensive optimization at base length
    print(f"\n=== Phase 1: Optimize at n={n_base} ===")
    rem = time_limit_global - time.monotonic()
    t1 = min(rem * 0.50, 800.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, sc_out = optimize_at_length(best_seq, t1, label=f"n{n_base}")
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  After phase 1: best={best_score:.6f}")

    # Phase 2: Try longer lengths (often better for this objective)
    print(f"\n=== Phase 2: Longer lengths (best={best_score:.6f}) ===")
    longer_lengths = [int(n_base * m) for m in [1.3, 1.5, 2.0, 2.5]]
    longer_lengths = [l for l in longer_lengths if l > n_base and l <= 5000]
    longer_lengths = sorted(set(longer_lengths))[:3]

    for ln in longer_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.20, 250.0)
        if t_alloc < 60:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_sequence(best_seq, ln)
        sc_init = 2.0 * ln * fast_convolve(init_long).max()
        cur = init_long
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"    Interpolation improved: {best_score:.6f}")

        a_out, sc_out = optimize_at_length(cur, t_alloc, label=f"n{ln}")
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"    After opt: best={best_score:.6f}")

    # Phase 3: Try shorter lengths
    print(f"\n=== Phase 3: Shorter lengths (best={best_score:.6f}) ===")
    shorter_lengths = [int(n_base * m) for m in [0.7, 0.5, 0.33]]
    shorter_lengths = [l for l in shorter_lengths if l < n_base and l >= 200]
    shorter_lengths = sorted(set(shorter_lengths))[:2]

    for sn in shorter_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 90:
            break
        t_alloc = min(rem * 0.12, 100.0)
        if t_alloc < 30:
            break
        print(f"  Trying n={sn}, budget={t_alloc:.0f}s")
        init_short = interpolate_sequence(best_seq, sn)
        a_out, sc_out = optimize_at_length(init_short, t_alloc, label=f"n{sn}")
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"    After opt: best={best_score:.6f}")

    # Phase 4: Final polish on the best sequence
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit_global - time.monotonic()
    if rem > 30:
        # Peak equalization (most effective for final polish)
        t_pe = min(rem * 0.40, 120.0)
        if t_pe > 10:
            print(f"  Peak equalization: {t_pe:.0f}s")
            a_fin, sc_fin = peak_equalization(best_seq, t_pe)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"    After PE: {best_score:.6f}")

        # More LSE fine-tuning
        rem = time_limit_global - time.monotonic()
        if rem > 15:
            t_lse = min(rem * 0.30, 60.0)
            if t_lse > 5:
                print(f"  LSE fine-tune: {t_lse:.0f}s")
                a_fin2 = lbgfs_logsumexp(best_seq, t_lse, T_start=10.0, T_end=0.005, n_stages=8)
                sc_fin2 = 2.0 * len(a_fin2) * fast_convolve(a_fin2).max()
                if sc_fin2 < best_score:
                    best_score = sc_fin2
                    best_seq = a_fin2.copy()
                    print(f"    After LSE: {best_score:.6f}")

        # More peak equalization
        rem = time_limit_global - time.monotonic()
        if rem > 10:
            t_pe2 = min(rem * 0.40, 40.0)
            if t_pe2 > 5:
                print(f"  More PE: {t_pe2:.0f}s")
                a_fin3, sc_fin3 = peak_equalization(best_seq, t_pe2)
                if sc_fin3 < best_score:
                    best_score = sc_fin3
                    best_seq = a_fin3.copy()
                    print(f"    After PE2: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    assert best_seq is not None and len(best_seq) > 0
    return [float(x) for x in best_seq]
