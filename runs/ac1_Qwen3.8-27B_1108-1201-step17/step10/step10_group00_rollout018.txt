import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    # Try solvers
    for s in [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value), 0)
                b_val = b_val / b_val.sum()
                return b_val, float(M_var.value)
        except:
            continue
    return None, None

def logsumexp_obj(z, n, T):
    a = softmax(z)
    b = fast_convolve(a)
    max_b = b.max()
    lse = max_b + T * np.log(np.sum(np.exp((b - max_b) / T)))
    obj = 2.0 * n * lse
    # Gradient
    w = np.exp((b - max_b) / T) / np.sum(np.exp((b - max_b) / T))
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def lse_lbgfs(a0, time_limit, T_stages):

    t0 = time.monotonic()

# === Core utilities ===

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))

    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def lse_obj(x, n, T):
    """LogSumExp smoothed objective."""
    a = softmax_from_z(x)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    lse = max_b + T * np.log(np.sum(np.exp(shifted)))
    return 2.0 * n * lse

def lse_grad(x, n, T):
    """Gradient of LSE objective w.r.t. z."""
    a = softmax_from_z(x)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    exp_terms = np.exp(shifted)
    w = exp_terms / exp_terms.sum()
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def nesterov_lp_refine(a0, time_limit, solver_name, n_max_iters=500000):

    old_x = np.linspace(0, 1, old_n)

    f_interp = np.maximum(f_interp, 0)

    if ss > 1e-15:

    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver_name):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver_name, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def lse_obj(x, n, T):
    """LogSumExp smoothed objective."""
    a = softmax_from_z(x)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    lse = max_b + T * np.log(np.sum(np.exp(shifted)))
    return 2.0 * n * lse


def lse_grad(x, n, T):
    """Gradient of LSE objective w.r.t. z."""
    a = softmax_from_z(x)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    exp_terms = np.exp(shifted)
    w = exp_terms / exp_terms.sum()

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def nesterov_lp_refine(a0, time_limit, solver_name, n_max_iters=500000):
    """Nesterov-accelerated LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver_name)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                # Structured perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise * 0.5
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

            # Convergence check: if LP response is very close to current, we're at fixed point
            if (it + 1) % 200 == 0:
                dist = np.linalg.norm(b_new - a)
                if dist < 1e-6:
                    print(f"    LP converged at iter {it+1}, dist={dist:.2e}")
                    break

    return best_a, best_score


def lse_lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=300):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for T in T_stages:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 3:
            break
        if T < 0.01 and elapsed > time_limit * 0.5:
            # Don't spend too much time on very low temperatures if already used half the budget
            if time.monotonic() - t0 > time_limit - 10:
                break
        try:
            res = minimize(
                fun=lse_obj, x0=z, jac=lse_grad,
                args=(n, T),
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-16, 'maxfun': maxiter_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def structured_init(n, rng, variant=0):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 1:
        a = np.sin(np.pi * x) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif variant == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif variant == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive Nesterov LP alternating, ultra-fine LSE L-BFGS,
    multi-length exploration, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, variant=0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Nesterov LP on parent (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.32, 540.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nesterov_lp_refine(best_seq, t_alloc, solver)
        if a_lp is not None:
            a_lp = np.maximum(a_lp, 0)
            a_lp = a_lp / a_lp.sum()
            sc = evaluate_local(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
                print(f"  NEW BEST [lp_1]: {sc:.8f}")

    # === Phase 2: Ultra-fine LSE L-BFGS ===
    print(f"\n=== Phase 2: LSE L-BFGS fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25,
                    0.15, 0.1, 0.07, 0.05, 0.035, 0.025, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002]
        print(f"  {len(T_stages)} stages, budget={t_alloc:.0f}s")
        a_lb = lse_lbgfs_refine(best_seq, t_alloc, T_stages, maxiter_per_stage=250)
        sc = evaluate_local(a_lb)
        if sc < best_score:
            best_score = sc
            best_seq = a_lb.copy()
            print(f"  NEW BEST [lse_fine]: {sc:.8f}")

    # === Phase 3: Second LP pass from updated best ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver)
        sc = evaluate_local(a_lp2)
        if sc < best_score:
            best_score = sc
            best_seq = a_lp2.copy()
            print(f"  NEW BEST [lp_2]: {sc:.8f}")

    # === Phase 4: Final LSE polish ===
    print(f"\n=== Phase 4: LSE Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        T_stages2 = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.001]
        a_lb2 = lse_lbgfs_refine(best_seq, t_alloc, T_stages2, maxiter_per_stage=200)
        sc = evaluate_local(a_lb2)
        if sc < best_score:
            best_score = sc
            best_seq = a_lb2.copy()
            print(f"  NEW BEST [lse_polish]: {sc:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        lengths_to_try = [1400, 1000, 1600, 2000, 800]
        for n_try in lengths_to_try:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.20, 120.0)
            print(f"  Trying n={n_try}, budget={t_sub:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_refine(init_try, t_sub * 0.65, solver)
            sc_try = evaluate_local(a_try)
            if sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
                print(f"  NEW BEST [lp_n{n_try}]: {sc_try:.8f}")
                # If improved, do LSE polish
                rem = deadline - time.monotonic()
                if rem > 30:
                    T_stages3 = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
                    a_try2 = lse_lbgfs_refine(best_seq, min(rem, 60.0), T_stages3, maxiter_per_stage=200)
                    sc2 = evaluate_local(a_try2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_seq = a_try2.copy()
                        print(f"  NEW BEST [lse_n{n_try}]: {sc2:.8f}")

    # === Phase 6: Perturbation restarts ===
    print(f"\n=== Phase 6: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            strength = 0.015 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength)
            t_sub = min(rem_sub * 0.25, 50.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver)
            sc_opt = evaluate_local(a_opt)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  NEW BEST [pert{trial}]: {sc_opt:.8f}")

    # === Final: LSE polish with lowest temperatures ===
    rem = deadline - time.monotonic()
    if rem > 15:
        T_final = [1.0, 0.5, 0.3, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        a_fin = lse_lbgfs_refine(best_seq, min(rem - 2, 60.0), T_final, maxiter_per_stage=150)
        sc_fin = evaluate_local(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"  NEW BEST [final_lse]: {sc_fin:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
