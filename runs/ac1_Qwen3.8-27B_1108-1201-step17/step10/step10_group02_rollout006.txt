```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LogSumExp-Adam optimization in log-space,
spectral nudging for peak reduction, multi-length exploration, and coordinate polish.
Key innovations: Adam optimizer with adaptive learning rate for high-dimensional
non-convex optimization, spectral nudging steps that directly target convolution peaks,
and multi-n exploration to escape local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_obj_grad(a, T):
    """
    Compute LogSumExp-smoothed objective and gradient wrt a.
    Objective: 2n * LSE_T(conv(a,a)) where LSE_T(x) = max(x) + T*log(mean(exp((x-max)/T)))
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    exp_scaled = np.exp(scaled)
    lse = max_b + T * np.log(exp_scaled.sum() / len(b))
    w = exp_scaled / exp_scaled.sum()

    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (valid indices only)
    # Using FFT: grad_a = 2 * (a conv w_rev)[n-1 : 2n-1][::-1]
    a_rev = a[::-1]
    size_conv = (2 * n - 1) + n - 1
    fft_size = 1 << (size_conv - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(W * AR, fft_size)
    grad_a = 2.0 * conv_result[n - 1:2 * n - 1][::-1]

    obj = 2.0 * n * lse
    return obj, grad_a


def softmax_gradient(fgrad, a):
    """Convert gradient from a-space to z-space for softmax parameterization."""
    return a * (fgrad - np.dot(fgrad, a))


def logsumexp_adam(a0, time_limit, T_start=50.0, T_end=0.3, lr=0.01,
                   beta1=0.9, beta2=0.999, eps=1e-8, nudge_interval=200,
                   nudge_strength=0.05, verbose=True):
    """
    Main optimizer: LogSumExp objective with Adam in log-space.
    Periodic spectral nudging to target convexity peaks.
    """
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a0 = a0 / a0.sum()
    n = len(a0)
    z = np.log(a0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a0.copy()
    best_score = evaluate_local(a0)
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)

    total_iter_target = max(1000, int(time_limit * 8))  # estimate iterations

    for it in range(max(500000, total_iter_target)):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal temperature
        frac = min(1.0, it / total_iter_target)
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 0.1)

        # Adaptive learning rate (decay over time)
        lr_t = lr * (1.0 / (1.0 + 0.001 * it))

        # Compute objective and gradient
        a = softmax(z)
        obj, fgrad = logsumexp_obj_grad(a, T)
        zgrad = softmax_gradient(fgrad / T, a)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * zgrad
        v = beta2 * v + (1 - beta2) * zgrad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic spectral nudge
        if it % nudge_interval == nudge_interval - 1 and it > nudge_interval:
            a_nudge = softmax(z)
            b_nudge = fast_convolve(a_nudge)
            max_b_nudge = b_nudge.max()
            threshold = max_b_nudge * 0.999
            peaks = np.where(b_nudge >= threshold)[0]
            if len(peaks) > 0:
                # Compute subgradient at peaks
                sg = np.zeros(n)
                for k in peaks:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    sg[idx] += a_nudge[k - idx]
                sg /= max(1, len(peaks))
                # Project to tangent space of simplex
                sg = sg - np.dot(sg, a_nudge) * a_nudge
                # Apply nudge in z-space
                z = z - nudge_strength * sg / (np.abs(sg).max() + 1e-10)

        # Periodic perturbation to escape local minima
        if it % 500 == 499:
            noise_scale = 0.002 * (1 + no_improve / 100.0)
            z = z + rng.standard_normal(n) * noise_scale

        # Evaluate
        a_new = softmax(z)
        score = evaluate_local(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if very stuck
        if no_improve > 500 and it > 1000:
            # Large perturbation
            z = z + rng.standard_normal(n) * 0.02
            no_improve = 0

        if verbose and (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam iter={it+1} ({elapsed:.0f}s): T={T:.3f} score={best_score:.8f} no_improve={no_improve}")

    return best_a, best_score


def spectral_flatten_step(a, n_steps=50, step_size=0.02):
    """
    Spectral flattening: iteratively reduce convolution peaks by adjusting
    the frequency content of the sequence.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    a = a / a.sum()
    n = len(a)

    for step in range(n_steps):
        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Target: flatten the convolution
        # Compute a "pressure" at each peak
        threshold = max_b * 0.9995
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        # Gradient pointing toward reducing peaks
        grad = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= max(1, len(peaks))

        # Line search: find best step size
        best_s = 0
        best_score = evaluate_local(a)
        for s in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
            a_trial = a - s * grad
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = evaluate_local(a_trial)
            if sc < best_score:
                best_score = sc
                best_s = s
                a = a_trial

    return a


def coordinate_polish(a0, time_limit, max_steps=50000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find the global peak
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try reducing top contributors
        top_m = min(8, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-14:
                continue
            for factor in [0.999, 0.998, 0.996, 0.994, 0.99, 0.985, 0.98, 0.97]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                # Redistribute to non-peak elements
                mask = np.ones(n, dtype=bool)
                # Don't redistribute to elements that also contribute to this peak
                for j in idx:
                    if a[peak_k - j] > 1e-12:
                        mask[j] = False
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    mask[pos] = True  # Allow redistribution to pos if nothing else
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPolish {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3 * (x - 0.3) ** 2 / 0.1)
    elif choice == 5:
        a = np.exp(-3 * (x - 0.7) ** 2 / 0.1)
    else:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * 8 * x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-10)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def quick_mirror_descent(a0, time_limit, eta=0.06):
    """Fast mirror descent for initial exploration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

        # Mirror descent update
        log_a = np.log(a)
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * (a + alpha * (a_new - a)) + alpha * a_new if it > 0 else a_new
        # Simplified: just use a_new
        a = a_new

        score = evaluate_local(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Multi-length exploration with fast mirror descent
    Phase 2: Deep LogSumExp-Adam optimization
    Phase 3: Spectral flattening and coordinate polish
    Phase 4: Final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_parent = len(best_seq) if best_seq is not None else 1200

    # === Phase 1: Multi-length exploration (150s) ===
    print(f"\n=== Phase 1: Multi-length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.10, 150.0)
    if t_phase1 > 30:
        print(f"  Budget: {t_phase1:.0f}s")

        # Determine which n values to try
        n_candidates = [n_parent]
        for factor in [0.75, 1.25, 1.5, 2.0]:
            n_try = int(n_parent * factor)
            if n_try >= 400 and n_try <= 4000 and n_try not in n_candidates:
                n_candidates.append(n_try)
        # Also try some absolute values
        for n_abs in [800, 1000, 1600, 2400]:
            if n_abs not in n_candidates and 400 <= n_abs <= 4000:
                n_candidates.append(n_abs)

        n_candidates = sorted(set(n_candidates))[:6]  # Limit to 6
        print(f"  Trying n values: {n_candidates}")

        candidates = []
        t_per_n = t_phase1 / max(1, len(n_candidates) * 2)

        for n_try in n_candidates:
            for init_choice in range(2):
                if time.monotonic() >= deadline - t_phase1 * 0.5:
                    break

                if init_choice == 0 and parent is not None and n_try == n_parent:
                    a_init = parent.copy()
                elif init_choice == 0 and parent is not None:
                    a_init = interpolate_sequence(parent, n_try)
                else:
                    a_init = structured_init(n_try, rng)

                a_opt, sc = quick_mirror_descent(a_init, t_per_n, eta=0.05)
                if sc < best_score:
                    candidates.append((sc, a_opt.copy()))
                    try_update(a_opt, f"expl_n{n_try}_i{init_choice}")

        # Keep top candidates for phase 2
        candidates.sort(key=lambda x: x[0])
        print(f"  Phase 1 done. Best: {best_score:.8f}")

    # === Phase 2: Deep LogSumExp-Adam optimization (1200s) ===
    print(f"\n=== Phase 2: LogSumExp-Adam (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.70, 1200.0)
    if t_phase2 > 60 and best_seq is not None:
        print(f"  Budget: {t_phase2:.0f}s, n={len(best_seq)}")

        # Split phase 2 into two halves: first with higher T, second with lower T
        t_half = t_phase2 / 2

        a_opt, _ = logsumexp_adam(best_seq, t_half,
                                   T_start=40.0, T_end=3.0,
                                   lr=0.008, nudge_interval=150,
                                   nudge_strength=0.04)
        try_update(a_opt, "logsumexp_adam_p2a")

        rem = deadline - time.monotonic()
        t_second = min(rem * 0.40, t_phase2 - t_half)
        if t_second > 30 and best_seq is not None:
            a_opt, _ = logsumexp_adam(best_seq, t_second,
                                       T_start=3.0, T_end=0.3,
                                       lr=0.003, nudge_interval=100,
                                       nudge_strength=0.02)
            try_update(a_opt, "logsumexp_adam_p2b")

    # === Phase 3: Spectral flattening + Coordinate polish (250s) ===
    print(f"\n=== Phase 3: Spectral + Coord (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.14, 250.0)
    if t_phase3 > 20 and best_seq is not None:
        print(f"  Budget: {t_phase3:.0f}s")

        # Spectral flattening
        t_spec = t_phase3 * 0.4
        a_spec = spectral_flatten_step(best_seq, n_steps=200, step_size=0.015)
        try_update(a_spec, "spectral_flatten")

        # Coordinate polish
        rem = deadline - time.monotonic()
        t_coord = min(rem * 0.60, t_phase3 * 0.6)
        if t_coord > 10 and best_seq is not None:
            a_cd, _ = coordinate_polish(best_seq, t_coord, max_steps=100000)
            try_update(a_cd, "coord_polish")

    # === Phase 4: Final refinement (remaining time) ===
    print(f"\n=== Phase 4: Final Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        print(f"  Budget: {rem:.0f}s")

        # One more round of low-temperature Adam
        t_final_adam = rem * 0.5
        if t_final_adam > 15:
            a_fin, _ = logsumexp_adam(best_seq, t_final_adam,
                                       T_start=1.0, T_end=0.1,
                                       lr=0.001, nudge_interval=50,
                                       nudge_strength=0.01)
            try_update(a_fin, "final_adam")

        # Final coordinate polish
        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None:
            a_polish, _ = coordinate_polish(best_seq, rem * 0.8, max_steps=50000)
            try_update(a_polish, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Also try evaluate_sequence for final check
        try:
            seq_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
            ev_score = evaluate_sequence(seq_list)
            if np.isfinite(ev_score):
                print(f"  evaluate_sequence confirms: {ev_score:.8f}")
        except Exception:
            pass

        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
