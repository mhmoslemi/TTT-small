    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    if s < 0.01:

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-300:

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast top-k subgradient descent in log-domain (primary workhorse)
2. FFT-accelerated gradient computation via cross-correlation
3. Multi-start exploration with diverse initializations
4. L-BFGS with ultra-fine LogSumExp for final polish
5. Adaptive momentum and learning rate schedules
Key innovations: vectorized gradient via FFT, adaptive active-set sizing,
and aggressive multi-restart strategy to escape local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(c, a):
    """Compute cross-correlation: result[i] = sum_k c[k] * a[k-i] (with bounds)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    C = rfft(c, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation via FFT: conj(fft(c)) * fft(a) gives correlation
    # But we need: sum_k c[k] * a[k-i] which is (c reversed) convolved with a
    # Actually: ifft(conj(fft(c)) * fft(a))[m] = sum_k c[k] * a[k+m]
    # We want sum_k c[k] * a[k-i], so we need shift by -i
    # Let's use: result[i] = ifft(conj(fft(c)) * fft(a))[fft_size - i] for i in [0, n)
    # Simpler: just compute the full convolution of c with flip(a)
    flip_a = a[::-1]
    R = rfft(flip_a, fft_size)
    corr = irfft(C * R, fft_size)
    # corr[k] = sum_j c[j] * flip_a[k-j] = sum_j c[j] * a[n-1-k+j]
    # We want g[i] = sum_k c[k] * a[k-i]
    # = sum_k c[k] * flip_a[n-1-k+i]
    # Let m = n-1+i: sum_k c[k] * flip_a[m-k] = corr[m] (if m is in range)
    # i ranges from 0 to n-1, so m ranges from n-1 to 2n-2
    result = corr[n - 1:2 * n - 1]
    return result


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_vec(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def topk_indices(b, k):
    """Find indices of top-k values in b."""
    if k >= len(b):
        return range(len(b))
    if k == 1:
        return [int(np.argmax(b))]
    if k <= 50:
        return list(np.argpartition(b, -k)[-k:])
    else:
        return list(np.argsort(b)[-k:])


def compute_grad_direct(a, active_indices):
    """Compute average gradient of b[k] over active set, direct method."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active_indices)
    return g


def compute_grad_fft(a, active_indices):
    """Compute average gradient of b[k] over active set, using FFT cross-correlation."""
    n = len(a)
    c = np.zeros(2 * n - 1)
    for k in active_indices:
        c[k] = 1.0
    g = fast_cross_corr(c, a)
    g *= 2.0 / len(active_indices)
    return g


def gradient_descent(a0, time_limit, topk_stages, lr_list, momentum_list, name="GD"):
    """
    Fast gradient descent in log-domain with adaptive top-k active set.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
        s = 1.0
    a0 = a0 / s

    z = np.log(a0 + 1e-30)
    z = z - z.mean()

    t0 = time.monotonic()
    deadline = t0 + time_limit
    best_a = a0.copy()
    best_score = evaluate_local(a0)

    a = softmax_vec(z)
    v = np.zeros(n)
    use_fft_grad = n >= 200  # Use FFT gradient for larger n

    for stage in range(len(topk_stages)):
        topk = min(topk_stages[stage], 2 * n - 1)
        lr = lr_list[min(stage, len(lr_list) - 1)]
        mu = momentum_list[min(stage, len(momentum_list) - 1)]

        stage_t0 = time.monotonic()
        no_improve = 0
        iters = 0

        while time.monotonic() < deadline:
            # Limit per-stage time
            stage_elapsed = time.monotonic() - stage_t0
            max_stage_time = time_limit / len(topk_stages) * 3
            if stage_elapsed > max_stage_time and stage < len(topk_stages) - 1:
                break

            # Compute convolution
            b = fast_convolve(a)
            max_b = b.max()

            # Find top-k active set
            if topk <= 20 and n < 500:
                active = topk_indices(b, topk)
                g_a = compute_grad_direct(a, active)
            else:
                active = topk_indices(b, topk)
                if use_fft_grad:
                    g_a = compute_grad_fft(a, active)
                else:
                    g_a = compute_grad_direct(a, active)

            # Transform gradient to z-domain (softmax Jacobian)
            S = np.dot(g_a, a)
            g_z = a * (g_a - S)

            # Momentum update
            v = mu * v + g_z
            z = z - lr * v

            # Update a
            a = softmax_vec(z)

            # Evaluate score (sum(a) = 1 by construction)
            b = fast_convolve(a)
            score = 2.0 * n * b.max()

            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            iters += 1

            # Adaptive: reset momentum if stuck too long
            if no_improve > 3000:
                v[:] = 0
                no_improve = 0

            # Debug print
            if iters % 200000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {name} stage {stage+1} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6f} topk={topk}")

        print(f"    {name} stage {stage+1} done ({time.monotonic()-stage_t0:.1f}s): best={best_score:.8f}")

    return best_a, best_score


def lbgfs_logsumexp_refine(a0, time_limit, T_stages):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
    a0 = a0 / s

    z = np.log(a0 + 1e-30)
    z = z - z.mean()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        maxiter = max(20, int(remaining / len(T_stages) * 3))

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            a_in = softmax_vec(z_in)
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_terms = np.exp((b_in - max_b) / T)
            lse = max_b + T * np.log(exp_terms.sum())
            return 2.0 * n * lse

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            a_in = softmax_vec(z_in)
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_terms = np.exp((b_in - max_b) / T)
            w = exp_terms / exp_terms.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S) * 2.0 * n / T

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_vec(z)


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-8 * (x - 0.5)))
    else:
        a = np.ones(n) * (0.5 + 0.2 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a using low-frequency cosine."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def perturb_sequence_random(a, rng, strength=0.05):
    """Create a perturbed version with Gaussian noise."""
    a_pert = a + strength * rng.standard_normal(len(a))
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
