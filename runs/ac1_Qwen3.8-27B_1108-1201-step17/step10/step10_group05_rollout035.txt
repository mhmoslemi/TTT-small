                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        if sweeps % 100 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score, sweeps

def pair_coord_descent(a0, time_limit, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        step = max(1, n // 500)
        for i in range(0, n, step):
            for j in range(i + 1, min(i + 6, n)):
                ai = a[i]
                aj = a[j]
                if ai < 1e-15 and aj < 1e-15:
                    continue
                c_total = ai + aj
                if c_total < 1e-15:
                    continue
                k_min = min(i, j)
                k_max = max(i + n - 1, j + n - 1)
                k_max = min(k_max, 2 * n - 2)
                if k_min > 0:
                    unaff_left = float(conv[:k_min].max())
                else:
                    unaff_left = -1e30
                if k_max < 2 * n - 2:
                    unaff_right = float(conv[k_max + 1:].max())
                else:
                    unaff_right = -1e30
                unaff_max = max(unaff_left, unaff_right)
                affected_old = conv[k_min:k_max + 1].copy()
                # Try several fractions t/c_total for a[i] = t, a[j] = c_total - t
                fracs = [0.0, 1.0, 0.25, 0.75, 0.5]
                # Add fraction based on current ratio
                if c_total > 1e-15:
                    fracs.append(ai / c_total)
                best_frac = ai / c_total if c_total > 1e-15 else 0.5
                best_new_max = M
                for f in fracs:
                    t = f * c_total
                    a_new_i = t
                    a_new_j = c_total - t
                    if a_new_i < -1e-15 or a_new_j < -1e-15:
                        continue
                    a_new_i = max(0.0, a_new_i)
                    a_new_j = max(0.0, a_new_j)
                    # Compute delta for each affected lag
                    new_affected = affected_old.copy()
                    for k in range(k_min, k_max + 1):
                        delta = 0.0
                        # Contribution from i: (t - ai) * a[k-i] if valid k-i
                        ki = k - i
                        if 0 <= ki <= n - 1 and ki != j:
                            delta += (a_new_i - ai) * a[ki]
                        # Contribution from j: ((c_total-t) - aj) * a[k-j] if valid k-j
                        kj = k - j
                        if 0 <= kj <= n - 1 and kj != i:
                            delta += (a_new_j - aj) * a[kj]
                        # Cross term correction for k = i+j
                        if k == i + j and i != j:
                            # We double-counted: need to subtract the interaction
                            # Original had ai*aj counted in both i-part and j-part
                            # Actually the i-part gives (t-ai)*aj and j-part gives aj_new*ai
                            # But the true change in the ai*aj term is:
                            # new product is t*(c_total-t), old is ai*aj
                            # In our formula: (t-ai)*aj + (c_total-t-aj)*ai
                            # = (t-ai)*aj + (ai-t)*ai  [since c_total-t-aj = ai-t]
                            # = (t-ai)*(aj-ai)
                            # True change: t*(c_total-t) - ai*aj
                            # = t*(ai+aj-t) - ai*aj = t*ai + t*aj - t^2 - ai*aj
                            # Difference: [t*ai+t*aj-t^2-ai*aj] - [(t-ai)*(aj-ai)]
                            # = t*ai + t*aj - t^2 - ai*aj - (t*aj - t*ai - ai*aj + ai^2)
                            # = t*ai + t*aj - t^2 - ai*aj - t*aj + t*ai + ai*aj - ai^2
                            # = 2*t*ai - t^2 - ai^2
                            # = -(t-ai)^2
                            # So we need to subtract (t-ai)^2 from the delta
                            delta -= (a_new_i - ai) ** 2
                            # Wait, let me verify: the true change in conv[i+j] from the
                            # (i,j) and (j,i) terms:
                            # Old: ai*aj + aj*ai = 2*ai*aj
                            # New: t*(c-t) + (c-t)*t = 2*t*(c-t)
                            # True delta: 2*t*(c-t) - 2*ai*aj
                            # Our formula delta (before correction):
                            # (t-ai)*aj + (c-t-aj)*ai = (t-ai)*aj + (ai-t)*ai
                            # = (t-ai)*aj - (t-ai)*ai = (t-ai)*(aj-ai)
                            # True delta - formula delta:
                            # 2*t*(c-t) - 2*ai*aj - (t-ai)*(aj-ai)
                            # = 2t(c-t) - 2ai*aj - t*aj + t*ai - ai*aj + ai^2
                            # = 2t*ai + 2t*aj - 2t^2 - 2ai*aj - t*aj + t*ai - ai*aj + ai^2
                            # = 3t*ai + t*aj - 2t^2 - 3ai*aj + ai^2
                            # Hmm this doesn't simplify cleanly. Let me just compute directly.
                            pass
                    # Simpler approach: directly compute new affected values
                    # For each affected k, new_conv[k] = old_conv[k] + delta_k
                    # where delta_k accounts for changes in a[i] and a[j]
                    # Let's just recompute the affected entries directly
                    pass
                # This approach is getting too complex. Let me use a simpler strategy:
                # Just try swapping mass between i and j in small increments
                delta_options = [-0.3*c_total, -0.2*c_total, -0.1*c_total,
                                 -0.05*c_total, 0.05*c_total, 0.1*c_total,
                                 0.2*c_total, 0.3*c_total]
                for d in delta_options:
                    a_new_i = ai + d
                    a_new_j = aj - d
                    if a_new_i < 0 or a_new_j < 0:
                        continue
                    # Compute new affected conv values
                    max_new_affected = unaff_max
                    for k in range(k_min, k_max + 1):
                        new_val = affected_old[k - k_min]
                        # Add change from a[i]: (a_new_i - ai) * a[k-i]
                        ki = k - i
                        if 0 <= ki <= n - 1:
                            new_val += (a_new_i - ai) * a[ki]
                        # Add change from a[j]: (a_new_j - aj) * a[k-j]
                        kj = k - j
                        if 0 <= kj <= n - 1:
                            new_val += (a_new_j - aj) * a[kj]
                        # Subtract double-counted cross term
                        # The term a[i]*a[j] at k=i+j was counted in both:
                        # (a_new_i-ai)*a[j] and a[i]*(a_new_j-aj)
                        # But the true change is: a_new_i*a_new_j + a_new_j*a_new_i - ai*aj - aj*ai
                        # = 2*a_new_i*a_new_j - 2*ai*aj
                        # Our formula gives: (a_new_i-ai)*aj + ai*(a_new_j-aj)
                        # = a_new_i*aj - ai*aj + ai*a_new_j - ai*aj
                        # = a_new_i*aj + ai*a_new_j - 2*ai*aj
                        # True: 2*a_new_i*a_new_j - 2*ai*aj
                        # Difference: 2*a_new_i*a_new_j - a_new_i*aj - ai*a_new_j
                        # = a_new_i*(2*a_new_j - aj) - ai*a_new_j
                        # = a_new_i*(2*(aj-d) - aj) - ai*(aj-d)
                        # = a_new_i*(aj - 2d) - ai*aj + ai*d
                        if k == i + j and i != j:
                            # Correct for double counting
                            new_val -= (a_new_i - ai) * (a_new_j - aj)
                        if new_val > max_new_affected:
                            max_new_affected = new_val
                    if max_new_affected < best_new_max - 1e-14:
                        best_new_max = max_new_affected
                        best_delta = d
                if 'best_delta' in dir() and best_new_max < M - 1e-14:
                    a[i] = max(ai + best_delta, 0.0)
                    a[j] = max(aj - best_delta, 0.0)
                    conv = fast_convolve(a)
                    M = float(conv.max())
                    cur_score = 2.0 * n * M
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True
                    del best_delta
        if sweeps % 100 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
        if sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score

def nesterov_smoothed(a0, time_limit, beta_init=25.0, lr_init=0.008, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))
        conv_y = fast_convolve(y_proj)
        max_c = float(conv_y.max())
        if max_c < 1e-30:
            break
        scaled = beta * (conv_y - max_c) / max_c
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z
        # Compute gradient via FFT-based approach
        # grad[i] = 2 * sum_k weights[k] * y_proj[k-i] for valid k-i
        # This is 2 * (weights_reversed * y_proj)[i + n - 1] using convolution
        w_rev = weights[::-1].copy()
        W = rfft(w_rev, 2 * (2 * n - 1))
        Y = rfft(y_proj, 2 * (2 * n - 1))
        corr = irfft(W * Y, 2 * (2 * n - 1))
        offset = n - 1
        idx = np.arange(offset, offset + n)
        valid = (idx >= 0) & (idx < len(corr))
        grad = np.zeros(n)
        grad[valid] = 2.0 * corr[idx[valid]]

        # Projected gradient step with backtracking
        a_new = a - lr * grad
        a_new = simplex_project(a_new)
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtracking
            if no_improve > 10:
                lr *= 0.7
                no_improve = 0
                continue

        a = a_new
        if no_improve > 60:
            beta *= 1.2
            no_improve = 0
            if beta > 10000:
                break
        elif it % 20 == 0:
            beta *= 1.005
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.1f}")
    return best_a, best_score

def fine_coord_descent(a0, time_limit, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)
            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            best_delta = 0.0
            best_new_max = M
            # Very fine factors
            factors = [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
                       0.99, 0.985, 0.98, 0.97, 0.95,
                       1.0001, 1.0002, 1.0005, 1.001, 1.002, 1.003, 1.005,
                       1.007, 1.01, 1.015, 1.02, 1.03, 1.05]
            for factor in factors:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta
            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        if not improved:
            no_improve_count += 1
            if no_improve_count > 5:
                break
        else:
            no_improve_count = 0
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
    return best_a, best_score

def generate_seeds(n, rng):
    """Generate diverse initial sequences."""
    seeds = []
    # Random
    r = rng.random(n)
    seeds.append(r / r.sum())
    # Spectrally flat: inverse of random PSD
    for trial in range(3):
        real = rng.standard_normal(n)
        imag = rng.standard_normal(n)
        spec = np.abs(real + 1j * imag) ** 2
        spec = spec + 0.1 * spec.mean()
        a_spec = np.zeros(n)
        # Try to find a non-negative sequence with this spectrum
        # Use iterative projection
        a_spec = rng.random(n)
        a_spec = a_spec / a_spec.sum()
        for _ in range(200):
            F = np.fft.fft(a_spec)
            F_new = np.sqrt(np.maximum(np.abs(F)**2, 0)) * np.exp(1j * np.angle(F))
            # Try to match target spectrum magnitude
            target_mag = np.sqrt(spec)
            current_mag = np.abs(F)
            if current_mag.sum() > 1e-15:
                F_new_mag = target_mag * (current_mag / current_mag.sum())
                F_new = F_new_mag * np.exp(1j * np.angle(F))
            a_spec = np.real(np.fft.ifft(F_new))
            a_spec = np.maximum(a_spec, 0)
            ss = a_spec.sum()
            if ss > 1e-15:
                a_spec = a_spec / ss
        if a_spec.sum() > 1e-15:
            seeds.append(a_spec / a_spec.sum())
    # Decaying
    x = np.linspace(0, 1, n)
    a_dec = np.exp(-3 * x)
    a_dec = a_dec / a_dec.sum()
    seeds.append(a_dec.copy())
    # Bump in middle
    x = np.linspace(0, 1, n)
    a_bump = np.exp(-20 * (x - 0.5)**2)
    a_bump = a_bump / a_bump.sum()
    seeds.append(a_bump.copy())
    # Two bumps
    x = np.linspace(0, 1, n)
    a_2b = np.exp(-30 * (x - 0.3)**2) + np.exp(-30 * (x - 0.7)**2)
    a_2b = a_2b / a_2b.sum()
    seeds.append(a_2b.copy())
    # Geometric
    r = 0.999
    a_geo = np.array([r**i for i in range(n)])
    a_geo = a_geo / a_geo.sum()
    seeds.append(a_geo.copy())
    return seeds

def peak_targeted_perturb(a0, rng, strength=0.05):
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.2):
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(20, n // 4):
        keep = np.ones(n, dtype=bool)
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(int(keep.sum())) * 0.4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 8.0 / (iters + 3))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()
        if iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_b={M_b:.8f}")
    return best_a, best_score, iters
