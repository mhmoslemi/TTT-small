def solve_lp_best_response(a_fixed):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b, M) where b is the optimal solution."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build T matrix: T[k, i] = a_fixed[k-i] if 0 <= k-i < n
    # Using sparse representation for speed
    rows = []
    cols = []
    vals = []
    for i in range(n):

        k_start = i  # k-i >= 0 → k >= i

            rows.append(k)
            cols.append(i)
            vals.append(a_fixed[k - i])

    # Variables: x = [b_0,...,b_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0

    b_ub = np.zeros(size)

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    M_val = res.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_val)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP block coordinate descent with fast sparse HiGHS solves
2. Subgradient steps between LP iterations for additional progress
3. Multi-length exploration around parent sequence length
4. Perturbation restarts on stall detection
5. Final polishing phase with tighter convergence
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import scipy.sparse as sp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_T_sparse(a):
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        k_start = i
        k_end = min(i + n - 1, size - 1)
        ks = range(k_start, k_end + 1)
        rows.extend(ks)
        cols.extend([i] * (k_end - k_start + 1))
        vals.extend(a[k - i] for k in ks)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return sp.csr_matrix((vals, (rows, cols)), shape=(size, n))


def solve_lp_fast(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_T_sparse(a_fixed)
    c = np.zeros(n + 1)
    c[n] = 1.0
    neg_ones_sparse = sp.csr_matrix(-np.ones((size, 1)))
    A_ub = sp.hstack([T, neg_ones_sparse], format='csr')
    b_ub = np.zeros(size)
    eq_row = sp.csr_matrix((np.ones(n), (np.zeros(n, dtype=np.int32), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=eq_row, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None, None
        b_val = np.maximum(res.x[:n], 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, float(res.x[n])
    except Exception:
        return None, None


def subgradient_step(a, step_size):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 1e-6)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([n - 1])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += 2.0 * a[k - j]
    g /= len(active)
    a_new = a - step_size * g
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_polish(a0, time_limit, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    y = a.copy()
    prev_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-8)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += 2.0 * y[k - j]
        g /= len(active)

        eta = 0.5 / np.sqrt(it + 1)
        momentum = (it) / (it + 3.0)
        grad_step = g + momentum * (y - prev_a)
        a_new = y - eta * grad_step
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        prev_a = y.copy()
        y = a_new

        sc = evaluate_local(y)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()

        if (it + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score
