        S = (a * grad_a).sum()
        return a * (grad_a - S)

    res = minimize(
        fun=_func, x0=z, jac=_jac, method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-20,
                 'gtol': 1e-14, 'maxfun': maxiter * 30,
                 'disp': False}
    )
    return res.x


def lbgfs_multistage(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=30):
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-30)
    a = a / a.sum()
    z = np.log(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    print(f"    LSE start: score={best_score:.6f}, z range=[{z.min():.2f}, {z.max():.2f}]")

    for si in range(n_stages):
        frac = (si + 1) / n_stages
        T_cur = T_start * (T_end / T_start) ** frac
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        maxiter = max(30, int(remaining * 50 / (n_stages - si)))
        maxiter = min(maxiter, 300)

        z = run_lbgfs_stage(z, T_cur, n, maxiter)
        a = softmax(z)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if si % 5 == 0 or si == n_stages - 1:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={T_cur:.3f}, {elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_mirror_descent(a0, time_limit, n_active_frac=0.05):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    z = np.log(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = 0.5
    stall = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - n_active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                g[j] += a[k - j]
        g /= len(active)

        z_new = z - eta * g
        a_new = softmax(z_new)
        sc = compute_score(a_new)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            z = z_new
            stall = 0
            eta = min(eta * 1.05, 5.0)
        else:
            stall += 1
            if stall > 50:
                eta = max(eta * 0.7, 0.05)
                if stall > 500:
                    rng = np.random.default_rng(it)
                    z = z + rng.normal(0, 0.05, n)
                    a = softmax(z)
                    stall = 0
                    eta = 0.5

        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter {it} ({elapsed:.0f}s): score={best_score:.6f}, eta={eta:.3f}")

    return best_a, best_score


def peak_reduction_search(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-30:
            break

        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])

        peak_areas = set()
        top_peaks = np.argsort(b)[::-1][:5]
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_areas.update(range(js2, je2 + 1))

        candidates = [j for j in range(n) if j not in peak_areas and j != src]
        if not candidates:
            candidates = [j for j in range(n) if j != src]
        dst = int(rng.choice(candidates))

        improved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            if rng.random() < 0.3:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.08)
                    a_trial = a.copy()
                    amt = a[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                            continue
            stall += 1
            if stall > 200:
                break

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSearch step {step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.4 * np.cos(3 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(4 * np.pi * x)
        a = np.power(np.maximum(a, 0), 0.7)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * x * (1 - x)) * (1 + 0.15 * np.cos(15 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def optimize_sequence(a_init, time_budget, label=""):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    def try_update(arr, src):
        nonlocal best_a, best_score
        arr = np.maximum(arr, 0)
        ss = arr.sum()
        if ss < 1e-15:
            return
        arr = arr / ss
        sc = compute_score(arr)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = arr.copy()
            elapsed = time.monotonic() - t0
            print(f"      NEW BEST [{src}]: {sc:.6f} (n={n}, t={elapsed:.0f}s)")

    # Phase A: LogSumExp L-BFGS (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.55, 600.0)
    if t_a > 30:
        print(f"    [{label}] LSE phase: budget={t_a:.0f}s")
        a_lse, _ = lbgfs_multistage(best_a, t_a, T_start=50.0, T_end=0.01, n_stages=30)
        try_update(a_lse, f"LSE_{label}")

    # Phase B: Subgradient mirror descent
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.25, 200.0)
    if t_b > 10:
        print(f"    [{label}] SubGrad phase: budget={t_b:.0f}s")
        a_sg, _ = subgradient_mirror_descent(best_a, t_b, n_active_frac=0.03)
        try_update(a_sg, f"SG_{label}")

    # Phase C: Peak reduction local search
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.20, 100.0)
    if t_c > 5:
        print(f"    [{label}] PeakSearch phase: budget={t_c:.0f}s")
        a_ps, _ = peak_reduction_search(best_a, t_c, n_steps=30000)
        try_update(a_ps, f"PS_{label}")

    # Phase D: More LogSumExp with lower T
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.20, 100.0)
    if t_d > 10:
        print(f"    [{label}] LSE fine phase: budget={t_d:.0f}s")
        a_lse2, _ = lbgfs_multistage(best_a, t_d, T_start=1.0, T_end=0.005, n_stages=15)
        try_update(a_lse2, f"LSE2_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    global _deadline
    rng = np.random.default_rng(seed)
    _deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)
    total_budget = _deadline - time.monotonic()

    print(f"Total budget: {total_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = compute_score(parent)
            best_seq = parent.copy()
            best_score = p_score
            print(f"Parent: n={n_base}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def global_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Main optimization at base length ===
    print(f"\n=== Phase 1: Optimize n={n_base} (score={best_score:.6f}) ===")
    rem = _deadline - time.monotonic()
    t1 = min(rem * 0.50, 800.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_sequence(best_seq, t1, label=f"base_n{n_base}")
        global_update(a_out, f"Phase1_n{n_base}")

    # === Phase 2: Try longer length ===
    print(f"\n=== Phase 2: Longer length (score={best_score:.6f}) ===")
    n_long = int(n_base * 1.5)
    if n_long > n_base and n_long <= 3000:
        rem = _deadline - time.monotonic()
        if rem > 120:
            t2 = min(rem * 0.25, 350.0)
            if t2 > 60:
                print(f"  Trying n={n_long}, budget={t2:.0f}s")
                init_long = interpolate_sequence(best_seq, n_long)
                sc_init = compute_score(init_long)
                print(f"  Interpolated init: score={sc_init:.6f}")
                global_update(init_long, f"interp_n{n_long}")
                a_out2, _ = optimize_sequence(best_seq if compute_score(best_seq) <= sc_init else init_long, t2, label=f"long_n{n_long}")
                global_update(a_out2, f"Phase2_n{n_long}")

    # === Phase 3: Try another variation at base length ===
    print(f"\n=== Phase 3: Variation at n={n_base} (score={best_score:.6f}) ===")
    rem = _deadline - time.monotonic()
    if rem > 100:
        t3 = min(rem * 0.25, 300.0)
        if t3 > 30:
            # Perturb current best
            noise = rng.normal(0, 0.02, n_base)
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            print(f"  Perturbed init: score={compute_score(a_pert):.6f}")
            a_out3, _ = optimize_sequence(a_pert, t3, label=f"perturb_n{n_base}")
            global_update(a_out3, f"Phase3_perturb")

    # === Phase 4: Try shorter length ===
    print(f"\n=== Phase 4: Shorter length (score={best_score:.6f}) ===")
    n_short = int(n_base * 0.75)
    if n_short < n_base and n_short >= 500:
        rem = _deadline - time.monotonic()
        if rem > 80:
            t4 = min(rem * 0.15, 200.0)
            if t4 > 30:
                print(f"  Trying n={n_short}, budget={t4:.0f}s")
                init_short = interpolate_sequence(best_seq, n_short)
                global_update(init_short, f"interp_short_n{n_short}")
                a_out4, _ = optimize_sequence(init_short, t4, label=f"short_n{n_short}")
                global_update(a_out4, f"Phase4_n{n_short}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.6f}, n={len(best_seq)}) ===")
    rem = _deadline - time.monotonic()
    if rem > 20:
        # Quick subgradient pass
        t5a = min(rem * 0.4, 60.0)
        if t5a > 5:
            a_fin, _ = subgradient_mirror_descent(best_seq, t5a, n_active_frac=0.02)
            global_update(a_fin, "final_sg")

        # Quick peak search
        rem = _deadline - time.monotonic()
        t5b = min(rem * 0.3, 40.0)
        if t5b > 5:
            a_fin2, _ = peak_reduction_search(best_seq, t5b, n_steps=20000)
            global_update(a_fin2, "final_ps")

        # Ultra-fine LSE
        rem = _deadline - time.monotonic()
        t5c = min(rem * 0.4, 50.0)
        if t5c > 5:
            a_fin3, _ = lbgfs_multistage(best_seq, t5c, T_start=0.5, T_end=0.003, n_stages=12)
            global_update(a_fin3, "final_lse")

    # Final verification
    final_score = evaluate_safe(best_seq)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
