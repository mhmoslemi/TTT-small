def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax_from_z(z):

    if s < 1e-300:

                a = y.copy()
                stall_count = 0

            if (it // 200) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}, lp_M={M_lp:.8f}")

    return best_a, best_score


def mirror_descent_refined(a0, time_limit, n_active_frac=0.05):
    """Refined mirror descent with adaptive step and subgradient averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.5
    grad_avg = np.zeros(n)

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        # Compute subgradient using active set
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Normalized Paternoster step size
        alpha = lr / (1.0 + lr * np.dot(g, g) ** 0.5)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        score_new = fast_convolve(a_new).max()

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
            lr = min(lr * 1.05, 5.0)
        else:
            lr = max(lr * 0.9, 0.05)
            # Shake if stuck
            if step % 500 == 499 and lr < 0.1:
                rng = np.random.default_rng(step)
                z += rng.normal(0, 0.01, n)
                a = softmax_from_z(z)
                best_score = fast_convolve(a).max()
                best_a = a.copy()

    return best_a, best_score


def logsumexp_lbfgs(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=10):
    """LogSumExp smoothed optimization with L-BFGS."""
    from scipy.optimize import minimize

    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        max_iter = max(10, int(30 * remaining / time_limit * 10))

        def func(z_in):
            a_local = softmax_from_z(z_in)
            b_local = fast_convolve(a_local)
            Tb = stage_T * b_local
            max_Tb = min(Tb.max(), 700)
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_in):
            a_local = softmax_from_z(z_in)
            b_local = fast_convolve(a_local)
            Tb = stage_T * b_local
            max_Tb = min(Tb.max(), 700)
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_local)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_local[k - idx]
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': max_iter * 200,
                         'disp': False}
            )
            if res.x is not None:
                z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def spectral_init(n, rng):
    """Generate initialization using random spectral phase."""
    # Create a random frequency response and take inverse
    freq = np.random.default_rng(rng.integers(100000)).random(2 * n)
    freq[0] = 0  # DC
    # Force non-negative by taking absolute value of IFFT
    a = np.abs(np.fft.ifft(freq))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = (1 - 4 * (x - 0.5) ** 2) ** 2
        a = np.maximum(a, 0)
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.2 * np.cos(3 * x)
    elif choice == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_phase(a0, time_budget, solver, label=""):
    """Run a full optimization phase at fixed length."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    t0 = time.monotonic()

    def try_update(a_new, src):
        nonlocal best_a, best_score
        if a_new is None:
            return
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 0.01:
            return
        a_new = a_new / ss
        sc = fast_convolve(a_new).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            elapsed = time.monotonic() - t0
            print(f"    [{src}] {elapsed:.0f}s: score={sc:.8f}")

    # Phase 1: LP Nesterov (main)
    rem = time_budget - (time.monotonic() - t0)
    t1 = rem * 0.45
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_a, t1, solver=solver)
        try_update(a_out, f"LP_{label}")

    # Phase 2: Peak deflation
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.2, 60.0)
    if t2 > 5:
        a_out = peak_deflation(best_a, n_steps=20000, time_limit=t2)
        try_update(a_out, f"PD_{label}")

    # Phase 3: Mirror descent
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.2, 60.0)
    if t3 > 5:
        a_out, _ = mirror_descent_refined(best_a, t3)
        try_update(a_out, f"MD_{label}")

    # Phase 4: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.15, 80.0)
    if t4 > 5:
        a_out = logsumexp_lbfgs(best_a, t4, T_start=20.0, T_end=0.1, n_stages=8)
        try_update(a_out, f"LSE_{label}")

    # Phase 5: More LP if time remains
    rem = time_budget - (time.monotonic() - t0)
    t5 = min(rem * 0.5, 60.0)
    if t5 > 10:
        a_out, _ = nesterov_lp_optimize(best_a, t5, solver=solver)
        try_update(a_out, f"LP2_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-strategy optimization for minimum objective."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_deadline = t_start + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Get parent or initialize
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass

    n_base = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_base = 960
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")

    def try_global_update(a_new, label):
        nonlocal best_seq, best_score
        if a_new is None:
            return
        a_new = np.maximum(a_new, 0).astype(np.float64)
        s = a_new.sum()
        if s < 0.01:
            return
        a_new = a_new / s
        sc = evaluate_safe(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** GLOBAL BEST [{label}]: {sc:.6f} (n={len(a_new)}, t={elapsed:.0f}s)")

    # === Phase 1: Optimize at base length (main effort) ===
    print(f"\n=== Phase 1: Optimize n={n_base} (score={best_score:.6f}) ===")
    rem = t_deadline - time.monotonic()
    t1 = min(rem * 0.5, 900.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_phase(best_seq, t1, solver, label=f"n{n_base}")
        try_global_update(a_out, f"opt_n{n_base}")

    # === Phase 2: Try longer lengths ===
    print(f"\n=== Phase 2: Longer lengths (best={best_score:.6f}) ===")
    longer_lengths = []
    for mult in [1.3, 1.5, 2.0]:
        ln = int(n_base * mult)
        if ln > n_base and ln <= 5000:
            longer_lengths.append(ln)
    longer_lengths = sorted(set(longer_lengths))[:3]

    for ln in longer_lengths:
        rem = t_deadline - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.2, 300.0)
        if t_alloc < 60:
            break
        print(f"  n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_sequence(best_seq, ln)
        a_out, _ = optimize_phase(init_long, t_alloc, solver, label=f"n{ln}")
        try_global_update(a_out, f"long_n{ln}")

    # === Phase 3: Try shorter lengths ===
    print(f"\n=== Phase 3: Shorter lengths (best={best_score:.6f}) ===")
    for mult in [0.75, 0.5]:
        sn = int(n_base * mult)
        if sn < n_base and sn >= 100:
            rem = t_deadline - time.monotonic()
            if rem < 90:
                break
            t_alloc = min(rem * 0.15, 150.0)
            if t_alloc < 30:
                break
            print(f"  n={sn}, budget={t_alloc:.0f}s")
            init_short = interpolate_sequence(best_seq, sn)
            a_out, _ = optimize_phase(init_short, t_alloc, solver, label=f"n{sn}")
            try_global_update(a_out, f"short_n{sn}")

    # === Phase 4: Fresh starts with different inits ===
    print(f"\n=== Phase 4: Alternative inits (best={best_score:.6f}) ===")
    for attempt in range(3):
        rem = t_deadline - time.monotonic()
        if rem < 90:
            break
        t_alloc = min(rem * 0.15, 200.0)
        if t_alloc < 30:
            break

        n_try = int(n_base * rng.uniform(0.8, 1.5))
        n_try = max(200, min(n_try, 5000))
        a_init = structured_init(n_try, rng)
        # Mix with current best
        if best_seq is not None:
            a_mix = interpolate_sequence(best_seq, n_try)
            a_init = 0.5 * a_init + 0.5 * a_mix
            a_init = np.maximum(a_init, 0)
            s = a_init.sum()
            if s > 0.01:
                a_init = a_init / s

        print(f"  Attempt {attempt+1}: n={n_try}, budget={t_alloc:.0f}s")
        a_out, _ = optimize_phase(a_init, t_alloc, solver, label=f"alt{attempt}")
        try_global_update(a_out, f"alt_{attempt}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = t_deadline - time.monotonic()
    if rem > 30:
        # LP refinement
        t_lp = min(rem * 0.4, 120.0)
        if t_lp > 10:
            a_fin, _ = nesterov_lp_optimize(best_seq, t_lp, solver=solver)
            try_global_update(a_fin, "final_lp")

        # Peak deflation
        rem = t_deadline - time.monotonic()
        if rem > 20:
            t_pd = min(rem * 0.3, 60.0)
            if t_pd > 5:
                a_fin = peak_deflation(best_seq, n_steps=20000, time_limit=t_pd)
                try_global_update(a_fin, "final_pd")

        # LSE fine-tune
        rem = t_deadline - time.monotonic()
        if rem > 15:
            t_lse = min(rem * 0.3, 60.0)
            if t_lse > 5:
                a_fin = logsumexp_lbfgs(best_seq, t_lse, T_start=5.0, T_end=0.05, n_stages=8)
                try_global_update(a_fin, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL VERIFIED SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(100)
        print("\nWARNING: No valid sequence found, returning uniform")

    elapsed_total = time.monotonic() - t_start
    print(f"Total time: {elapsed_total:.1f}s")
    return [float(x) for x in best_seq]
