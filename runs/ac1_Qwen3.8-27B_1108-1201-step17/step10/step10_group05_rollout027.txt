    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.PDLP, cp.SCS, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def compute_block_gradient(a_full, block, weights, s_full, beta, smooth_max):
    """Compute gradient of smoothed objective w.r.t. block coordinates."""
    n = len(a_full)
    B = len(block)
    grad = np.zeros(B)
    for vid, i in enumerate(block):
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            ds = 2.0 * a_full[ks - i]
            grad[vid] = np.dot(weights[k_lo:k_hi + 1], ds)
    return grad

def block_coordinate_descent(a0, time_limit, initial_block_size=8, beta_init=80.0, verbose=True, label=""):
    """
    Block coordinate descent with analytic gradients via L-BFGS-B.
    This is the main optimization engine.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    no_improve = 0
    block_size = initial_block_size
    beta = beta_init
    window_pos = 0
    use_window = False

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Choose block: alternate between random and sliding window
        if block_size >= n:
            block = list(range(n))
        elif use_window:
            start = window_pos % max(1, n - block_size)
            block = list(range(start, start + block_size))
            window_pos += max(1, block_size // 3)
        else:
            block = sorted(rng.choice(n, size=block_size, replace=False).tolist())

        v0 = np.array([a[i] for i in block], dtype=np.float64)
        s_rest = a.sum() - v0.sum()

        def bobj(v):
            v = np.maximum(v, 0.0)
            a_full = a.copy()
            for idx, i in enumerate(block):
                a_full[i] = v[idx]
            sf = s_rest + v.sum()
            if sf < 1e-10:
                return 1e10
            c = fast_convolve(a_full)
            mc = float(c.max())
            scaled = beta * (c - mc)
            scaled = np.clip(scaled, -100.0, 0.0)
            sm = mc + float(np.log(np.exp(scaled).sum())) / beta
            return 2.0 * n * sm / (sf * sf)

        def bgrad(v):
            v = np.maximum(v, 0.0)
            a_full = a.copy()
            for idx, i in enumerate(block):
                a_full[i] = v[idx]
            sf = s_rest + v.sum()
            if sf < 1e-10:
                return np.zeros(len(v))
            c = fast_convolve(a_full)
            mc = float(c.max())
            scaled = beta * (c - mc)
            scaled = np.clip(scaled, -100.0, 0.0)
            w = np.exp(scaled)
            Z = w.sum()
            if Z < 1e-30:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
                Z = 1.0
            w = w / Z
            sm = mc + float(np.log(Z)) / beta
            grad_block = compute_block_gradient(a_full, block, w, sf, beta, sm)
            return 2.0 * n * (grad_block / (sf * sf) - 2.0 * sm / (sf ** 3))

        bounds = [(0.0, 1000.0)] * len(block)
        max_iter = 40 if rem > 60 else 20

        try:
            result = minimize(bobj, v0, method='L-BFGS-B',
                             jac=bgrad, bounds=bounds,
                             options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
        except Exception:
            result = None

        improved = False
        if result is not None:
            v_opt = np.maximum(result.x, 0.0)
            a_new = a.copy()
            for idx, i in enumerate(block):
                a_new[i] = v_opt[idx]
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc = 2.0 * n * fast_convolve(a_new).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    no_improve = 0
                else:
                    a = a_new.copy()
                    no_improve += 1

        if not improved:
            no_improve += 1

        # Adaptive adjustments
        if no_improve > 150:
            no_improve = 0
            r = rng.random()
            if r < 0.3:
                block_size = max(3, block_size - 2)
            elif r < 0.6:
                block_size = min(25, block_size + 3)
            else:
                beta = max(20.0, beta * 0.7)
            use_window = not use_window

        if verbose and it % 200 == 0:

            print(f"    BCD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, bs={block_size}, beta={beta:.0f}")

def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with smart 1D candidate evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            top_k = min(5, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.5 * ai)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 100 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        alpha = min(1.0, 8.0 / (iters + 3))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        if iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.5 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def palindromic_smooth(a, alpha=0.5):
    """Blend a sequence with its reverse to encourage palindromic structure."""
    n = len(a)
    rev = a[::-1].copy()
    blended = alpha * a + (1 - alpha) * rev
    s = blended.sum()
    if s < 1e-15:
        return a.copy()
    return blended / s

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization:
    1. Quick CD to establish baseline
    2. Block coordinate descent (main workhorse)
    3. LP fictitious play (global refinement)
    4. More block CD from improved points
    5. CD + perturbation loops (escape local optima)
    6. Multi-length exploration
    7. Final fine polishing
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(99999)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # === Phase 1: Quick CD ===
    print(f"\n=== Phase 1: Quick CD (best={best_score:.6f}) ===")
    t1 = min(40.0, (deadline - time.monotonic()) * 0.03)
    print(f"  Budget: {t1:.0f}s, n={n_best}")
    a1, s1 = smart_coord_descent(best_seq, t1, verbose=True, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1")

    # === Phase 2: Block Coordinate Descent (MAIN) ===
    print(f"\n=== Phase 2: Block CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 600
    t2 = min(max(rem * 0.35, 120), 500.0)
    print(f"  Budget: {t2:.0f}s, n={n_best}")
    a2, s2 = block_coordinate_descent(best_seq, t2, initial_block_size=10,
                                      beta_init=80.0, verbose=True, label="p2")
    best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_bcd")

    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 60:
        t3 = min(max(rem * 0.25, 60), 300.0)
        print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
        a3, s3, it3 = lp_fictitious_play(best_seq, t3, solver=solver, label="p3")
        best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_lp")

    # === Phase 4: Block CD again from improved point ===
    print(f"\n=== Phase 4: Block CD round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t4 = min(max(rem * 0.3, 60), 300.0)
        print(f"  Budget: {t4:.0f}s, n={len(best_seq)}")
        a4, s4 = block_coordinate_descent(best_seq, t4, initial_block_size=6,
                                          beta_init=120.0, verbose=True, label="p4")
        best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_bcd")

    # === Phase 5: CD + Perturbation Loop ===
    print(f"\n=== Phase 5: CD + Perturb (best={best_score:.6f}) ===")
    for rnd in range(3):
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        t5a = min(rem * 0.25, 80.0)
        t5b = min(rem * 0.25, 60.0)

        # Perturb
        if rnd == 0:
            perturbed = peak_targeted_perturb(best_seq, rng, strength=0.04)
        elif rnd == 1:
            perturbed = palindromic_smooth(best_seq, alpha=0.7)
            perturbed = peak_targeted_perturb(perturbed, rng, strength=0.03)
        else:
            n_pert = len(best_seq)
            keep = rng.random(n_pert) > 0.15
            a_p = best_seq.copy()
            a_p[~keep] = best_seq[~keep] * 0.3
            a_p[keep] = best_seq[keep] * (1.0 + rng.standard_normal(keep.sum()) * 0.15)
            a_p = np.maximum(a_p, 0)
            s_p = a_p.sum()
            perturbed = a_p / s_p if s_p > 1e-10 else best_seq.copy()

        # Optimize perturbed
        a5a, s5a = block_coordinate_descent(perturbed, t5a, initial_block_size=8,
                                            beta_init=60.0, verbose=False, label=f"p5r{rnd}")
        best_seq, best_score = check_and_update(a5a, best_seq, best_score, f"p5r{rnd}_bcd")

        # Also try CD on the current best
        a5b, s5b = smart_coord_descent(best_seq, t5b, verbose=False, label=f"p5cd{rnd}")
        best_seq, best_score = check_and_update(a5b, best_seq, best_score, f"p5cd{rnd}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    alt_lengths = [n_best]
    for alt in [896, 1152, 1280, 1536, 1792, 2048]:
        if alt != n_best:
            alt_lengths.append(alt)
    alt_lengths = alt_lengths[:4]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic() - 100
        if rem < 40:
            break
        t6 = min(rem * 0.3, 100.0)
        if n_alt == n_best:
            init_alt = best_seq.copy()
        else:
            init_alt = interpolate_seq(best_seq, n_alt)
        a6, s6 = block_coordinate_descent(init_alt, t6, initial_block_size=8,
                                          beta_init=80.0, verbose=False, label=f"p6_n{n_alt}")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, f"p6_n{n_alt}")
        # Quick CD on it
        a6b, s6b = smart_coord_descent(a6, min(t6 * 0.5, 30.0), verbose=False, label=f"p6b_n{n_alt}")
        best_seq, best_score = check_and_update(a6b, best_seq, best_score, f"p6b_n{n_alt}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 30:
        t7a = min(rem * 0.4, 120.0)
        a7a, s7a = fine_coord_descent(best_seq, t7a, label="p7a")
        best_seq, best_score = check_and_update(a7a, best_seq, best_score, "p7a_fine")

    rem = deadline - time.monotonic() - 20
    if rem > 20:
        t7b = min(rem * 0.5, 80.0)
        a7b, s7b = block_coordinate_descent(best_seq, t7b, initial_block_size=5,
                                            beta_init=150.0, verbose=False, label="p7b")
        best_seq, best_score = check_and_update(a7b, best_seq, best_score, "p7b")

    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t7c = min(rem * 0.8, 40.0)
        a7c, s7c = fine_coord_descent(best_seq, t7c, label="p7c")
        best_seq, best_score = check_and_update(a7c, best_seq, best_score, "p7c_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
