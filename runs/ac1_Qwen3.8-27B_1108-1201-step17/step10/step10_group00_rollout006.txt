        k_start = j

        for k in range(k_start, k_end + 1):
            all_rows.append(k)
            all_cols.append(j)
            all_vals.append(a[k - j])
    T = cp.SparseMatrix(np.array(all_rows), np.array(all_cols), np.array(all_vals), shape=(2*n-1, n))
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_sparse_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[b_val < 1e-15] = 0
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def solve_lp_best_response_dense(a_fixed, solver):
    """LP best response using dense matrix (faster for moderate n)."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Build dense matrix
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        if n <= 1500:
            b_new, M_b = solve_lp_best_response_dense(y, solver)
        else:
            b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        avg_lp_time = np.mean(lp_times[-30:]) if len(lp_times) >= 10 else 0.5

        # Nesterov dual averaging step with adaptive alpha
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                # Structured perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                a = a + noise
                y = np.maximum(y, 0)
                a = np.maximum(a, 0)
                y = y / max(y.sum(), 1e-15)
                a = a / max(a.sum(), 1e-15)
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg_lp_time*1000:.0f}ms")

        if len(lp_times) > 50 and avg_lp_time > 2.5:
            print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping after {it+1} iters")
            break

    return best_a, best_score


def lse_objective_and_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. log-domain z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    lse = max_b + T * (shifted.max() + np.log(exp_b.sum()))
    
    # Objective: 2n * lse (since sum(a)=1)
    obj = 2.0 * n * lse

    # Gradient
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_val):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = lse_objective_and_grad(z_val, n, stage_T)
            return o

        def jac(z_val):
            if time.monotonic() > deadline:
                return np.zeros_like(z_val)
            _, g = lse_objective_and_grad(z_val, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    # Convert back
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    return a


def lse_momentum_refine(a0, time_limit, T_stages, lr=0.01, momentum=0.9):
    """Momentum-based gradient descent in log-domain with LogSumExp."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    best_score = 2.0 * n * fast_convolve(a0).max()
    best_a = a0.copy()
    step_count = 0

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        # Adaptive steps for this temperature
        n_steps = min(500, max(50, int((time_limit - (time.monotonic() - t0)) / 0.05)))
        current_lr = lr

        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = lse_objective_and_grad(z, n, T)
            v = momentum * v - current_lr * grad
            z = z + v
            # Remove mean (gauge freedom)
            z = z - z.mean()

            # Convert to a and check score
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            step_count += 1

            # Adaptive learning rate
            if step % 100 == 0:
                current_lr *= 0.9

    return best_a, best_score


def multiplicative_subgradient(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta_base = 0.03
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def diverse_init(n, rng, method=0):
    """Create diverse initial sequences."""
    x = np.linspace(0, 1, n)
    if method == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif method == 1:
        a = np.sin(np.pi * x) ** 2
    elif method == 2:
        a = np.exp(-3 * x * (1 - x))
    elif method == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif method == 4:
        a = (1 - x) ** 2 + 0.05
    elif method == 5:
        a = np.exp(-5 * (x - 0.4) ** 2) + 0.3 * np.exp(-5 * (x - 0.7) ** 2)
    elif method == 6:
        # Random smooth
        a = np.ones(n)
        for freq in range(1, 4):
            a += 0.3 / freq * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    elif method == 7:
        # Broad Gaussian
        a = np.exp(-1.0 * (x - 0.5) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def coordinate_descent_refine(a0, time_limit, n_iter=5000):
    """Coordinate-wise descent: try flipping small groups of coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        # Pick a random window and adjust
        w_size = rng.integers(5, min(50, n // 4))
        start = rng.integers(0, n - w_size)
        # Add a small bump or dip
        x_win = np.linspace(0, 1, w_size)
        bump = 0.01 * np.cos(np.pi * x_win)
        a_trial = a.copy()
        a_trial[start:start + w_size] += bump
        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial / s_trial
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP alternating, LogSumExp optimization (L-BFGS + momentum),
    multiplicative subgradient, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = diverse_init(n_init, rng, 0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP on parent ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 540.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        local_update(a_lp, "nlp1")

    # === Phase 2: LogSumExp L-BFGS ===
    print(f"\n=== Phase 2: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 360.0)
    if t_alloc > 30:
        T_stages = [20.0, 10.0, 5.0, 3.0, 1.5, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005, 0.002]
        print(f"  {len(T_stages)} stages, budget={t_alloc:.0f}s")
        a_lb = lse_lbgfs_refine(best_seq, t_alloc, T_stages, maxiter_per_stage=8000)
        local_update(a_lb, "lse_lbgfs")

    # === Phase 3: Momentum gradient descent ===
    print(f"\n=== Phase 3: Momentum GD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 15:
        T_stages3 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_mom, _ = lse_momentum_refine(best_seq, t_alloc, T_stages3, lr=0.02, momentum=0.9)
        local_update(a_mom, "momentum")

    # === Phase 4: Second Nesterov LP ===
    print(f"\n=== Phase 4: NLP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        local_update(a_lp2, "nlp2")

    # === Phase 5: Multiplicative subgradient ===
    print(f"\n=== Phase 5: MultSubGrad (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_ms, _ = multiplicative_subgradient(best_seq, t_alloc, n_steps=200000)
        local_update(a_ms, "mult_sub")

    # === Phase 6: Try n=1500 ===
    print(f"\n=== Phase 6: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        t_n1500 = min(rem * 0.20, 300.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = nesterov_lp_refine(init_1500, t_n1500 * 0.6, solver, n_max_iters=200000)
        local_update(a_1500, "lp_n1500")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1500 and rem > 40:
            T_s5 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_1500b = lse_lbgfs_refine(best_seq, min(rem, 120.0), T_s5, maxiter_per_stage=3000)
            local_update(a_1500b, "lse_n1500")

    # === Phase 7: Try n=1800 ===
    print(f"\n=== Phase 7: Try n=1800 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t_n1800 = min(rem * 0.18, 250.0)
        print(f"  Budget: {t_n1800:.0f}s")
        init_1800 = interpolate_sequence(best_seq, 1800)
        a_1800, _ = nesterov_lp_refine(init_1800, t_n1800 * 0.7, solver, n_max_iters=150000)
        local_update(a_1800, "lp_n1800")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1800 and rem > 30:
            T_s7 = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_1800b = lse_lbgfs_refine(best_seq, min(rem, 80.0), T_s7, maxiter_per_stage=2000)
            local_update(a_1800b, "lse_n1800")

    # === Phase 8: Perturbation escape ===
    print(f"\n=== Phase 8: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.015 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.25, 50.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=100000)
            local_update(a_opt, f"pert{trial}")

    # === Phase 9: Coordinate descent polish ===
    print(f"\n=== Phase 9: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_cd, _ = coordinate_descent_refine(best_seq, min(rem * 0.4, 40.0), n_iter=3000)
        local_update(a_cd, "coord_descent")

    # === Phase 10: Final polish ===
    print(f"\n=== Phase 10: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 40.0)
        if t_fin > 8:
            T_fin = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            a_fin = lse_lbgfs_refine(best_seq, t_fin, T_fin, maxiter_per_stage=3000)
            local_update(a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.5, 20.0), n_steps=50000)
        local_update(a_fin_ms, "final_mult")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin_lp, _ = nesterov_lp_refine(best_seq, min(rem * 0.6, 15.0), solver, n_max_iters=50000)
        local_update(a_fin_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
