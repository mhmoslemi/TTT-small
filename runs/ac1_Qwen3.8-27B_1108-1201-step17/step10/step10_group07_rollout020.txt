    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.PDLP)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.PDLP
    return cp.PDLP


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.PDLP)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Exact coordinate descent: for each coordinate, find the value that
    minimizes max(conv)/(sum)^2 using ternary search.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
        S = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    # Precompute unaffected region max (will update incrementally)
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_best_old = M

        # Randomize order slightly to break symmetry
        order = np.arange(n)
        if sweeps > 1:
            rng_sweep = np.random.default_rng(sweeps)
            rng_sweep.shuffle(order)

        for i in order:
            v = a[i]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Coefficients for the linear part
            a_shifted = a[k_start - i: k_end - i + 1]  # = a[0:min(n, k_end-i+1)]
            conv_affected = conv[k_start: k_end + 1]

            # Objective function: given x (new value of a[i]), compute max_conv / S_new^2
            def obj_val(x):
                S_new = S - v + x
                if S_new < 1e-15:
                    return 1e30
                # Linear update
                new_aff = conv_affected + 2.0 * (x - v) * a_shifted
                # Fix quadratic at k=2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] - v * v + x * x
                aff_max = float(new_aff.max())
                new_M = max(unaff_max, aff_max)
                return new_M / (S_new * S_new)

            # Current objective
            obj_cur = M / (S * S)

            # Determine search range
            x_lo = 0.0
            x_hi = max(2.0 * v + 0.5 / n, 0.1 / n)
            # Make sure x_hi gives a reasonable range
            obj_hi = obj_val(x_hi)
            if obj_hi < obj_cur:
                x_hi = min(x_hi * 2.0, S + 0.5 / n)
                obj_hi = obj_val(x_hi)
            obj_lo = obj_val(0.0)

            # If x=0 is best, check if it's worth it
            best_x = v
            best_obj = obj_cur
            if obj_lo < best_obj - 1e-12:
                best_x = 0.0
                best_obj = obj_lo
            if obj_hi < best_obj - 1e-12:
                best_x = x_hi
                best_obj = obj_hi

            # Ternary search in [x_lo, x_hi]
            lo = x_lo
            hi = x_hi
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                o1 = obj_val(m1)
                o2 = obj_val(m2)
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1

            x_mid = (lo + hi) / 2.0
            o_mid = obj_val(x_mid)
            if o_mid < best_obj - 1e-12:
                best_x = x_mid
                best_obj = o_mid

            # Apply update if improvement
            if best_obj < obj_cur - 1e-12:
                delta = best_x - v
                if abs(delta) < 1e-18:
                    continue

                # Get a_shifted before updating a[i]
                a_shifted_copy = a_shifted.copy() if (2 * i >= k_start and 2 * i <= k_end) else a_shifted

                a[i] = best_x
                conv[k_start:k_end + 1] += 2.0 * delta * a_shifted_copy
                # Fix quadratic term at k = 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    conv[2 * i] += delta * delta

                S = S - v + best_x
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (S * S)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 15:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"  ECD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    # Renormalize
    s_final = best_a.sum()
    if s_final > 1e-15:
        best_a = best_a / s_final
        conv_final = fast_convolve(best_a)
        best_score = 2.0 * n * float(conv_final.max())

    return best_a, best_score


def nesterov_smooth_descent(a0, time_limit, verbose=True, beta=200.0):
    """
    Nesterov-accelerated gradient descent on the smoothed objective:
    sum_k exp(beta * conv[k]) / (sum(a))^2
    This approximates max(conv) for large beta and is smooth.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    momentum = 0.9
    v_mom = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.8
            momentum = min(momentum + 0.02, 0.98)
            no_improve = 0
            if lr < 1e-12:
                break

        # Compute gradient of sum_k exp(beta * conv[k]) w.r.t. a
        # grad[i] = sum_k beta * exp(beta*conv[k]) * 2*a[k-i]
        # To avoid overflow, subtract max
        conv_shifted = conv - conv.max()
        weights = np.exp(beta * conv_shifted)
        # Normalize weights
        w_sum = weights.sum()
        if w_sum > 0:
            weights = weights / w_sum

        # grad[i] = 2 * sum_k weights[k] * a[k-i] = 2 * conv_weights[a, weights]
        # This is a convolution-like operation
        grad = 2.0 * convolve_ab(a, weights)
        grad = grad[:n]  # We only need the first n components (gradient w.r.t. a[0..n-1])

        # Actually, the gradient w.r.t. a[i] is:
        # d/d(a[i]) sum_k exp(beta*conv[k]) = sum_k beta*exp(beta*conv[k]) * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = 2*a[k-i] (if 0<=k-i<n) + [a[i] if k=2i... no]
        # Actually d(conv[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] for k != 2i
        # and = 2*a[i]... no, = a[i] + a[i] = 2*a[i] when k-i=i i.e. k=2i
        # Wait: conv[k] = sum_j a[j]*a[k-j]. d/d(a[i]) = a[k-i] + (i*delta term)
        # = a[k-i] (from j=i term) + a[k-i] (from k-j=i, i.e., j=k-i term)
        # = 2*a[k-i] when k-i != i, and 2*a[i] when k=2i (same thing since k-i=i)
        # So d(conv[k])/d(a[i]) = 2*a[k-i] for all valid k (where 0<=k-i<n)
        # So grad[i] = sum_k beta*exp(beta*conv[k]) * 2*a[k-i]
        # = 2*beta * sum_k weights_unnorm[k] * a[mod(k-i, n)]... 
        # Actually this is: 2*beta * (a * convolve with weights)[i]... 
        # Let me just compute it directly as a cross-correlation
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = beta * np.exp(beta * conv[k] - beta * M)  # shifted to avoid overflow
            if w < 1e-30:
                continue
            # d(conv[k])/d(a[i]) = 2*a[k-i] for i in [max(0,k-n+1), min(n-1,k)]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end >= i_start:
                grad[i_start:i_end + 1] += 2.0 * w * a[k - i_end: k - i_start + 1][::-1]

        # Nesterov update
        v_mom = momentum * v_mom - lr * grad
        a_new = a + v_mom
        a_new = simplex_project(a_new)
        a = a_new

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Nesterov it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    # Renormalize
    s_final = best_a.sum()
    if s_final > 1e-15:
        best_a = best_a / s_final
        best_score = 2.0 * n * float(fast_convolve(best_a).max())

    return best_a, best_score


def lp_alternating_enhanced(a0, time_limit, solver, verbose=True):
    """
    Enhanced LP alternating: tries multiple damping values and picks best.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP: best response b* to current a
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Try multiple dampings and pick the best
        alphas = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2]
        if it > 30:
            alphas = [0.5, 0.4, 0.3, 0.2, 0.15]

        best_candidate = None
        best_candidate_score = float('inf')

        for alpha in alphas:
            candidate = (1 - alpha) * a + alpha * b_new
            candidate = np.maximum(candidate, 0)
            s_c = candidate.sum()
            if s_c < 1e-15:
                continue
            candidate = candidate / s_c
            sc = 2.0 * n * float(fast_convolve(candidate).max())
            if sc < best_candidate_score:
                best_candidate_score = sc
                best_candidate = candidate.copy()

        if best_candidate is not None:
            if best_candidate_score < best_score:
                best_score = best_candidate_score
                best_a = best_candidate.copy()
            a = best_candidate
        else:
            a = b_new.copy()

        # Evaluate a
        score_a = 2.0 * n * float(fast_convolve(a).max())
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation check
        if score_a > best_score * (1 + 1e-6):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 20:
            noise_scale = 0.005 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert / s_p
                score_pert = 2.0 * n * float(fast_convolve(a_pert).max())
                if score_pert < best_score * 1.01:
                    a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-enhanced it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_init(n, rng):
    """Initialize with flat power spectrum (approximately flat auto-correlation)."""
    # Create random phase in frequency domain
    freqs = np.random.default_rng(rng).uniform(0, 2 * np.pi, n)
    magnitudes = np.ones(n)
    a = np.zeros(n)
    # Inverse FFT of random phase
    spectrum = magnitudes * np.exp(1j * freqs)
    a = np.real(np.fft.ifft(spectrum))
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    if init_type == 0:
        # Triangle
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        # U-shape
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        # Cosine window
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        # Two humps
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Spectral
        a = spectral_init(n, rng.integers(0, 2**31))
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
        S = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            v = a[i]
            if v < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i: k_end - i + 1]
            conv_affected = conv[k_start: k_end + 1]

            best_delta = 0.0
            best_new_score = best_score

            # Fine multiplicative steps
            for factor in [0.999, 0.997, 0.995, 0.998, 1.002, 1.004, 1.006, 1.003, 0.99, 1.01]:
                delta = v * (factor - 1.0)
                new_val = v + delta
                if new_val < -1e-15:
                    continue

                new_aff = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] - v * v + new_val * new_val

                new_M = max(unaff_max, float(new_aff.max()))
                new_S = S - v + new_val
                if new_S < 1e-15:
                    continue
                new_score = 2.0 * n * new_M / (new_S * new_S)

                if new_score < best_new_score - 1e-13:
                    best_new_score = new_score
                    best_delta = delta
                    best_new_M = new_M
                    best_new_S = new_S

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                delta = best_delta
                a_shifted_copy = a_shifted.copy()
                conv[k_start:k_end + 1] += 2.0 * delta * a_shifted_copy
                if 2 * i >= k_start and 2 * i <= k_end:
                    conv[2 * i] += delta * delta
                S = best_new_S
                M = best_new_M
                best_score = best_new_score
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    s_final = best_a.sum()
    if s_final > 1e-15:
        best_a = best_a / s_final
        best_score = 2.0 * n * float(fast_convolve(best_a).max())
    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining exact coordinate descent, LP alternating,
    Nesterov smoothed descent, and fine polishing. Explores multiple sequence lengths.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # Determine initial n
    if parent is not None:
        n_initial = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_initial = 1024
        best_seq = np.ones(n_initial) / n_initial
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n_initial}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP Alternating (30% time) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + min(540.0, (deadline - time.monotonic()) * 0.30)

    t_alloc = min(t1_end - time.monotonic(), 300.0)
    if t_alloc > 30:
        print(f"  LP enhanced from parent, budget={t_alloc:.0f}s")
        a_lp, s_lp = lp_alternating_enhanced(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp, "lp_parent")

    # Try from a diverse init
    if time.monotonic() < t1_end - 60:
        init_div = diverse_init(n_initial, rng, 2)
        t_alloc2 = min(t1_end - time.monotonic(), 200.0)
        if t_alloc2 > 30:
            print(f"  LP enhanced from cosine init, budget={t_alloc2:.0f}s")
            a_lp2, s_lp2 = lp_alternating_enhanced(init_div, t_alloc2, solver, verbose=False)
            try_update(a_lp2, "lp_div")

    # === Phase 2: Exact Coordinate Descent (35% time) ===
    print(f"\n=== Phase 2: Exact CD (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + min(630.0, (deadline - time.monotonic()) * 0.35)

    t_alloc = min(t2_end - time.monotonic(), 350.0)
    if t_alloc > 30:
        print(f"  ECD from best, budget={t_alloc:.0f}s")
        a_e1, s_e1 = exact_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_e1, "ecd_main")

    if time.monotonic() < t2_end - 120:
        t_alloc2 = min(t2_end - time.monotonic(), 200.0)
        if t_alloc2 > 30:
            print(f"  ECD from LP result, budget={t_alloc2:.0f}s")
            a_e2, s_e2 = exact_coord_descent(a_lp, t_alloc2, verbose=True, label="lp")
            try_update(a_e2, "ecd_lp")

    # Second ECD pass on best
    if time.monotonic() < t2_end - 60:
        t_alloc3 = min(t2_end - time.monotonic(), 150.0)
        if t_alloc3 > 20:
            print(f"  ECD round 2, budget={t_alloc3:.0f}s")
            a_e3, s_e3 = exact_coord_descent(best_seq, t_alloc3, verbose=True, label="r2")
            try_update(a_e3, "ecd_r2")

    # === Phase 3: Nesterov Smooth Descent (15% time) ===
    print(f"\n=== Phase 3: Nesterov (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + min(270.0, (deadline - time.monotonic()) * 0.15)

    if time.monotonic() < t3_end - 30:
        t_alloc = min(t3_end - time.monotonic(), 200.0)
        if t_alloc > 30:
            print(f"  Nesterov from best, budget={t_alloc:.0f}s")
            a_ne, s_ne = nesterov_smooth_descent(best_seq, t_alloc, beta=150.0, verbose=True)
            try_update(a_ne, "nesterov")

            # From LP result too
            if time.monotonic() < t3_end - 30:
                a_ne2, s_ne2 = nesterov_smooth_descent(a_lp, min(t3_end - time.monotonic(), 150.0),
                                                       beta=100.0, verbose=False)
                try_update(a_ne2, "nesterov_lp")

    # === Phase 4: Multi-n exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-n exploration (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + min(180.0, (deadline - time.monotonic()) * 0.10)

    # Try different n values
    for n_try in [512, 2048]:
        if time.monotonic() > t4_end - 30:
            break
        if n_try == n_initial:
            continue
        t_alloc = min(t4_end - time.monotonic() - 20, 120.0)
        if t_alloc < 30:
            continue
        print(f"  Trying n={n_try}, budget={t_alloc:.0f}s")
        # Initialize
        for itype in [0, 2]:
            init_n = diverse_init(n_try, rng, itype)
            t_sub = min(t_alloc * 0.5, 80.0)
            if t_sub < 20:
                break
            a_tmp, s_tmp = exact_coord_descent(init_n, t_sub, verbose=False, label=f"n{n_try}_i{itype}")
            try_update(a_tmp, f"ecd_n{n_try}")
        if time.monotonic() > t4_end - 20:
            break

    # === Phase 5: Fine Polish (10% time) ===
    print(f"\n=== Phase 5: Fine Polish (best={best_score:.8f}) ===")
    t5_end = deadline - 30

    if time.monotonic() < t5_end - 10:
        t_alloc = min(t5_end - time.monotonic() - 10, 120.0)
        if t_alloc > 15:
            print(f"  FinePolish from best, budget={t_alloc:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_alloc, verbose=True)
            try_update(a_fp, "fine_polish")

    # Final ECD pass
    if time.monotonic() < t5_end - 10:
        t_final = min(t5_end - time.monotonic() - 10, 60.0)
        if t_final > 10:
            a_fin, s_fin = exact_coord_descent(best_seq, t_final, verbose=False, label="final")
            try_update(a_fin, "final_ecd")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
