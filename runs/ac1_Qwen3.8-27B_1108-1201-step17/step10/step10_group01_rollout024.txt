import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_T_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        idx = np.arange(j, j + n)
        rows.extend(idx)
        cols.extend([j] * n)
        vals.extend(a)
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


_lp_solver_name = None


def get_lp_solver():
    """Find a working LP solver globally."""
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return _lp_solver_name
        except Exception:
            continue
    _lp_solver_name = cp.SCS
    return _lp_solver_name


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_T_sparse(a_fixed)
    size = 2 * n - 1
    M_vec = M_var * np.ones(size)
    constraints = [T @ b_var <= M_vec, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=100000,
                         stall_patience=15, top_k_avg=3):
    """Nesterov dual-averaging LP alternation with top-k peak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 25:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise_scale = 0.03 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_LP={M_b:.8f}")

def entropy_proximal_subgradient(a0, time_limit, n_steps=200000,
                                 eta0=0.05, decay=0.9997, top_k=5):
    """
    Multiplicative weights (entropy proximal) subgradient method.
    Uses the geometry of the KL divergence for faster convergence on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        # Find top-k peaks for averaged subgradient
        peak_vals = np.sort(b)[::-1][:top_k]
        threshold = peak_vals[-1] * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute averaged subgradient: g[j] = (2/|active|) * sum_{k in active} a[k-j]
        g = np.zeros(n)
        for k in active:
            shift = k
            # a[k-j] for j=0..n-1 means a[shift-j]
            # This is a reversed and shifted version of a
            lo = max(0, shift - (n - 1))
            hi = min(n - 1, shift)
            # g[j] gets contribution a[shift-j] where 0 <= shift-j < n
            # i.e., j in [shift-(n-1), shift] intersect [0, n-1]
            j_start = max(0, shift - (n - 1))
            j_end = min(n - 1, shift)
            if j_start <= j_end:
                # For j in [j_start, j_end]: a[shift - j]
                # shift - j ranges from shift-j_start to shift-j_end
                idx = np.arange(j_start, j_end + 1)
                g[idx] += a[shift - idx]
        g *= 2.0 / len(active)

        # Multiplicative weights update
        log_a = np.log(np.maximum(a, 1e-300))
        new_log = log_a - eta * g
        a_new = np.exp(new_log - new_log.max())
        a_new = a_new / a_new.sum()

        new_score = 2.0 * n * fast_convolve(a_new).max()
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve >= 50:
            eta *= 0.95
            no_improve = 0
        eta = max(1e-8, eta)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EntropySG {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def smooth_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing, multi-stage temperature cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        n_active_its = min(maxiter_per_stage, int(remaining / 0.05))
        n_active_its = max(10, min(n_active_its, 8000))

        def func_loc(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            aa = np.exp(z - z.max())
            aa = aa / aa.sum()
            bb = fast_convolve(aa)
            max_b = bb.max()
            exp_b = np.exp(bb - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse

        def jac_loc(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            aa = np.exp(z - z.max())
            aa = aa / aa.sum()
            bb = fast_convolve(aa)
            max_b = bb.max()
            exp_b = np.exp(bb - max_b)
            w = exp_b / exp_b.sum()
            # Gradient of sum_k w[k] * (2 * a * shifted stuff)
            # d/d(a[j]) of conv(a,a)[k] = 2*a[k-j] if valid
            # So grad_a = sum_k w[k] * 2*a[k-j] (over valid k)
            grad_a = np.zeros(n)
            for k in range(len(bb)):
                if w[k] < 1e-12:
                    continue
                j_start = max(0, k - (n - 1))
                j_end = min(n - 1, k)
                if j_start <= j_end:
                    idx = np.arange(j_start, j_end + 1)
                    grad_a[idx] += w[k] * 2.0 * aa[k - idx]
            # Chain rule for softmax normalization
            S = np.dot(grad_a, aa)
            return aa * (grad_a / T_stage - S / T_stage)

        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': n_active_its, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': n_active_its * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_coord_descent(a0, time_limit, n_steps=100000, top_m=6):
    """Coordinate descent targeting top convolution peaks with simultaneous moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]

        improved = False
        # Try reducing top contributors to the worst peak
        for peak_k in top_peaks[:3]:
            j_start = max(0, peak_k - (n - 1))
            j_end = min(n - 1, peak_k)
            idx = np.arange(j_start, j_end + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.985, 0.98, 0.97, 0.96]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Simultaneous reduction
            peak_k = top_peaks[0]
            j_start = max(0, peak_k - (n - 1))
            j_end = min(n - 1, peak_k)
            idx = np.arange(j_start, j_end + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    # Perturb to escape
                    rng = np.random.default_rng(step)
                    a = a + 0.01 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    elif choice == 7:
        a = (1 - x) ** 0.7 + 0.1
    elif choice == 8:
        # Flat-top with smooth edges
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.5 * (1 + np.cos(np.pi * np.arange(edge) / edge))
        a[-edge:] = a[:edge][::-1]
    else:
        # Slightly non-uniform
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def block_coord_descent(a0, time_limit, block_size=20, n_steps=50000):
    """Block coordinate descent: optimize small blocks at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    n_blocks = n // block_size + 1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a random block (weighted toward regions contributing to peaks)
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Bias block selection toward the peak region
        if rng.random() < 0.5:
            # Pick block near peak
            center_idx = peak_k // 2
            block_idx = max(0, min(n_blocks - 1, center_idx // block_size + rng.integers(-2, 3)))
        else:
            block_idx = rng.integers(0, n_blocks)

        start = block_idx * block_size
        end = min(n, start + block_size)
        if end - start < 2:
            continue

        # Optimize this block using local line search
        block = a[start:end].copy()
        rest_sum = a.sum() - block.sum()

        for _ in range(5):
            # Compute gradient for this block
            grad_block = np.zeros(end - start)
            b = fast_convolve(a)
            max_b = b.max()
            # Subgradient from peak
            active_thresh = max_b * 0.9995
            active = np.where(b >= active_thresh)[0]
            for k in active:
                j_start = max(0, k - (n - 1))
                j_end = min(n - 1, k)
                for j in range(start, end):
                    if j_start <= j <= j_end:
                        grad_block[j - start] += 2.0 * a[k - j]
            grad_block /= len(active)

            # Try a few step sizes
            for lr in [0.005, 0.002, 0.001, 0.0005]:
                block_new = block - lr * grad_block
                block_new = np.maximum(block_new, 0)
                block_new = block_new / (block_new.sum() + 1e-15) * (rest_sum + block.sum())

                a_trial = a.copy()
                a_trial[start:end] = block_new
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    block = block_new
                    break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP alternation, entropy-proximal
    subgradient, smooth L-BFGS, multi-n exploration, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) was {old_score:.8f}")

    # === Phase 1: Primary LP alternation ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t1, solver=solver,
                                        max_iters=100000, stall_patience=12, top_k_avg=3)
        try_update(a_lp, "lp_phase1")

    # === Phase 2: Entropy Proximal Subgradient (NEW) ===
    print(f"\n=== Phase 2: Entropy Proximal, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 280.0)
    if t2 > 15:
        print(f"  Budget: {t2:.0f}s")
        a_eg, _ = entropy_proximal_subgradient(best_seq, t2, n_steps=300000,
                                                eta0=0.03, decay=0.9998, top_k=5)
        try_update(a_eg, "entropy_sg")

    # === Phase 3: Smooth L-BFGS ===
    print(f"\n=== Phase 3: Smooth L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 15:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]
        print(f"  T stages ({len(T_stages)}), budget={t3:.0f}s")
        a_lb = smooth_lbgfs(best_seq, t3, T_stages=T_stages, maxiter_per_stage=6000)
        try_update(a_lb, "lbgfs_phase3")

    # === Phase 4: Peak-directed coordinate descent ===
    print(f"\n=== Phase 4: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 10:
        a_cd, _ = peak_coord_descent(best_seq, t4, n_steps=100000, top_m=6)
        try_update(a_cd, "peak_cd")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n exploration, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        # Try a few different lengths
        candidate_ns = []
        for factor in [1.15, 1.3, 0.8, 1.5]:
            alt_n = int(n_cur * factor)
            if 400 <= alt_n <= 3000:
                candidate_ns.append(alt_n)
        # Deduplicate
        candidate_ns = sorted(set(candidate_ns))

        t_each = min(rem / max(1, len(candidate_ns)), 120.0)
        for alt_n in candidate_ns[:4]:
            if time.monotonic() > deadline - 40:
                break
            if t_each < 20:
                break
            print(f"  Trying n={alt_n}, t={t_each:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_optimize(init_alt, t_each * 0.6, solver=solver,
                                             max_iters=30000, stall_patience=10)
            try_update(a_alt, f"lp_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 15:
                a_alt2 = smooth_lbgfs(a_alt, min(rem * 0.3, 30.0),
                                       T_stages=[3.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                       maxiter_per_stage=3000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 6: Second LP pass on best ===
    print(f"\n=== Phase 6: LP Pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 180.0)
    if t6 > 20:
        a_lp2, _ = nesterov_lp_optimize(best_seq, t6, solver=solver,
                                         max_iters=50000, stall_patience=15)
        try_update(a_lp2, "lp_phase6")

    # === Phase 7: Multi-start perturbation ===
    print(f"\n=== Phase 7: Multi-start, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 50.0)
        for pi in range(3):
            if time.monotonic() > deadline - 30:
                break
            if t_each < 15:
                break
            if pi == 0:
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.04 * np.cos(3 * x) + 0.02 * np.cos(7 * x)
            elif pi == 1:
                pert = 0.03 * rng.standard_normal(n_cur)
            else:
                # Asymmetric perturbation
                x = np.linspace(0, 1, n_cur)
                pert = 0.03 * (x - 0.5) * np.sin(4 * np.pi * x)
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps < 1e-15:
                continue
            a_pert = a_pert / ps
            a_res, _ = nesterov_lp_optimize(a_pert, t_each, solver=solver,
                                             max_iters=15000, stall_patience=8)
            try_update(a_res, f"perturb_{pi}")

    # === Phase 8: Block coordinate descent ===
    print(f"\n=== Phase 8: Block CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.15, 120.0)
    if t8 > 10:
        a_bc, _ = block_coord_descent(best_seq, t8, block_size=15, n_steps=50000)
        try_update(a_bc, "block_cd")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 40.0)
        if t_fin > 8:
            a_fin = smooth_lbgfs(best_seq, t_fin,
                                  T_stages=[1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01],
                                  maxiter_per_stage=4000)
            try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_cd2, _ = peak_coord_descent(best_seq, min(rem * 0.5, 25.0), n_steps=50000, top_m=5)
        try_update(a_cd2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 8:
        # Quick entropy proximal pass
        a_eg2, _ = entropy_proximal_subgradient(best_seq, min(rem * 0.6, 15.0),
                                                 n_steps=50000, eta0=0.01,
                                                 decay=0.999, top_k=3)
        try_update(a_eg2, "final_entropy")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
