Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive p-norm optimization
with multi-start search over sequence lengths, FFT-based gradient computation,
LP best-response alternating, and targeted coordinate descent.
Key innovations: p-norm schedule from p=2 to p=inf for smooth-to-sharp
transition, multi-n search, and adversarial peak reduction.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_pnorm_grad(b, a, p):
    """
    Compute gradient of sum_k b_k^p w.r.t. a, where b = conv(a,a).
    grad[i] = 2 * sum_j p*b_{i+j}^{p-1} * a[j] (with b[k]=0 for k>=2n-1)
    Computed via FFT-based cross-correlation.
    """
    n = len(a)
    w = p * np.power(np.maximum(b, 0), p - 1)
    fft_size = 1 << (4 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cc = irfft(W * np.conj(A), fft_size)
    return 2.0 * cc[:n]


def compute_max_grad(b, a, active_indices):
    """Compute subgradient of max(b) at active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= max(1, len(active_indices))
    return g


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached

    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:

    b_var = cp.Variable(n, nonneg=True)

    T = build_conv_matrix(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

        prob.solve(solver=solver)

        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def pnorm_mirror_descent(a0, time_limit, p_start=2.0, p_end=200.0,
                         n_p_stages=12, nesterov=True):
    """
    Mirror descent with adaptive p-norm objective.
    Starts with smooth p-norm and increases p toward infinity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Create p schedule (geometric)
    p_schedule = np.geomspace(p_start, p_end, n_p_stages)
    iters_per_stage = 200000

    total_iters = iters_per_stage * n_p_stages
    global_it = 0
    no_improve = 0

    for stage_i, p_val in enumerate(p_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        stage_t0 = time.monotonic()
        stage_budget = time_limit - (stage_t0 - t0)

        eta = 0.04 / (1.0 + 0.1 * stage_i)
        prev_a = a.copy()
        prev_g = None

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute objective and gradient at Nesterov point
            b = fast_convolve(y)
            b_clipped = np.maximum(b, 0)

            if p_val < 50:
                # p-norm gradient
                b_p = np.power(b_clipped, p_val)
                obj_val = (b_p.sum()) ** (1.0 / p_val)
                w = p_val * np.power(b_clipped, p_val - 1)
                fft_size = 1 << (4 * n - 1).bit_length()
                W = rfft(w, fft_size)
                A = rfft(y, fft_size)
                cc = irfft(W * np.conj(A), fft_size)
                g = 2.0 * cc[:n]
            else:
                # Use soft-max weighting (approximates max)
                max_b = b_clipped.max()
                exp_b = np.exp(b_clipped - max_b)
                w = exp_b / exp_b.sum()
                fft_size = 1 << (4 * n - 1).bit_length()
                W = rfft(w * (2 * n - 1), fft_size)  # scale to approximate subgradient
                A = rfft(y, fft_size)
                cc = irfft(W * np.conj(A), fft_size)
                g = 2.0 * cc[:n]

            # Barzilai-Borwein step size
            if it >= 3 and prev_g is not None:
                s_vec = a - prev_a
                y_vec = g - prev_g
                s_dot_s = np.dot(s_vec, s_vec)
                s_dot_y = np.dot(s_vec, y_vec)
                y_dot_y = np.dot(y_vec, y_vec)
                if s_dot_s > 1e-18 and y_dot_y > 1e-18:
                    if abs(s_dot_y) > 1e-18:
                        eta_bb = s_dot_s / abs(s_dot_y)
                        eta = 0.6 * eta + 0.4 * eta_bb
                    else:
                        eta = s_dot_s / y_dot_y
                    eta = max(1e-6, min(eta, 0.5))
                else:
                    eta *= 0.995

            prev_a = a.copy()
            prev_g = g.copy()

            # Mirror descent (exponential) update
            log_a = np.log(np.maximum(a, 1e-300))
            g_shifted = g - g.min()
            g_shifted = g_shifted / (g_shifted.max() + 1e-300)
            log_a_new = log_a - eta * g_shifted
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-300:
                a_new = a.copy()
            a_new = a_new / a_new_sum

            # Nesterov
            if nesterov:
                alpha = 2.0 / (global_it + 2)
                y = (1.0 - alpha) * y + alpha * a_new
                y = np.maximum(y, 0)
                ys = y.sum()
                if ys < 1e-15:
                    y = a_new.copy()
                else:
                    y = y / ys

            a = a_new
            global_it += 1

            # Evaluate
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score - 1e-13:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= 500:
                rng = np.random.default_rng(global_it * 13 + 7)
                noise_scale = 0.003 * np.sqrt(no_improve / 500)
                a = a + rng.standard_normal(n) * noise_scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                y = a.copy()
                no_improve = 0
                eta = max(eta, 0.02)

        if (stage_i + 1) % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    pNorm stage={stage_i+1}/{n_p_stages} p={p_val:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def logsumexp_gradient(a, tau):
    """Compute gradient of tau*log(sum(exp(b_k/tau))) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / tau)
    w = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    fft_size = 1 << (4 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cc = irfft(W * np.conj(A), fft_size)
    obj = b_max + tau * np.log(exp_b.sum())
    return obj, 2.0 * cc[:n]


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 2:
            break

        def func(z_inner):
            a_inner = softmax_safe(z_inner)
            obj, _ = logsumexp_gradient(a_inner, T)
            return obj

        def jac(z_inner):
            a_inner = softmax_safe(z_inner)
            _, grad_a = logsumexp_gradient(a_inner, T)
            S = np.dot(grad_a, a_inner)
            return a_inner * (grad_a - S)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxiter_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_safe(z)


def lp_alternating_refine(a0, time_limit, solver=None, n_max_iters=30000):
    """Nesterov-accelerated LP alternating method."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(99999)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-11:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 15:
                noise_scale = 0.01 / np.sqrt(stall)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_targeting_refine(a0, time_limit, n_steps=20000):
    """
    Targeted coordinate descent: find peak positions, identify most
    contributing elements, and reduce them while preserving sum.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995, 0.99]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-18:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                # Try multi-coordinate adjustment
                peak_k = int(np.argmax(fast_convolve(a)))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:5]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-14:
                        a_trial[pos] *= 0.9995
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
            if stall % 500 == 0 and stall > 0:
                # Periodic noise
                rng = np.random.default_rng(step)
                a = a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakTarget {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng, init_type=0):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if init_type == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif init_type == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif init_type == 2:
        a = np.exp(-4 * x * (1 - x))
    elif init_type == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif init_type == 4:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.15)
    elif init_type == 5:
        # Sine window with slight asymmetry
        a = np.sin(np.pi * x ** 0.8) * np.sin(np.pi * x ** 1.2)
        a = np.maximum(a, 0)
    elif init_type == 6:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Smooth random
        a = np.ones(n)
        for _ in range(3):
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            a = a * (1.0 + 0.3 * np.sin(2 * np.pi * freq * x + phase))
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multi_n_search(start_seq, start_score, time_budget, rng):
    """
    Quick search over different sequence lengths to find the best n.
    Returns best (sequence, score, n) found.
    """
    best_seq = start_seq.copy()
    best_score = start_score
    t0 = time.monotonic()

    n_base = len(start_seq)
    n_candidates = sorted(set([n_base // 2, int(n_base * 0.75), n_base,
                                int(n_base * 1.25), int(n_base * 1.5),
                                int(n_base * 2)]))
    time_per_n = max(5.0, time_budget / len(n_candidates))

    for n in n_candidates:
        if time.monotonic() - t0 > time_budget - 3:
            break
        if n < 50 or n > 10000:
            continue

        a_test = interpolate_sequence(start_seq, n)
        # Quick p-norm optimization
        a_test, sc = pnorm_mirror_descent(a_test, time_per_n,
                                           p_start=2.0, p_end=50.0,
                                           n_p_stages=4, nesterov=True)
        if sc < best_score:
            best_score = sc
            best_seq = a_test.copy()
            print(f"  MultiN: n={n} score={sc:.8f} (NEW BEST)")
        else:
            print(f"  MultiN: n={n} score={sc:.8f}")

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive p-norm mirror descent, LP alternating, LogSumExp L-BFGS,
    and peak-targeting coordinate descent with multi-n search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s, deadline in {budget_s - 20:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, init_type=0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 0: Multi-n search (find best length) ===
    print(f"\n=== Phase 0: Multi-n Search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mn, sc_mn = multi_n_search(best_seq, best_score, t_alloc, rng)
        if sc_mn < best_score:
            best_seq = a_mn
            best_score = sc_mn
            n_best = len(best_seq)
        print(f"  After multi-n: score={best_score:.8f}, n={n_best}")

    # === Phase 1: P-norm mirror descent (main workhorse) ===
    print(f"\n=== Phase 1: P-norm Mirror Descent (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn, sc = pnorm_mirror_descent(best_seq, t_alloc,
                                         p_start=2.0, p_end=300.0,
                                         n_p_stages=10, nesterov=True)
        try_update(a_pn, "pnorm_p1")

    # === Phase 2: LogSumExp L-BFGS (smooth refinement) ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 20:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s, T_stages={T_stages[:5]}...")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                                maxiter_per_stage=3000)
        try_update(a_lb, "lbgfs_p2")

    # === Phase 3: LP Alternating (disactive set decisions) ===
    print(f"\n=== Phase 3: LP Alternating (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc = lp_alternating_refine(best_seq, t_alloc, solver=solver,
                                          n_max_iters=50000)
        try_update(a_lp, "lp_alter")

    # === Phase 4: Second p-norm pass (finer) ===
    print(f"\n=== Phase 4: P-norm Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn2, sc = pnorm_mirror_descent(best_seq, t_alloc,
                                          p_start=10.0, p_end=500.0,
                                          n_p_stages=5, nesterov=True)
        try_update(a_pn2, "pnorm_p4")

    # === Phase 5: Peak targeting (coordinate descent) ===
    print(f"\n=== Phase 5: Peak Targeting (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pt, sc = peak_targeting_refine(best_seq, t_alloc, n_steps=50000)
        try_update(a_pt, "peak_target")

    # === Phase 6: Final LogSumExp polish ===
    print(f"\n=== Phase 6: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        T_stages_fin = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages_fin,
                                 maxiter_per_stage=2000)
        try_update(a_fin, "final_lbgfs")

    # === Phase 7: Final peak targeting polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 40.0)
    if t_alloc > 5:
        a_polish, sc = peak_targeting_refine(best_seq, t_alloc, n_steps=30000)
        try_update(a_polish, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
