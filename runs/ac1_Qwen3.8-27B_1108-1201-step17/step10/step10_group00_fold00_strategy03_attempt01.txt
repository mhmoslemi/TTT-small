# Non-trivial finite differences through Laplace parallelization
def finite_differences(sequence):
    """
    # Calculate finite differences for gradient descent via Laplacian matrix.
    """
    # Stochastic Laplacian matrix construction for gradient estimation.
    n = len(sequence)
    smoothed_sequence = gaussian_filter1d(sequence, sigma=0.2)
    # Calculate smooth derivatives with centered difference with adaptive sigma.
    logit_sum = np.zeros(n)
    for i in range(n):
        term = np.exp(logit_sum[i] - smoothed_sequence[i]) / (1 + np.exp(logit_sum[i] - smoothed_sequence[i]))
        logit_sum[i] = np.log(term).mean()

    grad = np.zeros(n)
    weights = np.exp(-np.abs(logit_sum))
    for i in range(n):
        left = max(0, i-1)
        right = max(0, i+1)
        response_left = smoothed_sequence[left] - smoothed_sequence[right]
        grad[i] = weights[left] * (response_left * np.exp(logit_sum[right])) + weights[right] * (response_right * np.exp(logit_sum[left]))
    
    return grad
