import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    # Use FFT-based approach instead of explicit matrix for large n
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed):
    """Solve LP to find b minimizing max conv(a, b) subject to b >= 0, sum(b) = 1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None
                return b_val / s
        except:
            continue
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach combining
accelerated mirror descent, LP best-response iteration, LogSumExp-smoothed L-BFGS,
and targeted coordinate descent. Key improvements: adaptive backtracking step size,

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    if n > 3000:
        return None
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None
                return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(1, len(active_indices))
    return g


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def mirror_descent_optimize(a0, time_limit, eta0=0.06, active_frac=0.999,
                            nesterov=True, max_iters=5000000):
    """
    Mirror descent with backtracking line search and Nesterov acceleration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()

        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = compute_subgradient(y, active)
        g_shifted = g - g.min()

        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum

        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new
        score = evaluate_local(a)
        score_y = 2.0 * n * fast_convolve(y).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 200:
            eta *= 0.95
        elif no_improve < 50:
            eta = min(eta * 1.02, 0.5)

        if no_improve >= 500:
            rng = np.random.default_rng(it * 7 + 13)
            noise_scale = 0.003 + 0.003 * np.sqrt(min(no_improve / 500.0, 10.0))
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.3)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f} |A|={len(active)}")

    return best_a, best_score


def logsumexp_obj_and_grad(z, T_temp):
    """Compute LogSumExp objective and gradient for softmax parameterization."""
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()

    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    logsumexp = max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T_temp) * logsumexp

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_obj_and_grad(z_inner, T)
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_obj_and_grad(z_inner, T)
            return grad

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_safe(z)

def structured_init(n, rng):
    """Create a diverse set of structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.2)
    elif choice == 5:
        a = x ** 0.5 * (1 - x) ** 0.5
        a = np.maximum(a, 1e-6)

        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def peak_flattening(a0, time_limit, n_rounds=1000):
    """
    Direct peak flattening: iteratively identify the dominant peak and
    reduce it by adjusting contributing coordinates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Find the top contributor and reduce it
        max_contrib_pos = idx[np.argmax(contribs)]
        if a[max_contrib_pos] < 1e-10:
            break

        # Try multiple reduction factors
        best_factor = None
        best_trial_score = best_score
        for factor in [0.99999, 0.99998, 0.99995, 0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.995, 0.99]:
            a_trial = a.copy()
            a_trial[max_contrib_pos] *= factor
            diff = a[max_contrib_pos] - a_trial[max_contrib_pos]
            mask = np.ones(n, dtype=bool)
            mask[max_contrib_pos] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                break
            # Redistribute to the "other" element in the pair (the one at peak_k - pos)
            partner = peak_k - max_contrib_pos
            if 0 <= partner < n:
                a_trial[partner] += diff
            else:
                a_trial[mask] = others + diff * others / others_sum
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = evaluate_local(a_trial)
            if score < best_trial_score:
                best_trial_score = score
                best_factor = factor
                best_a_trial = a_trial.copy()

        if best_factor is not None:
            a = best_a_trial
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_a = a.copy()
        else:
            # Try pair adjustment: reduce both elements contributing to the peak
            top2 = idx[np.argsort(contribs)[::-1][:2]]
            improved = False
            for f1 in [0.9999, 0.9998, 0.9995, 0.999]:
                for f2 in [0.9999, 0.9998, 0.9995, 0.999]:
                    a_trial = a.copy()
                    a_trial[top2[0]] *= f1
                    a_trial[top2[1]] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ts = a_trial.sum()
                    if ts < 1e-15:
                        continue
                    a_trial /= ts
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            if not improved and rnd % 100 == 0:
                # Small random perturbation
                a_trial = a + rng.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()

        if (rnd + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakFlat {elapsed:.0f}s round={rnd+1}: score={best_score:.8f}")

    return best_a, best_score

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    print(f"Starting propose_candidate, budget={budget_s:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Mirror Descent (main workhorse) ===
    print(f"\n=== Phase 1: Mirror Descent (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(best_seq, t_alloc,
                                          eta0=0.06, active_frac=0.999,
                                          nesterov=True)
        try_update(a_md, "mirror_descent_p1")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, n_max_iters=30000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 3: LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        T_stages = [10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                               maxiter_per_stage=4000)
        try_update(a_lb, "lbgfs")

    # === Phase 4: Peak Flattening (direct approach) ===
    print(f"\n=== Phase 4: Peak Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pf, _ = peak_flattening(best_seq, t_alloc, n_rounds=5000)
        try_update(a_pf, "peak_flattening")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=60000)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Second Mirror Descent (finer) ===
    print(f"\n=== Phase 6: Refine MD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 100.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md2, _ = mirror_descent_optimize(best_seq, t_alloc,
                                           eta0=0.01, active_frac=0.9999,
                                           nesterov=True)
        try_update(a_md2, "mirror_descent_p2")

    # === Phase 7: Final L-BFGS polish ===
    print(f"\n=== Phase 7: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 15:
        T_stages_fin = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages_fin,
                                maxiter_per_stage=2000)
        try_update(a_fin, "final_lbgfs")

    # === Phase 8: Final coordinate polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = max(remaining if (remaining := rem * 0.10) else 10.0, 10.0)
    t_alloc = min(t_alloc, 50.0)
    if t_alloc > 5:
        print(f"  Budget: {t_alloc:.0f}s")
        a_polish, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=40000)
        try_update(a_polish, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
