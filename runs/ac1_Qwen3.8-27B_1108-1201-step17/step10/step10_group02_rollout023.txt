        top_pos = int(idx[np.argmax(contribs)])
        
        # Best target: position with minimum a value (excluding source)
        target = int(np.argmin(np.where(np.arange(n) == top_pos, 1e10, a)))

        for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97]:
            delta = a[top_pos] * (1 - factor)
            if delta < 1e-15:
                break
            a_trial = a.copy()
            a_trial[top_pos] -= delta
            a_trial[target] += delta
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                break
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try more targets
            for target2 in ...:  # try a few more targets
                ...
        
        it += 1
        if it % 5000 == 0:
            print(f"  CD iter={it}: score={best_score:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated LP
alternation, peak-redirect coordinate descent, and LogSumExp L-BFGS polishing.
Explores multiple sequence lengths for better optimization surface.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    """Project to probability simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_T(a):
    """Build matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            v = cp.Variable(5, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(v)), [cp.sum(v) == 1])
            p.solve(solver=name)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = name
                return name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_best_response(a, T, solver, b_warm=None):
    """Solve LP: min M s.t. (Tb)_k <= M, sum(b)=1, b>=0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    if b_warm is not None:
        b_var.value = b_warm
        M_var.value = float((T @ b_warm).max())
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
            return b_val, float(M_var.value)
    except Exception:
        pass
    # Fallback: try other solvers
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            b_var2 = cp.Variable(n, nonneg=True)
            M_var2 = cp.Variable()
            prob2 = cp.Problem(
                cp.Minimize(M_var2),
                [T @ b_var2 <= M_var2, cp.sum(b_var2) == 1]
            )
            prob2.solve(solver=alt)
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var2.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val /= s
                return b_val, float(M_var2.value)
        except Exception:
            continue
    return None, None


def nesterov_lp(a0, T, solver, time_limit):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    b_warm = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(99)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = solve_best_response(y, T, solver, b_warm=b_warm)
        if b_new is None:
            break
        b_warm = b_new

        # Nesterov momentum
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        else:
            y = a.copy()

        a = ((it + 1.0) * a + alpha * (it + 2.0) * b_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a /= sa
        else:
            a = y.copy()

        sc_a = compute_score(a)
        sc_y = compute_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        elif sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stuck
        if stall > 200:
            noise = rng.standard_normal(n) * 0.01
            y = y + noise
            y = normalize(y)
            a = y.copy()
            b_warm = y.copy()
            stall = 0
            prev_best = best_score

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt it={it+1} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_val:.8f}")

    return best_a, best_score
