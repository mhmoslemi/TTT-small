def golden_section_search(f, lo, hi, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def spectral_refinement(a0, time_limit, verbose=False, label=""):
    """Iterative refinement in frequency domain to flatten the convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())
        # Compute the "excess" at each lag relative to the target
        # We want all convolution values <= target
        target = max_val * 0.999
        # Soft penalty: increase weights at high-convolution positions
        # and decrease at low positions
        weights = np.ones(n)
        for k in range(2 * n - 1):
            if conv[k] > target:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    weights[i] += 0.01 * (conv[k] - target) / max_val
        a_new = a / weights
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 1e-10:
            a_new = a
        else:
            a_new = a_new / cs
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.2, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)

    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_avg = (iters * a_avg + a_new) / (iters + 1)
        sc_avg = compute_score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        a = a_new
        lr *= 0.999
        if no_improve > 100:
            lr *= 0.7
            no_improve = 0
        if no_improve > 500:
            break

    return best_a, best_score
