"""
FFT-accelerated minimization of 2*n*max(conv(a,a))/(sum(a))^2 using:
- LogSumExp smoothing with multi-stage temperature cooling
- L-BFGS-B on log-parameterization (softmax)
- Nesterov-accelerated projected gradient on simplex
- Final hard-max coordinate descent polish
- Multi-length and multi-start exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute cross-correlation: result[i] = sum_m w[i+m] * a[m] for valid indices."""
    n = len(a)
    w_flip = w[::-1]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    WF = rfft(w_flip, fft_size)
    AF = rfft(a, fft_size)
    conv_result = irfft(WF * AF, fft_size)[:size]
    return conv_result[n - 1::-1]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_smoothed_obj_and_grad(z, n, T):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. z (log-parameter)."""
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s
    b = fast_convolve(a)
    # LogSumExp
    max_b = b.max()
    exp_Tb = np.exp(T * (b - max_b))
    lse = max_b + np.log(exp_Tb.sum()) / T
    # Gradient of LSE w.r.t. a
    w = exp_Tb / exp_Tb.sum()  # softmax weights
    # g_a[i] = 2 * sum_k w[k] * a[k-i] = 2 * cross_corr(w, a)[i]
    g_a = 2.0 * fast_cross_corr(w, a)
    # Gradient w.r.t. z: g_z[i] = a[i] * (g_a[i] - dot(a, g_a))
    dot_ag = np.dot(a, g_a)
    g_z = a * (g_a - dot_ag)
    obj = 2.0 * n * lse
    return obj, g_z


def lse_lbgfs(a0, time_limit, T_stages=None, maxiter=None):
    """L-BFGS-B optimization with LogSumExp smoothing and multi-stage T cooling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(a + 1e-300)
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.01]
    
    if maxiter is None:
        maxiter = 5000
    
    best_z = z.copy()
    best_score = float('inf')
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_eval):
            obj, _ = lse_smoothed_obj_and_grad(z_eval, n, T)
            return obj
        
        def jac(z_eval):
            _, g = lse_smoothed_obj_and_grad(z_eval, n, T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter * 100, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"  L-BFGS error at T={T}: {e}")
        
        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        score = evaluate_local(a_cur)
        if score < best_score:
            best_score = score
            best_z = z.copy()
        
        if len(T_stages) > 2:
            print(f"  T={T:.4f}: score={score:.10f}, elapsed={time.monotonic()-t0:.1f}s, status={res.status if 'res' in dir() else 'N/A'}")
    
    a_out = np.exp(best_z - best_z.max())
    a_out = a_out / a_out.sum()
    return a_out


def nesterov_projected_grad(a0, time_limit, T=0.1, max_iters=1000000):
    """Nesterov-accelerated projected gradient on the simplex (hard/soft max)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient of LSE at y
        b = fast_convolve(y)
        max_b = b.max()
        exp_Tb = np.exp(T * (b - max_b))
        w = exp_Tb / exp_Tb.sum()
        g = 2.0 * fast_cross_corr(w, y)
        
        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / (2.0 * (it + 1) + 1.0)
        # Actually standard Nesterov: beta_k = (1 + sqrt(1 + 4*k^2)) / (2*k + 1)
        # y_{k+1} = x_{k+1} + (beta_{k+1} - 1) * beta_{k+2} / beta_{k+1} * (x_{k+1} - x_k)
        # Simplified: y_{k+1} = x_{k+1} + (t_k - 1)/t_{k+1} * (x_{k+1} - x_k) where t_k = (1+sqrt(1+4k))/2
        
        # Learning rate (Polyak step size)
        lr = 1.0 / (T * (it + 1))
        # Actually, for LSE with temperature T, the function is 1/T-Lipschitz (in some metric)
        # Let's use a decreasing learning rate
        lr = 2.0 / (T * (it + 2))
        
        # Gradient step
        y_trial = y - lr * g
        
        # Project to simplex (sum=1, non-negative)
        # Simple projection: normalize
        y_trial = np.maximum(y_trial, 0)
        st = y_trial.sum()
        if st < 1e-15:
            y_trial = y.copy()
        else:
            y_trial = y_trial / st
        
        # Nesterov update
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * max(0, it - 1))) / 2.0
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * it)) / 2.0
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1))) / 2.0
        momentum = (t_prev - 1.0) / t_next if t_next > 0 else 0.0
        
        a_new = y_trial
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_new.copy()
        else:
            y = y / sy
        
        a_prev = a
        a = a_new
        
        # Track best
        if (it + 1) % 100 == 0:
            score = evaluate_local(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if (it + 1) % 10000 == 0:
            print(f"    NesterovPG iter={it+1}, T={T}: score={best_score:.10f}, elapsed={time.monotonic()-t0:.1f}s")
    
    # Final evaluation
    score = evaluate_local(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_steps=200000):
    """Direct coordinate descent on the non-smoothed objective (hard max)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:min(5, len(b))]
        
        improved = False
        for peak_k in top_peaks:
            # Elements contributing to this peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            # Try reducing top contributors
            order = np.argsort(contribs)[::-1][:8]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-14:
                    continue
                for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.9]:
                    a_trial = a.copy()
                    red = a[pos] * (1.0 - factor)
                    if red < 1e-16:
                        continue
                    a_trial[pos] = a[pos] * factor
                    # Redistribute to others
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + red * others / others_sum
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try simultaneous reduction
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top5 = idx[np.argsort(contribs)[::-1][:5]]
            a_trial = a.copy()
            for pos in top5:
                a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
            score = evaluate_local(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break
        
        if (step + 1) % 50000 == 0:
            print(f"    CD polish step={step+1}: score={best_score:.10f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    elif choice == 7:
        a = (1 - x) ** 0.7 + 0.1
    elif choice == 8:
        # Triangle
        a = 1 - np.abs(2 * x - 1)
    else:
        # Flat-top with soft edges
        a = np.minimum(1.0, 2.0 * np.minimum(x, 1 - x) + 0.3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
FFT-accelerated minimization of 2*n*max(conv(a,a))/(sum(a))^2 using:
- LogSumExp smoothing with multi-stage temperature cooling via L-BFGS-B
- Nesterov-accelerated projected gradient on the simplex
- Aggressive coordinate descent polish on the hard-max objective

- Alternative length exploration
All gradients computed in O(n log n) via FFT-based cross-correlation.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute cross-correlation: result[i] = sum_m w[i+m] * a[m] for valid indices."""
    n = len(a)
    w_flip = w[::-1]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    WF = rfft(w_flip, fft_size)
    AF = rfft(a, fft_size)
    conv_result = irfft(WF * AF, fft_size)[:size]
    return conv_result[n - 1::-1]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_obj_and_grad(z, n, T):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. z (log-parameter)."""
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s
    b = fast_convolve(a)
    max_b = b.max()
    exp_Tb = np.exp(T * (b - max_b))
    lse = max_b + np.log(exp_Tb.sum()) / T
    w = exp_Tb / exp_Tb.sum()
    g_a = 2.0 * n * fast_cross_corr(w, a)
    dot_ag = np.dot(a, g_a)
    g_z = a * (g_a - dot_ag)
    obj = 2.0 * n * lse
    return obj, g_z


class LSEOptimizer:
    """Stateful optimizer for L-BFGS callback."""

    def __init__(self, n, T):
        self.n = n
        self.T = T

    def func(self, z):
        obj, _ = lse_obj_and_grad(z, self.n, self.T)
        return obj

    def jac(self, z):
        _, g = lse_obj_and_grad(z, self.n, self.T)
        return g


def lse_lbgfs(a0, time_limit, T_stages=None, maxiter=3000):
    """L-BFGS-B optimization with LogSumExp smoothing and multi-stage T cooling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(a + 1e-300)
    t0 = time.monotonic()

    if T_stages is None:

                    0.018, 0.012, 0.008]

    best_z = z.copy()
    best_score = float('inf')

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:

        remaining = time_limit - (time.monotonic() - t0)

        opt = LSEOptimizer(n, T)

def nesterov_projected_grad(a0, time_limit, T=0.1, max_iters=1000000):
    """Nesterov-accelerated projected gradient on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    t_k = 2.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at y
        b = fast_convolve(y)
        max_b = b.max()
        exp_Tb = np.exp(T * (b - max_b))
        w = exp_Tb / exp_Tb.sum()
        g = 2.0 * n * fast_cross_corr(w, y)

        # Learning rate
        lr = 2.0 / (T * (it + 2.0))

        # Gradient step
        y_trial = y - lr * g

        # Project to simplex
        y_trial = np.maximum(y_trial, 0)
        st = y_trial.sum()
        if st < 1e-15:
            y_trial = y.copy()
        else:
            y_trial = y_trial / st

        # Nesterov momentum update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1.0))) / 2.0
        momentum = (t_k - 1.0) / t_next
        t_k = t_next

        a_prev = a
        a = y_trial
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Track best periodically
        if (it + 1) % 200 == 0:
            score = evaluate_local(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if (it + 1) % 50000 == 0:
            print(f"    NesterovPG iter={it + 1}, T={T}: score={best_score:.10f}, "
                  f"elapsed={time.monotonic() - t0:.1f}s")

    score = evaluate_local(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            if time.monotonic() > deadline - 20:
                break
            if t_each < 12:
                break

            if pi == 0:
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.05 * np.cos(2 * x) + 0.03 * np.cos(5 * x)
            elif pi == 1:
                pert = 0.03 * rng.standard_normal(n_cur)
            elif pi == 2:
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.04 * np.sin(3 * x)
            elif pi == 3:
                pert = np.zeros(n_cur)
                pos = rng.integers(0, n_cur)
                width = max(1, n_cur // 50)
                pert[max(0, pos - width):pos + width] = 0.08
            else:
                a_struct = structured_init(n_cur, rng)
                pert = 0.3 * a_struct - 0.3 * best_seq

            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps < 1e-15:
                continue
            a_pert = a_pert / ps
            T_quick = [8.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1]
            a_res = lse_lbgfs(a_pert, t_each, T_stages=T_quick, maxiter=2500)
            best_seq, best_score = update_best(best_seq, best_score, a_res, f"multi_{pi}")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alt lengths, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 70:
        n_cur = len(best_seq)
        for alt_factor in [1.25, 1.5, 0.8]:
            if time.monotonic() > deadline - 30:
                break
            alt_n = int(n_cur * alt_factor)
            t_alt = min(rem * 0.3, 100.0)
            if t_alt < 20:
                break
            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            T_alt = [15.0, 8.0, 5.0, 3.0, 2.0, 1.3, 0.85, 0.55, 0.38,
                     0.25, 0.17, 0.12, 0.08, 0.05, 0.03, 0.02]
            a_alt = lse_lbgfs(init_alt, t_alt, T_stages=T_alt, maxiter=3500)
            best_seq, best_score = update_best(best_seq, best_score, a_alt, f"alt_n{alt_n}")
            rem = deadline - time.monotonic()

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final polish, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.45, 50.0)
        if t_fin > 8:
            T_fin = [2.0, 1.2, 0.8, 0.55, 0.38, 0.25, 0.17, 0.12, 0.08,
                     0.055, 0.038, 0.026, 0.018, 0.012, 0.008, 0.005]
            a_fin = lse_lbgfs(best_seq, t_fin, T_stages=T_fin, maxiter=6000)
            best_seq, best_score = update_best(best_seq, best_score, a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_cd2, _ = coordinate_descent_polish(best_seq, min(rem * 0.5, 25.0), n_steps=400000)
        best_seq, best_score = update_best(best_seq, best_score, a_cd2, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_ng3, _ = nesterov_projected_grad(best_seq, rem * 0.6, T=0.08, max_iters=200000)
        best_seq, best_score = update_best(best_seq, best_score, a_ng3, "final_nesterov")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
