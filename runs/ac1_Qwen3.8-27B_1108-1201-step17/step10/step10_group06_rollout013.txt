The standard algorithm (from Duchi et al. 2011):

4. Return `max(v - θ, 0)`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.fft import rfft, irfft
import time

# --- Core utilities ---

def fast_convolve_auto(a):
    """Compute full auto-convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve(a, b):
    """Compute full convolution a * b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve_auto(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_corr_sliding(a, w):
    """Compute y[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0..n-1.
    a has length n, w has length >= 2n-1."""
    n = len(a)
    w_short = w[:2*n-1]
    w_rev = w_short[::-1]
    fft_size = 1 << (3*n - 3).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_rev, fft_size)
    full_conv = irfft(A * W, fft_size)[:3*n-2]
    return full_conv[2*n-2 - np.arange(n)]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lpd_continuation(a0, time_limit, p_schedule=None):
    """Smoothed Lp continuation with projected gradient descent."""
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128]
    
    a = np.asarray(a0, dtype=np.float64).copy()
    a = normalize(a)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Optimize f(a) = (sum_k conv[k]^p)^(1/p) via projected gradient
        # with backtracking line search
        eta = 0.01  # initial step size
        a_curr = a.copy()
        
        max_iters = 200
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_convolve_auto(a_curr)
            weights = conv ** (p - 1)
            S = float(np.sum(conv**p))
            if S < 1e-30:
                break
            factor = S ** (1.0/p - 1.0)
            
            # Gradient: grad[i] = 2 * factor * sum_j a[j] * weights[i+j]
            y = cross_corr_sliding(a_curr, weights)
            grad = 2.0 * factor * y
            
            # Projected gradient step with backtracking
            best_obj = float(np.sum(conv**p)) ** (1.0/p)
            for _ in range(20):
                a_trial = simplex_project(a_curr - eta * grad)
                conv_trial = fast_convolve_auto(a_trial)
                obj_trial = float(np.sum(conv_trial**p)) ** (1.0/p)
                if obj_trial < best_obj:
                    a_curr = a_trial
                    break
                eta *= 0.5
            
            # Check score
            sc = compute_score(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
        
        a = a_curr.copy()
        print(f"  Lp p={p}: score={best_score:.8f}")
    
    return best_a, best_score

def lp_fictitious_play(a0, time_limit, verbose=False):
    """LP Fictitious Play for the zero-sum game."""
    import cvxpy as cp
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = normalize(a)

    iters = 0
    a_avg = a.copy()
    no_improve = 0

        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = name
                break
        except Exception:
            continue
    if solver is None:
        solver = 'ECOS'
    
    print(f"  LP solver: {solver}")
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        
        # Solve LP: min M s.t. T @ b <= M, sum(b) = 1, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = []
        # Build constraints: for each lag k, sum_i T[k,i] * b[i] <= M
        # T[k,i] = a[k-i] if 0 <= k-i < n, else 0
        for k in range(2*n - 1):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            coeffs = np.zeros(n)
            for i in range(i_min, i_max + 1):
                coeffs[i] = a[k - i]
            # Only add constraint if coeffs is non-zero
            if np.any(coeffs != 0):
                constraints.append(coeffs @ b_var <= M_var)
        constraints.append(cp.sum(b_var) == 1)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except Exception:
                continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        bs = b_val.sum()
        if bs < 1e-15:
            continue
        b_val = b_val / bs
        M_val = float(M_var.value)
        
        # Update a using best response
        best_t = 0.0
        best_ls = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - t) * a + t * b_val
            cand = normalize(cand)
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = t
        
        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b_val
            a_new = normalize(a_new)
            if best_ls < best_score - 1e-12:
                best_score = best_ls
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            # Use averaged strategy
            alpha = 3.0 / (iters + 3.0)
            a = (1.0 - alpha) * a + alpha * b_val
            a = normalize(a)
            no_improve += 1
        
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        # Check all candidates
        for cand in [a, a_avg, b_val]:
            cand = normalize(cand)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
        
        if verbose and iters % 5 == 0:
            print(f"    FP it={iters}: best={best_score:.8f}")
        
        if no_improve > 50:
            break
    
    return best_a, best_score, iters

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section search."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = normalize(a)
    conv = fast_convolve_auto(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            # Affected convolution values
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            n_aff = k_hi - k_lo + 1
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            quad_idx = (2 * i - k_lo) if has_quad else -1
            
            # Golden section search
            def score_fn(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = max(3.0 * v, 5.0 / n)
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = score_fn(c)
            fd = score_fn(d)
            for _ in range(30):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d; d = c; fd = fc
                    c = hi - gr * (hi - lo); fc = score_fn(c)
                else:
                    lo = c; c = d; fc = fd
                    d = lo + gr * (hi - lo); fd = score_fn(d)
            best_x = (lo + hi) * 0.5
            
            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose:
            print(f"    ECD {label}: best={best_score:.8f}")
    
    return best_a, best_score

def basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping with diverse structured perturbations."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = normalize(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0
    
    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.35, 90.0)
        
        pert_type = count % 6
        perturbed = a.copy()
        
        if pert_type == 0:
            # Gaussian noise
            sigma = 0.01 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Peak-focused reduction
            conv_p = fast_convolve_auto(a)
            peaks = np.where(conv_p > 0.9 * conv_p.max())[0]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 5)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                perturbed[picks] *= (1.0 - 0.1 * rng.random())
        elif pert_type == 2:
            # Frequency domain perturbation
            fft_size = 1 << (n - 1).bit_length()
            A = rfft(a, fft_size)
            A = A * np.exp(1j * rng.standard_normal(len(A)) * 0.1)
            perturbed = irfft(A, fft_size)[:n]
        elif pert_type == 3:
            # Block swap
            blen = rng.integers(max(1, n//10), max(2, n//4))
            i1 = rng.integers(0, max(1, n - blen))
            i2 = rng.integers(0, max(1, n - blen))
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1+blen].copy()
                perturbed[i1:i1+blen] = perturbed[i2:i2+blen]
                perturbed[i2:i2+blen] = temp
        elif pert_type == 4:
            # Antisymmetric perturbation: mix with reversed
            shift = rng.integers(1, max(2, n//6))
            rev = a[::-1]
            shifted_rev = np.roll(rev, shift)
            perturbed = (1 - 0.2) * a + 0.2 * shifted_rev
        else:
            # Sparse perturbation: zero out random elements
            frac = 0.05 + 0.1 * rng.random()
            mask = rng.random(n) > frac
            perturbed = a * mask
        
        perturbed = normalize(perturbed)
        
        a_opt, s_opt = exact_coord_descent(perturbed, t_alloc, label=f"bh_{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: {best_score:.8f}")
    
    return best_a, best_score
