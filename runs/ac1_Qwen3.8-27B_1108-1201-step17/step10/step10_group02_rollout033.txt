def peak_transfer_descent(a, time_limit):
    """
    At each step, find the maximum peak, identify the most contributing
    coordinate pair, and transfer mass from the higher-contributing coordinate
    to a coordinate that would reduce the peak.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_k = int(np.argmax(b))
        max_b = b[max_k]
        
        # Find the coordinate pair (i, j) with i+j = max_k that maximizes a[i]*a[j]
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        pairs = a[idx] * a[max_k - idx]
        top_pair_idx = idx[np.argmax(pairs)]
        top_pair_j = max_k - top_pair_idx
        
        # Find a target coordinate m that would reduce b[max_k]
        # We want a[max_k - m] < a[max_k - top_pair_idx] or a[max_k - m] < a[max_k - top_pair_j]
        source = top_pair_idx if a[top_pair_idx] >= a[top_pair_j] else top_pair_j
        
        # Find best target
        # Moving from source to m changes b[max_k] by 2*delta*(a[max_k-m] - a[max_k-source])
        # We want a[max_k-m] to be small
        target_candidates = np.arange(n)
        target_values = a[np.clip(max_k - target_candidates, 0, n-1)]
        # Prefer targets where a[max_k - m] is small, and a[m] is not too large (to avoid creating new peaks)
        best_target = target_candidates[np.argmin(target_values)]
        
        if a[best_target] >= a[source]:
            # Can't improve, try something else
            continue
        
        # Compute optimal delta
        # We want: 2*delta*(a[max_k-best_target] - a[max_k-source]) + delta^2 * (...) < 0
        # First order: delta < (a[source] - a[best_target]) / 2 roughly
        # Be conservative: transfer a small fraction
        delta = min(a[source], (a[source] - a[best_target]) * 0.1)
        
        if delta < 1e-15:
            continue
        
        a_trial = a.copy()
        a_trial[source] -= delta
        a_trial[best_target] += delta
        a_trial = np.maximum(a_trial, 0)
        a_trial /= a_trial.sum()
        
        new_score = evaluate_local(a_trial)
        if new_score < best_score - 1e-14:
            a = a_trial
            best_score = new_score
            best_a = a_trial.copy()
        else:
            # Try smaller delta
            delta *= 0.5
            if delta > 1e-15:
                a_trial = a.copy()
                a_trial[source] -= delta
                a_trial[best_target] += delta
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                new_score = evaluate_local(a_trial)
                if new_score < best_score - 1e-14:
                    a = a_trial
                    best_score = new_score
                    best_a = a_trial.copy()
                else:
                    # Stall - try a random perturbation
                    if step > 100:
                        rng = np.random.default_rng(step)
                        a = a + rng.standard_normal(n) * 0.001
                        a = np.maximum(a, 0)
                        a /= a.sum()
    
    return best_a, best_score

def fixed_point_correlation(h_weights, n):
    """
    Given kernel h (length 2n-1), solve for a (length n) such that
    cross-correlation of a with h is constant, and sum(a) = 1.
    """
    # Build the linear system M @ a = c * 1, 1^T @ a = 1
    # M[j, i] = h[j+i] for valid j, i
    M = np.zeros((n, n))
    for j in range(n):
        for i in range(min(n, 2*n-1-j)):
            M[j, i] = h[j+i]
    
    # Augmented system: [M | -1; 1^T | 0] @ [a; c] = [0; 1]
    A = np.zeros((n+1, n+1))
    A[:n, :n] = M
    A[:n, n] = -1.0
    A[n, :n] = 1.0
    A[n, n] = 0.0
    rhs = np.zeros(n+1)
    rhs[n] = 1.0
    
    try:
        sol = np.linalg.solve(A, rhs)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
    
    a = sol[:n]
    c = sol[n]
    return a, c

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_found = None

    _lp_solver_found = cp.ECOS
    return _lp_solver_found


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:

    M_var = cp.Variable()

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):

        pass

                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()

    return b_val, float(M_var.value)


def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at given active peak positions."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= max(1, len(active_indices))
    return g


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    if t0_override is not None:
        t0 = t0_override
    else:
        t0 = time.monotonic()

    prev_g = None

    while time.monotonic() - t0 < time_limit:

        else:
            active_frac = 0.999

        max_b = b.max()
        threshold = max_b * active_frac

        g = compute_subgradient(y, active)

        if bb_adapt and total_iters >= 3 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)

def logsumexp_obj_and_grad(z, T):
    """Compute LogSumExp smoothed objective and gradient w.r.t. z."""
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)
    # Log-sum-exp of b
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()
    logsumexp = max_b + np.log(exp_b.sum())

    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    obj = (2.0 * n) * logsumexp

    # Convert to z-gradient (softmax parameterization)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z

        stage_start = time.monotonic()
        stage_limit = min(stage_time * 1.2, time_limit - (time.monotonic() - t0))
        if stage_limit < 2:
            break

        def func(z_inner):
            obj, _ = logsumexp_obj_and_grad(z_inner, T)
            return obj

        def jac(z_inner):
            _, grad = logsumexp_obj_and_grad(z_inner, T)
            return grad

        try:
            maxiter = max(10, int(stage_limit * 100))
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxiter * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_safe(z)


def analytical_coord_descent(a0, time_limit, n_steps=200000):
    """
    Coordinate descent using analytical transfer computation.
    For each peak, compute the optimal amount to transfer from the most
    contributing coordinate to the best target coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        max_k = int(np.argmax(b))

        # Identify top peaks
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:5]:  # Try top 5 peaks
            if improved:
                break

            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]

            # Source: coordinate with highest contribution to this peak
            source = idx[np.argmax(contribs)]

            # Find best target: minimize a[peak_k - target] while being smaller than source
            valid_targets = np.arange(n)
            # a[peak_k - target] should be small
            target_keys = np.clip(peak_k - valid_targets, 0, n - 1)
            target_values = a[target_keys] + 0.01 * a[valid_targets]  # penalty for large targets

            # Exclude source
            target_values[source] = np.inf
            best_target = int(np.argmin(target_values))

            if a[best_target] >= a[source] * 0.95:
                continue

            # Compute transfer amount analytically
            # Moving delta from source to target changes b[peak_k] by:
            # 2*delta*(a[peak_k-target] - a[peak_k-source]) approximately
            # We want this to be negative, and delta <= a[source]
            diff = a[np.clip(peak_k - source, 0, n-1)] - a[np.clip(peak_k - best_target, 0, n-1)]
            if diff <= 0:
                continue

            # Try multiple delta fractions
            for frac in [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]:
                delta = a[source] * frac
                if delta < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[source] -= delta
                a_trial[best_target] += delta
                a_trial = np.maximum(a_trial, 0)
                a_trial_sum = a_trial.sum()
                if a_trial_sum < 1e-15:
                    break
                a_trial /= a_trial_sum
                score = evaluate_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Multi-coordinate perturbation
                a_trial = a.copy()
                peak_k = int(np.argmax(fast_convolve(a_trial)))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a_trial[idx] * a_trial[peak_k - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:5]]
                for pos in top_pos:
                    if a_trial[pos] > 1e-10:
                        a_trial[pos] *= 0.9995
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                elif stall > 800:
                    # Random kick
                    a = a + rng.standard_normal(n) * 0.002
                    a = np.maximum(a, 0)
                    a /= a.sum()
                stall = 0
        else:
            stall = 0

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence with diversity."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * rng.integers(1, 5) * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2 * (x - 0.5) ** 2 / (0.1 + 0.2 * rng.random()))
    elif choice == 5:
        a = 1.0 + 0.5 * np.sin(np.pi * x + rng.random() * np.pi)
        a = np.maximum(a, 0)
    elif choice == 6:
        # Two-peak structure
        c1, c2 = sorted(rng.random(2))
        a = np.exp(-10*(x-c1)**2) + np.exp(-10*(x-c2)**2)
    else:
        a = np.ones(n) * (0.5 + 0.5 * np.random.random(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def dual_fixed_point(a0, time_limit, n_iters=5000):
    """
    Dual fixed-point iteration: solve linear system for given active set weights,
    then update weights based on current autocorrelation peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Compute weights: softmax over b with temperature that tightens
        T = max(0.01, max_b * 0.01 * (1.0 - it / n_iters))
        b_shifted = b - max_b
        w = np.exp(b_shifted / T)
        w = w / w.sum()

        # Build linear system: for each j, sum_k w[k] * a[k-j] = c
        # M[j, i] = w[j+i] (where j+i < 2n-1)
        M = np.zeros((n, n))
        for i in range(n):
            valid_k = np.arange(i, min(i + n, 2 * n - 1))
            M[:len(valid_k), i] = w[valid_k]

        # Solve [M | -1; 1^T | 0] [a; c] = [0; 1]
        A = np.zeros((n + 1, n + 1))
        A[:n, :n] = M
        A[:n, n] = -1.0
        A[n, :n] = 1.0
        rhs = np.zeros(n + 1)
        rhs[n] = 1.0

        try:
            sol = np.linalg.solve(A, rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
            except Exception:
                continue

        a_new = sol[:n]

        # Handle negative entries: project and iterate
        n_iter_inner = 0
        while np.any(a_new < -1e-10) and n_iter_inner < 10:
            mask = a_new > 0.01 * max(1.0, a_new.max())
            n_active = mask.sum()
            if n_active < 2:
                break
            # Solve on active set only
            A_sub = np.zeros((n_active + 1, n_active + 1))
            cols = np.where(mask)[0]
            for row_i in range(n_active):
                j = cols[row_i]
                for col_i in range(n_active):
                    i = cols[col_i]
                    k_start = j
                    k_end = min(j + n, 2 * n - 1)
                    if k_start <= i < k_end:
                        A_sub[row_i, col_i] = w[j + i]
                A_sub[row_i, n_active] = -1.0
            A_sub[n_active, :n_active] = 1.0
            rhs_sub = np.zeros(n_active + 1)
            rhs_sub[n_active] = 1.0
            try:
                sol_sub = np.linalg.solve(A_sub, rhs_sub)
            except np.linalg.LinAlgError:
                break
            a_new = np.zeros(n)
            a_new[cols] = sol_sub[:n_active]
            n_iter_inner += 1

        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn

        # Damping
        alpha = 0.5 + 0.5 * (it / max(1, n_iters))  # Increase mixing over time
        a_trial = (1 - alpha) * a + alpha * a_new
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()

        score = evaluate_local(a_trial)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        elif it % 100 == 0:
            # Occasionally accept non-improving steps to explore
            if rng.random() < 0.1:
                a = a_trial

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines mirror descent, LP alternating, LogSumExp L-BFGS, dual fixed-point,
    analytical coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Total budget: {budget_s:.0f}s, deadline in {budget_s-20:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Multi-restart Mirror Descent (diversification) ===
    print(f"\n=== Phase 1: Multi-restart MD (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.15, 200.0)
    if t_phase1 > 30:
        n_restarts = 4
        t_each = t_phase1 / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            if r == 0:
                a_start = best_seq.copy()
            else:
                a_start = best_seq + rng.standard_normal(n_best) * 0.02
                a_start = np.maximum(a_start, 0)
                a_start = a_start / a_start.sum()
            print(f"  Restart {r+1}/{n_restarts}, budget={t_each:.0f}s")
            a_md, _ = mirror_descent_optimize(a_start, t_each, eta0=0.07,
                                              nesterov=True, bb_adapt=True,
                                              adaptive_active=True,
                                              t0_override=time.monotonic())
            try_update(a_md, f"md_restart_{r}")

    # === Phase 2: LP Alternating (global structure) ===
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.30, 500.0)
    if t_phase2 > 30:
        print(f"  Budget: {t_phase2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_phase2, solver=solver, n_max_iters=50000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 3: Dual Fixed-Point (new method) ===
    print(f"\n=== Phase 3: Dual Fixed-Point (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.15, 250.0)
    if t_phase3 > 15:
        print(f"  Budget: {t_phase3:.0f}s")
        a_fp, _ = dual_fixed_point(best_seq, t_phase3, n_iters=10000)
        try_update(a_fp, "dual_fp")

    # === Phase 4: LogSumExp L-BFGS (smooth, precise) ===
    print(f"\n=== Phase 4: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase4 = min(rem * 0.20, 300.0)
    if t_phase4 > 15:
        T_stages = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        print(f"  T_stages: {len(T_stages)} stages, budget={t_phase4:.0f}s")
        a_lb = logsumexp_lbgfs(best_seq, t_phase4, T_stages=T_stages, maxiter_per_stage=5000)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Mirror Descent refinement (low eta) ===
    print(f"\n=== Phase 5: Refine MD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase5 = min(rem * 0.12, 180.0)
    if t_phase5 > 10:
        a_md2, _ = mirror_descent_optimize(best_seq, t_phase5, eta0=0.02,
                                           nesterov=True, bb_adapt=True,
                                           adaptive_active=True,
                                           t0_override=time.monotonic())
        try_update(a_md2, "md_refine")

    # === Phase 6: Analytical Coordinate Descent ===
    print(f"\n=== Phase 6: Analytical Coord Desc (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase6 = min(rem * 0.10, 150.0)
    if t_phase6 > 10:
        a_cd, _ = analytical_coord_descent(best_seq, t_phase6, n_steps=100000)
        try_update(a_cd, "coord_desc")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase7 = min(rem * 0.08, 100.0)
    if t_phase7 > 20:
        n_candidates = [1000, 1400, 1600, 2000]
        n_candidates = [x for x in n_candidates if x != n_best]
        t_each = t_phase7 / max(1, len(n_candidates))
        for n_try in n_candidates:
            if time.monotonic() > deadline - 10:
                break
            if t_each < 8:
                break
            print(f"  Trying n={n_try}, budget={t_each:.0f}s")
            a_interp = interpolate_sequence(best_seq, n_try)
            a_m, _ = mirror_descent_optimize(a_interp, t_each * 0.7, eta0=0.05,
                                             nesterov=True, bb_adapt=True,
                                             adaptive_active=True,
                                             t0_override=time.monotonic())
            try_update(a_m, f"n={n_try}")

    # === Phase 8: Final polish (very fine) ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase8 = min(rem * 0.10, 100.0)
    if t_phase8 > 10:
        # Quick coord descent
        t_cd = t_phase8 * 0.5
        a_polish, _ = analytical_coord_descent(best_seq, t_cd, n_steps=50000)
        try_update(a_polish, "polish_cd")
        # Quick L-BFGS at very low T
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            T_fin = [5.0, 2.0, 1.0, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01]
            a_fin = logsumexp_lbgfs(best_seq, min(rem2 - 5, t_phase8 * 0.5),
                                    T_stages=T_fin, maxiter_per_stage=2000)
            try_update(a_fin, "polish_lbgfs")

    # Final: one more MD pass with whatever time is left
    rem = deadline - time.monotonic()
    if rem > 5:
        a_final, _ = mirror_descent_optimize(best_seq, rem - 2, eta0=0.01,
                                             nesterov=True, bb_adapt=True,
                                             adaptive_active=True,
                                             t0_override=time.monotonic())
        try_update(a_final, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
