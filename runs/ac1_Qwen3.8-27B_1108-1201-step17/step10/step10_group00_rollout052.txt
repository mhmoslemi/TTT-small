    """Approximate best response using active set heuristic."""
    n = len(a)
    b_conv = fast_convolve(a)  # This is a*a, but we need a*b
    # Actually we need conv(a, b) for various b
    # Simple approach: b[j] should be small where a has large "footprint"
    # a's contribution to position k through b[j] is a[k-j]
    # We want to minimize max_k sum_j a[k-j]*b[j]
    # So b[j] should be small when a has large values at positions k-j where k is a peak
    
    # Heuristic: b[j] ∝ 1 / (1 + sum_{k in active} a[k-j])
    active = np.argsort(b_conv)[-n_active:]  # top n_active peaks
    denom = np.zeros(n)

        je = min(n-1, k)
        idx = np.arange(js, je+1)
        denom[idx] += a[k-idx]
    b = 1.0 / (1.0 + denom)
    b = np.maximum(b, 0)
    s = b.sum()
    if s > 1e-15:

    return b

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Compute convolve(a, b) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def get_lp_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except:
            continue
    return cp.ECOS

def solve_reduced_lp(a, K=300, solver=None, warm_start=None):
    n = len(a)
    if K > 2*n - 1:
        K = 2*n - 1
    
    # Find active constraints using a*a peaks

    top_idx = np.argsort(aa)[-K:]
    
    b_var = cp.Variable(n, nonneg=True)

    M_var = cp.Variable()
    constraints = [cp.sum(b_var) == 1]
    
    for k in top_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        constraints.append(a[k - idx] @ b_var[idx] <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]

        try:

                sv = b_val.sum()
                if sv > 1e-15:

"""
Hybrid optimization for minimum peak autocorrelation step functions.
Combines: (1) Active-set LP alternating with Nesterov momentum, (2) Subgradient 
descent with adaptive step, (3) Simulated annealing perturbations, (4) Coordinate
descent polish. Multi-length exploration with adaptive time allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-correlation convolve(a,b) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation of the score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def solve_reduced_lp(a, K=300, solver=None, warm_start=None):
    """Solve best-response LP with only top-K active constraints for speed."""
    n = len(a)
    if K > 2 * n - 1:
        K = 2 * n - 1

    aa = fast_convolve(a)
    top_idx = np.argsort(aa)[-K:]

    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start

    M_var = cp.Variable()
    constraints = [cp.sum(b_var) == 1]

    for k in top_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        constraints.append(a[k - idx] @ b_var[idx] <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    seen = set()
    for s in solvers:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                sv = b_val.sum()
                if sv > 1e-15:
                    return b_val / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def subgradient_step(a, eta, n_active=20):
    """One subgradient descent step on max_k (a*a)[k]."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b <= 0:
        return a.copy()

    threshold = max_b * (1.0 - 1e-4)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    if len(active) > n_active:
        active = active[np.argsort(b[active])[-n_active:]]

    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active)

    a_new = a - eta * g
    return np.maximum(a_new, 0)

        if sy < 1e-15:

            sy = 1.0

def coordinate_descent_polish(a0, time_budget, n_sweeps=100):
    """Cyclic coordinate descent: optimize each coordinate one at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    # Precompute the convolution structure
    # For coordinate i, (a*a)[k] changes only for k in [i, i+n-1] (intersected with [0, 2n-2])
    # d(a*a)[k]/d(a[i]) = 2*a[k-i] for max(0,k-n+1) <= i <= min(n-1,k)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_budget:
            break

        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                    break

            # Current contribution: (a*a)[k] for relevant k
            # If we change a[i] from v to v', the change in (a*a)[k] is:
            #   2 * a[k-i] * (v' - v) for k where k-i is in [0, n-1] and i is in [max(0,k-n+1), min(n-1,k)]
            # i.e., k in [i, i+n-1], and k-i in [0, n-1] is automatic
            # So affected k: k from i to min(2n-2, i+n-1)

            k_start = i
            k_end = min(2 * n - 2, i + n - 1)

            # Current (a*a) values at affected positions
            aa = fast_convolve(a)
            current_max = aa.max()

            # The new (a*a)[k] = old_aa[k] + 2 * a[k-i] * (v_new - v_old)
            # We want to minimize max_k new_aa[k]
            # This is a 1D problem: find v_new >= 0 minimizing max_k (c_k + d_k * v_new)
            # where c_k = old_aa[k] - 2*a[k-i]*a[i], d_k = 2*a[k-i]
            # (since old_aa[k] includes the term 2*a[i]*a[k-i])

            v = a[i]
            d = 2.0 * a[k_start - i:k_end - i + 1]  # d_k = 2*a[k-i] for k in range
            c = aa[k_start:k_end + 1] - d * v  # c_k without the v contribution

            # Minimize max(c + d*v) over v >= 0
            # The optimal v is where the maximum is minimized
            # This is a 1D convex (piecewise linear) problem
            # Use bisection on the subgradient

            # Widget: the objective is max_k (c[k] + d[k]*v)
            # At v=0: obj = max(c)
            # As v increases, terms with large d grow faster
            # Optimal v: find where the "most active" constraints balance

            # Simple approach: try v=0 and v that equalizes top constraints
            best_v = 0.0
            best_obj = c.max()  # at v=0

            # Try a few candidate v values
            if d.max() > 1e-15:
                # The optimal v satisfies: for active set S, c[k] + d[k]*v = M for k in S
                # Try: v such that the top-2 constraints are equal
                # Or just use a small grid
                v_max = max(0, (best_obj - c.min()) / max(d.max(), 1e-15))
                for v_try in [0.0, 0.5 * v, 0.3 * v, 0.7 * v, 0.1 * v]:
                    if v_try < 0:
                        continue
                    obj = (c + d * v_try).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_v = v_try

                # Also try: v that makes the max constraint equal to the second max
                # Sort by d descending
                if d.max() > 1e-15:
                    v_candidates = []
                    # For pairs of active constraints
                    top_d_idx = np.argsort(d)[-5:]
                    top_c_idx = np.argsort(c)[-5:]
                    for i1 in top_d_idx:
                        for i2 in top_c_idx:
                            if i1 != i2 and d[i1] != d[i2]:
                                v_pair = (c[i2] - c[i1]) / (d[i1] - d[i2])
                                if v_pair >= 0 and v_pair < 10 * v:
                                    v_candidates.append(v_pair)
                    for v_cand in v_candidates:
                        obj = (c + d * v_cand).max()
                        if obj < best_obj:
                            best_obj = obj
                            best_v = v_cand

            a[i] = max(0.0, best_v)

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s

        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if sweep % 20 == 0 and sweep > 0:

            print(f"    CoordDescent sweep={sweep} ({elapsed:.0f}s): score={best_score:.8f}")

def simulated_annealing_phase(a0, time_budget, rng):

    n = len(a0)

    t0 = time.monotonic()

    T = 0.05  # Initial temperature (relative to score scale)
    T_min = 0.001
    cooling = 0.999

    for step in range(500000):
        if time.monotonic() - t0 > time_budget:
            break

        a_new[idx] = np.maximum(a_new[idx] + perturb, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        score_new = evaluate_local(a_new)

        delta = score_new - best_score  # Compare against best
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            if score_new < best_score:

        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1 * x
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.4 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Asymmetric: concentrated on one side
        a = np.exp(-5 * x) + 0.1
    else:
        # Two-scale structure
        a = 0.6 * np.exp(-2 * (x - 0.25) ** 2) + 0.4 * np.exp(-2 * (x - 0.75) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version with low-frequency structure."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 7)
    freq2 = rng.integers(1, 7)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.cos(freq2 * x + p2))
    a_pert = a + pert
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP alternating, simulated annealing, coordinate descent,
    and multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick multi-length exploration (90s) ===
    print(f"\n=== Phase 1: Multi-length exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.05, 90.0)
    if t_phase1 > 20:
        n_candidates = [n_best]
        if n_best != 1000:
            n_candidates.append(1000)
        if n_best != 1500:
            n_candidates.append(1500)
        if n_best != 2000:
            n_candidates.append(2000)
        if n_best != 800:
            n_candidates.append(800)

        per_n = t_phase1 / len(n_candidates)
        for n_try in n_candidates:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 10:
                break
            t_sub = min(per_n, rem_sub * 0.3)
            if t_sub < 8:
                continue
            if n_try == n_best:
                init = best_seq.copy()
            else:
                init = interpolate_sequence(best_seq, n_try)
            # Quick LP optimization
            a_try, _ = nesterov_lp_main(init, t_sub, solver, n_max_iters=5000, verbose=False)
            local_update(a_try, f"explore_n{n_try}")

    # === Phase 2: Main Nesterov LP (700s) ===
    print(f"\n=== Phase 2: Main Nesterov LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.40, 700.0)
    if t_phase2 > 60:
        print(f"  Budget: {t_phase2:.0f}s")
        a_lp, _ = nesterov_lp_main(best_seq, t_phase2, solver, n_max_iters=1000000, verbose=True)
        local_update(a_lp, "main_lp")

    # === Phase 3: Simulated annealing (200s) ===
    print(f"\n=== Phase 3: Simulated Annealing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.12, 200.0)
    if t_phase3 > 30:
        print(f"  Budget: {t_phase3:.0f}s")
        a_sa, _ = simulated_annealing_phase(best_seq, t_phase3, rng)
        local_update(a_sa, "sim_anneal")

    # === Phase 4: Second LP round from updated best (350s) ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase4 = min(rem * 0.20, 350.0)
    if t_phase4 > 30:
        # Use a perturbed version for diversity
        a_pert = perturbed_sequence(best_seq, rng, strength=0.015)
        print(f"  Budget: {t_phase4:.0f}s (perturbed start)")
        a_lp2, _ = nesterov_lp_main(a_pert, t_phase4, solver, n_max_iters=500000, verbose=True)
        local_update(a_lp2, "lp_round2")

    # === Phase 5: Coordinate descent polish (200s) ===
    print(f"\n=== Phase 5: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase5 = min(rem * 0.12, 200.0)
    if t_phase5 > 20:
        print(f"  Budget: {t_phase5:.0f}s")
        a_cd, _ = coordinate_descent_polish(best_seq, t_phase5, n_sweeps=200)
        local_update(a_cd, "coord_descent")

    # === Phase 6: Final LP + subgradient polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 120.0)
        if t_fin > 10:
            a_fin, _ = nesterov_lp_main(best_seq, t_fin, solver, n_max_iters=200000, verbose=False)
            local_update(a_fin, "final_lp")

        rem = deadline - time.monotonic()
        if rem > 15:
            # Final subgradient with very small steps
            a = best_seq.copy()
            n = len(a)
            for step in range(5000):
                if time.monotonic() > deadline - 5:
                    break
                eta = 0.005 / (1 + 0.001 * step)
                a_new = subgradient_step(a, eta, n_active=50)
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_new.copy()
                    a = a_new
            print(f"  Subgradient polish: score={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
