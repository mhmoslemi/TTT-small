            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            # Golden section search for optimal a[i]
            def score_func(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = upper
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = score_func(c_pt)
            fd = score_func(d_pt)
            for _ in range(30):

                if fc <= fd:

                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = score_func(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = score_func(d_pt)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 10 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weights descent (O(n log n) per iteration via FFT cross-correlation)
2. DFT-based spectral shaping for coarse optimization
3. Exact coordinate descent with golden section search for polishing
4. Basin hopping with structured perturbations
5. Multi-length exploration and adaptive temperature scheduling
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)


def cross_correlate(w, a):
    n_a = len(a)
    n_w = len(w)
    N = 1 << (n_w + n_a - 2).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    result = irfft(W * np.conj(A), N)
    return result[:n_a]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def multiplicative_weights_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    tau = 0.05
    eta = 5.0
    tau_min = 1e-5
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - t0 < time_limit:
        iters += 1
        c = fast_convolve(a)
        max_c = float(c.max())
        mean_c = float(c.mean())
        spread = max_c - mean_c
        if spread < 1e-15:
            break

        tau_eff = max(tau * spread, tau_min)
        c_shifted = c - max_c
        w = np.exp(c_shifted / tau_eff)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = 2.0 * cross_correlate(w, a)
        grad = np.maximum(grad, 0.0)

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            break
        a_new = a_new / a_new_sum

        score = 2.0 * n * float(fast_convolve(a_new).max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        tau *= 0.9998

        if no_improve > max_no_improve:
            eta *= 0.5
            no_improve = 0
        if no_improve > max_no_improve * 3:
            tau = 0.03 * (max_c - mean_c)
            no_improve = 0
        if no_improve > max_no_improve * 6:
            break

        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWD {label} it={iters}, t={elapsed:.1f}s, score={best_score:.8f}")

    return best_a, best_score


def dft_spectral_flattening(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    N = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    alpha = 0.3

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, N)
        target_mag = np.ones(N // 2 + 1)
        target_mag[0] = 0.0
        mags = np.abs(A)
        for k in range(1, N // 2 + 1):
            if mags[k] > 1e-30:
                target_mag[k] = 1.0

        mean_mag = np.mean(mags[1:N // 2 + 1])
        if mean_mag < 1e-30:
            break
        target = mean_mag

        for k in range(1, N // 2 + 1):
            if mags[k] > 1e-30:
                correction = (target / mags[k]) ** alpha
                A[k] *= correction
                A[N - k] = np.conj(A[k])

        a_new = irfft(A, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-15:
            break
        a_new = a_new / ns

        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        alpha = max(alpha * 0.999, 0.05)

    return best_a, best_score


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            upper = max(3.0 * v, 5.0 / n)
            lo = 0.0
            hi = upper
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)

            def _sc(x, _cv=c_vals, _ap=a_partners, _qm=unaff_max,
                    _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                    _vv=v, _nn=n):
                ns = _cs + (x - _vv)
                if ns < 0.01:
                    return 1e10
                nv = _cv + 2.0 * x * _ap
                if _hq:
                    nv[ _qi] = _cv[_qi] + x * x
                nm = _qm
                local_max = float(nv.max())
                if local_max > nm:
                    nm = local_max
                return 2.0 * _nn * nm / (ns * ns)

            fc = _sc(c_pt)
            fd = _sc(d_pt)
            for _ in range(35):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = _sc(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = _sc(d_pt)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score


def projected_subgradient_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    lr = 0.02
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        c = fast_convolve(a)
        max_c = float(c.max())
        mean_c = float(c.mean())
        spread = max_c - mean_c
        if spread < 1e-15:
            break

        tau_eff = 0.05 * spread
        c_shifted = c - max_c
        w = np.exp(c_shifted / tau_eff)
        w_sum = w.sum()
        w = w / w_sum

        grad = 2.0 * cross_correlate(w, a)
        grad = np.maximum(grad, 0.0)

        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        if verbose and iters % 2000 == 0:

def basin_hopping(a0, time_limit, rng, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

        a = a / s

    best_a = a.copy()

    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.5, 120.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            conv_p = fast_convolve(a)
            top_k = min(15, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 5)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.08 * rng.random())
        elif pert_type == 1:
            sigma = 0.03 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            blen = max(2, rng.integers(n // 10, n // 4))
            i1 = rng.integers(0, max(1, n - blen))
            i2 = rng.integers(0, max(1, n - blen))
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1 + blen].copy()
                perturbed[i1:i1 + blen] = perturbed[i2:i2 + blen]
                perturbed[i2:i2 + blen] = temp
        elif pert_type == 3:
            shift = rng.integers(1, max(2, n // 5))
            shifted = np.roll(a, int(shift))
            perturbed = 0.7 * a + 0.3 * shifted
        else:
            mask = rng.random(n) < 0.3
            perturbed[mask] *= rng.uniform(0.3, 2.0, mask.sum())

        perturbed = np.maximum(perturbed, 0.0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt = multiplicative_weights_descent(perturbed, t_alloc * 0.6)

            best_a = a_opt.copy()
            a = a_opt.copy()
            if verbose:

def init_asymmetric(n, rng, shift=0.3, width_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = np.ones(n_freq)
    a = irfft(amps * np.exp(1j * phases), n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine_modulated(n, rng):
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    envelope = np.exp(-0.5 * ((t - n / 2) / (n * 0.3)) ** 2)
    a = envelope * (1.0 + 0.3 * np.cos(2 * np.pi * freq * t / n + phase))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    print(f"\n=== Phase 3: DFT Shaping (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t_dft = min(rem * 0.3, 150.0)
        a_dft, s_dft = dft_spectral_flattening(best_seq, t_dft, verbose=True, label="main")
        if s_dft < best_score:
            best_score = s_dft
            best_seq = a_dft.copy()
            print(f"  NEW BEST DFT: {best_score:.8f}")

    # Phase 4: Coordinate descent polishing
    print(f"\n=== Phase 4: ECD Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(4):
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        t_cd = min(rem * 0.2, 150.0)
        print(f"  ECD round {rnd + 1}, budget={t_cd:.0f}s")
        a_cd, s_cd = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph4_r{rnd}")
        if s_cd < best_score:
            best_score = s_cd
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD: {best_score:.8f}")
        elif rnd > 0:
            perturbed = best_seq + rng.standard_normal(n_work) * 0.015
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_cd2, s_cd2 = exact_coord_descent(perturbed, min(t_cd * 0.5, 60.0))
            if s_cd2 < best_score:
                best_score = s_cd2
                best_seq = a_cd2.copy()
                print(f"  NEW BEST ECD-pert: {best_score:.8f}")

    # Phase 5: Basin hopping
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_bh = min(rem * 0.7, 350.0)
        print(f"  BH budget={t_bh:.0f}s")
        a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, verbose=True, label="ph5")
        if s_bh < best_score:
            best_score = s_bh
            best_seq = a_bh.copy()
            print(f"  NEW BEST BH: {best_score:.8f}")

    # Phase 6: Try different lengths from best shape
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        n_cur = len(best_seq)
        for alt_n in [n_cur + 64, n_cur - 64, n_cur + 128, n_cur // 2, n_cur + 256]:
            if alt_n < 256 or alt_n > 3072:
                continue
            rem2 = deadline - time.monotonic() - 120
            if rem2 < 20:
                break
            a_init = interpolate_seq(best_seq, alt_n)
            t_alt = min(rem2 * 0.3, 80.0)
            a_alt, s_alt = multiplicative_weights_descent(a_init, t_alt)
            if s_alt < best_score:
                best_score = s_alt
                best_seq = a_alt.copy()
                print(f"  NEW BEST alt_n={alt_n}: {best_score:.8f}")
                # Quick polish
                rem3 = deadline - time.monotonic() - 100
                if rem3 > 10:
                    a_pol, s_pol = exact_coord_descent(best_seq, min(rem3 * 0.4, 40.0))
                    if s_pol < best_score:
                        best_score = s_pol
                        best_seq = a_pol.copy()

    # Phase 7: Final MWD + ECD
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_final_mwd = min(rem * 0.4, 60.0)
        a_fm, s_fm = multiplicative_weights_descent(best_seq, t_final_mwd)
        if s_fm < best_score:
            best_score = s_fm
            best_seq = a_fm.copy()

    rem = deadline - time.monotonic() - 15
    if rem > 8:
        t_final_cd = min(rem * 0.6, 50.0)
        a_fc, s_fc = exact_coord_descent(best_seq, t_final_cd, verbose=True, label="final")
        if s_fc < best_score:
            best_score = s_fc
            best_seq = a_fc.copy()

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
