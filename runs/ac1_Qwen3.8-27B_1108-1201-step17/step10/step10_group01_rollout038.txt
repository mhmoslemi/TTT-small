def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_matrix_vec(T_func, a, b):

def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)

    B = rfft(b, fft_size)

            no_improve += 1

        if no_improve >= 30:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-8, lr)

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f} |active|={len(active)}")

    return best_a, best_score


def peak_coord_descent(a0, time_budget, n_steps=50000):
    """Targeted coordinate descent reducing top convolution peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve_auto(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(5, len(b))]

        improved = False
        for peak_k in top_peaks[:3]:
            pk = int(peak_k)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            order = np.argsort(contribs)[::-1][:3]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-13:
                    continue
                for factor in [0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        break
                    # Redistribute to random positions
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_idx = np.where(mask)[0]
                    # Put redistributed mass where it contributes least to this peak
                    contributions = np.zeros(len(others_idx))
                    for ii, oi2 in enumerate(others_idx):
                        if 0 <= pk - oi2 < n:
                            contributions[ii] = a[pk - oi2]
                    # Pick positions with low contribution
                    low_idx = np.argsort(contributions)[:max(1, len(others_idx) // 10)]
                    a_trial[others_idx[low_idx]] += diff / len(low_idx)
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        break
                    a_trial = a_trial / st
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a diverse structured initialization."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-5 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        a = (1 - x) ** 1.2 + 0.05
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def spectral_perturb(a, rng, strength=0.03):
    """Perturb sequence in frequency domain for diversity."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    # Perturb magnitude
    phase = np.angle(A)
    mag = np.abs(A)
    noise = 1.0 + strength * rng.standard_normal(len(mag))
    new_mag = np.abs(mag * noise)
    new_A = new_mag * np.exp(1j * phase)
    a_new = irfft(new_A, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses smoothed alternating minimization (no LP), Nesterov acceleration,
    subgradient polishing, peak coordinate descent, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Temperature schedules (from smooth to sharp)
    T_schedule_full = [15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.3, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.06, 0.04, 0.03, 0.02]
    T_schedule_med = [10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.3, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12]
    T_schedule_short = [5.0, 3.5, 2.5, 1.8, 1.3, 1.0, 0.7, 0.5, 0.35]

    # === Phase 1: Main Nesterov SM (primary, longest) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Nesterov SM n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 720.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s, T_schedule has {len(T_schedule_full)} stages")
        iters_per_T = max(50, int(t1 / len(T_schedule_full) / 0.08))
        a_main, _ = nesterov_smoothed_alt(best_seq, T_schedule_full, iters_per_T, t1,
                                           inner_iters=35, seed=seed)
        try_update(a_main, "nesterov_main")

    # === Phase 2: Subgradient polish ===
    print(f"\n=== Phase 2: Subgradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 180.0)
    if t2 > 15:
        a_sg, _ = subgradient_polish(best_seq, t2, n_steps=30000)
        try_update(a_sg, "subgrad")

    # === Phase 3: Peak coordinate descent ===
    print(f"\n=== Phase 3: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 120.0)
    if t3 > 10:
        a_cd, _ = peak_coord_descent(best_seq, t3, n_steps=40000)
        try_update(a_cd, "peak_cd")

    # === Phase 4: Second Nesterov pass (refined) ===
    print(f"\n=== Phase 4: Nesterov Pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 180.0)
    if t4 > 30:
        a_n2, _ = nesterov_smoothed_alt(best_seq, T_schedule_med, max(30, int(t4 / len(T_schedule_med) / 0.08)),
                                         t4, inner_iters=30, seed=seed + 1)
        try_update(a_n2, "nesterov_pass2")

    # === Phase 5: Spectral perturbation + Nesterov ===
    print(f"\n=== Phase 5: Spectral perturbation, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_each = min(rem / 4.0, 60.0)
        for pi in range(3):
            if time.monotonic() > deadline - 30:
                break
            if t_each < 20:
                break
            a_pert = spectral_perturb(best_seq, rng, strength=0.04 * (pi + 1))
            a_res, _ = nesterov_smoothed_alt(a_pert, T_schedule_short, max(20, int(t_each / len(T_schedule_short) / 0.08)),
                                              t_each, inner_iters=25, seed=seed + 10 + pi)
            try_update(a_res, f"spectral_{pi}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5]:
            alt_n = int(n_cur * factor)
            if 200 <= alt_n <= 5000:
                alt_lengths.append(alt_n)
        # Deduplicate
        alt_lengths = sorted(set(alt_lengths))

        t_each = min(rem / max(1, len(alt_lengths)), 80.0)
        for alen in alt_lengths[:3]:
            if time.monotonic() > deadline - 30:
                break
            if t_each < 25:
                break
            print(f"  Trying n={alen}, t={t_each:.0f}s")
            init_alt = interpolate_sequence(best_seq, alen)
            a_alt, _ = nesterov_smoothed_alt(init_alt, T_schedule_med, max(20, int(t_each / len(T_schedule_med) / 0.08)),
                                              t_each, inner_iters=30, seed=seed + 100)
            try_update(a_alt, f"n{alen}")
            # Quick subgradient on the new best
            rem2 = deadline - time.monotonic()
            if rem2 > 20 and best_seq is not None and len(best_seq) == alen:
                a_sg2, _ = subgradient_polish(best_seq, min(rem2 * 0.3, 30.0), n_steps=5000)
                try_update(a_sg2, f"sg_n{alen}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.5, 40.0)
        if t_fin > 10:
            a_fin, _ = nesterov_smoothed_alt(best_seq, [0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02],
                                              max(30, int(t_fin / 0.05)), t_fin, inner_iters=40, seed=seed + 200)
            try_update(a_fin, "final_nesterov")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_cd2, _ = peak_coord_descent(best_seq, min(rem * 0.6, 25.0), n_steps=20000)
        try_update(a_cd2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_sg3, _ = subgradient_polish(best_seq, min(rem * 0.7, 15.0), n_steps=10000)
        try_update(a_sg3, "final_sg")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("ERROR: No valid sequence found!")
        best_seq = [1.0]
        best_score = float('inf')

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
