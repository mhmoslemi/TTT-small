"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
(1) LP alternation for cross-convolution Nash equilibrium,
(2) Direct auto-convolution optimization via LSE-smoothed L-BFGS,
(3) Active-set KKT refinement,
(4) Multi-n exploration with interpolation,
(5) Subgradient polishing with adaptive step sizes.
Key innovation: after LP alternation, directly optimize the AUTO-convolution
objective (not the cross-convolution), which addresses the gap between the
Nash equilibrium value and the true objective.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective (normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(fixed, solver):
    """Given fixed vector (sum=1), find x >= 0, sum(x)=1 minimizing max_k conv(fixed, x)[k]."""
    n = len(fixed)
    x_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(fixed)
    constraints = [T @ x_var <= M_var, cp.sum(x_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver]
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt not in solvers_to_try:
            solvers_to_try.append(alt)
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.asarray(x_var.value, dtype=np.float64)
                x_val = np.maximum(x_val, 0)
                sv = x_val.sum()
                if sv > 1e-15:
                    return x_val / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_alternation(a0, time_limit, solver, max_iters=500000, verbose=True):
    """LP alternation with Nesterov acceleration for cross-convolution game."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Best response for b given y_a
        b_new, M1 = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        # Best response for a given y_b
        a_new, M2 = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        # Track best (evaluate auto-convolution, not cross-convolution!)
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 8:
                noise_scale = 0.01 / math.sqrt(stall)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall = 0

            if verbose and (it + 1) % 2000 == 0:
                print(f"    LP alt iter={it+1} ({time.monotonic()-t0:.0f}s): auto_score={best_score:.10f}")

    return best_a, best_score


def lse_lbgfs_auto(a0, time_limit, T_stages=None, verbose=True):
    """
    Direct optimization of auto-convolution using LSE smoothing + L-BFGS.
    This optimizes max(a*a) directly, not max(a*b).
    """
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(30, int(remaining * 30))

        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            # Gradient of sum_k w_k * (a*a)_k w.r.t. a:
            # d/da_j [(a*a)_k] = a_{k-j} + a_{k-j} (with boundaries) = 2*a_{k-j} (for auto-conv)
            # Actually for auto-conv: (a*a)_k = sum_j a_j * a_{k-j}
            # d/da_j = a_{k-j} + (if k-j != j) a_{k-j} ... 
            # More carefully: (a*a)_k = sum_j a_j a_{k-j}
            # d/da_m = a_{k-m} + a_{k-m} if m != k-m, else a_{k-m}
            # For the LSE: grad_a = 2 * sum_k w_k * shift_k(a) (approximately, ignoring the diagonal)
            # Let's use the exact formula:
            grad_a = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                # d/da_j [(a*a)_k] = a_{k-j} (from the "first" a) 
                # But since it's auto-conv, we also get a contribution from the "second" a
                # (a*a)_k = sum_j a_j * a_{k-j}, d/da_m = a_{k-m} + a_{k-m} * [m == k-m]... 
                # Actually: d/da_m sum_j a_j a_{k-j} = a_{k-m} (from j=m term in first sum)
                # plus d/da_m sum_j a_j a_{k-j} where m appears in second factor: a_m when k-m=j, i.e., j=k-m
                # So total: a_{k-m} + a_m * (contribution from second factor)
                # Wait: (a*a)_k = sum_j a_j a_{k-j}. If we differentiate w.r.t. a_m:
                # Term where j=m: a_{k-m}
                # Term where k-j=m (j=k-m): a_{k-m}
                # But these are the same term when j=m and k-j=m, i.e., k=2m.
                # So: d/da_m = a_{k-m} + a_{k-m} - a_{k-m}*[k=2m] = 2*a_{k-m} - a_{k-m}*[k=2m]
                # For simplicity and since n is large, the diagonal correction is negligible.
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            # Correction for diagonal terms (k = 2m): we double-counted
            # For k from 0 to 2n-2, if k is even and k/2 < n, subtract w[k]*a[k/2]
            for k in range(0, 2 * n - 1, 2):
                m = k // 2
                if m < n:
                    grad_a[m] -= w[k] * a_cur[m]

            # Chain rule for softmax parametrization
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-25, 'gtol': 1e-16,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                if verbose:
                    print(f"    LSE L-BFGS T={T}: score={best_score:.10f} ({time.monotonic()-t0:.0f}s)")
        except Exception:
            pass

    return best_a, best_score


def subgradient_auto(a0, time_limit, verbose=True):
    """Subgradient method directly on max(a*a) with Polyak step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Lower bound: 2n/(2n-1)
    lb = 2.0 * n / (2.0 * n - 1.0)
    
    lr_scale = 0.1
    no_improve = 0
    avg_sc = best_score
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        sc = 2.0 * n * M
        avg_sc = 0.99 * avg_sc + 0.01 * sc

        # Find active peaks
        threshold = M * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        # Diagonal correction
        for k in active:
            if k % 2 == 0:
                m = k // 2
                if m < n:
                    g[m] -= a[m]
        g /= len(active)

        # Polyak step size
        gap = sc - lb
        if gap < 1e-15:
            break
        lr = lr_scale * gap / (np.dot(g, g) + 1e-20)
        lr = min(lr, 0.05)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc_trial = 2.0 * n * fast_convolve(a_trial).max()
        if sc_trial < best_score - 1e-14:
            best_score = sc_trial
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr_scale = min(lr_scale * 1.02, 0.5)
        else:
            no_improve += 1
            if no_improve >= 50:
                lr_scale *= 0.9
                no_improve = 0

        if verbose and (step + 1) % 50000 == 0:
            print(f"    Subgrad step={step+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def active_set_kkt(a0, solver, time_limit, verbose=True):
    """
    Solve the KKT system for a given active set.
    If a is optimal with active set K, then:
    - (a*a)_k = M for k in K
    - (a*a)_k <= M for k not in K

    - a_j >= 0, a_j = 0 implies complementary slackness
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for outer_it in range(100):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        sc = 2.0 * n * M

        # Identify active set: positions where b[k] is within 1% of max
        threshold = M * 0.99
        active = np.where(b >= threshold)[0]
        m_active = len(active)
        
        if m_active == 0:
            break
        if m_active > 200:
            # Too many active constraints, tighten threshold
            threshold = M * 0.999
            active = np.where(b >= threshold)[0]
            m_active = len(active)
            if m_active == 0:
                active = np.array([int(np.argmax(b))])
                m_active = 1

        # (a'*a')_k = M' for k in active
        # minimize M'
        # This is nonlinear, so use a linearized version:
        # Linearize (a'*a')_k around a: (a'*a')_k ≈ 2 * (a * a')_k - (a*a)_k

        constraints = [cp.sum(a_var) == 1]

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                sv = a_val.sum()

                    a_val = a_val / sv
                    # Blend with current
                    for t_blend in [0.5, 0.3, 0.2, 0.1]:
                        a_trial = (1 - t_blend) * a + t_blend * a_val
                        a_trial = normalize(a_trial)
                        sc_trial = 2.0 * n * fast_convolve(a_trial).max()
                        if sc_trial < sc - 1e-12:
                            a = a_trial
                            best_score = sc_trial
                            best_a = a_trial.copy()
                            break
        except Exception:
            break

        if verbose and (outer_it + 1) % 20 == 0:
            print(f"    KKT iter={outer_it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def fourier_refine(a0, n_freq, time_limit, verbose=True):
    """Refine in Fourier domain using Nelder-Mead on low-frequency coefficients."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)

    n_free = 2 * n_freq + 1
    x0 = np.zeros(n_free)
    x0[0] = A[0].real
    for i in range(1, n_freq + 1):
        x0[2 * i - 1] = A[i].real
        x0[2 * i] = A[i].imag

    A_rest = A[n_freq + 1:].copy()

    def fourier_obj(x):
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()

    try:
        res = minimize(
            fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': max(1000, int(time_limit * 2)), 'xatol': 1e-15,
                     'fatol': 1e-15, 'adaptive': True}
        )
        x_opt = res.x
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s > 1e-10:
            a_candidate = a_candidate / s
            sc = 2.0 * n * fast_convolve(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
                if verbose:
                    print(f"    Fourier K={n_freq}: score={best_score:.10f}")
    except Exception:
        pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def parametric_search(a0, time_limit, verbose=True):
    """Try parametric families and optimize their parameters."""
    n = len(a0)
    best_a = normalize(a0)
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    x = np.linspace(0, 1, n)

    # Family 1: raised cosine with power
    def fam1(params):
        p = max(0.1, params[0])
        width = np.clip(params[1], 0.3, 1.0)
        a = np.cos(np.pi * (x - 0.5) / width / 2.0) ** (2 * p)
        a = np.where(np.abs(x - 0.5) < width / 2.0, a, 0)
        return np.maximum(a, 0)

    # Family 2: Gaussian mixture
    def fam2(params):
        c1, c2, s1, s2 = params
        a = c1 * np.exp(-0.5 * ((x - 0.5) / s1) ** 2) + c2 * np.exp(-0.5 * ((x - 0.3) / s2) ** 2) + c2 * np.exp(-0.5 * ((x - 0.7) / s2) ** 2)
        return np.maximum(a, 0)

    # Family 3: Polynomial
    def fam3(params):
        a = 1.0 + params[0] * np.cos(2 * np.pi * x) + params[1] * np.cos(4 * np.pi * x) + params[2] * np.cos(6 * np.pi * x)
        return np.maximum(a, 0)

    # Family 4: Beta-like
    def fam4(params):
        alpha = max(0.5, params[0])
        a = x ** (alpha - 1) * (1 - x) ** (alpha - 1)
        return np.maximum(a, 1e-10)

    families = [
        (fam1, np.array([2.0, 1.0]), 3),
        (fam2, np.array([1.0, 0.5, 0.15, 0.1]), 4),
        (fam3, np.array([0.2, 0.1, 0.05]), 3),
        (fam4, np.array([2.0]), 1),
    ]

    for fam_func, x_init, n_params in families:
        if time.monotonic() - t0 > time_limit:
            break

        def obj(params):
            a = fam_func(params)
            s = a.sum()
            if s < 1e-10:
                return 1e15
            a = a / s
            return 2.0 * n * fast_convolve(a).max()

        try:
            res = minimize(obj, x_init, method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True})
            a_opt = fam_func(res.x)
            s = a_opt.sum()
            if s > 1e-10:
                a_opt = a_opt / s
                sc = 2.0 * n * fast_convolve(a_opt).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_opt.copy()
                    if verbose:
                        print(f"    Parametric: score={best_score:.10f}")
        except Exception:
            pass

    return best_a, best_score


def peak_flattening_iteration(a0, time_limit, n_steps=1000, verbose=True):
    """
    Iterative peak flattening: at each step, identify the peak,
    and move mass from the peak region to reduce it.
    Uses a "water-filling" inspired approach.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        sc = 2.0 * n * M

        # Find the top peaks
        k_max = int(np.argmax(b))
        # The peak at k_max is caused by the correlation of a with its shift by k_max
        # To reduce it, we need to reduce the "overlap" of a and shift_{k_max}(a)
        # This means reducing a_j where both a_j and a_{k_max-j} are large
        
        # Compute the "contribution" of each pair
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[k_max - idx]
        
        # Reduce the elements with high contribution
        # Use softmax-weighted reduction
        max_contrib = contributions.max()
        if max_contrib > 1e-20:
            weights = contributions / (max_contrib + 1e-20)
            # Move mass from high-contribution positions to low-contribution positions
            g = np.zeros(n)
            g[idx] = weights
            # Subtract from the vector (reduces those positions)
            a_trial = a - lr * g
            # Also add a small amount of uniform mass to maintain sum
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                sc_trial = 2.0 * n * fast_convolve(a_trial).max()
                if sc_trial < best_score - 1e-14:
                    best_score = sc_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    lr = min(lr * 1.01, 0.05)
                else:
                    lr *= 0.95
            else:
                lr *= 0.9
        else:
            break

        if verbose and (step + 1) % 200 == 0:
            print(f"    PeakFlat step={step+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. LP alternation (cross-convolution Nash equilibrium)
    2. Direct auto-conv LSE-LBFGS (key improvement over parent)
    3. Subgradient on auto-conv
    4. Peak flattening iterations
    5. Fourier domain refinement
    6. Active set refinement
    7. Multi-n exploration
    8. Final intensive polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective deadline in {budget_s - 30:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}) ***")

    # === Phase 1: LP Alternation ===
    print(f"\n=== Phase 1: LP Alternation (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "lp_alt")

    # === Phase 2: Direct Auto-Conv LSE L-BFGS (KEY PHASE) ===
    print(f"\n=== Phase 2: Direct Auto-Conv LSE L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 450.0)
    if t2 > 30:
        print(f"  Time budget: {t2:.0f}s")
        # Use aggressive T schedule with many stages
        T_stages = [200.0, 100.0, 50.0, 25.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_auto, sc_auto = lse_lbgfs_auto(best_seq, t2, T_stages=T_stages)
        try_update(a_auto, "auto_lse")

    # === Phase 3: Subgradient on Auto-Conv ===
    print(f"\n=== Phase 3: Subgradient Auto-Conv (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 150.0)
    if t3 > 15:
        a_sg, sc_sg = subgradient_auto(best_seq, t3)
        try_update(a_sg, "subgrad_auto")

    # === Phase 4: Peak Flattening ===
    print(f"\n=== Phase 4: Peak Flattening (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 120.0)
    if t4 > 10:
        a_pf, sc_pf = peak_flattening_iteration(best_seq, t4, n_steps=5000)
        try_update(a_pf, "peak_flat")

    # === Phase 5: Second LP Alternation (warm start) ===
    print(f"\n=== Phase 5: LP Alternation Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 250.0)
    if t5 > 20:
        a_lp2, sc_lp2 = lp_alternation(best_seq, t5, solver)
        try_update(a_lp2, "lp_alt_2")

    # === Phase 6: Second Auto-Conv LSE (finer T) ===
    print(f"\n=== Phase 6: Auto-Conv LSE Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 200.0)
    if t6 > 20:
        T_fine = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_auto2, sc_auto2 = lse_lbgfs_auto(best_seq, t6, T_stages=T_fine)
        try_update(a_auto2, "auto_lse_2")

    # === Phase 7: Fourier Refinement ===
    print(f"\n=== Phase 7: Fourier Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        for n_freq in [32, 64, 128]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_sub = min(rem * 0.2, 40.0)
            a_four, sc_four = fourier_refine(best_seq, n_freq, t_sub)
            try_update(a_four, f"fourier_K{n_freq}")

    # === Phase 8: Multi-n Exploration ===
    print(f"\n=== Phase 8: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_candidates = []
        base_n = n_best
        for mult in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 1.75, 2.0, 2.5, 3.0]:
            cand_n = int(base_n * mult)
            if 400 <= cand_n <= 6000 and cand_n not in n_candidates and cand_n != n_best:
                n_candidates.append(cand_n)
        # Also try some absolute values
        for abs_n in [800, 1000, 1500, 2000, 2500, 3000, 4000]:
            if abs_n not in n_candidates and abs_n != n_best:
                n_candidates.append(abs_n)

        for cand_n in sorted(n_candidates, key=lambda x: abs(x - base_n)):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.25, 100.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            
            # Quick LSE for this n
            T_quick = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_cand, _ = lse_lbgfs_auto(init_cand, t_sub * 0.7, T_stages=T_quick, verbose=False)
            try_update(a_cand, f"lse_n{cand_n}")
            
            rem = deadline - time.monotonic()
            if rem > 20 and len(best_seq) != cand_n:
                # If this n gives a better score, also try LP
                t_lp_sub = min(rem * 0.3, 40.0)
                if t_lp_sub > 10:
                    a_cand2, _ = lp_alternation(a_cand, t_lp_sub, solver, verbose=False)
                    try_update(a_cand2, f"lp_n{cand_n}")

    # === Phase 9: Final Intensive Polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Very fine LSE
        t_fin1 = min(rem * 0.4, 60.0)
        if t_fin1 > 10:
            T_very_fine = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
            a_fin, _ = lse_lbgfs_auto(best_seq, t_fin1, T_stages=T_very_fine, verbose=False)
            try_update(a_fin, "final_lse_fine")

        # Subgradient
        rem = deadline - time.monotonic()
        t_fin2 = min(rem * 0.3, 40.0)
        if t_fin2 > 10:
            a_fin2, _ = subgradient_auto(best_seq, t_fin2, verbose=False)
            try_update(a_fin2, "final_subgrad")

        # Peak flattening
        rem = deadline - time.monotonic()
        t_fin3 = min(rem * 0.3, 30.0)
        if t_fin3 > 5:
            a_fin3, _ = peak_flattening_iteration(best_seq, t_fin3, n_steps=3000, verbose=False)
            try_update(a_fin3, "final_peak_flat")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
