import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

    # cross[m] = sum_j w[m+j] * a[j]
    # Let v[j] = w[j+n-1] for j in range(2n-1), padded to fft_size
    # Then cross[m] = sum_j v[j+m] * a[j]
    # This is the cross-correlation of v and a at lag m
    # = ifft(conj(FFT(v)) * FFT(a))[m]  (with appropriate padding)

    # cross[m] = sum_{j=0}^{n-1} w[m+j] * a[j]
    # Let's define u[t] = w[t] for t in [0, 2n-2], 0 otherwise
    # cross[m] = sum_{j=0}^{n-1} u[m+j] * a[j]
    # Let s = m+j, then j = s-m, range of s is [m, m+n-1]
    # cross[m] = sum_{s=m}^{m+n-1} u[s] * a[s-m]

    # = sum_{s=m}^{m+n-1} u[s] * a[s-m]  (since a is zero outside [0,n-1])
    # = cross[m] ✓

    # i.e., cross[m] = conv(u, flip(a))[m+n-1]
    
    flip_a = a[::-1]

    conv_size = 3*n - 1  # be safe
    fft_size2 = 1 << (conv_size - 1).bit_length()
    U = rfft(full_weights, fft_size2)
    FA = rfft(flip_a, fft_size2)
    conv_result = irfft(U * FA, fft_size2)
    cross = conv_result[n-1 : 2*n-1]  # cross[m] for m=0..n-1
    
    grad = 2.0 * cross
    return grad, conv, float(conv.max())

1. Log-sum-exp smoothed gradient descent with Nesterov acceleration and adaptive step size
2. Active-set focused optimization (top-m lags)

4. Structured initializations (sinusoidal, parabolic, cosine-bell)

6. Iterative beta annealing for progressive sharpening

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def smooth_gradient(a, beta, top_m=None):
    """Compute gradient of log-sum-exp smoothed max autocorrelation.
    Returns (grad, conv, M) where M = max of conv."""
    n = len(a)
    conv = fast_convolve(a)
    
    if top_m is not None and top_m < 2 * n - 1:
        idx = np.argpartition(conv, -top_m)[-top_m:]
        c_active = conv[idx]
        max_c = c_active.max()
        scaled = beta * (c_active - max_c)
        scaled = np.clip(scaled, -50, 0)
        w_active = np.exp(scaled)

            w_active[c_active.argmax()] = 1.0

        weights[idx] = w_active
    else:

def lse_smoothed_descent(a0, time_limit, beta_init=10.0, top_m=50, label=""):
    """Nesterov-accelerated log-sum-exp smoothed gradient descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    beta = beta_init
    lr = 1.0 / (2.0 * beta * n)
    gamma = 1.0
    y = a.copy()
    it = 0
    no_improve = 0
    patience = 100
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov momentum
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))
        
        # Compute gradient
        grad, conv, M = smooth_gradient(y, beta, top_m)
        
        # Adaptive step size: use the fact that the objective is bounded
        # Step: a_new = a - lr * grad, projected
        a_new = simplex_project(np.maximum(a - lr * grad, 0))
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        
        # Adaptive parameters
        if no_improve > patience:
            lr *= 0.85
            beta *= 1.2
            no_improve = 0
            patience = max(30, patience // 1.2)
            if lr < 1e-12:
                break
        elif it % 20 == 0:
            # Gradually increase beta for sharper approximation
            beta *= 1.005
            lr = 1.0 / (2.0 * beta * n * (1.0 + 0.1 * np.log1p(it)))
        
        if it % 50 == 0 and label:
            elapsed = time.monotonic() - t0
            print(f"    LSE {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, beta={beta:.1f}, lr={lr:.2e}")
    
    return best_a, best_score

def smart_coord_descent(a0, time_limit, label="", use_pairwise=False):
    """Coordinate descent with smart 1D candidate evaluation and optional pairwise moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    M = float(conv.max())

    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        if use_pairwise and sweeps % 3 == 0:
            # Pairwise move: transfer mass from i to j
            cand_i = np.where(a > 0.01 / n)[0]
            if len(cand_i) > 10:
                # Try a few random pairs
                n_tries = min(20, len(cand_i))
                idx = np.random.choice(len(cand_i), size=n_tries, replace=False)

                    i = cand_i[idx[t]]
                    j = np.random.randint(n)

                        continue
                    ai, aj = a[i], a[j]
                    if ai < 1e-12:
                        continue

                        # Add contribution to j

                            idx2 = k - j

                                new_conv[k] += (-delta) * (a[idx2] + delta if idx2 == j else a[idx2])
                        if 2*j <= 2*n-2:

        for i in range(n):

            k_end = min(i + n - 1, 2 * n - 2)

                unaff_left = float(conv[:k_start].max())
            else:

            unaff_max = max(unaff_left, unaff_right)

            for j in range(len(affected)):
                c = coeffs[j]

                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            top_idx = np.argsort(affected)[-top_k:]

                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:

            # Fractional moves
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)

            best_delta = 0.0
            best_new_max = M
            
            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        # Periodic recompute
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
            if 2.0 * n * M < best_score:
                best_score = 2.0 * n * M
                best_a = a.copy()
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            
            best_delta = 0.0
            best_new_max = M
            
            for factor in factors:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_score

def structured_init(n, shape="sine"):
    """Create structured initial sequences."""
    i = np.arange(n, dtype=np.float64)
    if shape == "sine":
        a = np.sin(np.pi * (i + 0.5) / n)
    elif shape == "parabola":
        a = np.sqrt(np.maximum(i * (n - 1 - i), 1.0))
    elif shape == "cosine_bell":
        a = 0.5 * (1 + np.cos(2 * np.pi * (i + 0.5) / n))
    elif shape == "gaussian":
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    elif shape == "uniform":
        a = np.ones(n)
    elif shape == "raised_cosine":
        a = 0.5 * (1 - np.cos(2 * np.pi * (i + 0.5) / n))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturb_sequence(a0, rng, method="peak_target", strength=0.05):
    """Perturb sequence to escape local optima."""
    n = len(a0)
    a = a0.copy()
    
    if method == "peak_target":
        conv = fast_convolve(a)
        M = conv.max()
        threshold = M * 0.85
        peak_positions = np.where(conv >= threshold)[0]
        
        pressure = np.zeros(n)
        for k in peak_positions:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                pressure[i] += a0[i] * a0[j]
        
        p_max = pressure.max()
        if p_max > 1e-20:
            pressure = pressure / p_max
        
        noise = rng.standard_normal(n) * 0.3
        a = a * (1.0 - strength * pressure + strength * noise)
        
    elif method == "random_zero":
        frac = strength
        mask = rng.random(n) < frac
        a[mask] = 0
        n_add = max(1, int(n * 0.1))
        zero_idx = np.where(a < 1e-15)[0]
        if len(zero_idx) > 0:
            add_idx = rng.choice(zero_idx, size=min(n_add, len(zero_idx)), replace=False)
            a[add_idx] = rng.random(len(add_idx)) * a.max() * 0.3
    
    elif method == "gaussian_noise":
        noise = rng.standard_normal(n) * strength
        a = a + noise * a.mean()
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def lp_best_response(a_fixed):
    """Solve LP for best response b to fixed a."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix T where (T @ b)[k] = conv(a_fixed, b)[k]
    # conv(a,b)[k] = sum_i a[i] * b[k-i]
    # For fixed a, this is linear in b: sum_j (sum_i a[i] * [k-i==j]) * b[j]
    # = sum_j C[k,j] * b[j] where C[k,j] = a[k-j] if 0<=k-j<n else 0
    
    constraints = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a_fixed[idx]
        constraints.append(row @ b_var <= M_var)
    constraints.append(cp.sum(b_var) == 1)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solver = get_lp_solver()
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fictitious_play(a0, time_limit, label=""):
    """LP fictitious play with adaptive averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, M_b = lp_best_response(a)
        if b_new is None:
            break
        
        # Adaptive averaging
        alpha = min(1.0, 8.0 / (it + 3))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        
        # Also check b directly
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()
        
        a = a_new
        
        if it % 10 == 0 and label:
            elapsed = time.monotonic() - t0
            print(f"    FP {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def propose_candidate():
