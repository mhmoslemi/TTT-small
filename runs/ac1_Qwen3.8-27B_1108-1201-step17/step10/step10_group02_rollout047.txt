import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val > 1e-15:
                    b_val = b_val / s_val
                    return b_val
                return None
        except Exception:
            continue
    return None

    # Wait, (a*a)_k = sum_i a_i * a_{k-i}. d/dx_j = a_{k-j} (from term i=j) + a_j (from term where k-i=j, i=k-j)
    # If j != k-j: d/dx_j = a_{k-j} + a_{k-j} = 2*a_{k-j}... no wait.
    # (a*a)_k = sum_{i=0}^{k} a_i * a_{k-i} (with a_i=0 outside [0,n-1])
    # d/dx_j of this = a_{k-j} (coefficient of a_j in the first slot) + a_j... 
    # No. Let me be more careful.
    # (a*a)_k = sum_i a_i * a_{k-i}
    # d/dx_j = a_{k-j} (when i=j, the derivative of a_i is 1) + sum_i a_i * d(a_{k-i})/dx_j
    # d(a_{k-i})/dx_j = 1 when k-i = j, i.e., i = k-j
    # So d/dx_j = a_{k-j} (from first term, i=j) + a_{k-j} (from second term, i=k-j)
    # = 2*a_{k-j} if j != k-j (i.e., j != k/2)
    # = a_{k-j} if j = k/2 (i.e., k is even and j = k/2, in which case both terms are a_j^2 and d/dx_j = 2*a_j)
    # Wait, if j = k-j, then (a*a)_k contains the term a_j * a_j = a_j^2, and d/dx_j = 2*a_j.
    # So in all cases: d/dx_j [(a*a)_k] = 2*a_{k-j} (with a_{k-j}=0 if k-j not in [0,n-1])
    # Wait no. If j = k-j (i.e., 2j=k), then the term is a_j^2, and its derivative is 2*a_j.
    # If j != k-j, then the terms are a_j*a_{k-j} and a_{k-j}*a_j, and the derivative w.r.t. a_j is a_{k-j} + a_{k-j} = 2*a_{k-j}.
    # So in all cases: d/dx_j [(a*a)_k] = 2*a_{k-j} (where a_{k-j}=0 if out of range)
    
    # Actually wait, I need to be more careful. Let's say k=4, n=5.
    # (a*a)_4 = a_0*a_4 + a_1*a_3 + a_2*a_2 + a_3*a_1 + a_4*a_0
    # d/dx_2 = 2*a_2 (from the a_2^2 term)
    # d/dx_0 = a_4 + a_4 = 2*a_4
    # d/dx_1 = a_3 + a_3 = 2*a_3
    # Yes, so d/dx_j [(a*a)_k] = 2*a_{k-j} (with a_m = 0 if m < 0 or m >= n)
    
    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-15:
            continue
        coeff = 2.0 * w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad_a[j] += coeff * a[k - j]
    
    # Smoothed max value
    smooth_val = max_b + T * np.log(exp_b.sum())
    return smooth_val, grad_a, w


def mirror_descent_optimize(a0, time_limit, eta0=0.06, active_frac=0.999, nesterov=True):
    """Mirror descent with Nesterov acceleration for the peak minimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient: average of a[k-j] for active k's
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je+1] += a[k - np.arange(js, je+1)]
        g /= len(active)
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys
        
        a = a_new
        score = evaluate_local(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 500:
            rng = np.random.default_rng(it * 7 + 13)
            noise = rng.standard_normal(n) * 0.005
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def lp_alternate_covector(a0, time_limit, solver=None):
    """
    LP alternating with covector (averaged) method for the final answer.
    Key improvement: maintains running average of all iterates (covector),
    which has better convergence properties for the duality gap.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # Covector: running average of all a iterates
    covector_a = a.copy()
    covector_y = a.copy()
    count = 1
    
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov update for y
        alpha = 2.0 / (it + 2)
        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / sy
        
        # Updated iterate for a (weighted average)
        a_new = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa
        
        y = y_new
        a = a_new
        
        # Update covector (simple running average)
        covector_a = (count * covector_a + a_new) / (count + 1)
        covector_y = (count * covector_y + y_new) / (count + 1)
        count += 1
        
        # Check all candidates
        candidates = [a, y, covector_a, covector_y]
        for cand in candidates:
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
        
        # Stall detection
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                prev_best = best_score
                stall = 0
            else:
                stall += 1
            
            if stall >= 15:
                noise_scale = 0.02 / np.sqrt(min(stall, 100))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                # Also perturb covector
                covector_a = covector_a + noise * 0.5
                covector_a = np.maximum(covector_a, 0)
                covector_a = covector_a / covector_a.sum()
                stall = 0
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            cov_score = 2.0 * n * fast_convolve(covector_a).max()
            print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} cov={cov_score:.8f}")
    
    return best_a, best_score


def logsumexp_refine(a0, time_limit, T_stages=None):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        def func(z_inner):
            a = np.exp(z_inner - z_inner.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T)
            return max_b + T * np.log(exp_b.sum())
        
        def jac(z_inner):
            a = np.exp(z_inner - z_inner.max())
            a = a / a.sum()
            _, grad_a, _ = smooth_max_grad(a, T)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        maxiter = 2000
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'disp': False})
            z = res.x
        except Exception:
            pass
        
        elapsed = time.monotonic() - t0
        a_check = np.exp(z - z.max())
        a_check = a_check / a_check.sum()
        print(f"    LSE T={T:.3f} ({elapsed:.0f}s): score={evaluate_local(a_check):.8f}")
    
    result = np.exp(z - z.max())
    result = result / result.sum()
    return result


def coordinate_descent(a0, time_limit):
    """Targeted coordinate descent for peak reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        improved = False
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        for pos in top_positions:
            for factor in [0.999, 0.998, 0.997, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-15:
                    continue
                # Redistribute to non-peak positions
                mask = np.ones(n, dtype=bool)
                # Don't redistribute to positions that contribute to the peak
                for p in range(js, je+1):
                    mask[p] = False
                    mask2 = peak_k - p
                    if 0 <= mask2 < n:
                        mask[mask2] = False
                if mask.sum() == 0:
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0
        
        if (stall + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.5 * (x - 0.5) ** 2 / 0.25)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
