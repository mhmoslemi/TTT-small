from scipy.fft import rfft, irfft

    a = softmax(z), objective = (2n/T) * logsumexp(T * conv(a,a))
    """

    sa = a.sum()

    b = fast_convolve(a)
    max_b = b.max()
    Tb = T * (b - max_b)
    exp_Tb = np.exp(Tb)
    sum_exp = exp_Tb.sum()
    w = exp_Tb / sum_exp
    lse = max_b + np.log(sum_exp) / T
    obj = (2.0 * n / T) * lse

def peak_equalization(a0, time_limit, n_max_steps=100000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.85
    no_improve = 0

        if time.monotonic() - t0 > time_limit:
            break

        max_b = b.max()

        # Active set: peaks within 0.1% of max
        threshold = max_b * 0.9999

        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)

        g_centered = g - g_mean

        else:
            a_trial = a_trial / st

        score = evaluate_local(a_trial)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq step={step} ({elapsed:.0f}s): score={best_score:.8f}, n_active={len(active)}, lr={lr:.6f}")

    return best_a, best_score


def kkt_fixed_point(a0, time_limit, n_max_iters=50000):
    """KKT-inspired fixed point: at optimum, a[k*-i] ≈ M/S for active peaks k*.
    Iterate: identify active peaks, compute target, relax towards it."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    mixing = 0.3

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Target: for each active peak k, a[k-i] should equal max_b (since S=1)
        # Average target over active peaks
        target = np.zeros(n)
        target_count = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            vals = k - idx  # These are the positions j such that a[j] * a[k-j] contributes
            # a[k-i] for i in idx means positions k-idx
            target[k - idx] += max_b
            target_count[k - idx] += 1

        valid = target_count > 0
        target[valid] /= target_count[valid]

        # Relax: move a towards target where target is defined, keep elsewhere
        a_trial = a.copy()
        a_trial[valid] = (1 - mixing) * a[valid] + mixing * target[valid]
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            continue
        a_trial = a_trial / st

        score = evaluate_local(a_trial)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller mixing
            mixing *= 0.95
            if mixing < 0.01:
                break

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT-FP iter={it} ({elapsed:.0f}s): score={best_score:.8f}, n_active={len(active)}, mix={mixing:.4f}")

    return best_a, best_score


def lbgfs_logsumexp_refined(a0, time_limit, T_stages=None, maxiter_total=50000):
    """Refined L-BFGS on LogSumExp with better gradient."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300) + 1e-300)
    z = z - z.max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(50, maxiter_total // n_stages)

    best_a = a0.copy()
    best_score = evaluate_local(a0)

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_loc):
            a_loc = np.exp(z_loc - z_loc.max())
            a_loc = a_loc / a_loc.sum()
            b_loc = fast_convolve(a_loc)
            max_b_loc = b_loc.max()
            exp_b = np.exp(T_stage * (b_loc - max_b_loc))
            lse = max_b_loc + np.log(exp_b.sum()) / T_stage
            return (2.0 * n) * lse

        def jac(z_loc):
            obj_val, grad_val = lse_obj_grad_topk(z_loc, n, T_stage, topk=150)
            # Scale: lse_obj_grad gives gradient of (2n/T)*LSE, we want gradient of 2n*LSE
            return grad_val * T_stage / (2.0 * n) * 2.0 * n  # = grad_val

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
            a_out = np.exp(z - z.max())
            a_out = a_out / a_out.sum()
            sc = evaluate_local(a_out)
            if sc < best_score:
                best_score = sc
                best_a = a_out.copy()
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetric_refine(a0, time_limit):
    """Optimize assuming symmetry: a_j = a_{n-1-j}."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    # Symmetrize
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / a_sym.sum()
    score_sym = evaluate_local(a_sym)

    if score_sym >= evaluate_local(a):
        return a  # Symmetrization didn't help

    # Now optimize the symmetric part (first half)
    half_n = (n + 1) // 2
    a_half = a_sym[:half_n].copy()

    t0 = time.monotonic()
    lr = 0.03
    momentum_half = np.zeros(half_n)
    best_score = score_sym
    best_a_half = a_half.copy()
    no_improve = 0

    for step in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Reconstruct full symmetric sequence
        if n % 2 == 0:
            a_full = np.concatenate([a_half, a_half[::-1]])
        else:
            a_full = np.concatenate([a_half, a_half[-2::-1]])
        a_full = np.maximum(a_full, 0)
        sf = a_full.sum()
        if sf < 1e-15:
            break
        a_full = a_full / sf

        b = fast_convolve(a_full)
        max_b = b.max()
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]

        # Gradient on half
        g_half = np.zeros(half_n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a_full[k - idx]
            # Map back to half indices
            for pos in idx:
                if pos < half_n:
                    g_half[pos] += contribs[pos]
                else:
                    mirror = n - 1 - pos
                    if mirror < half_n:
                        g_half[mirror] += contribs[pos]
        g_half /= max(1, len(active))
        g_half -= g_half.mean()

        momentum_half = 0.85 * momentum_half + 0.15 * g_half
        a_half = a_half + lr * momentum_half
        a_half = np.maximum(a_half, 0)

        # Check score
        if n % 2 == 0:
            a_check = np.concatenate([a_half, a_half[::-1]])
        else:
            a_check = np.concatenate([a_half, a_half[-2::-1]])
        a_check = np.maximum(a_check, 0)
        sc = a_check.sum()
        if sc < 1e-15:
            break
        a_check = a_check / sc
        score = evaluate_local(a_check)

        if score < best_score - 1e-13:
            best_score = score
            best_a_half = a_half.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.8
                no_improve = 0
                if lr < 1e-8:
                    break

    # Reconstruct
    if n % 2 == 0:
        a_out = np.concatenate([best_a_half, best_a_half[::-1]])
    else:
        a_out = np.concatenate([best_a_half, best_a_half[-2::-1]])
    a_out = np.maximum(a_out, 0)
    a_out = a_out / a_out.sum()
    return a_out


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining adaptive LogSumExp gradient, peak
    equalization, KKT fixed-point, symmetric refinement, and multi-length search.
    """
    global rng_restart
    rng_restart = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Symmetric check ===
    print(f"\n=== Phase 1: Symmetric check ===")
    a_sym = (best_seq + best_seq[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / a_sym.sum()
    sc_sym = evaluate_local(a_sym)
    print(f"  Symmetric score: {sc_sym:.8f} vs original {best_score:.8f}")
    if sc_sym < best_score:
        best_seq = a_sym
        best_score = sc_sym
        print(f"  Using symmetric version")

    # === Phase 2: Adaptive LogSumExp (main workhorse) ===
    print(f"\n=== Phase 2: Adaptive LogSumExp, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_res, _ = adaptive_lse_optimize(best_seq, t2, T_init=20.0, T_min=0.005,
                                          n_max_steps=300000, use_topk=80)
        try_update(a_res, "adLSE_p2")

    # === Phase 3: Peak Equalization ===
    print(f"\n=== Phase 3: Peak Equalization, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        a_pe, _ = peak_equalization(best_seq, t3, n_max_steps=80000)
        try_update(a_pe, "peak_eq")

    # === Phase 4: KKT Fixed Point ===
    print(f"\n=== Phase 4: KKT Fixed Point, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 15:
        a_kkt, _ = kkt_fixed_point(best_seq, t4, n_max_iters=30000)
        try_update(a_kkt, "kkt_fp")

    # === Phase 5: L-BFGS LogSumExp refinement ===
    print(f"\n=== Phase 5: L-BFGS LogSumExp, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 300.0)
    if t5 > 20:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_lb, _ = lbgfs_logsumexp_refined(best_seq, t5, T_stages=T_stages,
                                           maxiter_total=30000)
        try_update(a_lb, "lbgfs_p5")

    # === Phase 6: Second Adaptive LogSumExp (finer) ===
    print(f"\n=== Phase 6: Fine Adaptive LogSumExp, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 200.0)
    if t6 > 20:
        a_res2, _ = adaptive_lse_optimize(best_seq, t6, T_init=5.0, T_min=0.002,
                                           n_max_steps=200000, use_topk=50)
        try_update(a_res2, "adLSE_p6")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        candidate_lengths = [int(n_cur * 0.85), int(n_cur * 1.15), int(n_cur * 1.3)]
        t_each = min(rem / (len(candidate_lengths) + 1), 80.0)
        for ci, cn in enumerate(candidate_lengths):
            if time.monotonic() > deadline - 30:
                break
            if t_each < 15:
                break
            print(f"  Trying n={cn}, t={t_each:.0f}s")
            init_cn = interpolate_sequence(best_seq, cn)
            a_cn, _ = adaptive_lse_optimize(init_cn, t_each * 0.6, T_init=10.0,
                                             T_min=0.01, n_max_steps=50000, use_topk=60)
            try_update(a_cn, f"len_{cn}")
            rem = deadline - time.monotonic()
            t_each = min(rem / (len(candidate_lengths) - ci - 1), 80.0) if ci < len(candidate_lengths) - 1 else 0

    # === Phase 8: Symmetric refinement ===
    print(f"\n=== Phase 8: Symmetric Refine, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sym = min(rem * 0.2, 60.0)
        a_s = symmetric_refine(best_seq, t_sym)
        try_update(a_s, "sym_refine")

    # === Phase 9: Final peak equalization ===
    print(f"\n=== Phase 9: Final Peak Eq, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 40.0)
        a_fe, _ = peak_equalization(best_seq, t_fin, n_max_steps=50000)
        try_update(a_fe, "final_pe")

    # === Phase 10: Final fine LogSumExp ===
    print(f"\n=== Phase 10: Final LogSumExp, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin2 = min(rem * 0.5, 30.0)
        a_f2, _ = adaptive_lse_optimize(best_seq, t_fin2, T_init=2.0, T_min=0.001,
                                         n_max_steps=50000, use_topk=30)
        try_update(a_f2, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
