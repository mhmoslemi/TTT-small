For the exact solution:

- We want to find `x >= 0` that minimizes `max_k f_k(x)`
- This is equivalent to finding the minimum `M` such that `f_k(x) <= M` for all `k`
- Each constraint `f_k(x) <= M` defines a region of `(x, M)` space
- The optimal `(x, M)` is on the lower envelope of these constraints

For the non-quadratic constraints (`k != 2i`):
`f_k(x) = conv_old[k] + (x - a[i]) * c_k <= M`
This is a linear constraint in `x`: `x <= (M - conv_old[k]) / c_k + a[i]` (if `c_k > 0`) or `x >= ...` (if `c_k < 0`, but `c_k = 2*a[k-i] >= 0` always).

Since `c_k >= 0`, each non-quadratic constraint gives an upper bound on `x` for a given `M`:
`x <= (M - conv_old[k]) / c_k + a[i]` (if `c_k > 0`)

If `c_k = 0`, the constraint is just `conv_old[k] <= M` (independent of x).

For the quadratic constraint (`k = 2i`):
`f_{2i}(x) = conv_old[2i] + (x - a[i]) * 2*a[i] + (x - a[i])^2 = x^2 <= M`
So `x <= sqrt(M)`.

The optimal solution is:
- Find the smallest `M >= 0` such that there exists `x >= 0` with:
  - `x <= (M - conv_old[k]) / c_k + a[i]` for all `k` with `c_k > 0` in the affected range
  - `x <= sqrt(M)` (for the quadratic constraint)
  - `conv_old[k] <= M` for all `k` with `c_k = 0` in the affected range
  - `x >= 0`

This can be solved by bisection on `M`:
- For a given `M`, compute the maximum allowed `x`:
  - `x_max = min of ((M - conv_old[k]) / c_k + a[i] for k with c_k > 0, M >= conv_old[k])`
  - `x_max = min(x_max, sqrt(M))`
  - Check that `M >= conv_old[k]` for all k with `c_k = 0`
- If `x_max >= 0`, then `M` is feasible
- Bisection to find the minimum feasible `M`

This gives an exact solution in O(n log(precision)) time per coordinate.

import numpy as np
from scipy.fft import rfft, irfft
from scipy import sparse
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_sparse_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLOP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a, solver):
    import cvxpy as cp
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    T = build_sparse_conv_matrix(a)
    constraints = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M.value)

def exact_1d_minimax_step(conv, a, i):
    """
    Find optimal x >= 0 for coordinate i using bisection on the minimax value.
    conv_new[k] = conv[k] + 2*(x - a[i]) * a[k-i] for k != 2i
    conv_new[2i] = conv[2i] - a[i]^2 + x^2
    Minimize max(conv_new) over x >= 0.
    """
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    affected = conv[k_start:k_end + 1]
    coeff = a[0:n]  # a[k-i] for k in [i, i+n-1] gives a[0],...,a[n-1]

    # Unaffected max
    if k_start > 0:
        unaff_left = conv[:k_start].max()
    else:
        unaff_left = 0.0
    if k_end < 2 * n - 2:
        unaff_right = conv[k_end + 1:].max()
    else:
        unaff_right = 0.0
    unaff_max = max(float(unaff_left), float(unaff_right))

    ai = a[i]
    quad_conv = conv[2 * i]  # conv value at k=2i

    # Bisection bounds
    lo = max(unaff_max, 0.0)
    hi = max(unaff_max, float(affected.max()), float(quad_conv))

    # Bisection
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if mid < unaff_max - 1e-15:
            lo = mid
            continue

        # Check zero-coeff constraints
        zero_mask = coeff < 1e-15
        if zero_mask.any():
            if float(affected[zero_mask].max()) > mid + 1e-12:
                lo = mid
                continue

        # Linear upper bounds on x
        pos_mask = coeff >= 1e-15
        if pos_mask.any():
            x_ub = float(np.min((mid - affected[pos_mask]) / (2.0 * coeff[pos_mask]) + ai))
        else:
            x_ub = float('inf')

        # Quadratic constraint: x^2 <= mid - quad_conv + ai^2
        q_rhs = mid - quad_conv + ai * ai
        if q_rhs < -1e-12:
            lo = mid
            continue
        x_ub = min(x_ub, np.sqrt(max(q_rhs, 0.0)))

        if x_ub >= -1e-12:
            hi = mid
        else:
            lo = mid

    # Recover optimal x
    M_opt = hi
    pos_mask = coeff >= 1e-15
    if pos_mask.any():
        x_ub = float(np.min((M_opt - affected[pos_mask]) / (2.0 * coeff[pos_mask]) + ai))
    else:
        x_ub = float('inf')
    q_rhs = M_opt - quad_conv + ai * ai
    if q_rhs >= 0:
        x_ub = min(x_ub, float(np.sqrt(q_rhs)))
    x_star = max(x_ub, 0.0)

    return x_star, M_opt

def smart_cd(a0, time_limit, verbose=True, label=""):
    """Smart coordinate descent with exact 1D minimax via bisection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            x_star, M_new = exact_1d_minimax_step(conv, a, i)
            delta = x_star - a[i]

            if abs(delta) < 1e-15:
                continue

            # Apply update
            ai = a[i]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            a[i] = x_star

            # Update conv
            conv[k_start:k_end + 1] += 2.0 * delta * a[0:n]
            conv[2 * i] += delta * delta  # quadratic correction

            M = max(M, 0.0)
            # Recompute M for affected + unaffected
            if k_start > 0:
                M = max(M, float(conv[:k_start].max()))
            M = max(M, float(conv[k_start:k_end + 1].max()))
            if k_end < 2 * n - 2:
                M = max(M, float(conv[k_end + 1:].max()))

            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
            improved = True

        # Normalize periodically to avoid drift
        if sweeps % 50 == 0:
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / s
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver, verbose=True, label=""):
    """LP fictitious play with 1/sqrt(t) step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            iters -= 1
            time.sleep(0.1)
            continue

        # 1/sqrt(t) step size with constant c
        alpha = 8.0 / np.sqrt(iters + 1.0)
        alpha = min(alpha, 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Also check the LP solution directly
        score_b = 2.0 * n * float(fast_convolve(b_new).max())
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, M_b={M_b:.8f}")

    return best_a, best_score, iters

def perturb_and_optimize(a_best, rng, time_per_trial, n_trials, solver, label=""):
    """Simulated annealing on CD basins: perturb, optimize, accept if better."""
    best_a = a_best.copy()
    best_score = 2.0 * len(a_best) * float(fast_convolve(a_best).max())
    n = len(a_best)
    t0 = time.monotonic()
    temp = 0.005
    trial = 0

    while time.monotonic() - t0 < time_per_trial:
        trial += 1
        # Generate perturbation
        strength = temp * (1.0 + 0.5 * np.random.default_rng(trial).random())
        a_pert = best_a * (1.0 + strength * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert / s

        # Quick CD to converge
        T = time_limit_for_trial = time_per_trial / max(n_trials, 1)
        a_opt, score_opt = smart_cd(a_pert, min(T, time_per_trial - (time.monotonic() - t0) - 5), verbose=False, label=f"{label}_t{trial}")

        # Accept/reject
        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            temp *= 0.95
        else:
            if np.random.default_rng(trial * 7).random() < np.exp((best_score - score_opt) / max(temp * best_score, 1e-10)):
                best_a = a_opt
                best_score = score_opt
            temp *= 0.98

        if trial % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Perturb {label} trial={trial} ({elapsed:.0f}s): best={best_score:.8f}, temp={temp:.6f}")

    return best_a, best_score

def smoothed_gradient_step(a, beta, lr):
    """One step of smoothed gradient descent using log-sum-exp approximation to max."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = beta * (conv - max_c)
    scaled = np.clip(scaled, -100, 0)
    weights = np.exp(scaled)
    Z = weights.sum()
    if Z < 1e-30:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
        Z = 1.0
    weights = weights / Z

    # Gradient of smoothed max with respect to a[i]
    grad = np.zeros(n)
    active_k = np.where(weights > 1e-10)[0]
    for k in active_k:
        w = weights[k]
        if w < 1e-12:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        # d(conv[k])/d(a[j]) = a[k-j] + a[j] (if j != k-j) or 2*a[j] (if j == k-j)
        other = k - j
        dconv = other.astype(np.float64)
        # Actually: d(conv[k])/d(a[j]) = a[k-j] (from term j in sum) + a[j] (from term k-j in sum)
        # But if j == k-j, it's 2*a[j]
        dconv = a[other] + np.where((j == other), a[j], 0.0)
        grad[j] += 2.0 * w * dconv

    # Multiplicative update (mirror descent on simplex)
    a = a * np.exp(-lr * grad)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization:
    1. Smart CD with exact 1D minimax (bisection)
    2. LP Fictitious Play with 1/sqrt(t) schedule
    3. Smoothed gradient descent
    4. Perturb + re-optimize (simulated annealing on basins)
    5. Final fine CD polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = 2.0 * n_init
        print(f"Fallback init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Smart CD with exact 1D minimax ===
    print(f"\n=== Phase 1: Smart CD (best={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t1 = min(250.0, rem * 0.2)
    print(f"  Budget: {t1:.0f}s")
    a1, s1 = smart_cd(best_seq, t1, verbose=True, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(350.0, rem * 0.25)
    print(f"  Budget: {t2:.0f}s")
    a2, s2, it2 = lp_fictitious_play(best_seq, t2, solver=solver, verbose=True, label="p2")
    best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2")

    # === Phase 3: More CD after LP ===
    print(f"\n=== Phase 3: CD after LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(200.0, rem * 0.15)
    print(f"  Budget: {t3:.0f}s")
    a3, s3 = smart_cd(best_seq, t3, verbose=True, label="p3")
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3")

    # === Phase 4: Smoothed Gradient Descent ===
    print(f"\n=== Phase 4: Smoothed GD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(150.0, rem * 0.12)
    print(f"  Budget: {t4:.0f}s")
    a4, s4 = smoothed_gradient_descent(best_seq, t4, beta_init=30.0, lr_init=0.005, label="p4")
    best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4")
    # CD polish after smoothed GD
    rem = deadline - time.monotonic()
    t4b = min(60.0, rem * 0.05)
    if t4b > 10:
        a4b, s4b = smart_cd(best_seq, t4b, verbose=False, label="p4b")
        best_seq, best_score = check_and_update(a4b, best_seq, best_score, "p4b")

    # === Phase 5: Perturb + Re-optimize (annealing on basins) ===
    print(f"\n=== Phase 5: Perturb+Optimize (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(300.0, rem * 0.25)
    print(f"  Budget: {t5:.0f}s")
    n_trials = max(5, int(t5 / 30))
    a5, s5 = perturb_and_optimize(best_seq, rng, t5, n_trials, solver, label="p5")
    best_seq, best_score = check_and_update(a5, best_seq, best_score, "p5")

    # === Phase 6: Final intensive CD polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.7, 200.0)
    print(f"  Budget: {t6:.0f}s")
    a6, s6 = fine_cd(best_seq, t6, label="p6")
    best_seq, best_score = check_and_update(a6, best_seq, best_score, "p6")

    # === Phase 7: One last LP FP pass if time remains ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 7: Final LP FP (best={best_score:.8f}) ===")
        t7 = min(rem * 0.5, 120.0)
        print(f"  Budget: {t7:.0f}s")
        a7, s7, it7 = lp_fictitious_play(best_seq, t7, solver=solver, verbose=False, label="p7")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, "p7")
        rem = deadline - time.monotonic()
        if rem > 30:
            a7b, s7b = smart_cd(best_seq, min(rem * 0.6, 60.0), verbose=False, label="p7b")
            best_seq, best_score = check_and_update(a7b, best_seq, best_score, "p7b")

    # === Phase 8: Absolute final fine polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 8: Final Fine Polish (best={best_score:.8f}) ===")
        t8 = min(rem * 0.8, 60.0)
        a8, s8 = fine_cd(best_seq, t8, label="p8")
        best_seq, best_score = check_and_update(a8, best_seq, best_score, "p8")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
