def project_simplex(v):
    """Project vector onto probability simplex (non-negative, sum=1)."""
    n = len(v)

def compute_subgradient(a, n_active_frac=0.1):
    """Compute subgradient of max_k (a*a)_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def lbgfs_logsumexp(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=12):
    """L-BFGS with LogSumExp approximation, multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, 100 // n_stages * 3)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            return lse_objective(z_in, stage_T)

        def jac(z_in):

            pass

def simulated_annealing(a0, time_limit, T_start=0.1, T_end=0.001, rng=None):
    """Simulated annealing with structured moves for non-convex problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    current_score = best_score
    t0 = time.monotonic()
    if rng is None:
        rng = np.random.default_rng(999)

    n_total = max(1000, int(time_limit * 50))
    runs = np.geomspace(T_start, T_end, n_total)

    for it in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break
        T = runs[it]

        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Gaussian perturbation on random subset
            k = rng.integers(max(2, n // 10), n // 2)
            idx = rng.choice(n, k, replace=False)
            a_trial = a.copy()
            a_trial[idx] += rng.normal(0, T * 0.1, k)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()

        elif move_type == 1:
            # Swap: move mass from high to low contributing position
            b = fast_convolve(a)
            peak_k = int(np.argmax(b))
            contrib = np.zeros(n)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contrib[idx] = a[idx] * a[peak_k - idx]
            src = np.argmax(contrib)
            dst = rng.integers(n)
            if dst == src:
                dst = (dst + 1) % n
            frac = rng.uniform(0.01, 0.1)
            a_trial = a.copy()
            amt = a[src] * frac
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()

        elif move_type == 2:
            # Segment flip (mirror a sub-segment)
            i = rng.integers(0, n - 2)
            j = rng.integers(i + 1, n)
            a_trial = a.copy()
            a_trial[i:j+1] = a_trial[i:j+1][::-1]
            a_trial = a_trial / a_trial.sum()

        else:
            # Scale random segment
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)
            factor = rng.uniform(0.5, 1.5)
            a_trial = a.copy()
            a_trial[i:j+1] *= factor
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()

        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        delta = score_trial - current_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            current_score = score_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()

    return best_a, best_score


def peak_equalization(a0, time_limit, n_steps=50000):

        peak_gap = top_vals[0] - top_vals[-1]

        if peak_gap < 1e-8 * top_vals[0]:

            pass

            contrib = a[idx] * a[k - idx]

            src = idx[src_local]
            # Find a position that doesn't contribute to any top peak
            peak_areas = set()
            for k2 in top_k:

                amt = a[src] * frac
                a_trial = a.copy()

                a_trial[dst] += amt

def structured_init(n, rng):

        a = np.sin(np.pi * x) ** 2

        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
