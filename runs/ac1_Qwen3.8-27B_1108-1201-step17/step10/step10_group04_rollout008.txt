import numpy as np

from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    """Fast full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max_b for normalized a (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def compute_subgradient(a, b, n_active=20):
    """Compute subgradient of max_k b_k w.r.t. a using top n_active peaks."""
    n = len(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n)
    # Find top peaks
    threshold = max_b * 0.95
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    # Limit to top n_active
    if len(active) > n_active:
        vals = b[active]
        top_idx = np.argsort(vals)[::-1][:n_active]
        active = active[top_idx]
    
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            g[js:je+1] += a[k - np.arange(js, je+1)]
    g /= len(active)
    return g


def frank_wolfe_away(a0, time_limit, n_active=20, use_away=True):
    """Frank-Wolfe with away steps for minimizing max autocorrelation peak.
    Works on normalized a (sum=1, a>=0)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

        a = a / s

    b = fast_convolve(a)

    t0 = time.monotonic()
    iter_count = 0
    support = set(np.where(a > 1e-10)[0].tolist())

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()

        g = compute_subgradient(a, b, n_active)

        # Towards step: argmin of g
        v_idx = int(np.argmin(g))

        # Away step: argmax of g over support
        if use_away and len(support) > 1:
            support_arr = np.array(list(support))
            away_idx = int(support_arr[np.argmax(g[support_arr])])
        else:
            away_idx = None

        # Compute step size
        iter_count += 1
        t_step = 2.0 / (iter_count + 2.0)

        # Try towards step

        a_trial[v_idx] += t_step
        a_trial -= t_step * (1.0 / n)  # Maintain sum=1

        ss = a_trial.sum()
        if ss > 1e-15:
            a_trial /= ss
        score_trial = 2.0 * n * fast_convolve(a_trial).max()

        # Try away step

        best_trial_score = score_trial
        if away_idx is not None and away_idx != v_idx:
            a_away = a.copy()
            a_away[v_idx] += t_step
            a_away[away_idx] -= t_step
            a_away = np.maximum(a_away, 0)
            ss = a_away.sum()
            if ss > 1e-15:

            score_away = 2.0 * n * fast_convolve(a_away).max()

            a = a_trial

        if iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-LS iter={iter_count} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=8):
    """LogSumExp gradient descent with annealing temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage, T in enumerate(temps):
        stage_t0 = time.monotonic()
        stage_limit = min(time_limit - (t0 - time.monotonic()), 
                          max(5.0, time_limit * 0.6 / n_stages))

        lr = 0.1
        no_improve = 0
        prev_sc = 2.0 * n * fast_convolve(a).max()

        while time.monotonic() - stage_t0 < stage_limit and (time.monotonic() - t0) < time_limit:
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            if max_Tb > 500:
                max_Tb = 500
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient: weighted average of shifts
            g = np.zeros(n)
            # Only use significant weights
            sig = w > 1e-8
            for k in np.where(sig)[0]:
                js = max(0, int(k) - n + 1)
                je = min(n - 1, int(k))
                if js <= je:
                    g[js:je+1] += w[k] * a[k - np.arange(js, je+1)]

            # Gradient step with projection
            a_new = a - lr * g
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = a
            else:
                a_new /= ss

            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < prev_sc - 1e-14:
                a = a_new
                prev_sc = sc
                no_improve = 0
                lr = min(lr * 1.05, 1.0)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a.copy()
            else:
                no_improve += 1
                lr *= 0.8
                if no_improve > 100:
                    break

        elapsed = time.monotonic() - t0
        print(f"    LSE stage {stage+1}/{n_stages} (T={T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def spectral_equalization(a0, time_limit, n_iters=100000):
    """Iterative spectral equalization: flatten |FFT(a)| while keeping a>=0."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Use full FFT for complex
    A = fft(a)
    phase = np.angle(A)
    magnitude = np.abs(A)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Try to flatten the magnitude
        target_mag = magnitude.mean()
        # Soft equalization
        alpha = 0.05
        new_mag = magnitude + alpha * (target_mag - magnitude)
        new_mag = np.maximum(new_mag, magnitude * 0.5)  # Don't reduce too much

        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new /= ss

        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            A = fft(a)
            phase = np.angle(A)
            magnitude = np.abs(A)

        if it % 10000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, n_trials=50000):
    """Local search that targets the highest peaks in the autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            continue

        # Pick the largest contributor
        src_local = int(np.argmax(contrib))
        src = idx[src_local]

        # Try moving mass from src to various destinations
        # Strategy 1: move to position that doesn't contribute to top peaks
        # Strategy 2: move to a random position
        n_dest = min(5, n)
        best_sc = best_score
        best_a_trial = None

        # Top peaks to avoid
        top_peaks = np.argsort(b)[::-1][:10]
        peak_zones = set()
        for pk in top_peaks:
            js2 = max(0, int(pk) - n + 1)
            je2 = min(n - 1, int(pk))
            peak_zones.update(range(js2, je2 + 1))

        # Candidate destinations
        candidates = []
        for d in range(n):
            if d != src and d not in peak_zones:
                candidates.append(d)
        if not candidates:
            candidates = [d for d in range(n) if d != src]

        # Sample some candidates
        if len(candidates) > n_dest:
            sampled = rng.choice(candidates, n_dest, replace=False)
        else:
            sampled = np.array(candidates)

        for dst in sampled:
            for frac in [0.02, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_sc:
                    best_sc = sc
                    best_a_trial = a_trial

        if best_a_trial is not None:
            a = best_a_trial
            best_score = best_sc
            best_a = a.copy()

        if trial % 5000 == 0 and trial > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave trial={trial} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=100000):
    """Randomized coordinate descent: update pairs of coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a random pair
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        # Try transferring mass from i to j
        for frac in [0.01, 0.03, 0.1]:
            amt = a[i] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial /= ss
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-13:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                break

        # Also try symmetric transfer j->i
        for frac in [0.01, 0.03, 0.1]:
            amt = a[j] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[j] -= amt
            a_trial[i] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial /= ss
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-13:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                break

        if rnd % 20000 == 0 and rnd > 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent round={rnd} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng, method=0):
    """Generate diverse structured initializations."""
    if method == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif method == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif method == 2:
        # Fejer-like
        x = np.linspace(-1, 1, n)
        a = (1 - x**2) * np.sinc(x * 2) ** 2
    elif method == 3:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.linspace(0, 1, n)))
    elif method == 4:
        # Random smooth
        n_f = max(3, n // 10)
        freqs = rng.integers(1, n // 2, n_f) * (2 * np.pi / n)
        phases = rng.uniform(0, 2 * np.pi, n_f)
        amps = rng.uniform(0.1, 1.0, n_f)
        x = np.arange(n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(f * x + p)
    elif method == 5:
        # Product of sines (creates multiple peaks in spectrum)
        x = np.linspace(0, 2 * np.pi, n)
        a = np.abs(np.sin(3 * x) * np.sin(7 * x)) + 0.1
    elif method == 6:
        # Bump with oscillation
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x**2) * (1 + 0.3 * np.cos(10 * x))
    else:
        # Pure random (smoothed)
        a = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_at_length(a_init, time_budget, label=""):
    """Run multi-method optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()

    def try_update(a, src):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 1e-15:
            return
        a = a / ss
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            print(f"      [{src}]: {sc:.6f} (n={n})")

    t0 = time.monotonic()

    # Method 1: Frank-Wolfe with away (fast, many iterations)
    rem = time_budget - (time.monotonic() - t0)
    t1 = min(rem * 0.35, 200.0)
    if t1 > 10:
        print(f"    Phase FW-Away: budget={t1:.0f}s")
        a1, _ = frank_wolfe_away(best_a, t1, n_active=15)
        try_update(a1, f"fw_away_{label}")

    # Method 2: LogSumExp gradient descent
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.25, 150.0)
    if t2 > 10:
        print(f"    Phase LSE-GD: budget={t2:.0f}s")
        a2, _ = lse_gradient_descent(best_a, t2, T_start=30.0, T_end=0.5, n_stages=6)
        try_update(a2, f"lse_gd_{label}")

    # Method 3: Spectral equalization
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.15, 80.0)
    if t3 > 5:
        print(f"    Phase Spectral: budget={t3:.0f}s")
        a3, _ = spectral_equalization(best_a, t3, n_iters=50000)
        try_update(a3, f"spec_{label}")

    # Method 4: Peak shaving
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.15, 80.0)
    if t4 > 5:
        print(f"    Phase PeakShave: budget={t4:.0f}s")
        a4, _ = peak_shaving(best_a, t4, n_trials=30000)
        try_update(a4, f"peak_{label}")

    # Method 5: Coordinate descent (if time remains)
    rem = time_budget - (time.monotonic() - t0)
    t5 = min(rem * 0.10, 60.0)
    if t5 > 5:
        print(f"    Phase CoordDesc: budget={t5:.0f}s")
        a5, _ = coordinate_descent(best_a, t5, n_rounds=50000)
        try_update(a5, f"coord_{label}")

    # Final: one more FW pass on whatever is best
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        t6 = min(rem * 0.5, 30.0)
        if t6 > 3:
            a6, _ = frank_wolfe_away(best_a, t6, n_active=10)
            try_update(a6, f"fw_final_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-method, multi-length optimization."""
    rng = np.random.default_rng(seed)
    abs_deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def global_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = structured_init(1000, rng, method=0)
        p_score = evaluate_safe(parent)
        print(f"Default init: score={p_score:.6f}")
        best_seq = parent.copy()
        best_score = p_score

    n_parent = len(parent)

    # Phase 1: Optimize at parent length with multiple methods
    print(f"\n=== Phase 1: Optimize at n={n_parent} (best={best_score:.6f}) ===")
    rem = abs_deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(parent, t1, label=f"n{n_parent}")
        global_try_update(a_out, f"opt_main_n{n_parent}")

    # Phase 2: Try different lengths
    print(f"\n=== Phase 2: Length exploration (best={best_score:.6f}) ===")
    # Try lengths: shorter and longer
    candidate_lengths = []
    for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.2]:
        ln = int(n_parent * mult)
        if 100 <= ln <= 3000:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))

    # Prioritize lengths near the current best
    for ln in candidate_lengths:
        if ln == n_parent:
            continue
        rem = abs_deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.20, 120.0)
        if t_alloc < 30:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        # Initialize by interpolating best
        init = interpolate_sequence(best_seq, ln)
        a_out, _ = optimize_at_length(init, t_alloc, label=f"n{ln}")
        global_try_update(a_out, f"opt_n{ln}")

    # Phase 3: Diverse initializations at the best length
    print(f"\n=== Phase 3: Diverse inits at n={len(best_seq)} (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for method in [2, 3, 4, 5]:
        rem = abs_deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.15, 80.0)
        if t_alloc < 15:
            break
        print(f"  Diverse init method={method}, budget={t_alloc:.0f}s")
        init = structured_init(n_best, rng, method=method)
        # Quick optimization
        a_out, _ = frank_wolfe_away(init, t_alloc, n_active=15)
        global_try_update(a_out, f"diverse_m{method}")
        # Also try LSE
        rem = abs_deadline - time.monotonic()
        if rem > 20:
            a_out2, _ = lse_gradient_descent(a_out, min(rem * 0.3, 40.0), T_start=20.0, T_end=0.5, n_stages=4)
            global_try_update(a_out2, f"diverse_m{method}_lse")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = abs_deadline - time.monotonic()
    if rem > 10:
        # More FW iterations
        t_fw = min(rem * 0.30, 100.0)
        if t_fw > 5:
            a_fin, _ = frank_wolfe_away(best_seq, t_fw, n_active=10)
            global_try_update(a_fin, "polish_fw")

        # LSE
        rem = abs_deadline - time.monotonic()
        if rem > 10:
            t_lse = min(rem * 0.25, 60.0)
            if t_lse > 5:
                a_fin2, _ = lse_gradient_descent(best_seq, t_lse, T_start=10.0, T_end=0.3, n_stages=5)
                global_try_update(a_fin2, "polish_lse")

        # Peak shaving
        rem = abs_deadline - time.monotonic()
        if rem > 10:
            t_pk = min(rem * 0.20, 50.0)
            if t_pk > 5:
                a_fin3, _ = peak_shaving(best_seq, t_pk, n_trials=20000)
                global_try_update(a_fin3, "polish_peak")

        # Coordinate descent
        rem = abs_deadline - time.monotonic()
        if rem > 10:
            t_cd = min(rem * 0.20, 40.0)
            if t_cd > 5:
                a_fin4, _ = coordinate_descent(best_seq, t_cd, n_rounds=30000)
                global_try_update(a_fin4, "polish_coord")

        # One final FW
        rem = abs_deadline - time.monotonic()
        if rem > 5:
            a_fin5, _ = frank_wolfe_away(best_seq, rem * 0.5, n_active=8)
            global_try_update(a_fin5, "final_fw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
