```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast gradient-based L-BFGS
with multi-stage LogSumExp cooling, FFT-accelerated gradient, multiple
structured initializations across varying lengths, symmetry exploitation,
and peak-directed coordinate descent polish.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_gradient_fft(p, a):
    """
    Compute c_j = sum_k p[k] * a[k-j] for j=0..n-1 using FFT.
    p has length 2n-1, a has length n.
    """
    n = len(a)
    # We need cross-correlation: c_j = sum_k p[k] * a[k-j]
    # This equals (p * rev_a)[j + n - 1] where rev_a[i] = a[n-1-i]
    # and * is linear convolution
    rev_a = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    P = rfft(p, fft_size)
    AR = rfft(rev_a, fft_size)
    conv = irfft(P * AR, fft_size)
    # c_j = conv[j + n - 1] for j in 0..n-1
    return conv[n - 1: 2 * n - 1]


def lse_func_grad(z, n, T):
    """
    LogSumExp-smoothed objective and gradient w.r.t. z.
    a = softmax(z), objective = 2n * logsumexp(conv(a,a) / T) * T / T
    Actually: f = 2n * (logsumexp(b/T)) / 1  where logsumexp includes the 1/T scaling.
    
    Let F = 2n * logsumexp(b / T) = 2n * (max(b)/T + log(sum(exp((b - max(b))/T))))
    dF/dz_i via chain rule.
    """
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s

    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    LSE = max_b / T + log_sum
    f = 2.0 * n * LSE

    # p_k = exp(b_k/T) / sum(exp(b'/T))
    p = exp_shifted / exp_shifted.sum()

    # c_j = sum_k p_k * a_{k-j}  (this is 1/2 of df/da_j divided by 2n)
    # Actually: df/da_j = 2n * (2/T) * c_j = (4n/T) * c_j
    # Then: df/dz_i = a_i * (df/da_i - sum_j (df/da_j) * a_j)
    
    c = compute_gradient_fft(p, a)
    df_da = (4.0 * n / T) * c
    S = np.dot(df_da, a)
    grad = a * (df_da - S)

    return f, grad


def lse_func(z, n, T):
    f, _ = lse_func_grad(z, n, T)
    return f


def lse_jac(z, n, T):
    _, g = lse_func_grad(z, n, T)
    return g


def optimize_lbfgs(a0, time_budget, T_stages=None, memory=50):
    """
    Multi-stage L-BFGS optimization with LogSumExp smoothing.
    a0: initial point (normalized, non-negative, length n)
    """
    n = len(a0)
    t0 = time.monotonic()
    z = np.log(np.maximum(a0, 1e-300))
    best_a = a0.copy()
    best_score = eval_objective(a0)

    if T_stages is None:
        # Fine cooling schedule
        T_stages = [n * 0.5, n * 0.25, n * 0.125, n * 0.0625,
                    n * 0.03, n * 0.015, n * 0.008, n * 0.004,
                    n * 0.002, n * 0.001, n * 0.0005, n * 0.00025,
                    n * 0.000125, n * 0.00006, n * 0.00003]

    n_stages = len(T_stages)
    # Allocate iterations proportional to stage difficulty (lower T needs more)
    base_iters = 200
    iters_per_stage = [int(base_iters * (1.0 + 2.0 * (i / max(1, n_stages - 1)))) for i in range(n_stages)]

    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_budget:
            break
        t_stage_start = time.monotonic()
        max_iter = iters_per_stage[stage_idx]

        # Adaptive: if stage is taking too long, reduce iterations
        elapsed_total = time.monotonic() - t0
        remaining = time_budget - elapsed_total
        if remaining < 2:
            break
        est_time_per_iter = 0.005  # rough estimate for n=1200
        max_iter = min(max_iter, int(remaining / est_time_per_iter) - 5)
        max_iter = max(max_iter, 20)

        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac, args=(n, T),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': max_iter * 200,
                         'maxls': 50, 'memory': memory, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS stage {stage_idx} failed: {e}")
            break

        # Evaluate and update best
        a_curr = np.exp(z - z.max())
        a_curr = a_curr / a_curr.sum()
        score = eval_objective(a_curr)
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()

        # Also check symmetrized version
        a_sym = (a_curr + a_curr[::-1]) / 2.0
        a_sym = np.maximum(a_sym, 0)
        ss = a_sym.sum()
        if ss > 1e-15:
            a_sym = a_sym / ss
            score_sym = eval_objective(a_sym)
            if score_sym < best_score:
                best_score = score_sym
                best_a = a_sym.copy()
                z = np.log(np.maximum(best_a, 1e-300))

        elapsed = time.monotonic() - t0
        if stage_idx % 3 == 0 or stage_idx == n_stages - 1:
            print(f"    T={T:.6f} ({elapsed:.1f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_coord_descent(a0, time_budget, n_steps=100000):
    """
    Peak-directed coordinate descent: reduce top contributors to max peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, eval_objective(a0)
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Get top peaks (within 1% of max)
        threshold = max_b * 0.999
        top_peaks = np.where(b >= threshold)[0]
        # Limit to top 5
        if len(top_peaks) > 5:
            peak_vals = b[top_peaks]
            top_peaks = top_peaks[np.argsort(peak_vals)[::-1][:5]]

        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            # Try reducing top contributors
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-14:
                    continue
                for factor in [0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        break
                    # Redistribute proportionally to other elements
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try symmetric reduction of top 2 contributors to the max peak
            peak_k = int(np.argmax(b))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top2 = idx[np.argsort(contribs)[::-1][:2]]
            a_trial = a.copy()
            for pos in top2:
                if a[pos] > 1e-14:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 200:
                    break

        if step % 20000 == 0 and step > 0:
            print(f"    PeakCD ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def subgradient_step(a0, time_budget, n_steps=50000):
    """
    Subgradient method on the active peak set with adaptive step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, eval_objective(a0)
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: within 0.5% of max
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: average of a[k-j] over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g *= 2.0  # factor of 2 from the quadratic

        # Update with projection onto simplex
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)

        if step % 10000 == 0 and step > 0:
            print(f"    SubGrad ({time.monotonic()-t0:.0f}s): score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score

    return a / a.sum()


def make_init_gaussian(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-2 * (x - 0.5) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_bell_asym(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-3 * (x - 0.35) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_flat_bump(n):
    """Try a trapezoidal shape."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    # Taper edges
    taper = max(1, n // 10)
    for i in range(taper):
        a[i] *= (i + 1) / taper
        a[n - 1 - i] *= (i + 1) / taper
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_multi_peak(n, rng):
    """Random multi-peak initialization."""
    n_peaks = rng.integers(2, 6)
    centers = rng.uniform(0, 1, n_peaks)
    widths = rng.uniform(0.05, 0.2, n_peaks)
    a = np.zeros(n)
    x = np.linspace(0, 1, n)
    for c, w in zip(centers, widths):
        a += np.exp(-0.5 * ((x - c) / w) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-15)
    return f_interp / f_interp.sum()


def coarse_to_fine(parent, coarse_n, fine_n, time_budget, T_stages_coarse=None):
    """
    Optimize at coarse resolution, then upsample and refine at fine resolution.
    """
    # Downsample parent to coarse_n
    a_coarse = interpolate_sequence(parent, coarse_n)
    # Optimize at coarse level
    a_coarse_opt, _ = optimize_lbfgs(a_coarse, time_budget * 0.3, T_stages_coarse)
    # Upsample to fine level
    a_fine_init = interpolate_sequence(a_coarse_opt, fine_n)
    # Refine at fine level
    a_fine_opt, _ = optimize_lbfgs(a_fine_init, time_budget * 0.7)
    return a_fine_opt


def symmetrize_check(a):
    """Try symmetrization and return best version."""
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    ss = a_sym.sum()
    if ss < 1e-15:
        return a
    a_sym = a_sym / ss
    s_orig = eval_objective(a)
    s_sym = eval_objective(a_sym)
    if s_sym < s_orig:
        return a_sym
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent sequence
    2. Try multiple (length, init) combinations with L-BFGS
    3. Coarse-to-fine refinement for promising configs
    4. Final polish with coordinate descent + subgradient
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_objective(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) [was {old:.8f}]")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_objective(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 1200

    # === Phase 1: Coarse-to-fine from parent ===
    print(f"\n=== Phase 1: Coarse-to-fine (n={n_best}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t1 = min(rem * 0.15, 150.0)
        coarse_n = max(200, n_best // 4)
        print(f"  Coarse n={coarse_n}, t={t1:.0f}s")
        a_cf = coarse_to_fine(best_seq, coarse_n, n_best, t1)
        try_update(a_cf, "coarse_fine")
        # Also try symmetrized
        try_update(symmetrize_check(a_cf), "coarse_fine_sym")

    # === Phase 2: Direct L-BFGS from parent with fine cooling ===
    print(f"\n=== Phase 2: L-BFGS parent (n={n_best}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t2 = min(rem * 0.2, 200.0)
        print(f"  Budget: {t2:.0f}s")
        T_stages = [n_best * f for f in
                    [0.3, 0.15, 0.075, 0.04, 0.02, 0.01, 0.005,
                     0.0025, 0.00125, 0.0006, 0.0003, 0.00015,
                     0.00008, 0.00004, 0.00002]]
        a_res, _ = optimize_lbfgs(best_seq, t2, T_stages=T_stages)
        try_update(a_res, "lbfgs_parent")
        try_update(symmetrize_check(a_res), "lbfgs_parent_sym")

    # === Phase 3: Different lengths ===
    print(f"\n=== Phase 3: Different lengths ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        lengths_to_try = []
        # Try larger and smaller than current
        for n_try in [int(n_best * 0.75), int(n_best * 1.25), int(n_best * 1.5)]:
            if n_try >= 100 and n_try <= 5000 and n_try != n_best:
                lengths_to_try.append(n_try)
        # Also try some fixed good lengths
        for n_fixed in [800, 1000, 1500, 2000]:
            if n_fixed not in lengths_to_try and n_fixed != n_best:
                lengths_to_try.append(n_fixed)
        lengths_to_try = sorted(set(lengths_to_try))[:4]

        for n_try in lengths_to_try:
            if time.monotonic() > deadline - 40:
                break
            t_each = min(rem / (len(lengths_to_try) + 1), 120.0)
            if t_each < 20:
                break
            print(f"  n={n_try}, budget={t_each:.0f}s")
            a_init = interpolate_sequence(best_seq, n_try)
            T_stages = [n_try * f for f in
                        [0.2, 0.1, 0.05, 0.025, 0.0125, 0.006,
                         0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]]
            a_res, _ = optimize_lbfgs(a_init, t_each, T_stages=T_stages)
            try_update(a_res, f"len_{n_try}")
            try_update(symmetrize_check(a_res), f"len_{n_try}_sym")
            rem = deadline - time.monotonic()

    # === Phase 4: Alternative initializations ===
    print(f"\n=== Phase 4: Alt inits (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        inits = [
            ("uniform", make_init_uniform(n_cur)),
            ("cosine", make_init_cosine(n_cur)),
            ("gaussian", make_init_gaussian(n_cur)),
            ("bell_asym", make_init_bell_asym(n_cur)),
            ("flat_bump", make_init_flat_bump(n_cur)),
            ("multi_peak", make_init_multi_peak(n_cur, rng)),
        ]
        # Blend with parent for warm start
        for name, init in inits:
            if time.monotonic() > deadline - 30:
                break
            t_each = min(rem / (len(inits) + 1), 60.0)
            if t_each < 15:
                break
            # Blend: 70% parent + 30% new init
            a_blend = 0.7 * best_seq + 0.3 * init
            a_blend = np.maximum(a_blend, 1e-15)
            a_blend = a_blend / a_blend.sum()
            T_stages = [n_cur * f for f in
                        [0.1, 0.05, 0.025, 0.0125, 0.006, 0.003,
                         0.0015, 0.0008, 0.0004, 0.0002, 0.0001]]
            a_res, _ = optimize_lbfgs(a_blend, t_each, T_stages=T_stages)
            try_update(a_res, f"init_{name}")
            rem = deadline - time.monotonic()

    # === Phase 5: Iterated L-BFGS on best (second pass) ===
    print(f"\n=== Phase 5: L-BFGS pass 2 (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t5 = min(rem * 0.25, 150.0)
        n_cur = len(best_seq)
        T_stages = [n_cur * f for f in
                    [0.01, 0.005, 0.0025, 0.00125, 0.0006, 0.0003,
                     0.00015, 0.00008, 0.00004, 0.00002, 0.00001, 0.000005]]
        print(f"  Fine cooling, budget={t5:.0f}s")
        a_res, _ = optimize_lbfgs(best_seq, t5, T_stages=T_stages)
        try_update(a_res, "lbfgs_pass2")
        try_update(symmetrize_check(a_res), "lbfgs_pass2_sym")

    # === Phase 6: Subgradient refinement ===
    print(f"\n=== Phase 6: Subgradient ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t6 = min(rem * 0.2, 60.0)
        a_sg, _ = subgradient_step(best_seq, t6, n_steps=80000)
        try_update(a_sg, "subgrad")
        try_update(symmetrize_check(a_sg), "subgrad_sym")

    # === Phase 7: Peak coordinate descent ===
    print(f"\n=== Phase 7: Peak CD ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.3, 60.0)
        a_cd, _ = peak_coord_descent(best_seq, t7, n_steps=100000)
        try_update(a_cd, "peak_cd")

    # === Phase 8: Final L-BFGS polish (very low T) ===
    print(f"\n=== Phase 8: Final polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.4, 40.0)
        n_cur = len(best_seq)
        T_stages = [n_cur * f for f in
                    [0.0001, 0.00005, 0.000025, 0.00001, 0.000005, 0.000002, 0.000001]]
        a_fin, _ = optimize_lbfgs(best_seq, t8, T_stages=T_stages)
        try_update(a_fin, "final_polish")

    # === Phase 9: Final coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        a_cd2, _ = peak_coord_descent(best_seq, min(rem * 0.6, 30.0), n_steps=50000)
        try_update(a_cd2, "final_cd")

    # Final symmetrization attempt
    if best_seq is not None:
        a_sym_final = symmetrize_check(best_seq)
        try_update(a_sym_final, "final_sym")

    # Final verification
    print(f"\nTotal time: {time.monotonic() - t_start:.1f}s")
    if best_seq is not None:
        fs = eval_objective(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
