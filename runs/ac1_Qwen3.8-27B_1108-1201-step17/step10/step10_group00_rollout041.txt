import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_function(a):
    """Compute the score 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def gs_flatten(a0, n_iters=1000, target_dc_weight=0.0):
    """Gerchberg-Saxton style power spectrum flattening.
    
    Minimizes the max of the (approximately) circular convolution
    by driving the power spectrum toward flatness.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_function(a)
    
    # Target magnitude: flat spectrum
    # For sum(a) = 1: DC = 1, others should be ~ 1/sqrt(n) for flat
    # But we can't achieve this exactly, so we use it as a target
    target_mag = np.ones(n) / np.sqrt(n)  # for rfft length n
    target_mag[0] = 1.0  # DC component (sum = 1)
    
    # Actually, for rfft of a length-n sequence, we get n//2+1 bins.
    # Let me use full FFT for clarity.
    
    from scipy.fft import fft, ifft
    
    for it in range(n_iters):
        # FFT
        A = fft(a)
        
        # Replace magnitudes with target (keep phases)
        mag = np.abs(A)
        # Avoid division by zero
        angles = np.angle(A)
        
        # Blend between current and target magnitude
        # Use a soft approach: move toward target
        new_mag = target_mag.copy()
        # Keep some of the current magnitude to avoid discontinuities
        # Actually, pure GS: replace all magnitudes
        A_new = new_mag * np.exp(1j * angles)
        
        # Inverse FFT
        a_new = np.real(ifft(A_new))
        
        # Project onto non-negativity
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check score
        sc = score_function(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

Now, let me also think about the L-BFGS approach. The LogSumExp smoothing is:
LSE_T(b) = T * log(sum(exp(b[j]/T)))

This approximates max(b) as T → 0. The gradient w.r.t. a[i] is:
d/d(a[i]) LSE_T = sum_k w[k] * d/d(a[i]) b[k]
where w[k] = exp(b[k]/T) / sum_j exp(b[j]/T)

And d/d(a[i]) b[k] = 2*a[k-i] (for valid k-i).

So the gradient is:
grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)

This is a "weighted average" of shifted versions of a, weighted by the softmax of b.

For small T, the weights concentrate on the max, and the gradient approaches the subgradient. For large T, the weights are spread out, and the gradient is smoother.

The L-BFGS optimization in the "z-space" (where a[i] = exp(z[i]) / sum(exp(z))) is the standard approach for optimizing over the simplex.
