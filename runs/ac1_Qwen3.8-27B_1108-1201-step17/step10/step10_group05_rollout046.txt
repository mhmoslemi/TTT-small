import numpy as np
from scipy.fft import rfft, irfft
import time
import math

_lp_solver_cached = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.PDLP, cp.SCS, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS

def compute_pressure(a, lambda_vec):
    """Compute pressure on each coordinate given dual lambda distribution on lags."""
    n = len(a)
    n_lag = len(lambda_vec)
    # pressure_i = sum_k lambda_k * a_{|k - i|} (circular-ish, but bounded)
    # p_i = sum_k lambda_k * a_{k-i} where 0 <= k-i < n
    # This is a cross-correlation-like operation
    pressure = np.zeros(n)
    # For efficiency, use FFT-based approach
    # p_i = sum_k lambda_k * a_{k-i} for valid k-i
    # = sum_j a_j * lambda_{j+i} for valid j+i (where lambda is extended)
    # This is the cross-correlation of a and lambda, evaluated at specific lags
    
    # Use direct computation for small lambda support, FFT for large
    active = np.where(lambda_vec > 1e-12)[0]
    if len(active) < n // 2:
        for k in active:
            w = lambda_vec[k]
            if w < 1e-14:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            pressure[j] += w * a[k - j]
    else:
        # FFT-based: pressure = ifft(fft(a) * conj(fft(lambda_padded)))
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        L = rfft(lambda_vec, fft_size)
        # Cross-correlation
        corr = irfft(A * np.conj(L), fft_size)[:size]
        # pressure[i] = corr[n-1+i] ... need to get the right indexing
        # Actually: (a ⋆ λ)_m = sum_i a_i * λ_{i+m}
        # We want p_i = sum_k λ_k * a_{k-i} = sum_j a_j * λ_{j+i} = (a ⋆ λ)_i
        # The FFT cross-correlation gives (a ⋆ λ)_m for m in range
        # corr[m] = sum_i a_i * conj(λ)_{i+m} = sum_i a_i * λ_{i+m} (since λ is real)
        # So pressure = corr[0:n]
        pressure = corr[:n]
    
    return pressure

def solve_lp_and_dual(a_fixed):
    """Solve the LP best response and get dual variables."""
    import cvxpy as cp
    solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint Tb <= M*1
    # T is (2n-1) x n, T[k,i] = a_fixed[k-i] if 0 <= k-i < n else 0
    # Instead of building T explicitly (memory), use the structure
    
    # For moderate n, build the matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None, None
    b_val = b_val / s
    
    # Extract dual variables (lambda) from the constraints
    # The dual of the first constraint (T @ b <= M) gives lambda
    lambda_val = None
    try:
        # In cvxpy, dual values are on the constraints
        # constraints[0] is T @ b_var <= M_var
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            lambda_val = np.asarray(dual_val, dtype=np.float64)
            lambda_val = np.maximum(lambda_val, 0)
            ls = lambda_val.sum()
            if ls > 1e-15:
                lambda_val = lambda_val / ls
            else:
                lambda_val = None
    except Exception:
        pass
    
    return b_val, float(M_var.value), lambda_val, T

def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with smart 1D candidate evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            top_k = min(5, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.5 * ai)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 100 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
                          0.99, 0.985, 0.98, 0.97,
                          1.0001, 1.0002, 1.0005, 1.001, 1.002, 1.005,
                          1.01, 1.015, 1.02, 1.03]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def lp_guided_optimization(a0, time_limit, label=""):
    """LP-guided optimization using dual pressure for water-filling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    eta = 0.1  # step size for mirror descent
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem_time = time_limit - (time.monotonic() - t0)
        
        # Solve LP to get best response and dual
        b, M_val, lambda_vec, _ = solve_lp_and_dual(a)
        
        if b is None:
            break
        
        # Update toward best response (fictitious play step)
        alpha = min(1.0, 5.0 / (iters + 2))
        a_new = (1 - alpha) * a + alpha * b
        a_new = simplex_project(np.maximum(a_new, 0))
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        # Also try the LP solution directly
        score_b = 2.0 * n * fast_convolve(b).max()
        if score_b < best_score - 1e-15:
            best_score = score_b
            best_a = b.copy()
        
        # Water-filling step using dual pressure
        if lambda_vec is not None and lambda_vec.sum() > 1e-10:
            pressure = compute_pressure(a, lambda_vec)
            # Mirror descent: move mass from high pressure to low pressure
            log_a = np.log(np.maximum(a, 1e-30))
            # Normalize pressure to have mean 0 for the active set
            active = a > 1e-10
            if active.sum() > 1:
                p_active = pressure[active]
                p_mean = p_active.mean()
                # Scaled pressure
                p_scaled = np.zeros(n)
                p_scaled[active] = (pressure[active] - p_mean)
                log_a -= eta * p_scaled
                log_a -= log_a.max()
                a_wf = np.exp(log_a)
                a_wf = simplex_project(a_wf)
                
                score_wf = 2.0 * n * fast_convolve(a_wf).max()
                if score_wf < best_score - 1e-15:
                    best_score = score_wf
                    best_a = a_wf.copy()
                
                # Take a step toward the water-filling solution
                a_new2 = 0.7 * a + 0.3 * a_wf
                a_new2 = simplex_project(np.maximum(a_new2, 0))
                score_af = 2.0 * n * fast_convolve(a_new2).max()
                if score_af < best_score - 1e-15:
                    best_score = score_af
                    best_a = a_new2.copy()
                a = a_new2
        
        # Adaptive step size
        if iters % 10 == 0:
            eta = max(0.01, eta * 0.95)
        
        if iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Guided {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")
        
        if rem_time < 30:
            break

    return best_a, best_score

def block_coord_descent(a0, time_limit, block_size=3, label=""):
    """Block coordinate descent: optimize blocks of coordinates together."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(0, n, block_size):
            block = slice(i, min(i + block_size, n))
            block_idx = np.arange(block.start, block.stop)
            a_block = a[block].copy()
            if a_block.sum() < 1e-15:
                continue
            
            # Find all affected lags
            k_min = block.start
            k_max = min(block.stop - 1 + n - 1, 2 * n - 2)
            
            # For block moves, the update is quadratic in the block variables
            # conv[k] = original + sum_j in block delta_j * (a[k-j] + a[j-k] if valid)
            #           + sum_{j1,j2 in block} delta_j1 * delta_j2 * delta_{j1+j2-k offset}
            # This is complex; use random search within the block instead
            
            best_block_delta = np.zeros(len(block_idx))
            best_block_max = M
            
            # Try several random perturbations of the block
            rng_block = np.random.default_rng(sweep + i)
            for trial in range(20):
                if trial == 0:
                    delta = np.zeros(len(block_idx))
                else:
                    delta = rng_block.standard_normal(len(block_idx)) * a_block.mean() * 0.5
                    delta = np.clip(delta, -a_block, None)
                
                a_test = a.copy()
                a_test[block_idx] = a_block + delta
                a_test = np.maximum(a_test, 0)
                
                # Compute new conv (only the affected parts)
                # For efficiency, compute the full conv
                new_conv = fast_convolve(a_test)
                new_max = float(new_conv.max())
                
                if new_max < best_block_max - 1e-14:
                    best_block_max = new_max
                    best_block_delta = delta.copy()
                    conv = new_conv
                    M = new_max
            
            if np.any(best_block_delta != 0) and best_block_max < M - 1e-14:
                a[block_idx] = np.maximum(a_block + best_block_delta, 0)
                # Recompute conv
                conv = fast_convolve(a)
                M = best_block_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved and sweeps > 20:
            break

    return best_a, best_score

def simulated_annealing_search(a0, time_limit, rng, label=""):
    """Simulated annealing to escape local optima."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    score = 2.0 * n * M
    best_a = a.copy()
    best_score = score
    t0 = time.monotonic()
    
    T_init = 0.02
    T_final = 0.001
    total_iters = 500
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = it / total_iters
        T = T_init * (T_final / T_init) ** progress
        
        # Generate a perturbation
        pert_type = rng.random()
        if pert_type < 0.3:
            # Small random perturbation
            noise = rng.standard_normal(n) * 0.01
            a_new = a * (1 + noise)
        elif pert_type < 0.5:
            # Move mass between random pairs
            i, j = rng.integers(0, n, 2)
            if i != j:
                amount = a[min(i,j)] * 0.1 * rng.random()
                a_new = a.copy()
                a_new[i] -= amount
                a_new[j] += amount
            else:
                a_new = a.copy()
        elif pert_type < 0.7:
            # Smooth perturbation
            kernel = np.ones(5) / 5.0
            a_smooth = np.convolve(a, kernel, mode='same')
            a_new = (1 - 0.1) * a + 0.1 * a_smooth
        else:
            # Peak-targeted reduction
            conv = fast_convolve(a)
            peak_k = conv.argmax()
            i_start = max(0, peak_k - n + 1)
            i_end = min(n - 1, peak_k)
            candidates = np.arange(i_start, i_end + 1)
            # Reduce the coordinate with highest contribution to the peak
            contribs = a[candidates] * a[peak_k - candidates]
            worst = candidates[np.argmax(contribs)]
            a_new = a.copy()
            a_new[worst] *= 0.9
            # Redistribute
            diff = a[worst] - a_new[worst]
            a_new += diff / n
        
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        new_conv = fast_convolve(a_new)
        new_M = float(new_conv.max())
        new_score = 2.0 * n * new_M
        
        if new_score < score:
            a = a_new
            score = new_score
            conv = new_conv
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
        else:
            # Accept with probability exp(-(new_score - score) / T)
            delta = new_score - score
            if delta < 10 * T:
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    a = a_new
                    score = new_score
                    conv = new_conv
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA {label} it={it} ({elapsed:.0f}s): current={score:.6f}, best={best_score:.6f}, T={T:.4f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def nesterov_smoothed_md(a0, time_limit, beta_init=30.0, lr_init=0.008, label=""):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y_proj)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        grad = np.zeros(n)
        active_k = np.where(weights > 1e-10)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            lr *= 0.85
            beta *= 1.4
            no_improve = 0
            if lr < 1e-10:
                break
        elif it % 30 == 0:
            beta *= 1.01

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score

def subgradient_nesterov(a0, time_limit, label=""):
    """Subgradient method with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    avg_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y)
        M = conv_y.max()
        threshold = M - 0.02 * M
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_peak)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += rho * 2.0 * y[k - j]

        a_new = a - step * grad
        a_new = simplex_project(a_new)

        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new

        for cand in [a_new, avg_a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score
