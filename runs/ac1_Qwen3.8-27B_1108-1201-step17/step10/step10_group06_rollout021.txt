"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp continuation method with projected gradient descent
2. FFT-based efficient gradient computation for the smoothed objective
3. Exact coordinate descent with golden section search
4. Dual-guided focused coordinate descent
5. Multi-scale basin hopping with length adaptation
6. Spectral initialization and windowing strategies
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) > 0:
        r = rho[-1]
        theta = (cssv[r] - 1.0) / (r + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def lse_objective_and_grad_a(a, T):
    """Compute log-sum-exp(T*conv) / T and its gradient w.r.t. a."""
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = T * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = (max_c + np.log(sum_exp)) / T
    p = exp_shifted / sum_exp
    # g_i = 2 * sum_j a[j] * p[i+j]
    a_rev = a[::-1]
    fft_len = 1 << (2 * len(a) - 1).bit_length()
    A_rev = rfft(a_rev, fft_len)
    P = rfft(p, fft_len)
    cross = irfft(A_rev * P, fft_len)
    n = len(a)
    g = 2.0 * cross[n - 1:2 * n - 1]
    return obj, g

def lse_projected_grad_descent(a0, T_values, time_limit, rng):
    """Log-sum-exp continuation with projected gradient descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for T_idx, T in enumerate(T_values):
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 2:
            break
        lr = 2.0 / (T * n * 0.1)
        grad_scale = 0.01
        no_improve = 0
        max_iter = max(20, int(time_left * 30))

        for it in range(max_iter):
            if time.monotonic() - t0 > time_limit - 1:
                break
            obj, g = lse_objective_and_grad_a(a, T)
            # Projected gradient step
            a_new = a - lr * g
            a_new = simplex_project(a_new)
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
            if no_improve > 50:
                lr *= 0.8
                no_improve = 0
            if lr < 1e-8:
                break

    return best_a, best_score

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-16:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def coord_descent_score(x, c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n):
    ns = current_sum + (x - v)
    if ns < 0.01:
        return 1e10
    nv = c_vals + 2.0 * x * a_partners
    if has_quad:
        nv = nv.copy()
        nv[quad_idx] = c_vals[quad_idx] + x * x
    nm = max(unaff_max, float(nv.max()))
    return 2.0 * n * nm / (ns * ns)

def exact_coord_descent(a0, time_limit, verbose=False, label=""):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

    if s < 1e-15:

    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                upper = 5.0 / n
                if upper < 1e-15:
                    continue
            else:
                upper = max(3.0 * v, 5.0 / n)

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = -1
            if has_quad:
                quad_idx = 2 * i - k_lo

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            best_x = golden_section_search(
                lambda x: coord_descent_score(x, c_vals, a_partners, unaff_max,
                                              has_quad, quad_idx, current_sum, v, n),
                0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def dual_guided_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent that focuses on coordinates affecting the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        conv = fast_convolve(a)
        peak_val = conv.max()
        # Find coordinates that contribute to peaks above 95% of max
        threshold = 0.95 * peak_val
        peak_lags = np.where(conv > threshold)[0]

        # Determine priority order: coordinates that appear in MANY peak lags first
        coord_count = np.zeros(n)
        for k in peak_lags:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            coord_count[i_start:i_end + 1] += 1

        order = np.argsort(-coord_count)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = -1
            if has_quad:
                quad_idx = 2 * i - k_lo

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            upper = max(3.0 * v if v > 1e-15 else 0, 5.0 / n)
            best_x = golden_section_search(
                lambda x: coord_descent_score(x, c_vals, a_partners, unaff_max,
                                              has_quad, quad_idx, current_sum, v, n),
                0.0, upper, max_iter=28)

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    DCD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def init_gaussian(n, rng, center_frac, width_frac):
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine_window(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    a = irfft(np.exp(1j * phases), n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_multi_peak(n, rng, n_peaks=3):
    a = np.zeros(n)
    for _ in range(n_peaks):
        center = rng.integers(0, n)
        width = rng.uniform(n * 0.05, n * 0.2)
        amp = rng.uniform(0.5, 1.5)
        t = np.arange(n, dtype=np.float64)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_hamming_like(n, rng):
    """Hamming window variant with random shift."""
    t = np.arange(n, dtype=np.float64)
    shift = rng.uniform(-0.1, 0.1) * n
    x = (t + shift) / n
    a = 0.54 + 0.46 * np.cos(2.0 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def multiplicative_weights_update(a, conv, n, step_size):
    """Update using multiplicative weights inspired by the peak structure."""
    a_new = a.copy()
    threshold = conv.max() * 0.9
    for k in range(len(conv)):
        if conv[k] > threshold:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            excess = (conv[k] - threshold) / conv.max()
            for i in range(i_start, i_end + 1):
                partner = a[k - i]
                if partner > 1e-15:
                    a_new[i] *= (1.0 - step_size * excess * 0.5)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

def mw_iteration(a0, time_limit, step_size=0.05, verbose=False):
    """Multiplicative weights iteration to flatten convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    no_improve = 0
    current_step = step_size

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        a_new = multiplicative_weights_update(a, conv, n, current_step)
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            current_step *= 0.9
            if current_step < 1e-6:
                break
        if no_improve > 200:
            break
        if verbose and iters % 50 == 0:
            print(f"    MW it={iters}: best={best_score:.8f}, step={current_step:.6f}")

    return best_a, best_score

def basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping with diverse perturbation strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 60.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Peak-directed reduction
            conv_p = fast_convolve(a)
            peaks = np.where(conv_p > 0.9 * conv_p.max())[0]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                n_reduce = max(1, (i_end - i_start + 1) // 5)
                idx = rng.choice(np.arange(i_start, i_end + 1), size=min(n_reduce, i_end - i_start + 1), replace=False)
                for ii in idx:
                    perturbed[ii] *= (1.0 - 0.1 * rng.random())
        elif pert_type == 2:
            # Block permutation
            blen = max(4, n // 16)
            n_blocks = n // blen
            if n_blocks >= 2:
                perm = rng.permutation(n_blocks)
                for old_i in range(n_blocks):
                    new_i = perm[old_i]
                    perturbed[new_i * blen:(new_i + 1) * blen] = a[old_i * blen:(old_i + 1) * blen]
        elif pert_type == 3:
            # Mirror mix
            shift = rng.integers(1, max(2, n // 5))
            shifted = np.roll(a, shift)
            mix = rng.uniform(0.3, 0.7)
            perturbed = mix * a + (1 - mix) * shifted
        else:
            # Sparse perturbation: randomly zero and refill
            n_zero = max(1, n // 20)
            zero_idx = rng.choice(n, size=n_zero, replace=False)
            mass = perturbed[zero_idx].sum()
            perturbed[zero_idx] = 0
            fill_idx = rng.choice(np.setdiff1d(np.arange(n), zero_idx), size=n_zero, replace=False)
            perturbed[fill_idx] += mass / n_zero

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: improved to {best_score:.8f}")

    return best_a, best_score

def lse_refinement(a0, time_limit, verbose=False):
    """Log-sum-exp refinement with continuation."""
    n = len(a0)
    T_values = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    # Allocate time proportionally
    total_time = time_limit
    per_T = total_time / len(T_values)
    best_a = np.asarray(a0, dtype=np.float64)
    best_score = compute_score(best_a)

    for T in T_values:
        rem = time_limit - (time.monotonic() - t0) if 't0' in dir() else time_limit
        a_new, s_new = lse_projected_grad_descent(best_a, [T], per_T, 0)
        if s_new < best_score - 1e-12:
            best_score = s_new
            best_a = a_new.copy()
            if verbose:
                print(f"    LSE T={T}: {best_score:.8f}")

    return best_a, best_score

def propose_candidate():

    multiplicative weights, and basin hopping to minimize the convolution peak.
    """

            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")

        parent = None

    best_seq = None
    best_score = float('inf')

        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    # Phase 1: Diverse initialization and quick screening
    print("\n=== Phase 1: Initialization & Screening ===")
    t1_start = time.monotonic()
    n_parent = len(parent) if parent is not None else 1024

    n_values = list(set([n_parent, 768, 1024, 1152, 1280, 1536]))
    n_values = [n for n in n_values if 256 <= n <= 2048]
    n_values.sort()

    candidates = []
    for n in n_values:
        if time.monotonic() - t1_start > 60:
            break
        inits = []
        if parent is not None:
            inits.append(("parent_interp", interpolate_seq(parent, n)))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.8f}")

    for s_c, a_c, n_c in candidates[:5]:
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST p1: {sc:.8f} (n={n_c})")

    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = compute_score(best_seq)

    n_work = len(best_seq)
    print(f"  Working with n={n_work}, score={best_score:.8f}")

    # Phase 2: LSE Continuation (main optimization driver)
    print(f"\n=== Phase 2: LSE Continuation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 700
