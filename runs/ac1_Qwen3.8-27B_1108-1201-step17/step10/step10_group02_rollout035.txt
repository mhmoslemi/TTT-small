        col_ind.extend([i] * n)

    T = csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, T_sparse, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_sparse @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try fallback solver
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(t_var.value) if t_var.value is not None else fast_convolve(a_fixed) @ b_val
    if M_val is None or np.isnan(M_val):
        actual_conv = fast_convolve(a_fixed) @ b_val
        M_val = float(np.max(actual_conv))
    return b_val, M_val


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=100000):
    """
    Nesterov LP alternating with adaptive exploration.
    Uses structured perturbations toward the second-highest peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    stall_threshold = 8
    prev_best = best_score
    rng = np.random.default_rng(12345)

    # Build sparse convolution matrix (cached for this n)
    T_sparse = build_sparse_conv_matrix(np.ones(n) / n)
    # Actually T depends on a, so we need to rebuild each iteration
    # But for the LP "best response" step, we fix a and solve for b
    # So T uses the CURRENT a

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Build T for current y
        T_sparse = build_sparse_conv_matrix(y)
        b_new, M_b = solve_lp_best_response(y, T_sparse, solver)
        if b_new is None:
            continue

        # Adaptive Nesterov step
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Weighted average for a (FTRL-style)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 20 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            # Adaptive exploration on stall
            if stall >= stall_threshold:
                # Structured perturbation: push away from the dominant peak
                b_cur = fast_convolve(y)
                peak_idx = np.argmax(b_cur)
                # Find the "contributing" coordinates to the peak
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                idx = np.arange(js, je + 1)
                contribs = y[idx] * y[peak_idx - idx]

                # Reduce top contributors, increase others
                n_reduce = max(3, len(idx) // 10)
                top_pos = idx[np.argsort(contribs)[::-1][:n_reduce]]
                for pos in top_pos:
                    if y[pos] > 1e-10:
                        y[pos] *= (1.0 - 0.05 * np.sqrt(stall / stall_threshold))

                # Add some random noise too
                noise = rng.standard_normal(n) * (0.01 * np.sqrt(stall / stall_threshold))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall = 0
                stall_threshold = min(20, stall_threshold + 1)

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_obj_and_grad(a, T):
    """
    Compute LogSumExp smoothed objective and gradient.
    Objective: (2n/T) * LogSumExp(conv(a,a)/T)
    Gradient wrt a computed via cross-correlation.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()

    # grad_a[i] = (2n/T) * sum_k w[k] * 2*a[k-i] (for valid k-i)
    # = (4n/T) * sum_{j=0}^{n-1} w[i+j] * a[j]
    grad_a = np.zeros(n)
    for i in range(n):
        # Sum over j: w[i+j] * a[j], where i+j < 2n-1
        valid_j = min(n, 2 * n - 1 - i)
        if valid_j > 0:
            grad_a[i] = 2.0 * w[i:i + valid_j] @ a[:valid_j]
    grad_a *= (2.0 * n) / T

    # Objective value
    logsumexp_val = max_b / T + np.log(exp_b.sum())
    obj = (2.0 * n / T) * logsumexp_val

    return obj, grad_a


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS on LogSumExp smoothed objective (operates in a-space with projection)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 12.0, 7.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.1]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)
    best_a = a.copy()
    best_obj = float('inf')

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit - 2:
                return 1e15
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = logsumexp_obj_and_grad(a_trial, T)
            if obj < best_obj:
                best_obj = obj
                best_a = a_trial.copy()
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit - 2:
                return np.zeros_like(z_inner)
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z_inner)
            a_trial = a_trial / st
            _, grad = logsumexp_obj_and_grad(a_trial, T)
            return grad

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            pass

        if (time.monotonic() - t0) % 10 < 1:
            elapsed = time.monotonic() - t0
            sc = visualize_score(a)
            print(f"    LSE-LB T={T:.2f} ({elapsed:.0f}s): score={sc:.8f}")

    return best_a


def visualize_score(a):
    """Quick score visualization."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def mirror_descent_adam(a0, time_limit, eta0=0.05):
    """
    Mirror descent with Adam-like adaptive step size.
    Uses per-coordinate exponential moving averages for step adaptation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = visualize_score(a)
    t0 = time.monotonic()

    # Adam state
    m_g = np.zeros(n)  # First moment
    v_g = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_count = 0
    no_improve = 0

    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

        # Adam update
        t_count += 1
        m_g = beta1 * m_g + (1 - beta1) * g
        v_g = beta2 * v_g + (1 - beta2) * g * g
        m_hat = m_g / (1 - beta1 ** t_count)
        v_hat = v_g / (1 - beta2 ** t_count)
        adaptive_eta = eta0 / (np.sqrt(v_hat + eps) + eps)
        # Adaptive eta should be moderate
        adaptive_eta = np.clip(adaptive_eta, 1e-6, 2.0)
        # Use scalar average for stability
        eta_eff = float(np.mean(adaptive_eta))
        eta_eff = max(1e-5, min(eta_eff, 1.0))

        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta_eff * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum
        a = a_new

        score = visualize_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            m_g = np.zeros(n)
            v_g = np.zeros(n)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-Adam iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=100000):
    """Targeted coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = visualize_score(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:3]:  # Top 3 peaks
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = visualize_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                # Simultaneous small reduction of top contributors
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:4]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.998
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = visualize_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    else:
        # Bimodal
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver)
        try_update(a_lp, "LP_p1")

    # === Phase 2: Mirror Descent with Adam ===
    print(f"\n=== Phase 2: Mirror Descent Adam (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_adam(best_seq, t_alloc, eta0=0.04)
        try_update(a_md, "MD_adam")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        candidate_ns = sorted(set([max(400, n_best - 200), max(600, n_best + 200),
                                    max(400, n_best // 2), n_best * 2]))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, {per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 10:
                break
            if n_try == n_best:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 10:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_lp_refine(a_interp, t_sub, solver=solver, n_max_iters=50000)
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 4: Deep LP Refinement ===
    print(f"\n=== Phase 4: Deep LP Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_deep, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_deep, "deep_lp")

    # === Phase 5: LogSumExp L-BFGS ===
    print(f"\n=== Phase 5: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]
        print(f"  Budget: {t_alloc:.0f}s, T_stages={T_stages}")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                               maxiter_per_stage=4000)
        try_update(a_lb, "lbgfs")

    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=200000)
        try_update(a_cd, "coord_desc")

    # === Phase 7: Final quick LP ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_fin, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
