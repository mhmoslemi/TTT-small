        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    import cvxpy as cp
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = 'ECOS'
    return find_lp_solver._cached


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    import cvxpy as cp
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def nesterov_smooth_descent(a0, time_limit, beta_start=5.0, beta_end=300.0,
                            lr_start=0.03, momentum=0.9, verbose=True, label=""):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start / (1.0 + 0.005 * it)

        M = conv.max()
        shifted = beta * (conv - M)
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()

        grad = np.zeros(n)
        thresh = max(1e-8, 0.005 / n)
        active_mask = w > thresh
        active_indices = np.where(active_mask)[0]

        for k in active_indices:
            k = int(k)
            wk = w[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx_arr = np.arange(i_start, i_end + 1)
            k_minus_i = k - idx_arr
            a_kmi = a[k_minus_i]
            a_ii = a[idx_arr]
            grad[i_start:i_end + 1] += wk * (a_kmi + a_ii)

        grad *= 2.0 * n

        y = a + momentum * (a - a_prev)
        a_prev = a.copy()
        a_new = y - lr * grad
        a = simplex_project(a_new)

        conv = fast_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
            momentum = min(0.95, momentum + 0.01)
            if lr < 1e-12:
                break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSD[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.0f}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        else:
            alpha = max(0.4, 15.0 / (it + 5.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a = a / a.sum()

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if score_a > best_score * (1 + 5e-6):
            stagnation += 1
        else:
            stagnation = 0

        if stagnation > 20:
            noise_scale = 0.02 / np.sqrt(n)
            rng = np.random.default_rng(it * 12345 + 678)
            a_pert = a + rng.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.005:
                a = a_pert
            stagnation = 0

        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_precise(a0, time_limit, verbose=True, label=""):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end + 1].copy()
            conv_affected = conv[k_start:k_end + 1].copy()
            a_old = a[i]

            def eval_coord(t):
                new_affected = conv_affected + 2.0 * (t - a_old) * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] += t * t - a_old * a_old
                return max(unaff_max, float(new_affected.max()))

            # Golden section search
            t_lower = 0.0
            t_upper = 3.0 * a_old
            gr = (np.sqrt(5.0) + 1.0) / 2.0
            t1 = t_upper - (t_upper - t_lower) / gr
            t2 = t_lower + (t_upper - t_lower) / gr
            f1 = eval_coord(t1)
            f2 = eval_coord(t2)

            for _ in range(40):
                if f1 < f2:
                    t_upper = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_upper - (t_upper - t_lower) / gr
                    f1 = eval_coord(t1)
                else:
                    t_lower = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_lower + (t_upper - t_lower) / gr
                    f2 = eval_coord(t2)

            t_opt = (t_lower + t_upper) / 2.0
            f_opt = eval_coord(t_opt)

            if abs(t_opt - a_old) > 1e-12 and f_opt < M - 1e-14:
                a[i] = max(t_opt, 0.0)
                changes = 2.0 * (a[i] - a_old) * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += a[i] * a[i] - a_old * a_old
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 15:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end + 1]
            conv_affected = conv[k_start:k_end + 1]
            a_old = a[i]

            # Very fine multiplicative steps
            for factor in [0.9995, 0.999, 0.998, 0.997, 1.001, 1.002, 1.003, 1.004]:
                delta = a_old * (factor - 1.0)
                new_val = a_old + delta
                if new_val < -1e-16:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a_old + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < M - 1e-15:
                    a[i] = new_val
                    changes = 2.0 * delta * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        changes = changes.copy()
                        changes[idx] = 2.0 * delta * (a_old - delta) + delta * delta
                    conv[k_start:k_end + 1] += changes
                    M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                    cur_score = 2.0 * n * M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Fine[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_init(n, rng, mode=0):
    if mode == 0:
        target_mag = np.ones(n // 2 + 1)
        target_mag[0] = 0.1
    elif mode == 1:
        target_mag = 1.0 + 0.3 * np.cos(np.pi * np.arange(n // 2 + 1) / (n // 4 + 1))
    else:
        target_mag = 1.0 + 0.5 * np.random.default_rng(42).uniform(-0.5, 0.5, n // 2 + 1)
        target_mag = np.maximum(target_mag, 0.5)

    phase = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    phase[0] = 0.0
    A = target_mag * np.exp(1j * phase)

    if n // 2 > 0:
        A_full = np.zeros(n, dtype=complex)
        A_full[:n // 2 + 1] = A
        A_full[n // 2 + 1:] = np.conj(A_full[1:n // 2 + 1][::-1])
    else:
        A_full = A

    a = np.real(np.fft.ifft(A_full))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def diverse_init(n, rng, init_type=0):
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    else:
        a = spectral_init(n, rng, mode=init_type % 3)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
