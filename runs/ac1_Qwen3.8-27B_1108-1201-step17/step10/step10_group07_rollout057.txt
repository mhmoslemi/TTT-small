    """
    For coordinate i, find the optimal delta (change to a[i]) that minimizes
    the maximum convolution value.
    
    Returns (best_delta, best_new_M)
    """
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    
    # Unaffected maximum
    unaff_max = 0.0
    if k_start > 0:
        unaff_max = max(unaff_max, conv[:k_start].max())
    if k_end < 2 * n - 2:
        unaff_max = max(unaff_max, conv[k_end + 1:].max())
    
    # Affected convolution values as function of delta:
    # For k in [k_start, k_end], k != 2i:
    #   conv_new[k] = conv[k] + 2*delta*a[k-i]
    # For k = 2i:
    #   conv_new[2i] = conv[2i] + 2*delta*a[i] + delta^2
    
    # We want to find delta in [-a[i], ∞) that minimizes:
    #   max(unaff_max, conv[k] + 2*delta*a[k-i] for k in affected, k != 2i,
    #       conv[2i] + 2*delta*a[i] + delta^2)
    
    # The function f(delta) = max of all these is convex in delta
    # (max of affine and quadratic functions)
    # So we can use bisection on the derivative (or ternary search)
    
    # Let's use bisection on the subgradient:
    # f'(delta) = 0 at the optimum
    
    # Actually, let's just use ternary search since f is convex.
    
    # Lower bound: delta >= -a[i] (to keep a[i] >= 0)
    # Upper bound: we need to find where f(delta) starts increasing
    # A reasonable upper bound: delta_max = a[i] (double the value)
    
    lo = -a[i]
    hi = a[i]  # Try up to doubling
    
    # Actually, for better bounds:
    # If delta is too large, the quadratic term at k=2i dominates:
    # conv[2i] + 2*delta*a[i] + delta^2
    # At delta = a[i]: conv[2i] + 2a[i]^2 + a[i]^2 = conv[2i] + 3a[i]^2
    # This is likely much larger than M, so hi = a[i] might be too large
    # But ternary search will handle this.
    
    # Let's extend the range a bit
    hi = max(hi, M / (2 * max(a[i], 1e-10)))  # Rough bound

        if f1 < f2:

        else:
            lo = m1
    
    best_delta = (lo + hi) / 2
    best_new_M = eval_conv_max(a, conv, i, best_delta, unaff_max)
    return best_delta, best_new_M


def eval_conv_max(a, conv, i, delta, unaff_max):
    """Evaluate max convolution after changing a[i] by delta."""
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    
    # Affected values
    max_val = unaff_max
    for k in range(k_start, k_end + 1):
        if k == 2 * i:
            val = conv[k] + 2 * delta * a[i] + delta * delta
        else:
            val = conv[k] + 2 * delta * a[k - i]
        if val > max_val:
            max_val = val
    return max_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent via ternary search per coordinate (main innovation)
2. LP-based best-response for escaping local minima
3. Multi-restart with random perturbations
4. Fine polishing
The key improvement over previous approaches is solving the 1D subproblem
exactly for each coordinate rather than using a fixed set of trial steps.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_quick(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def conv_max_after_delta(conv, a, i, delta, unaff_max):
    """Compute max conv value after changing a[i] by delta. Vectorized."""
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    a_shifted = a[k_start - i : k_end - i + 1]
    new_vals = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
    if 2 * i >= k_start and 2 * i <= k_end:
        idx = 2 * i - k_start
        new_vals = new_vals.copy()
        new_vals[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
    local_max = float(new_vals.max())
    if local_max < unaff_max:
        return unaff_max
    return local_max


def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Exact coordinate descent: for each coordinate, find the optimal delta
    using ternary search on the convex function f(delta) = max conv value.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_start_M = M

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Ternary search for optimal delta in [-a[i], hi]
            lo = -a[i]
            # Upper bound: try up to a[i] initially, extend if needed
            hi = a[i]
            
            # Quick check: is the current value (delta=0) already optimal?
            # Check subgradient at delta=0
            # f'(0-) and f'(0+) signs tell us if 0 is optimal
            # For now, just do the ternary search
            
            # Ternary search
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = conv_max_after_delta(conv, a, i, m1, unaff_max)
                f2 = conv_max_after_delta(conv, a, i, m2, unaff_max)
                if f1 <= f2:
                    hi = m2
                else:
                    lo = m1

            best_delta = (lo + hi) / 2.0
            best_new_M = conv_max_after_delta(conv, a, i, best_delta, unaff_max)

            if best_new_M < M - 1e-14:
                # Apply the update
                a[i] = max(a[i] + best_delta, 0.0)
                # Update convolution
                k_s = i
                k_e = min(i + n - 1, 2 * n - 2)
                a_shifted = a[k_s - i : k_e - i + 1]
                # Note: a[i] has already been updated, so a_shifted[i] = a[i] + best_delta
                # We need to be careful: the change in conv is:
                # For k != 2i: conv[k] += 2 * best_delta * a_old[k-i]
                # For k = 2i: conv[2i] += 2 * best_delta * a_old[i] + best_delta^2
                # But a_shifted[i] = a_new[i] = a_old[i] + best_delta
                # So a_old[k-i] = a_shifted[k-i] for k-i != i, and a_old[i] = a_shifted[i] - best_delta
                # Let's just recompute:
                delta = best_delta
                old_ai = a[i] - delta  # recover old value
                changes = np.zeros(k_e - k_s + 1)
                a_old_shifted = a[k_s - i : k_e - i + 1].copy()
                a_old_shifted[i] = old_ai  # fix the i-th element
                changes = 2.0 * delta * a_old_shifted
                if 2 * i >= k_s and 2 * i <= k_e:
                    idx = 2 * i - k_s
                    changes[idx] = 2.0 * delta * old_ai + delta * delta
                conv[k_s:k_e + 1] += changes
                M = max(unaff_max, float(conv[k_s:k_e + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
            # Check convergence within sweep
            if M < sweep_start_M * 0.999:
                pass  # Good progress

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score

def fast_convolve(a):

    return irfft(A * A, fft_size)[:size]


def score_quick(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def conv_max_after_delta(conv, a, i, delta, unaff_max):
    """Compute max conv value after hypothetically changing a[i] by delta."""
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    a_shifted = a[k_start - i : k_end - i + 1]
    new_vals = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
    if 2 * i >= k_start and 2 * i <= k_end:
        idx = 2 * i - k_start
        new_vals = new_vals.copy()
        new_vals[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
    local_max = float(new_vals.max())
    if local_max < unaff_max:
        return unaff_max
    return local_max


def apply_delta_to_conv(a, conv, i, delta, old_ai):
    """Apply delta to coordinate i and update convolution in-place."""
    k_s = i
    k_e = min(i + len(a) - 1, 2 * len(a) - 2)
    a_old_shifted = a[k_s - i : k_e - i + 1].copy()
    a_old_shifted[i] = old_ai
    changes = 2.0 * delta * a_old_shifted
    if 2 * i >= k_s and 2 * i <= k_e:
        idx = 2 * i - k_s
        changes[idx] = 2.0 * delta * old_ai + delta * delta
    conv[k_s:k_e + 1] += changes


def compute_coord_importance(a, conv, top_k=10):
    """Compute importance score for each coordinate based on peak contributions."""
    n = len(a)
    importance = np.zeros(n)
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    for k in peak_indices:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            importance[i] += a[i] * a[j]
    return importance


def exact_cd(a0, time_limit, verbose=True, label=""):
    """
    Exact coordinate descent using ternary search per coordinate.
    Processes coordinates in order of importance to top peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Initial importance ordering
    importance = compute_coord_importance(a, conv)
    order = np.argsort(importance)[::-1]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_start_M = M

        # Adaptive search parameters
        if sweeps <= 5:
            scale = 1.0
            num_iters = 50
        elif sweeps <= 20:
            scale = 0.5
            num_iters = 45
        else:
            scale = 0.25
            num_iters = 35

        for idx in range(n):
            i = int(order[idx])
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            lo = -ai
            hi = ai * scale
            if hi < 1e-15:
                hi = 1e-15

            for _ in range(num_iters):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = conv_max_after_delta(conv, a, i, m1, unaff_max)
                f2 = conv_max_after_delta(conv, a, i, m2, unaff_max)
                if f1 <= f2:
                    hi = m2
                else:
                    lo = m1

            best_delta = (lo + hi) / 2.0
            best_new_M = conv_max_after_delta(conv, a, i, best_delta, unaff_max)

            if best_new_M < M - 1e-14:
                delta = best_delta
                a[i] = max(ai + delta, 0.0)
                apply_delta_to_conv(a, conv, i, delta, ai)
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

        # Update importance every 10 sweeps
        if sweeps % 10 == 0:
            importance = compute_coord_importance(a, conv)
            order = np.argsort(importance)[::-1]

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        if it <= 10:
            a = b_new.copy()
        else:
            alpha = max(0.2, 8.0 / it)
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if score_a > best_score * (1 + 1e-8):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 20:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def try_update_state(state, candidate, label):
    """Try to update the best sequence in state dict."""
    candidate = np.maximum(candidate, 0.0)
    s = candidate.sum()
    if s < 0.01:
        return
    candidate = candidate / s
    try:
        sc = float(evaluate_sequence(candidate.tolist()))
    except Exception:
        sc = float('inf')
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = candidate.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f}")


def multi_restart_cd(a0, time_limit, rng, num_restarts, verbose=True):
    """Multiple random perturbations + exact CD from each."""
    n = len(a0)
    best_a = a0.copy()
    s = a0.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    else:
        best_a = a0 / s
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    for r in range(num_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 40:
            break
        t_alloc = min(remaining / max(num_restarts - r, 1), 150.0)

        scale = 0.03 / np.sqrt(n)
        a_pert = best_a + rng.standard_normal(n) * scale
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        a_res, s_res = exact_cd(a_pert, t_alloc, verbose=False, label=f"r{r}")
        if s_res < best_score:
            best_score = s_res
            best_a = a_res.copy()
            if verbose:
                print(f"    Restart {r}: improved to {best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if init_type == 0:
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 10))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * t * 3 / n) + 0.2 * np.cos(2 * np.pi * t * 7 / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    state = {'best_seq': None, 'best_score': float('inf')}

        p = np.asarray(height_sequence_1, dtype=np.float64)

            state['best_seq'] = p / p.sum()
            state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
            print(f"Parent: n={len(state['best_seq'])}, score={state['best_score']:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if state['best_seq'] is None:
        n = 1024
        state['best_seq'] = np.ones(n) / n
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"Uniform init: n={n}, score={state['best_score']:.8f}")

    n = len(state['best_seq'])
    solver = find_lp_solver()
    print(f"Config: n={n}, solver={solver}")

    # === Phase 1: Exact CD (main workhorse, ~35% time) ===
    remaining = deadline - time.monotonic()
    t1 = min(remaining * 0.35, 450.0)
    print(f"\n=== Phase 1: Exact CD ({t1:.0f}s budget) ===")
    a1, s1 = exact_cd(state['best_seq'], t1, verbose=True, label="main")
    try_update_state(state, a1, "cd1")

    # === Phase 2: LP Alternating (~15% time) ===
    remaining = deadline - time.monotonic()
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: LP Alternating ({t2:.0f}s budget) ===")
    a2, s2 = lp_alternating(state['best_seq'], t2, solver, verbose=True)
    try_update_state(state, a2, "lp")

    # Exact CD on LP result
    if time.monotonic() < deadline - 600:
        t2b = min((deadline - time.monotonic()) * 0.10, 120.0)
        a2b, s2b = exact_cd(a2, t2b, verbose=True, label="lp_cd")
        try_update_state(state, a2b, "lp_cd")

    # === Phase 3: More Exact CD (~20% time) ===
    remaining = deadline - time.monotonic()
    t3 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 3: More Exact CD ({t3:.0f}s budget) ===")
    a3, s3 = exact_cd(state['best_seq'], t3, verbose=True, label="main2")
    try_update_state(state, a3, "cd2")

    # === Phase 4: Multi-restart (~20% time) ===
    remaining = deadline - time.monotonic()
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: Multi-restart ({t4:.0f}s budget) ===")
    a4, s4 = multi_restart_cd(state['best_seq'], t4, rng, num_restarts=6, verbose=True)
    try_update_state(state, a4, "restart")

    # Also try diverse inits with quick CD
    if time.monotonic() < deadline - 120:
        for init_type in [0, 2, 4]:
            if time.monotonic() > deadline - 100:
                break
            init = diverse_init(n, rng, init_type)
            t_init = min(40.0, (deadline - time.monotonic()) * 0.05)
            if t_init < 10:
                break
            a_i, s_i = exact_cd(init, t_init, verbose=False, label=f"div{init_type}")
            try_update_state(state, a_i, f"div{init_type}")

    # === Phase 5: Final polish (~5% time) ===
    remaining = deadline - time.monotonic()
    t5 = min(remaining - 15, 100.0)
    if t5 > 10:
        print(f"\n=== Phase 5: Final Polish ({t5:.0f}s budget) ===")
        a5, s5 = exact_cd(state['best_seq'], t5, verbose=True, label="final")
        try_update_state(state, a5, "final")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs) and fs < state['best_score']:
        state['best_score'] = fs

    print(f"\nFINAL: score={state['best_score']:.8f}, n={len(state['best_seq'])}")
    return [float(x) for x in state['best_seq']]
