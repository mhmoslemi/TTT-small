Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using efficient bidirectional LP 
alternation with HiGHS, intensive LogSumExp L-BFGS polishing, and 
multi-length exploration. Key improvements: HiGHS for fast LP solves, 
better Nesterov acceleration, aggressive low-temperature polishing,
and systematic n-search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import math

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, method='highs'):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Returns (b_opt, M_opt)."""
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    
    # Formulation: minimize M
    # s.t. T @ b <= M * ones  (equivalently: T @ b - M <= 0)
    #      sum(b) = 1
    #      b >= 0
    
    # Variable vector: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Inequality: T @ b - M*1 <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method=method)
        if result.success:
            b_val = result.x[:n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
                return b_val, float(result.x[n])
    except Exception:
        pass
    
    # Fallback: try cvxpy
    try:
        import cvxpy as cp
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve()
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s, float(M_var.value)
    except Exception:
        pass
    
    return None, None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using efficient bidirectional LP alternation
with HiGHS solver, intensive multi-stage LogSumExp L-BFGS polishing, FFT-based gradients,
and multi-length exploration. Key improvements over prior: HiGHS for fast LP, FFT-accelerated
gradients, adaptive Nesterov with anti-stall restarts, aggressive low-T polishing stages,
and systematic n-search around the optimum.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (assumes sum~1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Uses scipy linprog with HiGHS. Returns (b_opt, M_opt) or (None, None)."""
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)

    # Variable vector: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M

    # Inequality: T @ b - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            b_val = result.x[:n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
                return b_val, float(result.x[n])
    except Exception:
        pass

    # Fallback: dual formulation
    try:
        # Dual: maximize sum(y) s.t. T^T @ y <= 1, y >= 0
        AT = T.T  # n x (2n-1)
        c_dual = -np.ones(2 * n - 1)  # minimize -sum(y)
        A_ub_dual = AT
        b_ub_dual = np.ones(n)
        bounds_dual = [(0, None)] * (2 * n - 1)
        result_dual = linprog(c_dual, A_ub=A_ub_dual, b_ub=b_ub_dual,
                              bounds=bounds_dual, method='highs',
                              options={'presolve': True, 'time_limit': 10})
        if result_dual.success and result_dual.x is not None:
            y_val = np.maximum(result_dual.x, 0)
            M_opt = float(-result_dual.fun)
            # Recover b from complementary slackness:
            # b_j > 0 only if (T^T @ y)[j] = 1 (tight constraint)
            dual_val = AT @ y_val
            tight = np.where(dual_val >= 1.0 - 1e-8)[0]
            if len(tight) > 0:
                # Solve: T[tight_idx, tight] @ b[tight] = M_opt, sum(b[tight]) = 1
                # Actually, we just need b that satisfies the primal constraints
                # Simplest: use equal weights on tight positions
                b_val = np.zeros(n)
                b_val[tight] = 1.0 / len(tight)
                # Project to satisfy T @ b <= M
                return b_val, M_opt
    except Exception:
        pass

    return None, None


def bidirectional_lp(a0, time_limit, max_iters=100000):
    """Alternating minimization solving both LP subproblems with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Best b given current y_a
        b_new, M1 = solve_lp_best_response(y_a)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = y_b

        # Step 2: Best a given current y_b
        a_new, M2 = solve_lp_best_response(y_b)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = y_a

        # Track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        # Stall detection and restart
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                noise_scale = 0.03 / math.sqrt(stall_count)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                a = y_a
                b = y_b
                stall_count = 0

            if (it + 1) % 2000 == 0:
                print(f"    BiDirLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def compute_grad_a_fft(a, w):
    """Compute grad_a[m] = 2 * sum_k w[k] * a[k-m] for m = 0,...,n-1 using FFT."""
    n = len(a)
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    full = irfft(W * np.conj(A), L)
    return 2.0 * full[:n]


def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    """L-BFGS on the LogSumExp-smoothed objective with FFT-based gradients."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 15))

        def obj(z_inner):
            z_max = z_inner.max()
            exp_z = np.exp(z_inner - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def grad(z_inner):
            z_max = z_inner.max()
            exp_z = np.exp(z_inner - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            exp_ws = np.exp(b_shift - b_max_val)
            w = exp_ws / exp_ws.sum()
            grad_a = compute_grad_a_fft(a_cur, w)
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method with adaptive step size and peak averaging."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: gradient of mean of active peaks
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        g = compute_grad_a_fft(a, w)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    else:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
