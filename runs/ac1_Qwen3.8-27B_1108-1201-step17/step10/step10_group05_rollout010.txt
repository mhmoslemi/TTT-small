"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Fictitious Play with peak-avoidance constraints (primary)
2. Smart coordinate descent with exact 1D minimax
3. Multi-length exploration
4. Final fine polishing
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

_lp_solver = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.PDLP, cp.GLOP, cp.SCS, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_best_response(a, solver=None, peak_constraints=None):
    """Solve the best response LP: min M s.t. (a*b)[k] <= M, b>=0, sum(b)=1."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build the convolution matrix T: T[k,i] = a[k-i] if 0<=k-i<n else 0
    # Instead of building the full matrix, use the structure
    # (a*b)[k] = sum_i a[k-i]*b[i]
    # For efficiency, precompute the matrix
    T = np.zeros((2*n-1, n))
    for i in range(n):
        T[i:i+n, i] = a
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    if peak_constraints is not None:
        for k in peak_constraints:
            constraints.append(T[k, :] @ b_var <= peak_constraints[k])
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

def build_T(a):
    """Build convolution matrix T where T[k,i] = a[k-i] if valid."""
    n = len(a)

    for i in range(n):

def solve_bp(a, solver, extra_constraints=None):
    """Solve best response LP."""
    import cvxpy as cp
    n = len(a)
    T = build_T(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [T @ b <= M, cp.sum(b) == 1]
    if extra_constraints is not None:
        for k, bound in extra_constraints:
            cons.append(T[k] @ b <= bound)
    prob = cp.Problem(cp.Minimize(M), cons)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    bval = np.asarray(b.value, dtype=np.float64)
    bval = np.maximum(bval, 0)
    s = bval.sum()
    if s < 1e-15:
        return None, None
    return bval / s, float(M.value)

def interpolate_seq(f, new_n):
    old_n = len(f)

        return f.copy()
    s = f.sum()

        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss

import numpy as np

import time

_solver_name = None

def fast_convolve(a):
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_solver():
    global _solver_name
    if _solver_name is not None:
        return _solver_name
    import cvxpy as cp
    for s in [cp.MOSEK, cp.PDLP, cp.GLOP, cp.SCS, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(100, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _solver_name = s
                print(f"  Selected LP solver: {s}")

        except Exception:

    print(f"  Selected LP solver: ECOS (fallback)")
    return _solver_name

def build_T(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T

def solve_bp(a, solver, extra_cons=None):
    import cvxpy as cp
    n = len(a)
    T = build_T(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [T @ b <= M, cp.sum(b) == 1]
    if extra_cons is not None:
        for k, bound in extra_cons:
            cons.append(T[k] @ b <= bound)
    prob = cp.Problem(cp.Minimize(M), cons)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    bval = np.asarray(b.value, dtype=np.float64)
    bval = np.maximum(bval, 0)
    s = bval.sum()
    if s < 1e-15:
        return None, None
    return bval / s, float(M.value)

def smart_cd(a0, time_limit, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            affected = conv[k_start:k_end + 1].copy()
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)
            top_k = min(6, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)
            for frac in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.3 * ai)

            best_delta = 0.0
            best_new_max = M
            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                new_aff = affected + delta * coeffs
                if has_quad:
                    new_aff = new_aff.copy()
                    new_aff[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_aff.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur = 2.0 * n * M
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a).copy()
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 150:
                break
        else:
            no_improve = 0

    return best_a, best_score
