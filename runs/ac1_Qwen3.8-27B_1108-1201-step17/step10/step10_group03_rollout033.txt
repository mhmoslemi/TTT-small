1. Finding the t where the maximum changes (i.e., where two quadratics cross at the top)
2. Evaluating at critical points

def exact_line_search(a, b, conv_a, conv_b, conv_ab):
    """Find t in [0,1] minimizing max_k conv((1-t)a+tb, (1-t)a+tb)[k].
    
    conv((1-t)a+tb, (1-t)a+tb) = (1-t)^2 * conv_a + 2t(1-t) * conv_ab + t^2 * conv_b
    
    Each k gives a quadratic: q_k(t) = (1-t)^2*conv_a[k] + 2t(1-t)*conv_ab[k] + t^2*conv_b[k]
    = conv_a[k] + t*(2*conv_ab[k] - 2*conv_a[k]) + t^2*(conv_a[k] - 2*conv_ab[k] + conv_b[k])
    
    We want min_t max_k q_k(t) for t in [0,1].

    C = conv_a  # (1-t)^2 * conv_a -> at t=0, this is conv_a
    # Actually: (1-t)^2 * C_k + 2t(1-t) * D_k + t^2 * E_k
    # = C_k - 2*C_k*t + C_k*t^2 + 2*D_k*t - 2*D_k*t^2 + E_k*t^2
    # = C_k + (2*D_k - 2*C_k)*t + (C_k - 2*D_k + E_k)*t^2
    # where C_k = conv_a[k], D_k = conv_ab[k], E_k = conv_b[k]
    
    A = conv_a - 2 * conv_ab + conv_b
    B = 2 * (conv_ab - conv_a)
    C = conv_a
    
    # objective(t) = max_k (A_k * t^2 + B_k * t + C_k)
    # Use scipy minimize_scalar with bounded
    
    def obj(t):
        vals = A * t * t + B * t + C
        return vals.max()
    
    res = minimize_scalar(obj, bounds=(0, 1), method='bounded', options={'xatol': 1e-12})
    return res.x, res.fun

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def local_eval(a):

    s = a.sum()

def normalize(a):

    s = a.sum()
    if s < 1e-15:

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver):

def exact_line_search(conv_aa, conv_bb, conv_ab):
    C = conv_aa
    D = conv_ab
    E = conv_bb
    qa = C - 2.0 * D + E
    qb = 2.0 * (D - C)
    qc = C

    def obj_fn(t):
        return (qa * t * t + qb * t + qc).max()

    try:
        res = minimize_scalar(obj_fn, bounds=(0.0, 1.0), method='bounded',

def lse_obj_grad(a, T, n):
    b = fast_convolve(a)
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()
    lse = b_max + np.log(exp_ws.sum())
    val = 2.0 * n * lse
    grad_a = np.zeros(n)
    active = np.where(w > 1e-14)[0]
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    return val, grad_a


def topk_lse_obj_grad(a, T, k_frac, n):
    b = fast_convolve(a)
    size = 2 * n - 1
    k_top = max(1, int(k_frac * size))
    sorted_idx = np.argsort(b)[::-1]
    top_idx = sorted_idx[:k_top]
    b_top = b[top_idx]
    b_max = b_top.max()
    exp_ws = np.exp(b_top / T - b_max / T)
    w_top = exp_ws / exp_ws.sum()
    lse = b_max / T + np.log(exp_ws.sum())
    val = 2.0 * n * lse
    w = np.zeros(size)
    w[top_idx] = w_top
    grad_a = np.zeros(n)
    active = np.where(w > 1e-14)[0]
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    return val, grad_a


def log_param_obj(z, T, n, k_frac):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a_cur = exp_z / exp_z.sum()
    if k_frac is None or k_frac >= 1.0:
        val, grad_a = lse_obj_grad(a_cur, T, n)
    else:
        val, grad_a = topk_lse_obj_grad(a_cur, T, k_frac, n)
    S = np.dot(grad_a, a_cur)
    grad_z = a_cur * (grad_a - S)
    return val, grad_z

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_tk, _ = adam_minimize(z_init, n_cur, T_stages, k_schedule, t4, seed=seed)
        try_update(a_tk, "topk_adam")

    # Phase 5: Second peak equalization pass
    print(f"\n=== Phase 5: Peak Equalization Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 200.0)
    if t5 > 20:
        a_pe2, _ = peak_equalization(best_seq, solver, t5, max_steps=1500)
        try_update(a_pe2, "peak_eq_2")

    # Phase 6: Additional n exploration near best
    print(f"\n=== Phase 6: Nearby n exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        nearby_ns = []
        for mult in [0.9, 1.1, 1.25, 0.8]:
            cand_n = int(best_n * mult)
            if 400 <= cand_n <= 4000 and cand_n != best_n and cand_n not in nearby_ns:
                nearby_ns.append(cand_n)
        t6_total = min(rem * 0.25, 200.0)
        per_n6 = t6_total / max(1, len(nearby_ns))
        for cand_n in nearby_ns:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_sub = min(per_n6, rem)
            if t_sub < 15:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            a_cand, _ = lp_alternation_exact(init_cand, solver, t_sub, max_iters=10000)
            try_update(a_cand, f"lp_nearby_n{cand_n}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7a = min(rem * 0.3, 60.0)
        a_fin, _ = peak_equalization(best_seq, solver, t7a, max_steps=500)
        try_update(a_fin, "final_pe")
    rem = deadline - time.monotonic()
    if rem > 20:
        n_cur = len(best_seq)
        a_clipped = np.maximum(best_seq, 1e-300)
        z_init = np.log(a_clipped) - np.log(a_clipped.sum())
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        k_schedule = [0.05, 0.02, 0.01, 0.5, 0.5, 0.5, 0.5, 0.5]
        t7b = min(rem * 0.4, 60.0)
        a_fin2, _ = adam_minimize(z_init, n_cur, T_stages, k_schedule, t7b, seed=seed + 1)
        try_update(a_fin2, "final_adam")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
