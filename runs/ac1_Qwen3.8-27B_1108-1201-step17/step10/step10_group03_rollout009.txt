"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided gradient descent,
LogSumExp L-BFGS with FFT-accelerated gradients, Fourier domain refinement,
and multi-length exploration.
Key innovations: (1) using LP dual weights for exact subgradient computation,
(2) FFT-accelerated gradient evaluation, (3) momentum-based gradient descent
with backtracking, (4) aggressive multi-n search.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def local_eval(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_subgradient(a, w):
    """Compute subgradient of max_k conv(a,a)[k] using weights w.
    
    grad[j] = 2 * sum_k w[k] * a[k-j]
    where w has length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # c[j] = sum_m w[j+m] * a[m] = cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    C = irfft(W * np.conj(A), N)[:n]
    return 2.0 * C


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def solve_lp_dual_weights(a, solver, time_limit=10.0):
    """Solve the best-response LP and return the dual weights w.
    
    LP: minimize M
        subject to: conv(a, b)[k] <= M for all k
                    sum(b) = 1
                    b >= 0
    
    Returns: (w, b, M) where w are dual weights (normalized to sum=1),
             b is the optimal best response, M is the optimal value.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: T @ b <= M
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Use sparse matrix for efficiency
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    t0 = time.monotonic()
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
            
            # Get dual values
            # The constraint T @ b_var <= M_var has dual variable for each row
            # We need to extract the dual values
            # In cvxpy, dual values are accessible via constraint.dual_value
            dual_vals = np.zeros(size)
            # The first constraint is T @ b_var <= M_var
            if prob.constraints[0].dual_value is not None:
                dv = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
                if len(dv) == size:
                    dual_vals = -dv  # cvxpy gives negative values for <= constraints
                    dual_vals = np.maximum(dual_vals, 0)
                    ds = dual_vals.sum()
                    if ds > 1e-15:
                        dual_vals = dual_vals / ds
                    else:
                        # Fallback: use empirical weights based on proximity to max
                        conv_ab = np.array([np.dot(a[max(0,k-n+1):min(n,k+1)], 
                                                   b_val[max(0,n-1-k):n-1-k+n+1]) for k in range(size)])
                        # Actually let me compute it properly
                        conv_ab = _convolve_arrays(a, b_val)
                        M_val = conv_ab.max()
                        threshold = M_val * 0.95
                        active = conv_ab >= threshold
                        if active.sum() > 0:
                            dual_vals = np.zeros(size)
                            dual_vals[active] = 1.0 / active.sum()
                        else:
                            dual_vals = np.zeros(size)
                            dual_vals[np.argmax(conv_ab)] = 1.0
            
            M_val = float(M_var.value)
            return dual_vals, b_val, M_val
    except Exception as e:
        pass
    
    # Fallback: compute weights from empirical peaks
    # Just use uniform weights on near-peak positions
    b_guess = a.copy()  # use a itself as a rough best response
    conv_aa = fast_convolve(a)
    M_val = conv_aa.max()
    threshold = M_val * 0.9
    active = conv_aa >= threshold
    w = np.zeros(size)
    if active.sum() > 0:
        w[active] = 1.0 / active.sum()
    else:
        w[np.argmax(conv_aa)] = 1.0
    return w, b_guess, M_val


def _convolve_arrays(a, b):
    """Compute linear convolution of two arrays using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def lp_guided_gradient_descent(a0, solver, time_limit=600, re_solve_interval=100):
    """Gradient descent on the original (non-convex) problem using LP dual weights.
    
    At each step:
    1. (Every re_solve_interval steps) Solve LP to get optimal dual weights w.
    2. Compute subgradient: g = 2 * cross_corr(w, a)
    3. Projected gradient step with momentum and backtracking.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Get initial LP weights
    w, b_lp, M_lp = solve_lp_dual_weights(a, solver)
    print(f"    Initial LP: M={M_lp:.8f}, active peaks={np.sum(w > 0.01)}")
    
    # Momentum
    velocity = np.zeros(n)
    beta = 0.9  # momentum coefficient
    lr = 0.1  # initial learning rate
    lr_min = 1e-8
    lr_max = 0.5
    
    it = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Periodically re-solve LP
        if it % re_solve_interval == 1:
            w, b_lp, M_lp = solve_lp_dual_weights(normalize(a), solver)
            # Update momentum based on new weights
            beta = min(0.95, 0.9 + 0.01 * min(it // re_solve_interval, 5))
        
        # Compute subgradient
        g = compute_subgradient(a, w)
        
        # Momentum step
        velocity = beta * velocity + (1 - beta) * g
        direction = -velocity
        
        # Backtracking line search
        current_obj = 2.0 * n * fast_convolve(a).max()
        step_taken = False
        for backtrack in range(15):
            trial_lr = lr / (2 ** backtrack)
            a_trial = project_simplex(a + trial_lr * direction)
            obj_trial = 2.0 * n * fast_convolve(a_trial).max()
            if obj_trial < current_obj - 1e-14:
                a = a_trial
                step_taken = True
                break
            elif obj_trial <= current_obj:
                a = a_trial
                step_taken = True
                break
        
        if not step_taken:
            # No improvement, reduce learning rate
            lr = max(lr * 0.5, lr_min)
            no_improve += 1
            
            if no_improve >= max_no_improve:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = normalize(a + rng.standard_normal(n) * 0.01)
                velocity *= 0.1
                no_improve = 0
                lr = 0.05
                # Re-solve LP
                w, b_lp, M_lp = solve_lp_dual_weights(a, solver)
            continue
        
        # Update best
        obj_new = 2.0 * n * fast_convolve(a).max()
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, lr_max)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr = max(lr * 0.9, lr_min)
        
        # Progress print
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-GD iter={it} ({elapsed:.0f}s): score={best_score:.10f}, lr={lr:.6f}")
    
    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with FFT-accelerated gradients."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(30, int(remaining * 10))
        
        # Precompute FFT size
        fft_size = 1 << (2 * n - 1).bit_length()
        N = 1 << (3 * n - 2).bit_length()
        
        def obj(z_x):
            z_max = z_x.max()
            exp_z = np.exp(z_x - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse
        
        def grad(z_x):
            z_max = z_x.max()
            exp_z = np.exp(z_x - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            
            # Compute gradient using FFT
            # grad_a[j] = 2 * sum_k w[k] * a_cur[k-j]
            # = 2 * cross_corr(w, a_cur)[j]
            g_a = compute_subgradient(a_cur, w)
            
            # Chain rule for softmax: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(g_a, a_cur)
            return a_cur * (g_a - S)
        
        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-25, 'gtol': 1e-16,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = np.asarray(res.x, dtype=np.float64)
            z = z - z.max()  # numerical stability
            a_new = np.exp(z)
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
        except Exception as e:
            print(f"    L-BFGS error at T={T}: {e}")
        
        # Update z for next temperature
        a_cur = best_a
        a_clipped = np.maximum(a_cur, 1e-300)
        z = np.log(a_clipped) - np.log(a_clipped.sum())
    
    return best_a, best_score

    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    
    n_free = 2 * n_freq + 1
    x0 = np.zeros(n_free)
    x0[0] = A[0].real
    for i in range(1, n_freq + 1):
        x0[2 * i - 1] = A[i].real
        x0[2 * i] = A[i].imag
    
    A_rest = A[n_freq + 1:].copy()
    
    def fourier_obj(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()
    
    try:
        res = minimize(
            fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': 3000, 'xatol': 1e-15, 'fatol': 1e-15, 'adaptive': True}
        )
        x_opt = res.x
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s > 1e-10:
            a_candidate = a_candidate / s
            sc = 2.0 * n * fast_convolve(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
    except Exception:
        pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    else:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS
