import numpy as np

import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_grad_a(w, a):
    """Compute grad_a[i] = sum_{j=0}^{n-1} w[i+j] * a[j] using FFT correlation."""
    n = len(a)
    # Full linear correlation of w (len 2n-1) with a (len n) has length 3n-2
    full_len = (2 * n - 1) + n - 1
    fft_size = 1 << (full_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    # Valid correlation (where both signals fully overlap) is at [n-1, n-1+n)
    return corr[n - 1:n - 1 + n]

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    return np.log(np.maximum(a, 1e-300))

    grad_z = a * (d_obj_da - S)

    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK]
    solved = False

        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_val = b_val.sum()
    if s_val < 1e-15:

    b_val = b_val / s_val

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() > deadline:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        avg_lp = np.mean(lp_times[-30:]) if len(lp_times) >= 10 else 0.5

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 500 == 0:
                print(f"    NesterovLP iter={it+1}: score={best_score:.8f} avg_lp={avg_lp*1000:.0f}ms")

        if len(lp_times) > 50 and avg_lp > 5.0:
            break

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_sweeps, deadline):
    """Coordinate descent: for each coordinate, optimize it in 1D via golden section."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s_total = a.sum()
    if s_total < 1e-15:
        return a0.copy()
    a_norm = a / s_total
    best_score = evaluate_local(a_norm)
    t0 = time.monotonic()

    # Precompute: for each coordinate i, the constant parts of Q_k
    # Q_k(x) for coordinate i:
    #   k = 2i: x^2 + c_{2i}
    #   k in [i, i+n-1], k != 2i: 2*x*a[k-i] + c_k
    #   else: c_k (constant)
    # where c_k = Q_k(0) = conv(a_old, a_old)[k] - terms involving a[i]

    def score_at(x, i, a_other_sum, c_vals, linear_coeffs):
        """Compute score when a[i] = x, others fixed."""
        # c_vals[k] = constant part of Q_k (not depending on x)
        # linear_coeffs: list of (coef, const) pairs for the linear/quadratic terms
        # Max of: x^2 + c[2i], max_j (2*x*lincoeff[j] + c[valid_k]), max of constants
        quad_val = x * x + c_vals[2 * i] if 2 * i < len(c_vals) else 0
        max_val = quad_val
        for (k_idx, coef) in linear_coeffs:
            val = 2.0 * x * coef + c_vals[k_idx]
            if val > max_val:
                max_val = val
        # Also check all constant constraints
        max_const = np.max(c_vals)
        if max_const > max_val:
            max_val = max_const
        S = a_other_sum + x
        if S < 1e-15:
            return 1e15
        return 2.0 * n * max_val / (S * S)

    # For efficiency, precompute c_vals and which k's matter
    # c_k = conv(a,a)[k] with a[i] set to 0
    a_zero = a_norm.copy()
    original_ai = a_zero[i] if False else 0  # placeholder

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit or time.monotonic() > deadline:
            break

        for i in range(n):
            if time.monotonic() - t0 > time_limit or time.monotonic() > deadline:
                break

            # Temporarily set a[i] to 0
            old_val = a_norm[i]
            a_norm[i] = 0.0
            s_other = a_norm.sum()

            # Compute c_k = conv(a_with_zero, a_with_zero)[k]
            b_zero = fast_convolve(a_norm)

            # For the coordinate i, the terms involving x = a[i]:
            # Q_k(x) = b_zero[k] + x * (a[k-i] if valid) + (x * a[i-... hmm
            # Actually: Q_k(x) = sum_j a'[j] * a'[k-j] where a'[i] = x, a'[j] = a_norm[j] for j != i
            # = b_zero[k] + x * a_norm[k-i] (if 0 <= k-i < n and k-i != i)
            #   + x * a_norm[i-(k)] ... wait
            # Let me think again.
            # conv(a',a')[k] = sum_j a'[j] * a'[k-j]
            # = sum_{j != i} a'[j] * a'[k-j] + a'[i] * a'[k-i] + a'[k-i] * a'[i] (if k-i != i)
            # = b_zero[k] + x * a'[k-i] + (if k-i != i and 0<=k-i<n) x * a'[k-i]
            # Wait, b_zero already has a'[i] = 0, so all terms involving a'[i] are 0 in b_zero.
            # The terms that involve a'[i] = x are:
            #   j = i: a'[i] * a'[k-i] = x * a_norm[k-i] (valid if 0 <= k-i < n)
            #   j = k-i: a'[k-i] * a'[i] = a_norm[k-i] * x (valid if 0 <= k-i < n and k-i != i)
            # If k-i == i (i.e., k = 2i): only one term, x * x = x^2
            # If k-i != i: two terms, 2x * a_norm[k-i]

            # So: Q_k(x) = b_zero[k] + [x^2 if k==2i] + [2x*a_norm[k-i] if 0<=k-i<n and k!=2i]
            #           = b_zero[k] + [2x*a_norm[k-i] if 0<=k-i<n and k!=2i] + [x^2 if k==2i]

            # The max of Q_k(x) over k is what matters.
            # For large n, most k don't depend on x, so the max is determined by:
            #   - The constant max (b_zero at k's that don't involve i)
            #   - The terms at k in {i, ..., i+n-1} (where a_norm[k-i] is valid)

            # For efficiency, only consider k in the range where x appears:
            # k = 2i (quadratic), and k in [i, i+n-1] \ {2i} (linear)
            # Plus the global max of b_zero (constant lower bound on the max)

            k_range = list(range(max(0, i), min(2 * n - 1, i + n)))
            # Add k = 2i to the range if not already there
            if 2 * i not in k_range and 2 * i < 2 * n - 1:
                k_range.append(2 * i)

            # Precompute coefficients
            c_vals_list = []
            lin_coeffs = []  # (k_index, a_norm[k-i])
            for k in k_range:
                if k == 2 * i:
                    c_vals_list.append(b_zero[k])  # will add x^2
                elif 0 <= k - i < n:
                    c_vals_list.append(b_zero[k])
                    lin_coeffs.append((k, a_norm[k - i]))
                else:
                    c_vals_list.append(b_zero[k])

            # Also need the max of b_zero over all k NOT in k_range (constant constraint)
            mask = np.ones(2 * n - 1, dtype=bool)
            for k in k_range:
                if 0 <= k < 2 * n - 1:
                    mask[k] = False
            const_max = np.max(b_zero[mask]) if mask.any() else 0.0

            c_arr = np.array([b_zero[k] for k in k_range])

            # Golden section search for x in [0, x_max]
            x_max = max(old_val * 3.0, s_other / n * 4.0, 0.01)
            x_max = min(x_max, s_other * 2.0)  # don't let x exceed 2x the rest

            phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

            def f_of_x(x):
                # max of: x^2 + c[2i_idx], 2*x*lincoeff + c[k_idx] for each linear term, const_max
                max_val = const_max
                for idx, k in enumerate(k_range):
                    if k == 2 * i:
                        val = x * x + c_arr[idx]
                    elif idx < len(lin_coeffs) and (k, a_norm[k - i]) in lin_coeffs:
                        # Find the coefficient
                        # Actually let me precompute this differently
                        val = c_arr[idx]
                    else:
                        val = c_arr[idx]
                    if val > max_val:
                        max_val = val
                # Handle linear terms properly
                for (k_idx, coef) in lin_coeffs:
                    val = 2.0 * x * coef + b_zero[k_idx]
                    if val > max_val:
                        max_val = val
                if k_range and 2 * i in k_range:
                    val = x * x + b_zero[2 * i]
                    if val > max_val:
                        max_val = val
                S = s_other + x
                if S < 1e-15:
                    return 1e15
                return 2.0 * n * max_val / (S * S)

            # Golden section
            a_g = 0.0
            b_g = x_max
            c_g = b_g - phi * (b_g - a_g)
            d_g = a_g + phi * (b_g - a_g)
            fc = f_of_x(c_g)
            fd = f_of_x(d_g)

            for _ in range(40):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - phi * (b_g - a_g)
                    fc = f_of_x(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + phi * (b_g - a_g)
                    fd = f_of_x(d_g)

            x_opt = (a_g + b_g) / 2.0
            f_opt = f_of_x(x_opt)

            # Also try x = 0 and x = old_val for comparison
            f_zero = f_of_x(0.0)
            f_old = f_of_x(old_val)

            if f_opt < f_zero and f_opt < f_old:
                a_norm[i] = x_opt
            elif f_zero < f_old:
                a_norm[i] = 0.0
            else:
                a_norm[i] = old_val

        # Renormalize and check improvement
        a_norm = np.maximum(a_norm, 0)
        sn = a_norm.sum()
        if sn < 1e-15:
            break
        a_norm = a_norm / sn
        sc = evaluate_local(a_norm)
        if sc < best_score:
            best_score = sc

        if sweep % 3 == 0:
            print(f"    CoordDescent sweep {sweep}: score={best_score:.8f}")

    return a_norm


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 + 0.1 * np.sin(2 * np.pi * 5 * x)
    else:
        a = x ** 0.5 * (1 - x) ** 0.3 + 0.05

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_subgradient(a0, time_limit, n_steps, deadline):
    """Mirror descent targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.04
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit or time.monotonic() > deadline:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])

        # Use FFT to compute gradient
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        grad_a = fast_grad_a(w, a)

        eta = eta_base * scale
        log_scale = -eta * grad_a
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 60:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.CBC, cp.GLOP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return 'ECOS'


def propose_candidate(seed=42, budget_s=1800.0):
    """

    """

            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200

        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    print(f"\n=== Phase 1: L-BFGS LogSumExp (score={best_score:.8f}) ===")

    rem = deadline - time.monotonic()

                     0.003, 0.002, 0.001, 0.0005]

                             deadline=deadline)

    # === Phase 4: Second Nesterov LP ===

    # === Phase 7: Multiplicative subgradient ===

    rem = deadline - time.monotonic()
    if rem > 15:

                                          n_sweeps=20, deadline=deadline)
        local_update(a_cd2, "final_coord")

    rem = deadline - time.monotonic()

                                           solver, n_max_iters=50000, deadline=deadline)
        local_update(a_lp_final, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):

        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        elapsed = time.monotonic() - t_start
        print(f"Total time: {elapsed:.1f}s")

    # Ensure we return valid output
    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
