import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if cond.any():
        rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def softmax_shifted(x, alpha):
    m = x.max()
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    return exp_shifted / exp_shifted.sum()

def soft_gradient(a, alpha):
    n = len(a)
    conv = fast_convolve(a)
    weights = softmax_shifted(conv, alpha)
    # grad[i] = 2 * alpha * sum_j weights[i+j] * a[j] for j=0..n-1
    # This is 2*alpha * correlate(weights, a, 'valid')
    grad = 2.0 * alpha * np.correlate(weights, a, mode='valid')
    return grad, conv

def soft_grad_descent(a0, time_limit, alpha_start=50.0, alpha_end=5000.0,

def build_conv_matrix(a):

def solve_lp_best_response(a_fixed, solver=None):

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'SCS', 'GLPK', 'CBC', 'PDLP']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            n_aff = k_hi - k_lo + 1
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = max(3.0 * v, 5.0 / n)
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)

            def _score_fn(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            fc = _score_fn(c)
            fd = _score_fn(d)
            for _ in range(25):
                if hi - lo < 1e-14:
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _score_fn(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _score_fn(d)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score

def peak_shaving(a0, time_limit, verbose=False):
    """Directly reduce the maximum convolution peak by adjusting contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())
        peaks = np.where(conv >= max_val * 0.999)[0]

        for k in peaks:
            # Find which a[i] contribute to conv[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            partners = (k - idx)

            # For each contributing pair (i, k-i), reduce the larger one
            for ii in idx:
                jj = k - ii
                if jj < 0 or jj >= n:
                    continue
                # Reduce a[ii] slightly, redistribute to reduce peak
                reduction = 0.01 * a[ii]
                a[ii] -= reduction
                # Add to a random other position to maintain sum
                j_pos = np.random.randint(0, n)
                a[j_pos] += reduction

        a = np.maximum(a, 0)
        cs = a.sum()
        if cs < 1e-10:
            break
        a = a / cs
        sc = compute_score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

def multi_start_local_search(a0, time_limit, rng, verbose=False):
    """Multiple random perturbations followed by coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        t_alloc = min(rem * 0.3, 60.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Gaussian noise
            sigma = 0.01 + 0.03 * rng.random()
            perturbed += rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Targeted: reduce high-convolution contributors
            conv_p = fast_convolve(a)
            top_k = min(10, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.random.choice(np.arange(i_start, i_end + 1),
                                       size=min(3, i_end - i_start + 1), replace=False)
                for ii in idx:
                    perturbed[ii] *= (1.0 - 0.05 * rng.random())
        elif pert_type == 2:
            # Block swap
            blen = rng.integers(n // 10, max(2, n // 4))
            i1 = rng.integers(0, n - blen)
            i2 = rng.integers(0, n - blen)
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1+blen].copy()
                perturbed[i1:i1+blen] = perturbed[i2:i2+blen]
                perturbed[i2:i2+blen] = temp
        elif pert_type == 3:
            # Mirror blend
            shifted = np.roll(a, rng.integers(1, n // 8))
            t = 0.1 + 0.2 * rng.random()
            perturbed = (1 - t) * a + t * shifted
        else:
            # Sparse perturbation: change a few random elements
            n_change = rng.integers(3, 10)
            idx = rng.choice(n, size=n_change, replace=False)
            perturbed[idx] *= (1.0 + rng.standard_normal(n_change) * 0.1)

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"msl_{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"      MSL {count}: improved to {best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.2, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_ring(n, rng, freq):
    """Initialize with a ring-modulated structure."""
    t = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(2 * np.pi * freq * t / n)) ** 2
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        # Create initial guess
        n_init = 1152
        parent = init_asymmetric(n_init, rng, 0.3)
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    # Phase 1: Soft gradient descent with continuation (fast)
    print("\n=== Phase 1: Soft Gradient Descent ===")
    n_work = len(best_seq)
    t1_budget = min(200.0, (deadline - time.monotonic()) * 0.2)
    print(f"  Budget: {t1_budget:.0f}s, n={n_work}")
    a_sgd, s_sgd = soft_grad_descent(best_seq, t1_budget,
                                     alpha_start=20.0, alpha_end=2000.0,
                                     alpha_growth=1.05, verbose=True)
    sc = float(evaluate_sequence(a_sgd.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_sgd.copy()
        print(f"  NEW BEST SGD: {sc:.8f}")

    # Phase 2: LP Fictitious Play
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_fp = min(rem * 0.5, 300.0)
        print(f"  Budget: {t_fp:.0f}s, n={n_work}")
        a_fp, s_fp = lp_fictitious_play(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            print(f"  NEW BEST FP: {sc:.8f}")

        # Soft gradient after FP
        rem2 = deadline - time.monotonic() - 500
        if rem2 > 30:
            t_sg2 = min(rem2 * 0.3, 100.0)
            a_sg2, s_sg2 = soft_grad_descent(best_seq, t_sg2,
                                             alpha_start=100.0, alpha_end=5000.0,
                                             alpha_growth=1.06)
            sc2 = float(evaluate_sequence(a_sg2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_sg2.copy()
                print(f"  NEW BEST SGD2: {sc2:.8f}")

    # Phase 3: Exact coordinate descent polishing
    print(f"\n=== Phase 3: ECD Polishing (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(4):
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        t_cd = min(rem * 0.2, 120.0)
        print(f"  Round {rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd, _ = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph3_{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD: {sc:.8f}")

    # Phase 4: Multi-start local search
    print(f"\n=== Phase 4: Multi-Start Local Search (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_msl = min(rem * 0.6, 250.0)
        print(f"  Budget: {t_msl:.0f}s")
        a_msl, s_msl = multi_start_local_search(best_seq, t_msl, rng, verbose=True)
        sc = float(evaluate_sequence(a_msl.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_msl.copy()
            print(f"  NEW BEST MSL: {sc:.8f}")

    # Phase 5: Try different sequence lengths
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for alt in [n_best - 64, n_best - 128, n_best + 64, n_best + 128, 1024, 1280, 1536]:
        if 256 <= alt <= 2500 and alt != n_best and alt not in alt_lengths:
            alt_lengths.append(alt)

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alt = min(rem * 0.25, 80.0)
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")

        # Interpolate from best
        a_init = interpolate_seq(best_seq, alt_n)
        # Quick ECD
        a_opt, s_opt, _ = exact_coord_descent(a_init, t_alt * 0.6, label=f"alt_{alt_n}")
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST alt n={alt_n}: {sc:.8f}")

    # Phase 6: Final intensive polishing
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 50
    if rem > 10:
        t_final = min(rem * 0.5, 100.0)
        print(f"  Final ECD budget: {t_final:.0f}s")
        a_fin, s_fin, _ = exact_coord_descent(best_seq, t_final, verbose=True, label="final")
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST Final: {sc:.8f}")

    # Phase 7: One more round of soft gradient + ECD
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t_sg = min(rem * 0.4, 50.0)
        a_sg3, s_sg3 = soft_grad_descent(best_seq, t_sg,
                                         alpha_start=500.0, alpha_end=10000.0,
                                         alpha_growth=1.04)
        sc = float(evaluate_sequence(a_sg3.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_sg3.copy()

        rem2 = deadline - time.monotonic() - 10
        if rem2 > 10:
            a_last, s_last, _ = exact_coord_descent(best_seq, min(rem2 * 0.7, 30.0), label="last")
            sc = float(evaluate_sequence(a_last.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_last.copy()

    # Final cleanup: tiny coordinate descent pass
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a_last2, s_last2, _ = exact_coord_descent(best_seq, min(rem * 0.8, 15.0), label="last2")
        sc = float(evaluate_sequence(a_last2.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last2.copy()

    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(max(0.0, x)) for x in best_seq]
