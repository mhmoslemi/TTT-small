    n = len(a0)
    a = a0.copy() / a0.sum()
    conv = fast_convolve(a)
    M = conv.max()
    K = 10  # number of active constraints
    t = time.monotonic()
    
    while time.monotonic() - t < time_limit:
        # Find active constraints
        if K < 2*n-1:
            active = np.argpartition(conv, -K)[-K:]
        else:
            active = np.arange(2*n-1)
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_start = max(0, k-n+1)
            i_end = min(n-1, k)
            grad[i_start:i_end+1] += 2.0 * a[k-i_end:k-i_start+1][::-1]
        # Handle the diagonal (where i = k-i)
        for k in active:
            k = int(k)
            if k % 2 == 0 and 0 <= k//2 < n:
                grad[k//2] -= a[k//2]  # subtract the double count
        
        grad /= len(active)

        j = np.argmin(grad)
        e_j = np.zeros(n)
        e_j[j] = 1.0
        
        # Step size (line search or fractional)

            val = conv_trial.max()

                best_step = alpha

            conv = fast_convolve(a)
            M = conv.max()
    
    return a, 2.0*n*M

from scipy.optimize import linprog

def solve_lp_best_response(a, c_prev=None):
    """Solve: min M s.t. T@b <= M*1, sum(b)=1, b>=0"""
    n = len(a)
    
    # Build T matrix
    T = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(n-1, k)
        T[k, i_start:i_end+1] = a[k-i_end:k-i_start+1][::-1]
    
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M → c = [0,0,...,0,1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Inequality constraints: T@b - M <= 0
    # A_ub @ [b, M] <= b_ub
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = T
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M unbounded
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    b_opt = result.x[:n]
    M_opt = result.x[-1]
    return b_opt, M_opt

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):

    s = a.sum()

        return float('inf')

    return float(2.0 * n * conv.max() / (s * s))


def build_T(a):
    """Build convolution matrix T where T[k,i] = a[k-i] if valid."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        T[k, i_start:i_end+1] = a[k - i_end:k - i_start + 1][::-1]
    return T


def lp_best_response(a, T=None):
    """Solve LP: min M s.t. T@b <= M*1, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a)
    if T is None:
        T = build_T(a)
    
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: T@b - M <= 0
    A_ub = np.hstack([T, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # M >= 0 (since a >= 0, conv >= 0)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    b_opt = np.maximum(result.x[:n], 0)
    s = b_opt.sum()
    if s < 1e-15:
        return None, None
    b_opt = b_opt / s
    M_opt = float(result.x[-1])
    return b_opt, M_opt

Let me try Anderson acceleration. The idea is:
- Keep a history of the last m iterates: a_0, a_1, ..., a_{m-1}
- At each step, find coefficients gamma_0, ..., gamma_{m-1} that minimize ||sum gamma_i * (a_i - F(a_i))||^2
- where F is the fixed-point map (F(a) = best_response(a))
- The new iterate is sum gamma_i * a_i

    F_hist = [a.copy()]  # F(a) values

        if time.monotonic() - t0 > time_limit:
            break

        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Anderson acceleration
        F_hist.append(b_new.copy())
        A_hist.append(a.copy())

            A_hist = A_hist[-m:]
            F_hist = F_hist[-m:]

            # g_i = F(a_i) - a_i
            g = np.zeros((len(A_hist), n))
            for i in range(len(A_hist)):
                g[i] = F_hist[i] - A_hist[i]
            
            # Solve: min ||sum gamma_i * g_i||^2 s.t. sum gamma_i = 1
            # This is: min ||G^T * gamma||^2 s.t. 1^T * gamma = 1
            # G is (m x n), gamma is (m,)
            # G^T * G is (n x n)... no, that's wrong.

            # where G is (m x n), so G^T is (n x m), gamma is (m,)
            # min (gamma^T G G^T gamma) s.t. sum(gamma) = 1
            # G G^T is (m x m)

            # [1^T    0] [nu  ] = [1]
            m_kk = len(g)
            KKT = np.zeros((m_kk+1, m_kk+1))
            KKT[:m_kk, :m_kk] = 2 * GGT
            KKT[:m_kk, m_kk] = 1
            KKT[m_kk, :m_kk] = 1
            rhs = np.zeros(m_kk+1)
            rhs[m_kk] = 1
            
            try:
                sol = np.linalg.solve(KKT + 1e-10 * np.eye(m_kk+1), rhs)
                gamma = sol[:m_kk]

                gamma = np.ones(m_kk) / m_kk

        a_lp3, s_lp3 = lp_alternate(best_seq, t_alloc, verbose=True, use_anderson=True, m_anderson=5)
        safe_update(a_lp3, "lp_round2")

    # Phase 3: Coordinate Descent Polish
    t3_end = deadline - 30
    if time.monotonic() < t3_end:
        t_alloc = min(t3_end - time.monotonic(), 200.0)
        print(f"\n=== Phase 3: CD Polish (best={best_score:.10f}) ===")
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, s_cd = cd_polish(best_seq, t_alloc, verbose=True)
        safe_update(a_cd, "cd_polish")

    # Phase 4: Final fine polish
    if time.monotonic() < t3_end - 10:
        t_alloc = min(t3_end - time.monotonic() - 10, 60.0)
        if t_alloc > 15:
            print(f"\n=== Phase 4: Final polish (budget={t_alloc:.0f}s) ===")
            a_fin, s_fin = cd_polish(best_seq, t_alloc, verbose=False)
            safe_update(a_fin, "final")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
