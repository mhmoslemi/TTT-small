def freq_gradient_descent(a0, time_limit, label=""):
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    N = 1 << (2*n - 1 - 1).bit_length()  # FFT size
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    lr = 0.1  # initial learning rate
    it = 0

        it += 1
        
        # Compute FFT of a
        A = rfft(a, N)
        
        # Compute autocorrelation
        c = irfft(A * np.conj(A), N)[:2*n-1]
        M = c.max()
        
        # Find peak positions (within tolerance)
        threshold = M - 0.01 * M
        peak_mask = c >= threshold
        
        # Compute subgradient in frequency domain
        # grad_a[i] = 2 * sum_{k in peaks} c_weight[k] * a[k-i] (for valid k-i)
        # In frequency domain: grad_A[m] = conj(A[m]) * sum_{k in peaks} c_weight[k] * exp(2*pi*i*m*k/N)
        # where c_weight is uniform over peaks
        
        # Actually, let's just compute the subgradient directly in the time domain
        # It's simpler and the cost is O(n * num_peaks)

        rho = 1.0 / num_peaks
        grad = np.zeros(n)
        for k in np.where(peak_mask)[0]:
            w = rho
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                grad[i] += 2.0 * w * a[k - i]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack
            lr *= 0.9
            if lr < 1e-10:
                break
        
        if it % 100 == 0:
            lr = min(lr * 1.1, 0.5)  # slowly increase lr
        
        if it % 200 == 0:
            print(f"    FreqGD {label} it={it}: best={best_score:.6f}, lr={lr:.4f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain gradient descent with log-sum-exp smoothing
2. Smart coordinate descent with adaptive candidate generation
3. LP fictitious play with accelerated convergence
4. Multi-length exploration with interpolation
5. Random restarts with structured perturbation
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

# Cached LP solver
_lp_solver_cached = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLOP, cp.SCS, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain smoothed gradient descent with Nesterov momentum
2. Fast fictitious play with first-order QP sub-solver (Nash averaging)
3. Smart coordinate descent with adaptive candidate generation
4. Multi-length exploration with interpolation
5. Random restarts to escape local optima
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

_lp_solver = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_smooth_gradient(a, c, beta):
    """Compute smooth gradient of log-sum-exp smoothed max(c) w.r.t. a."""
    n = len(a)
    # w[k] = softmax(beta * c[k])
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    w /= w.sum()
    
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j in [0, n-1]
    # Using FFT: cross_correlate(w_padded, a_padded)
    N = 1 << (3 * n - 3).bit_length()  # >= 3n-2
    V = np.zeros(N)
    V[:2*n-1] = w
    A = np.zeros(N)
    A[:n] = a
    cross = irfft(rfft(V) * np.conj(rfft(A)), N)
    grad = 2.0 * cross[:n]
    return grad

def smooth_gradient_descent(a0, time_limit, label="", beta_start=10.0):
    """Nesterov-accelerated smoothed gradient descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    c = fast_convolve(a)
    best_score = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    beta = beta_start
    lr = 1.0 / (4.0 * beta)  # initial step size
    it = 0
    y = a.copy()  # Nesterov variable
    gamma = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = (1.0 + gamma) * a - gamma * y
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Compute gradient at y
        c_y = fast_convolve(y)
        grad = get_smooth_gradient(y, c_y, beta)
        
        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Evaluate
        c_new = fast_convolve(a_new)
        score = 2.0 * n * c_new.max()
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack
            lr *= 0.8
            if lr < 1e-8:
                lr = 1.0 / (4.0 * beta)
                beta *= 1.5
        
        # Update Nesterov momentum
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        
        # Adapt beta
        if it % 200 == 0:
            beta *= 1.1
            lr = 1.0 / (4.0 * beta)
        
    return best_a, best_score

def fast_qp_solve(p, a_init, n_steps=200):
    """Solve min a^T Q_p a s.t. a>=0, sum(a)=1 using projected gradient.
    Q_p[i,j] = p[i+j] for i+j < 2n-1."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    # Adaptive step size
    # Lipschitz constant of 2*Q_p@x is at most 2*||Q_p||_2 <= 2*max_colsum
    # max_colsum of Q_p: sum_j p[i+j] for j in [0,n-1] <= sum p = 1
    # So L <= 2
    step = 0.4 / (1.0 + p.max() * n)  # conservative
    
    for _ in range(n_steps):
        # Compute Q_p @ a
        # (Q_p @ a)[i] = sum_j p[i+j] * a[j]
        N = 1 << (3 * n - 3).bit_length()
        V = np.zeros(N)
        V[:2*n-1] = p
        A = np.zeros(N)
        A[:n] = a
        Qa = irfft(rfft(V) * np.conj(rfft(A)), N)[:n]
        
        # Gradient step
        a_new = a - step * 2.0 * Qa
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a = a_new
    
    return a

1. Smoothed projected gradient descent with adaptive beta annealing

import numpy as np
from scipy.fft import rfft, irfft
import time

_lp_solver_cached = None


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():

def compute_smooth_gradient(a, c, beta):
    """
    Compute the gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    grad[i] = 2 * sum_j w[i+j] * a[j], where w = softmax(beta*c)
    """
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    w /= w.sum()

    N = 1 << (3 * n - 3).bit_length()
    V = np.zeros(N)
    V[:2 * n - 1] = w
    A = np.zeros(N)
    A[:n] = a
    cross = irfft(rfft(V) * np.conj(rfft(A)), N)
    grad = 2.0 * cross[:n]
    return grad


def qp_matvec(p, a):
    """Compute Q_p @ a where (Q_p)[i,j] = p[i+j]. Result: (Q_p@a)[i] = sum_j p[i+j]*a[j]."""
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()
    V = np.zeros(N)
    V[:2 * n - 1] = p
    A = np.zeros(N)
    A[:n] = a
    Qa = irfft(rfft(V) * np.conj(rfft(A)), N)[:n]
    return Qa


def smoothed_gradient_descent(a0, time_limit, label=""):
    """Adaptive-beta smoothed projected gradient descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = fast_convolve(a)
    best_score = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = 10.0
    lr = 0.5 / beta
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        c_y = fast_convolve(a)
        grad = compute_smooth_gradient(a, c_y, beta)

        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        c_new = fast_convolve(a_new)
        score = 2.0 * n * c_new.max()

        if score < best_score - 1e-16:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.7
            if no_improve > 50:
                beta *= 1.8
                lr = 0.5 / beta
                no_improve = 0
                if beta > 1e6:
                    break

        if it % 400 == 0:
            elapsed = time.monotonic() - t0

    return best_a, best_score


def fast_qp_solve(p, a_init, n_steps=80):
    """Solve min a^T Q_p a s.t. a>=0, sum(a)=1 via projected gradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    step = 0.45
    n_active = max(1, int(np.sum(p > p.max() * 0.5)))
    step = min(step, 1.0 / (1.0 + p.max() * n_active))

    for _ in range(n_steps):
        Qa = qp_matvec(p, a)
        a_new = a - step * 2.0 * Qa
        a_new = simplex_project(a_new)
        a = a_new

    return a
