```
def fft_convolution(a):
```

```
def lse_weights(b, T):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return np.array([], dtype=np.float64)
    bmax = float(b.max())
    if T <= 0.0 or not np.isfinite(T):
        active = np.where(b == bmax)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(b))])
        w = np.zeros(b.size, dtype=np.float64)
        w[active] = 1.0 / float(active.size)
        return w
    sh = (b - bmax) / T
    # sh <= 0 theoretically, but clip to 0 to avoid overflow if numerical
    np.minimum(sh, 0.0, out=sh)
    ex = np.exp(sh)
    sm = float(ex.sum())
    if sm <= 1e-300 or not np.isfinite(sm):
        active = np.where(b == bmax)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(b))])
        w = np.zeros(b.size, dtype=np.float64)
        w[active] = 1.0 / float(active.size)
        return w
    return ex / sm
```

```
def fft_core_gradient(w, a):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    expected = 2*n - 1
    if w.size != expected:
        # pad or crop
        tmp = np.zeros(expected, dtype=np.float64)
        m = min(w.size, expected)
        if m > 0:
            tmp[:m] = w[:m]
        w = tmp
    if float(w.sum()) <= 1e-300:
        return np.zeros(n, dtype=np.float64)
    full_len = 3*n - 2
    fft_size = 1 << (full_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a[::-1], fft_size)
    conv = irfft(W * A, fft_size)
    out = 2.0 * conv[n-1 : 2*n-1]
    np.maximum(out, 0.0, out=out)  # gradient should nonnegative? Is it always nonnegative? G_j=2 sum w a >=0. yes. But due numerical small negative. Clipping ok.
    return out
```

```
def subgradient_polish(a0, time_budget, rng=None):
    start = time.monotonic()
    a = normalize_sequence(a0)
    best = a.copy()
    best_sc = local_score(a)
    if time_budget <= 0.0:
        return best, best_sc
    n = a.size
    scale = 1.0
    step_count = 0
    while (time.monotonic() - start) < time_budget:
        b = fft_convolution(a)
        M = float(b.max())
        if M <= 1e-30:
            break
        Tpol = max(5e-6 * M, 1e-14)
        w = lse_weights(b, Tpol)
        G = fft_core_gradient(w, a)
        G = G - float(np.mean(G))
        gmax = float(np.max(np.abs(G)))
        if gmax <= 1e-16:
            break
        accepted = False
        try_factor = 0.1 * scale
        for fac in (1.0, 0.5, 0.25):
            trial = a - try_factor * fac * G
            trial = np.maximum(trial, 0.0)
            s = float(trial.sum())
            if s <= 1e-15:
                continue
            trial = trial / s
            bt = fft_convolution(trial)
            sc = float(2.0 * n * bt.max())
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best = trial
                best_sc = sc
                a = trial
                accepted = True
                break
        if accepted:
            scale = min(1.0, scale * 1.05)
        else:
            scale *= 0.5
            if scale < 1e-5:
                break
        step_count += 1
        if step_count % 2000 == 0:
            print(f"    subgrad step={step_count} score={best_sc:.10f}")
    return best, best_sc
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a)**2) for non-negative step heights.

This solver combines several non-convex local optimizers:
  * a Dirichlet-parameterized L-BFGS-B optimizer for a temperature-anncaled
    LogSumExp-smoothed maximum-convolution objective, with FFT-based gradients;
  * multi-resolution multi-start search, including asymmetric odd-mode starts
    and a small differential-evolution global search at low resolution;
  * an exact subgradient / trust-step polish;
  * an optional CVXPY active-set linearized trust-region polish.

The objective is scale invariant, so all internal optimization is done on the
simplex sum(a)=1. The final returned sequence is normalized to sum 1.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from operator import itemgetter

try:
    from scipy.optimize import differential_evolution
except Exception:
    differential_evolution = None

try:
    import cvxpy as cp
    _HAVE_CP = True
except Exception:
    cp = None
    _HAVE_CP = False


# ---------------------------------------------------------------------------
# Global optimization state for the top-level L-BFGS-B callbacks.
# ---------------------------------------------------------------------------
_CUR_N = 1
_CUR_ANCHOR = 0
_CUR_OTHER = np.array([], dtype=int)
_CUR_T = 1.0
_CUR_DEADLINE = 0.0

_CACHE_N = -1
_CACHE_ANCHOR = -1
_CACHE_T = -1.0
_CACHE_Y = None
_CACHE_A = None
_CACHE_VAL = None


# ---------------------------------------------------------------------------
# Basic helpers.
# ---------------------------------------------------------------------------
def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    n = a.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def fft_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def local_score(a):
    a = normalize_sequence(a)
    n = a.size
    b = fft_convolution(a)
    if b.size == 0:
        return float("inf")
    return float(2.0 * n * float(b.max()))


def interpolate_sequence(a, new_n):
    a = normalize_sequence(a)
    old_n = a.size
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    if new_n == old_n:
        return a.copy()
    if old_n <= 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, a)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        out = np.ones(new_n, dtype=np.float64) / float(new_n)
    else:
        out = out / s
    return out


# ---------------------------------------------------------------------------
# LogSumExp-smoothed objective and gradient.
# ---------------------------------------------------------------------------
def lse_weights(b, T):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return np.array([], dtype=np.float64)
    bmax = float(b.max())
    if T <= 0.0 or not np.isfinite(T):
        active = np.where(b == bmax)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(b))])
        w = np.zeros(b.size, dtype=np.float64)
        w[active] = 1.0 / float(active.size)
        return w

    sh = (b - bmax) / T
    np.minimum(sh, 0.0, out=sh)
    ex = np.exp(sh)
    sm = float(ex.sum())
    if sm <= 1e-300 or not np.isfinite(sm):
        active = np.where(b == bmax)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(b))])
        w = np.zeros(b.size, dtype=np.float64)
        w[active] = 1.0 / float(active.size)
        return w
    return ex / sm


def lse_value(a, T):
    b = fft_convolution(a)
    if b.size == 0:
        return 0.0
    bmax = float(b.max())
    if T <= 0.0 or not np.isfinite(T):
        return bmax
    sh = (b - bmax) / T
    np.minimum(sh, 0.0, out=sh)
    ex = np.exp(sh)
    sm = float(ex.sum())
    if sm <= 1e-300 or not np.isfinite(sm):
        return float(bmax)
    return float(bmax + T * np.log(sm))


def fft_core_gradient(w, a):
    """
    If w_k are weights on convolution coefficients b_k, return
    d(sum_k w_k b_k)/d(a).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)

    expected = 2 * n - 1
    if w.size != expected:
        tmp = np.zeros(expected, dtype=np.float64)
        m = min(w.size, expected)
        if m > 0:
            tmp[:m] = w[:m]
        w = tmp

    ws = float(w.sum())
    if ws <= 1e-300:
        return np.zeros(n, dtype=np.float64)

    full_len = 3 * n - 2
    fft_size = 1 << (full_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a[::-1], fft_size)
    conv = irfft(W * A, fft_size)
    out = 2.0 * conv[n - 1:2 * n - 1]
    np.maximum(out, 0.0, out=out)
    return out


def lse_gradient(a, T):
    b = fft_convolution(a)
    if b.size == 0:
        return np.zeros(a.size, dtype=np.float64)
    w = lse_weights(b, T)
    return fft_core_gradient(w, a)


# ---------------------------------------------------------------------------
# Dirichlet parameterization:
#   a[anchor] = 1 / S
#   a[i]      = y[i] / S
# where S = 1 + sum(y), y >= 0. This removes scale invariance and keeps all
# entries non-negative.
# ---------------------------------------------------------------------------
def set_opt_globals(n, anchor):
    global _CUR_N, _CUR_ANCHOR, _CUR_OTHER
    _CUR_N = int(n)
    _CUR_ANCHOR = int(anchor)
    if _CUR_N > 1:
        arr = np.arange(_CUR_N, dtype=int)
        _CUR_OTHER = arr[arr != _CUR_ANCHOR]
    else:
        _CUR_OTHER = np.array([], dtype=int)


def clear_opt_cache():
    global _CACHE_N, _CACHE_ANCHOR, _CACHE_T, _CACHE_Y, _CACHE_A, _CACHE_VAL
    _CACHE_N = -1
    _CACHE_ANCHOR = -1
    _CACHE_T = -1.0
    _CACHE_Y = None
    _CACHE_A = None
    _CACHE_VAL = None


def opt_from_y(y):
    n = _CUR_N
    if n <= 0:
        return None
    if n == 1:
        return np.array([1.0], dtype=np.float64)

    y = np.asarray(y, dtype=np.float64)
    if y.ndim != 1:
        y = y.reshape(-1)
    if y.size != n - 1:
        return None

    y = np.where(np.isfinite(y), y, 0.0)
    y = np.maximum(y, 0.0)

    S = 1.0 + float(y.sum())
    if S <= 1e-15 or not np.isfinite(S):
        return None

    a = np.empty(n, dtype=np.float64)
    a[_CUR_ANCHOR] = 1.0 / S
    if _CUR_OTHER.size > 0:
        a[_CUR_OTHER] = y / S

    s = float(a.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return None
    if abs(s - 1.0) > 1e-12:
        a = a / s
    return a


def stop_callback(x):
    return time.monotonic() > _CUR_DEADLINE


def opt_fun(y):
    global _CACHE_N, _CACHE_ANCHOR, _CACHE_T, _CACHE_Y, _CACHE_A, _CACHE_VAL

    if _CUR_N <= 1:
        return 0.0

    use_cache = False
    if (_CACHE_Y is not None and _CACHE_N == _CUR_N and
            _CACHE_ANCHOR == _CUR_ANCHOR and _CACHE_T == _CUR_T):
        if np.array_equal(_CACHE_Y, y):
            use_cache = True

    if use_cache and _CACHE_A is not None and _CACHE_VAL is not None:
        return float(_CACHE_VAL)

    a = opt_from_y(y)
    if a is None:
        return 1e100

    val = lse_value(a, _CUR_T)
    if not np.isfinite(val):
        return 1e100

    _CACHE_N = _CUR_N
    _CACHE_ANCHOR = _CUR_ANCHOR
    _CACHE_T = _CUR_T
    _CACHE_Y = np.array(y, dtype=np.float64)
    _CACHE_A = a
    _CACHE_VAL = float(val)
    return float(_CACHE_VAL)


def opt_grad(y):
    global _CACHE_N, _CACHE_ANCHOR, _CACHE_T, _CACHE_Y, _CACHE_A, _CACHE_VAL

    if _CUR_N <= 1:
        return np.zeros(0, dtype=np.float64)

    a = None
    if (_CACHE_Y is not None and _CACHE_N == _CUR_N and
            _CACHE_ANCHOR == _CUR_ANCHOR and _CACHE_T == _CUR_T):
        if np.array_equal(_CACHE_Y, y):
            a = _CACHE_A

    if a is None:
        a = opt_from_y(y)
        if a is None:
            return np.zeros(_CUR_N - 1, dtype=np.float64)
        _CACHE_N = _CUR_N
        _CACHE_ANCHOR = _CUR_ANCHOR
        _CACHE_T = _CUR_T
        _CACHE_Y = np.array(y, dtype=np.float64)
        _CACHE_A = a
        if _CACHE_VAL is None:
            _CACHE_VAL = lse_value(a, _CUR_T)

    G = lse_gradient(a, _CUR_T)
    if G.size != a.size:
        return np.zeros(_CUR_N - 1, dtype=np.float64)

    Sdot = float(np.dot(G, a))
    row = float(a[_CUR_ANCHOR]) * (G - Sdot * a)
    out = np.empty(_CUR_N - 1, dtype=np.float64)
    out[:] = row[_CUR_OTHER]
    return np.where(np.isfinite(out), out, 0.0)


# ---------------------------------------------------------------------------
# Exact subgradient polish.
# ---------------------------------------------------------------------------
def subgradient_polish(a0, time_budget, rng=None):
    start = time.monotonic()
    a = normalize_sequence(a0)
    best = a.copy()
    best_sc = local_score(a)
    if time_budget <= 0.0 or a.size <= 1:
        return best, best_sc

    n = a.size
    scale = 1.0
    step_count = 0

    while (time.monotonic() - start) < time_budget:
        b = fft_convolution(a)
        M = float(b.max())
        if M <= 1e-30:
            break

        Tpol = max(1e-5 * M, 1e-14)
        w = lse_weights(b, Tpol)
        G = fft_core_gradient(w, a)
        G = G - float(np.mean(G))

        if float(np.max(np.abs(G))) <= 1e-16:
            break

        accepted = False
        base_step = 0.1 * scale
        for fac in (1.0, 0.5, 0.25):
            trial = a - base_step * fac * G
            trial = np.maximum(trial, 0.0)
            s = float(trial.sum())
            if s <= 1e-15:
                continue
            trial = trial / s
            bt = fft_convolution(trial)
            sc = float(2.0 * n * bt.max())
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best = trial
                best_sc = sc
                a = trial
                accepted = True
                break

        if accepted:
            scale = min(1.0, scale * 1.05)
        else:
            scale *= 0.5
            if scale < 1e-5:
                break

        step_count += 1
        if step_count % 2500 == 0:
            print(f"    subgrad step={step_count} score={best_sc:.10f}")

    return best, float(best_sc)


# ---------------------------------------------------------------------------
# L-BFGS-B temperature-annealed optimizer.
# ---------------------------------------------------------------------------
def run_lbgfs(a0, time_budget, T_multipliers, max_iter_stage=300, rng=None, label=""):
    start = time.monotonic()
    a = normalize_sequence(a0)
    best = a.copy()
    best_sc = local_score(a)
    n = a.size

    if time_budget <= 1.0 or n <= 1:
        return best, float(best_sc)

    bmax_now = float(max(fft_convolution(best).max(), 0.0))
    scale = max(bmax_now, 1.0 / float(n), 1e-12)
    T_list = [max(float(m) * scale, 1e-14) for m in T_multipliers]

    reserve = min(90.0, 0.28 * time_budget)
    optimize_time = max(1.0, time_budget - reserve)
    prev_best = best_sc

    for idx, T in enumerate(T_list):
        elapsed = time.monotonic() - start
        rem = time_budget - elapsed
        if rem < 1.0:
            break

        remaining_stages = max(1, len(T_list) - idx)
        stage_budget = min(rem - 1.0, max(2.0, optimize_time / float(remaining_stages)))
        if stage_budget < 1.0:
            stage_budget = max(1.0, rem - 1.0)

        anchor = int(np.argmax(a))
        set_opt_globals(n, anchor)

        y0 = np.zeros(n - 1, dtype=np.float64)
        if _CUR_OTHER.size > 0:
            y0[:] = a[_CUR_OTHER] / max(float(a[anchor]), 1e-300)
        y0 = np.where(np.isfinite(y0), y0, 0.0)
        y0 = np.clip(y0, 0.0, 1000.0)

        _CUR_T = float(T)
        _CUR_DEADLINE = time.monotonic() + stage_budget
        clear_opt_cache()

        maxiter = int(min(float(max_iter_stage), max(20.0, stage_budget * 8.0)))
        bounds = [(0.0, 1000.0)] * (n - 1)

        try:
            res = minimize(
                opt_fun,
                y0,
                jac=opt_grad,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxiter,
                    "ftol": 1e-14,
                    "gtol": 1e-12,
                    "maxcor": 20,
                    "maxls": 20,
                    "disp": False,
                },
                callback=stop_callback,
            )
            x = res.x
            a_new = opt_from_y(x)
            if a_new is not None:
                a_new = normalize_sequence(a_new)
                sc_new = local_score(a_new)
                if np.isfinite(sc_new):
                    a = a_new
                    if sc_new < best_sc:
                        best = a_new
                        best_sc = sc_new
                else:
                    a = best.copy()
            else:
                a = best.copy()
        except Exception:
            a = best.copy()

        if idx < len(T_list) - 1 and rng is not None:
            if abs(best_sc - prev_best) < 1e-13:
                noise = rng.standard_normal(n) * (0.04 / float(n))
                a = normalize_sequence(a + noise)
                prev_best = best_sc
            else:
                prev_best = best_sc
        else:
            prev_best = best_sc

    elapsed = time.monotonic() - start
    rem = time_budget - elapsed
    if rem > 3.0:
        polish_time = min(rem, max(2.0, reserve))
        if polish_time > 1.0:
            a_pol, sc_pol = subgradient_polish(best, polish_time, rng)
            if np.isfinite(sc_pol) and sc_pol < best_sc:
                best = a_pol
                best_sc = sc_pol

    return best, float(best_sc)


# ---------------------------------------------------------------------------
# Candidate generation.
# ---------------------------------------------------------------------------
def make_starts(n, rng, count=12):
    x = np.linspace(0.0, 1.0, int(n))
    tau = 2.0 * np.pi
    starts = []

    for i in range(int(count)):
        kind = i % 8

        if kind == 0:
            amp1 = float(rng.uniform(0.5, 1.0))
            amp2 = float(rng.uniform(0.1, 0.4))
            amp3 = float(rng.uniform(0.0, 0.2))
            base = (
                1.0
                + amp1 * np.sin(tau * x)
                + amp2 * np.sin(2.0 * tau * x)
                + amp3 * np.sin(3.0 * tau * x)
            )

        elif kind == 1:
            q = float(rng.uniform(0.4, 2.0))
            env = np.sin(np.pi * x) ** q
            amp = float(rng.uniform(0.4, 1.0))
            base = env * (
                1.0
                + amp * np.sin(tau * x)
                + 0.25 * amp * np.sin(2.0 * tau * x)
            )

        elif kind == 2:
            r = float(rng.uniform(0.5, 3.0))
            base = np.exp(-r * x) + 0.3 * np.exp(-r * (1.0 - x))
            base = base * (1.0 + 0.3 * np.sin(tau * x))

        elif kind == 3:
            c1 = float(rng.uniform(0.05, 0.40))
            c2 = float(rng.uniform(0.60, 0.95))
            w = float(rng.uniform(1.0, 8.0))
            a2 = float(rng.uniform(0.2, 1.0))
            base = (
                np.exp(-w * (x - c1) ** 2)
                + a2 * np.exp(-w * (x - c2) ** 2)
                + 0.01
            )

        elif kind == 4:
            A = float(rng.uniform(2.0, 12.0))
            c = float(rng.uniform(0.15, 0.85))
            base = 1.0 / (1.0 + np.exp(-A * (x - c))) + 0.01

        elif kind == 5:
            slope = float(rng.uniform(0.5, 1.8))
            base = 1.0 + slope * (x - 0.5) + 0.2 * np.sin(tau * x)

        elif kind == 6:
            base = np.ones(n, dtype=np.float64)
            for k in range(1, 6):
                base = (
                    base
                    + float(rng.uniform(-0.25, 0.25)) * np.sin(tau * k * x)
                    + float(rng.uniform(-0.15, 0.15)) * np.cos(tau * k * x)
                )

        else:
            base = 1.0 + 0.8 * np.sin(tau * x) + 0.4 * rng.standard_normal(n)

        base = np.asarray(base, dtype=np.float64)
        if base.size != n:
            base = np.resize(base, n)
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.maximum(base, 1e-4)

        mean_b = float(base.mean())
        if mean_b > 0.0:
            base = base + rng.standard_normal(n) * (0.01 * mean_b)
        base = np.maximum(base, 1e-6)
        s = float(base.sum())
        if s > 1e-15:
            starts.append(base / s)
        else:
            starts.append(np.ones(n, dtype=np.float64) / float(n))

    return starts


def raw_de_score(x):
    a = np.asarray(x, dtype=np.float64)
    s = float(a.sum())
    if s <= 1e-8:
        return 1e6
    a = a / s
    return local_score(a)


def de_search(n, time_budget, rng):
    if differential_evolution is None or time_budget < 5.0:
        return None, float("inf")
    n = int(n)
    if n < 2:
        return None, float("inf")

    bounds = [(0.0, 1.0)] * n
    seed = int(rng.integers(0, 2 ** 31 - 1))
    maxiter = 45
    popsize = 12
    if n > 96:
        maxiter = 30
        popsize = 10

    try:
        res = differential_evolution(
            raw_de_score,
            bounds,
            maxiter=maxiter,
            popsize=popsize,
            tol=1e-5,
            seed=seed,
            polish=False,
        )
        if res.x is not None:
            a = normalize_sequence(res.x)
            sc = local_score(a)
            if np.isfinite(sc):
                return a, float(sc)
    except Exception:
        pass

    return None, float("inf")


# ---------------------------------------------------------------------------
# Candidate bookkeeping.
# ---------------------------------------------------------------------------
def add_candidate(cands, a, max_keep=40):
    a = normalize_sequence(a)
    if a.size <= 0:
        return
    sc = local_score(a)
    if not np.isfinite(sc):
        return
    cands.append((sc, a))
    if len(cands) > max_keep:
        cands.sort(key=itemgetter(0))
        del cands[max_keep:]


def mix_sequences(a, b, t):
    a = normalize_sequence(a)
    b = normalize_sequence(b)
    if b.size != a.size:
        b = interpolate_sequence(b, a.size)
    m = float(t) * a + (1.0 - float(t)) * b
    return normalize_sequence(m)


# ---------------------------------------------------------------------------
# Optional CVXPY active-set linearized trust-region polish.
# ---------------------------------------------------------------------------
def build_half_rows(a, active):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    active = np.asarray(active, dtype=int)
    m = active.size
    rows = np.zeros((m, n), dtype=np.float64)
    for i in range(m):
        k = int(active[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1, dtype=int)
            rows[i, js:je + 1] = a[k - idx]
    return rows


def active_linear_refine(a0, time_budget, label=""):
    if not _HAVE_CP or time_budget <= 0.0:
        a = normalize_sequence(a0)
        return a, local_score(a)

    start = time.monotonic()
    a = normalize_sequence(a0)
    best = a.copy()
    best_sc = local_score(a)
    n = a.size
    if n <= 1:
        return best, float(best_sc)

    solvers = []
    if _HAVE_CP:
        for name in ("CLARABEL", "ECOS", "SCS", "PDLP", "MOSEK"):
            attr = getattr(cp, name, None)
            if attr is not None:
                solvers.append(attr)
    if not solvers:
        return best, float(best_sc)

    rho_list = (0.05, 0.02, 0.10, 0.01)
    loops = 0

    while (time.monotonic() - start) < time_budget and loops < 8:
        rem = time_budget - (time.monotonic() - start)
        if rem < 10.0:
            break

        b = fft_convolution(a)
        M = float(b.max())
        if M <= 1e-30:
            break

        max_active = min(220, b.size)
        idx = np.argsort(b)[-max_active:]
        rows = build_half_rows(a, idx)
        vals = b[idx]

        improved = False

        for rho in rho_list:
            if improved:
                break
            try:
                d = cp.Variable(n)
                Mvar = cp.Variable(initial_value=float(M))
                constraints = [
                    cp.sum(d) == 0,
                    a + d >= 0,
                    cp.norm(d, 1) <= float(rho),
                    vals + 2.0 * (rows @ d) <= Mvar,
                ]
                prob = cp.Problem(cp.Minimize(Mvar), constraints)

                for sol in solvers:
                    if improved:
                        break
                    try:
                        prob.solve(solver=sol, verbose=False)
                        if prob.status in ("optimal", "optimal_inaccurate"):
                            dv = np.asarray(d.value, dtype=np.float64)
                            dv = np.where(np.isfinite(dv), dv, 0.0)
                            if float(np.max(np.abs(dv))) < 1e-14:
                                continue

                            for t in (1.0, 0.5, 0.25, 0.1):
                                trial = a + float(t) * dv
                                trial = np.maximum(trial, 0.0)
                                ss = float(trial.sum())
                                if ss <= 1e-15:
                                    continue
                                trial = trial / ss
                                sc = local_score(trial)
                                if np.isfinite(sc) and sc < best_sc - 1e-12:
                                    best = trial
                                    best_sc = sc
                                    a = trial
                                    improved = True
                                    break
                    except Exception:
                        continue
            except Exception:
                continue

        if not improved:
            break

        loops += 1
        if loops % 2 == 0:
            print(f"    active LP polish {loops}: score={best_sc:.10f}")

    return best, float(best_sc)


# ---------------------------------------------------------------------------
# Main entry point.
# ---------------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1800.0):
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)
    cands = []

    # Load parent construction if available.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1:
            p = p.reshape(-1)
        if p.size > 0 and np.isfinite(p).all() and float(np.sum(p)) > 1e-12:
            parent = p
    except Exception:
        parent = None

    if parent is not None:
        if parent.size > 4000:
            p = interpolate_sequence(parent, 2400)
        else:
            p = normalize_sequence(parent)
        sc = local_score(p)
        if np.isfinite(sc):
            add_candidate(cands, p)
            print(f"Parent loaded: n={p.size}, score={sc:.10f}")
    else:
        for a0 in make_starts(1200, rng, 4):
            add_candidate(cands, a0)

    # Ensure at least one candidate.
    if not cands:
        u = np.ones(1200, dtype=np.float64) / 1200.0
        cands.append((local_score(u), u))
    else:
        cands.sort(key=itemgetter(0))
        print(f"Initial best: {cands[0][0]:.10f} (n={cands[0][1].size})")

    # ------------------------------------------------------------------
    # Phase 1: quick refinement of current best (usually parent).
    # ------------------------------------------------------------------
    cands.sort(key=itemgetter(0))
    best_seq = cands[0][1]
    rem = deadline - time.monotonic()
    t1 = min(130.0, rem * 0.18)
    if t1 > 25.0:
        print(f"Phase 1: refine initial best, budget={t1:.0f}s")
        a1, sc1 = run_lbgfs(
            best_seq,
            t1,
            [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002],
            300,
            rng,
            "init",
        )
        add_candidate(cands, a1)
        print(f"Phase 1 best: {sc1:.10f}")

    # ------------------------------------------------------------------
    # Phase 2: low-resolution multi-start and small global search.
    # ------------------------------------------------------------------
    rem = deadline - time.monotonic()
    t2 = min(260.0, rem * 0.26)
    if t2 > 40.0:
        print(f"Phase 2: low-res search, budget={t2:.0f}s")
        low_n = 128
        n_starts = 10
        starts = make_starts(low_n, rng, n_starts)
        per_start = max(4.0, t2 / float(n_starts + 2))
        for idx, a0 in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 10.0:
                break
            bb = min(per_start, rem * 0.35)
            a_l, sc_l = run_lbgfs(
                a0,
                bb,
                [0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                160,
                rng,
                f"low{idx}",
            )
            add_candidate(cands, a_l)

        rem = deadline - time.monotonic()
        if rem > 30.0:
            t_de = min(60.0, rem * 0.18)
            a_de, sc_de = de_search(96, t_de, rng)
            if a_de is not None:
                add_candidate(cands, a_de)
                print(f"DE candidate: {sc_de:.10f} (n={a_de.size})")

    # ------------------------------------------------------------------
    # Phase 3: interpolate promising low-res candidates to mid/high n.
    # ------------------------------------------------------------------
    cands.sort(key=itemgetter(0))
    top = cands[:min(3, len(cands))]

    jobs = []
    for sc, a_src in top:
        if a_src.size < 1500:
            jobs.append((a_src, 1600))
            jobs.append((a_src, 2400))
            jobs.append((a_src, 3200))
        elif a_src.size < 2400:
            jobs.append((a_src, 2400))
            jobs.append((a_src, 3200))
        elif a_src.size < 3200:
            jobs.append((a_src, 3200))

    # Also ensure parent-like high-n jobs if parent exists.
    if parent is not None:
        psrc = interpolate_sequence(parent, 1600)
        jobs.append((psrc, 2400))
        jobs.append((psrc, 3200))

    # Limit jobs.
    jobs = jobs[:6]

    rem = deadline - time.monotonic()
    t3 = rem * 0.42
    if t3 > 40.0 and jobs:
        print(f"Phase 3: high-res jobs={len(jobs)}, budget={t3:.0f}s")
        per_job = max(20.0, t3 / float(max(1, len(jobs))))
        for jidx, (a_src, nn) in enumerate(jobs):
            rem = deadline - time.monotonic()
            if rem < 30.0:
                break
            bb = min(per_job, rem * 0.40)
            a_init = interpolate_sequence(a_src, int(nn))
            a_h, sc_h = run_lbgfs(
                a_init,
                bb,
                [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.0007],
                360,
                rng,
                f"n{nn}_{jidx}",
            )
            add_candidate(cands, a_h)
            print(f"  high-res job {jidx} n={nn}: {sc_h:.10f}")

    # ------------------------------------------------------------------
    # Phase 4: optional active-set LP polish of the best candidate.
    # ------------------------------------------------------------------
    cands.sort(key=itemgetter(0))
    if cands:
        best_seq = cands[0][1]
    rem = deadline - time.monotonic()
    if rem > 120.0 and _HAVE_CP and best_seq.size <= 3500:
        t4 = min(90.0, rem * 0.18)
        print(f"Phase 4: active LP polish, budget={t4:.0f}s")
        a_act, sc_act = active_linear_refine(best_seq, t4, "act")
        add_candidate(cands, a_act)
        print(f"Active LP polish: {sc_act:.10f}")

    # ------------------------------------------------------------------
    # Phase 5: final intensive refinement.
    # ------------------------------------------------------------------
    cands.sort(key=itemgetter(0))
    if not cands:
        best_seq = np.ones(1200, dtype=np.float64)
    else:
        best_seq = cands[0][1]

    rem = deadline - time.monotonic()
    if rem > 30.0:
        if best_seq.size < 2400:
            final_n = 2400
        elif best_seq.size > 3500:
            final_n = 3200
        else:
            final_n = best_seq.size

        a_fin = interpolate_sequence(best_seq, int(final_n)) if int(final_n) != best_seq.size else best_seq.copy()
        t5 = min(rem * 0.68, 320.0)
        if t5 > 20.0:
            print(f"Phase 5: final refine n={final_n}, budget={t5:.0f}s")
            a_f, sc_f = run_lbgfs(
                a_fin,
                t5,
                [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004],
                700,
                rng,
                "final",
            )
            add_candidate(cands, a_f)
            print(f"Final L-BFGS: {sc_f:.10f}")

    # ------------------------------------------------------------------
    # Final selection.
    # ------------------------------------------------------------------
    cands.sort(key=itemgetter(0))
    if cands:
        best_seq = cands[0][1]

    best_seq = normalize_sequence(best_seq)
    if best_seq.size > 8000:
        best_seq = interpolate_sequence(best_seq, 3200)

    # A tiny final subgradient on the exact returned sequence, if time remains.
    fin_t = deadline - time.monotonic()
    if fin_t > 20.0 and best_seq.size <= 3500:
        a_last, sc_last = subgradient_polish(best_seq, min(15.0, fin_t), rng)
        if np.isfinite(sc_last) and sc_last < local_score(best_seq):
            best_seq = a_last

    best_seq = normalize_sequence(best_seq)
    out = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return out
```
