"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Ultra-fine LogSumExp L-BFGS (T: 20 -> 0.001) with FFT-accelerated gradients via cross-correlation
2. Nesterov-accelerated LP alternating with efficient solvers

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.signal import correlate


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally (fast, no security checks)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector (sum=1)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_fast(z, n, T):
    """
    LogSumExp objective and gradient w.r.t. z using FFT-based cross-correlation.
    Returns (obj, grad_z).
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    s_exp = exp_terms.sum()
    lse = max_b + T * np.log(s_exp)
    obj = 2.0 * n * lse

    w = exp_terms / s_exp  # length 2n-1
    # grad_a[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m] = 2 * correlate(w, a, 'valid')[i]
    grad_a = 2.0 * correlate(w, a, mode='valid')

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def make_lse_func(n, T, deadline):
    """Create objective closure-free factory."""
    def func(z):
        if time.monotonic() > deadline:
            return 1e15
        obj, _ = lse_obj_grad_fast(z, n, T)
        return obj
    return func


def make_lse_jac(n, T, deadline):
    """Create gradient closure-free factory."""
    def jac(z):
        if time.monotonic() > deadline:
            return np.zeros_like(z)
        _, g = lse_obj_grad_fast(z, n, T)
        return g
    return jac


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=200):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(5, maxiter_per_stage // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        func = make_lse_func(n, stage_T, deadline)
        jac = make_lse_jac(n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': iters_per_stage,
                    'ftol': 1e-25,
                    'gtol': 1e-18,
                    'maxfun': iters_per_stage * 200,
                    'disp': False
                }
            )
            z = res.x
        except Exception:
            pass

        # Normalize
        a_cur = softmax_from_z(z)
        elapsed = time.monotonic() - t0
        if (stage_i + 1) % 3 == 0 or stage_T < 0.1:
            sc = evaluate_local(a_cur)
            print(f"    LSE T={stage_T:.4f} ({elapsed:.0f}s): score={sc:.8f}")

    return softmax_from_z(z)


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                b_val = b_val / sv
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp(a0, time_limit, solver, n_max_iters):
    """Nesterov dual averaging LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def multiplicative_weights(a0, time_limit, n_steps=50000):
    """Mirror descent (multiplicative weights) for exact max objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.02
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])

        # Subgradient: average over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            eta *= 0.85
            no_improve = 0
            if eta < 0.001:
                break

    return best_a, best_score


def coordinate_descent_log(a0, time_limit, n_sweeps=10):
    """Coordinate descent in log-domain with golden section search."""
    n = len(a0)
    z = z_from_a(a0).copy()
    a = softmax_from_z(z)
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    inv_golden = 1.0 / golden

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Golden section search on z[i]
            # Objective: evaluate_local(softmax_from_z(z))
            lo = z[i] - 3.0
            hi = z[i] + 3.0
            c = hi - inv_golden * (hi - lo)
            d = lo + inv_golden * (hi - lo)

            z[i] = c
            a_c = softmax_from_z(z)
            fc = evaluate_local(a_c)
            z[i] = d
            a_d = softmax_from_z(z)
            fd = evaluate_local(a_d)

            for _ in range(20):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - inv_golden * (hi - lo)
                    z[i] = c
                    a_c = softmax_from_z(z)
                    fc = evaluate_local(a_c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + inv_golden * (hi - lo)
                    z[i] = d
                    a_d = softmax_from_z(z)
                    fd = evaluate_local(a_d)

            # Keep the better
            if fc < fd:
                z[i] = c
            else:
                z[i] = d

            # Update best
            a_cur = softmax_from_z(z)
            sc = evaluate_local(a_cur)
            if sc < best_score:
                best_score = sc

        a = softmax_from_z(z)
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
        if (sweep + 1) % 2 == 0:
            print(f"    CoordDescent sweep={sweep+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return softmax_from_z(z), best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Flat spectrum approximation: use raised cosine with slight tilt
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 0.7 + 0.02
    else:
        # Triangular with bump
        a = 1.0 - np.abs(2 * x - 1)
        a = a ** 2 + 0.05 * np.exp(-20 * (x - 0.5) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def freq_perturb(a, rng, strength=0.03):
    """Perturb in frequency domain for structured exploration."""
    n = len(a)
    a_copy = a.copy()
    # Compute real FFT
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a_copy, fft_size)
    # Perturb magnitudes (keep DC and Nyquist fixed)
    for k in range(1, len(A) - 1):
        A[k] *= (1.0 + strength * rng.standard_normal())
    # Inverse
    a_new = irfft(A, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a_copy
    return a_new / s

        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = nesterov_lp(init_1500, t_n1500 * 0.5, solver, n_max_iters=200000)
        _local_update(a_1500, "lp_n1500")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1500 and rem > 40:
            T_1500 = [3.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005, 0.002, 0.001]
            a_1500b = lbgfs_refine(best_seq, min(rem, 120.0), T_1500, maxiter_per_stage=3000)
            _local_update(a_1500b, "lbgfs_n1500")

    # === Phase 7: Try n=1000 (shorter might be easier to optimize) ===
    print(f"\n=== Phase 7: Try n=1000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n1000 = min(rem * 0.10, 120.0)
        print(f"  Budget: {t_n1000:.0f}s")
        init_1000 = interpolate_sequence(best_seq, 1000)
        a_1000, _ = nesterov_lp(init_1000, t_n1000 * 0.6, solver, n_max_iters=100000)
        _local_update(a_1000, "lp_n1000")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1000 and rem > 30:
            T_1000 = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005, 0.001]
            a_1000b = lbgfs_refine(best_seq, min(rem, 80.0), T_1000, maxiter_per_stage=2500)
            _local_update(a_1000b, "lbgfs_n1000")

    # === Phase 8: Coordinate Descent Polish ===
    print(f"\n=== Phase 8: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.15, 100.0)
        n_cd = len(best_seq)
        n_sweeps = max(2, int(t_cd / max(1, n_cd * 0.015)))
        print(f"  n={n_cd}, ~{n_sweeps} sweeps, budget={t_cd:.0f}s")
        a_cd, _ = coordinate_descent_log(best_seq, t_cd, n_sweeps=n_sweeps)
        _local_update(a_cd, "coord_desc")

    # === Phase 9: Frequency-domain perturbation restarts ===
    print(f"\n=== Phase 9: Freq Perturb (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            strength = 0.02 + 0.01 * trial
            a_pert = freq_perturb(best_seq, rng, strength=strength)
            # Quick LSE polish
            T_quick = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            t_sub = min(rem_sub * 0.25, 50.0)
            a_opt = lbgfs_refine(a_pert, t_sub, T_quick, maxiter_per_stage=1500)
            _local_update(a_opt, f"freq_pert{trial}")

    # === Phase 10: Final ultra-fine LSE ===
    print(f"\n=== Phase 10: Final Ultra-Fine LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.4, 60.0)
        T_final = [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04,
                   0.03, 0.025, 0.02, 0.015, 0.012, 0.01, 0.008, 0.006,
                   0.005, 0.004, 0.003, 0.0025, 0.002, 0.0015, 0.001]
        a_fin = lbgfs_refine(best_seq, t_fin, T_final, maxiter_per_stage=3000)
        _local_update(a_fin, "final_ultra_lse")

    # === Phase 11: Final multiplicative weights ===
    rem = deadline - time.monotonic()
    if rem > 15:
        a_fw, _ = multiplicative_weights(best_seq, min(rem * 0.5, 25.0), n_steps=30000)
        _local_update(a_fw, "final_mw")

    # === Phase 12: Last LP shot ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_fl, _ = nesterov_lp(best_seq, min(rem * 0.5, 15.0), solver, n_max_iters=50000)
        _local_update(a_fl, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Verify with the actual evaluate_sequence
        try:
            test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
            actual = evaluate_sequence(test_seq)
            print(f"\nFINAL: local={best_score:.8f}, actual={actual:.8f}, n={len(best_seq)}")
        except Exception as e:
            print(f"\nFINAL: local={best_score:.8f}, actual eval error: {e}, n={len(best_seq)}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
