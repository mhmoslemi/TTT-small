def build_T_for_a(a):
    """Build matrix T (2n-1 x n) such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        # a[j] * b[k-j] contributes to position k = j + j', where j' is index in b
        # So T[j + j', j] = a[j] for j' = 0, ..., n-1
        T[j:j + n, j] = a[j]
    return T


def get_lp_solver():

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 7)
                noise = 0.04 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={np.mean(lp_times[-50:])*1000:.1f}ms")

    return best_a, best_score


def peak_shaving_local_search(a0, time_limit, n_steps=50000, verbose_interval=10000):
    """Direct active-set peak shaving via projected subgradient on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.02
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])

        # Compute subgradient: g[i] = (2/|S|) * sum_{k in S} a[k-i] (valid)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g *= 2.0 / len(active)

        # Multiplicative update (mirror descent)
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

        if step % verbose_interval == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f} scale={scale:.4f} |S|={len(active)}")

    return best_a, best_score


def swap_local_search(a0, time_limit, n_trials=200000, verbose_interval=50000):
    """Structured swap local search: move mass from peak-contributing to non-peak coords."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Precompute: for each coordinate i, which lags does it contribute to?
    # i contributes to lag k if 0 <= k-i < n, i.e., k in [i, min(i+n-1, 2n-2)]
    # For speedup, we'll work with the top peaks

    delta_frac = 0.01  # fraction of a[i] to move
    T_anneal = 0.1
    T_min = 0.001

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks (within small epsilon of max)
        threshold = max_b * (1.0 - 5e-5)
        active_peaks = np.where(b >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([n - 1])

        # Pick a source coordinate that contributes to a peak
        k_src = active_peaks[rng.integers(len(active_peaks))]
        # Source coords: i such that 0 <= k_src - i < n, i.e., i in [max(0,k_src-n+1), min(n-1,k_src)]
        js = max(0, k_src - n + 1)
        je = min(n - 1, k_src)
        src_candidates = np.arange(js, je + 1)
        i_src = src_candidates[rng.integers(len(src_candidates))]

        if a[i_src] < 1e-8:
            continue

        # Pick a destination coordinate that contributes LESS to peaks
        # Dest should NOT contribute to any active peak, or contribute to a non-peak lag
        # i_dst contributes to lag k if i_dst in [max(0,k-n+1), min(n-1,k)]
        # We want i_dst NOT in any active peak's range
        # Simpler: just pick a random coordinate and check
        i_dst = rng.integers(n)
        if i_dst == i_src:
            continue
        if a[i_dst] > 0.5 * s / n:  # don't move too much mass
            continue

        delta = delta_frac * a[i_src]
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        if a_new[i_src] < 0:
            continue

        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        score_new = 2.0 * n * max_new

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Simulated annealing acceptance
            T_cur = T_anneal * (T_min / T_anneal) ** (trial / max(1, n_trials))
            ratio = (score_new - best_score) / max(1e-10, best_score)
            if ratio < 0 or rng.random() < np.exp(-ratio / T_cur):
                a = a_new

        if trial % verbose_interval == 0 and trial > 0:
            elapsed = time.monotonic() - t0
            print(f"    SwapLS {elapsed:.0f}s: score={best_score:.8f} T={T_cur:.6f}")

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 1:
        a = np.sin(np.pi * x) ** 2
    elif mode == 2:
        a = np.exp(-4 * x * (1 - x))
    elif mode == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif mode == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif mode == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif mode == 6:
        # Chirp-like
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (3 * x + x**2))
        a = np.maximum(a, 0.01)
    elif mode == 7:
        # Two-sided decay
        center = n // 2
        idx = np.abs(np.arange(n) - center)
        a = np.exp(-2.0 * idx / n)
    else:
        a = np.ones(n) * 0.5 + 0.2 * rng.random(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def spectral_perturbation(a, rng, strength=0.03):
    """Perturb in frequency domain to preserve global shape."""
    n = len(a)
    a_copy = a.copy()
    # Go to frequency domain
    A = np.fft.rfft(a_copy)
    # Perturb phases of non-zero frequencies
    for i in range(1, len(A)):
        if rng.random() < 0.3:
            phase = rng.uniform(-strength, strength)
            A[i] *= np.exp(1j * phase)
    a_pert = np.fft.irfft(A, n=n)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(15, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    Uses Nesterov LP, peak-shaving, swap local search, L-BFGS, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Nesterov LP on parent (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=2000000, verbose_interval=500)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 2: Peak shaving local search ===
    print(f"\n=== Phase 2: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_ps, _ = peak_shaving_local_search(best_seq, t_alloc, n_steps=100000)
        local_update(a_ps, "peak_shave_1")

    # === Phase 3: Nesterov LP round 2 (from updated best) ===
    print(f"\n=== Phase 3: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000, verbose_interval=300)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 4: L-BFGS with aggressive cooling ===
    print(f"\n=== Phase 4: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
        print(f"  {len(T_stages)} T-stages, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages, maxiter_per_stage=10000)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 5: Swap local search ===
    print(f"\n=== Phase 5: Swap Local Search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_sw, _ = swap_local_search(best_seq, t_alloc, n_trials=100000)
        local_update(a_sw, "swap_ls")

    # === Phase 6: Peak shaving round 2 ===
    print(f"\n=== Phase 6: Peak Shaving Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 8:
        a_ps2, _ = peak_shaving_local_search(best_seq, t_alloc, n_steps=50000)
        local_update(a_ps2, "peak_shave_2")

    # === Phase 7: Try n=1800 ===
    print(f"\n=== Phase 7: Try n=1800 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80 and len(best_seq) != 1800:
        t_n1800 = min(rem * 0.12, 120.0)
        print(f"  Budget: {t_n1800:.0f}s")
        init_1800 = interpolate_sequence(best_seq, 1800)
        a_1800, _ = nesterov_lp_refine(init_1800, t_n1800 * 0.6, solver, n_max_iters=200000, verbose_interval=200)
        local_update(a_1800, "lp_n1800")
        # Peak shave on n=1800
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1800 and rem > 15:
            a_1800b, _ = peak_shaving_local_search(best_seq, min(rem, 30.0), n_steps=30000)
            local_update(a_1800b, "peak_n1800")

    # === Phase 8: Try n=2500 ===
    print(f"\n=== Phase 8: Try n=2500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and len(best_seq) != 2500:
        t_n2500 = min(rem * 0.10, 100.0)
        print(f"  Budget: {t_n2500:.0f}s")
        init_2500 = interpolate_sequence(best_seq, 2500)
        a_2500, _ = nesterov_lp_refine(init_2500, t_n2500 * 0.6, solver, n_max_iters=100000, verbose_interval=200)
        local_update(a_2500, "lp_n2500")

    # === Phase 9: Spectral perturbation + re-optimize ===
    print(f"\n=== Phase 9: Spectral Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            a_pert = spectral_perturbation(best_seq, rng, strength=0.02 + 0.01 * trial)
            t_sub = min(rem_sub * 0.25, 40.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=50000, verbose_interval=100)
            local_update(a_opt, f"spectral_pert{trial}")
            # Also try peak shaving on perturbed
            rem_sub = deadline - time.monotonic()
            if rem_sub > 10:
                a_opt2, _ = peak_shaving_local_search(a_opt, min(rem_sub, 15.0), n_steps=20000)
                local_update(a_opt2, f"spectral_pert{trial}_ps")

    # === Phase 10: Final polish ===
    print(f"\n=== Phase 10: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.4, 30.0)
        if t_fin > 8:
            T_stages3 = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages3, maxiter_per_stage=3000)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin_ps, _ = peak_shaving_local_search(best_seq, min(rem * 0.5, 15.0), n_steps=30000)
        local_update(a_fin_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 8:
        a_fin_lp, _ = nesterov_lp_refine(best_seq, min(rem * 0.6, 15.0), solver, n_max_iters=20000, verbose_interval=50)
        local_update(a_fin_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
