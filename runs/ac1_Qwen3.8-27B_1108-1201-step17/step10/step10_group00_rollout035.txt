Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast best response via multiplicative weights (O(n log n) per step, no LP matrix)
2. Nesterov-accelerated alternating method (10x+ more iterations than LP-based)
3. Average strategy for better MW convergence
4. Coordinate descent polishing
5. LogSumExp L-BFGS fine-tuning with aggressive temperature annealing
6. Multi-length exploration (n=800, 1200, 1500, 2000)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_conv(a, b):
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]

def cross_corr_with_indicator(a, indicator):
    """v_j = sum_k a[k-j] * indicator[k] for j=0..len(a)-1"""
    n = len(a)
    rev_a = a[::-1]
    fft_size = 1 << (len(rev_a) + len(indicator) - 2).bit_length()
    R = rfft(rev_a, fft_size)
    I = rfft(indicator, fft_size)
    conv = irfft(R * I, fft_size)
    return conv[n-1 : n-1+n]

def fast_best_response_mw(a_fixed, eta=1.0, n_iter=3000, eps=1e-6, use_average=True):
    """Approximate best response using multiplicative weights on the dual game."""
    n = len(a_fixed)
    b = np.ones(n) / n
    b_avg = np.zeros(n)
    
    for t in range(n_iter):
        c = fast_cross_conv(a_fixed, b)
        max_c = c.max()
        if max_c < 1e-15:
            break
        active = (c >= max_c * (1.0 - eps)).astype(np.float64)
        n_active = active.sum()
        if n_active < 1:
            active = np.zeros(2*n-1)
            active[int(c.argmax())] = 1.0
            n_active = 1.0
        
        v = cross_corr_with_indicator(a_fixed, active) / n_active
        
        b *= np.exp(-eta * v)
        b = np.maximum(b, 1e-30)
        s = b.sum()
        b /= s
        b_avg += b
    
    if use_average:
        result = b_avg / n_iter
    else:
        result = b
    
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        result = np.ones(n) / n
    else:
        result /= s
    return result

def fast_best_response_subgrad(a_fixed, eta=0.1, n_iter=5000, eps=1e-4):
    """Best response via projected subgradient on the simplex."""
    n = len(a_fixed)
    b = np.ones(n) / n
    
    for t in range(n_iter):
        c = fast_cross_conv(a_fixed, b)
        max_c = c.max()
        if max_c < 1e-15:
            break
        # Active set
        threshold = max_c * (1.0 - eps)
        active = c >= threshold
        n_active = active.sum()
        if n_active < 1:
            active = np.zeros(2*n-1, dtype=bool)
            active[int(c.argmax())] = True
            n_active = 1
        
        v = cross_corr_with_indicator(a_fixed, active.astype(np.float64)) / n_active
        
        # Subgradient step
        b = b - eta * v
        # Project onto simplex
        # b_i = max(b_i - theta, 0), choose theta so sum = 1
        # Sort b in descending order
        b_sorted = np.sort(b)[::-1]
        css = np.cumsum(b_sorted)
        rho = np.nonzero(b_sorted * np.arange(1, n+1) > (css - 1))[0][-1]
        theta = (css[rho] - 1) / (rho + 1)
        b = np.maximum(b - theta, 0)
        s = b.sum()
        if s < 1e-15:
            b = np.ones(n) / n
        else:
            b /= s
    
    return b

    n = len(a0)

import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]


def cross_corr_indicator(a, indicator):
    """v_j = sum_k a[k-j] * indicator[k] for j=0..n-1 via FFT."""
    n = len(a)
    rev_a = a[::-1]
    total_len = n + len(indicator) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    R = rfft(rev_a, fft_size)
    I = rfft(indicator, fft_size)
    conv = irfft(R * I, fft_size)
    return conv[n - 1: n - 1 + n]


def fast_best_response(a_fixed, eta=1.0, n_iter=2000, eps=1e-6, use_average=True):
    """Approximate best response: find b>=0, sum(b)=1 minimizing max(conv(a_fixed, b)).
    Uses multiplicative weights on the dual zero-sum game with FFT acceleration."""
    n = len(a_fixed)
    b = np.ones(n) / n
    b_avg = np.zeros(n)

    for t in range(n_iter):
        c = fast_cross_conv(a_fixed, b)
        max_c = c.max()
        if max_c < 1e-15:
            break
        active = (c >= max_c * (1.0 - eps)).astype(np.float64)
        n_active = active.sum()
        if n_active < 1:
            active = np.zeros(2 * n - 1)
            active[int(c.argmax())] = 1.0
            n_active = 1.0

        v = cross_corr_indicator(a_fixed, active) / n_active

        log_b = -eta * v
        log_b -= log_b.max()
        b = np.exp(log_b)
        b = np.maximum(b, 1e-30)
        b /= b.sum()
        b_avg += b

    if use_average:
        result = b_avg / n_iter
    else:
        result = b

    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        result = np.ones(n) / n
    else:
        result /= s
    return result


def fast_best_response_subgrad(a_fixed, eta=0.05, n_iter=3000, eps=1e-4):
    """Best response via projected subgradient on the simplex (alternative to MW)."""
    n = len(a_fixed)
    b = np.ones(n) / n
    b_avg = np.zeros(n)

    for t in range(n_iter):
        c = fast_cross_conv(a_fixed, b)
        max_c = c.max()
        if max_c < 1e-15:
            break
        threshold = max_c * (1.0 - eps)
        active = (c >= threshold).astype(np.float64)
        n_active = active.sum()
        if n_active < 1:
            active = np.zeros(2 * n - 1)
            active[int(c.argmax())] = 1.0
            n_active = 1.0

        v = cross_corr_indicator(a_fixed, active) / n_active

        # Adaptive step size
        eta_t = eta / (1.0 + 0.01 * t)
        b = b - eta_t * v
        # Project onto simplex {x >= 0, sum(x) = 1}
        b_sorted = np.sort(b)[::-1]
        css = np.cumsum(b_sorted)
        rho = np.nonzero(b_sorted * np.arange(1, n + 1) > (css - 1.0))[0]

            s = b.sum()
            if s > 1e-15:

            else:

    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        result = np.ones(n) / n
    else:
        result /= s

def logsumexp_obj_grad(a, T):
    """Compute LogSumExp objective and gradient for a (sum=1)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    exp_terms = np.exp(shifted)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad = np.zeros(n)
    # grad_j = (1/T) * sum_k w_k * 2 * a[k-j] (for valid k-j)
    # Use FFT: grad = (2/T) * cross_corr(w, a) ... let me compute directly
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    grad /= T
    return obj, grad


def logsumexp_z_obj_grad(z, n, T):
    """LogSumExp objective in log-domain z (a = softmax(z))."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    obj_val, grad_a = logsumexp_obj_grad(a, T)
    # Chain rule: grad_z = a * (grad_a - a^T * grad_a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def lse_lbgfs_refine(a0, time_limit, T_stages=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing in log-domain."""
    n = len(a0)
    if T_stages is None:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    bfgs_iter_total = 3000

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        iters = max(5, bfgs_iter_total // n_stages)

        def func(z_x):
            if time.monotonic() > deadline:
                return 1e15
            return logsumexp_z_obj_grad(z_x, n, stage_T)[0]

        def jac(z_x):
            if time.monotonic() > deadline:
                return np.zeros_like(z_x)
            return logsumexp_z_obj_grad(z_x, n, stage_T)[1]

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-20,
                                    'gtol': 1e-14, 'maxfun': iters * 200, 'disp': False})
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        # Randomize coordinate order
        order = rng.permutation(n)

        for j in order:
            f = a[j]
            # Compute base convolution (without contribution of a[j])
            a[j] = 0.0
            base = fast_autocorr(a)
            # Restore
            a[j] = f

            # For new value f_new:
            # conv[k] = base[k] + 2*f_new*a[k-j] (k != 2j, k-j in [0,n-1))
            # conv[2j] = base[2j] + f_new^2
            # We want to minimize max(conv)

            # Candidate values to try
            candidates = [0.0, f]

            # Find the peak position when f=0
            max_base = base.max()
            peak_k = int(base.argmax())

            # Try to reduce the peak: if peak involves a[j], reduce f
            # Analytical: minimize max(base + 2*f*a[k-j] for k, base[2j] + f^2)
            # Try value that equalizes: base[peak_k] + 2*f*a[peak_k-j] = base[2j] + f^2
            if 0 <= peak_k - j < n and peak_k != 2 * j:
                ak = a[peak_k - j]
                if ak > 1e-15:
                    # base[peak_k] + 2*f*ak = base[2j] + f^2
                    # f^2 - 2*ak*f + (base[2j] - base[peak_k]) = 0
                    disc = 4 * ak * ak - 4 * (base[2j] - base[peak_k])
                    if disc >= 0:
                        sq_disc = np.sqrt(disc)
                        f1 = (2 * ak + sq_disc) / 2.0
                        f2 = (2 * ak - sq_disc) / 2.0
                        if f1 > 0:
                            candidates.append(f1)
                        if f2 > 0:
                            candidates.append(f2)

            # Try the value that makes the linear peak = quadratic peak
            if 2 * j < len(base):
                # Find max of (base[k] - base[2j]*0 + 2*f*a[k-j]) over k
                # Simplified: just try f that would halve the peak
                candidates.append(f * 0.5)
                candidates.append(f * 1.5)
                candidates.append(f * 0.1)

            best_f = f
            best_val = best_score

            for f_cand in candidates:
                f_cand = max(0.0, f_cand)
                # Quick evaluation
                a[j] = f_cand
                new_conv = fast_autocorr(a)
                a[j] = f  # restore temporarily
                new_max = new_conv.max()
                # Scale: score = 2n * max_conv / (sum)^2
                new_sum = s  # unchanged since we're working with sum=1... 
                # Actually sum changes! s - f + f_cand
                new_s = s - f + f_cand
                if new_s < 0.01:
                    continue
                new_score = 2.0 * n * new_max / (new_s * new_s)
                if new_score < best_val:
                    best_val = new_score
                    best_f = f_cand

            if best_f != f:
                a[j] = best_f
                new_s = a.sum()
                if new_s > 1e-15:
                    a = a / new_s
                    s = 1.0
                sc = evaluate_local(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if sweep % 5 == 0 and sweep > 0:
            if time.monotonic() - t0 > time_limit:
                break

    return best_a, best_score


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Dolph-Chebyshev-like
        a = np.abs(np.cos(np.pi * (2 * x - 1))) ** 2
    else:
        # Random smooth
        k = rng.integers(3, 10)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast MW best-response (replacing LP), Nesterov acceleration,
    coordinate descent, and L-BFGS polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Main Nesterov alternating with fast MW ===
    print(f"\n=== Phase 1: Nesterov Alternating n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.55, 900.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, _ = nesterov_alternating(best_seq, t_alloc, eta_mw=1.0, mw_iters=1500, method='mw')
        local_update(a_opt, "nesterov_mw_1")

    # === Phase 2: Second round with different MW parameters ===
    print(f"\n=== Phase 2: Nesterov Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt2, _ = nesterov_alternating(best_seq, t_alloc, eta_mw=0.5, mw_iters=3000, method='mw')
        local_update(a_opt2, "nesterov_mw_2")

    # === Phase 3: Subgradient best response variant ===
    print(f"\n=== Phase 3: SubGrad Variant (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt3, _ = nesterov_alternating(best_seq, t_alloc, eta_mw=0.5, mw_iters=2000, method='subgrad')
        local_update(a_opt3, "nesterov_subgrad")

    # === Phase 4: Coordinate Descent Polish ===
    print(f"\n=== Phase 4: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_polish(best_seq, n_sweeps=50, time_limit=t_alloc)
        local_update(a_cd, "coord_descent")

    # === Phase 5: LogSumExp L-BFGS Fine-tuning ===
    print(f"\n=== Phase 5: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lb = lse_lbgfs_refine(best_seq, t_alloc, T_stages)
        local_update(a_lb, "lbgfs_lse")

    # === Phase 6: Try n=1500 ===
    print(f"\n=== Phase 6: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_n = min(rem * 0.25, 180.0)
        print(f"  Budget: {t_n:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = nesterov_alternating(init_1500, t_n * 0.7, eta_mw=1.0, mw_iters=1000, method='mw')
        local_update(a_1500, "n1500_mw")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1500 and rem > 30:
            a_1500b = lse_lbgfs_refine(best_seq, min(rem, 60.0),
                                        [1.0, 0.5, 0.2, 0.1, 0.05, 0.02])
            local_update(a_1500b, "n1500_lbgfs")

    # === Phase 7: Try n=800 (shorter, easier to optimize) ===
    print(f"\n=== Phase 7: Try n=800 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_n800 = min(rem * 0.20, 100.0)
        print(f"  Budget: {t_n800:.0f}s")
        init_800 = interpolate_sequence(best_seq, 800)
        a_800, _ = nesterov_alternating(init_800, t_n800 * 0.7, eta_mw=1.0, mw_iters=1000, method='mw')
        local_update(a_800, "n800_mw")

    # === Phase 8: Final polish on best ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 40.0)
        a_fin, _ = nesterov_alternating(best_seq, t_fin, eta_mw=1.5, mw_iters=3000, method='mw')
        local_update(a_fin, "final_mw")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin2 = lse_lbgfs_refine(best_seq, min(rem, 25.0),
                                  [0.5, 0.2, 0.1, 0.05, 0.02, 0.01])
        local_update(a_fin2, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin3, _ = coordinate_descent_polish(best_seq, n_sweeps=20, time_limit=min(rem, 15.0))
        local_update(a_fin3, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
