import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_simplex(a):
    """Project onto probability simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, b, n_peaks=20):
    """Compute subgradient of max_k (a*a)[k] using top-k active peaks.
    Returns (gradient_vector, max_value)."""
    n = len(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n), 0.0
    
    # Get top peaks with exponential weighting
    sorted_idx = np.argsort(b)[::-1]
    K = min(n_peaks, len(b))
    
    # Weight peaks by how close they are to the max
    top_vals = b[sorted_idx[:K]]
    # Soft weighting: peaks closer to max get more weight
    weights = np.exp(-(max_b - top_vals) / (max_b * 0.1 + 1e-15))
    weights /= weights.sum()
    
    g = np.zeros(n)
    for i in range(K):
        k = sorted_idx[i]
        w = weights[i]
        if w < 1e-10:
            continue
        # nabla_k(a)[j] = a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        g[js:je+1] += w * a[k - np.arange(js, je + 1)]
    
    return g, max_b


def mirror_descent_update(a, g, eta):
    """Multiplicative weights update on the simplex."""
    # a_j_new ∝ a_j * exp(-eta * g_j)
    log_a = np.log(np.maximum(a, 1e-300))
    log_new = log_a - eta * g
    log_new -= log_new.max()  # numerical stability
    new_a = np.exp(log_new)
    s = new_a.sum()
    if s < 1e-15:
        return a.copy()
    return new_a / s


def frank_wolfe_update(a, g, gamma):
    """Frank-Wolfe (conditional gradient) step on the simplex."""
    # LMO: pick coordinate with most negative g
    j = np.argmin(g)
    e_j = np.zeros(len(a))
    e_j[j] = 1.0
    return (1 - gamma) * a + gamma * e_j


def nesterov_mirror_step(a, y, g, eta, t):
    """Nesterov-accelerated mirror descent step.
    Returns (new_a, new_y)."""
    # Update y (lookahead)
    alpha = 2.0 / (t + 2)
    y_new = a + alpha * (y - a) if t > 0 else a.copy()
    y_new = normalize_simplex(y_new)
    
    # Mirror descent step from y
    a_new = mirror_descent_update(y_new, g, eta)
    return a_new, y_new


def simplified_lp_response(a, solver=None, max_iter=200):
    """Solve LP: min M s.t. (a*b)[k] <= M, sum(b)=1, b>=0.
    Returns (b, M) or (None, None)."""
    if solver is None:
        solver = _find_solver()
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build T: (T @ b)[k] = (a*b)[k]
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [solver]
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt not in solvers:
            solvers.append(alt)
    
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                bs = b_val.sum()
                if bs < 1e-15:
                    continue
                b_val = b_val / bs
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None

        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                return s
        except Exception:
            continue
    _solver_cache = cp.ECOS
    return cp.ECOS


def accelerated_subgradient(a0, time_limit, n_peaks=30, base_eta=0.5,
                           lp_every=500, solver=None, method="mirror",
                           print_interval=2000):
    """Accelerated subgradient / mirror descent with periodic LP correction."""
    n = len(a0)
    a = normalize_simplex(np.asarray(a0, dtype=np.float64))
    y = a.copy()
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_max
    rng = np.random.default_rng(12345)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution and subgradient at lookahead point y
        b_y = fast_convolve(y)
        max_y = b_y.max()
        g, _ = compute_subgradient(y, b_y, n_peaks)
        
        # Adaptive step size
        eta = base_eta / np.sqrt(t)
        
        # Update
        if method == "mirror":
            a, y = nesterov_mirror_step(a, y, g, eta, t)
        elif method == "fw":
            gamma = 2.0 / (t + 2)
            a = frank_wolfe_update(y, g, gamma)
        else:  # "grad"
            a_trial = a - eta * g
            a = normalize_simplex(a_trial)
            alpha = 2.0 / (t + 2)
            y = normalize_simplex(a + alpha * (y - a))
        
        # Track best
        b_a = fast_convolve(a)
        max_a = b_a.max()
        if max_a < best_max - 1e-12:
            best_max = max_a
            best_a = a.copy()
        
        # Periodic LP correction
        if lp_every > 0 and t % lp_every == 0 and method != "fw":
            b_lp, M_lp = simplified_lp_response(y, solver, max_iter=100)
            if b_lp is not None:
                # Blend: move towards LP solution
                blend = min(1.0, 5.0 / np.sqrt(t))
                a_trial = (1 - blend) * a + blend * b_lp
                a = normalize_simplex(a_trial)
                y = normalize_simplex(0.7 * y + 0.3 * a)
                b_a = fast_convolve(a)
                if b_a.max() < best_max:
                    best_max = b_a.max()
                    best_a = a.copy()
        
        # Stall detection with restart
        if t % 2000 == 0:
            if best_max < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_max
            
            if stall >= 8:
                # Perturb and restart
                noise_scale = 0.01 / np.sqrt(stall)
                y = y + rng.standard_normal(n) * noise_scale
                y = normalize_simplex(y)
                a = y.copy()
                stall = 0
                # Increase n_peaks for more diversified subgradient
                n_peaks = min(n_peaks + 10, 100)
            
            if print_interval and t % print_interval == 0:
                elapsed = time.monotonic() - t0
                score = 2.0 * n * best_max
                print(f"    AGD t={t} ({elapsed:.0f}s): score={score:.8f} max_b={best_max:.8f}")
    
    return best_a, 2.0 * n * best_max


def logsumexp_obj_and_grad(z, n, T):
    """Compute log-sum-exp smoothed objective and gradient."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    lse = max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T) * lse
    
    # Gradient
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0 / T
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_smooth_refine(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS refinement on log-sum-exp smoothed objective."""
    n = len(a0)
    a = normalize_simplex(np.asarray(a0, dtype=np.float64))
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]
    
    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_loc(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_obj_and_grad(z, n, T_stage)
            return obj
        
        def jac_loc(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, grad = logsumexp_obj_and_grad(z, n, T_stage)
            return grad
        
        try:
            iters = min(maxiter_per_stage, int(remaining * 20))
            iters = max(50, iters)
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def spectral_init(n, rng):
    """Create diverse spectral-based initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    
    if choice == 0:
        # Cosine raised to power
        a = np.cos(np.pi * x / 2) ** 3
    elif choice == 1:
        # Sinc-like
        t = np.pi * (x - 0.5)
        with np.errstate(divide='ignore', invalid='ignore'):
            a = np.sinc(t) ** 2
        a = np.nan_to_num(a)
    elif choice == 2:
        # Exponential decay from both ends
        a = np.exp(-3 * x) + np.exp(-3 * (1 - x))
    elif choice == 3:
        #two bumps
        a = np.exp(-8 * (x - 0.2) ** 2) + np.exp(-8 * (x - 0.8) ** 2) + 0.1
    elif choice == 4:
        # Flat middle, high edges
        a = 1.0 - 0.7 * np.exp(-20 * (x - 0.5) ** 2)
    elif choice == 5:
        # Power law
        a = (x * (1 - x)) ** 0.3 + 0.05
    elif choice == 6:
        # Random smooth
        freqs = rng.choice(np.arange(1, 10), size=5, replace=False)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.3, size=5)
        a = np.ones(n) + sum(amps[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i]) for i in range(5))
    elif choice == 7:
        # Triangular (B-spline-like)
        a = np.maximum(1 - np.abs(2 * x - 1), 0) ** 2 + 0.05
    elif choice == 8:
        # Hyperbolic
        a = 1.0 / (1 + 5 * (2 * x - 1) ** 2)
    else:
        # High-frequency oscillation damped
        a = (1 + 0.3 * np.cos(20 * np.pi * x)) * np.exp(-2 * (x - 0.5) ** 2) + 0.02
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def peak_targeted_cd(a0, time_limit, n_iterations=50000):
    """Coordinate descent specifically targeting the top convolution peaks."""
    n = len(a0)
    a = normalize_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        order = np.argsort(contribs)[::-1][:3]
        improved = False
        for oi in order:
            pos = idx[oi]
            if a[pos] < 1e-12:
                continue
            for factor in [0.98, 0.96, 0.94]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] * (1 - factor)
                if diff < 1e-15:
                    continue
                # Redistribute to other positions (favor positions that reduce other peaks)
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask].copy()
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = normalize_simplex(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining fast first-order methods with
    strategic LP and L-BFGS refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = _find_solver()
    print(f"Solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = spectral_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    n_best = len(best_seq)
    
    # === Phase 1: Fast accelerated subgradient (mirror descent) ===
    print(f"\n=== Phase 1: Mirror Descent n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_res, sc = accelerated_subgradient(best_seq, t1, n_peaks=30, base_eta=0.8,
                                            lp_every=300, solver=solver, method="mirror")
        try_update(a_res, "mirror_p1")
        
        # Also try Frank-Wolfe
        rem = deadline - time.monotonic()
        t1b = min(rem * 0.15, 200.0)
        if t1b > 20:
            print(f"  FW Budget: {t1b:.0f}s")
            a_fw, _ = accelerated_subgradient(best_seq, t1b, n_peaks=20, base_eta=1.0,
                                              lp_every=0, solver=solver, method="fw")
            try_update(a_fw, "fw_p1")
    
    # === Phase 2: Log-sum-exp L-BFGS ===
    print(f"\n=== Phase 2: L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 15:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]
        print(f"  T_stages: {len(T_stages)}, budget={t2:.0f}s")
        a_lb = lbgfs_smooth_refine(best_seq, t2, T_stages=T_stages, maxiter_per_stage=8000)
        try_update(a_lb, "lbgfs_p2")
    
    # === Phase 3: Peak-directed coordinate descent ===
    print(f"\n=== Phase 3: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 150.0)
    if t3 > 10:
        a_cd, _ = peak_targeted_cd(best_seq, t3)
        try_update(a_cd, "peak_cd")
    
    # === Phase 4: Second pass mirror descent with more peaks ===
    print(f"\n=== Phase 4: Mirror Descent (more peaks), score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 20:
        a_res2, _ = accelerated_subgradient(best_seq, t4, n_peaks=50, base_eta=0.5,
                                            lp_every=200, solver=solver, method="mirror")
        try_update(a_res2, "mirror_p4")
    
    # === Phase 5: Alternative length exploration ===
    print(f"\n=== Phase 5: Alt lengths, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        alt_lengths = [int(n_cur * 0.75), int(n_cur * 1.25), int(n_cur * 1.5)]
        t_each = min(rem / 4.0, 120.0)
        for alt_n in alt_lengths:
            if time.monotonic() > deadline - 30:
                break
            if t_each < 15:
                break
            # Interpolate to new length
            old_x = np.linspace(0, 1, n_cur)
            new_x = np.linspace(0, 1, alt_n)
            init_alt = np.interp(new_x, old_x, best_seq)
            init_alt = normalize_simplex(init_alt)
            
            print(f"  Trying n={alt_n}, t={t_each:.0f}s")
            a_alt, _ = accelerated_subgradient(init_alt, t_each * 0.6, n_peaks=30,
                                               base_eta=0.8, lp_every=400, solver=solver)
            try_update(a_alt, f"mirror_n{alt_n}")
            
            rem = deadline - time.monotonic()
            if rem > 20:
                a_alt_lb = lbgfs_smooth_refine(a_alt, rem * 0.3,
                                                T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02],
                                                maxiter_per_stage=3000)
                try_update(a_alt_lb, f"lbgfs_n{alt_n}")
    
    # === Phase 6: Multi-start with different initializations ===
    print(f"\n=== Phase 6: Multi-start, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 80.0)
        for si in range(3):
            if time.monotonic() > deadline - 20:
                break
            if t_each < 10:
                break
            a_init = spectral_init(n_cur, rng)
            # Blend with current best
            a_init = 0.5 * a_init + 0.5 * best_seq
            a_init = normalize_simplex(a_init)
            a_ms, _ = accelerated_subgradient(a_init, t_each, n_peaks=25,
                                              base_eta=0.7, lp_every=0, solver=solver)
            try_update(a_ms, f"multistart_{si}")
    
    # === Phase 7: Final LP-based refinement ===
    print(f"\n=== Phase 7: LP refinement, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t7 = min(rem * 0.3, 150.0)
        if t7 > 30:
            a_lp, _ = accelerated_subgradient(best_seq, t7, n_peaks=40, base_eta=0.6,
                                              lp_every=100, solver=solver, method="mirror")
            try_update(a_lp, "lp_refine")
    
    # === Phase 8: Final L-BFGS polish ===
    print(f"\n=== Phase 8: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.5, 60.0)
        if t8 > 5:
            a_fin = lbgfs_smooth_refine(best_seq, t8,
                                         T_stages=[1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01],
                                         maxiter_per_stage=5000)
            try_update(a_fin, "final_polish")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
