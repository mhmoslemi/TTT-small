def fast_convolve(a):
    """Compute full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_row(k, n):
    """Get the row k of the convolution matrix (which b positions are involved)."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    return js, je

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Anderson-accelerated LP best-response iteration (main workhorse)
2. LogSumExp L-BFGS with FFT-based gradient and multi-stage temperature
3. Nesterov subgradient descent on log-space
4. Diverse structured initializations and final polishing
Key: proper Anderson acceleration, efficient FFT gradients, robust time management.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global time reference
_t_end_global = 0.0


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def normalize_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def build_T_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_T_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def anderson_lp_iteration(a_init, time_limit, solver=None, memory=5, damp=1.0):
    global _t_end_global
    t_end = time.monotonic() + time_limit
    n = len(a_init)
    a = normalize_simplex(a_init)
    best_a = a.copy()
    best_score = compute_score(a)
    print(f"    AndersonLP start: score={best_score:.6f}, n={n}")

    X_hist = [a.copy()]
    G_hist = []

    it = 0
    while time.monotonic() < t_end:
        it += 1
        b, M_val = solve_best_response(a, solver)
        if b is None:
            break

        score_b = compute_score(b)
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
            if it % 50 == 0:
                print(f"      AndersonLP it={it}: best={best_score:.6f}, t={time.monotonic()+time_limit-t_end:.0f}s")

        g = b - a
        X_hist.append(a.copy())
        G_hist.append(g.copy())

        if len(X_hist) > memory:
            X_hist.pop(0)
            G_hist.pop(0)

        k = len(G_hist)
        if k == 1:
            a_next = b.copy()
        elif k >= 2:
            D = np.zeros((k, k))
            for i in range(k):
                for j in range(i, k):
                    D[i, j] = np.dot(G_hist[i], G_hist[j])
                    D[j, i] = D[i, j]
            sys_mat = np.zeros((k + 1, k + 1))
            sys_mat[:k, :k] = D + 1e-12 * np.eye(k)
            sys_mat[:k, k] = 1.0
            sys_mat[k, :k] = 1.0
            rhs = np.zeros(k + 1)
            rhs[k] = 1.0
            try:
                sol = np.linalg.solve(sys_mat, rhs)
                c = sol[:k]
            except np.linalg.LinAlgError:
                c = np.ones(k) / k
            a_next = np.zeros(n)
            for i in range(k):
                a_next += c[i] * X_hist[i]

        a_next = np.maximum(a_next, 0)
        s = a_next.sum()
        if s < 1e-15:
            a_next = np.ones(n) / n
        else:
            a_next = a_next / s

        a = a_next

        if it % 100 == 0 and time.monotonic() > t_end - time_limit + 30:
            elapsed = time.monotonic() + time_limit - t_end
            print(f"    AndersonLP it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lse_grad_a(a, T_temp, n):
    b = fast_convolve(a)
    Tb = T_temp * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    Z = exp_Tb.sum()
    p = exp_Tb / Z

    # Cross-correlation C[j] = sum_k p[k] * a[k-j] for valid indices
    fft_size = 1 << (3 * n - 2).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    C_full = irfft(P * np.conj(A), fft_size)
    C = C_full[:n]

    # Window sums W[j] = sum of p[k] for j <= k <= j+n-1 (clipped)
    p_cumsum = np.concatenate([[0.0], np.cumsum(p)])
    W = np.zeros(n)
    for j in range(n):
        k_start = max(0, j)
        k_end = min(2 * n - 2, j + n - 1)
        W[j] = p_cumsum[k_end + 1] - p_cumsum[k_start]

    grad_a = 2.0 * n * (C + a * W)
    return grad_a, p, b


def lse_lbgfs_optimize(a_init, time_limit, T_start=50.0, T_end=0.1, n_stages=10):
    global _t_end_global
    t_end = time.monotonic() + time_limit
    n = len(a_init)
    z = z_from_a(a_init).copy()
    best_a = normalize_simplex(a_init)
    best_score = compute_score(best_a)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() > t_end:
            break

        def func_obj(z_in):
            if time.monotonic() > t_end:
                return 1e15
            a_curr = softmax_from_z(z_in)
            b_curr = fast_convolve(a_curr)
            Tb_curr = T_val * b_curr
            max_Tb = Tb_curr.max()
            if max_Tb > 700:
                return 1e10
            exp_Tb = np.exp(Tb_curr - max_Tb)
            log_Z = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_Z

        def grad_obj(z_in):
            if time.monotonic() > t_end:
                return np.zeros_like(z_in)
            a_curr = softmax_from_z(z_in)
            grad_a, _, _ = lse_grad_a(a_curr, T_val, n)
            S = np.dot(grad_a, a_curr)
            return a_curr * (grad_a - S)

        max_iter_stage = max(15, min(50, int((time_limit / n_stages) * 2)))
        try:
            res = minimize(
                fun=func_obj, x0=z, jac=grad_obj, method='L-BFGS-B',
                options={'maxiter': max_iter_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': max_iter_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_curr = softmax_from_z(z)
        sc = compute_score(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
            print(f"    LSE-T{T_val:.1f}: score={best_score:.6f}")

    return best_a, best_score


def subgradient_nesterov(a_init, time_limit, n_steps=50000):
    t_end = time.monotonic() + time_limit
    n = len(a_init)
    a = normalize_simplex(a_init)
    z = z_from_a(a).copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.5
    momentum = 0.9

    for step in range(n_steps):
        if time.monotonic() > t_end:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        # Subgradient of max_b w.r.t. a (using active set)
        threshold = max_b - 0.05 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Nesterov momentum in z-space
        factor = (1 + np.sqrt(1 + 4 * (step + 1) ** 2)) / 2
        a_prev = softmax_from_z(z)
        v = momentum * v + (1 - momentum) * g / (a + 1e-10)
        z_new = z - lr * v
        a_new = softmax_from_z(z_new)

        # Check improvement
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z = z_new
            a = a_new
        else:
            lr = max(lr * 0.95, 0.01)

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() + time_limit - t_end
            print(f"    SubGrad it={step} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(5 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        a = np.ones(n) / n
        a += 0.1 * np.cos(np.pi * np.arange(n) / n)
    elif choice == 6:
        # Pseudo-random based
        a = rng.random(n)
        a = a ** 2
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2) * (1 + 0.2 * np.cos(10 * np.pi * x))
    elif choice == 8:
        # Fourier-based: flat spectrum target
        freq = rng.standard_normal(n)
        freq[0] = abs(freq[0])
        a = irfft(np.abs(freq) * np.exp(1j * rng.standard_normal(n)), n * 2)[:n]
        a = np.abs(a)
    else:
        a = np.ones(n)
        a += 0.3 * rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_equalization(a_init, time_limit, n_steps=20000):
    t_end = time.monotonic() + time_limit
    n = len(a_init)
    a = normalize_simplex(a_init)
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() > t_end:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        # Find contributing elements
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]

        # Move mass from high contributor to random low position
        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])
        dst = int(rng.integers(n))
        if dst == src:
            dst = (dst + 1) % n

        for frac in [0.01, 0.02, 0.05, 0.1]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                break
        else:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()

    return best_a, best_score


def multi_start_anderson_lp(a_list, time_limit, solver=None):
