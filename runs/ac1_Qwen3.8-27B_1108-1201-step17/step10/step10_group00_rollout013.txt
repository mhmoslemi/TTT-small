"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP alternating with warm starts (cvxpy PDLP/CLARABEL)
2. Adam-based LogSumExp optimization in log-space with adaptive temperature
3. Active-set-focused subgradient with Nesterov momentum
4. Multi-length exploration (n=1200 primary, 1500 secondary)
5. Structured perturbations and multi-restart strategy

Key differences from previous:
- Adam optimizer (O(n) per step, 100k+ steps/sec vs L-BFGS)
- Adaptive temperature based on solution roughness
- Warm-started LP for faster per-iteration convergence
- Top-k peak focusing for numerical stability
- Better time allocation across phases
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver, warm_start=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, warm_start=(warm_start is not None))
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0.0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp, top_k=None):
    """Compute LogSumExp objective and gradient w.r.t. z using top-k focusing."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    if top_k is not None and top_k < len(b):
        # Focus on top-k peaks
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx]
        max_b_top = b_top.max()
        exp_terms = np.exp((b_top - max_b_top) / T_temp)
        w_top = exp_terms / exp_terms.sum()
        # Reconstruct full weight vector (sparse)
        w = np.zeros(len(b))
        w[idx] = w_top
        lse = max_b_top + T_temp * np.log(exp_terms.sum())
    else:
        max_b = b.max()
        exp_terms = np.exp((b - max_b) / T_temp)
        w = exp_terms / exp_terms.sum()
        lse = max_b + T_temp * np.log(exp_terms.sum())
    
    obj = 2.0 * n * lse

    # Gradient: g_a[i] = (4n/T) * sum_k w[k] * a[k-i]
    # = (4n/T) * correlate(w, a)[n-1 + i]
    # Use FFT for efficiency
    fft_size = 1 << (2 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    g_a = irfft(W * A_rev, fft_size)[:n]
    # Actually let me just compute it directly via correlation
    # g_a[i] = sum_k w[k] * a[k-i] for valid k-i
    # This is np.correlate(w, a, 'full')[n-1:2*n-1]
    # But for speed, use FFT:
    # correlate(w, a)[i] = sum_k w[k+i] * a[k] (numpy convention)
    # I want: sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] = correlate(w, a) at lag i
    
    # Using np.correlate directly (it's C-optimized)
    g_a_raw = np.correlate(w, a, mode='full')
    g_a = g_a_raw[n - 1:2 * n - 1] * 2.0  # factor of 2 from derivative
    
    # Total gradient of objective w.r.t. a
    grad_a = (2.0 * n / T_temp) * g_a
    
    # Convert to z-space gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, 2.0 * n * grad_z / (2.0 * n)  # The 2n factor is already in grad_a


def adam_logsumexp(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    total_stages = len(T_stages)
    steps_per_stage = max(5000, int(time_limit * 50000 / total_stages))
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break
        
        top_k = max(5, int(top_k_frac * (2 * n - 1)))
        
        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            if (step + 1) % 5000 == 0 and time.monotonic() - t0 > 30:
                # Check time
                if time.monotonic() > deadline:
                    break
            
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute gradient
            if T_stage < 0.01:
                # Use pure subgradient on active set
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                # LogSumExp with top-k
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()
            
            # Gradient in a-space
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:2 * n - 1] * 2.0
            
            # Gradient in z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init * (lr_decay ** stage_idx) / (1.0 + 0.1 * step / steps_per_stage)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Keep z bounded to prevent numerical issues
            z -= z.mean()  # Centering (doesn't change softmax)
            
            # Progress print
            if (step + 1) % 20000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")
    
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(999)
    lp_times = []
    B_prev = None

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, warm_start=B_prev)
        B_prev = b_new
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        avg_lp_time = np.mean(lp_times[-20:]) if len(lp_times) >= 20 else 1.0

        # Nesterov dual averaging step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 20:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg_lp_time*1000:.1f}ms")

        if len(lp_times) > 50 and avg_lp_time > 5.0:
            print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping after {it+1} iters")
            break

    return best_a, best_score


def multiplicative_subgradient(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.03
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive active set
        threshold = max_b * (1.0 - max(1e-5, 1e-3 / (1 + step / 1000)))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubGrad {elapsed:.0f}s: score={best_score:.8f} eta_scale={scale:.4f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast LP alternating, Adam LogSumExp, multiplicative subgradient,
    and multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s (effective deadline at {deadline - time.monotonic():.0f}s)")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Quick Adam warm-up (fast, to get smoother landscape) ===
    print(f"\n=== Phase 1: Adam Warm-up n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages_warm = [20.0, 10.0, 5.0, 3.0, 2.0]
        a_adam, _ = adam_logsumexp(best_seq, t_alloc, T_stages_warm,
                                    lr_init=0.005, lr_decay=0.8, top_k_frac=0.3)
        local_update(a_adam, "adam_warm")

    # === Phase 2: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: Fine Adam with temperature annealing ===
    print(f"\n=== Phase 3: Adam Fine Annealing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages_fine = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
        a_adam2, _ = adam_logsumexp(best_seq, t_alloc, T_stages_fine,
                                     lr_init=0.003, lr_decay=0.85, top_k_frac=0.1)
        local_update(a_adam2, "adam_fine")

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Multiplicative subgradient polish ===
    print(f"\n=== Phase 5: MultiSubGrad Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_ms, _ = multiplicative_subgradient(best_seq, t_alloc, n_steps=200000)
        local_update(a_ms, "mult_subgrad")

    # === Phase 6: Try n=1500 ===
    print(f"\n=== Phase 6: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        t_n1500 = min(rem * 0.20, 300.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        # Quick Adam first
        t_adam_1500 = t_n1500 * 0.4
        a_1500a, _ = adam_logsumexp(init_1500, t_adam_1500,
                                     [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02],
                                     lr_init=0.003, lr_decay=0.85, top_k_frac=0.15)
        local_update(a_1500a, "adam_n1500")
        
        # Then LP
        rem = deadline - time.monotonic()
        if rem > 60:
            t_lp_1500 = min(rem * 0.5, 150.0)
            a_1500, _ = nesterov_lp_refine(best_seq, t_lp_1500, solver, n_max_iters=200000)
            local_update(a_1500, "lp_n1500")

    # === Phase 7: Perturbation escape ===
    print(f"\n=== Phase 7: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_pert = min(rem * 0.25, 200.0)
        n_cur = len(best_seq)
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.01 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.25, 50.0)
            # Quick Adam from perturbation
            a_opt, _ = adam_logsumexp(a_pert, t_sub,
                                       [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                       lr_init=0.003, lr_decay=0.85, top_k_frac=0.1)
            local_update(a_opt, f"pert_adam_t{trial}")

    # === Phase 8: Final polish with all methods ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.35, 40.0)
        if t_fin > 5:
            T_final = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
            a_fin, _ = adam_logsumexp(best_seq, t_fin, T_final,
                                       lr_init=0.001, lr_decay=0.9, top_k_frac=0.05)
            local_update(a_fin, "final_adam")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_final_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.5, 20.0), n_steps=50000)
        local_update(a_final_ms, "final_mult")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_final_lp, _ = nesterov_lp_refine(best_seq, min(rem * 0.6, 15.0), solver, n_max_iters=20000)
        local_update(a_final_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
