- Simpler: just use a random initialization and rely on the optimization

Actually, let me think more carefully about the Nesterov update. The standard FISTA update is:

t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2
y_{k+1} = a_{k+1} + (t_k - 1) / t_{k+1} * (a_{k+1} - a_k)
a_{k+1} = proj(y_{k+1} - (1/L) * grad(y_{k+1}))

But for our case, since we're on the simplex and using projection, the standard projected gradient with Nesterov acceleration is:

t_k = (1 + sqrt(1 + 4*(t_{k-1})^2)) / 2
y_k = a_k + (t_{k-1} - 1) / t_k * (a_k - a_{k-1})
a_{k+1} = proj_simplex(y_k - step_k * grad(y_k))

where step_k is the step size.

For the step size, since we're using a smoothed objective with parameter β, the Lipschitz constant L of the gradient is roughly β^2 * (max c range) * n / something. It's hard to bound precisely, so I'll use an adaptive step size with backtracking.

Actually, let me simplify. Since the problem is relatively low-dimensional (n ~ 1000) and we have 1800 seconds, I can afford to be a bit more careful:

1. Use a fixed (but well-chosen) step size for the smoothed gradient

- This naturally keeps a > 0 and might work better for this problem
- Normalization: divide by sum

Mirror descent has the advantage that it doesn't "kill" small coordinates as aggressively as Euclidean projection, which is important because we might need small but non-zero values.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated smoothed (log-sum-exp) gradient descent on the simplex
2. Mirror descent variant with exponential updates
3. Exact subgradient polishing with near-peak averaging
4. Smart coordinate descent for final refinement
5. Multi-length and multi-start exploration

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fft_cross_corr(w, a):
    """Compute (w ⊛ a)[m] = sum_k w[k+m]*a[k] for m=0..len(a)-1."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    return irfft(W * np.conj(A), N)[:n]

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(c, w, a):
    """Compute gradient of smoothed objective: g[m] = 2 * sum_j w[m+j] * a[j]."""
    return 2.0 * fft_cross_corr(w, a)

def nesterov_smoothed_cd(a0, time_limit, beta_start=10.0, beta_end=500.0, label=""):
    """
    Nesterov-accelerated gradient descent with log-sum-exp smoothing.
    Uses adaptive step size and increasing beta (annealing).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    prev_a = a.copy()
    t_k = 1.0
    
    # Precompute FFT sizes
    conv_N = 1 << (2 * n - 1).bit_length()
    corr_N = 1 << (3 * n - 2).bit_length()
    
    total_iters = 0
    no_improve = 0
    step = 0.005  # initial step size
    
    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        
        # Animate beta
        progress = min(1.0, total_iters / 5000.0)
        beta = beta_start + (beta_end - beta_start) * progress**0.5
        
        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a + (t_k - 1.0) / t_new * (a - prev_a)
        y = np.maximum(y, 0.0)
        y = simplex_project(y)
        t_k = t_new
        
        # Compute autocorrelation of y
        A = rfft(y, conv_N)
        c = irfft(A * A, conv_N)[:2*n-1]
        
        # Compute softmax weights
        c_shifted = beta * (c - c.max())
        c_shifted = np.clip(c_shifted, -50.0, 0.0)
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2*n-1)
            w[c.argmax()] = 1.0
            w_sum = 1.0
        w = w / w_sum
        
        # Compute gradient
        W = rfft(w, corr_N)
        Y = rfft(y, corr_N)
        g = 2.0 * irfft(W * np.conj(Y), corr_N)[:n]
        
        # Gradient step + projection
        a_new = y - step * g
        a_new = simplex_project(a_new)
        
        # Evaluate
        c_check = fast_convolve(a_new)
        score = 2.0 * n * c_check.max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        prev_a = a
        a = a_new
        
        # Adaptive step size
        if no_improve > 100:
            step *= 0.8
            no_improve = 0
            if step < 1e-10:
                break
        elif total_iters % 200 == 0:
            step = min(step * 1.01, 0.02)
        
        if total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSGD {label} it={total_iters} ({elapsed:.0f}s) beta={beta:.1f} step={step:.6f}: best={best_score:.6f}")
    
    return best_a, best_score

def mirror_descent(a0, time_limit, beta_start=10.0, beta_end=300.0, label=""):
    """Mirror descent with entropy mirror (exponential updates)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    conv_N = 1 << (2 * n - 1).bit_length()
    corr_N = 1 << (3 * n - 2).bit_length()
    
    total_iters = 0
    no_improve = 0
    step = 0.01
    
    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        
        progress = min(1.0, total_iters / 3000.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute autocorrelation
        A = rfft(a, conv_N)
        c = irfft(A * A, conv_N)[:2*n-1]
        
        # Softmax weights
        c_shifted = beta * (c - c.max())
        c_shifted = np.clip(c_shifted, -50.0, 0.0)
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2*n-1)
            w[c.argmax()] = 1.0
            w_sum = 1.0
        w = w / w_sum
        
        # Gradient
        W = rfft(w, corr_N)
        A2 = rfft(a, corr_N)
        g = 2.0 * irfft(W * np.conj(A2), corr_N)[:n]
        
        # Mirror descent step: a_new[i] ∝ a[i] * exp(-step * g[i])
        log_update = np.log(a) - step * g
        log_update -= log_update.max()  # for numerical stability
        a_new = np.exp(log_update)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        c_check = fast_convolve(a_new)
        score = 2.0 * n * c_check.max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 150:
            step *= 0.85
            no_improve = 0
            if step < 1e-10:
                break
        elif total_iters % 100 == 0:
            step = min(step * 1.005, 0.05)
        
        if total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={total_iters} ({elapsed:.0f}s) beta={beta:.1f}: best={best_score:.6f}")
    
    return best_a, best_score

def exact_subgradient_polish(a0, time_limit, label=""):
    """Polish using exact subgradient with averaging over near-peak indices."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    avg_a = a.copy()
    t_k = 1.0
    prev_a = a.copy()
    step = 0.003
    
    total_iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        
        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a + (t_k - 1.0) / t_new * (a - prev_a)
        y = np.maximum(y, 0.0)
        y = simplex_project(y)
        t_k = t_new
        
        # Compute autocorrelation of y
        c = fast_convolve(y)
        M = c.max()
        
        # Near-peak averaging
        threshold = M * 0.95  # use peaks within 5% of max
        near_peak = np.where(c >= threshold)[0]
        weights = c[near_peak]  # weight by their value
        
        # Gradient: g[m] = 2 * sum_k w_k * a[k-m] / sum(w_k)
        # = 2 * (w_normalized ⊛ a)[m]
        w_norm = weights / weights.sum()
        # Create full weight vector
        w_full = np.zeros(2*n-1)
        w_full[near_peak] = w_norm
        
        # Cross-correlation
        g = 2.0 * fft_cross_corr(w_full, y)
        
        # Step
        a_new = y - step * g
        a_new = simplex_project(a_new)
        
        # Nesterov average
        gamma_it = 2.0 / (total_iters + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new
        
        # Evaluate both
        for cand in [a_new, avg_a]:
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = cand.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
        
        prev_a = a
        a = a_new
        
        if no_improve > 200:
            step *= 0.9
            no_improve = 0
            if step < 1e-10:
                break
        elif total_iters % 50 == 0:
            step = min(step * 1.002, 0.01)
        
        if total_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} it={total_iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def coord_descent_fast(a0, time_limit, label=""):
    """Fast coordinate descent with 1D minimax per coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end+1:].max()))
            
            affected = conv[k_start:k_end+1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            
            # Find best delta by checking where unconstrained max equals affected max
            # The new max after changing a[i] by delta is:
            # max(unaff_max, max_j (affected[j] + delta*coeffs[j]) [, affected[quad_idx] + delta^2])
            
            # Candidate deltas:
            candidates = [0.0, -ai]
            
            # Where each affected line crosses unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai <= d <= 2 * ai:
                        candidates.append(d)
            
            # Where top affected lines cross each other
            top_k = min(8, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai <= d <= 2 * ai:
                            candidates.append(d)
            
            # Also try the "equalize" point: where max of affected = some target
            # Binary search for optimal delta
            best_delta = 0.0
            best_new_max = M
            
            for delta in candidates:
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-13:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end+1] += best_delta * coeffs
                if has_quad:
                    conv[2*i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def random_init(n, rng):

def structured_init(n, rng, method="gaussian"):
    """Structured initialization."""
    if method == "gaussian":
        # Half-Gaussian (folded normal)
        a = np.abs(rng.standard_normal(n))
        a = a / a.sum()
    elif method == "exponential":
        a = np.exp(-np.abs(np.arange(n) - n//2) / (n * 0.3))
        a = a / a.sum()
    elif method == "cosine":
        x = np.linspace(0, np.pi, n)
        a = np.cos(x * 0.5) ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
