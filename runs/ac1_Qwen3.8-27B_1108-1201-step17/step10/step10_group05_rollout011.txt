import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def coord_descent(a0, time_limit, verbose=False, label="", top_k_cross=8):
    """High-efficiency coordinate descent with exact 1D minimax."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            # Generate candidate deltas
            candidates = [-ai]  # minimum (set to zero)

            # Where individual lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-16:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.1 < d < 0.5 * ai:
                        candidates.append(d)

            # Where top affected lines cross each other
            top_k = min(top_k_cross, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-16:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.1 < d < 0.5 * ai:
                            candidates.append(d)

            # Also try setting peak to equal unaff_max more precisely
            # Binary search for the delta that minimizes the max
            # (This is a convex piecewise-linear (plus one quadratic) function)
            #(do a few bisection steps around the apparent minimum)
            
            # Fractional moves
            for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 0.95, 0.99]:
                candidates.append(-frac * ai)
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                candidates.append(frac * ai * 0.5)

            # Evaluate candidates
            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            # If we found an improvement, also try to refine with a few bisection steps
            if best_delta != 0.0 and best_new_max < M - 1e-15:
                # Try a few local bisection refinements
                lo = best_delta - 0.05 * ai
                hi = best_delta + 0.05 * ai
                lo = max(lo, -ai)
                hi = min(hi, 0.5 * ai)
                for _ in range(10):
                    mid = (lo + hi) / 2
                    new_aff = affected + mid * coeffs
                    if has_quad:
                        new_aff = new_aff.copy()
                        new_aff[quad_idx] += mid * mid
                    m_val = max(unaff_max, float(new_aff.max()))
                    # Determine which side is better
                    mid_left = (lo + mid) / 2
                    mid_right = (mid + hi) / 2
                    new_aff_l = affected + mid_left * coeffs
                    if has_quad:
                        new_aff_l = new_aff_l.copy()
                        new_aff_l[quad_idx] += mid_left * mid_left
                    ml = max(unaff_max, float(new_aff_l.max()))
                    new_aff_r = affected + mid_right * coeffs
                    if has_quad:
                        new_aff_r = new_aff_r.copy()
                        new_aff_r[quad_idx] += mid_right * mid_right
                    mr = max(unaff_max, float(new_aff_r.max()))
                    if ml <= mr:
                        hi = mid
                    else:
                        lo = mid
                
                # Use the best of bisection result and original
                for delta2 in [lo, hi, (lo+hi)/2]:
                    delta2 = max(delta2, -ai)
                    new_aff = affected + delta2 * coeffs
                    if has_quad:
                        new_aff = new_aff.copy()
                        new_aff[quad_idx] += delta2 * delta2
                    m_val = max(unaff_max, float(new_aff.max()))
                    if m_val < best_new_max - 1e-16:
                        best_new_max = m_val
                        best_delta = delta2

            if best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Periodically recompute to avoid drift
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
            if M < best_score / (2.0 * n):
                best_score = 2.0 * n * M
                best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps

def lp_fictitious_play_nesterov(a0, time_limit, solver=None, label=""):
    """LP fictitious play with Nesterov-style averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    avg_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            break

        # Nesterov-style step
        alpha = 2.0 / (iters + 2.0)
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Nesterov average
        avg_a = (1 - alpha) * avg_a + alpha * a_new

        a = a_new

        # Evaluate both the iterate and the average
        for cand in [a_new, avg_a, b_new]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        if iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP-N {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def pressure_perturb(a0, rng, strength=0.05):
    """Perturb based on pressure at each coordinate."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    
    # Compute pressure: how much each coordinate contributes to the peaks
    pressure = np.zeros(n)
    # Use all positions where conv is above 80% of max
    threshold = M * 0.8
    active_k = np.where(conv >= threshold)[0]
    
    for k in active_k:
        w = (conv[k] - threshold) / (M - threshold + 1e-30)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += w * a[i] * a[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Perturb: reduce high-pressure coordinates, increase low-pressure
    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure) + strength * 0.1 * np.abs(noise) * a.max()
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.15):
    """Randomly restructure a portion of the sequence."""
    n = len(a0)
    a = a0.copy()
    mask = rng.random(n) < frac
    a[mask] = 0
    a[~mask] = a[~mask] * (1.0 + rng.standard_normal((~mask).sum()) * 0.2)
    a = np.maximum(a, 0)
    
    # Add some random energy to zeroed positions
    zero_mask = a < 1e-16
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.3))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.3

    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def gaussian_perturb(a0, rng, sigma=0.02):
    """Simple Gaussian perturbation."""
    n = len(a0)
    a = a0 * (1.0 + rng.standard_normal(n) * sigma)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fine_polish(a0, time_limit, label=""):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. High-efficiency coordinate descent with exact 1D minimax and bisection refinement
2. LP fictitious play with Nesterov averaging for global basin-hopping

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLOP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

            break

        alpha = 2.0 / (iters + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        avg_a = (1.0 - alpha) * avg_a + alpha * a_new
        a = a_new

        for cand in [a_new, avg_a, b_new]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP-N {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def pressure_perturb(a0, rng, strength=0.04):
    """Perturb based on pressure at each coordinate (peak contribution)."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()

    pressure = np.zeros(n)
    threshold = M * 0.75
    active_k = np.where(conv >= threshold)[0]

    for k in active_k:
        w = (conv[k] - threshold) / (M - threshold + 1e-30)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += w * a[i] * a[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure) + strength * 0.05 * np.abs(noise) * a.max()
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def random_restructure(a0, rng, frac=0.12):
    """Randomly restructure a portion of the sequence."""
    n = len(a0)
    a = a0.copy()
    mask = rng.random(n) < frac
    a[mask] = 0
    n_keep = (~mask).sum()
    if n_keep > 0:
        a[~mask] = a[~mask] * (1.0 + rng.standard_normal(n_keep) * 0.15)
    a = np.maximum(a, 0)

    zero_mask = a < 1e-16
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.25))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.2

    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def gaussian_perturb(a0, rng, sigma=0.015):
    """Simple Gaussian multiplicative perturbation."""
    n = len(a0)
    a = a0 * (1.0 + rng.standard_normal(n) * sigma)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def fine_polish(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
                           0.99, 0.985, 0.98, 1.001, 1.002, 1.005, 1.01, 1.015, 1.02]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-17:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            if sweeps > 5:
                break

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Strategy: alternate between coordinate descent (local) and LP fictitious play
    (global basin-hopping), with pressure-based perturbations and multi-n exploration.
    """
    deadline = time.monotonic() + 1765.0
    rng = np.random.default_rng(777)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = p.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Init: n={n_init}, score={best_score:.6f}")

    # === Phase 1: Initial intensive CD ===
    print(f"\n=== Phase 1: CD (best={best_score:.6f}) ===")
    t1 = min(100.0, (deadline - time.monotonic()) * 0.08)
    print(f"  Budget: {t1:.0f}s, n={len(best_seq)}")
    a1, s1, _ = coord_descent(best_seq, t1, verbose=True, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_cd")

    # === Phase 2: LP Fictitious Play (basin hop) ===
    print(f"\n=== Phase 2: LPFP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 900
    t2 = min(max(rem * 0.2, 30), 150.0)
    print(f"  Budget: {t2:.0f}s")
    a2, s2, _ = lp_fictitious_play_nesterov(best_seq, t2, solver=solver, label="p2")
    best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_lpfp")

    # === Phase 3: CD after LP (exploit new basin) ===
    print(f"\n=== Phase 3: CD post-LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 700
    t3 = min(max(rem * 0.2, 30), 120.0)
    print(f"  Budget: {t3:.0f}s")
    a3, s3, _ = coord_descent(best_seq, t3, verbose=False, label="p3")
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_cd")

    # === Phase 4: Alternating LP + CD with perturbations ===
    print(f"\n=== Phase 4: Alternating + Perturb (best={best_score:.6f}) ===")
    temperature = 0.03
    for rnd in range(5):
        rem = deadline - time.monotonic() - 400
        if rem < 80:
            break

        # Sometimes do LP, sometimes perturb
        if rnd % 2 == 0 and rem > 200:
            # LP step (shorter)
            t_lp = min(40.0, rem * 0.15)
            a_lp, s_lp, _ = lp_fictitious_play_nesterov(best_seq, t_lp, solver=solver, label=f"p4lp{rnd}")
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, f"p4_lp{rnd}")
        else:
            # Perturbation
            pert_type = rng.integers(0, 3)
            if pert_type == 0:
                perturbed = pressure_perturb(best_seq, rng, strength=temperature)
            elif pert_type == 1:
                perturbed = random_restructure(best_seq, rng, frac=0.08 + 0.04 * rnd)
            else:
                perturbed = gaussian_perturb(best_seq, rng, sigma=temperature * 0.5)

        # CD to optimize
        t_cd = min(60.0, rem * 0.25)
        a4, s4, _ = coord_descent(best_seq if best_score < s4 else perturbed if 'perturbed' in dir() else best_seq,
                                   t_cd, label=f"p4r{rnd}")
        best_seq, best_score = check_and_update(a4, best_seq, best_score, f"p4r{rnd}")
        temperature *= 0.75

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n (best={best_score:.6f}) ===")
    n_cur = len(best_seq)
    alt_ns = [n for n in [896, 960, 1152, 1280, 1408] if abs(n - n_cur) > 50]
    for n_alt in alt_ns[:3]:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        t5 = min(rem * 0.2, 50.0)
        init_alt = interpolate_seq(best_seq, n_alt)
        a5, s5, _ = coord_descent(init_alt, t5, label=f"p5_n{n_alt}")
        best_seq, best_score = check_and_update(a5, best_seq, best_score, f"p5_n{n_alt}")

    # === Phase 6: Final alternate LP+CD (if time) ===
    print(f"\n=== Phase 6: Final LP+CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 120
    if rem > 80:
        t6 = min(50.0, rem * 0.3)
        a6, s6, _ = lp_fictitious_play_nesterov(best_seq, t6, solver=solver, label="p6lp")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, "p6_lp")

        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t6b = min(60.0, rem * 0.6)
            a6b, s6b, _ = coord_descent(best_seq, t6b, label="p6cd")
            best_seq, best_score = check_and_update(a6b, best_seq, best_score, "p6_cd")

    # === Phase 7: Intensive final polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t7 = min(rem * 0.5, 80.0)
        a7, s7 = fine_polish(best_seq, t7, label="p7")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, "p7_fine")

    rem = deadline - time.monotonic() - 10
    if rem > 15:
        t7b = min(rem * 0.7, 40.0)
        a7b, s7b, _ = coord_descent(best_seq, t7b, label="p7b")
        best_seq, best_score = check_and_update(a7b, best_seq, best_score, "p7b_cd")

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a7c, s7c = fine_polish(best_seq, min(rem * 0.8, 20.0), label="p7c")
        best_seq, best_score = check_and_update(a7c, best_seq, best_score, "p7c_fine")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
