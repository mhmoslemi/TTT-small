"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using dual-guided alternating LP,
peak-shaving targeted LPs, geometric interpolation, adaptive perturbation,
and multi-n exploration with intensive L-BFGS polishing.
Key novel ideas: (1) using LP dual variables to form weighted combination directions,
(2) aggressive peak-shaving that targets only the top few convolution positions,
(3) geometric-mean interpolation for better near-boundary progress,
(4) perturbation-restart to escape local minima.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def geometric_interp(a, b, t):
    """Geometric-mean interpolation, projected to simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-300)
    b = np.maximum(np.asarray(b, dtype=np.float64), 1e-300)
    g = np.power(a, 1.0 - t) * np.power(b, t)
    s = g.sum()
    if s < 1e-15:
        return normalize(a)
    return g / s


def solve_lp_best_response(a_fixed, solver, top_frac=1.0):
    """
    Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Optionally only constrain the top_frac fraction of positions (for speed).
    Returns (b, M, dual_vars) or (None, None, None).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    size = 2 * n - 1

    # Compute current conv to identify which positions matter
    b = T @ a_fixed
    M_current = b.max()

    constraints = [cp.sum(b_var) == 1]

    if top_frac < 1.0:
        # Only constrain positions where current conv is close to max
        threshold = M_current * (1.0 - (1.0 - top_frac) * 0.5)
        active_mask = b >= threshold
        if active_mask.sum() == 0:
            active_mask = np.zeros(size, dtype=bool)
            active_mask[int(np.argmax(b))] = True
        T_active = T[active_mask]
        y_dual = cp.Variable(active_mask.sum(), nonneg=True)
        constraints.append(T_active @ b_var <= M_var)
    else:
        T_active = T
        constraints.append(T @ b_var <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for attempt_solver in [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            prob.solve(solver=attempt_solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val = b_val / s
                    M_val = float(M_var.value)
                    return b_val, M_val
        except Exception:
            continue
    return None, None


def solve_lp_peak_shave(a_fixed, solver, top_k=5, margin=0.0):
    """
    Targeted peak shaving: minimize the max over only the top_k positions
    of conv(a', a_fixed), while keeping all other positions below current_max + margin.
    """
    n = len(a_fixed)
    b = fast_convolve(a_fixed)
    M_current = b.max()
    size = 2 * n - 1

    # Get top_k positions
    top_positions = np.argsort(b)[-top_k:][::-1]

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)

    constraints = [cp.sum(b_var) == 1]

    # Tight constraint on top positions
    T_top = T[top_positions]
    constraints.append(T_top @ b_var <= M_var)

    # Loose constraint on remaining positions
    remaining = np.setdiff1d(np.arange(size), top_positions)
    if len(remaining) > 0:
        T_rem = T[remaining]
        constraints.append(T_rem @ b_var <= M_current * (1.0 + margin))

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for attempt_solver in [solver, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=attempt_solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val = b_val / s
                    return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def dual_guided_alternation(a0, solver, time_limit, max_iters=50000):
    """
    Alternating LP with dual-guided updates and geometric interpolation.
    Uses top_frac to reduce LP size, and geometric interpolation for updates.
    """
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Nesterov-style averages
    y_a = a.copy()
    y_b = b.copy()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Best b given y_a
        b_new, M1 = solve_lp_best_response(y_a, solver, top_frac=1.0)
        if b_new is None:
            b_new, M1 = solve_lp_best_response(y_a, solver, top_frac=0.8)

        # Step 2: Best a given y_b
        a_new, M2 = solve_lp_peak_shave(y_b, solver, top_k=8, margin=0.05)
        if a_new is None:
            a_new, M2 = solve_lp_best_response(y_b, solver, top_frac=1.0)

        if it % 500 == 0 and it > 0:
            sc_cur = 2.0 * n * fast_convolve(a).max()
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                # Perturb to escape
                noise = rng.standard_normal(n) * 0.01 / math.sqrt(stall_count)
                y_a = normalize(y_a + noise)
                y_b = normalize(y_b + noise * 0.5)
                stall_count = 0
            if (it + 1) % 2000 == 0:
                print(f"    DGA iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

        if b_new is not None:
            alpha = 2.0 / (it + 2)
            # Use geometric interpolation for the update
            y_b = normalize((1.0 - alpha) * y_b + alpha * b_new)
            # Nesterov combination
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        if a_new is not None:
            alpha_a = 2.0 / (it + 3)
            y_a = normalize((1.0 - alpha_a) * y_a + alpha_a * a_new)
            a = ((it + 1) * a + (alpha_a * (it + 3)) * a_new) / (it + 1 + alpha_a * (it + 3))
            a = normalize(a)

        # Track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        # Also try geometric interpolation of best with new
        if it % 100 == 0:
            for t in [0.3, 0.5, 0.7]:
                g = geometric_interp(best_a, a, t)
                sc_g = 2.0 * n * fast_convolve(g).max()
                if sc_g < best_score:
                    best_score = sc_g
                    best_a = g.copy()

    return best_a, best_score


def intensive_peak_shave(a0, solver, time_limit, max_rounds=200):
    """
    Intensive peak shaving: iteratively identify top peaks and solve
    targeted LPs to reduce them, with adaptive margin.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        current_score = 2.0 * n * M

        # Get top positions with decreasing tolerance
        frac = max(0.3, 1.0 - 0.5 * rnd / max_rounds)
        threshold = M * (1.0 - frac * 0.1)
        active = np.where(b >= threshold)[0]
        top_k = min(len(active), 10 + 5 * (rnd // 20))

        # Solve peak-shaving LP with decreasing margin
        margin = max(0.0, 0.05 * (1.0 - rnd / max_rounds))
        a_new, M_new = solve_lp_peak_shave(a, solver, top_k=top_k, margin=margin)

        if a_new is not None:
            # Try several step sizes
            for step in [1.0, 0.7, 0.5, 0.3]:
                a_trial = normalize(a + step * (a_new - a))
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    break

        # Also try solving the other direction
        b_new, M_b = solve_lp_peak_shave(a, solver, top_k=top_k, margin=margin)
        if b_new is not None:
            for step in [1.0, 0.7, 0.5]:
                b_trial = normalize(a + step * (b_new - a))
                sc = 2.0 * n * fast_convolve(b_trial).max()
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = b_trial.copy()
                    a = b_trial
                    break

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(30, int(remaining * 15))

        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent_shave(a0, time_limit, max_steps=100000):
    """
    Coordinate descent: for each position j, find the optimal value of a[j]
    (given all other a fixed) that minimizes the max convolution.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Precompute: for each j, which conv positions involve a[j]
    # conv[k] involves a[j] when k-j is in [0, n-1], i.e., j <= k <= j+n-1
    # So a[j] affects conv[j], conv[j+1], ..., conv[j+n-1]

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a coordinate to update (cyclic with random perturbation)
        j = step % n if step < n * 3 else rng.integers(0, n)

        # Save old value
        old_val = a[j]

        # Fix all other entries, find best a[j]
        # conv[k] = a[j]*a[k-j] + (rest) for k in [j, j+n-1]
        # Also conv[k] = a[k-j]*a[j] + (rest) — same thing by symmetry
        # The contribution of a[j] to conv[k] is: a[j] * (a[k-j] if 0<=k-j<n and k-j != j)
        #                                          + a[j]^2 if k-j == j (i.e., k = 2j)

        # Let's compute the "background" conv without a[j]
        a_no_j = a.copy()
        a_no_j[j] = 0
        b_bg = fast_convolve(a_no_j)

        # The full conv as a function of x = a[j]:
        # conv[k] = b_bg[k] + x * c[k] + (x^2 if k == 2j)
        # where c[k] = a[k-j] if 0 <= k-j < n and k-j != j (i.e., the cross term)
        #             Actually: conv[k] = sum_i a[i]*a[k-i]
        # With a[j] = x: conv[k] includes x*a[k-j] (if k-j != j and valid)
        #                           + a[j]*x (if k-j == j, i.e., k=2j) = x^2
        #                           + x*a[j-k] (if j-k != j and valid, i.e., k != 0 and j-k valid)

        # More carefully: conv[k] = sum_i a[i] * a[k-i]
        # Terms involving a[j]: i=j gives a[j]*a[k-j] = x*a[k-j] (if 0<=k-j<n)
        #                       i=k-j gives a[k-j]*a[j] = a[k-j]*x (if 0<=k-j<n and k-j != j)
        # So if k-j != j: conv[k] has 2*x*a[k-j] from these two terms
        # If k-j == j (i.e., k=2j): conv[k] has x*a[j] = x^2 from the i=j term,
        #   and also a[k-j]*a[j] = a[j]*x = x^2 from the i=k-j=j term... wait that's the same term.
        # Actually no: when i=j, the term is a[j]*a[k-j]. When i=k-j, the term is a[k-j]*a[j].
        # If k-j = j, both give a[j]*a[j] = x^2, but they're the SAME term (i=j=k-j).
        # So: if k = 2j: contribution is x^2 (one term).
        #     if k != 2j and 0 <= k-j < n: contribution is 2*x*a[k-j] (two terms, i=j and i=k-j).
        #     But also if 0 <= j-k < n and j-k != j: contribution is 2*x*a[j-k]...
        #     Wait, I'm double counting. Let me be very explicit.

        # conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
        # Terms where i = j: a[j] * a[k-j] (valid if 0 <= k-j < n)
        # Terms where k-i = j, i.e., i = k-j: a[k-j] * a[j] (valid if 0 <= k-j < n)
        # These are the same term when i = j = k-j, i.e., k = 2j.

        # So the contribution of x = a[j] to conv[k]:
        # If k = 2j: x * a[k-j] = x * a[j] = x^2. Just one term.
        # If k != 2j and 0 <= k-j < n: two terms: x*a[k-j] + a[k-j]*x = 2*x*a[k-j]
        #   (provided k-j != j, which is true since k != 2j)
        # Otherwise: 0

        # So: conv[k] = b_bg[k] + 2*x*a[k-j] (if 0<=k-j<n and k!=2j)
        #                     + x^2 (if k==2j)

        # We want to minimize max_k conv[k] = max_k (b_bg[k] + linear_or_quad_in_x)
        # This is a 1D optimization in x >= 0, with the constraint that sum(a) = 1
        # means x = 1 - sum_{i!=j} a[i] = 1 - (S - old_val) where S = sum(a)

        # Since we normalized, S = 1, so x is determined! We can't change a single coordinate
        # without renormalizing. So instead, let's do: change a[j] to x, then renormalize.
        # After renormalization, all entries scale by 1/(1 - old_val + x).

        # This makes the 1D problem more complex. Let's use a simpler approach:
        # Binary search / golden section on x.

        remaining_sum = 1.0 - old_val
        # x can range from 0 to 1 (then renormalize)
        lo, hi = 0.0, max(2.0 * old_val, 0.1)
        best_x = old_val
        best_f = best_score

        for _ in range(20):  # 20 bisection steps
            mid = (lo + hi) / 2.0
            a_trial = a.copy()
            a_trial[j] = mid
            a_trial = normalize(a_trial)
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_f:
                best_f = sc
                best_x = mid
                lo = mid
            else:
                hi = mid

        a[j] = best_x
        a = normalize(a)
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def fourier_refine(a0, n_freq, time_limit):
    """Refine in Fourier domain with Nelder-Mead."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)

    n_free = 2 * n_freq + 1
    x0 = np.zeros(n_free)
    x0[0] = A[0].real
    for i in range(1, n_freq + 1):
        x0[2 * i - 1] = A[i].real
        x0[2 * i] = A[i].imag

    A_rest = A[n_freq + 1:].copy()

    def fourier_obj(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()

    try:
        res = minimize(
            fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': 8000, 'xatol': 1e-14, 'fatol': 1e-14, 'adaptive': True}
        )
        x_opt = res.x
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s > 1e-10:
            a_candidate = a_candidate / s
            sc = 2.0 * n * fast_convolve(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
    except Exception:
        pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: dual-guided LP alternation, intensive peak shaving,
    coordinate descent, Fourier refinement, L-BFGS polish, multi-n exploration,
    and perturbation-restart for escaping local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        # Try a few structured inits
        candidates = []
        a1 = np.sin(np.pi * x) ** 2
        a2 = (1 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x))
        a3 = np.exp(-2 * (x - 0.5) ** 2 / 0.1)
        for c in [a1, a2, a3]:
            c = np.maximum(c, 0)
            c = c / c.sum()
            sc = local_eval(c)
            candidates.append((c, sc))
        best_seq = candidates[0][0]
        best_score = candidates[0][1]
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Dual-guided LP alternation (main workhorse) ===
    print(f"\n=== Phase 1: Dual-Guided LP (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = dual_guided_alternation(best_seq, solver, t1)
        try_update(a_lp, "dual_lp")

    # === Phase 2: Intensive peak shaving ===
    print(f"\n=== Phase 2: Peak Shaving (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_ps, sc_ps = intensive_peak_shave(best_seq, solver, t2)
        try_update(a_ps, "peak_shave")

    # === Phase 3: Second round of LP alternation from improved point ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        a_lp2, _ = dual_guided_alternation(best_seq, solver, t3)
        try_update(a_lp2, "lp_round2")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 15:
        a_cd, _ = coordinate_descent_shave(best_seq, t4, max_steps=50000)
        try_update(a_cd, "coord_desc")

    # === Phase 5: L-BFGS LogSumExp polish ===
    print(f"\n=== Phase 5: L-BFGS Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 250.0)
    if t5 > 20:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lb, _ = lbgfs_logsumexp(best_seq, t5, T_stages=T_stages)
        try_update(a_lb, "lbgfs")

    # === Phase 6: Fourier refinement ===
    print(f"\n=== Phase 6: Fourier Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        for nf in [32, 64, 128]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_f = min(rem * 0.2, 50.0)
            a_f, _ = fourier_refine(best_seq, nf, t_f)
            try_update(a_f, f"fourier_{nf}")

    # === Phase 7: Perturbation + restart ===
    print(f"\n=== Phase 7: Perturbation Restart (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for pert_scale in [0.01, 0.02, 0.05]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            noise = rng.standard_normal(n_best) * pert_scale
            a_pert = normalize(best_seq + noise)
            t_p = min(rem * 0.3, 60.0)
            a_r, _ = dual_guided_alternation(a_pert, solver, t_p)
            try_update(a_r, f"pert_{pert_scale}")
            # Also try L-BFGS from perturbed point
            rem = deadline - time.monotonic()
            if rem > 20:
                a_r2, _ = lbgfs_logsumexp(a_r, min(rem * 0.3, 40.0),
                                          T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02])
                try_update(a_r2, f"pert_lb_{pert_scale}")

    # === Phase 8: Multi-n exploration ===
    print(f"\n=== Phase 8: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_tries = [int(n_best * m) for m in [0.8, 0.9, 1.1, 1.3, 1.5]]
        n_tries = [n for n in n_tries if 400 <= n <= 3000 and n != n_best]
        seen_n = set()
        for cand_n in n_tries:
            if cand_n in seen_n:
                continue
            seen_n.add(cand_n)
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_sub = min(rem * 0.25, 80.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}")
            a_interp = interpolate_sequence(best_seq, cand_n)
            a_n, _ = dual_guided_alternation(a_interp, solver, t_sub * 0.5)
            try_update(a_n, f"lp_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_n2, _ = lbgfs_logsumexp(a_n, min(rem * 0.4, 40.0),
                                          T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05])
                try_update(a_n2, f"lb_n{cand_n}")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Peak shave
        t_fs = min(rem * 0.3, 40.0)
        a_ps2, _ = intensive_peak_shave(best_seq, solver, t_fs)
        try_update(a_ps2, "final_ps")

        # L-BFGS very low T
        rem = deadline - time.monotonic()
        if rem > 15:
            t_fl = min(rem * 0.4, 40.0)
            a_fl, _ = lbgfs_logsumexp(best_seq, t_fl,
                                      T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
            try_update(a_fl, "final_lb")

        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fc, _ = coordinate_descent_shave(best_seq, min(rem * 0.4, 20.0), max_steps=20000)
            try_update(a_fc, "final_cd")

        # Geometric interpolation between best and slightly perturbed
        rem = deadline - time.monotonic()
        if rem > 5:
            for t in [0.1, 0.3, 0.5]:
                noise = rng.standard_normal(n_best) * 0.001
                a_g = geometric_interp(best_seq, normalize(best_seq + noise), t)
                try_update(a_g, f"geom_{t}")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
