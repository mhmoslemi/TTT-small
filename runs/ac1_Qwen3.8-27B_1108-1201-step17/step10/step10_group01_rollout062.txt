    # This is a 1D piecewise linear minimization
    # The minimum is at a point where two or more peaks are tied at the max
    
    # For efficiency, only consider the top few peaks
    top_k = np.argsort(conv)[::-1][:50]
    c = conv[top_k]
    dd = d[top_k]
    
    # For each pair of peaks (p, q), find where c[p] + 2*delta*dd[p] = c[q] + 2*delta*dd[q]
    # => delta = (c[q] - c[p]) / (2*(dd[p] - dd[q]))
    
    candidates = [0.0]
    for p in range(len(c)):
        for q in range(p+1, len(c)):
            if abs(dd[p] - dd[q]) < 1e-15:
                continue
            delta = (c[q] - c[p]) / (2*(dd[p] - dd[q]))
            if delta_range[0] <= delta <= delta_range[1]:
                candidates.append(delta)
    
    best = min(conv.max())
    best_delta = 0
    for delta in candidates:
        val = (c + 2*delta*dd).max()
        if val < best:
            best = val
            best_delta = delta
    return best_delta

            scaled = np.maximum(scaled, -700)  # Avoid underflow

            # Compute gradient

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive multi-stage LogSumExp
L-BFGS with FFT-accelerated gradients, balanced-peak coordinate descent, direct
subgradient polishing, and multi-start perturbation. Key improvements: FFT-based
cross-correlation in the gradient (enabling 10x more iterations), T cooling to
0.001, smarter peak-targeting CD with pair swaps, and adaptive time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_gradient(a, b, T):
    """Compute gradient of 2n*LSE_T(conv(a,a)) w.r.t. log(a) using FFT cross-correlation."""
    n = len(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled = np.clip(scaled, -700.0, 0.0)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()

    # g_i = 2 * sum_k p[k] * a[k-i] = 2 * cross_correlation(p, a)[i]
    fft_len = 1 << (3 * n - 1).bit_length()
    P = rfft(p, fft_len)
    A = rfft(a, fft_len)
    cross = irfft(np.conj(P) * A, fft_len)
    g = 2.0 * cross[:n]

    dot_ga = np.dot(g, a)
    return 2.0 * n * a * (g - dot_ga)


def lse_value(a, b, T):
    """Compute 2n*LSE_T(b)."""
    n = len(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled = np.clip(scaled, -700.0, 0.0)
    lse = max_b + T * np.log(np.exp(scaled).sum())
    return 2.0 * n * lse


def lse_lbgfs(a0, time_limit, T_stages=None):
    """Aggressive LogSumExp L-BFGS with FFT-accelerated gradients."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0
    if T_stages is None:
        T_stages = [30, 15, 8, 5, 3, 2, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12,
                    0.08, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]

    best_z = z.copy()
    best_val = float('inf')

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 8:
            break

        # Precompute FFT size
        fft_len = 1 << (3 * n - 1).bit_length()

        def func(z_in):
            z_sh = z_in - z_in.max()
            exp_z = np.exp(z_sh)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            return lse_value(a, b, T), None

        def jac(z_in):
            z_sh = z_in - z_in.max()
            exp_z = np.exp(z_sh)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            grad = lse_gradient(a, b, T)
            return grad

        def fun_and_jac(z_in):
            z_sh = z_in - z_in.max()
            exp_z = np.exp(z_sh)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            f_val = lse_value(a, b, T)
            grad = lse_gradient(a, b, T)
            return f_val, grad

        # Scale iterations by remaining time
        max_iter = max(30, int(remaining / 0.015))
        try:
            res = minimize(fun_and_jac, z, jac=True, method='L-BFGS-B',

                                    'gtol': 1e-14, 'maxfun': max_iter * 300})
            z = res.x
            z_sh = z - z.max()
            exp_z = np.exp(z_sh)
            a_check = exp_z / exp_z.sum()
            b_check = fast_convolve(a_check)
            v = lse_value(a_check, b_check, T)
            if v < best_val:

                best_z = z.copy()
        except Exception:
            pass

    # Use the best z found
    z = best_z
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def pair_swap_cd(a0, time_limit, n_steps=100000):
    """Coordinate descent using pair swaps to reduce peak convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        k_max = int(np.argmax(b))

        improved = False
        # Find top contributors to the max peak
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_max - idx]
        order = np.argsort(contribs)[::-1][:8]

        for oi in order:
            pos = idx[oi]
            if a[pos] < 1e-10:
                continue
            # Try reducing pos and redistributing
            for factor in [0.97, 0.95, 0.92, 0.88]:
                delta = a[pos] * (1.0 - factor)
                if delta < 1e-13:
                    break
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[pos] -= delta
                a_trial[mask] += delta * a[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Try pair swap: move weight from high-contributor to low-impact position
            if step % 10 == 0:
                # Find a "safe" target position (low contribution to all top peaks)
                top_peaks = np.argsort(b)[::-1][:5]
                target_scores = np.zeros(n)
                for pk in top_peaks:
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    idx2 = np.arange(js2, je2 + 1)
                    target_scores[idx2] += a[pk - idx2]
                target_scores /= len(top_peaks)

                # Find position with lowest target score (safe to add weight)
                safe_positions = np.argsort(target_scores)[:20]
                # Move from top contributor to safest position
                src = idx[order[0]] if len(order) > 0 else -1
                if src >= 0 and a[src] > 1e-10:
                    for tgt in safe_positions[:5]:
                        if tgt == src:
                            continue
                        for delta_frac in [0.02, 0.04, 0.06]:
                            delta = a[src] * delta_frac
                            if delta < 1e-13:
                                break
                            a_trial = a.copy()
                            a_trial[src] -= delta
                            a_trial[tgt] += delta
                            st = a_trial.sum()
                            a_trial = a_trial / st
                            sc = 2.0 * n * fast_convolve(a_trial).max()
                            if sc < best_score - 1e-12:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break
                        if improved:
                            break

            if not improved:
                stall += 1
                if stall > 200:
                    # Random perturbation to escape
                    pert = rng.standard_normal(n) * 0.01
                    a_trial = a + pert
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                    stall = 0
                    if stall > 500:
                        break

        if step % 20000 == 0:
            print(f"    PairCD {time.monotonic()-t0:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def direct_subgrad(a0, time_limit, n_steps=80000):
    """Direct subgradient method on the actual max-convolution objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.008
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: peaks within 0.2% of max
        threshold = max_b * 0.998
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: average of gradients over active peaks
        # d(conv[k])/da[i] = 2*a[k-i] (for valid k-i)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Nesterov-like step with momentum
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.93
                no_improve = 0
        lr = max(lr, 1e-9)

    return best_a, best_score


def peak_equalization(a0, time_limit, n_steps=50000):
    """Try to equalize the top K peaks by adjusting the sequence."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    K = 5
    lr = 0.003

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Get top K peaks
        top_idx = np.argsort(b)[::-1][:K]
        top_vals = b[top_idx]
        mean_top = top_vals.mean()

        # Gradient: push down peaks above mean, push up those below
        g = np.zeros(n)
        for ki, k in enumerate(top_idx):
            weight = (top_vals[ki] - mean_top) / max(max_b, 1e-15)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] -= weight * 2.0 * a[k - idx]

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            lr = min(lr * 1.005, 0.05)
        else:
            lr *= 0.97
            lr = max(lr, 1e-10)

    return best_a, best_score
