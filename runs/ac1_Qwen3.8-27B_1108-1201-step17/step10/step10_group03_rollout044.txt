import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def local_eval(a):
    """Fast local evaluation of the objective (normalized sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s != solver]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv > 1e-15:
                    return b_val / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def bidirectional_lp_alternation(a0, time_limit, solver, max_iters=500000):
    """Alternating minimization solving both LP subproblems with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Best b given current y_a
        b_new, M1 = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 3)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 3)) * b_new) / (it + 1 + alpha * (it + 3))
            b = normalize(b)

        # Step 2: Best a given current y_b
        a_new, M2 = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 4)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 4)) * a_new) / (it + 1 + alpha * (it + 4))
            a = normalize(a)

        # Track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

                # More aggressive noise injection
                noise_scale = 0.05 / math.sqrt(stall)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall = 0

            if (it + 1) % 1000 == 0:
                print(f"    BiDirLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

def peak_targeted_refinement(a0, solver, time_limit):
    """Targeted LP refinement focusing on the peak lags of the autocorrelation."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for outer in range(100):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Find peaks within 1% of max
        threshold = M * 0.99
        peak_indices = np.where(b >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(b))])

        # LP: minimize sum of b_k over peak indices, subject to all b_k <= M, sum(a)=1, a>=0
        a_var = cp.Variable(n, nonneg=True)
        T = build_conv_matrix(a)
        m = len(peak_indices)

        # Objective: minimize sum of linearized peak values
        # b_k = sum_j a[k-j] * a_var[j] for linearized version
        peak_rows = T[peak_indices, :]
        obj = cp.Minimize(cp.sum(peak_rows @ a_var))

        constraints = [
            cp.sum(a_var) == 1,
        ]
        # Constrain all convolution entries to not exceed current M
        # This is the key constraint - keep all b_k <= M
        # But this is nonlinear (a * a_var), so we use the linearized version
        # For the LP, we fix one side: minimize sum of T[peaks] @ a_var
        # subject to T @ a_var <= M (linear upper bound on all entries)
        constraints.append(T @ a_var <= M)

        prob = cp.Problem(obj, constraints)
        solvers_to_try = [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL] if s != solver]
        solved = False
        for s in solvers_to_try:
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    sv = a_val.sum()
                    if sv > 1e-15:
                        a_val = a_val / sv
                        # Blend with current
                        t_blend = 0.3
                        a_trial = (1 - t_blend) * a + t_blend * a_val
                        a_trial = normalize(a_trial)
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_trial.copy()
                        a = a_trial
                        solved = True
                        break
            except Exception:
                continue
        if not solved:
            # Try smaller step
            break

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on the LogSumExp-smoothed objective in log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    # Center z for numerical stability
    z = z - z.mean()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        # Allocate more iterations to lower T
        frac = min(1.0, 10.0 / remaining)
        max_iter = max(50, int(remaining * 30 * max(1.0, 5.0 / T)))

        def obj_and_grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            lse = b_max + np.log(exp_ws.sum())
            obj_val = (2.0 * n) * lse

            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]

            # Chain rule for softmax: grad_z = a * (grad_a - sum(grad_a * a))
            S = np.dot(grad_a, a_cur)
            grad_z = (2.0 * n) * a_cur * (grad_a - S)
            return obj_val, grad_z

        def obj_fn(z):
            return obj_and_grad(z)[0]

        def grad_fn(z):
            return obj_and_grad(z)[1]

        try:
            res = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=50000):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for rd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a random coordinate
        j = rng.integers(0, n)
        old_val = a[j]

        # Compute convolution without the j-th term contribution
        # b_k = sum_i a_i * a_{k-i}
        # The contribution of a_j is: for each k, a_j * a_{k-j} (if k-j in range)
        # So b_k = b_k_rest + a_j * a[k-j] (for valid k-j)
        # We want to minimize max_k (b_k_rest + x * a[k-j]) over x >= 0
        # where b_k_rest is the convolution of a with a[j] removed

        # Efficient: b_k = sum_i a_i * a_{k-i}
        # Remove a_j: b'_k = b_k - a_j * a[k-j] (for valid k-j)
        # Then new b_k = b'_k + x * a[k-j]
        # We want to choose x to minimize max_k (b'_k + x * a[k-j])

        # The terms a[k-j] for valid k are a[0], a[1], ..., a[n-1] (shifted)
        # Specifically, for k=j: a[k-j] = a[0]
        # For k=j+1: a[k-j] = a[1]
        # ...
        # For k=j+n-1: a[k-j] = a[n-1]
        # For other k, a[k-j] = 0

        # So the new convolution is: b'_k + x * a[k-j] for k in {j, j+1, ..., j+n-1}
        # And b'_k for other k.

        # To find optimal x: we want to minimize max over all k of (b_k_new)
        # This is a 1D optimization. The max is the upper envelope of lines in x.
        # The lines with positive slope are b'_k + x * a[k-j] for k in {j..j+n-1}
        # The lines with zero slope are b'_k for other k.
        # The maximum of zero-slope lines is M0 = max of b'_k for k not in {j..j+n-1}
        # The maximum of positive-slope lines is M1(x) = max_k (b'_k + x * a[k-j])
        # We want min_x max(M0, M1(x))

        # M1(x) is convex in x (max of linear functions with non-negative slopes)
        # M0 is constant.
        # Optimal x is where M1(x) = M0 (if possible) or at the minimum of M1(x) clamped at x=0.

        # M1(x) at x=0: M1(0) = max of b'_k for k in {j..j+n-1}
        # If M1(0) <= M0, then x=0 is optimal (or we can increase x slightly)
        # Otherwise, we need to find x where M1(x) = M0.

        # For simplicity, let's just do a few evaluations:
        # x=0, x=old_val, and a few in between.

        b = fast_convolve(a)
        M = b.max()

        # Try x = 0
        a[j] = 0
        s = a.sum()
        if s > 1e-15:
            a_norm = a / s
            b_trial = fast_convolve(a_norm)
            sc_trial = 2.0 * n * b_trial.max()
        else:
            sc_trial = float('inf')
        a[j] = old_val

        # Try x = 2 * old_val
        a[j] = 2.0 * old_val
        s = a.sum()
        if s > 1e-15:
            a_norm = a / s
            b_trial = fast_convolve(a_norm)
            sc_trial2 = 2.0 * n * b_trial.max()
        else:
            sc_trial2 = float('inf')
        a[j] = old_val

        # Try x = 0.5 * old_val
        a[j] = 0.5 * old_val
        s = a.sum()
        if s > 1e-15:
            a_norm = a / s
            b_trial = fast_convolve(a_norm)
            sc_trial3 = 2.0 * n * b_trial.max()
        else:
            sc_trial3 = float('inf')
        a[j] = old_val

        # Pick best
        candidates = [(0.0, sc_trial), (old_val, local_eval(a)), (2.0 * old_val, sc_trial2), (0.5 * old_val, sc_trial3)]
        candidates.sort(key=lambda x: x[1])
        best_x, best_sc = candidates[0]

        if best_sc < best_score - 1e-14:
            a[j] = best_x
            a = normalize(a)
            best_score = best_sc
            best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    elif choice == 8:
        a = np.sin(np.pi * x) * (0.5 + 0.5 * np.cos(2 * np.pi * 2 * x))
        a = np.maximum(a, 0)
    elif choice == 9:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 10:
        a = 0.3 + 0.7 * np.cos(np.pi * x) ** 4
    else:
        a = np.ones(n) * (1.0 + 0.05 * rng.standard_normal(n))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Focus: LP alternation (main), LogSumExp L-BFGS (polish), peak targeting,
    coordinate descent, multi-n search.
    """
    import math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}, budget: {budget_s:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Bidirectional LP alternation (main workhorse) ===
    print(f"\n=== Phase 1: BiDir LP (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = bidirectional_lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "bidi_lp")

    # === Phase 2: Peak-targeted LP refinement ===
    print(f"\n=== Phase 2: Peak Targeting (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 20:
        a_pt, sc_pt = peak_targeted_refinement(best_seq, solver, t2)
        try_update(a_pt, "peak_target")

    # === Phase 3: LogSumExp L-BFGS polishing ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 350.0)
    if t3 > 30:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  Budget: {t3:.0f}s")
        a_lb, sc_lb = logsumexp_lbgfs(best_seq, t3, T_stages=T_stages)
        try_update(a_lb, "lbgfs")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 100.0)
    if t4 > 10:
        a_cd, sc_cd = coordinate_descent(best_seq, t4, n_rounds=50000)
        try_update(a_cd, "coord_desc")

    # === Phase 5: Second LP round ===
    print(f"\n=== Phase 5: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if t5 > 30:
        a_lp2, sc_lp2 = bidirectional_lp_alternation(best_seq, t5, solver)
        try_update(a_lp2, "bidi_lp_2")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_candidates = []
        base_n = n_best
        for mult in [0.7, 0.8, 0.9, 1.1, 1.2, 1.5, 2.0, 3.0]:
            cand_n = int(base_n * mult)
            if 200 <= cand_n <= 8000 and cand_n not in n_candidates:
                n_candidates.append(cand_n)
        # Also try some absolute values
        for abs_n in [500, 800, 1500, 2400, 3600, 5000]:
            if abs_n not in n_candidates and abs_n != n_best:
                n_candidates.append(abs_n)

        n_candidates = n_candidates[:8]  # Limit to 8 candidates

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.25, 120.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # LP
            a_cand, _ = bidirectional_lp_alternation(init_cand, t_sub * 0.5, solver)
            try_update(a_cand, f"lp_n{cand_n}")
            # L-BFGS
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cand2, _ = logsumexp_lbgfs(a_cand, min(rem * 0.4, 60.0),
                                             T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02])
                try_update(a_cand2, f"lbgfs_n{cand_n}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very low T L-BFGS
        t_fin = min(rem * 0.5, 80.0)
        if t_fin > 15:
            a_fin, _ = logsumexp_lbgfs(best_seq, t_fin,
                                       T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
            try_update(a_fin, "final_lbgfs")

        # Quick coordinate descent
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin_cd, _ = coordinate_descent(best_seq, min(rem * 0.3, 30.0), n_rounds=20000)
            try_update(a_fin_cd, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
