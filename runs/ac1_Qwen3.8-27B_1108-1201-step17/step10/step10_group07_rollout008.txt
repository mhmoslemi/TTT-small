def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).

    T = build_conv_matrix(a)

    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

                # a_old[k-i] = a[k-i] - (delta if k-i == i else 0)
                a_old = a - best_delta
                a_shifted_old = a_old[k_start - i:k_end - i + 1]

                if k_start > 0:
                    unaff_left = conv[:k_start].max()
                else:
                    unaff_left = -1e30
                if k_end < 2 * n - 2:
                    unaff_right = conv[k_end + 1:].max()
                else:
                    unaff_right = -1e30
                unaff_max = max(unaff_left, unaff_right)

                new_affected = conv[k_start:k_end + 1] + 2.0 * best_delta * a_shifted_old
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected[idx] = conv[2 * i] + 2.0 * a_old[i] * best_delta + best_delta * best_delta

                conv[k_start:k_end + 1] = new_affected
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved or M > sweep_best_before * (1 + 1e-10):
            no_improve_sweeps += 1
            if no_improve_sweeps > 50:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExactCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def fast_coord_descent(a0, time_limit, verbose=True, label=""):
    """Faster coordinate descent with incremental updates and adaptive factors."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_M = M

            # Adaptive factors
            base = max(a[i], 1e-8)
            factors = [-0.8, -0.5, -0.3, -0.1, -0.03, 0.03, 0.1, 0.3, 0.5, 0.8]

            for factor in factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 40:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FastCD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, verbose=True):
    """
    Focus on reducing the top peaks by redistributing mass from
    high-contributing coordinates to low-contributing ones.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Find top peaks
        K = 8
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.97:
                continue

            # Find contributing coordinates
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # Contribution of coordinate i to conv[k_star]: 2*a[i]*a[k_star-i] (or a[i]^2 if i == k_star-i)
            contributions = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = k_star - i
                if i == j:
                    contributions[i] = a[i] * a[i]
                else:
                    contributions[i] = 2.0 * a[i] * a[j]

            # Find the coordinate with highest contribution that we can reduce
            top_contrib = np.argsort(contributions)[::-1][:5]

            for target in top_contrib:
                if contributions[target] < 1e-12:
                    continue

                # Try reducing target and see if max decreases
                for frac in [0.15, 0.25, 0.4, 0.6]:
                    delta = -a[target] * frac
                    if delta == 0:
                        continue

                    k_s = target
                    k_e = min(target + n - 1, 2 * n - 2)
                    a_shifted = a[k_s - target:k_e - target + 1]

                    new_affected = conv[k_s:k_e + 1] + 2.0 * delta * a_shifted
                    if 2 * target >= k_s and 2 * target <= k_e:
                        idx = 2 * target - k_s
                        new_affected = new_affected.copy()
                        new_affected[idx] = conv[2 * target] + 2.0 * delta * a[target] + delta * delta

                    if k_s > 0:
                        unaff_left = conv[:k_s].max()
                    else:
                        unaff_left = -1e30
                    if k_e < 2 * n - 2:
                        unaff_right = conv[k_e + 1:].max()
                    else:
                        unaff_right = -1e30
                    unaff_max = max(unaff_left, unaff_right)

                    new_M = max(unaff_max, float(new_affected.max()))
                    if new_M < M - 1e-12:
                        a[target] = max(a[target] + delta, 0.0)
                        conv[k_s:k_e + 1] = new_affected
                        M = new_M
                        improved = True
                        cur_score = 2.0 * n * M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                        break

        if not improved and rounds > 100:
            break

        if verbose and rounds % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave round={rounds} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    # Renormalize
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
        best_score = 2.0 * n * fast_convolve(best_a).max()

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    """
    LP-based best-response alternating minimization with geometric mean updates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        # Solve LP: best response b* to current a
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Evaluate b_new
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Update strategy: use geometric mean for better stability
        if it <= 10:
            alpha = 0.8
        elif it <= 30:
            alpha = 0.6
        elif it <= 60:
            alpha = 0.4
        else:
            alpha = max(0.25, 8.0 / (it + 5.0))

        # Geometric mean update (with epsilon to avoid log(0))
        eps = 1e-15
        log_a = np.log(a + eps)
        log_b = np.log(b_new + eps)
        log_new = (1 - alpha) * log_a + alpha * log_b
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        # Evaluate updated a
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation detection
        if score_a > best_score * (1 + 5e-6):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 2)

        # Random perturbation if stuck
        if stagnation > 40:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a_pert = a_pert / s_pert
                score_pert = 2.0 * n * fast_convolve(a_pert).max()
                if score_pert < best_score * 1.005:
                    a = a_pert
            stagnation = 0

        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPAlt it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f} alpha={alpha:.3f}")

    return best_a, best_score


def lp_alternating_arith(a0, time_limit, solver, verbose=True):
    """LP alternating with arithmetic mean updates (alternative)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        if it <= 15:
            alpha = 0.7
        elif it <= 40:
            alpha = 0.5
        else:
            alpha = max(0.3, 12.0 / (it + 10.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s_new = a.sum()
        a = a / s_new

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if score_a > best_score * (1 + 5e-6):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 2)

        if stagnation > 35:
            noise_scale = 0.015 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a_pert = a_pert / s_pert
                score_pert = 2.0 * n * fast_convolve(a_pert).max()
                if score_pert < best_score * 1.003:
                    a = a_pert
            stagnation = 0

        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPAlt(arith) it={it} ({elapsed:.0f}s): best={best_score:.8f} alpha={alpha:.3f}")

    return best_a, best_score


def subgradient_descent(a0, time_limit, lr_init=0.015, top_k=25, verbose=True):
    """Subgradient descent with multi-peak weighting and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-12:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]

        temp = 0.004 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[i_end - (k - i_start):i_end + 1][::-1]

        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.2 + 0.8 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.5, 4.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.18)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(4, 12))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        t = np.arange(n)
        a = 1.0 - 0.4 * np.cos(6 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Sparse comb-like pattern
        a = np.zeros(n)
        spacing = int(rng.integers(3, 7))
        for i in range(0, n, spacing):
            a[i] = rng.uniform(0.5, 1.0)
    elif init_type == 7:
        # Random smooth
        freqs = rng.uniform(0, 5, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        t = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * t + p)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Very fine multiplicative steps
            factors = [0.997, 0.995, 0.998, 0.999, 1.001, 1.003, 1.006, 1.002]
            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    # === Phase 1b: LP Alternating - Arithmetic (alternative) ===
    print(f"\n=== Phase 1b: LP Alt Arithmetic (best={best_score:.8f}) ===")
    t1b_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_alloc = min(t1b_end - time.monotonic(), 300.0)
    if t_alloc > 30:
        print(f"  Budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_alternating_arith(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_arith")

    # LP from diverse inits
    for init_type in [0, 2, 6]:
        if time.monotonic() > t1b_end:
            break
        remaining = t1b_end - time.monotonic()
        if remaining < 30:
            break
        init = diverse_init(n, rng, init_type)
        t_alloc = min(remaining * 0.35, 100.0)
        print(f"  LP from init{init_type}, budget={t_alloc:.0f}s")
        a_tmp, s_tmp = lp_alternating(init, t_alloc, solver, verbose=False)
        try_update(a_tmp, f"lp_init{init_type}")

    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.22

    t_alloc = min(t2_end - time.monotonic(), 250.0)
    if t_alloc > 30:
        print(f"  FastCD budget={t_alloc:.0f}s")
        a_cd, s_cd = fast_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "cd_main")

        if time.monotonic() < t2_end - 60:
            t_alloc2 = min(t2_end - time.monotonic(), 120.0)
            a_cd2, s_cd2 = fast_coord_descent(a_lp1, t_alloc2, verbose=True, label="lp")
            try_update(a_cd2, "cd_lp")

    # === Phase 3: Exact Coordinate Descent ===
    print(f"\n=== Phase 3: Exact CD (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.12

    t_alloc = min(t3_end - time.monotonic(), 150.0)
    if t_alloc > 30:
        print(f"  ExactCD budget={t_alloc:.0f}s")
        a_etc, s_etc = exact_coordinate_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_etc, "exact_cd")

    # === Phase 4: Peak Shaving + Fine Polish ===
    print(f"\n=== Phase 4: Peak Shave + Fine Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 30

    if time.monotonic() < t4_end - 80:
        t_ps = min(t4_end - time.monotonic() - 80, 100.0)
        if t_ps > 20:
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_ps, s_ps = peak_shaving(best_seq, t_ps, verbose=True)
            try_update(a_ps, "peak_shave")

    if time.monotonic() < t4_end - 50:
        t_fp = min(t4_end - time.monotonic() - 50, 80.0)
        if t_fp > 15:
            print(f"  FinePolish budget={t_fp:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
            try_update(a_fp, "fine_polish")

    # Final passes
    if time.monotonic() < t4_end - 20:
        t_final = min(t4_end - time.monotonic() - 20, 40.0)
        if t_final > 10:
            a_fin, s_fin = fast_coord_descent(best_seq, t_final, verbose=False, label="final")
            try_update(a_fin, "final_cd")

    if time.monotonic() < t4_end - 5:
        t_fp2 = min(t4_end - time.monotonic() - 5, 20.0)
        if t_fp2 > 5:
            a_fp2, s_fp2 = fine_polish(best_seq, t_fp2, verbose=False)
            try_update(a_fp2, "final_fine")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
