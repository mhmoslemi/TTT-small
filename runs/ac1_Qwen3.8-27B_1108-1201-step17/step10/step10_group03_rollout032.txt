import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    return None

def bidirectional_lp_alternation(a0, time_limit, solver, max_iters=100000):
    """Alternating minimization solving both LP subproblems with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        a_new = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 15:
                noise_scale = 0.03
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall = 0

            if (it + 1) % 500 == 0:
                print(f"    BiDirLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score

def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    """L-BFGS on the LogSumExp-smoothed objective in the simplex."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    z -= z.max()  # for numerical stability
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 25))

        def lse_obj(z_cur):
            z_max = z_cur.max()
            exp_z = np.exp(z_cur - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def lse_grad(z_cur):
            z_max = z_cur.max()
            exp_z = np.exp(z_cur - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                if T <= 1.0:
                    print(f"    LSE L-BFGS T={T}: score={sc:.10f}")
        except Exception:
            pass

    return best_a, best_score

def adam_logsumexp(a0, time_limit, T_stages=None, lr_base=0.001):
    """Adam optimizer on LogSumExp-smoothed objective with softmax parameterization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    z -= z.max()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        lr = lr_base
        
        while time.monotonic() - t0 < time_limit and (time.monotonic() - t0) < remaining:
            t_adam += 1
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - S)
            
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic evaluation
            if t_adam % 100 == 0:
                a_new = np.exp(z - z.max())
                a_new = a_new / a_new.sum()
                sc = 2.0 * n * fast_convolve(a_new).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                # Decay learning rate
                if t_adam % 1000 == 0:
                    lr *= 0.95

    return best_a, best_score

def subgradient_method(a0, time_limit, n_steps=50000):
    """Subgradient method with adaptive step size and Nesterov momentum."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03
    v = np.zeros(n)  # momentum
    momentum = 0.9
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        v = momentum * v - lr * g
        a_trial = a + v
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
            v = np.zeros(n)
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.95
                momentum = max(0.5, momentum * 0.99)
            if no_improve > 500:
                # Random restart in neighborhood
                a_trial = a + np.random.default_rng(step).standard_normal(n) * 0.02
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                a = a_trial
                v = np.zeros(n)
                no_improve = 0
                lr = 0.03
                momentum = 0.9

        lr = max(1e-8, lr)

    return best_a, best_score

def stochastic_perturbation(a0, time_limit, n_perturb=50):

def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 1:
        a = np.sin(np.pi * x) ** 2
    elif mode == 2:
        a = np.exp(-4 * x * (1 - x))
    elif mode == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif mode == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif mode == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif mode == 6:

        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

def random_init(n, rng):
    """Random initial sequence on the simplex."""
    a = rng.exponential(1.0, size=n)
    return a / a.sum()

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    return None


def bidirectional_lp_alternation(a0, time_limit, solver, max_iters=100000):
    """Alternating minimization with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(54321)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        a_new = solve_lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-11:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 12:
                noise_scale = 0.03 / math.sqrt(stall)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall = 0

            if (it + 1) % 500 == 0:
                print(f"    BiDirLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def compute_lse_grad_z(z, T):
    """Compute gradient of LogSumExp objective w.r.t. z (log-softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a_cur = exp_z / exp_z.sum()
    n = len(a_cur)
    b = fast_convolve(a_cur)
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()

    # grad_a[i] = sum_k 2*w[k]*a[k-i] for valid k-i
    # Use FFT: this is cross-correlation of (2w) and a
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    tw = 2.0 * w
    a_flip = a_cur[::-1]
    W = rfft(tw, fft_size)
    AF = rfft(a_flip, fft_size)
    corr = irfft(W * AF, fft_size)[:size]
    # grad_a[i] = corr[n-1+i]
    grad_a = corr[n - 1:n - 1 + n]

    S = np.dot(grad_a, a_cur)
    grad_z = a_cur * (grad_a - S)
    return grad_z


def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with FFT-accelerated gradients."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    z -= z.max()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                    0.01, 0.005, 0.002, 0.001]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        # Allocate more iterations for lower T
        if T >= 5.0:
            max_iter = max(50, int(remaining * 30))
        elif T >= 1.0:
            max_iter = max(80, int(remaining * 40))
        else:
            max_iter = max(100, int(remaining * 60))

        def lse_obj(z_cur):
            z_max = z_cur.max()
            exp_z = np.exp(z_cur - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def lse_grad(z_cur):
            return compute_lse_grad_z(z_cur, T)

        try:
            res = minimize(
                lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-16,
                         'maxfun': max_iter * 1000, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"    LSE T={T}: score={sc:.10f}")
        except Exception:
            pass

    return best_a, best_score


def adam_logsumexp(a0, time_limit, T_stages=None, lr_base=0.001):
    """Adam optimizer on LogSumExp-smoothed objective."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    z -= z.max()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        t_adam = 0
        lr = lr_base

        while time.monotonic() - t0 < time_limit - 2:
            t_adam += 1
            grad_z = compute_lse_grad_z(z, T)

            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if t_adam % 200 == 0:
                s = np.exp(z - z.max())
                a_new = s / s.sum()
                sc = 2.0 * n * fast_convolve(a_new).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                if t_adam % 2000 == 0:
                    lr *= 0.9

    return best_a, best_score


def subgradient_method(a0, time_limit, n_steps=50000):
    """Subgradient with Nesterov momentum and adaptive restart."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.04
    v = np.zeros(n)
    momentum = 0.85
    no_improve = 0
    rng = np.random.default_rng(99)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient via FFT
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        grad_z = compute_lse_grad_z(np.log(np.maximum(a, 1e-300)), 1.0)
        # Actually let's just compute directly for the exact max case
        grad_a = np.zeros(n)
        fft_size = 1 << (2 * n - 2).bit_length() + 1
        fft_size = 1 << (2 * n - 1).bit_length()
        wt = 2.0 * w
        a_flip = a[::-1]
        W = rfft(wt, fft_size)
        AF = rfft(a_flip, fft_size)
        corr = irfft(W * AF, fft_size)[:2 * n - 1]
        grad_a = corr[n - 1:n - 1 + n]

        v = momentum * v - lr * grad_a
        a_trial = a + v
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
            v = np.zeros(n)
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.97
            if no_improve > 300:
                # Structured restart
                noise = rng.standard_normal(n) * 0.03
                a_pert = a + noise
                a = normalize(a_pert)
                v = np.zeros(n)
                no_improve = 0
                lr = 0.04
                momentum = 0.85

        lr = max(1e-8, lr)

    return best_a, best_score


def stochastic_perturbation(a0, time_limit, n_perturb=40):
    """Iteratively perturb and locally optimize with subgradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(31415)

    for i in range(n_perturb):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive perturbation scale
        scale = 0.04 * (1.0 + 0.5 * math.sin(i * 0.3))
        a_pert = a + rng.standard_normal(n) * scale
        a_pert = normalize(a_pert)

        # Quick local optimization
        t_sub = min(15.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        if t_sub < 3:
            break
        a_opt, sc_opt = subgradient_method(a_pert, t_sub, n_steps=3000)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  StochPert {i}: new best {sc_opt:.10f}")

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create diverse structured initial sequences (asymmetric)."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.cos(np.pi * x / 2) ** 2 + 0.05
    elif mode == 1:
        a = np.sin(np.pi * x) ** 2 + 0.05
    elif mode == 2:
        a = np.exp(-3 * x * (1 - x)) + 0.1
    elif mode == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(4 * np.pi * 5 * x)
        a = np.maximum(a, 0.01)
    elif mode == 4:
        a = (1 - x) ** 2 + 0.05 * x + 0.05
    elif mode == 5:
        a = np.exp(-4 * (x - 0.35) ** 2) + 0.4 * np.exp(-4 * (x - 0.75) ** 2) + 0.05
    elif mode == 6:
        a = 1.0 + 0.25 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x) + 0.05 * np.cos(6 * np.pi * x)
    elif mode == 7:
        # Asymmetric ramp with oscillation
        a = (0.3 + 0.7 * x) * (1.0 + 0.2 * np.cos(2 * np.pi * 7 * x)) + 0.05
    elif mode == 8:
        # Bimodal asymmetric
        a = np.exp(-8 * (x - 0.2) ** 2) + 0.6 * np.exp(-8 * (x - 0.8) ** 2) + 0.05
    else:
        a = np.ones(n) * 0.1
    a = np.maximum(a, 0.001)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random initial sequence on the simplex (exponential distribution)."""
    a = rng.exponential(1.0, size=n)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def quick_optimize(a_init, time_budget, solver, n_subgrad=5000):
    """Quick optimization: few LP iterations + subgradient."""
    best_a = normalize(a_init)
    best_score = 2.0 * len(best_a) * fast_convolve(best_a).max()

    # Quick LP (limited time)
    t_lp = min(time_budget * 0.4, 20.0)
    if t_lp > 3:
        a_lp, sc_lp = bidirectional_lp_alternation(best_a, t_lp, solver)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp

    # Subgradient
    rem = time_budget - (time_budget - t_lp) if t_lp < time_budget else 0
    rem = time_budget - min(t_lp, time_budget)
    if rem > 2:
        t_sg = min(rem * 0.8, 15.0)
        a_sg, sc_sg = subgradient_method(best_a, t_sg, n_steps=n_subgrad)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg

    return best_a, best_score

            parent = p / p.sum()

    n_best = len(best_seq)

    # --- Phase 1: Multi-length diverse initialization (120s) ---
    print(f"\n=== Phase 1: Multi-length Init (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.10, 150.0)
    if t_phase1 > 20:

                    print(f"  NEW BEST [interp_n{cand_n}]: {sc_cand:.10f}")

        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
