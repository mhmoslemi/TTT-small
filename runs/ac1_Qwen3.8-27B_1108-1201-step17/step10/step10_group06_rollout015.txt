"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Fictitious Play with dual-guided adaptive steps
2. Gerchberg-Saxton frequency domain spectral flattening
3. Exact coordinate descent with golden section search
4. Multiplicative weights (mirror descent) refinement

6. Multi-length exploration around parent length

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

_LP_SOLVER = None
_LP_SOLVER_CHECKED = False


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)


def get_lp_solver():
    global _LP_SOLVER, _LP_SOLVER_CHECKED
    if _LP_SOLVER_CHECKED:
        return _LP_SOLVER
    _LP_SOLVER_CHECKED = True
    import cvxpy as cp
    for name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return _LP_SOLVER


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    dual_vals = None
    try:
        dual_vals = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            try:
                dual_vals[k] = constraints[k].dual_value
            except Exception:
                pass
    except Exception:
        pass
    return b_val, M_val, dual_vals


def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners].copy()
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            def score_func(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                           _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                           _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(5.0 * v, 10.0 / n)
            best_x = golden_section_search(score_func, 0.0, upper, max_iter=28)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_a, best_score, sweeps


def freq_domain_refine(a0, time_limit, verbose=False):
    """Gerchberg-Saxton inspired spectral flattening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    alpha = 0.85
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        iters += 1

        conv = fast_convolve(a)
        max_val = float(conv.max())
        mean_val = 1.0 / (2 * n - 1)

        if max_val < 1.01 * mean_val:
            break

        N = 1 << (2 * n - 1).bit_length()
        P = fft(a, N)

        magnitude = np.abs(P)
        phase = np.angle(P)

        target_mag = np.full(N, 1.0 / np.sqrt(N))
        compression = alpha
        new_mag = (magnitude ** compression) * (target_mag / (magnitude ** compression).max())
        new_mag = np.maximum(new_mag, 1e-10)

        new_P = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(new_P, N))[:n]

        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = a
        else:
            a_new = a_new / ss

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            blend = 0.3
            a = (1.0 - blend) * a + blend * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve += 1

        if no_improve > 20:
            alpha = max(0.5, alpha * 0.95)
            no_improve = 0
            perturb = a + np.random.standard_normal(n) * 0.01
            perturb = np.maximum(perturb, 0)
            perturb = perturb / perturb.sum()
            sc_p = compute_score(perturb)
            if sc_p < best_score:
                best_score = sc_p
                best_a = perturb.copy()
            a = perturb

    return best_a, best_score


def multiplicative_weights(a0, time_limit, verbose=False):
    """Mirror descent / multiplicative weights for non-negative variables."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.1
    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())

        top_k = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        temp = 0.005 * max_val
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        grad_mean = grad.mean()
        grad_centered = grad - grad_mean
        log_weights = -lr * grad_centered
        log_weights -= log_weights.max()
        new_weights = np.exp(log_weights)
        ws2 = new_weights.sum()
        if ws2 < 1e-15:
            break
        a_new = a * new_weights / ws2
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = a
        else:
            a_new = a_new / ss

        sc = compute_score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
                a_new = 0.7 * a + 0.3 * best_a
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()

        a = a_new

        if no_improve > 200:
            break

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_asymmetric(n, rng, shift=0.3):

    center = n * shift
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners].copy()
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            def score_fn(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                         _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                         _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(5.0 * v, 10.0 / n)
            best_x = golden_section_search(score_fn, 0.0, upper, max_iter=25)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0

    return best_a, best_score
