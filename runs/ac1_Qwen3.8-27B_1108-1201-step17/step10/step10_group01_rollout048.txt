**Cleaner approach: Projected gradient with line search**

$$\partial f(a) = \text{conv}\{\nabla (a*a)[k] : k \in \mathcal{A}(a)\}$$

        A_ub = np.hstack([T, -np.eye(len(active))])
        b_ub = np.zeros(len(active))
        A_eq = np.concatenate([np.ones(n), [0.0]])
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq.reshape(1, -1), b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                damping = max(damping * 0.5, 0.01)
                no_improve_count += 1
                continue

            a_lp = res.x[:n]
            a_lp = np.maximum(a_lp, 0)
            s_lp = a_lp.sum()
            if s_lp < 1e-15:
                continue
            a_lp = a_lp / s_lp
        except Exception:
            damping = max(damping * 0.5, 0.01)
            no_improve_count += 1
            continue

        # Try damped step
        accepted = False
        for damp in [damping, 0.5 * damping, 0.25 * damping, 0.1 * damping]:
            a_trial = (1.0 - damp) * a + damp * a_lp
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve_count = 0
                damping = min(damping * 1.1, 1.0)
                accepted = True
                # Tighten active set as we improve
                if best_score < 1.504:
                    active_frac = max(active_frac * 0.95, 0.0001)
                break

        if not accepted:
            no_improve_count += 1
            damping = max(damping * 0.7, 0.01)
            if no_improve_count > 50:
                # Reset with small perturbation
                rng_local = np.random.default_rng(hash(it) % 2**32)
                a = a + 0.001 * rng_local.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                damping = 0.3
                no_improve_count = 0

        if (it + 1) % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    SLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} "
                  f"|A|={len(active)} damp={damping:.3f} no_imp={no_improve_count}")

        # Early termination if very close
        if best_score < 1.5026 and no_improve_count > 20:
            break

    return best_a, best_score


def lp_best_response(a_fixed, method='highs'):
    """Solve the LP: min max_k (a_fixed * b)[k] s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    c_obj = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([T, -np.eye(size)])
    b_ub = np.zeros(size)
    A_eq = np.concatenate([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq.reshape(1, -1), b_eq=b_eq,
                      bounds=bounds, method=method)
        if res.success:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def nesterov_lp(a0, time_limit, max_iters=100000, greedy_frac=0.3,
                stall_patience=15, print_interval=2000, method='highs'):
    """
    Accelerated LP alternation with greedy component.
    Combines Nesterov averaging with periodic greedy updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = lp_best_response(y, method=method)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Primal iterated average
        a = ((it + 1) * a + (alpha * (it + 2.0)) * b_new) / (it + 1 + alpha * (it + 2.0))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Greedy step: try using b_new directly (or damped)
        if (it + 1) % int(1.0 / greedy_frac) == 0:
            for gy in [1.0, 0.7, 0.5]:
                a_greedy = (1.0 - gy) * a + gy * b_new
                a_greedy = np.maximum(a_greedy, 0)
                sg = a_greedy.sum()
                if sg < 1e-15:
                    continue
                a_greedy = a_greedy / sg
                sc = 2.0 * n * fast_convolve(a_greedy).max()
                if sc < best_score - 1e-14:
                    a = a_greedy
                    best_score = sc
                    best_a = a.copy()
                    break

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise_scale = 0.01 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                stall = 0

            if (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): "
                      f"score={best_score:.8f} M_LP={M_val:.8f}")

    return best_a, best_score


def coordinate_peak_shave(a0, time_limit, n_steps=30000, top_m=5):
    """
    Coordinate descent that targets the top peaks individually.
    For each top peak, find the single coordinate to reduce most.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:top_m]

        improved = False
        for pk in top_peaks:
            # Find which coordinate contributes most to this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_contribs = np.argsort(contribs)[::-1][:3]

            for ci in top_contribs:
                pos = idx[ci]
                if a[pos] < 1e-14:
                    continue
                for factor in [0.98, 0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        break
                    # Redistribute to other coordinates proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def subgradient_momentum(a0, time_limit, n_steps=50000, top_k=20):
    """
    Subgradient method with momentum targeting top-k peaks.
    Uses the average subgradient over the active set.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    v = np.zeros(n)  # Momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    beta = 0.9
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: top-k peaks
        active = np.argsort(b)[::-1][:top_k]

        # Average subgradient
        g = np.zeros(n)
        for k in active:
            g += peak_grad_vec(k, a)
        g /= len(active)

        # Project gradient to sum=0 hyperplane
        g -= g.mean()

        # Momentum update
        v = beta * v + (1 - beta) * g
        a_trial = a - lr * v
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            lr *= 0.85
            beta = min(beta + 0.005, 0.99)
            no_improve = 0
        lr = max(lr, 1e-6)

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradMom {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization pipeline:
    1. LP alternation (greedy + Nesterov)
    2. SLP peak-balancing refinement
    3. Multi-length exploration
    4. Final SLP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    if best_seq is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LP Alternation (greedy + Nesterov) ===
    print(f"\n=== Phase 1: LP Alternation n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 450.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t1, max_iters=200000,
                              greedy_frac=0.3, stall_patience=12,
                              print_interval=3000, method='highs')
        try_update(a_lp, "nlp_phase1")

    # === Phase 2: SLP Refinement ===
    print(f"\n=== Phase 2: SLP Refinement, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_slp, _ = slp_refine(best_seq, t2, max_iter=50000,
                              active_frac=0.002, max_active=200,
                              print_interval=500)
        try_update(a_slp, "slp_phase2")

    # === Phase 3: Subgradient with momentum ===
    print(f"\n=== Phase 3: SubGrad+Momentum, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 120.0)
    if t3 > 10:
        a_sg, _ = subgradient_momentum(best_seq, t3, n_steps=50000, top_k=15)
        try_update(a_sg, "subgrad_mom")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_cur = len(best_seq)
        alt_lengths = []
        # Try nearby and larger lengths
        for factor in [0.8, 1.2, 1.5, 2.0]:
            alt_n = int(n_cur * factor)
            if alt_n != n_cur and 200 <= alt_n <= 5000:
                alt_lengths.append(alt_n)
        # Remove duplicates and sort
        alt_lengths = sorted(set(alt_lengths))

        t_each = min(rem / max(1, len(alt_lengths)) * 0.7, 100.0)
        for alt_n in alt_lengths[:3]:  # Try at most 3
            if time.monotonic() > deadline - 60:
                break
            if t_each < 25:
                break
            print(f"  Trying n={alt_n}, t={t_each:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)

            # Quick LP + SLP
            t_lp = t_each * 0.4
            t_slp = t_each * 0.5
            a_alt, _ = nesterov_lp(init_alt, t_lp, max_iters=50000,
                                   greedy_frac=0.4, stall_patience=8,
                                   method='highs')
            try_update(a_alt, f"lp_n{alt_n}")
            if time.monotonic() < deadline - 30:
                a_alt2, _ = slp_refine(best_seq if len(best_seq) == alt_n else a_alt,
                                       t_slp, max_iter=20000,
                                       active_frac=0.002, max_active=150)
                try_update(a_alt2, f"slp_n{alt_n}")

    # === Phase 5: Final SLP polish on best ===
    print(f"\n=== Phase 5: Final SLP Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t5 = min(rem * 0.6, 200.0)
        print(f"  Budget: {t5:.0f}s")
        a_fin, _ = slp_refine(best_seq, t5, max_iter=50000,
                              active_frac=0.001, max_active=100,
                              print_interval=500)
        try_update(a_fin, "final_slp")

    # === Phase 6: Coordinate peak shaving ===
    print(f"\n=== Phase 6: Peak Shaving, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t6 = min(rem * 0.5, 60.0)
        a_cs, _ = coordinate_peak_shave(best_seq, t6, n_steps=20000, top_m=4)
        try_update(a_cs, "peak_shave")

    # === Phase 7: One more SLP with tighter active set ===
    print(f"\n=== Phase 7: Tight SLP, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t7 = min(rem * 0.7, 40.0)
        a_tight, _ = slp_refine(best_seq, t7, max_iter=20000,
                                active_frac=0.0005, max_active=50,
                                print_interval=200)
        try_update(a_tight, "tight_slp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
