from scipy.fft import rfft, irfft

import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_of_normalized(a):
    """Compute 2*n*max(conv) for normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())

    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):

    return T

            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def exact_coord_descent_step(a, conv, M, i):
    """
    Find the optimal value of coordinate i to minimize max(conv).
    Returns (best_a_i, best_new_M, best_new_conv_segment).
    """
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    
    # Unaffected max
    unaff_max = 0.0
    if k_start > 0:
        unaff_max = max(unaff_max, float(conv[:k_start].max()))
    if k_end < 2 * n - 2:
        unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
    
    a_shifted = a[k_start - i:k_end + 1].copy()  # a[k-i] for k in [k_start, k_end]
    conv_aff = conv[k_start:k_end + 1].copy()
    
    # Check if k=2i is in the affected range
    has_quadratic = (2 * i >= k_start) and (2 * i <= k_end)
    
    # Candidate deltas to try
    # 1. delta = 0 (no change)
    # 2. For each affected k, the delta where conv[k] + 2*delta*a[k-i] = M
    # 3. The delta where the quadratic = M
    # 4. delta = -a[i] (set to zero)
    
    candidates = [0.0, -a[i]]
    
    for idx in range(k_end - k_start + 1):
        k = k_start + idx
        if k == 2 * i:
            continue  # Handle quadratic separately
        coeff = 2.0 * a_shifted[idx]
        if abs(coeff) < 1e-15:
            continue
        # conv[k] + coeff * delta = M => delta = (M - conv[k]) / coeff
        delta = (M - conv_aff[idx]) / coeff
        if delta >= -a[i] - 1e-15:
            candidates.append(delta)
    
    if has_quadratic:
        qidx = 2 * i - k_start
        # conv[2i] + 2*delta*a[i] + delta^2 = M
        # delta^2 + 2*a[i]*delta + (conv[2i] - M) = 0
        A_q = 1.0
        B_q = 2.0 * a[i]
        C_q = conv_aff[qidx] - M
        disc = B_q * B_q - 4 * A_q * C_q
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            d1 = (-B_q + sqrt_disc) / (2 * A_q)
            d2 = (-B_q - sqrt_disc) / (2 * A_q)
            if d1 >= -a[i] - 1e-15:
                candidates.append(d1)
            if d2 >= -a[i] - 1e-15:
                candidates.append(d2)
        # Also try the vertex of the quadratic: delta = -a[i]
        # (already included)
    
    # Evaluate each candidate
    best_delta = 0.0
    best_new_M = M
    
    for delta in candidates:
        if delta < -a[i] - 1e-15:
            continue
        delta = max(delta, -a[i])
        
        new_vals = conv_aff + 2.0 * delta * a_shifted
        if has_quadratic:
            qidx = 2 * i - k_start
            new_vals[qidx] = conv_aff[qidx] + 2.0 * delta * a[i] + delta * delta
        # Actually wait, the formula is:
        # new_conv[k] = conv[k] + 2*delta*a[k-i] + delta^2*[k==2i]
        # But when k==2i, a[k-i] = a[i], so the linear part gives 2*delta*a[i]
        # and the quadratic part gives delta^2. Total: 2*delta*a[i] + delta^2. ✓
        # For k != 2i: new_conv[k] = conv[k] + 2*delta*a[k-i]. ✓
        
        new_M_candidate = max(unaff_max, float(new_vals.max()))
        if new_M_candidate < best_new_M - 1e-15:
            best_new_M = new_M_candidate
            best_delta = delta
    
    # Also try a few small perturbations around the best candidate
    if best_delta != 0.0:
        for eps in [0.01, -0.01, 0.001, -0.001]:
            d = best_delta * (1 + eps)
            if d < -a[i]:
                d = -a[i]
            new_vals = conv_aff + 2.0 * d * a_shifted
            if has_quadratic:
                qidx = 2 * i - k_start
                new_vals[qidx] = conv_aff[qidx] + 2.0 * d * a[i] + d * d
            new_M_candidate = max(unaff_max, float(new_vals.max()))
            if new_M_candidate < best_new_M - 1e-15:
                best_new_M = new_M_candidate
                best_delta = d
    
    if best_delta == 0.0:
        return a[i], M, None
    
    # Apply the change
    new_a_i = max(a[i] + best_delta, 0.0)
    changes = 2.0 * best_delta * a_shifted
    if has_quadratic:
        qidx = 2 * i - k_start
        changes[qidx] = 2.0 * best_delta * (a[i]) + best_delta * best_delta
        # Wait, the change in conv[2i] is:
        # (a[i]+d)^2 - a[i]^2 = 2*a[i]*d + d^2
        # And the formula gives: 2*d*a[2i-i] + d^2 = 2*d*a[i] + d^2 ✓
    
    return new_a_i, best_new_M, changes

def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """Coordinate descent with exact 1D subproblem solving."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-18:
                continue
            
            new_val, new_M, changes = exact_coord_descent_step(a, conv, M, i)
            
            if new_val != a[i] and changes is not None and new_M < M - 1e-15:
                a[i] = new_val
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                conv[k_start:k_end + 1] += changes
                M = new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"     CD[{label}] sweep={sweeps} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break
        
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        # Damping schedule
        if it <= 10:
            alpha = 1.0  # Full iterate
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.5
        else:
            alpha = max(0.2, 8.0 / (it + 5.0))
        
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Stagnation handling
        if score_a > best_score * (1 + 1e-7):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)
        
        if stagnation > 20:
            # Random perturbation
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + rng.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
                score_pert = 2.0 * n * fast_convolve(a_pert).max()
                if score_pert < best_score * 1.01:
                    a = a_pert
            stagnation = 0
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"     LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        # Uniform
        a = np.ones(n)
    elif init_type == 1:
        # Cosine window
        t = np.arange(n)
        x = t / (n - 1) * 2 - 1
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif init_type == 2:
        # Two humps
        c1 = n * 0.25
        c2 = n * 0.75
        w = n * 0.15
        a = np.exp(-0.5 * ((t if 't' in dir() else np.arange(n)) - c1) / w) ** 2 + \
            np.exp(-0.5 * ((np.arange(n)) - c2) / w) ** 2
    elif init_type == 3:
        # Random smooth
        k = max(1, n // 16)
        coeffs = rng.standard_normal(k)
        t = np.arange(n)
        a = np.zeros(n)
        for j in range(k):
            a += coeffs[j] * np.cos(2 * np.pi * (j + 1) * t / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a + 2.0, 0.01)
    elif init_type == 4:
        # Power law
        t = np.arange(n)
        x = t / (n - 1) * 2 - 1
        p = rng.uniform(0.5, 2.0)
        a = (1 - p * x * x * 0.5) ** p
        a = np.maximum(a, 0.01)
    elif init_type == 5:
        # Periodic
        period = int(rng.integers(3, 15))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    else:
        # Sparse random
        a = rng.random(n)
        a[a < 0.3] = 0
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def multiplicative_weights_descent(a0, time_limit, lr_init=0.01, top_k=50, verbose=True):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-12:
                break
        
        # Find top k peaks
        K = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -K)[-K:]
        pv = conv[peaks]
        
        # Softmax weights
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        # Multiplicative update
        a = a * np.exp(-lr * sg)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s
        
        conv = fast_convolve(a)
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"     MW it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score

def local_search(a0, time_limit, verbose=False):
    """Random perturbation local search (simulated annealing style)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

    if s < 1e-15:

    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    T = 0.1  # Initial "temperature"
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Generate perturbation
        # Option 1: Add Gaussian noise to random subset
        # Option 2: Swap mass between two random coordinates
        # Option 3: Scale a random coordinate
        
        choice = rng.integers(0, 3)
        if choice == 0:
            # Gaussian noise on random 10% of coordinates
            mask = rng.random(n) < 0.1
            a_new = a.copy()
            a_new[mask] += rng.standard_normal(mask.sum()) * 0.05
            a_new = np.maximum(a_new, 0)
        elif choice == 1:
            # Move mass between two coordinates
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            amt = a[i] * rng.uniform(0.1, 0.5)
            a_new = a.copy()
            a_new[i] -= amt
            a_new[j] += amt
            a_new = np.maximum(a_new, 0)
        else:
            # Scale a random block
            start = rng.integers(0, n - 10)
            length = rng.integers(5, min(50, n - start))
            factor = rng.uniform(0.5, 1.5)
            a_new = a.copy()
            a_new[start:start + length] *= factor
            a_new = np.maximum(a_new, 0)
        
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            conv = conv_new
            no_improve = 0
        elif score_new < best_score * (1 + T):
            # Accept with probability (Metropolis)
            if np.log(rng.random()) < (best_score - score_new) / T:
                a = a_new.copy()
                conv = conv_new
        else:
            no_improve += 1
        
        T *= 0.9995
        
        if no_improve > 500:
            T = 0.05
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"     LS it={it} ({elapsed:.0f}s): best={best_score:.8f} T={T:.4f}")
    
    return best_a, best_score

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:

def lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var * np.ones(2 * n - 1),
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')

    M_val = float(M_var.value) if M_var.value is not None else float('inf')

    Coordinate descent with exact 1D subproblem solving.
    For each coordinate, analytically finds the best value
    by considering all critical points of the piecewise
    linear/quadratic objective.

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

                if v > unaff_max:

                new_vals = conv_aff + 2.0 * d * a_shifted
                if has_quad:
                    new_vals[qidx] = conv_aff[qidx] + 2.0 * d * ai + d * d
                nm = max(unaff_max, float(new_vals.max()))
                if nm < best_new_M - 1e-15:
                    best_new_M = nm
                    best_delta = d

            # Refinement: perturb around best candidate
            if best_delta != 0.0:
                for eps in [0.02, -0.02, 0.005, -0.005]:
                    d = best_delta * (1.0 + eps)
                    if d < -ai:
                        d = -ai
                    new_vals = conv_aff + 2.0 * d * a_shifted
                    if has_quad:
                        new_vals[qidx] = conv_aff[qidx] + 2.0 * d * ai + d * d
                    nm = max(unaff_max, float(new_vals.max()))
                    if nm < best_new_M - 1e-15:
                        best_new_M = nm
                        best_delta = d

            if best_delta == 0.0 or best_new_M >= M - 1e-15:
                continue

            # Apply
            a[i] = max(ai + best_delta, 0.0)
            changes = 2.0 * best_delta * a_shifted
            if has_quad:
                changes[qidx] = 2.0 * best_delta * ai + best_delta * best_delta
            conv[k_start:k_end + 1] += changes
            M = best_new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP best-response alternating with progressive damping and randomization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Progressive damping
        if it <= 5:
            alpha = 1.0
        elif it <= 15:
            alpha = 0.8
        elif it <= 40:
            alpha = 0.5
        elif it <= 80:
            alpha = 0.3
        else:
            alpha = max(0.15, 6.0 / (it + 10.0))

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation: random perturbation
        if score_a > best_score * (1.0 + 1e-7):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 25:
            ns = 0.02 / np.sqrt(n)
            a_pert = a + rng.standard_normal(n) * ns
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
                sc = 2.0 * n * fast_convolve(a_pert).max()
                if sc < best_score * 1.005:
                    a = a_pert
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multiplicative_weights(a0, time_limit, lr_init=0.008, top_k=40, verbose=True):
    """Multiplicative weights (mirror descent) on the probability simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            lr *= 0.88
            no_improve = 0
            if lr < 1e-12:
                break

        # Top-k peaks with softmax weighting
        K = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -K)[-K:]
        pv = conv[peaks]
        temp = 0.008 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Subgradient computation
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Multiplicative update
        a = a * np.exp(-lr * sg)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s
        conv = fast_convolve(a)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def local_search(a0, time_limit, verbose=True):
    """Simulated annealing style local search with diverse move types."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    T = 0.08
    it = 0
    no_improve = 0
    cur_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Choose move type
        choice = rng.integers(0, 4)
        if choice == 0:
            # Gaussian noise on random subset
            mask = rng.random(n) < 0.08
            a_new = a.copy()
            a_new[mask] += rng.standard_normal(int(mask.sum())) * 0.04
            a_new = np.maximum(a_new, 0.0)
        elif choice == 1:
            # Move mass between two random coordinates
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            amt = a[i] * rng.uniform(0.1, 0.4)
            a_new = a.copy()
            a_new[i] -= amt
            a_new[j] += amt
            a_new = np.maximum(a_new, 0.0)
        elif choice == 2:
            # Scale a random contiguous block
            start = int(rng.integers(0, max(1, n - 20)))
            ln = int(rng.integers(5, min(40, n - start)))
            factor = rng.uniform(0.7, 1.3)
            a_new = a.copy()
            a_new[start:start + ln] *= factor
            a_new = np.maximum(a_new, 0.0)
        else:
            # Pairwise swap: reduce one, increase another (anti-correlation exploit)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            a_new = a.copy()
            a_new[i] *= rng.uniform(0.6, 0.9)
            a_new[j] *= rng.uniform(1.1, 1.4)
            a_new = np.maximum(a_new, 0.0)

        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            cur_score = score_new
            no_improve = 0
        elif score_new < cur_score * (1.0 + T * 2.0):
            # Metropolis acceptance
            if np.log(rng.random()) < (cur_score - score_new) / max(T, 1e-10):
                a = a_new
                cur_score = score_new
        else:
            no_improve += 1

        T = max(T * 0.9997, 0.001)

        if no_improve > 800:
            T = 0.04
            no_improve = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS it={it} ({elapsed:.0f}s): best={best_score:.8f} T={T:.4f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / max(center, 1.0)

    if init_type == 0:
        a = np.ones(n)
    elif init_type == 1:
        a = 0.5 * (1.0 + np.cos(np.pi * x))
    elif init_type == 2:
        p = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * x)), 0.0) ** p + 0.05
    elif init_type == 3:
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 12))
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * t / period)
    elif init_type == 5:
        k = max(1, n // 20)
        a = np.ones(n)
        for j in range(k):
            freq = 2.0 * np.pi * (j + 1) / n
            amp = rng.standard_normal() * 0.1 / (j + 1)
            a += amp * np.cos(freq * t + rng.uniform(0, 2 * np.pi))
    elif init_type == 6:
        # U-shape
        a = 0.3 + 0.7 * x * x
    else:
        a = 0.5 + 0.3 * np.sin(2.0 * np.pi * t / n * rng.integers(1, 5))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    Phase 1: LP alternating minimization (main workhorse)

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n}, score={best_score:.8f}")

    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return

        try:
            sc = float(evaluate_sequence(candidate.tolist()))

        if np.isfinite(sc) and sc < best_score:
