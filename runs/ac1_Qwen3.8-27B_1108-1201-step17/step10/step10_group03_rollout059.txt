The KKT conditions for `min a^T H a` s.t. `a >= 0`, `sum(a) = 1` are:
1. `2 H a + mu * 1 - nu = 0` (stationarity)
2. `sum(a) = 1` (primal feasibility)
3. `a >= 0`, `nu >= 0` (dual feasibility)
4. `a_j * nu_j = 0` for all j (complementary slackness)

From (1): `nu = 2 H a + mu * 1`.
From (4): for `j` in the active set `S` (where `a_j > 0`): `(2 H a + mu * 1)_j = 0`.
For `j` not in `S`: `(2 H a + mu * 1)_j >= 0`.

So for the active set `S`:
`2 H_SS a_S + mu * 1_S = 0`
`sum_S a_S = 1`

This is a linear system in `(a_S, mu)`:
`[2 H_SS, 1_S; 1_S^T, 0] [a_S; mu] = [0; 1]`

Once we have `a_S` and `mu`, we check if `nu_j = (2 H a + mu * 1)_j >= 0` for `j` not in `S`. If yes, we're at the optimum. If not, we add the violating `j` to the active set and repeat.

This is the active-set method for QP. It's efficient when the active set is small (which it should be, since `a >= 0` and `sum(a) = 1` means most entries of `a` are zero or very small).

Actually, for a smooth sequence like ours, the active set is probably all of `S = {0, ..., n-1}` (all entries are positive). In that case, the KKT conditions reduce to:
`2 H a + mu * 1 = 0`
`sum(a) = 1`

Which is a linear system:
`[2 H, 1; 1^T, 0] [a; mu] = [0; 1]`

This is an (n+1) x (n+1) linear system, which can be solved in O(n^3) time. For n=1200, that's about 1.7 billion operations, which might take a few seconds with numpy.

But if we solve this linear system directly, we bypass the need for a QP solver entirely! This is much faster.

def solve_qp_hankel(lambda_vec, max_active_set_iters=100):
    n = (len(lambda_vec) + 1) // 2
    idx = np.arange(n)
    H = lambda_vec[idx[:, None] + idx[None, :]]
    
    # Start with full active set
    active = set(range(n))
    
    for it in range(max_active_set_iters):
        active_list = sorted(active)
        m = len(active_list)
        H_sub = H[np.ix_(active_list, active_list)]
        
        # Solve [2*H_sub  1; 1^T  0] [a_sub; mu] = [0; 1]
        KKT = np.zeros((m+1, m+1))
        KKT[:m, :m] = 2 * H_sub
        KKT[:m, m] = 1
        KKT[m, :m] = 1
        rhs = np.zeros(m+1)
        rhs[m] = 1
        
        try:
            sol = np.linalg.solve(KKT, rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(KKT, rhs, rcond=None)[0]
        
        a_sub = sol[:m]
        mu = sol[m]
        
        # Check non-negativity of a_sub
        if np.all(a_sub >= -1e-10):
            # Check KKT for inactive set
            a_full = np.zeros(n)
            a_full[active_list] = np.maximum(a_sub, 0)
            a_full = a_full / a_full.sum()
            
            # Check if inactive variables want to enter
            grad = 2 * H @ a_full + mu
            inactive = set(range(n)) - active
            violates = False
            for j in inactive:
                if grad[j] < -1e-10:
                    # Add j to active set
                    active.add(j)
                    violates = True
                    break
            if not violates:
                return a_full
        
        # Remove the most negative variable from active set
        min_idx = np.argmin(a_sub)
        if a_sub[min_idx] < 0:
            active.discard(active_list[min_idx])
        else:
            # Shouldn't happen, but break to avoid infinite loop
            a_full = np.zeros(n)
            a_full[active_list] = np.maximum(a_sub, 0)
            a_full = a_full / a_full.sum()
            return a_full
    
    # Fallback
    a_full = np.zeros(n)
    active_list = sorted(active)
    a_full[active_list] = np.maximum(sol[:len(active_list)], 0)
    a_full = a_full / a_full.sum()

active-set QP solver, and multi-n exploration.

Hankel-matrix QP that directly optimizes a in the direction suggested by the LP dual.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_fixed_a(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Returns (b_opt, M_opt, dual_weights)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    con = T @ b_var <= M_var
    constraints = [con, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for attempt in range(5):
        try:
            sv = solver if attempt == 0 else [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP][attempt - 1]
            prob.solve(solver=sv)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    b_val = b_val / s
                    lam = np.zeros(2 * n - 1)
                    try:
                        dual_val = con.dual_value
                        if dual_val is not None:
                            lam = np.asarray(dual_val, dtype=np.float64)
                            lam = np.maximum(lam, 0)
                    except Exception:
                        pass
                    return b_val, float(M_var.value), lam
        except Exception:
            continue
    return None, None, None

def solve_qp_hankel(lambda_vec):
    """Solve min a^T H a s.t. a>=0, sum(a)=1 where H[j,i]=lambda[j+i].
    Uses KKT/active-set approach."""
    n = (len(lambda_vec) + 1) // 2
    idx = np.arange(n)
    H = lambda_vec[idx[:, None] + idx[None, :]]

    # Symmetrize for numerical stability
    H = 0.5 * (H + H.T)

    # Try full active set first
    def try_active_set(active_list):
        m = len(active_list)
        if m == 0:
            return None, None
        H_sub = H[np.ix_(active_list, active_list)]
        KKT = np.zeros((m + 1, m + 1))
        KKT[:m, :m] = 2.0 * H_sub
        KKT[:m, m] = 1.0
        KKT[m, :m] = 1.0
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0
        try:
            sol = np.linalg.solve(KKT, rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(KKT, rhs, rcond=None)[0]
            except Exception:
                return None, None
        a_sub = sol[:m]
        mu = sol[m]
        return a_sub, mu

    # Start with full set
    active_list = list(range(n))
    a_sub, mu = try_active_set(active_list)
    if a_sub is None:
        return np.ones(n) / n

    for inner_it in range(50):
        if np.all(a_sub >= -1e-8):
            # Check KKT for inactive variables
            a_full = np.zeros(n)
            a_full[active_list] = np.maximum(a_sub, 0)
            s = a_full.sum()
            if s < 1e-15:
                return np.ones(n) / n
            a_full = a_full / s
            grad = 2.0 * H @ a_full + mu
            inactive = [j for j in range(n) if j not in set(active_list)]
            violated = False
            for j in inactive:
                if grad[j] < -1e-8:
                    active_list.append(j)
                    violated = True
                    break
            if not violated:
                return a_full
            # Solve with new active set
            a_sub, mu = try_active_set(sorted(active_list))
            if a_sub is None:
                return a_full
        else:
            # Remove most negative
            min_idx = int(np.argmin(a_sub))
            active_list.pop(min_idx)
            if len(active_list) < 2:
                a_full = np.zeros(n)
                a_full[active_list] = np.maximum(a_sub, 0)
                s = a_full.sum()
                if s < 1e-15:
                    return np.ones(n) / n
                return a_full / s
            a_sub, mu = try_active_set(sorted(active_list))
            if a_sub is None:
                return np.ones(n) / n

    # Fallback
    a_full = np.zeros(n)
    a_full[active_list] = np.maximum(a_sub, 0)
    s = a_full.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_full / s


def make_hankel_from_weights(weight_vec):
    """Create Hankel matrix H[j,i] = weight[j+i] from a vector of length 2n-1."""
    n = (len(weight_vec) + 1) // 2
    idx = np.arange(n)
    H = weight_vec[idx[:, None] + idx[None, :]]
    return 0.5 * (H + H.T)


def lp_qp_alternation(a0, time_limit, solver, max_iters=500):
    """LP-QP alternation: LP to get duals, QP to update a."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    prev_best = best_score
    last_lambda = None

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: LP to find best b given current a, get dual weights
        b_new, M1, lam_b = solve_lp_fixed_a(a, solver)
        if b_new is not None:
            b = b_new
            if lam_b is not None and lam_b.sum() > 1e-10:
                last_lambda = lam_b

        # Step 2: QP to find best a given dual weights
        if last_lambda is not None and last_lambda.sum() > 1e-10:
            a_qp = solve_qp_hankel(last_lambda)
            if a_qp is not None:
                # Blend with current a for stability
                alpha = 0.7 / (1.0 + 0.1 * it)
                a_new = (1.0 - alpha) * a + alpha * a_qp
                a = normalize(a_new)

        # Also do LP for a (for direct comparison)
        a_lp, M2, lam_a = solve_lp_fixed_b(b, solver)
        if a_lp is not None:
            # Compare LP and QP solutions, pick the better one
            sc_a = 2.0 * n * fast_convolve(a).max()
            sc_lp = 2.0 * n * fast_convolve(a_lp).max()
            if sc_lp < sc_a:
                # Blend LP solution
                alpha = 0.5 / (1.0 + 0.05 * it)
                a = normalize((1.0 - alpha) * a + alpha * a_lp)
            if lam_a is not None and lam_a.sum() > 1e-10:
                last_lambda = lam_a

        # Track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                noise_scale = 0.01 / math.sqrt(stall)
                a = normalize(a + rng.standard_normal(n) * noise_scale)
                b = normalize(b + rng.standard_normal(n) * noise_scale)
                stall = 0

            if (it + 1) % 200 == 0:
                print(f"    LP-QP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def lp_alternation(a0, time_limit, solver, max_iters=300):
    """Pure LP alternation with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()

    t0 = time.monotonic()

    stall = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M1, _ = solve_lp_fixed_a(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = normalize((1.0 - alpha) * y_b + alpha * b_new)
            b = normalize(((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2)))

        a_new, M2, _ = solve_lp_fixed_b(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = normalize((1.0 - alpha) * y_a + alpha * a_new)
            a = normalize(((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3)))

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                noise_scale = 0.015 / math.sqrt(stall)
                y_a = normalize(y_a + rng.standard_normal(n) * noise_scale)
                y_b = normalize(y_b + rng.standard_normal(n) * noise_scale)
                stall = 0
            if (it + 1) % 200 == 0:
                print(f"    LP alt iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 15))

        def lse_obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            lse = b_max_val + np.log(np.exp(b_shift - b_max_val).sum())
            return (2.0 * n) * lse

        def lse_grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max_val = b_shift.max()
            exp_ws = np.exp(b_shift - b_max_val)
            w = exp_ws / exp_ws.sum()
            # Gradient w.r.t. a: 2 * sum_k w[k] * a[k-j]
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-15,
                                    'maxfun': max_iter * 300, 'disp': False})
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"      LSE T={T}: score={sc:.10f}")
        except Exception:
            pass

    return best_a, best_score


def direct_subgradient(a0, time_limit, n_steps=50000):
    """Direct subgradient with adaptive step and peak tracking."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03
    no_improve = 0
    momentum = np.zeros(n)
    beta = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Momentum
        momentum = beta * momentum + (1 - beta) * g
        g_use = momentum / (1 - beta ** (step + 1))  # bias correction

        a_trial = a - lr * g_use
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.06)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetric_refine(a0, time_limit, solver):

    n = len(a0)

                if sc < best_score - 1e-12:

        except Exception:

def peak_target_qp(a0, lambda_weights):
    """Solve QP with modified weights focusing on peaks."""

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = normalize(best_seq)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP Alternation ===
    print(f"\n=== Phase 1: LP Alternation (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "lp_alt")

    # === Phase 2: LP-QP Alternation ===
    print(f"\n=== Phase 2: LP-QP Alternation (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_qp, sc_qp = lp_qp_alternation(best_seq, t2, solver)
        try_update(a_qp, "lp_qp")

    # === Phase 3: LogSumExp L-BFGS (low T) ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 30:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  Budget: {t3:.0f}s, T_stages={T_stages}")
        a_lse, sc_lse = logsumexp_lbgfs(best_seq, t3, T_stages=T_stages)
        try_update(a_lse, "lse_lowT")

    # === Phase 4: Subgradient with momentum ===
    print(f"\n=== Phase 4: Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 15:
        a_sg, sc_sg = direct_subgradient(best_seq, t4, n_steps=30000)
        try_update(a_sg, "subgrad")

    # === Phase 5: Symmetry-constrained LP ===
    print(f"\n=== Phase 5: Symmetry LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 15:
        a_sym, sc_sym = symmetric_refine(best_seq, t5, solver)
        try_update(a_sym, "symmetric")

    # === Phase 6: Second LP-QP round ===
    print(f"\n=== Phase 6: LP-QP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.12, 150.0)
    if t6 > 20:
        a_qp2, sc_qp2 = lp_qp_alternation(best_seq, t6, solver)
        try_update(a_qp2, "lp_qp_2")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_candidates = []
        base_n = n_best
        for mult in [0.8, 0.9, 1.1, 1.25, 1.5, 2.0]:
            cand_n = int(base_n * mult)
            if 300 <= cand_n <= 5000 and cand_n != n_best:
                n_candidates.append(cand_n)

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.25, 100.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick LP-QP
            a_cand, _ = lp_qp_alternation(init_cand, t_sub * 0.5, solver)
            try_update(a_cand, f"lpqp_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 30:
                T_sub = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
                a_cand2, _ = logsumexp_lbgfs(a_cand if a_cand is not None else init_cand,
                                             min(rem * 0.4, 60.0), T_stages=T_sub)
                try_update(a_cand2, f"lse_n{cand_n}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very low T L-BFGS
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            a_fin, _ = logsumexp_lbgfs(best_seq, t_fin,
                                       T_stages=[0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
            try_update(a_fin, "final_lse")

        # Quick subgradient
        rem = deadline - time.monotonic()
        if rem > 15:
            t_sg = min(rem * 0.3, 30.0)
            a_fin_sg, _ = direct_subgradient(best_seq, t_sg, n_steps=10000)
            try_update(a_fin_sg, "final_sg")

        # Final LP-QP
        rem = deadline - time.monotonic()
        if rem > 15:
            t_fin2 = min(rem * 0.4, 40.0)
            a_fin2, _ = lp_qp_alternation(best_seq, t_fin2, solver)
            try_update(a_fin2, "final_lpqp")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
