def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    # Try alternate solvers
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    return None

    t_k = 1.0

        a_prev = a.copy()
        t_k = 1.0
        
        n_steps = 5000
        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            
            # Adaptive temperature
            progress = min(1.0, (restart * n_steps + step) / (n_restarts * n_steps))
            T = T_init * (T_final / T_init) ** progress
            # eta decreases as 1/sqrt(step)
            eta = eta_init / (1.0 + 0.01 * step)

            for k in range(len(b)):

                js = max(0, k - n + 1)
                je = min(n - 1, k)

            if s < 1e-15:

            else:
                a_new = a_new / s

            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
            
            a_prev = a
            a = a_new
            t_k = t_next

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_gradient(a, w):
    """
    Compute gradient of weighted sum of convolution entries.
    grad[j] = sum_k w[k] * a[k-j] for valid k-j in [0, n-1].
    Uses FFT: grad[j] = conv(w, a_rev)[n-1+j]
    """
    n = len(a)
    a_rev = a[::-1]
    # conv(w, a_rev) has length (2n-1) + n - 1 = 3n-2
    c = convolve_ab(w, a_rev)
    # grad[j] = c[n-1+j] for j = 0, ..., n-1
    return c[n-1:n-1+n]


def compute_soft_weights(b, T):
    """Compute softmax weights for the convolution entries."""
    b_max = b.max()
    b_shift = (b - b_max) / T
    exp_b = np.exp(b_shift)
    w = exp_b / exp_b.sum()
    return w


def nesterov_multiplicative(a0, time_limit, T_init, T_final, eta_scale, n_restarts):
    """
    Nesterov-accelerated multiplicative (mirror) descent.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    steps_per_restart = 3000
    
    for restart in range(n_restarts):
        a = best_a.copy()
        a_prev = a.copy()
        t_k = 1.0
        
        for step in range(steps_per_restart):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Adaptive temperature
            total_step = restart * steps_per_restart + step
            total_steps = n_restarts * steps_per_restart
            progress = min(1.0, total_step / max(1, total_steps))
            T = T_init * (T_final / T_init) ** progress
            
            # Learning rate
            eta = eta_scale / (1.0 + 0.05 * step)
            
            # Nesterov extrapolation
            t_next = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k))
            theta = (t_k - 1.0) / t_next
            y = a + theta * (a - a_prev)
            y = np.maximum(y, 1e-30)
            y_sum = y.sum()
            if y_sum < 1e-15:
                y = a.copy()
            else:
                y = y / y_sum
            
            # Compute convolution and gradient
            b = fast_convolve(y)
            w = compute_soft_weights(b, T)
            g = compute_gradient(y, w)
            
            # Multiplicative update
            log_update = -eta * g * n
            log_update = np.clip(log_update, -40.0, 40.0)
            a_new = y * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-30)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s
            
            # Track best
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
            
            a_prev = a
            a = a_new
            t_k = t_next
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_score


def peak_pushing(a0, n_steps, time_limit):
    """
    Peak-pushing heuristic: identify positions contributing to the max convolution
    entry and redistribute their mass to positions that don't contribute to any peak.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        
        # Find active peaks (within 1% of max)
        threshold = b_max * 0.999
        active_k = np.where(b >= threshold)[0]
        
        # Find positions contributing most to the active peaks
        # contribution[j] = sum_{k in active} a[k-j] * a[j] / b[k]
        # Simplified: just find j where a[j] * a[k*-j] is large for some active k*
        
        # For each active k, the contributors are positions j where a[k-j] > 0
        # The "responsibility" of position j for peak k is a[j] * a[k-j]
        
        # Compute total responsibility
        resp = np.zeros(n)
        for k in active_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            resp[idx] += a[j]  # Wait, this is wrong
            # resp[j] should be a[j] * a[k-j]
            resp[idx] += a[idx] * a[k - idx]
        
        # Normalize responsibility
        resp_max = resp.max()
        if resp_max < 1e-15:
            break
        
        # "Safe" positions: low responsibility
        safe_threshold = resp_max * 0.1
        safe_positions = np.where(resp <= safe_threshold)[0]
        hot_positions = np.where(resp > resp_max * 0.5)[0]
        
        if len(safe_positions) == 0 or len(hot_positions) == 0:
            break
        
        # Move mass from hot to safe
        move_frac = 0.01
        total_mass_to_move = move_frac * a[hot_positions].sum()
        a[hot_positions] *= (1.0 - move_frac)
        # Distribute to safe positions proportional to current value
        safe_weights = a[safe_positions] + 1e-10
        safe_weights = safe_weights / safe_weights.sum()
        a[safe_positions] += total_mass_to_move * safe_weights
        
        # Also move from the peak position itself (center of convolution)
        # The center is at k = n-1, contributed by a[j]*a[n-1-j]
        # Positions near the center of a contribute most
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def lp_best_response(a_fixed, solver, max_time=30):
    """Solve LP: find b >= 0, sum(b)=1 minimizing max_k conv(a_fixed, b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    t_start = time.monotonic()
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    
    # Try alternate solvers
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        if time.monotonic() - t_start > max_time:
            break
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s
        except Exception:
            continue
    
    return None


def alternating_lp(a0, solver, time_limit, n_iters=100):
    """Alternating LP best-response method."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Best response for b given a
        b_new = lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        
        # Best response for a given b
        a_new = lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        # Track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()
        
        if (it + 1) % 20 == 0:
            print(f"    AltLP iter={it+1}: score={best_score:.10f}")
    
    return best_a, best_score


def structured_init(n, method):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if method == 0:
        # Sine-squared
        a = np.sin(np.pi * x) ** 2
    elif method == 1:
        # Cosine taper
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif method == 2:
        # Fejer kernel (triangular autocorrelation)
        # DFT of tent function
        k = np.arange(n)
        tent = np.maximum(0, 1 - np.abs(k - n//2) / (n//2))
        a = np.abs(irfft(rfft(tent, 2*n), 2*n)[:n]) ** 2
        # Actually, for Fejer kernel: a[j] = (1 - j/(n+1))^2 for j=0..n
        a = np.maximum(0, 1 - np.arange(n) / n) ** 2
    elif method == 3:
        # Gaussian
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif method == 4:
        # Bump (flat top)
        a = np.maximum(0, np.cos(np.pi * (x - 0.5))) ** 4
    elif method == 5:
        # Uniform
        a = np.ones(n)
    elif method == 6:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Power taper
        a = (1 - np.abs(2*x - 1)) ** 1.5
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_n(a, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

        b = fast_convolve(a)
        b_max = b.max()
        
        # Active set: peaks within 0.1% of max
        threshold = b_max * 0.9999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je+1] += a[k - js:k - js + (je - js + 1)]
        g /= len(active)
        
        # Step
        a_trial = a - lr * g * n
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.9
                no_improve = 0
                # Add small perturbation
                a = a + np.random.default_rng(step).standard_normal(n) * 0.001
                a = normalize(a)
    
    return best_a, best_score
