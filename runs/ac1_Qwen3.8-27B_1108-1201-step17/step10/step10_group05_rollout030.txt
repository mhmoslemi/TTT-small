def fft_conv(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    sz = n + m - 1
    fs = 1 << (sz - 1).bit_length()
    return irfft(rfft(a, fs) * rfft(b, fs), fs)[:sz]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return 1e10
    c = fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rho = np.sum(u > (cs - 1.0) / np.arange(1, n + 1))
    if rho == 0:
        return np.ones(n) / n
    theta = (cs[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build T where T[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, solver=None):
    """Solve: min M s.t. conv(a,b)[k] <= M for all k, sum(b)=1, b>=0."""
    import cvxpy as cp

    M = cp.Variable()
    T = build_conv_matrix(a)
    cons = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)

def coord_descent(a0, time_limit, verbose=False):
    """Coordinate descent with exact 1D minimax per coordinate."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = proj_simplex(a)
    conv = fft_conv(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        any_change = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-13:
                continue

            ks = i
            ke = min(i + n - 1, 2 * n - 2)

            if ks > 0:
                um = float(conv[:ks].max())
            else:
                um = 0.0
            if ke < 2 * n - 2:
                um = max(um, float(conv[ke + 1:].max()))

            affected = conv[ks:ke + 1]
            coeffs = 2.0 * a[(ks - i):(ke - i + 1)]
            has_quad = (2 * i >= ks) and (2 * i <= ke)
            qidx = 2 * i - ks if has_quad else -1

            # Generate candidates
            cands = set()
            cands.add(0.0)
            cands.add(-ai * 0.5)
            cands.add(-ai * 0.1)
            cands.add(-ai * 0.9)
            cands.add(ai * 0.1)

            # Where each affected line hits um
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-13:
                    d = (um - affected[j]) / c
                    if -ai * 1.1 < d < ai * 2:
                        cands.add(d)

            # Top-5 pairwise crossings
            top_k = min(5, len(affected))
            ti = np.argsort(affected)[-top_k:]
            for x in range(len(ti)):
                for y in range(x + 1, len(ti)):
                    i1, i2 = ti[x], ti[y]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-13:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.1 < d < ai * 2:
                            cands.add(d)

            best_d = 0.0
            best_nm = M

            for d in cands:
                d = max(d, -ai)
                if abs(d) < 1e-15:
                    continue
                na = affected + d * coeffs
                if has_quad:
                    na = na.copy()
                    na[qidx] += d * d
                nm = max(um, float(na.max()))

            conv = fft_conv(a)
            M = float(conv.max())

        if not improved and sweep > 15:
            break

def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient method with Nesterov acceleration for final polish."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = proj_simplex(a)
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    it = 0
    gamma = 1.0
    y = a.copy()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        # Nesterov lookahead
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y = proj_simplex(np.maximum(y, 0.0))

        conv_y = fft_conv(y)
        M = conv_y.max()
        thresh = M * 0.95
        near_peak = np.where(conv_y >= thresh)[0]
        rho = 1.0 / max(1, len(near_peak))

        grad = np.zeros(n)
        for k in near_peak:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                j = np.arange(i_lo, i_hi + 1)
                grad[j] += rho * 2.0 * y[k - j]

        a_new = proj_simplex(a - step * grad)
        sc = score_of(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if verbose and it % 50 == 0:
            print(f"    SubGrad it={it} ({time.monotonic()-t0:.0f}s) best={best_score:.6f}")

        if stall > 300:
            break

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    fi /= fi.sum() if fi.sum() > 1e-15 else 1.0
    return fi

            sc = float(evaluate_sequence(p.tolist()))
            if np.isfinite(sc):
                best_seq = p.copy()
                best_score = sc
                print(f"Parent: n={len(p)}, score={sc:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = 2.0
        print(f"Default init: n={n}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP Fictitious Play (main workhorse) ===
    print(f"\n=== Phase 1: LP Fictitious Play ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 700.0)
    print(f"  Budget: {t1:.0f}s, n={n0}")
    a1, s1, it1 = lp_fictitious_play(best_seq, t1, solver=solver, verbose=True)
    sc1 = float(evaluate_sequence(a1.tolist()))
    if np.isfinite(sc1) and sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  Phase 1 result: {best_score:.6f} ({it1} iters)")

    # === Phase 2: Coordinate Descent polishing ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 350.0)
    print(f"  Budget: {t2:.0f}s, n={len(best_seq)}")
    a2, s2 = coord_descent(best_seq, t2, verbose=True)
    sc2 = float(evaluate_sequence(a2.tolist()))
    if np.isfinite(sc2) and sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  Phase 2 result: {best_score:.6f}")

    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.15, 200.0)
        a3, s3 = subgradient_polish(best_seq, t3, verbose=True)
        sc3 = float(evaluate_sequence(a3.tolist()))
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  Phase 3 result: {best_score:.6f}")

    # === Phase 4: Another LP FP round (on polished solution) ===
    print(f"\n=== Phase 4: LP FP Round 2 ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(rem * 0.30, 300.0)
        print(f"  Budget: {t4:.0f}s, n={len(best_seq)}")
        a4, s4, it4 = lp_fictitious_play(best_seq, t4, solver=solver, verbose=True)
        sc4 = float(evaluate_sequence(a4.tolist()))
        if np.isfinite(sc4) and sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  Phase 4 result: {best_score:.6f} ({it4} iters)")

        # Final CD
        rem = deadline - time.monotonic()
        if rem > 30:
            t4b = min(rem * 0.5, 100.0)
            a4b, s4b = coord_descent(best_seq, t4b)
            sc4b = float(evaluate_sequence(a4b.tolist()))
            if np.isfinite(sc4b) and sc4b < best_score:
                best_score = sc4b
                best_seq = a4b.copy()
                print(f"  Phase 4b CD result: {best_score:.6f}")

    # === Phase 5: Try alternative lengths ===
    print(f"\n=== Phase 5: Alternative Lengths ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    alt_ns = []
    for dn in [128, 256, 512, -128, -256]:
        n_alt = n_best + dn
        if 512 <= n_alt <= 2560 and n_alt != n_best:
            alt_ns.append(n_alt)

    for n_alt in alt_ns[:2]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        a_interp = interpolate_seq(best_seq, n_alt)
        t_alt = min(rem * 0.30, 150.0)
        print(f"  Trying n={n_alt}, budget={t_alt:.0f}s")
        ar, sr, _ = lp_fictitious_play(a_interp, t_alt, solver=solver, verbose=False)
        # Quick CD polish
        rem2 = deadline - time.monotonic()
        if rem2 > 30:
            ar2, sr2 = coord_descent(ar, min(rem2 * 0.4, 60.0))
            sr_eval = float(evaluate_sequence(ar2.tolist()))
            if np.isfinite(sr_eval) and sr_eval < best_score:
                best_score = sr_eval
                best_seq = ar2.copy()
                print(f"  n={n_alt}: NEW BEST {best_score:.6f}")
            else:
                print(f"  n={n_alt}: {sr_eval:.6f} (no improvement)")
        else:
            sr_eval = float(evaluate_sequence(ar.tolist()))
            if np.isfinite(sr_eval) and sr_eval < best_score:
                best_score = sr_eval
                best_seq = ar.copy()
                print(f"  n={n_alt}: NEW BEST {best_score:.6f}")
            else:
                print(f"  n={n_alt}: {sr_eval:.6f} (no improvement)")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t6 = min(rem * 0.6, 80.0)
        a6, s6 = coord_descent(best_seq, t6)
        sc6 = float(evaluate_sequence(a6.tolist()))
        if np.isfinite(sc6) and sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  Final CD: {best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
