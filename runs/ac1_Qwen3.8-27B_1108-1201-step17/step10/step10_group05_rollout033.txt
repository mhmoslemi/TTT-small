def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Compute sum_j b[i+j] * a[j] for i=0..len(a)-1"""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_grad(a, weights):
    """
    Compute gradient of weighted autocorrelation.
    weights: length 2n-1 (softmax or uniform over active set)
    Returns grad of length n.
    """
    n = len(a)
    # h[i] = sum_j weights[i+j] * a[j]
    # = (weights_rev * a)[2n-2-i]
    w_rev = weights[::-1]
    size = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(W * A, fft_size)
    idx = 2 * n - 2 - np.arange(n)
    h = conv[idx]
    # Add the diagonal correction: grad[i] += weights[2*i] * a[i]
    diag = np.zeros(n)
    valid = 2 * np.arange(n) < (2 * n - 1)
    diag[valid] = weights[2 * np.arange(n)[valid]] * a[valid]
    return 2.0 * (h + diag)

def compute_variance_grad(a):
    """
    Compute gradient of variance of autocorrelation.
    loss = sum_k (conv_k - mean)^2 where mean = 1/(2n-1)
    grad_i = 2 * sum_k (conv_k - mean) * (a_{k-i} + [k=2i]*a_i)
    """
    n = len(a)
    conv = fast_convolve(a)
    mean = 1.0 / (2 * n - 1)
    residuals = conv - mean  # length 2n-1
    # grad_i = 2 * sum_k residuals[k] * a_{k-i} + 2 * residuals[2i] * a_i
    # First term: h[i] = sum_k residuals[k] * a_{k-i}
    r_rev = residuals[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    R = rfft(r_rev, fft_size)
    A = rfft(a, fft_size)
    conv_ra = irfft(R * A, fft_size)
    idx = 2 * n - 2 - np.arange(n)
    h = conv_ra[idx]
    # Diagonal correction
    diag = np.zeros(n)
    valid = 2 * np.arange(n) < (2 * n - 1)
    diag[valid] = residuals[2 * np.arange(n)[valid]] * a[valid]
    return 2.0 * (h + diag)

def mirror_descent_varmin(a0, time_limit, lr_init=0.01, label=""):

    n = len(a0)

    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_var = float(np.var(conv))
    t0 = time.monotonic()
    
    lr = lr_init
    it = 0
    no_improve = 0
    gamma = 1.0
    y = a.copy()

        it += 1

        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        a_new = np.exp(log_a)

            no_improve = 0
        else:
            no_improve += 1

import numpy as np

import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_weighted_grad(a, weights):
    n = len(a)
    w_rev = weights[::-1]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(W * A, fft_size)
    idx = 2 * n - 2 - np.arange(n)
    h = conv[idx]
    diag = np.zeros(n)
    valid = 2 * np.arange(n) < (2 * n - 1)
    diag[valid] = weights[2 * np.arange(n)[valid]] * a[valid]
    return 2.0 * (h + diag)


def compute_variance_grad(a):
    n = len(a)
    conv = fast_convolve(a)
    mean = 1.0 / (2 * n - 1)
    residuals = conv - mean
    r_rev = residuals[::-1]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    R = rfft(r_rev, fft_size)
    A = rfft(a, fft_size)
    conv_ra = irfft(R * A, fft_size)
    idx = 2 * n - 2 - np.arange(n)
    h = conv_ra[idx]
    diag = np.zeros(n)
    valid = 2 * np.arange(n) < (2 * n - 1)
    diag[valid] = residuals[2 * np.arange(n)[valid]] * a[valid]
    return 2.0 * (h + diag)


def mirror_descent_homotopy(a0, time_limit, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    it = 0
    no_improve = 0
    gamma_nest = 1.0
    y = a.copy()
    lr = 0.006
    beta = 15.0
    total_time = time_limit

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (total_time * 0.85))
        gamma = progress ** 2.0

        if it > 1:
            y = (1.0 + gamma_nest) * a - gamma_nest * y
        y = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y)
        M = float(conv_y.max())
        mean_c = 1.0 / (2 * n - 1)
        size = 3 * n - 2
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(y, fft_size)

        grad = np.zeros(n)

        if gamma > 0.005:
            scaled = beta * (conv_y - M)
            scaled = np.clip(scaled, -80, 0)
            w_peak = np.exp(scaled)
            Z = w_peak.sum()
            if Z < 1e-30:
                w_peak = np.zeros_like(conv_y)
                w_peak[np.argmax(conv_y)] = 1.0
                Z = 1.0
            w_peak = w_peak / Z
            w_rev = w_peak[::-1]
            Wp = rfft(w_rev, fft_size)
            conv_p = irfft(Wp * A, fft_size)
            h_p = conv_p[2 * n - 2 - np.arange(n)]
            diag_p = np.zeros(n)
            valid = 2 * np.arange(n) < (2 * n - 1)
            diag_p[valid] = w_peak[2 * np.arange(n)[valid]] * y[valid]
            grad += gamma * 2.0 * (h_p + diag_p)

        if gamma < 0.995:
            residuals = conv_y - mean_c
            r_rev = residuals[::-1]
            Wr = rfft(r_rev, fft_size)
            conv_r = irfft(Wr * A, fft_size)
            h_r = conv_r[2 * n - 2 - np.arange(n)]
            diag_r = np.zeros(n)
            valid = 2 * np.arange(n) < (2 * n - 1)
            diag_r[valid] = residuals[2 * np.arange(n)[valid]] * y[valid]
            grad += (1.0 - gamma) * 2.0 * (h_r + diag_r)

        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        gamma_nest = (1.0 + np.sqrt(1.0 + 4.0 * gamma_nest * gamma_nest)) / 2.0

        if no_improve > 50:
            lr *= 0.88
            beta = min(beta * 1.35, 400.0)
            no_improve = 0
            gamma_nest = 1.0
            y = a.copy()
            if lr < 1e-12:
                break

        if it % 50 == 0:
            beta = min(beta * 1.005, 400.0)

        if it % 200 == 0:
            print(f"    Homotopy {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, gamma={gamma:.3f}")

    return best_a, best_score

def hotspot_coord_descent(a0, time_limit, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Identify hot spots: positions where conv is close to M
        threshold = M * 0.85
        hot_spots = np.where(conv >= threshold)[0]

        # For each coordinate, compute its "pressure" (contribution to hot spots)
        pressure = np.zeros(n)
        for k in hot_spots:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                pressure[i] += a[i] * a[j] * (conv[k] / M)

        p_max = pressure.max()
        if p_max < 1e-20:
            pressure = np.zeros(n)
        else:
            pressure = pressure / p_max

        # Priority: process high-pressure coordinates first
        order = np.argsort(pressure)[::-1]

        for i in order:
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai * 0.5]

            # Where lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.2 < d < ai * 0.6:
                        candidates.append(d)

            # Crossings of top affected lines
            top_k = min(8, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.2 < d < ai * 0.6:
                            candidates.append(d)

            # Fractional moves biased toward reducing (high pressure) or small increase (low pressure)
            if pressure[i] > 0.5:
                fracs = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
            else:
                fracs = [0.05, 0.1, 0.2, 0.1, 0.05, -0.1, -0.2]
            for frac in fracs:
                candidates.append(-frac * ai)
                candidates.append(frac * ai * 0.3)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-13:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 200 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 200:
                break
        else:
            no_improve_sweeps = 0

        if sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    HotCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def pressure_perturb(a0, rng, strength=0.04):
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.8
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[i] * a[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
