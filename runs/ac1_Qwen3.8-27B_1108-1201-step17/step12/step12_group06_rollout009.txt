def solve_best_response_linprog(a, solver='highs'):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a)
    size = 2 * n - 1
    
    # Build T (size x n)
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    # Variables: [b_0, ..., b_{n-1}, M] (n+1 variables)
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: T @ b - M <= 0
    # => [T, -1] @ [b, M] <= 0
    A_ub = np.column_stack([T, -np.ones(size)])
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1
    # => [1, ..., 1, 0] @ [b, M] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_j >= 0, M free (or M >= 0)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        b_val = result.x[:n]
        M_val = result.x[n]
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val = b_val / s
        return b_val, M_val
    return None, None

    c[-1] = 1.0
    
    # Inequality: T @ b - M <= 0
    if use_sparse and n > 200:
        # Build sparse A_ub
        from scipy.sparse import identity, hstack
        A_ub = hstack([T, -identity(size).tocsc()])
    else:
        A_ub = np.column_stack([T, -np.ones(size)])
    
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        # Fallback to dense
        T_d = build_T_dense(a)
        A_ub_d = np.column_stack([T_d, -np.ones(size)])
        try:
            result = linprog(c, A_ub=A_ub_d, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            return None, None
    
    if result.success:
        b_val = result.x[:n]
        M_val = result.x[n]
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val = b_val / s
        else:
            return None, None
        return b_val, float(M_val)
    return None, None


def nesterov_lp_fast(a0, time_limit, max_iters=1000000, verbose=True):
    """Nesterov-accelerated LP best-response with fast linprog."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_lp = solve_best_response(y)
        if b_new is None:
            continue
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if verbose and (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M={M_lp:.6f}")
                last_print = elapsed
    
    return best_a, best_score


def lse_objective_grad(z_in, T_temp, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_temp * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_temp) * log_sum_exp
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_optimize(a0, time_limit, T_start=100.0, T_min=0.1, n_stages=30):
    """Adaptive LogSumExp L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    T = T_start
    iters_per_stage = max(10, int(300000 / n))
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            # Adaptive temperature
            gap = (sc - best_score) / (best_score + 1e-15)
            if gap > 0.003:
                T = max(T, T * 1.3)
            else:
                T = max(T_min, T * 0.7)
            
            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")
                
        except Exception:
            T = max(T_min, T * 0.8)
    
    return best_a, best_score


def peak_subgradient_fast(a0, time_limit, n_steps=500000):
    """Peak-targeted subgradient with Polyak step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        # Active peaks
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        S = np.dot(g, a)
        grad_z = a * (g - S)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break
        
        # Polyak step
        gap = cur_score - lower_bound
        if gap < 1e-12:
            break
        eta = gap / (grad_norm * grad_norm)
        eta = min(eta, 0.1)
        
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)
        
        # Line search if needed
        if sc_new > cur_score * 1.05:
            for _ in range(4):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new <= cur_score:
                    break
        
        z = z_new
        a = a_new
        
        # Average
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 3000:
            avg_count = 3000
            avg_a = (avg_a * 3000 + a) / 3001
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
        
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
        
        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")
    
    return best_a, best_score


def peak_equalization(a0, time_limit, n_iters=100000):
    """Iteratively equalize the peaks of the autocorrelation.
    
    Idea: If the autocorrelation has peaks at certain lags, we want to
    redistribute mass to reduce the max. The update is:
    a_new[j] = a[j] * exp(-eta * sum_k y_k * a[|k-j|])
    where y_k is a distribution over the active peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        # Define weight distribution over peaks (exponential tilting)
        T = 0.01 * max_b + 1e-10
        w = np.exp(b / T)
        w = w / w.sum()
        
        # Compute the "pressure" at each position
        # pressure[j] = sum_k w[k] * a[k-j] (when valid)
        # This is (w * a reversal) / ... actually it's the convolution of w with reversed a
        # pressure[j] = sum_k w[k] * a[|k-j|] ... hmm not exactly
        # Actually: the gradient of max_b (approximated by softmax) at position j is:
        # grad[j] = 2 * sum_k w[k] * a[k-j] (when 0 <= k-j < n)
        #         = 2 * (w * a_shifted)[j]
        
        # More precisely: (a*a)[k] = sum_j a[j]*a[k-j]
        # d(a*a)[k]/da[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
        # Wait, for the full convolution:
        # (a*a)[k] = sum_{j=0}^{n-1} a[j] * a[k-j] where a[m]=0 if m<0 or m>=n
        # d(a*a)[k]/da[j] = a[k-j] (the term where the derivative hits the first a)
        #                              + a[j] * ... no
        # Actually d/da[j] of sum_i a[i]*a[k-i] = a[k-j] + a[j] (if both terms exist)
        # Wait no: d/da[j] of sum_i a[i]*a[k-i] 
        # = (d a[j]/da[j]) * a[k-j] + sum_{i != j} a[i] * (d a[k-i]/da[j])
        # = a[k-j] + a[j] * (d a[k-j']/da[j] where j = k-i', i.e., i' = k-j)
        # Hmm, this is getting confusing. Let me think again.
        
        # (a*a)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (with a[m]=0 outside [0,n-1])
        # d(a*a)[k]/da[j] = a[k-j] * (d a[j]/da[j]) + a[j] * (d a[k-j]/da[j])
        # Wait, the second term: a[k-i] contributes to d/da[j] only when k-i = j, i.e., i = k-j.
        # So d(a*a)[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] (if 0 <= k-j < n)
        # Wait that's not right either. Let me be very explicit.
        
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # d/da[j]: 
        #   - The term i=j in the sum gives: (da[j]/da[j]) * a[k-j] = a[k-j]
        #   - The term where k-i=j, i.e., i=k-j, gives: a[k-j] * (da[j]/da[j]) = a[k-j]
        #   - But wait, these are the same term if j = k-j, i.e., k=2j!
        #   - So: d(a*a)[k]/da[j] = a[k-j] + a[j-k] (where a[m]=0 if m notin [0,n-1])
        #   - Since a is the same sequence, a[j-k] = a[|j-k|]... no, a is not symmetric in general.
        #   - a[k-j] is valid if 0 <= k-j <= n-1, i.e., j <= k and k-j < n

        #   - So: d(a*a)[k]/da[j] = (a[k-j] if 0<=k-j<n) + (a[j-k] if 0<=j-k<n)
        #   - But since the sum is over i from 0 to n-1, and a[k-i] is 0 if k-i notin [0,n-1]:
        #     d(a*a)[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j] * [1 if exists i!=j with k-i=j]
        #     The second case: k-i=j means i=k-j, and we need i!=j, so k-j!=j, i.e., k!=2j
        #     And we need 0<=k-j<n for the term to exist
        #     So: d(a*a)[k]/da[j] = a[k-j] + a[j] if 0<=k-j<n and k!=2j
        #         d(a*a)[k]/da[j] = 2*a[j] if k=2j (and 0<=j<n)

        # Let me just do it with a matrix product for the active peaks.
        
        # Simple approach: build the matrix implicitly
        # To compute grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j] for j = 0, ..., n-1
        # This is: grad = 2 * (H @ a) where H[j, m] = w[m+j]
        # H is an (n x n) upper-triangular-like matrix.
        # We can build it or use FFT.
        
        # FFT approach: pad a to length L >= 2n-1, pad w to length L
        # Then the linear cross-correlation can be computed via FFT.
        # corr_x[j] = sum_m a[m] * conj(w[m+j]) ... this is for complex.
        # For real: corr[j] = sum_m a[m] * w[m+j] = (a * w_flipped)[j + (2n-2) - (n-1)]
        # Hmm, let me just use scipy.signal.correlate or compute directly.
        
        # Direct vectorized computation:
        # Create a matrix where row j has w[j:j+n]
        # This is a Hankel-like matrix.
        
        # For n=1200, building this (1200x1200 matrix) and doing a matvec is fast.
        # But we do this every iteration... Let me think of a faster way.
        
        # Actually, I realize I can just use the convolution directly:
        # grad[j]/2 = sum_m a[m] * w[m+j]
        # Define b[m] = a[m] for m=0..n-1, 0 otherwise (length L)
        # Define c[k] = w[k] for k=0..2n-2, 0 otherwise (length L)
        # Then grad[j]/2 = sum_m b[m] * c[m+j]
        # This is the cross-correlation of b and c at lag j: (b ⊛ c)(j) = sum_m b[m] c[m+j]
        # = (b * c_flipped)(j + L - 1) where * is linear convolution and c_flipped is c reversed
        # 
        # Let me just use numpy's correlate or scipy.signal:
        
        from scipy.signal import correlate
        # grad[j]/2 = correlate(w, a, mode='valid') ... let me check
        # correlate(w, a, mode='valid')[j] = sum_m w[m] * a[m+j] ... 
        # Hmm, numpy's correlate: correlate(a, v, mode)[k] = sum_n a[n+k] * conj(v[n])
        # So correlate(w, a, mode='full')[k] = sum_n w[n+k] * a[n] ... not quite what I want.
        
        # I want: sum_m a[m] * w[m+j] for m=0..n-1
        # = sum_m a[m] * w[m+j]
        # numpy.correlate(w, a, 'valid') has length max(len(w),len(a)) - min(len(w),len(a)) + 1
        # = (2n-1) - n + 1 = n. 
        # correlate(w, a, 'valid')[j] = sum_m w[m+j] * a[m] for m=0..n-1. YES!
        
        # So: grad = 2 * correlate(w, a, mode='valid')
        # But wait, 'valid' mode requires the first argument to be at least as long as the second,
        # or it swaps them. Since len(w) = 2n-1 >= n = len(a), it should work.
        
        # Actually let me double-check: numpy.correlate(a, v, mode='valid')
        # "The output is the valid part of the cross-correlation, which has length max(M, N) - min(M, N) + 1"
        # where M = len(a), N = len(v).
        # If M >= N: correlate(a, v, 'valid')[k] = sum_{n=0}^{N-1} a[n+k] * v[n]
        # So correlate(w, a, 'valid')[j] = sum_{m=0}^{n-1} w[m+j] * a[m]. YES!
        
        corr_vals = np.correlate(w, a, mode='valid')
        grad = 2.0 * corr_vals
        
        # Project to tangent space of simplex
        S = np.dot(grad, a)
        grad_z = a * (grad - S)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break
        
        # Step size: use a simple decreasing schedule + Polyak
        gap = cur_score - lower_bound
        eta = min(0.5 * gap / (grad_norm * grad_norm + 1e-10), 0.05)
        
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        z = z_new
        a = a_new
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PEQ iter={it+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")
    
    return best_a, best_score
