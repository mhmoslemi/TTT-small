        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_conv_matrix(a):
    """Build the matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def hedge_fw(a0, time_limit, eta=1.0, refine_every=0, solver=None, verbose=True):
    """
    Hedge + Frank-Wolfe with Nesterov.
    Each iteration: O(n log n) via FFT.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Virtual iterate
    num_lags = 2 * n - 1
    q = np.ones(num_lags) / num_lags  # Dual distribution
    n_iters = 0

    # Precompute FFT sizes
    fft_size_auto = 1 << (2 * n - 1).bit_length()
    # For gradient: need FFT of length >= (2n-1) + n - 1 = 3n-2
    fft_size_grad = 1 << (3 * n - 2).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Compute autocorrelation of y
        Y = rfft(y, fft_size_auto)
        conv_yy = irfft(Y * Y, fft_size_auto)[:num_lags]

        # Step 2: Compute gradient g_i = sum_k q_k * (y_{k-i} + y_{i-k})
        # g = ifft((FFT(q) + conj(FFT(q))) * FFT(y_padded))[:n]
        # where y_padded = [y, zeros] of length num_lags
        
        # Actually, let's just use the formula directly:
        # We need FFT of q (length num_lags) and y (length n), padded to same size.
        Qf = rfft(q, fft_size_grad)
        Yf = rfft(y, fft_size_grad)
        g = irfft((Qf + np.conj(Qf)) * Yf, fft_size_grad)[:n]

        # Step 3: Frank-Wolfe vertex
        i_min = int(np.argmin(g))
        v = np.zeros(n)
        v[i_min] = norm

        # Step 4: Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Step 5: Update q via multiplicative weights
        max_conv = conv_yy.max()
        if max_conv > 1e-15:
            log_q = eta * conv_yy / max_conv
            log_q -= log_q.max()
            q_new = np.exp(log_q)
            s_q = q_new.sum()
            if s_q > 1e-15:
                q = q_new / s_q

        # Step 6: Track best
        score_curr = conv_yy.max()
        if score_curr < best_score:
            best_score = score_curr
            best_a = y.copy()

        # Periodic check of a
        if t % 20 == 0:
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

        # Periodic LP refinement
        if refine_every > 0 and t % refine_every == 0 and solver is not None:
            M_curr = conv_yy.max()
            b = lp_best_response(y, M_curr, solver)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    g_dir = b * norm / b_sum
                    # Try blending
                    for blend in [0.3, 0.5, 0.7]:
                        y_trial = (1 - blend) * y + blend * g_dir
                        y_trial = np.maximum(y_trial, 0.0)
                        s_tr = y_trial.sum()
                        if s_tr > 1e-15:
                            y_trial = y_trial * norm / s_tr
                            score_tr = fast_autocorr(y_trial).max()
                            if score_tr < best_score:
                                best_score = score_tr
                                best_a = y_trial.copy()
                                y = y_trial
                                break

        n_iters += 1
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    HedgeFW t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
