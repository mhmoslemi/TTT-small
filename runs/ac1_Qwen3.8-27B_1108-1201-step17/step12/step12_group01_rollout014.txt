"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LSE (log-sum-exp) gradient descent with Adam (fast, high T)
2. LP Nesterov zero-sum game with CLARABEL (precise, main workhorse)
3. High-T LSE for peak-focused refinement
4. Coordinate descent and peak shaving for final polish
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_corr(w, a):
    """Compute sum_k w[k]*a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    N = 1 << ((2 * len(w) - 1) - 1).bit_length()  # at least 4n-3
    W = fft(w, N)
    A = fft(a, N)
    result = ifft(W * np.conj(A))[:n].real
    return result

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def lse_score(a, T):
    """Compute LSE_T of autocorrelation."""
    c = fast_autocorr(a)
    Tc = T * c
    return Tc.max() + np.log(np.exp(Tc - Tc.max()).sum()) / T

def lse_grad(a, T):
    """Compute gradient of LSE_T(autocorr(a)) w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    # grad_a[j] = 2 * sum_k w[k] * a[k-j]
    grad_a = 2.0 * fast_cross_corr(w, a)
    return lse_score(a, T), grad_a

def adam_lse_optimize(a0, time_limit, T_start, T_end, n_stages, n_iter_per_stage):
    """Adam-optimized LSE gradient descent in log-space."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a_init.sum()
    a_init = a_init / s
    z = np.log(a_init)
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        lr = 0.01 * (0.7 ** si)
        
        for it in range(n_iter_per_stage):
            if time.monotonic() - t0 > time_limit - 10:
                break
            
            a = softmax_fn(z)
            _, grad_a = lse_grad(a, T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 50 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam stage {si}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft


def fast_autocorr(a):
    """Full autocorrelation (linear convolution) a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_corr(w, a):
    """Compute cross-correlation: result[j] = sum_k w[k]*a[k-j] (valid range)."""
    n = len(a)
    # Need FFT size >= (2n-1) + n - 1 = 3n-2
    N = 1 << ((3 * n - 2) - 1).bit_length()
    W = fft(np.asarray(w, dtype=np.float64), N)
    A = fft(np.asarray(a, dtype=np.float64), N)
    result = ifft(W * np.conj(A), N)[:n].real
    return result


def softmax_fn(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def solve_lp_best_response(y, solver, verbose=False):
    """Solve LP: min M s.t. conv(y,b) <= M, sum(b)=1, b>=0."""

def lse_optimize(a0, time_limit, T_start, T_end, n_stages, iters_per_stage, lr_scale=1.0):
    """

    """

    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)

    else:

    z = np.log(a_init)
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    base_lr = 0.01 * lr_scale * min(1.0, 50.0 / n)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        lr = base_lr * (0.75 ** si)
        
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            a = softmax_fn(z)
            c = fast_autocorr(a)
            Tc = T * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            
            # Gradient: grad_a[j] = 2 * sum_k w[k]*a[k-j]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 100 == 0:
                a_check = softmax_fn(z)
                sc = compute_score(a_check)

            print(f"    LSE stage {si}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
