import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_topk_grad(a, T, topk):
    """Compute top-k LSE objective and gradient w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    
    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    
    # grad_a[j] = 2 * sum_{k in topk} w_k * a[k-j] (for valid k-j)
    # More efficiently: for each k, add 2*w_k * a[k-j] to grad_a[j]
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        # For position j, the contribution is 2*wK*a[K-j] if 0 <= K-j < n
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    return lse_val, grad_a

def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def adam_optimize_z(a0, time_limit, T_start, T_end, n_stages, topk, lr_scale=1.0, seed=42):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        base_lr = 3e-3 * min(1.0, 100.0 / n) * lr_scale
        lr = base_lr * (0.85 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        # Adaptive max iterations based on remaining time
        # Estimate time per iteration
        it0 = time.monotonic()
        topk_lse_grad_z(z, stage_T, topk)
        time_per_it = time.monotonic() - it0
        max_it = max(5, int(remaining * 0.95 / max(time_per_it, 1e-3)))
        max_it = min(max_it, 5000)
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 5 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                elif (it + 1) % 25 == 0:
                    # Reset to best occasionally to prevent drift
                    z = np.log(np.maximum(best_a, 1e-30))
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
    
    a_final = softmax_fn(z)
    score_final = 2.0 * n * fast_autocorr(a_final).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()
    
    return best_a, best_score

        a = np.abs(1 - 3*x + 3*x**2)  # Not quite right, let's use a better one
        a = np.exp(-4*(x-0.5)**2) * (1 + 0.3*np.cos(8*np.pi*x))
    elif style == 8:
        # Sinusoidal envelope
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1 + 0.2 * np.cos(6 * np.pi * x))
    else:
        # Screenshot-like (smooth with structure)
        x = np.linspace(0, 1, n)
        a = np.exp(-2*(x-0.5)**2) * (0.7 + 0.3*np.cos(2*np.pi*2*x))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def nesterov_lp_optimize_v2(a0, time_limit, max_iters=500000):
    """
    Nesterov-accelerated zero-sum game approach.
    Solves: min_a max_k (conv(a,a))[k] / (sum a)^2
    Using the zero-sum game:
    - Player 1 (row): chooses a (probability vector)
    - Player 2 (column): chooses k (lag) 
    - Payoff: (conv(a,a))[k]
    
    Actually, the proper zero-sum game formulation:
    - Row player chooses distribution p over {1,...,n} (representing a)
    - Column player chooses distribution q over {0,...,2n-2} (representing lag weights)
    - Payoff matrix: M[p,q] = sum_k q[k] * (conv(p,p))[k]
    
    But this isn't bilinear in p and q because conv(p,p) is quadratic in p.
    
    Alternative formulation (from the previous code):
    - Fix a, find optimal g: max sum(g) s.t. conv(a,g) <= M, g >= 0, sum(g)=1
    - This IS linear in g for fixed a
    """
    import cvxpy as cp
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Find solver
    solver = None
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = solver_name
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    # Build matrix T where T @ b = convolve(a_fixed, b) for fixed a_fixed
    # T has shape (2n-1) x n, T[k, j] = a_fixed[k-j] if 0 <= k-j < n else 0
    def build_T(a_fixed):
        n_f = len(a_fixed)
        T = np.zeros((2 * n_f - 1, n_f))
        for j in range(n_f):
            T[j:j + n_f, j] = a_fixed
        return T
    
    # Pre-build T (it changes as a changes, but we rebuild each iteration)
    
    stall_count = 0
    avg = a.copy()
    n_avg = 1
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        # Solve LP: max sum(b) s.t. T(y) @ b <= M, sum(b) = 1, b >= 0
        # where M is the current max of conv(y,y)
        # Actually, the standard approach: find b that maximizes the minimum
        # Let's use: max_b [conv(y,b)]_min s.t. sum(b)=1, b>=0
        # This is: max t s.t. T(y) @ b >= t*ones, sum(b)=1, b>=0
        
        T = build_T(y)
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [T @ b_var >= t_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(t_var), constraints)
        try:
            prob.solve(solver=solver, max_time=2.0)
        except Exception:
            pass
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Fallback: use the argmax of conv(y,y) as a point mass
            c_y = fast_autocorr(y)
            b_star = np.zeros(n)
            # The best response in the game is to put mass where conv(y,.) is max
            # For the LP formulation: max sum(b) s.t. conv(y,b) <= M
            # with M = max(conv(y,y)), the optimal b has mass on positions where
            # conv(y, e_j) is smallest
            # conv(y, e_j)[k] = y[k-j] for valid k
            # sum_k indicator(conv(y,e_j)[k] <= M for all k) ... this is complex
            # Let's just use a simple heuristic
            b_star = np.ones(n) / n
        else:
            b_star = np.asarray(b_var.value, dtype=np.float64)
            b_star = np.maximum(b_star, 0)
            bs = b_star.sum()
            if bs < 1e-15:
                b_star = np.ones(n) / n
            else:
                b_star = b_star / bs
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ys
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Track running average
        avg = avg + (a - avg) / n_avg
        n_avg += 1
        
        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if n_avg % 50 == 0 and n_avg >= 100:
            score_avg = 2.0 * n * fast_autocorr(avg).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg.copy()
                stall_count = 0
        
        # Adaptive restart
        if stall_count > 500 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            avg = best_a.copy()
            n_avg = 1
            stall_count = 0
            if t % 1000 == 0:
                print(f"      NLP restart at t={t}, best={best_score:.6f}")
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def guided_local_search(a0, time_limit, seed=42):
    """
    Local search that uses the gradient structure to guide moves.
    More principled than random peak shaving.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        cmax = c.max()
        peak = c.argmax()
        
        # Gradient of c[peak] w.r.t. a
        # dc[peak]/da[j] = 2*a[peak-j] (approximately, when peak-j is valid and peak != 2j)
        js = max(0, peak - (n - 1))
        je = min(n - 1, peak)
        if js > je:
            # Peak is out of range for this n (shouldn't happen)
            break
        idx = np.arange(js, je + 1)
        grad = 2.0 * a[peak - idx]
        
        # We want to reduce c[peak]. The most effective move is to
        # decrease a at positions with large gradient (large a[peak-j])
        # and increase a at positions with small gradient contribution to
        # ALL peaks (not just the current one)
        
        # Find the "critical" positions (large contribution to peak)
        # and "safe" positions (small contribution to all top peaks)
        top_k = min(20, len(c))
        top_peaks = np.argsort(c)[::-1][:top_k]
        
        # Compute total "danger" for each position
        danger = np.zeros(n)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            idx2 = np.arange(js2, je2 + 1)
            danger[idx2] += c[k2] * a[k2 - idx2]
        
        # Source: position with highest danger
        src_candidates = idx[np.argsort(grad)[::-1][:5]]
        # Destination: position with lowest danger
        dst_candidates = np.argsort(danger)[:max(5, n // 10)]
        dst_candidates = dst_candidates[dst_candidates != src_candidates[0]]
        
        improved = False
        for src in src_candidates:
            for dst in dst_candidates[:3]:
                if src == dst:
                    continue
                for frac in [0.003, 0.008, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random exploration with cooling
            scale = max(0.001, 0.02 / (1 + stall * 0.01))
            a_trial = a.copy()
            n_pert = rng.integers(1, 4)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    amt = a_trial[i] * rng.uniform(0.001, scale)
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial /= st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall < 50:
                    # Accept to explore (simulated annealing style)
                    T = max(0.001, 0.05 * np.exp(-stall * 0.05))
                    if rng.random() < np.exp(-(score - best_score) / T):
                        a = a_trial
            
            if stall > 300:
                break
    
    return best_a, best_score
