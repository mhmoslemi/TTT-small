"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using asymmetric initializations and
mirror descent with FFT-accelerated subgradients. Key insight: asymmetric
sequences (e.g., linear ramps) break the symmetric lower bound of 2, allowing
scores well below 2. Uses diverse asymmetric starts, adaptive mirror descent,
frequency-domain initialization, peak shaving, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_conv(a):
    """Compute full linear convolution of a with itself via FFT. Length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_from_peaks(a, b, top_k, weights=None):
    """
    Compute subgradient of max b[k] w.r.t. a.
    b = conv(a,a), length 2n-1.
    g[i] = 2 * sum_m w_m * a[k_m - i] (with a[j]=0 for j out of [0,n)).
    """
    n = len(a)
    if weights is None:
        top_idx = np.argsort(b)[::-1][:top_k]
        weights = np.ones(top_k) / top_k
    else:
        top_idx = np.argsort(b)[::-1][:top_k]

    g = np.zeros(n)
    for ii in range(top_k):
        m = int(top_idx[ii])
        w = 2.0 * weights[ii]
        # g[i] += w * a[m-i] for valid (0 <= m-i < n), i.e., m-n+1 <= i <= m
        i_start = max(0, m - n + 1)
        i_end = min(n - 1, m)
        if i_start > i_end:
            continue
        indices = np.arange(i_start, i_end + 1)
        g[indices] += w * a[m - indices]
    return g


def mirror_descent_asym(a0, time_limit_s, eta0=0.08, decay=0.55,
                        k_start=1, k_max=20, k_ramp=500,
                        noise_every=400, seed=42, verbose=True):
    """
    Mirror descent (multiplicative weights) on the simplex for minimizing
    max conv(a,a). Uses asymmetric-friendly settings.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    last_best = best_score

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit_s:
            break

        b = fast_conv(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        frac = min(1.0, t / float(k_ramp))
        k = max(k_start, min(k_max, int(k_start + frac * (k_max - k_start))))

        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx] - b_mean
        if excess.max() > 1e-12:
            es = excess - excess.max()
            weights = np.exp(es) / np.exp(es).sum()
        else:
            weights = np.ones(k) / k

        g = subgradient_from_peaks(a, b, k, weights)

        eta = eta0 / (1.0 + t / 200.0) ** decay

        a_new = a * np.exp(-eta * g)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a = a_new / s_new

        # Periodic score check
        if t < 200:
            interval = 2
        elif t < 2000:
            interval = 5
        else:
            interval = 15

        if t % interval == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Noise injection to escape local optima
        if stall > 0 and t % noise_every == 0 and stall > noise_every // 3:
            ns = min(0.04, 0.005 * (stall / 50.0))
            a = a + ns * rng.standard_normal(n)
            a = normalize(a)
            stall = 0

        if verbose and t % 10000 == 0:
            el = time.monotonic() - t0
            print(f"      MD t={t} ({el:.0f}s): best={best_score:.6f}, bmax={b_max:.4e}, k={k}")

    return best_a, best_score


def lse_optimize(a0, time_limit_s, T_start=50.0, T_end=0.3, n_stages=10, seed=42):
    """LogSumExp-smoothed optimization via L-BFGS with annealing."""
    from scipy.optimize import minimize

    n = len(a0)
    a_init = normalize(a0.copy())
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit_s:
            break
        remaining = time_limit_s - (time.monotonic() - t0)
        if remaining < 3:
            break

        max_iter = max(10, min(60, int(15 + 45 * si / max(1, n_stages))))

        def func(z_in):
            z_s = z_in - z_in.max()
            e = np.exp(z_s)
            a = e / e.sum()
            b = fast_conv(a)
            Tb = T * b
            Tb_s = Tb - Tb.max()
            return Tb.max() + np.log(np.exp(Tb_s).sum())

        def jac(z_in):
            z_s = z_in - z_in.max()
            e = np.exp(z_s)
            a = e / e.sum()
            b = fast_conv(a)
            Tb = T * b
            Tb_s = Tb - Tb.max()
            exp_Tb = np.exp(Tb_s)
            w = exp_Tb / exp_Tb.sum()
            # grad_a[i] = 2 * sum_k w[k] * a[k-i]
            # Use cross-correlation via FFT
            a_rev = a[::-1].copy()
            conv_len = n + len(w) - 1
            fft_sz = 1 << (conv_len - 1).bit_length()
            W = rfft(w, fft_sz)
            AR = rfft(a_rev, fft_sz)
            cc = irfft(W * AR, fft_sz)
            grad_a = 2.0 * cc[n - 1:n - 1 + n]
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            return grad_z

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-15,
                                    'gtol': 1e-9, 'maxfun': max_iter * 300,
                                    'disp': False})
            z = res.x
            z_s = z - z.max()
            a = np.exp(z_s)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 3 == 0:
            el = time.monotonic() - t0
            print(f"      LSE stage {si+1}/{n_stages} (T={T:.1f}, {el:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit_s, seed=123):

    n = len(a0)
    rng = np.random.default_rng(seed)

    stall = 0

        top_idx = np.argsort(b)[::-1][:min(8, 2 * n - 1)]
        improved = False

        for k_idx in top_idx:
            k_idx = int(k_idx)
            i_start = max(0, k_idx - n + 1)
            i_end = min(n - 1, k_idx)
            idx = np.arange(i_start, i_end + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_idx - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue

            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            # Find target: low-value positions far from the peak
            peak_region = np.zeros(n, dtype=bool)
            for k2 in top_idx[:4]:
                k2 = int(k2)
                ps = max(0, k2 - n + 1)
                pe = min(n - 1, k2)
                peak_region[ps:pe + 1] = True

            candidates = np.where(~peak_region)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]
            sorted_c = candidates[np.argsort(a[candidates])[:max(3, len(candidates) // 5)]]
            if len(sorted_c) == 0:
                continue
            dst = int(sorted_c[int(rng.integers(len(sorted_c)))])

            for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(4) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:

                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 300:
                break

    return best_a, best_score


def freq_domain_init(n, rng, variant=0):
    """

    """
    if variant == 0:

        a = x + 1.0

        x = np.arange(n)
        a = (n - x) + 1.0
    elif variant == 2:
        # Quadratic ramp
        x = np.arange(n, dtype=np.float64) / n
        a = x ** 2 + 0.1
    elif variant == 3:
        # Exponential ramp
        x = np.arange(n, dtype=np.float64) / n
        a = np.exp(3.0 * x)
    elif variant == 4:
        # Double bump (asymmetric)
        x = np.arange(n, dtype=np.float64) / n
        a = np.exp(-30 * (x - 0.3) ** 2) + 0.5 * np.exp(-20 * (x - 0.7) ** 2)
    elif variant == 5:
        # S-curve (sigmoid)
        x = np.arange(n, dtype=np.float64) / n
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    elif variant == 6:
        # Cosine taper (asymmetric peak)
        x = np.arange(n, dtype=np.float64)
        a = np.cos(np.pi * x / (2 * n)) ** 2 + 0.05
    elif variant == 7:
        # Power law
        x = np.arange(n, dtype=np.float64) / (n - 1)
        a = (x + 0.1) ** 1.5
    elif variant == 8:
        # Asymmetric Gaussian
        x = np.arange(n, dtype=np.float64) / n
        a = np.exp(-15 * (x - 0.4) ** 2) + 0.3 * np.exp(-10 * (x - 0.8) ** 2)
    else:

        a = np.ones(n)
        for _ in range(8):
            c = rng.uniform(0, n)
            w = rng.uniform(n * 0.1, n * 0.4)

def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        t_alloc = min(remaining * 0.06, 80.0)
        if t_alloc < 10:
            break
        init = freq_domain_init(n_cur, rng, variant=vi)
        # Perturb with current best to create a blend
        a_blend = 0.5 * init + 0.5 * best_seq
        a_blend = normalize(a_blend)
        a_out, _ = mirror_descent_asym(a_blend, t_alloc, eta0=0.03, decay=0.5,
                                        k_start=2, k_max=15, k_ramp=200,
                                        noise_every=250, seed=400 + vi, verbose=False)
        try_update(a_out, f"blend_v{vi}_n{n_cur}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None:
        # Short MD with high k for smooth refinement
        t_f1 = min(remaining * 0.20, 60.0)
        if t_f1 > 10:
            a_out, _ = mirror_descent_asym(best_seq, t_f1, eta0=0.01, decay=0.7,
                                            k_start=8, k_max=30, k_ramp=50,
                                            noise_every=500, seed=555, verbose=False)
            try_update(a_out, "final_md")

        # Very low temp LSE
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.25, 80.0)
        if t_f2 > 10:
            a_out, _ = lse_optimize(best_seq, t_f2, T_start=10.0, T_end=0.05,
                                     n_stages=8, seed=333)
            try_update(a_out, "final_lse")

        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.35, 100.0)
        if t_f3 > 5:
            a_out, _ = peak_shave(best_seq, t_f3, seed=999)
            try_update(a_out, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        # Fallback: linear ramp
        n_fallback = 1600
        best_seq = freq_domain_init(n_fallback, rng, variant=0)
        best_score = fast_score(best_seq)
        print(f"FINAL (fallback): score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
