"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast MWU-Nesterov for rapid exploration
2. LP-based Nesterov with adaptive restarts (main precise optimizer)
3. LSE L-BFGS multi-stage annealing for fine polishing
4. Multi-resolution and length exploration
5. Random perturbation search for escaping local minima
Key improvements over previous: MWU fast iterations, adaptive LP frequency,

def fast_convolve_self(a):
    """Compute auto-convolution of a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve2(a, b):
    """Compute convolution of a and b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve_self(a)
    return 2.0 * n * b.max()

    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:

                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def mwu_nesterov(a0, time_limit, K=50, eta_init=0.3, max_iters=5000000):
    """Fast MWU-Nesterov optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    b = a.copy()  # Initialize b = a
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta_b = eta_init
    eta_a = eta_init
    last_improve = 0
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if (it + 1) % 10000 == 0 and (it + 1) - last_improve > 50000:
            break

        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        # Update b: MWU to minimize max conv(y, b)
        c_yb = fast_convolve2(y, b)
        if K < len(c_yb):
            topk_idx = np.argpartition(c_yb, -K)[-K:]
        else:
            topk_idx = np.arange(len(c_yb))
        topk_vals = c_yb[topk_idx]
        w = np.exp(topk_vals - topk_vals.max())
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(topk_idx)) / len(topk_idx)
        else:
            w = w / w_sum

        # Gradient for b
        grad_b = np.zeros(n)
        for k_idx in range(len(topk_idx)):
            k = topk_idx[k_idx]
            wk = w[k_idx]
            if wk < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                grad_b[idx] += wk * y[k - idx]

        exponent_b = -eta_b * grad_b
        exponent_b -= exponent_b.max()
        b_new = b * np.exp(exponent_b)
        sb = b_new.sum()
        if sb < 1e-300:
            eta_b *= 0.5
            continue
        b_new = b_new / sb

        # Update a: Nesterov step toward b_new
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            last_improve = it

        # Adaptive restart
        if sc > best_score * 1.002 and t_nest > 10.0:
            t_nest = 1.0
        else:
            t_nest = t_new

        a_prev = a
        a = a_new
        b = b_new

        # Decay eta
        if (it + 1) % 1000 == 0:
            eta_b *= 0.99
            eta_a *= 0.99

        if (it + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 5:
                print(f"    MWU iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f}")
                last_print = elapsed

    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    """Compute LSE objective and gradient in z-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve_self(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def random_perturbation_search(a0, time_limit, n_trials=50000, sigma_range=(0.001, 0.1)):
    """Random perturbation search to escape local minima."""
    n = len(a0)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive sigma
        sigma = rng.uniform(sigma_range[0], sigma_range[1])
        # Random direction
        direction = rng.standard_normal(n)
        a_pert = best_a + sigma * direction
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 0.01:
            continue
        a_pert = a_pert / s

        sc = score_normalized(a_pert)
        if sc < best_score:
            best_score = sc
            best_a = a_pert.copy()

    return best_a, best_score
