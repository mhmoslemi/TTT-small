```
    y = b + momentum * (b - b_prev)
    y = proj_simplex(y)  # ensure y is in simplex
    
    # Compute subgradient at y
    c = conv(a_fixed, y)
    k* = argmax(c)
    g[j] = a_fixed[k* - j] if 0 <= k* - j < n else 0
    
    # Gradient step
    L = ||a_fixed||_2
    b_new = proj_simplex(y - (1/L) * g)
    
    b_prev = b
    b = b_new
```

This is O(n) per step (the convolution and the projection are both O(n) or O(n log n)).

```python
import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def get_lp_solver():
    """Find a working LP solver."""
    try:
        import cvxpy as cp
        for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
        return cp.ECOS
    except ImportError:
        return None


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0. Returns b*, M."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    n = len(a0)

        else:
            y_new = y_new / y_sum
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)

            else:

        y = y_new
        a = a_new
        avg_long += a

            if score_a < best_score:
                best_score = score_a

def nesterov_subgradient_optimize(a0, time_limit):
    """
    Nesterov-accelerated subgradient method for min_b max_k conv(a_fixed, b)_k.
    This is the "inner" problem of the self-play game, solved without LP.
    Each step is O(n log n) due to FFT convolution and simplex projection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a_fixed = a.copy()
    L = np.linalg.norm(a_fixed, 2)
    if L < 1e-15:
        L = 1.0 / np.sqrt(n)

    b = a.copy()
    b_prev = b.copy()
    best_b = b.copy()
    best_val = fast_autocorr(b).max()
    t0 = time.monotonic()
    avg = np.zeros(n)
    n_avg = 0
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        momentum = (t - 1) / (t + 2)
        y = b + momentum * (b - b_prev)
        y = project_simplex(y)

        conv_y = fast_autocorr(y)
        k_star = np.argmax(conv_y)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        g = np.zeros(n)
        g[js:je + 1] = a_fixed[k_star - np.arange(js, je + 1)]

        step_size = 1.0 / (L * np.sqrt(2.0 * t))
        b_new = project_simplex(y - step_size * g)

        b_prev = b
        b = b_new
        avg += b
        n_avg += 1

        if t % 50 == 0:
            val = fast_autocorr(b).max()
            if val < best_val:
                best_val = val
                best_b = b.copy()
                stall = 0
            else:
                stall += 1
            if n_avg % 200 == 0 and n_avg >= 400:
                avg_b = avg / n_avg
                val_avg = fast_autocorr(avg_b).max()
                if val_avg < best_val:
                    best_val = val_avg
                    best_b = avg_b.copy()
                    stall = 0

        if stall > 5000 and t > 10000:
            b = best_b.copy()
            b_prev = best_b.copy()
            avg = np.zeros(n)
            n_avg = 0
            stall = 0

        if t % 50000 == 0:
            print(f"    NSG t={t}: best={best_val:.8f}")

    return best_b, best_val


def multiplicative_weights_optimize(a0, time_limit, lr_init=0.5, decay=0.99):
    """
    Multiplicative weights (mirror descent with entropy) for the self-play game.
    Update: a_new[i] = a[i] * exp(-lr * g[i]) / Z
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    avg = np.zeros(n)
    n_avg = 0
    lr = lr_init
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_a = fast_autocorr(a)
        k_star = np.argmax(conv_a)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        g = np.zeros(n)
        g[js:je + 1] = a[k_star - np.arange(js, je + 1)] * 2.0

        log_w = np.log(np.maximum(a, 1e-300)) - lr * g / np.linalg.norm(g, 2)
        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new = w / w_sum

        a = a_new
        avg += a
        n_avg += 1
        lr *= decay

        if t % 30 == 0:
            val = fast_autocorr(a).max()
            if val < best_score:
                best_score = val
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if n_avg % 300 == 0 and n_avg >= 600:
                avg_a = avg / n_avg
                val_avg = fast_autocorr(avg_a).max()
                if val_avg < best_score:
                    best_score = val_avg
                    best_a = avg_a.copy()
                    stall = 0

        if stall > 3000 and t > 5000:
            a = best_a.copy()
            avg = np.zeros(n)
            n_avg = 0
            lr = lr_init * 0.5
            stall = 0

        if t % 100000 == 0:
            print(f"    MW t={t}: best={best_score:.8f}")

    return best_a, best_score


def peak_equalization_polish(a0, time_limit):
    """
    Direct peak equalization: repeatedly reduce the maximum peak by
    moving mass from high-contributing positions to low-impact positions.
    Uses exact subgradient with adaptive step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    prev_grad = None
    prev_b = None

    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_a = fast_autocorr(a)
        k_star = np.argmax(conv_a)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        g = np.zeros(n)
        g[js:je + 1] = 2.0 * a[k_star - np.arange(js, je + 1)]

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_b is not None:
            s_vec = a - prev_b
            y_vec = g - prev_grad
            ss = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            if abs(sy) > 1e-30:
                lr = ss / max(sy, 1e-30)
                lr = np.clip(lr, 1e-10, 1.0)
            else:
                lr = 0.01
        else:
            lr = 0.01

        prev_grad = g.copy()
        prev_b = a.copy()

        a_new = project_simplex(a - lr * g)

        # Occasionally add small random perturbation to escape plateaus
        if it % 5000 == 0 and stall > 500:
            noise = rng.normal(0, 0.001, n)
            a_new = project_simplex(a_new + noise)

        a = a_new
        val = fast_autocorr(a).max()
        if val < best_score:
            best_score = val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            a = best_a.copy()
            stall = 0
            # Perturb to escape
            noise = rng.normal(0, 0.01, n)
            a = project_simplex(a + noise * a)

        if it % 100000 == 0:
            print(f"    PE it={it}: best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):

        try_update(a_out, f"pe_n{n_base}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Multi-length (budget {remaining:.0f}s) ===")
    if remaining > 60:
        for n_try in [n_base + 100, n_base + 200, n_base + 400]:
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            t_each = min(remaining * 0.25, 150.0)
            if t_each < 15:
                break
            print(f"  Trying n={n_try} ({t_each:.0f}s)")
            init_long = interpolate_sequence(best_seq, n_try)
            # Run fast Nesterov subgradient
            a_out, _ = nesterov_subgradient_optimize(init_long, t_each * 0.6)
            try_update(a_out, f"nsg_n{n_try}")
            # Quick peak equalization
            t_pe = min(remaining * 0.1, 60.0)
            if t_pe > 10:
                a_out2, _ = peak_equalization_polish(a_out, t_pe)
                try_update(a_out2, f"pe_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global_start)

    # === Phase 6: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        n_final = len(best_seq)
        print(f"=== Phase 6: Final Polish n={n_final} (budget {remaining:.0f}s) ===")
        t6 = min(remaining * 0.5, 150.0)
        if t6 > 10:
            # Alternate between subgradient and peak equalization
            t_half = t6 / 2
            a_out, _ = nesterov_subgradient_optimize(best_seq, t_half)
            try_update(a_out, "final_nsg")
            a_out, _ = peak_equalization_polish(best_seq, t_half)
            try_update(a_out, "final_pe")

    # === Phase 7: Second round of LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        n_final = len(best_seq)
        print(f"=== Phase 7: Second LP round n={n_final} (budget {remaining:.0f}s) ===")
        t7 = min(remaining * 0.7, 200.0)
        if t7 > 30:
            solver = get_lp_solver()
            if solver is not None:
                a_out, _ = nesterov_lp_optimize(best_seq, t7, solver)
                try_update(a_out, f"lp2_n{n_final}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
