def fast_conv(a):

    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]


def fast_score(a):
    """2n * max(conv(a,a)) / (sum(a))^2."""

def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > cv - s
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)


def lse_obj_grad_a(a, T):
    """LSE objective and gradient w.r.t. a (unconstrained)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (where indices valid)
    # This is 2 * cross-correlation of w (length 2n-1) with a (length n)
    # Using FFT:
    fs = 1 << (n + len(w) - 1).bit_length()
    W = rfft(w, fs)
    Ar = rfft(a[::-1], fs)
    cc = irfft(W * Ar, fs)
    grad_a = 2.0 * cc[n - 1:n - 1 + n]
    return obj, grad_a


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in z-space (a = softmax(z))."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    obj, grad_a = lse_obj_grad_a(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Polynomial warm-start for smoother optimization landscape
2. Nesterov-accelerated mirror descent with LSE-smoothed subgradient
3. Multi-restart LSE + L-BFGS with diverse temperature schedules
4. Simulated annealing for local optimum escape

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

_C_T = 1.0


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]


def fast_score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * fast_conv(a).max() / (s * s)

def proj_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > cv - s
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)


def _cross_corr(w, a):
    n = len(a)
    fs = 1 << (n + len(w) - 1).bit_length()
    W = rfft(w, fs)
    Ar = rfft(a[::-1], fs)
    cc = irfft(W * Ar, fs)
    return cc[n - 1:n - 1 + n]


def lse_obj_z(z, T):
    n = len(z)
    zs = z - z.max()
    e = np.exp(zs)
    a = e / e.sum()
    b = fast_conv(a)
    Tb = T * b
    Ts = Tb - Tb.max()
    eT = np.exp(Ts)
    obj = (Tb.max() + np.log(eT.sum())) / T
    w = eT / eT.sum()
    ga = 2.0 * _cross_corr(w, a)
    S = np.dot(ga, a)
    gz = a * (ga - S)
    return obj, gz


def _lse_func(z):
    return lse_obj_z(z, _C_T)[0]


def _lse_jac(z):
    return lse_obj_z(z, _C_T)[1]


def nesterov_md(a0, time_limit, eta0=0.04, decay=0.55, T_lse=3.0,
                momentum=0.85, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = proj_simplex(a, 1.0)
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    y = a.copy()

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_conv(y)
        Tb = T_lse * b
        Ts = Tb - Tb.max()
        eT = np.exp(Ts)
        w = eT / eT.sum()
        grad_y = 2.0 * _cross_corr(w, y)
        eta = eta0 / (1.0 + t / 400.0) ** decay
        a_new = y * np.exp(-eta * grad_y)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / max(a_new.sum(), 1e-15)
        y = proj_simplex((1 + momentum) * a_new - momentum * a, 1.0)
        a = a_new
        interval = 2 if t < 200 else (5 if t < 2000 else 15)
        if t % interval == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        if stall > 400:
            ns = min(0.04, 0.003 * stall / 50.0)
            a = proj_simplex(a + ns * rng.standard_normal(n), 1.0)
            y = a.copy()
            stall = 0
        if t % 5000 == 0:
            print(f"    NMD t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def lse_lbfgs(a0, time_limit, T_start=60.0, T_end=0.2, n_stages=15, seed=42):
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a_init = proj_simplex(a_init, 1.0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)
    global _C_T
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        _C_T = float(temps[si])
        iters = max(10, min(50, int(10 + 40 * si / n_stages)))
        try:
            res = minimize(_lse_func, z, jac=_lse_jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': iters * 200})
            z = res.x
            zs = z - z.max()
            a = np.exp(zs)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass
        if si % 4 == 0:
            print(f"    LSE {si+1}/{n_stages} (T={float(temps[si]):.1f}): best={best_score:.6f}")
    return best_a, best_score


def simulated_annealing(a0, time_limit, T_init=0.005, T_end=1e-7, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = proj_simplex(a, 1.0)
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    n_iter = 0
    for step in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, step / 50000.0)
        T_cur = T_init * (T_end / T_init) ** progress
        scale = max(0.001, 0.01 * (1.0 - progress))
        idx = rng.integers(0, n, size=min(4, n))
        a_trial = a.copy()
        for i in idx:
            delta = scale * rng.standard_normal()
            a_trial[i] = max(0, a_trial[i] + delta)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        sc_trial = fast_score(a_trial)
        sc_cur = fast_score(a)
        if sc_trial < sc_cur:
            a = a_trial
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a_trial.copy()
        else:
            accept_prob = np.exp(-(sc_trial - sc_cur) / max(T_cur, 1e-12))
            if rng.random() < accept_prob:
                a = a_trial
        n_iter += 1
        if step % 5000 == 0:
            print(f"    SA step={step} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")
    return best_a, best_score

def interp_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi

def poly_warm_start(a0, time_limit, degree=6, seed=42):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / max(a.sum(), 1e-15)
    x = np.linspace(0, 1, n)
    coeffs = np.polyfit(x, a, degree)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_score(a)
    global _C_T
    for T in [20.0, 5.0, 1.0, 0.3]:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break
        _C_T = T

        def _poly_obj(c_in):
            p = np.poly1d(c_in)
            a_t = np.maximum(p(x), 0)
            s = a_t.sum()
            if s < 1e-10:
                return 1e10
            a_t = a_t / s
            b = fast_conv(a_t)
            Tb = _C_T * b
            Ts = Tb - Tb.max()
            eT = np.exp(Ts)
            return (Tb.max() + np.log(eT.sum())) / _C_T

        c_fit = coeffs.copy()
        for it in range(3):
            try:
                res = minimize(_poly_obj, c_fit, method='Nelder-Mead',
                              options={'maxiter': 300, 'xatol': 1e-10,
                                       'fatol': 1e-12, 'disp': False})
                c_fit = res.x
                p = np.poly1d(c_fit)
                a_t = np.maximum(p(x), 0)
                s = a_t.sum()
                if s > 1e-10:
                    a_t = a_t / s
                    sc = fast_score(a_t)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_t.copy()
            except Exception:
                break
        print(f"    Poly T={T}: score={best_score:.6f}")
    return best_a, best_score
