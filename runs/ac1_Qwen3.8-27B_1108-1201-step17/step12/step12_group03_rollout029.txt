Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric shape exploration (fast initial search)
2. Mirror descent with smoothed subgradient (fast pre-optimization)
3. Nesterov-accelerated Frank-Wolfe with LP best response (scipy HiGHS)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Compute the objective score locally."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_autocorr(a)
    return float(2.0 * n * c.max() / (s ** 2))


def normalize(a):
    """Normalize to simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def build_T(a):
    """Build (2n-1) x n convolution matrix where T@b = conv(a,b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_response(a):
    """Solve LP: min M s.t. conv(a,b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a)
    T = build_T(a)
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8})
        if res.success:
            b_opt = np.maximum(res.x[:n], 0.0)
            s = b_opt.sum()
            if s < 1e-15:
                return None, float('inf')
            return b_opt / s, float(res.x[n])
    except Exception:
        pass
    return None, float('inf')


def solve_lp_restricted(a, k_max_constraints=500):
    """Solve LP with only top-k constraints (faster, approximate)."""
    n = len(a)
    T = build_T(a)
    conv_vals = T @ (np.ones(n) / n)
    top_k = np.argsort(conv_vals)[::-1][:k_max_constraints]
    T_red = T[top_k, :]
    A_ub = np.hstack([T_red, -np.ones((len(top_k), 1))])
    b_ub = np.zeros(len(top_k))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3})
        if res.success:
            b_opt = np.maximum(res.x[:n], 0.0)
            s = b_opt.sum()
            if s < 1e-15:
                return None, float('inf')
            return b_opt / s, float(res.x[n])
    except Exception:
        pass
    return None, float('inf')


def mirror_descent_fast(a0, time_budget, lr_init=0.05, n_peak_smooth=15):
    """Mirror descent with smoothed subgradient on the simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    n_avg = 0

    while time.monotonic() - t0 < time_budget:
        it = n_avg
        lr = lr_init / (1.0 + 0.003 * it)

        c = fast_autocorr(a)
        max_c = c.max()

        n_peaks = min(n_peak_smooth, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:n_peaks]
        weights = np.exp((c[top_k] - max_c) * 25.0)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += w * a[other]
            grad[other] += w * a[idx]
        grad *= 2.0

        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= lr * grad
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()

        avg_a += a
        n_avg += 1

        if n_avg % 50 == 0:
            sc = eval_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if n_avg >= 300 and n_avg % 200 == 0:
                sc_avg = eval_local(avg_a / n_avg)

                    best_score = sc_avg
                    best_a = (avg_a / n_avg).copy()

        sc_fin = eval_local(avg_a / n_avg)
        if sc_fin < best_score:
            best_score = sc_fin
            best_a = (avg_a / n_avg).copy()

    return best_a, best_score


def nesterov_fw(a0, time_budget, verbose=True, use_restricted_lp=False):
    """Nesterov-accelerated Frank-Wolfe with LP best response."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    stall = 0
    lp_fails = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_budget:
            break

        if use_restricted_lp:
            b_star, M_val = solve_lp_restricted(y, k_max_constraints=400)
        else:
            b_star, M_val = solve_lp_response(y)

        if b_star is None:
            lp_fails += 1
            if lp_fails > 50:
                break
            continue
        lp_fails = 0

        gamma_t = 2.0 / (t + 1.0)
        y_new = (1.0 - gamma_t) * y + gamma_t * b_star
        y_new = normalize(y_new)

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize(a_new)

        y = y_new
        a = a_new

        sc_a = eval_local(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        else:
            sc_y = eval_local(y)
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
                stall = 0
            else:
                stall += 1

        if stall > 300:
            a = y.copy()
            stall = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def qp_refine(a0, time_budget, n_active_max=25):
    """QP refinement on active constraint set using cvxpy."""
    import cvxpy as cp

    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    n_active = 12

    for rnd in range(100):
        if time.monotonic() - t0 > time_budget:
            break

        c = fast_autocorr(a)
        max_c = c.max()
        threshold = max_c * 0.998
        active = np.where(c >= threshold)[0]
        if len(active) > n_active_max:
            active = active[np.argsort(c[active])[::-1][:n_active_max]]

        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, M_var >= 0]

        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other])
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        for slv in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=slv, max_iter=20000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = normalize(a_new)

        sc = eval_local(a_new)
        improved = False
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            improved = True
            print(f"    QP rnd {rnd}: {best_score:.6f}")
        else:
            for frac in [0.2, 0.4, 0.6, 0.8]:
                a_mix = normalize((1.0 - frac) * a + frac * a_new)
                sc_mix = eval_local(a_mix)
                if sc_mix < best_score:
                    best_score = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    improved = True
                    break

        if not improved:
            n_active = min(n_active * 2, 50)
            if rnd > 8:
                break

    return best_a, best_score

        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]

        ci = int(np.argmax(contrib))
        i1 = int(idx[ci])
        i2 = k_max - i1

        top_peaks = np.argsort(c)[::-1][:10]
        peak_positions = set()
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            for p in range(js2, je2 + 1):
                peak_positions.add(int(p))

        best_move_score = best_score
        best_move = None

        n_tries = 30
        for _ in range(n_tries):
            if i1 == i2:
                src = i1
                frac = rng.uniform(0.003, 0.015)
            else:
                if a[i1] >= a[i2]:
                    src = i1
                else:
                    src = i2
                frac = rng.uniform(0.003, 0.01)

            amt = a[src] * frac
            if amt < 1e-15:
                continue

            for _ in range(8):
                dst = int(rng.integers(0, n))
                if dst == src:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = eval_local(a_trial)
                if sc < best_move_score:
                    best_move_score = sc
                    best_move = a_trial.copy()

        if best_move is not None:
            a = best_move
            best_score = best_move_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                n_pert = int(rng.integers(2, 6))
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for s, d in zip(sources, targets):
                    frac = rng.uniform(0.002, 0.01)
                    a_trial[s] -= a_trial[s] * frac
                    a_trial[d] += a_trial[s] * frac
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = eval_local(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Linear interpolation of a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def parametric_search(n, time_budget):
    """Try parametric families and return best."""
    rng = np.random.default_rng(7)
    best_a = None
    best_score = float('inf')
    t0 = time.monotonic()

    # Cosine power family: a[i] = cos^p(pi*i/n)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0, 8.0]:
        if time.monotonic() - t0 > time_budget:
            break
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** p
        a = normalize(a)
        sc = eval_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Gaussian family: a[i] = exp(-beta*(i/n - 0.5)^2)
    for beta in [0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0]:
        if time.monotonic() - t0 > time_budget:
            break
        x = np.linspace(0, 1, n)
        a = np.exp(-beta * (x - 0.5) ** 2)
        a = normalize(a)
        sc = eval_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Raised cosine with power: a[i] = (1 - (2*i/n - 1)^2)^p
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        if time.monotonic() - t0 > time_budget:
            break
        x = np.linspace(0, 1, n)
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 2, 0.0) ** p
        a = normalize(a)
        sc = eval_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Modulated: a[i] = (1 + alpha*cos(2*pi*i/n))^+
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        if time.monotonic() - t0 > time_budget:
            break
        x = np.linspace(0, 2 * np.pi, n, endpoint=False)
        a = np.maximum(1.0 + alpha * np.cos(x), 0.0)
        a = normalize(a)
        sc = eval_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Two-humped: a[i] = 1 + alpha*cos(4*pi*i/n)
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        if time.monotonic() - t0 > time_budget:
            break
        x = np.linspace(0, 2 * np.pi, n, endpoint=False)
        a = np.maximum(1.0 + alpha * np.cos(2.0 * x), 0.0)
        a = normalize(a)
        sc = eval_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score
