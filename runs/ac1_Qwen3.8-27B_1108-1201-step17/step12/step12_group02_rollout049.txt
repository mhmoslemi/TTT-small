import numpy as np
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)

def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def peak_pressure_cd(a0, time_limit, seed=42):
    """
    Coordinate descent guided by 'peak pressure'.
    For each element a[j], compute the total 'pressure' from all peaks it contributes to.
    Move mass from high-pressure elements to low-pressure elements.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        b_max = b.max()
        
        # Compute pressure on each a[j]:
        # a[j] contributes to conv[m] for m in [j, j+n-1] (capped at [0, 2n-2])
        # with weight a[m-j]
        # pressure[j] = sum_m: valid m, b[m] > threshold * max(b) => a[m-j]
        # Or more simply: pressure[j] = sum_m weights[m] * a[m-j] where weights are based on peak heights
        
        # Use top-20% of peaks
        threshold = np.percentile(b, 80)
        active_peaks = np.where(b > threshold)[0]
        
        pressure = np.zeros(n)
        for m in active_peaks:
            m = int(m)
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            weight = b[m] / b_max  # Normalize by max
            pressure[js] += weight * a[m - js]
        
        # Find high-pressure and low-pressure elements
        p_thresh_high = np.percentile(pressure, 85)
        p_thresh_low = np.percentile(pressure, 15)
        high_idx = np.where(pressure > p_thresh_high)[0]
        low_idx = np.where(pressure < p_thresh_low)[0]
        
        if len(high_idx) == 0 or len(low_idx) == 0:
            stall += 1
            if stall > 100:
                break
            # Random move
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.01
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            continue
        
        # Try moving mass from high to low pressure
        improved = False
        for _ in range(20):
            src = high_idx[rng.integers(len(high_idx))]
            dst = low_idx[rng.integers(len(low_idx))]
            if a[src] < 1e-10:
                continue
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score
