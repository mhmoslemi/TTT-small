The LP is: max `sum(b)` s.t. `T @ b <= M * 1`, `b >= 0`.

The dual is: min `M * sum(lambda)` s.t. `T^T @ lambda >= 1`, `lambda >= 0`.

By strong duality, at the optimum: `sum(b_i) = M * sum(lambda_k)`.

The KKT conditions give: `b_i > 0` implies `(T^T lambda)_i = 1`.

We can use the multiplicative weight algorithm:
- Initialize `lambda_k = 1/(2n-1)` for all `k`
- At each iteration:
  - `b_i = M / (T^T lambda)_i` (clipped to non-negative)
  - Update `lambda_k *= exp(eta * (T @ b)_k / M - 1)` (or similar)

This gives an approximate solution to the LP in O(n^2) per iteration (since T is Toeplitz-like, we can do the matrix-vector products in O(n log n) using FFT).

def lse_logsumexp_objective(a_flat, n, T_temp):
    a = a_flat.reshape(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)
    return obj


def lse_logsumexp_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()

def coordinate_descent_polish(a0, time_limit, seed=42):

        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:8]

        improved = False
        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-10:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 3)]

                for dst in candidates_sorted[:3]:
                    for frac in [0.01, 0.03, 0.08, 0.15]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt

                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)

                        if s_t < 1e-10:

                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 10 == 0:
                for _ in range(20):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:

                    best_a = avg_check.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        for pw in [3, 5, 7]:
            a = np.exp(-pw * (x - 0.5) ** 2)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1200, 1600, 2000, 2400]
    scan_results = {}

    for n_scan in scan_lengths:
        if time.monotonic() - t_global > time_budget * 0.15:
            break
        scan_time = min((time_budget * 0.15 - (time.monotonic() - t_global)) /
                        max(1, len(scan_lengths) - scan_lengths.index(n_scan)), 80.0)
        if scan_time < 10:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

    # Pick best from scan
    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP-Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.45
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_adaptive=True)
    do_try_update(a_out, f"lp_deep_n{n_base}")

    # Phase 2: LSE + scipy at base length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=30.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_scipy_n{n_best}")

    # Phase 3: Try longer length
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP n={n_long} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, use_adaptive=True)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # Phase 4: Subgradient Nesterov (fast, different trajectory)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SubGrad n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_nesterov(a_norm, t4, verbose=False)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=12345)
        do_try_update(a_out, "coord_polish")

    # Phase 6: LSE polish with lower temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t6, T_start=5.0, T_end=0.5, n_stages=4)
        do_try_update(a_out, "lse_polish")

    # Phase 7: Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.5
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=77777)
        do_try_update(a_out, "coord_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
