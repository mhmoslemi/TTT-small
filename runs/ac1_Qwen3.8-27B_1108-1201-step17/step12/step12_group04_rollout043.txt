def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

            if time.monotonic() - t0 > time_limit:

        # h[k_max] = sum_m a[m] * a[k_max - m]
        # Moving δ from i to j: change in h[k_max] ≈ -δ*a[k_max-i] + δ*a[k_max-j] (first order)
        # We want this to be negative: a[k_max-j] < a[k_max-i]

        je = min(n - 1, k_max)

def structured_init(n, seed):
    """Generate a structured initialization."""
    rng = np.random.default_rng(seed)
    choice = int(rng.integers(0, 6))
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = np.exp(-4 * (x - 0.3) ** 2) + np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.sqrt(np.maximum(a, 0))  # Square root to make it more "peaked"
    return normalize(a)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LogSumExp-smoothed L-BFGS optimization (primary)
2. Peak-directed coordinate descent for targeted peak reduction
3. Fine-grained polishing with partner-value analysis
4. Multi-length exploration with interpolation transfer
Key design: FFT-based O(n log n) convolution, adaptive temperature scheduling,
and careful time management to maximize quality within budget.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import math


def fast_conv_full(a):
    """Full self-convolution a*a using FFT, O(n log n). Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]


def compute_score(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    h = fast_conv_full(a)
    return 2.0 * n * h.max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_and_grad_a(a, T):
    """
    Compute LSE smoothed objective and gradient w.r.t. a (on the simplex).
    F_T(a) = (1/T)*log(sum_k exp(T*h_k)) where h = conv(a,a).
    Gradient: dF_T/da_i = 2 * sum_m w_{i+m} * a_m
    """
    n = len(a)
    h = fast_conv_full(a)
    Th = T * h
    Th_max = Th.max()
    exp_Th = np.exp(Th - Th_max)
    exp_sum = exp_Th.sum()
    obj = (Th_max + math.log(exp_sum)) / T

    w = exp_Th / exp_sum

    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size2 = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size2)
    A_rev = rfft(a_rev, fft_size2)
    conv_w_arev = irfft(W * A_rev, fft_size2)

    grad = 2.0 * conv_w_arev[n - 1:n - 1 + n]
    return obj, grad


def lse_optimize(a0, time_limit, T_start=500.0, T_end=5.0, n_stages=20):
    """
    Multi-stage LogSumExp optimization with L-BFGS.
    Uses softmax reparameterization: a = softmax(z).
    """
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-30)).astype(np.float64)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si in range(n_stages):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        stage_T = temps[si]

        # Later stages: more iterations for finer optimization
        if si < n_stages // 3:
            iters = 50
        elif si < 2 * n_stages // 3:
            iters = 100
        else:
            iters = 150

        def func_final(z_in):
            a_in = softmax_fn(z_in)
            val, _ = lse_obj_and_grad_a(a_in, stage_T)
            return val

        def jac_final(z_in):
            a_in = softmax_fn(z_in)
            _, grad_a = lse_obj_and_grad_a(a_in, stage_T)
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=func_final, x0=z, jac=jac_final, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        # Also check the raw a (not just softmax output)
        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si}/{n_stages} T={stage_T:.1f} "
                  f"({elapsed:.0f}s, rem={remaining:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_directed_descent(a0, time_limit, seed=42):
    """
    Peak-directed coordinate descent: identify top peaks and redistribute mass.
    Uses partner-value analysis to find optimal source-target pairs.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        h = fast_conv_full(a)
        n_peaks = min(8, 2 * n - 1)
        top_peaks = np.argsort(h)[::-1][:n_peaks]

        improved = False

        for pk in range(min(4, len(top_peaks))):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(top_peaks[pk])

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            top_contrib = np.argsort(contrib)[::-1][:4]

            for tc in top_contrib:
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-11:
                    continue

                # Find best target: minimize contribution to top peaks
                best_target = -1
                best_penalty = float('inf')

                # Pre-compute partner values for top peaks
                partner_vals = np.zeros(n)
                for k2 in top_peaks[:4]:
                    for pos in range(n):
                        partner = k2 - pos
                        if 0 <= partner < n:
                            partner_vals[pos] += a[partner]

                # Sample candidates
                n_samples = min(150, n)
                candidates = rng.integers(0, n, size=n_samples)

                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    penalty = partner_vals[dst] + 0.005 * a[dst]
                    if penalty < best_penalty:
                        best_penalty = penalty
                        best_target = dst

                if best_target < 0:
                    continue

                for frac in [0.001, 0.003, 0.008, 0.015, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_target] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial /= s
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

                if improved:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = float(rng.uniform(0.001, 0.01))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s
                sc = compute_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                    break

            if stall > 400:
                break

        if stall > 0 and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PDD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_polish(a0, time_limit, seed=777):
    """
    Very fine polishing: tiny moves targeting the single maximum peak
    using partner-value gradient analysis.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        h = fast_conv_full(a)
        k_max = int(np.argmax(h))

        # Partner value: for each position i, how much does a[k_max - i] contribute?
        partner = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        for i in range(js, je + 1):
            p = k_max - i
            if 0 <= p < n:
                partner[i] = a[p]

        # Source: high partner value (large contribution to peak)
        # Target: low partner value (small contribution)
        src_order = np.argsort(partner)[::-1]
        dst_order = np.argsort(partner)

        improved = False
        n_src = min(15, len(src_order))
        n_dst = min(15, len(dst_order))

        for si in range(n_src):
            if time.monotonic() - t0 > time_limit:
                break
            src = int(src_order[si])
            if a[src] < 1e-12:
                continue
            for di in range(n_dst):
                dst = int(dst_order[di])
                if dst == src:
                    continue
                for frac in [0.0003, 0.001, 0.003]:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial /= s
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-16:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 150:
                break

    return best_a, best_score


def lse_optimize_direct(a0, time_limit, T_start=200.0, T_end=2.0, n_stages=15):
    """
    LogSumExp optimization WITHOUT softmax reparameterization.
    Uses L-BFGS-B with manual projection onto the simplex.
    This avoids numerical issues with softmax for large n.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si in range(n_stages):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        stage_T = temps[si]

        if si < n_stages // 3:
            iters = 40
        elif si < 2 * n_stages // 3:
            iters = 80
        else:
            iters = 120

        # Work in log space: x = log(a), with constraint sum(exp(x)) = 1
        # Use x directly and project
        x = np.log(np.maximum(a, 1e-30))

        def func_x(x_in):
            a_in = np.exp(x_in - x_in.max())
            a_in = a_in / a_in.sum()
            val, _ = lse_obj_and_grad_a(a_in, stage_T)
            return val

        def jac_x(x_in):
            a_in = np.exp(x_in - x_in.max())
            a_in = a_in / a_in.sum()
            _, grad_a = lse_obj_and_grad_a(a_in, stage_T)
            # dx -> da: da_i = a_i * dx_i - a_i * sum_j(a_j * dx_j)
            # grad_x = a * (grad_a - dot(grad_a, a))
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=func_x, x0=x, jac=jac_x, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': iters * 300, 'disp': False}
            )
            x = res.x
            a = np.exp(x - x.max())
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE2 stage={si}/{n_stages} T={stage_T:.1f} "
                  f"({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = int(rng.integers(0, 8))
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(6 * np.pi * x) + 0.1 * np.cos(14 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 5:
        a = (1 - x) ** 1.5
    elif choice == 6:
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) ** 0.8
    else:
        a = np.ones(n)
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
