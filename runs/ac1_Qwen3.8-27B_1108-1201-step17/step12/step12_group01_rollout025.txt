- Move a small amount from `a[j*]` (or `a[k*-j*]`, whichever is larger) to a position `p` that minimizes its contribution to all top peaks

This is a form of "coordinate descent" where we select the coordinate to update based on the current bottleneck.

Let me now write the final code:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Nesterov zero-sum game (subgradient) for global optimization
2. Smoothed max gradient descent with Adam (top-k LSE with adaptive temperature)
3. Coordinate descent peak shaving (targeted mass transfer)
4. Multi-length exploration and final polish
Key improvement: proper simplex projection + adaptive peak targeting + 
more aggressive local search in the final phase.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def simplex_project(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_smooth_max_grad(a, T, topk):
    """
    Compute smooth max (top-k LSE) of autocorrelation and its gradient.
    Returns (value, gradient).
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    # Gradient: d/d a[j] of c[k] = 2 * a[k-j] (for valid k-j)
    # So grad_a[j] = 2 * sum_k w[k] * a[k-j]
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def exact_max_grad(a):
    """
    Compute exact max of autocorrelation and a subgradient.
    Returns (max_val, subgradient, argmax_positions).
    """
    n = len(a)
    c = fast_autocorr(a)
    max_val = c.max()
    # Find all positions within 1e-10 of max (for subgradient averaging)
    threshold = max_val - 1e-10
    peak_positions = np.where(c >= threshold)[0]
    
    grad_a = np.zeros(n)
    for K in peak_positions:
        weight = 1.0 / len(peak_positions)
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * weight * a[K - idx_range]
    
    return max_val, grad_a, peak_positions


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_nesterov(a0, time_limit, solver, max_iters=50000):
    """
    Nesterov-accelerated LP best-response (zero-sum game).
    Uses running averages and adaptive restart.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg = np.zeros(n)
    count = 0
    WINDOW = 50
    buf = [None] * WINDOW
    buf_idx = 0
    buf_sum = np.zeros(n)
    stall = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Running average
        if buf[buf_idx] is not None:
            buf_sum -= buf[buf_idx]
        buf[buf_idx] = a.copy()
        buf_sum += a
        buf_idx = (buf_idx + 1) % WINDOW
        count += 1
        avg = a  # Just track current for simplicity

        # Evaluate
        score_a = 2.0 * n * fast_autocorr(a).max()
        score_y = 2.0 * n * fast_autocorr(y).max()
        
        if count >= WINDOW and count % 20 == 0:
            avg_w = buf_sum / WINDOW
            score_avg = 2.0 * n * fast_autocorr(avg_w).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_w.copy()
                stall = 0
                continue

        current_best = min(score_a, score_y)
        if current_best < best_score:
            best_score = current_best
            best_a = a.copy() if score_a < score_y else y.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restart
        if stall > 500 and t > 100:
            y = best_a.copy()
            a = simplex_project(best_a + rng_local.normal(0, 0.01, n))
            stall = 0

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def gradient_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=30, topk=15):
    """
    Smoothed max gradient descent with Adam.
    Uses log-softmax parameterization and adaptive temperature annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Learning rate: start higher, decay across stages
        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.85 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(50, int(remaining * 5))  # aim for ~5 iters/sec initially

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            a = softmax_fn(z)
            obj, grad_a = compute_smooth_max_grad(a, stage_T, topk)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 16 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        # Reset z to best periodically
        if si % 10 == 9:
            z = np.log(np.maximum(best_a, 1e-30))

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Grad stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, rng, n_trials_per_iter=50):
    """
    Coordinate descent: try moving mass from one position to another,
    accepting only if it reduces the max autocorrelation.
    Uses smart source/destination selection based on peak contributions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 5:
        c = fast_autocorr(a)
        max_c = c.max()
        # Top peaks
        top_k = 5
        peak_positions = np.argsort(c)[::-1][:top_k]
        peak_threshold = c[peak_positions[-1]] * 1.0  # Use the lowest of top-k as threshold
        
        # Build "danger" map: for each position j, how much does it contribute to peaks?
        danger = np.zeros(n)
        for K in peak_positions:
            js = max(0, K - (n - 1))
            je = min(n - 1, K)
            for j in range(js, je + 1):
                contrib = a[j] * a[K - j]
                danger[j] += contrib
        
        improved = False
        for trial in range(n_trials_per_iter):
            # Choose source: position with high danger and high value
            if danger.max() < 1e-15:
                break
            # Weight by danger * value
            src_weights = danger * np.maximum(a, 0)
            total_w = src_weights.sum()
            if total_w < 1e-15:
                break
            src = rng.choice(n, p=src_weights / total_w)
            
            if a[src] < 1e-10:
                continue
            
            # Choose destination: low danger, not the source
            dst_weights = np.maximum(1e-6 - danger, 0) + 1e-10
            dst_weights[src] = 0
            total_dw = dst_weights.sum()
            if total_dw < 1e-15:
                continue
            dst = rng.choice(n, p=dst_weights / total_dw)
            
            # Try different step sizes
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                
                c_trial = fast_autocorr(a_trial)
                if c_trial.max() < best_score / (2.0 * n) - 1e-15:
                    a = a_trial
                    best_score = 2.0 * n * c_trial.max()
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                c_trial = fast_autocorr(a)
                if c_trial.max() * 2 * n < best_score:
                    best_score = 2.0 * n * c_trial.max()
                    best_a = a.copy()
                    stall = 0
            
            if stall > 200:
                # Bigger perturbation
                n_pert = rng.integers(3, 10)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i2 = rng.integers(n)
                    j2 = rng.integers(n)
                    if i2 != j2 and a_trial[i2] > 1e-10:
                        frac2 = rng.uniform(0.01, 0.05)
                        amt2 = a_trial[i2] * frac2
                        a_trial[i2] -= amt2
                        a_trial[j2] += amt2
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    c_trial = fast_autocorr(a_trial)
                    if c_trial.max() * 2 * n < best_score:
                        best_score = 2.0 * n * c_trial.max()
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        stall = 0
                
                if stall > 500:
                    break

    return best_a, best_score


def subgradient_polish(a0, time_limit, rng):
    """
    Subgradient method with exact max + adaptive step size.
    Works directly in a-space with simplex projection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    stall = 0
    
    step_size = 1e-4
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        max_val, grad_a, peaks = exact_max_grad(y)
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = y - alpha * step_size * grad_a
        y_new = simplex_project(y_new)
        
        if t == 1:
            a_new = y_new
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = simplex_project(a_new)
        
        y = y_new
        a = a_new
        
        # Check
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive step size
        if stall > 100:
            step_size *= 0.5
            stall = 0
        
        if stall > 500:
            # Perturb and restart
            a = simplex_project(best_a + rng.normal(0, 0.005, n))
            y = a.copy()
            step_size = 1e-4
            stall = 0
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad t={t} ({elapsed:.0f}s): best={best_score:.6f}, step={step_size:.2e}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.2) ** 2) + 0.5 * np.exp(-5 * (x - 0.8) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        # Asymmetric Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.4) ** 2)
    elif choice == 8:
        # Sine-weighted
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x ** 0.5) ** 2
    else:
        # Multi-bump asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.3) ** 2) + 0.3 * np.exp(-10 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set LP Nesterov (zero-sum game) for global optimization
2. Smoothed max gradient descent with Adam and adaptive temperature
3. Subgradient method with Nesterov momentum and simplex projection
4. Coordinate descent with danger-map-based source/destination selection
5. Multi-length exploration and multi-phase polish
Key improvements: active-set LP (faster), proper simplex projection,
danger-map coordinate descent, and more aggressive local search.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def simplex_project(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_smooth_max_grad(a, T, topk):
    """Compute top-k LSE of autocorrelation and gradient w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = int(top_idx[idx])
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    return lse_val, grad_a


def exact_max_subgrad(a):
    """Compute exact max autocorrelation value and averaged subgradient."""
    n = len(a)
    c = fast_autocorr(a)
    max_val = c.max()
    threshold = max_val - 1e-10 * max(1.0, max_val)
    peak_positions = np.where(c >= threshold)[0]
    grad_a = np.zeros(n)
    for K in peak_positions:
        weight = 1.0 / len(peak_positions)
        js = max(0, int(K) - (n - 1))
        je = min(n - 1, int(K))
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * weight * a[K - idx_range]
    return max_val, grad_a, peak_positions


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, n_active=None):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Uses active set if n_active is specified."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    if n_active is not None and n_active < 2 * n - 1:
        c = fast_autocorr(a_fixed)
        top_lags = np.argsort(c)[::-1][:n_active]
        active_T = T[top_lags]
        constraints = [active_T @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_nesterov(a0, time_limit, solver, max_iters=100000, n_active=80):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    t0 = time.monotonic()

    WINDOW = 50
    buf = [None] * WINDOW
    buf_idx = 0
    buf_sum = np.zeros(n)
    stall = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Update active set periodically
        na = n_active if (t % 100 == 0 or t <= 10) else None
        b_star, M_val = solve_lp_best_response(y, solver, n_active=na)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Running average
        if buf[buf_idx] is not None:
            buf_sum -= buf[buf_idx]
        buf[buf_idx] = a.copy()
        buf_sum += a
        buf_idx = (buf_idx + 1) % WINDOW

        # Evaluate every few iterations
        if t % 5 == 0:
            score_a = 2.0 * n * fast_autocorr(a).max()
            score_y = 2.0 * n * fast_autocorr(y).max()
            if t >= WINDOW and t % 20 == 0:
                avg_w = buf_sum / WINDOW
                score_avg = 2.0 * n * fast_autocorr(avg_w).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_w.copy()
                    stall = 0
                    continue

            current_best = min(score_a, score_y)
            if current_best < best_score - 1e-12:
                best_score = current_best
                best_a = a.copy() if score_a < score_y else y.copy()
                stall = 0
            else:
                stall += 1

        # Adaptive restart with perturbation

            pert = rng_local.normal(0, 0.02, n)
            a_restart = simplex_project(best_a + pert)
            y = a_restart.copy()
            a = a_restart.copy()
            buf_sum = np.zeros(n)
            buf = [None] * WINDOW
            buf_idx = 0
            stall = 0

        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.6f}")

    return best_a, best_score


def gradient_optimize(a0, time_limit, T_start=50.0, T_end=0.03, n_stages=35, topk=15):
    """Smoothed max gradient descent with Adam and log-softmax parameterization."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 4e-3 * min(1.0, 120.0 / n)
        lr = base_lr * (0.88 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(50, int(remaining * 8))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            a = softmax_fn(z)
            obj, grad_a = compute_smooth_max_grad(a, stage_T, topk)
            Sc = np.dot(grad_a, a)
            grad_z = a * (grad_a - Sc)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 20 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 10 == 9:
            z = np.log(np.maximum(best_a, 1e-30))

        if si % 7 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Grad stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=777, n_trials=80):
    """Coordinate descent with danger-map based source/destination selection."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 5:
        c = fast_autocorr(a)
        max_c = c.max()
        top_k = min(8, len(c))
        peak_positions = np.argsort(c)[::-1][:top_k]

        # Build danger map
        danger = np.zeros(n)
        for K in peak_positions:
            K = int(K)
            js = max(0, K - (n - 1))
            je = min(n - 1, K)
            for j in range(js, je + 1):
                danger[j] += a[j] * a[K - j]

        improved = False
        for trial in range(n_trials):
            if time.monotonic() - t0 > time_limit - 5:
                break
            if danger.max() < 1e-15:
                break

            src_weights = danger * np.maximum(a, 0)
            total_w = src_weights.sum()
            if total_w < 1e-15:
                break
            src = rng.choice(n, p=src_weights / total_w)

            if a[src] < 5e-8:
