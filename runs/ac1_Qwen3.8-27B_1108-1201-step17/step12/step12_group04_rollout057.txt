"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diversified structured initialization

import numpy as np
import time
import math
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the score 2n*max(autocorr)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a / len(a)
    return a / s

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*b_k)) w.r.t. a.
    Returns (lse_value, gradient).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = (max_Tb + np.log(exp_Tb.sum() / len(exp_Tb))) / T  # Hmm, this isn't quite right
    
    # Actually: (1/T)*log(sum exp(T*b_k)) = b_max + (1/T)*log(sum exp(T*(b_k - b_max)))
    lse_val = b.max() + np.log(exp_Tb.sum()) / T
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i in [0,n-1])
    # This is 2 * (w * a_rev)[n-1 : n-1+n] where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_len = n + (2*n-1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n-1 : n-1+n]
    
    return lse_val, grad, w

def adam_step(grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. Returns updated (m, v, step)."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v, step

    z = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)

        if time.monotonic() - t0 > time_limit:

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Learning rate: start moderate, decay within stage

        max_iters = min(2000, max(50, int(remaining * 20)))
        
        m = np.zeros(n)

        t_adam = 0

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    Instead of full LP, use a probabilistic best response (sample from soft-max of constraints).
    """

    a = normalize(a0.copy())
    y = a.copy()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute cross-correlation of y with all unit vectors (i.e., y itself reversed)
        # The "best response" in the game is to put mass where y has low autocorrelation
        # Approximate: b_star_i ∝ 1 / (sum_k |y[k-i]|^2 + eps)  -- not quite right

        # Approximate with: b_star proportional to the "gap" at each position

        b_star = 1.0 / np.sqrt(y + eps)
        b_star = b_star / b_star.sum()
        
        alpha = 2.0 / (t + 2.0)
        y = (1 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Nesterov extrapolation
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            a = y + momentum * (y - a)
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        sc_y = score_of(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NL t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

If the peak is at lag `k*`, we want to reduce `b[k*] = sum_i a[i]*a[i+k*]`. The most effective move is to reduce the largest term in this sum. If `a[i]*a[i+k*]` is the largest, we should move mass out of `i` (or `i+k*`).

A smarter approach: for each candidate source position `i`, compute the "cost" of moving mass from `i`:
- It decreases `b[k]` by `2*delta*a[k-i]` for all lags `k`
- It increases `b[k]` by `2*delta*a[k-j]` (for the target `j`)
- Net change: `-2*delta*(a[k-i] - a[k-j])`

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LogSumExp-smoothed optimization with Adam optimizer and deep annealing
2. Intensive peak-aware local search with simulated annealing escape
3. Nesterov-style fixed point iteration (no LP, faster)

5. Final polishing with very fine local search
Key innovations: peak-aware move scoring, adaptive fraction search,

import numpy as np
import time
import math
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2n*max(autocorr)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a / len(a)
    return a / s


def softmax_stable(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_and_grad(a, T):
    """
    Compute LogSumExp value and its gradient w.r.t. a.
    Returns (lse_value, gradient, weights).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = b.max() + np.log(exp_Tb.sum()) / T

    # grad[i] = 2 * sum_k w[k] * a[k-i] = 2*(w * a_rev)[n-1 : 2n-2]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1: n - 1 + n]

    return lse_val, grad, w


def adam_update(grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer. Returns (m_new, v_new, step)."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v, step


def lse_optimize(a0, time_limit, T_start=200.0, T_end=0.01, n_stages=40):
    """
    LogSumExp-smoothed optimization with Adam and multi-stage geometric annealing.
    Works in log-space (z = log(a)) with softmax projection.
    """
    n = len(a0)
    a = normalize(a0.copy())
    z = np.log(np.maximum(a, 1e-30))

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        stage_T = temps[si]
        # Learning rate: adaptive based on stage
        progress = si / max(1, n_stages - 1)
        lr = 0.08 * (1.0 - 0.5 * progress)
        max_iters = max(30, min(3000, int(remaining * 15 * (1.0 + progress))))

        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break

            lse_val, grad_a, w = compute_lse_and_grad(a, stage_T)

            # Project gradient to z-space (simplex constraint)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m, v, step = adam_update(grad_z, m, v, t_adam, lr)

            # Gradient step in z-space
            z = z - step
            a = softmax_stable(z)

            # Periodic score check
            if it % 50 == 0:
                sc = score_of(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        # End-of-stage check
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE stage {si}/{n_stages} T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    n = len(a0)

    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    it = 0
    stall = 0
    temperature = 0.01  # SA temperature for escape

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_autocorr(a)
        k_max = int(np.argmax(b))
        b_max = b[k_max]

        # Get top 5 peaks for scoring
        top_peaks = np.argsort(b)[::-1][:5]

        # Identify the top contributor to the main peak
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]

        # Try several source positions (top contributors + random)
        n_sources = min(len(idx), 5)
        top_contrib_order = np.argsort(contrib)[::-1][:n_sources]
        sources = [idx[tc] for tc in top_contrib_order]

        # Add some random sources
        for _ in range(3):
            sources.append(int(rng.integers(n)))

        improved = False

        for src in sources:
            if a[src] < 1e-12:
                continue

            # Score candidate targets by their total contribution to top peaks
            n_cands = min(n, 300)
            if n > n_cands:
                cands = rng.choice(n, size=n_cands, replace=False)
            else:
                cands = np.arange(n)

            best_dst = -1
            best_dst_score = float('inf')

            for dst in cands:
                if dst == src:
                    continue
                ts = 0.0
                for kp in top_peaks:
                    js2 = max(0, kp - n + 1)
                    je2 = min(n - 1, kp)
                    if js2 <= dst <= je2:
                        partner = kp - dst
                        if 0 <= partner < n:
                            ts += a[dst] * a[partner]
                ts += 0.0005 * a[dst]
                if ts < best_dst_score:
                    best_dst_score = ts
                    best_dst = dst

            if best_dst < 0:
                continue

            # Try multiple fractions
            fracs = [0.002, 0.005, 0.01, 0.02, 0.04, 0.06, 0.08, 0.1, 0.15]
            # Shuffle for diversity
            rng.shuffle(fracs)

            for frac in fracs[:6]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s

                sc = score_of(a_trial)
                if sc < best_score - 1e-16:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            stall += 1
            temperature = max(0.001, 0.02 * math.exp(-stall / 200.0))

            # Random exploration with SA acceptance
            n_random = min(20, 5 + stall // 50)
            for _ in range(n_random):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                frac = float(rng.uniform(0.002, 0.15))
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = score_of(a_trial)

                delta = sc - best_score
                if delta < 0 or (rng.random() < math.exp(-delta / temperature) and stall > 100):
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    stall = 0
                    break

            if stall > 300:
                # Large perturbation to escape
                n_perturb = 3
                for _ in range(n_perturb):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j:
                        frac = float(rng.uniform(0.1, 0.3))
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_of(a_trial)
                        if sc < best_score * 1.05:  # Accept if not too bad
                            a = a_trial
                            if sc < best_score:
                                best_score = sc
                                best_a = a.copy()
                            stall = 0
                            break

            if stall % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"  LS stall={stall} ({elapsed:.0f}s): best={best_score:.8f}, T={temperature:.5f}")

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LS it={it} ({elapsed:.0f}s): cur={score_of(a):.8f}, best={best_score:.8f}")

    return best_a, best_score


def nesterov_fixed_point(a0, time_limit, seed=7):
    """
    Nesterov-accelerated fixed point iteration.
    Uses soft best-response instead of LP (much faster).
    """
    n = len(a0)
    a = normalize(a0.copy())
    y = a.copy()
    prev_y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Soft best response: put more mass where y has low autocorrelation contribution
        # b_cross[i] = sum_k y[k] * y[k-i] approximated by y[i] * sum_y
        # Better: use the actual cross-correlation with a reference
        # Simple heuristic: b_star ∝ 1/sqrt(y + eps)
        eps = 1e-8
        b_star = 1.0 / np.sqrt(y + eps)

        # Add noise for exploration
        noise = rng.standard_normal(n) * 0.01
        b_star = b_star + y * noise
        b_star = np.maximum(b_star, 0)
        s = b_star.sum()
        if s < 1e-15:
            b_star = 1.0 / np.sqrt(y + eps)
            s = b_star.sum()
        b_star = b_star / s

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / s

        # Nesterov extrapolation
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            a = y_new + momentum * (y_new - y)
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = y_new.copy()
            else:
                a = a / s

        y = y_new

        if t % 50 == 0:
            sc_a = score_of(a)
            sc_y = score_of(y)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NF t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 12))
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.exp(-20.0 * (x - 0.5) ** 2)
    elif choice == 2:
        a = np.exp(-5.0 * (x - 0.5) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 3:
        xf = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * xf) + 0.15 * np.cos(4 * xf) + 0.05 * np.cos(6 * xf)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1.0 - x) ** 3 + 0.1
    elif choice == 5:
        a = 1.0 / (1.0 + 100.0 * (x - 0.5) ** 4)
    elif choice == 6:
        a = np.exp(-3.0 * (x - 0.4) ** 2) + 0.8 * np.exp(-3.0 * (x - 0.6) ** 2)
    elif choice == 7:
        a = np.cos(np.pi * x) ** 4
    elif choice == 8:
        a = 0.3 + 0.4 * np.exp(-10.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-10.0 * (x - 0.2) ** 2) + 0.3 * np.exp(-10.0 * (x - 0.8) ** 2)
    elif choice == 9:
        a = np.ones(n)
        q = n // 4
        a[:q] *= 0.2
        a[-q:] *= 0.2
    elif choice == 10:
        xf = np.linspace(0, 2 * np.pi, n)
        a = 0.4 + 0.25 * np.cos(3 * xf) + 0.15 * np.cos(5 * xf) + 0.08 * np.cos(7 * xf) + 0.04 * np.cos(11 * xf)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-8.0 * (x - 0.5) ** 2) * (1.0 + 0.3 * np.sin(20.0 * np.pi * x))
        a = np.maximum(a, 0)

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fine_coarse_search(a0, time_limit, seed=77):
    """
    Two-scale local search: first make larger moves (coarse), then smaller (fine).
    This helps escape local minima while still converging.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    # Phase A: Coarse search (larger moves)
    t_coarse = time_limit * 0.4
    t_fine = time_limit * 0.6

    coarse_end = t0 + t_coarse
    fine_end = t0 + time_limit

    for phase, phase_end, frac_range in [("coarse", coarse_end, (0.02, 0.2)), ("fine", fine_end, (0.001, 0.05))]:
        while time.monotonic() < phase_end:
            b = fast_autocorr(a)
            k_max = int(np.argmax(b))

            # Top contributor to peak
            js = max(0, k_max - n + 1)
            je = min(n - 1, k_max)
            if je >= js:
                idx = np.arange(js, je + 1)
                contrib = a[idx] * a[k_max - idx]
                src = int(idx[np.argmax(contrib)])
            else:
                src = int(rng.integers(n))

            if a[src] < 1e-12:
                src = int(np.argmax(a))

            # Target: minimize contribution to all peaks
            top_peaks = np.argsort(b)[::-1][:3]
            best_dst = -1
            best_cost = float('inf')
            n_cands = min(n, 150)
            cands = rng.choice(n, size=n_cands, replace=False) if n > n_cands else np.arange(n)

            for dst in cands:
                if dst == src:
                    continue
                cost = 0.0
                amp = float(rng.uniform(frac_range[0], frac_range[1]))
                amt = a[src] * amp
                if amt < 1e-14:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                sc = score_of(a_t)
                if sc < best_cost:
                    best_cost = sc
                    best_dst = dst
                    best_a_t = a_t

            if best_dst >= 0 and best_cost < best_score - 1e-16:
                a = best_a_t
                best_score = best_cost
                best_a = a.copy()
            else:
                # Small random step
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    amp = float(rng.uniform(0.001, 0.03))
                    amt = a[i] * amp
                    a_t = a.copy()
                    a_t[i] -= amt
                    a_t[j] += amt
                    a_t = np.maximum(a_t, 0)
                    a_t = a_t / a_t.sum()
                    if score_of(a_t) < best_score:
                        a = a_t
                        best_score = score_of(a)
                        best_a = a.copy()

        if phase == "coarse":
            elapsed = time.monotonic() - t0
            print(f"  FineCoarse coarse done ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
