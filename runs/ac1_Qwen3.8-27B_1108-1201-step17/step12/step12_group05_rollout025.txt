import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_b/S^2 without calling evaluate_sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * np.max(b) / (S * S)


def evaluate_safe(arr):

    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_shaping(a, n_iter=1000, alpha=0.2, time_limit=None, t_start=None):
    """
    Spectral shaping: iteratively flatten power spectrum.
    Adaptive: reduces step size as improvement slows.
    """
    n = len(a)
    a = normalize_nonneg(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    if t_start is None:
        t_start = time.monotonic()
    
    for i in range(n_iter):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        
        # Adaptive alpha: reduce as we converge
        eff_alpha = alpha * max(0.1, 1.0 - i / max(1, n_iter))
        
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, eff_alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize_nonneg(a_new)
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

def lse_grad(a, T):
    """
    Compute gradient of LSE_T = (1/T) * log(sum exp(T * conv(a,a))) w.r.t. a.
    Returns (lse_value, gradient).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = (max_Tb + np.log(exp_Tb.sum() if exp_Tb.sum() > 0 else 1)) / T
    
    # Gradient: 2 * sum_k w[k] * a_reversed_at_k
    # (grad conv(a,a)/da)[k] = 2 * conv(a, a_reversed) projected
    # Using FFT: grad = 2 * IFFT(FFT(a_reversed) * FFT(w)) [n-1 : n-1+n]
    fft_size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv_result[n-1:n-1+n]
    
    return lse_val, grad


def lse_grad_z(z, T):
    """LSE objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = lse_grad(a, T)
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def analytical_peak_shaving(a0, time_limit, top_k=5, max_stall=200):
    """
    Peak shaving with analytical move computation.
    For each peak, finds the best (src, dst) pair that reduces it most.
    Tries to reduce multiple peaks simultaneously.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_indices = np.argsort(b)[::-1][:top_k]
        max_b = b[top_indices[0]]
        
        # Find best move: for the worst peak, find the pair (src, dst)
        # that most reduces it
        best_improvement = 0
        best_move = None
        
        k_worst = top_indices[0]
        
        # For peak at position k_worst, the contribution from position j is a[j] * a[k_worst - j]
        # Moving delta from src to dst changes b[k_worst] by:
        # delta * (a[k_worst - dst] - a[k_worst - src]) [approximately]
        # We want this to be negative: a[k_worst - dst] < a[k_worst - src]
        
        # Compute sensitivity for the worst peak
        js = max(0, k_worst - n + 1)
        je = min(n - 1, k_worst)
        valid_idx = np.arange(js, je + 1)
        sensitivity = np.zeros(n)
        sensitivity[valid_idx] = a[k_worst - valid_idx]  # = a[valid_idx] for symmetric
        
        # Source: position with highest a[j] * sensitivity[j] (contributes most to peak)
        contrib = a * sensitivity
        src_candidates = np.argsort(contrib)[::-1][:max(3, n // 100)]
        
        # For each source, find best destination
        for src in src_candidates:
            if a[src] < 1e-12:
                continue
            # Destination should have low sensitivity to all top peaks
            dst_sens = np.zeros(n)
            for k2 in top_indices:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                valid_idx2 = np.arange(js2, je2 + 1)
                dst_sens[valid_idx2] += a[k2 - valid_idx2]
            
            # Also avoid positions that are already high (to not create new peaks)
            dst_score = dst_sens + 0.1 * a
            dst_candidates = np.argsort(dst_score)[:max(3, n // 100)]
            
            for dst in dst_candidates:
                if dst == src:
                    continue
                # Estimate improvement
                # Move delta from src to dst
                # Change in b[k_worst] ≈ -delta * sensitivity[src] + delta * a[k_worst - dst]
                # (the negative term removes contribution from src, positive adds from dst)
                est_change = -a[src] * sensitivity[src] / max(a[src], 1e-15) * a[src]
                # Simplified: improvement proportional to sensitivity[src] - a[k_worst-dst]
                # (times the amount moved)
                improvement = sensitivity[src] - (a[k_worst - dst] if 0 <= k_worst - dst < n else 0)
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_move = (src, dst)
        
        if best_move is None:
            # Fall back to random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                stall += 1
                if stall > max_stall:
                    break
                continue
        
        src, dst = best_move
        
        # Try various step sizes (from small to larger)
        improved = False
        for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            if a_trial[src] < 0:
                continue
            a_trial[dst] += amt
            a_trial = normalize_nonneg(a_trial)
            score = compute_score(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            # Try random perturbation (annealing)
            stall += 1
            if stall > max_stall:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize_nonneg(a_trial)
                score = compute_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize_nonneg(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize_nonneg(f_interp)

Wait, I have a problem. The `analytical_peak_shaving` function might be too slow because for each iteration, it computes the full sensitivity vector and sorts it. For n=1600, this is O(n log n) per iteration. If I do 1000 iterations, that's 1000 * 1600 * log(1600) ≈ 16 million operations, which should be fine.

But the inner loop over `src_candidates` (which is `n // 100 = 16` candidates) and `dst_candidates` (also 16) gives 256 evaluations per iteration, each requiring a convolution (O(n log n)). So total per "big step" is 256 * 1600 * 11 ≈ 4.5 million operations. Over 100 steps (limited by time), that's 450 million operations. This might be too slow.

Let me optimize: instead of trying all (src, dst) pairs, I should just try the best src with the best dst. That reduces it to O(n) per iteration.

Actually, let me reconsider the peak shaving approach. The issue is that each "try" requires a full convolution to compute the score. For n=1600, a single FFT-based convolution takes about n*log(n) ≈ 18000 operations. So if I can do 1000 convolutions per second (rough estimate), I can try 1000 moves per second. Over 300 seconds, that's 300,000 moves.

The key question is: can I find a move that improves the score in O(n) or O(n log n) time (without needing to evaluate the full convolution)?

**Analytical move evaluation:**

If I move `delta` from position `src` to position `dst`, the change in `conv(a,a)[k]` is:
`delta * (a[k-dst] - a[k-src])` (for the linear approximation, valid for small delta)

More precisely:
`conv(a', a')[k] - conv(a, a')[k] = delta * (a[k-dst] - a[k-src])` where `a' = a - delta*e_src + delta*e_dst`

And `conv(a', a')[k] = conv(a,a)[k] + delta*(a[k-dst] - a[k-src]) + O(delta^2)`

So to check if a move reduces the maximum, I need:
`conv(a,a)[k_max] + delta * (a[k_max - dst] - a[k_max - src]) < current_max`

This can be checked in O(1) for a specific (src, dst, delta, k_max) tuple!

But I also need to verify that no OTHER peaks are increased. For that, I need:
`conv(a,a)[k] + delta * (a[k-dst] - a[k-src]) <= current_max + epsilon` for all k.

This is O(n) to check.

So the algorithm is:
1. Find the current peak position k_max
2. For a candidate (src, dst):
   a. Check: `delta * (a[k_max - dst] - a[k_max - src]) < 0` (reduces the peak)
   b. Check: for all k, `conv(a,a)[k] + delta * (a[k-dst] - a[k-src]) <= current_max`
3. If both conditions hold, apply the move

The O(n) check in step 2b can be vectorized:
`max_k (conv(a,a)[k] + delta * (a_shifted_dst[k] - a_shifted_src[k])) <= current_max`

Where `a_shifted_dst[k] = a[k-dst]` (with appropriate wrapping/bounding).

This is much faster than computing a full new convolution!

Let me implement this optimized version:

```python
def fast_peak_shaving(a0, time_limit, top_k=5, max_stall=500):
    """
    Fast peak shaving using analytical move evaluation (O(n) per move check).
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    b = fast_autocorr(a)  # current convolution
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(999)
    
    # Precompute shifted versions for quick access
    # a_shifted[s][k] = a[k-s] (with zeros for out-of-bounds)
    
    while time.monotonic() - t0 < time_limit:
        # Find top peaks
        top_indices = np.argsort(b)[::-1][:top_k]
        current_max = b[top_indices[0]]
        
        # For the worst peak at k_worst, find best (src, dst)
        k_worst = top_indices[0]
        
        # Sensitivity at k_worst: sens[j] = a[k_worst - j] (if valid)
        sens = np.zeros(n)
        js = max(0, k_worst - n + 1)
        je = min(n - 1, k_worst)
        valid_idx = np.arange(js, je + 1)
        sens[valid_idx] = a[k_worst - valid_idx]
        
        # Source: high a[j] * sens[j] (contributes most to peak)
        contrib = a * sens
        top_src_idx = np.argsort(contrib)[::-1][:min(20, n // 20)]
        
        found_move = False
        for src in top_src_idx:
            if a[src] < 1e-12:
                continue
            
            # For each src, the "sensitivity vector" for checking all peaks is:
            # shift_src[k] = a[k - src] (for valid k - src)
            shift_src = np.zeros(2 * n - 1)
            for k in range(n):
                ks = k + src  # position in conv where a[src] contributes
                if ks < 2 * n - 1:
                    shift_src[ks] = a[k]  # Wait, this isn't right
                    # Actually conv(a,a)[k] includes the term a[src]*a[k-src]
                    # So the sensitivity of conv[k] to a[src] is a[k-src]
            
            # Let me think differently.
            # If I decrease a[src] by delta, the change in conv[k] is:
            # -delta * a[k-src] (for the term a[src]*a[k-src])
            #   -delta * a[src] (for the term a[k-src]*a[src], if k-src is the index)
            # Wait, conv(a,a)[k] = sum_j a[j]*a[k-j]
            # d/d(a[src]) conv(a,a)[k] = a[k-src] + a[k-src] = 2*a[k-src] (if 0<=k-src<n)
            # So decreasing a[src] by delta changes conv[k] by -2*delta*a[k-src]
            
            # If I increase a[dst] by delta, the change in conv[k] is:
            # +2*delta*a[k-dst] (if 0<=k-dst<n)
            
            # Net change: 2*delta*(a[k-dst] - a[k-src])
            
            # For the worst peak: 2*delta*(a[k_worst-dst] - a[k_worst-src]) < 0
            # i.e., a[k_worst-dst] < a[k_worst-src] = sens[src]
            
            # For all other peaks k_i: 2*delta*(a[k_i-dst] - a[k_i-src]) <= 0
            # i.e., a[k_i-dst] <= a[k_i-src] for all i
            
            # So dst should have low values at positions k_i - dst for all top peaks k_i.
            
            # Build "forbidden sensitivity" vector for dst:
            forbidden = np.zeros(n)
            for k2 in top_indices:
                k2_dst_offset = k2  # position in a that matters for peak k2
                # a[k2 - dst] should be small
                # For a given dst, this is a[k2 - dst]
                # We want: max over k2 of a[k2 - dst] to be small
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                # a[k2 - j] for j in [js2, je2]
                # This means we want a[0 : k2+1] to be small at position (k2-dst)
            
            # Simplified: for dst, compute max sensitivity = max over top peaks k2 of a[k2-dst] (if valid)
            # We want this to be < sens[src] (to ensure the move helps)
            
            # Vectorized: for each dst, max_sens[dst] = max over k2 of a[k2-dst]
            # This is like a "MaxPool" over the sensitivity patterns.
            
            # Let me just compute it directly for the top dst candidates:
            # dst should have small a[k_worst - dst] and small a[k_i - dst] for all i
            # Equivalently: a[k_worst - dst] + a[k_1 - dst] + ... + a[k_m - dst] should be small
            
            dst_constraint = np.zeros(n)
            for k2 in top_indices:
                # a[k2 - dst] is non-zero when 0 <= k2-dst < n, i.e., k2-n+1 <= dst <= k2
                lo = max(0, k2 - n + 1)
                hi = min(n - 1, k2)
                if lo <= hi:
                    # a[k2 - dst] for dst in [lo, hi] = a[k2-hi : k2-lo+1] = a[k2-k2 : k2-(k2-n+1)+1] = a[0 : n-lo]
                    # More precisely: for dst=lo, k2-dst = k2-lo; for dst=hi, k2-dst = k2-hi
                    dst_constraint[lo:hi+1] += a[k2-hi : k2-lo+1][::-1]
            
            # Best dst: minimize dst_constraint[dst] + 0.1*a[dst] (avoid high values)
            dst_score = dst_constraint + 0.1 * a
            dst_score[src] = np.inf  # can't move to self
            top_dst_idx = np.argsort(dst_score)[:min(10, n // 50)]
            
            for dst in top_dst_idx:
                # Check if this move actually helps: a[k_worst-dst] < a[k_worst-src]
                if k_worst - dst < 0 or k_worst - dst >= n:
                    a_kd = 0
                else:
                    a_kd = a[k_worst - dst]
                if a_kd >= sens[src]:
                    continue  # doesn't help the worst peak
                
                # Try different step sizes
                for frac in [0.002, 0.005, 0.01, 0.02]:
