            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver=None, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def multiplicative_weights(a0, time_limit, eta0=0.1, verbose=True):
    """Mirror descent / multiplicative weights on the simplex."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = eta0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        
        # Subgradient of max_k conv(a,a)_k = conv(a,a)_{k_star}
        # d/d a_i [sum_j a_j * a_{k-i}] = a_{k-i} + a_{i-k} (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                val += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                val += a[j2]
            grad[i] = 2.0 * val
        
        # Multiplicative update: a_i *= exp(-eta * grad_i)
        # Then renormalize
        log_shift = eta * grad
        # Clip to avoid overflow
        log_shift = np.clip(log_shift, -50, 50)
        a_new = a * np.exp(-log_shift)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * norm / s
        else:
            a_new = a.copy()
        
        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Decay eta
        eta = eta0 / (1.0 + 0.01 * t)
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW t={t} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score

def smoothed_gradient_descent(a0, time_limit, T_start=10.0, T_end=0.1, verbose=True):
    """Gradient descent on the LogSumExp smoothed max."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    total_templates = 20
    templates = np.geomspace(T_start, T_end, total_templates)
    
    for si, T in enumerate(templates):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        n_steps = int(remaining * 3)
        n_steps = max(50, min(5000, n_steps))
        
        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_autocorr(a)
            b_shifted = b / T
            b_max = b_shifted.max()
            exp_b = np.exp(b_shifted - b_max)
            w = exp_b / exp_b.sum()
            
            # Gradient: grad_i = 2 * sum_k w_k * (a[k-i] + a[i-k])
            # Compute via convolution: grad = 2 * (w * a + w_rev * a)
            # Actually: sum_k w_k * a[k-i] is the cross-correlation of w and a
            fft_size = 1 << (3 * n - 2).bit_length()
            A_f = rfft(a, fft_size)
            W_f = rfft(w, fft_size)
            # cross-correlation: sum_k w_k * a[k-i] = ifft(W * conj(A))[i] ... 
            # Actually let me think...
            # (w * a)[i] = sum_k w_k * a[i-k] (convolution)
            # We want sum_k w_k * a[k-i] which is the cross-correlation
            # cross_corr[i] = sum_k w_k * a[k-i] = (w * a_rev)[i] where a_rev[j] = a[n-1-j]
            # Hmm, let me just do it directly for correctness
            
            # grad_i = 2 * sum_k w_k * (a_{k-i} + a_{i-k})
            #         = 2 * (correlation of w with a at lag i + correlation of a with w at lag i)
            #         = 2 * correlation(w, a)[i]  (since correlation is symmetric for real signals)
            
            # correlation(w, a)[i] = sum_k w_k * a_{k-i}
            # This is the cross-correlation, which can be computed as:
            # ifft(FFT(w) * conj(FFT(a)))[i] ... no
            # Actually: cross_corr[i] = sum_k w[k] * a[k-i]
            # Let's use: cross_corr = ifft(FFT(w) .* conj(FFT(a)))
            # Wait, that gives sum_k w[k] * a[k+n-i] which is circular correlation
            
            # For linear cross-correlation:
            # cross_corr[i] = sum_k w[k] * a[k-i] for k-i in [0, n-1]
            # = sum_j a[j] * w[j+i] for j+i in [0, n-1]
            # This is the correlation of a and w at lag i
            
            # Using FFT: zero-pad both to size >= 2n-1
            # corr = ifft(FFT(w_padded) .* conj(FFT(a_padded)))[n-1 : 2n-2]
            # Hmm, this is getting complicated. Let me use a simpler approach.
            
            # For each i: grad_i = 2 * sum_k w_k * (a[k-i] if 0<=k-i<n else 0 + a[i-k] if 0<=i-k<n else 0)
            # = 2 * (sum_{k=i}^{i+n-1} w_k * a[k-i] + sum_{k=max(0,i-n+1)}^{i} w_k * a[i-k])
            # First term: cross-corr of w and a at lag i
            # Second term: cross-corr of w and a at lag i (same thing by symmetry of the sum)
            # Actually no: first term sums w_k * a_{k-i} for k in [i, min(i+n-1, 2n-2)]
            # Second term sums w_k * a_{i-k} for k in [max(0, i-n+1), i]
            # Let j = k-i in first: sum_j w_{i+j} * a_j for j in [0, min(n-1, 2n-2-i)]
            # Let j = i-k in second: sum_j w_{i-j} * a_j for j in [0, min(n-1, i)]
            # So grad_i = 2 * (sum_j w_{i+j} * a_j [j: 0 to min(n-1,2n-2-i)] + sum_j w_{i-j} * a_j [j: 0 to min(n-1,i)])
            
            # This is 2 * (cross_corr(w,a)[i] + cross_corr(w,a)[-i])... 
            # Actually it's simpler to just compute it via convolution.
            # grad = 2 * (w * a + w[::-1] * a)[:n] ... no, this isn't right either.
            
            # Let me just do it with a loop for now (n is at most ~3000, so it's fine)
            # Actually for n=1600, a loop of 1600 iterations each doing O(n) work is O(n^2) = 2.5M, which is fine.
            
            # Better: use the FFT approach
            # grad_i = 2 * sum_k w_k * (a[k-i] + a[i-k]) with bounds
            # = 2 * (sum_{k=i}^{i+n-1} w[k]*a[k-i] + sum_{k=0}^{i} w[k]*a[i-k]) [with bounds on w]
            # First sum: let j = k-i, range j=0 to min(n-1, 2n-1-i)
            #   = sum_j w[i+j] * a[j] (cross-correlation of w and a at lag i)
            # Second sum: let j = i-k, range j=0 to min(i, n-1, 2n-2)
            #   = sum_j w[i-j] * a[j] (cross-correlation of w and a at lag -i)
            
            # So grad = 2 * (xcorr(w,a) at positive lags + xcorr(w,a) at negative lags)
            # where xcorr(w,a)[i] = sum_j w[i+j] * a[j]
            
            # Using FFT: xcorr(w,a) = ifft(conj(FFT(w))) .* FFT(a)) ... 
            # Standard: xcorr(x, y)[l] = sum_n x[n+l] * y[n] = ifft(FFT(x) .* conj(FFT(y)))[l + N - len(y)] (circular)
            # For linear: zero-pad to size >= len(x) + len(y) - 1
            
            # Let me just use scipy.signal.correlate or do it manually
            # For efficiency with large n, let's use FFT
            
            fft_sz = 1 << (2 * n).bit_length()
            W = rfft(w, fft_sz)
            A = rfft(a, fft_sz)
            # xcorr(w, a)[l] = sum_j w[j+l] * a[j] = ifft(conj(W) * A)[l] (for circular)
            # For linear correlation:
            # xcorr = irfft(conj(W) * A, fft_sz)
            # The linear correlation is xcorr[n-1 : 2n-2] (lags from -(n-1) to n-1)
            # Wait, I need to be more careful.
            
            # Let's define: corr[l] = sum_j w[j+l] * a[j] for valid j (both 0<=j<n and 0<=j+l<2n-1)
            # Using circular correlation with zero-padding:
            # xcorr_circ[l] = sum_j w[j+l mod N] * a[j] where N = fft_sz
            # Since w has length 2n-1 and a has length n, and N >= 2n, we get the linear correlation
            # xcorr_circ[l] for l = 0, ..., N-1
            # The linear correlation we want is xcorr_circ[l] for l = -(n-1) to n-1
            # i.e., xcorr_circ[N-(n-1) : N] concatenated with xcorr_circ[0 : n]
            
            xcorr = irfft(np.conj(W) * A, fft_sz)
            # xcorr[l] = sum_j w[(j+l) % N] * a[j]
            # For l = 0: sum_j w[j] * a[j] = dot(w[:n], a)
            # For l = 1: sum_j w[j+1] * a[j]
            # ...
            # For l = N-1 = -1 mod N: sum_j w[j-1] * a[j]
            
            # grad_i = 2 * (xcorr[i] + xcorr[(N-i) % N])
            # where xcorr[i] = sum_j w[j+i]*a[j] (positive lag)
            # and xcorr[(N-i)%N] = sum_j w[j-i]*a[j] (negative lag)
            
            grad = 2.0 * (xcorr[:n] + xcorr[(np.arange(n)) % N if N else np.arange(n)][::-1])
            # Hmm this is getting confusing. Let me just compute it directly.
            
            # Actually, let me reconsider. The gradient of sum_k w_k * conv(a,a)_k with respect to a_i is:
            # = 2 * sum_k w_k * d(conv(a,a)_k)/da_i
            # = 2 * sum_k w_k * (a[k-i] * 1_{0<=k-i<n} + a[i-k] * 1_{0<=i-k<n})
            
            # Let's compute this as: 2 * (cross_corr_1 + cross_corr_2)
            # where cross_corr_1[i] = sum_k w_k * a[k-i] * 1_{0<=k-i<n}
            # and cross_corr_2[i] = sum_k w_k * a[i-k] * 1_{0<=i-k<n}
            
            # cross_corr_1[i] = sum_{k=i}^{i+n-1} w_k * a[k-i] = sum_j w_{i+j} * a_j (j=0 to n-1, i+j < 2n-1)
            # cross_corr_2[i] = sum_{k=0}^{i+n-1} w_k * a[i-k] * 1_{i-k>=0} = sum_{k=0}^{i} w_k * a[i-k] (i-k < n)
            #                  = sum_j w_{i-j} * a_j (j=0 to min(i, n-1), i-j >= 0)
            
            # So grad[i] = 2 * (sum_j w[i+j]*a[j] + sum_j w[i-j]*a[j])
            #            = 2 * (xcorr(w,a)[i] + xcorr(w,a)[-i])
            
            # Where xcorr(w,a)[i] = sum_j w[i+j]*a[j] (this is the cross-correlation at lag i)
            
            # Using FFT: xcorr_full = irfft(conj(FFT(w)) * FFT(a), N)
            # xcorr_full[l] = sum_j w[j+l]*a[j] (circular, with zero-padding making it linear)
            # We want xcorr_full[i] for i = 0..n-1 and xcorr_full[N-i] for i = 1..n-1 (negative lags)
            # xcorr_full[N-i] = sum_j w[j-i]*a[j] (since (j+N-i) mod N = j-i for j>=i)
            
            # So grad[i] = 2 * (xcorr_full[i] + xcorr_full[N-i])
            # For i=0: grad[0] = 2 * xcorr_full[0] = 2 * sum_j w[j]*a[j]
            
            xcorr_full = irfft(np.conj(W) * A, fft_sz)
            grad = 2.0 * (xcorr_full[:n] + xcorr_full[fft_sz - np.arange(1, n)])
            if n > 1:
                grad[0] = 2.0 * xcorr_full[0]
            
            # Gradient descent step
            lr = norm / (T * n * (1.0 + step * 0.01))
            a = a - lr * grad
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SGD T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent with random coordinate selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Pick two random coordinates
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # Try to transfer mass from i to j
        a_i = a[i]
        if a_i < 1e-10:
            continue
        
        best_frac = 0
        best_trial = None
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            amt = a_i * frac
            a_trial = a.copy()
            a_trial[i] = a_i - amt
            a_trial[j] += amt
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial * norm / s
            score = fast_autocorr(a_trial).max()
            if score < best_score:
                best_score = score
                best_frac = frac
                best_trial = a_trial
        
        if best_trial is not None:
            a = best_trial
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Random restart perturbation
                pert = rng.uniform(-0.01, 0.01, n)
                pert[i] = 0
                a_trial = a + pert
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-15:
                    a_trial = a_trial * norm / s
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_a = a.copy()
                        best_score = score
                        stall = 0
        
        n_iters += 1
        if verbose and n_iters % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def get_solver():
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def cosine_init(n, freq=1.0, order=2):
    """Generate a cosine-based initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0
    for k in range(order):
        a *= (1.0 + 0.3 * np.cos(2 * np.pi * (k + 1) * freq * x))
    a = np.maximum(a, 0.0)
