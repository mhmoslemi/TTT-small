import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full autocorrelation (self-convolution) a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

    A = rfft(a, N)
    W = rfft(w, N)

                        idx = rng.integers(n)
                        amt = a[idx] * 0.01
                        a[idx] -= amt
                        a[rng.integers(n)] += amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        b = fast_autocorr(a)
                        stuck_count = 0
        
        if verbose and move_count % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairMove: {move_count} moves ({elapsed:.0f}s), "
                  f"score={best_score:.6f}, bmax={b.max():.6f}, T={T:.1f}")
    
    return best_a, best_score
