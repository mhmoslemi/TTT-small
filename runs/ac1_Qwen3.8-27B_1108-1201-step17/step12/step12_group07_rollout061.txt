def fast_convolve(a, b=None):
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_convolve_a(a):
    """Self-convolution of a."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a_fixed):

def solve_lp_best_response(a_fixed, solver=None):

    if prob.status not in ('optimal', 'optimal_inaccurate'):

    b_val = np.asarray(b_var.value, dtype=np.float64)

def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0005 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve_a(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_grad_fast(a, T_val, n, b=None, w=None):
    """Fast LSE gradient computation using FFT where possible."""
    if b is None:

    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Use FFT-based convolution for gradient
    # grad_a[i] = T * sum_k w[k] * 2*a[k-i] (approx, ignoring diagonal)
    # This is 2*T * conv(w, a)[i + n - 1] ... let me compute directly
    # Using the identity: sum_k w[k]*a[k-i] for valid k-i
    # = cross-correlation
    
    # Compute via FFT: we want for each i, sum_k w[k]*a[k-i] where 0<=k-i<n
    # This is (w * reverse(a))[i + n - 1] ... 
    # Actually: let's use the direct loop with numpy vectorization
    # For each k with significant w[k], add 2*w[k]*a[shifted] to grad
    
    # Efficient: use the top contributions
    top_k = 200
    if len(w) > top_k:
        top_indices = np.argsort(w)[-top_k:]

def coordinate_descent_exact(a0, time_limit, max_sweeps=5000):
    """Coordinate descent minimizing max(conv(a,a)) with incremental convolution updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()

        a = np.ones(n) / n
        s = 1.0
    a = a / s

    curr_max = conv.max()
    best_a = a.copy()
    best_max = curr_max
    t0 = time.monotonic()
    last_improve = t0

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - last_improve > max(30.0, time_limit * 0.1):
            break

        sweep_improved = False
        for i in range(n):
            old = a[i]
            if old < 1e-15:
                continue

            # Compute outer_max: max of conv outside [i, i+n-1]
            k_start = i
            k_end = i + n - 1
            outer_max = 0.0
            if k_start > 0:
                outer_max = max(outer_max, conv[0:k_start].max())
            if k_end + 1 < len(conv):
                outer_max = max(outer_max, conv[k_end + 1:].max())

            # base[j] = conv[i+j] - 2*old*a[j] for j != i
            #          conv[i+j] - old^2 for j == i (i.e., k=2i)
            # Wait: conv[i+j] includes the contribution of old at position i.
            # For j != i: contribution is 2*old*a[j], so base[j] = conv[i+j] - 2*old*a[j]
            # For j == i: contribution is old^2, so base[i] = conv[2*i] - old^2
            
            base = conv[k_start:k_end + 1].copy()
            base = base - 2.0 * old * a
            base[i] = conv[2 * i] - old * old

            # f(x) = max(outer_max, max_{j!=i}(base[j] + 2*x*a[j]), base[i] + x^2)
            # Note: base[j] + 2*x*a[j] for j!=i is a line in x
            # base[i] + x^2 is a parabola
            
            # Find upper bound for x:
            # x^2 + base[i] <= curr_max => x <= sqrt(max(0, curr_max - base[i]))
            x_parab_bound = np.sqrt(max(0.0, curr_max - base[i])) if curr_max > base[i] else 0.0
            
            # For lines: base[j] + 2*x*a[j] <= curr_max => x <= (curr_max - base[j])/(2*a[j])
            # Use the minimum positive bound
            pos_a = a > 1e-15
            if np.any(pos_a):
                line_bounds = np.full(n, np.inf)
                line_bounds[pos_a] = (curr_max - base[pos_a]) / (2.0 * a[pos_a])
                line_bounds = line_bounds[base < curr_max]  # only where constraint is active
                if len(line_bounds) > 0:
                    x_line_bound = max(0.0, line_bounds.min())
                else:
                    x_line_bound = 1.0
            else:
                x_line_bound = 1.0

            x_hi = min(max(x_parab_bound, x_line_bound) + 0.1, max(old * 3.0, 1.0 / n * 2))
            if x_hi < 1e-15:
                x_hi = old * 1.5
            x_lo = 0.0

            # Ternary search for minimum of f(x) on [x_lo, x_hi]
            # First, bracket: evaluate at a few points
            def f_of_x(x):
                # max(outer_max, max_j(base[j] + 2*x*a[j]) with j!=i corrected, base[i]+x^2)
                terms = base + 2.0 * x * a
                terms[i] = base[i] + x * x
                return max(outer_max, terms.max())

            # Coarse ternary search
            for _ in range(15):
                m1 = x_lo + (x_hi - x_lo) / 3.0
                m2 = x_hi - (x_hi - x_lo) / 3.0
                if f_of_x(m1) < f_of_x(m2):
                    x_hi = m2
                else:
                    x_lo = m1

            # Fine ternary search
            for _ in range(20):
                m1 = x_lo + (x_hi - x_lo) / 3.0
                m2 = x_hi - (x_hi - x_lo) / 3.0
                if f_of_x(m1) < f_of_x(m2):
                    x_hi = m2
                else:
                    x_lo = m1

            x_opt = (x_lo + x_hi) / 2.0
            f_opt = f_of_x(x_opt)

            if f_opt < curr_max - 1e-12:
                sweep_improved = True
                if f_opt < best_max:
                    best_max = f_opt
                    best_a = a.copy()
                    # Save current state
                    best_a[i] = x_opt
                    last_improve = time.monotonic()

            # Update conv incrementally
            # conv[k_start:k_end+1] += 2*(x_opt - old)*a (all positions)
            # conv[2*i] += (x_opt^2 - old^2) - 2*(x_opt - old)*a[i]
            conv[k_start:k_end + 1] += 2.0 * (x_opt - old) * a
            conv[2 * i] += (x_opt * x_opt - old * old) - 2.0 * (x_opt - old) * a[i]
            a[i] = x_opt

            # Update curr_max (approximate - full max is expensive, track locally)
            local_max = max(outer_max, conv[k_start:k_end + 1].max())
            if local_max > curr_max:
                curr_max = local_max
            # Don't decrease curr_max here (will be exact at end of sweep)

        # At end of sweep, recompute exact max
        curr_max = conv.max()
        if curr_max < best_max:
            best_max = curr_max
            best_a = a.copy()
            last_improve = time.monotonic()

        if (sweep + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent sweep={sweep+1} ({elapsed:.0f}s): max_b={curr_max:.10f} best={best_max:.10f}")

    return best_a, best_max


def multiplicative_weights_subgrad(a0, time_limit, max_steps=100000):
    """Multiplicative weights with subgradient of max conv."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.5
    last_improve = t0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - last_improve > max(20.0, time_limit * 0.15):
            break

        b = fast_convolve_a(a)
        max_b = b.max()
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = time.monotonic()
            eta = min(eta * 1.01, 1.5)
        else:
            eta *= 0.95
            if eta < 1e-8:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coord_descent_lse_hybrid(a0, time_limit):
    """Alternate between coordinate descent on exact max and LSE L-BFGS steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Phase A: Coordinate descent (60% of time)
    t_a = time_limit * 0.6
    a_out, _ = coordinate_descent_exact(a, t_a, max_sweeps=3000)
    sc = score_normalized(a_out)
    if sc < best_score:
        best_score = sc
        best_a = a_out.copy()
        a = a_out.copy()

    # Phase B: LSE polish (30% of time)
    t_b = time_limit * 0.3
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a_lse, sc_lse = lbgfs_logsumexp(a, remaining, T_start=30.0, T_end=0.003, n_stages=20)
        if sc_lse < best_score:
            best_score = sc_lse
            best_a = a_lse.copy()

    # Phase C: Final coordinate descent (10% of time)
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 5:
        a_out2, _ = coordinate_descent_exact(best_a, remaining, max_sweeps=2000)
        sc2 = score_normalized(a_out2)
        if sc2 < best_score:
            best_score = sc2
            best_a = a_out2.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init_asymmetric(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 1: Multiplicative weights (60s) ===
    print(f"\n=== Phase 1: Multiplicative weights at n={n_base} (score={best_score:.8f}) ===")
    t1 = min(60.0, (_global_deadline - time.monotonic()) * 0.05)
    if t1 > 5:
        a_out, _ = multiplicative_weights_subgrad(best_seq, t1, max_steps=200000)
        try_update(a_out, "mul_weights_p1")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: LP-based Nesterov (550s) ===
    print(f"\n=== Phase 2: Nesterov LP at n={best_n} (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(550.0, remaining * 0.35)
    if t2 > 30:
        a_out, _ = nesterov_lp_adaptive(best_seq, t2, solver=solver)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: LSE L-BFGS continuation (400s) ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(400.0, remaining * 0.25)
    if t3 > 20:
        a_lse, sc_lse = lbgfs_logsumexp(best_seq, t3, T_start=80.0, T_end=0.005, n_stages=30)
        try_update(a_lse, "lse_continuation")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Coordinate descent on exact max (500s) ===
    print(f"\n=== Phase 4: Coordinate descent exact (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(500.0, remaining * 0.3)
    if t4 > 30:
        a_cd, _ = coordinate_descent_exact(best_seq, t4, max_sweeps=5000)
        try_update(a_cd, "coord_descent")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: LP Nesterov round 2 (150s) ===
    print(f"\n=== Phase 5: Nesterov LP round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(150.0, remaining * 0.15)
    if t5 > 15:
        a_out2, _ = nesterov_lp_adaptive(best_seq, t5, solver=solver)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Hybrid coord+LSE (100s) ===
    print(f"\n=== Phase 6: Hybrid coordination (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6 = min(100.0, remaining * 0.12)
    if t6 > 15:
        a_h, _ = coord_descent_lse_hybrid(best_seq, t6)
        try_update(a_h, "hybrid_coord_lse")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 80:
        for n_try in [best_n + 150, best_n + 300, best_n - 100, best_n + 500]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 20:
                break
            if n_try < 200:
                continue
            t7 = min(45.0, remaining * 0.3)
            if t7 < 8:
                break
            print(f"  Trying n={n_try}, budget={t7:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick coordinate descent for new length
            a_try, _ = coordinate_descent_exact(init_try, t7, max_sweeps=2000)
            try_update(a_try, f"len_{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
