            if current_gap > 1e-15:
                # Estimate gradient norm
                g = np.zeros(n)
                js = max(0, k_star - n + 1)
                je = min(n - 1, k_star)
                for j in range(js, je + 1):
                    g[j] = 2.0 * a[k_star - j]
                g_norm_sq = np.dot(g, g)
                if g_norm_sq > 1e-20:
                    eta = current_gap / g_norm_sq
                    eta = min(eta, 2.0)
                    scale_down = 1.0

        a = a_new

        if it % 50000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_f
            print(f"      SubGrad it={it} ({elapsed:.0f}s): best_f={best_f:.8f}, score={score:.6f}, eta={eta*scale_down:.6f}")

    return best_a, best_f


def mirror_descent_optimize(a0, time_limit, max_iters=10000000):
    """
    Mirror descent (multiplicative weight) for min max_k (a*a)_k
    subject to sum(a)=1, a>=0. Uses entropy as mirror map.
    Update: a_j <- a_j * exp(-eta * g_j) / Z
    """
    n = len(a0)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    f_lb = 1.0 / (2 * n - 1)
    best_a = a.copy()
    best_f = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = a.copy()
    n_avg = 0
    eta = 0.5  # initial learning rate
    stall_count = 0
    scale = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        f_val = b[k_star]

        # Subgradient g_j = 2 * a_{k_star - j}
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            g[j] = 2.0 * a[k_star - j]

        # Mirror descent update: a_j <- a_j * exp(-eta * g_j) / Z
        log_a = np.log(np.maximum(a, 1e-30))
        new_log_a = log_a - eta * scale * g
        new_log_a = new_log_a - new_log_a.max()  # numerical stability
        weights = np.exp(new_log_a)
        Z = weights.sum()
        if Z < 1e-300:
            break
        a_new = weights / Z
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        if f_val < best_f:
            best_f = f_val
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        avg_a += a_new
        n_avg += 1

        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            f_avg = fast_autocorr(avg_check).max()
            if f_avg < best_f:
                best_f = f_avg
                best_a = avg_check.copy()
                stall_count = 0

        if stall_count > 2000:
            scale *= 0.9
            stall_count = 0
            if scale < 0.005:
                scale = 0.5
                a = best_a.copy()

        a = a_new

        if it % 100000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_f
            print(f"      MirrorDescent it={it} ({elapsed:.0f}s): best_f={best_f:.8f}, score={score:.6f}, eta={eta*scale:.6f}")

    return best_a, best_f


def lse_smoothed_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a using FFT.
    Returns (f_val, grad).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    f_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    # grad_j = 2 * sum_k w_k * a_{k-j}
    # This is 2 * (w_rev * a)_j convolved appropriately
    # (w * a) in the flipped sense: grad = 2 * flip(w) * a (convolution)
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # We need: grad_j = 2 * sum_k w_k * a_{k-j}
    # This is the cross-correlation of w and a, or equivalently conv(w_rev, a)
    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)
    conv = irfft(W_rev * A, fft_size)
    # The result at index j corresponds to sum_k w_rev[k-j] * a[j] ... 
    # Let me think more carefully.
    # We want: grad_j = 2 * sum_{k: k-j valid} w_k * a_{k-j}
    # Let m = k - j, so k = m + j: grad_j = 2 * sum_m w_{m+j} * a_m
    # This is cross-correlation: <w, shift_j(a)> = (w * flip(a))_j ... 
    # Actually: sum_m w_{m+j} * a_m = (w_rev * a)_{j} where w_rev[k] = w_{len-1-k}
    # No, let's use the direct formula:
    # grad_j = 2 * sum_m a_m * w_{m+j}
    # This is the convolution of a with w, evaluated at index j... 
    # (a * w)_j = sum_m a_m * w_{j-m} ... that's not quite right either.
    # Let's just compute it directly for correctness, using the full autocorrelation approach:
    # Actually, the simplest: grad = 2 * (w convolved with a) in the right shift
    # sum_k w_k * a_{k-j} for each j
    # = (w * flip(a)) at position (n-1-j) ... this is getting confusing
    # Let me just use the Toeplitz structure:
    # grad_j = 2 * sum_{k=max(j,0)}^{min(n-1+j, 2n-2)} w_k * a_{k-j} (with a_i=0 outside [0,n-1])
    
    # Simplest correct approach: use the fact that grad = 2 * T^T w where T is the
    # convolution matrix. T^T w at position j = sum_k T[k,j] * w_k = sum_k a_{k-j} * w_k
    # (where T[k,j] = a_{k-j} if 0<=k-j<n, else 0)
    # So grad_j = 2 * sum_k a_{k-j} * w_k
    # This is the same as: for each j, sum over k of a[k-j] * w[k]
    # = convolution of a and w at specific offsets
    
    # The cleanest way: pad and use FFT
    # grad_j = 2 * sum_{k} a_{k-j} * w_k for k-j in [0, n-1]
    # Let's define: for fixed j, grad_j/2 = sum_{m=0}^{n-1} a_m * w_{m+j}
    # This is the cross-correlation of a and w at lag j.
    # cross_corr(a,w)[j] = sum_m a_m * w_{m+j}
    # = (flip(a) * w)[n-1+j] ... 
    
    # Let me just do it with a direct FFT-based cross-correlation:
    # cross_corr(a, w) at lag j = sum_m a_m * w_{m+j}
    # = IDFT( conj(FFT(a)) * FFT(w) ) at appropriate index
    
    FFT_size = 1 << (max(n, len(w)) + n - 1 - 1).bit_length()
    FA = rfft(a, FFT_size)
    FW = rfft(w, FFT_size)
    # Cross-correlation: cc[j] = sum_m a_m * w_{m+j}
    # In frequency domain: cc = IDFT( conj(FA) * FW )... no
    # Actually: cross_corr(a,w)[tau] = sum_m a[m] * w[m+tau]
    # = sum_m a[m] * conj(conj(w[m+tau]))
    # The standard formula: cross_corr(a,w) = IFFT(FFT(a) * conj(FFT(w))) gives sum_m a[m+tau]*w[m]
    # So cross_corr(a,w)[tau] = sum_m a[m+tau]*w[m]
    # We want sum_m a[m]*w[m+tau] = cross_corr(w,a)[tau] = IFFT(FFT(w)*conj(FFT(a)))[tau]
    
    cc = irfft(FW * np.conj(FA), FFT_size)
    # cc[tau] = sum_m w[m+tau] * a[m] for tau >= 0
    # We need grad_j = 2 * cc[j] for j = 0, ..., n-1
    # But cc has length FFT_size, and for tau = j (0-indexed), cc[j] = sum_m w[m+j]*a[m]
    # which is exactly what we need!
    
    grad = 2.0 * cc[:n]
    return f_val, grad


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=15, lr=0.01):
    """LogSumExp-smoothed optimization with Adam optimizer."""
    n = len(a0)
    a_init = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(30, min(200, int(remaining * 100 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            obj, grad_a = lse_smoothed_grad(a, stage_T)

            # Chain rule for z-parameterization
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      AdamLSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_targeting_spectral(a0, n_iter=500, alpha=0.2):
    """
    Peak-targeting spectral shaping: identify autocorrelation peaks and
    selectively suppress the frequency components that contribute to them.
    """
    n = len(a0)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        b = fast_autocorr(a)
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        # Find top peak lags
        k_peak = int(np.argmax(b))
        b_peak = b[k_peak]

        # The peak at lag k_peak is the IDFT of |A|^2 at that position
        # We want to reduce |A(f)|^2 at frequencies where the phase aligns
        # to constructively interfere at lag k_peak
        # More simply: reduce all non-DC spectral components proportionally
        # to how much they contribute to the peak

        # Compute contribution of each frequency to lag k_peak
        # (a*a)_k = sum_f |A(f)|^2 * exp(2*pi*i*f*k/fft_size) / fft_size
        # The "in-phase" contribution is |A(f)|^2 * cos(2*pi*f*k/fft_size)
        # So frequencies where cos is positive contribute to the peak

        freqs = np.arange(len(A))
        phase_angle = 2 * np.pi * freqs * k_peak / fft_size
        in_phase = np.cos(phase_angle)

        # Target: reduce in-phase components more than out-of-phase
        scale = np.ones(len(A))
        # Reduce components that are in-phase with the peak
        in_phase_non_dc = in_phase[1:].copy()
        in_phase_non_dc = np.maximum(in_phase_non_dc, -1.0)
        scale[1:] = 1.0 - alpha * np.maximum(in_phase_non_dc, 0)
        scale[1:] = np.maximum(scale[1:], 0.1)  # don't eliminate completely

        A_new = A * np.sqrt(scale)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize_nonneg(a_new)

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # If this didn't help, try a gentler step
            alpha_local = alpha * 0.5
            scale_local = np.ones(len(A))
            scale_local[1:] = 1.0 - alpha_local * np.maximum(in_phase_non_dc, 0)
            scale_local[1:] = np.maximum(scale_local[1:], 0.1)
            A_new2 = A * np.sqrt(scale_local)
            a_new2 = irfft(A_new2, fft_size)[:n]
            a_new2 = normalize_nonneg(a_new2)
            score2 = 2.0 * n * fast_autocorr(a_new2).max()
            if score2 < best_score:
                best_score = score2
                best_a = a_new2.copy()
                a = a_new2
            else:
                # Reduce alpha
                alpha *= 0.95
                if alpha < 0.01:
                    break

    return best_a


def full_spectrum_flatten(a0, n_iter=1000, alpha=0.3):
    """Standard spectral flattening - flatten |A(f)|^2 to reduce peak."""
    n = len(a0)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(np.maximum(non_dc, 1e-30))))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize_nonneg(a_new)

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build convolution matrix T where (T@b)_k = (conv(a_fixed, b))_k."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    elif choice == 7:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 8:
        # Smoother bump
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.5) ** 2)
    else:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-40 * (x - 0.3) ** 2) + 0.8 * np.exp(-40 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining:
    - Peak-targeting spectral shaping
    - Subgradient with Polyak step
    - Mirror descent (multiplicative weights)
    - LogSumExp Adam
    - LP Nesterov
    - Peak shaving
    With multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def check_time():
        return time_budget - (time.monotonic() - t_global_start)

    def try_update(cand, tag):
        nonlocal best_score, best_seq
        if cand is None:
            return
        cand = np.maximum(cand, 0)
        sc = evaluate_safe(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand / cand.sum()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{tag}]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick spectral shaping at base length (10% time) ===
    remaining = check_time()
    print(f"\n=== Phase 1: Spectral shaping at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.08, 80.0)
    if t1 > 10:
        # Full flattening
        a_spec = full_spectrum_flatten(best_seq, n_iter=2000, alpha=0.3)
        try_update(a_spec, f"flatten_n{n_base}")
        # Peak-targeting
        a_pt = peak_targeting_spectral(best_seq, n_iter=1000, alpha=0.15)
        try_update(a_pt, f"peaktarget_n{n_base}")

    # === Phase 2: Subgradient with Polyak step (25% time) ===
    remaining = check_time()
    print(f"\n=== Phase 2: Subgradient (Polyak) at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 300.0)
    if t2 > 20:
        a_sg, _ = polyak_subgradient_optimize(best_seq, t2)
        try_update(a_sg, f"subgrad_n{n_base}")

    # === Phase 3: Mirror descent (15% time) ===
    remaining = check_time()
    print(f"\n=== Phase 3: Mirror descent at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_md, _ = mirror_descent_optimize(best_seq, t3)
        try_update(a_md, f"mirrordescent_n{n_base}")

    # === Phase 4: Adam LSE (15% time) ===
    remaining = check_time()
    print(f"\n=== Phase 4: Adam LSE at n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_lse, _ = adam_lse_optimize(best_seq, t4, T_start=80.0, T_end=0.05, n_stages=20, lr=0.005)
        try_update(a_lse, f"adam_lse_n{n_base}")

    # === Phase 5: LP Nesterov at base length (20% time) ===
    remaining = check_time()
    print(f"\n=== Phase 5: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 60:
        a_lp, _ = nesterov_lp_optimize(best_seq, t5, solver)
        try_update(a_lp, f"lp_n{n_base}")

    # === Phase 6: Try longer length (10% time) ===
    remaining = check_time()
    n_long = int(round(n_base * 1.3))
    print(f"\n=== Phase 6: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral + subgradient
        a_spec_long = full_spectrum_flatten(init_long, n_iter=500, alpha=0.3)
        try_update(a_spec_long, f"flatten_n{n_long}")
        a_sg_long, _ = polyak_subgradient_optimize(a_spec_long, t6 * 0.5)
        try_update(a_sg_long, f"subgrad_n{n_long}")
        a_lse_long, _ = adam_lse_optimize(a_spec_long, t6 * 0.3, T_start=50.0, T_end=0.1, n_stages=10, lr=0.003)
        try_update(a_lse_long, f"adam_lse_n{n_long}")

    # === Phase 7: Peak shaving polish (10% time) ===
    remaining = check_time()
    print(f"\n=== Phase 7: Peak shaving (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 120.0)
    if t7 > 10:
        a_ps, _ = peak_shaving(best_seq, t7, seed=456)
        try_update(a_ps, "peakshave_1")

    # === Phase 8: Final Adam LSE polish with low temperature (5% time) ===
    remaining = check_time()
    print(f"\n=== Phase 8: Final Adam LSE polish (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.10, 80.0)
    if t8 > 10:
        a_final, _ = adam_lse_optimize(best_seq, t8, T_start=10.0, T_end=0.01, n_stages=15, lr=0.002)
        try_update(a_final, "final_adam_lse")

    # === Phase 9: Another peak shaving round ===
    remaining = check_time()
    if remaining > 20:
        t9 = min(remaining * 0.3, 60.0)
        if t9 > 10:
            a_ps2, _ = peak_shaving(best_seq, t9, seed=789)
            try_update(a_ps2, "peakshave_2")

    # === Phase 10: Try a slightly shorter length (sometimes shorter is better due to 2n factor) ===
    remaining = check_time()
    n_short = int(round(n_base * 0.85))
    if remaining > 60:
        print(f"\n=== Phase 10: Shorter n={n_short} (remaining={remaining:.0f}s) ===")
        t10 = min(remaining * 0.3, 100.0)
        if t10 > 20:
            init_short = interpolate_sequence(best_seq, n_short)
            a_spec_s = full_spectrum_flatten(init_short, n_iter=1000, alpha=0.3)
            try_update(a_spec_s, f"flatten_n{n_short}")
            a_sg_s, _ = polyak_subgradient_optimize(a_spec_s, t10 * 0.5)
            try_update(a_sg_s, f"subgrad_n{n_short}")

    # Final peak shaving with remaining time
    remaining = check_time()
    if remaining > 15:
        t_final = min(remaining * 0.8, 60.0)
        if t_final > 5:
            a_ps_f, _ = peak_shaving(best_seq, t_final, seed=1000)
            try_update(a_ps_f, "final_peakshave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
