    b = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    fft_size = 1 << (2 * n - 2).bit_length()

    t0 = time.monotonic()
    t = 0
    eval_interval = 200

    # Geometric average with adaptive rate
    avg_a = a.copy()
    alpha_avg = 0.02

    # Tail buffer for recent average
    W = 3000
    tail_sum = np.zeros(n)
    tail_buf = np.zeros((W, n))
    tail_idx = 0
    tail_filled = 0

    while time.monotonic() - t0 < time_limit - 1:
        # Compute conv(a, b)
        A_fft = rfft(a, fft_size)
        B_fft = rfft(b, fft_size)
        conv_ab = irfft(A_fft * B_fft, fft_size)[:2 * n - 1]

        k_star = int(np.argmax(conv_ab))

        # Subgradients
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)

        a_new = a.copy()
        b_new = b.copy()

        if js <= je:
            idx = np.arange(js, je + 1)
            sg_a = b[k_star - idx]
            sg_b = a[k_star - idx]
            a_new[idx] = a[idx] * np.exp(-eta * sg_a)
            b_new[idx] = b[idx] * np.exp(eta * sg_b)
            # For indices outside [js, je], sg=0 so exp(0)=1, no change

        a_new = np.maximum(a_new, 0)
        na = a_new.sum()
        if na < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / na

        b_new = np.maximum(b_new, 0)
        nb = b_new.sum()
        if nb < 1e-300:
            b_new = np.ones(n) / n
        else:
            b_new = b_new / nb

        a = a_new
        b = b_new

        # Update geometric average
        avg_a = (1.0 - alpha_avg) * avg_a + alpha_avg * a

        # Update tail buffer
        if tail_filled < W:
            tail_sum += a
            tail_buf[tail_filled] = a
            tail_filled += 1
        else:
            old = tail_buf[tail_idx]
            tail_sum -= old
            tail_sum += a
            tail_buf[tail_idx] = a
            tail_idx = (tail_idx + 1) % W

        t += 1

        if t % eval_interval == 0:
            # Check geometric average
            sc = 2.0 * n * fast_autocorr(avg_a).max()
            if sc < best_score:
                best_score = sc
                best_a = avg_a.copy()

            # Check tail average
            tavg = tail_sum / max(1, tail_filled)
            sc_t = 2.0 * n * fast_autocorr(tavg).max()
            if sc_t < best_score:
                best_score = sc_t
                best_a = tavg.copy()

            # Check current iterate
            sc_a = 2.0 * n * fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()

            # Check column player
            sc_b = 2.0 * n * fast_autocorr(b).max()
            if sc_b < best_score:
                best_score = sc_b
                best_a = b.copy()

            if verbose and t % (eval_interval * 25) == 0:
                elapsed = time.monotonic() - t0
                print(f"      MW t={t} ({elapsed:.0f}s): best={best_score:.6f}, L={conv_ab.max():.8f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL, cp.MOSEK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, warm_b=None):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix T where (T @ b)[k] = conv(a_fixed, b)[k]
    # T[k, j] = a_fixed[k - j] if 0 <= k-j < n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP best-response with adaptive restart.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Running averages
    avg_long = np.zeros(n)
    n_long = 0
    WINDOW_SHORT = 50
    short_buf = [None] * WINDOW_SHORT
    short_idx = 0
    short_sum = np.zeros(n)
    short_count = 0

    stall_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 1:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Long running average
        avg_long += a
        n_long += 1

        # Short running average
        if short_buf[short_idx] is not None:
            short_sum -= short_buf[short_idx]
        else:
            short_count += 1
        short_buf[short_idx] = a.copy()
        short_sum += a
        short_idx = (short_idx + 1) % WINDOW_SHORT

        # Evaluate candidates
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if n_long % 50 == 0 and n_long >= 100:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        eff_short = min(short_count, WINDOW_SHORT)
        if eff_short >= WINDOW_SHORT and n_long % 50 == 0:
            avg_s = short_sum / eff_short
            score_s = 2.0 * n * fast_autocorr(avg_s).max()
            if score_s < best_score:
                best_score = score_s
                best_a = avg_s.copy()
                stall_count = 0

        # Adaptive restart
        if stall_count > 500 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            short_buf = [None] * WINDOW_SHORT
            short_idx = 0
            short_sum = np.zeros(n)
            short_count = 0
            stall_count = 0

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def compute_topk_grad(a, T_temp, topk):
    """
    Compute top-k LSE objective and gradient w.r.t. a.
    obj = (1/T)*log(sum_{k in top_k} exp(T*c_k)) where c = autocorr(a)
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T_temp * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T_temp
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T_temp, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_grad(a, T_temp, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=25, topk=12):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = normalize_nonneg(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 2:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        base_lr = 3e-3 * min(1.0, 60.0 / n)
        lr = base_lr * (0.75 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 100.0) * 4))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Multi-scale peak shaving with adaptive step sizes.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.99
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:20]
        top_peaks = top_peaks[:30]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:20]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            n_pert = rng.integers(1, 4)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    else:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence."""
    a = normalize_nonneg(a0)
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining MW game, LP Nesterov, LSE, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: MW Game (fast bulk optimization) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: MW Game n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_out, _ = mw_game(best_seq, t1, eta=2.0, verbose=True)
        try_update(a_out, f"mw_n{n_base}_eta2")

        # Try a second MW run with different eta
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 30:
            t1b = min(remaining * 0.15, 200.0)
            if t1b > 20:
                a_out2, _ = mw_game(best_seq, t1b, eta=4.0, verbose=False)
                try_update(a_out2, f"mw_n{n_base}_eta4")

    # === Phase 2: LP Nesterov (precision refinement) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LP Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 3: LSE with Adam (smooth polish) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LSE Adam n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = adam_topk_lse(best_seq, t3, T_start=80.0, T_end=0.3,
                                  n_stages=20, topk=20)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        print(f"\n=== Phase 4: Multi-length ({remaining:.0f}s left) ===")

        # Try slightly longer sequence with LP
        n_long = n_base + 200
        t4a = min(remaining * 0.20, 250.0)
        if t4a > 40:
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = nesterov_lp_optimize(init_long, t4a, solver)
            try_update(a_out, f"lp_n{n_long}")

        # Try MW at longer length
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 40:
            n_long2 = n_base + 400
            t4b = min(remaining * 0.25, 300.0)
            if t4b > 30:
                init_long2 = interpolate_sequence(best_seq, n_long2)
                a_out, _ = mw_game(init_long2, t4b, eta=2.0, verbose=False)
                try_update(a_out, f"mw_n{n_long2}")

    # === Phase 5: LSE with larger topk ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"\n=== Phase 5: LSE large-topk ({remaining:.0f}s left) ===")
        t5 = min(remaining * 0.20, 200.0)
        if t5 > 15:
            a_out, _ = adam_topk_lse(best_seq, t5, T_start=50.0, T_end=0.5,
                                      n_stages=15, topk=50)
            try_update(a_out, f"lse_big_topk_n{n_base}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving ({remaining:.0f}s left) ===")
    t6 = min(remaining * 0.25, 150.0)
    if t6 > 10:
        a_out, _ = peak_shaving(best_seq, t6, seed=456)
        try_update(a_out, "peak_shave_1")

    # === Phase 7: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"\n=== Phase 7: Final LSE polish ({remaining:.0f}s left) ===")
        t7 = min(remaining * 0.30, 150.0)
        if t7 > 10:
            a_out, _ = adam_topk_lse(best_seq, t7, T_start=5.0, T_end=0.1,
                                      n_stages=10, topk=10)
            try_update(a_out, "final_lse")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final peak shave ({remaining:.0f}s left) ===")
            a_out, _ = peak_shaving(best_seq, t8, seed=789)
            try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
