def lp_best_response(a, M, solver):

        return None
    b_val = b_var.value
    if b_val is None:
        return None
    b_val = np.maximum(np.asarray(b_val, dtype=np.float64).flatten(), 0.0)
    if b_val.sum() < 1e-15:

    s0 = a.sum()

    else:
        a = a * norm / s0

    t0 = time.monotonic()
    
    y = a.copy()
    prev_y = a.copy()
    n_iters = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:

        else:

        else:

            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:

            else:
                x_new = x_new * norm / s_x
        
        prev_y = y.copy()

def coordinate_polish(a0, time_limit, seed=42):
    """Coordinate descent / peak shaving polish."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:20]
        
        improved = False
        for k in top_k:
            if best_score < 1.42:  # Near theoretical minimum, stop
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Find the pair (i, j) with i+j=k that contributes most
            max_contrib_idx = np.argmax(contrib)
            src = idx[max_contrib_idx]
            
            # Find destination: position with low contribution to all top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Pick among candidates with lowest value
            candidates = candidates[np.argsort(a[candidates])][:max(1, len(candidates)//4)]
            dst = candidates[rng.integers(len(candidates))]
            
            # Try different transfer fractions
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score

def structured_init(n, rng, shape=None):
    """Generate a structured initial sequence."""
    if shape is None:
        shape = rng.integers(0, 8)
    
    x = np.linspace(0.0, 1.0, n)
    
    if shape == 0:
        # Uniform
        a = np.ones(n)
    elif shape == 1:
        # Triangular
        a = 1.0 - np.abs(2.0*x - 1.0)
    elif shape == 2:
        # Arc-sine / U-shaped
        a = 1.0 / np.sqrt(np.maximum(x * (1-x), 1e-10))
    elif shape == 3:
        # Gaussian
        a = np.exp(-4.0 * (x - 0.5)**2)
    elif shape == 4:
        # Cosine taper
        a = np.cos(np.pi * x / 2.0)**2 + 0.01
    elif shape == 5:
        # Sine power
        a = np.sin(np.pi * x)**1.5
    elif shape == 6:
        # Multi-frequency
        a = 0.5 + 0.3*np.cos(4*np.pi*x) + 0.15*np.cos(10*np.pi*x) + 0.05*np.cos(20*np.pi*x)
        a = np.maximum(a, 0.01)
    else:
        # Concentrated at edges
        a = x**2 + (1-x)**2 + 0.1
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def entropy_best_response(y, M, n_iter=50, lr=0.1):
    """Entropy-regularized best response (multiplicative weights)."""
    n = len(y)
    b = np.ones(n) / n  # Initialize uniform
    
    for _ in range(n_iter):
        # Compute (y*b)_k for all k
        conv_yb = fast_convolve(y, b)
        # Find constraints that are violated
        excess = conv_yb - M
        # Update b multiplicatively: b_j *= exp(-lr * max(0, excess at positions involving j))
        # Simpler: compute gradient of max_k (y*b)_k - M at b
        k_star = int(np.argmax(conv_yb))
        # grad[j] = y[k_star - j] if valid
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = y[idx]
        
        # Multiplicative update
        b = b * np.exp(-lr * grad / (lr * grad).max() + 1e-10)
        b = np.maximum(b, 1e-15)
        b = b / b.sum()
    
    return b

def fast_convolve(a, b):
    """Fast linear convolution of two sequences."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]
