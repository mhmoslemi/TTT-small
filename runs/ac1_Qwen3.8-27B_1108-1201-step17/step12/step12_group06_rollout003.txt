                for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
                    delta = frac * a_jmax
                    if delta <= 0 or delta >= a[j_max]:
                        continue
                    a_trial = a.copy()
                    a_trial[j_max] -= delta
                    a_trial[i_target] += delta
                    if a_trial[j_max] < 0:
                        continue
                    sc_trial = score_normalized(a_trial)
                    if sc_trial < best_new_score:
                        best_new_score = sc_trial
                        best_delta = (delta, i_target)
            
            if best_delta != 0:
                delta, i_target = best_delta
                a[j_max] -= delta
                a[i_target] += delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                
                cur_score = score_normalized(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            # Try symmetric moves
            for _ in range(5):
                if time.monotonic() - t0 >= time_limit:
                    break
                # Pick a random pair and try to equalize
                i1 = np.random.randint(0, n)
                i2 = np.random.randint(0, n)
                if i1 == i2:
                    continue
                diff = (a[i1] - a[i2]) * 0.1
                a[i1] -= diff
                a[i2] += diff
                a = np.maximum(a, 0)
                a = a / a.sum()
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, max_sweeps=100):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for sw in range(max_sweeps):
        if time.monotonic() - t0 >= time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        # Shuffled coordinate order
        order = np.random.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 >= time_limit:
                break
            
            # Optimize a[i] while keeping sum=1
            # a_new[i] = a[i] + delta, a_new[others] = a[others] * (1 - a[i] - delta) / (1 - a[i])
            # For simplicity: scale all other coords when changing a[i]
            # Actually, let's do: a_new = a; a_new[i] += delta; then renormalize
            # Or better: use the "simplex coordinate" parametrization
            
            # Simple approach: try a few values of a[i]
            ai = a[i]
            if ai < 1e-10:
                # Try increasing
                for new_ai in [ai * 2, ai + 0.001, 0.001, 0.01]:
                    if new_ai >= 1 - 1e-10:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_ai
                    # Scale others to maintain sum=1
                    others_sum = 1 - new_ai
                    cur_others_sum = a_trial.sum() - new_ai
                    if cur_others_sum > 1e-15:
                        a_trial[np.arange(n) != i] *= others_sum / cur_others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            else:
                # Try increasing and decreasing
                for frac in [0.5, 1.5, 0.8, 1.2, 2.0]:
                    new_ai = ai * frac
                    if new_ai <= 0 or new_ai >= 1 - 1e-10:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_ai
                    others_sum = 1 - new_ai
                    cur_others_sum = a_trial.sum() - new_ai
                    if cur_others_sum > 1e-15:
                        a_trial[np.arange(n) != i] *= others_sum / cur_others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length using cubic spline."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use cubic interpolation via scipy
    from scipy.interpolate import CubicSpline
    cs = CubicSpline(old_x, a_norm, bc_type='not-a-knot')
    a_new = cs(new_x)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new
