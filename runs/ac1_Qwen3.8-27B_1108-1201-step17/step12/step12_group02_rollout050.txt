"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Proximal gradient with Adam + Nesterov momentum on the simplex,
   optimizing the LogSumExp-smoothed objective via FFT-based gradients
2. Multi-restart with diverse structured initializations
3. Adaptive multi-stage temperature annealing
4. Multi-length exploration (interpolation + re-optimization)
5. Final coordinate-descent peak shaving

Key innovation: Direct simplex-constrained Adam optimization (no log-space
parameterization), with efficient O(n log n) gradient via FFT cross-correlation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly. Assumes a >= 0."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}. O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    sorted_v = np.sort(v)[::-1]
    css = np.cumsum(sorted_v)
    rho = np.where(sorted_v > (css - 1.0) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (css[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)


def cross_corr_for_grad(w, a):
    """
    Compute c[i] = sum_{k=0}^{n-1} w[i+k]*a[k] for i = 0..n-1.
    w has length 2n-1, a has length n.
    This gives the gradient of the LSE objective (up to factor of 2).
    """
    n = len(a)
    w_len = len(w)  # 2n-1
    a_rev = a[::-1].copy()
    conv_len = w_len + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_grad_simplex(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*b)) w.r.t. a, where b = a*a.
    Uses FFT-based cross-correlation. O(n log n).
    a is assumed to be on the simplex (sum=1, a>=0).
    """
    n = len(a)
    b = fast_autocorr(a)
    # Softmax weights
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    w = exp_Tb / exp_Tb.sum()
    # Gradient: 2 * cross_corr(w, a)
    cc = cross_corr_for_grad(w, a)
    grad = 2.0 * cc
    return grad, w


def adam_simplex_lse(a0, time_limit, T_start=200.0, T_end=1.0,
                     n_stages=20, lr_init=0.05, beta1=0.9, beta2=0.999,
                     eps=1e-8, seed=42, verbose=True):
    """
    Adam optimizer with Nesterov momentum on the simplex,
    minimizing the LogSumExp-smoothed max-convolution objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    v_t = np.zeros(n)  # Nesterov velocity
    step_count = 0

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, min(500, int(time_limit / n_stages / 0.05)))

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        stage_iters = min(iters_per_stage, int(remaining / 0.05) + 1)
        lr = lr_init / (1.0 + 0.1 * si)

        for it in range(stage_iters):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov lookahead
            a_look = a + v_t * beta1

            # Gradient at lookahead point
            grad, w = lse_grad_simplex(a_look, stage_T)

            # Adam update
            step_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** step_count)
            v_hat = v / (1.0 - beta2 ** step_count)

            # Nesterov velocity update
            v_new = beta1 * v_t - lr * m_hat / (np.sqrt(v_hat) + eps)
            v_t = v_new

            # Step with projection
            a_new = a + v_t
            a_new = simplex_project(a_new)

            a = a_new

            # Check score periodically
            if it < 100:
                check = 5
            elif it < 1000:
                check = 20
            else:
                check = 50

            if it % check == 0:
                sc = fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if verbose and si % 4 == 0:
            elapsed = time.monotonic() - t0
            sc = fast_score(a)
            print(f"      Adam-LSE stage {si+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): "
                  f"cur={sc:.6f}, best={best_score:.6f}")

    # Final check
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def lse_lbfgs_refined(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=15, seed=42):
    """
    Refined LSE + L-BFGS in log-space with finer temperature schedule.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(30, min(150, int(50 + 100 * si / max(1, n_stages))))

        def func_z(z_in):
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a_local = e / e.sum()
            b = fast_autocorr(a_local)
            Tb = stage_T * b
            Tb_shift = Tb - Tb.max()
            return (Tb.max() + np.log(np.exp(Tb_shift).sum())) / stage_T

        def jac_z(z_in):
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a_local = e / e.sum()
            b = fast_autocorr(a_local)
            Tb = stage_T * b
            Tb_shift = Tb - Tb.max()
            exp_Tb = np.exp(Tb_shift)
            w = exp_Tb / exp_Tb.sum()
            cc = cross_corr_for_grad(w, a_local)
            grad_a = 2.0 * cc
            S = np.dot(grad_a, a_local)
            grad_z = a_local * (grad_a - S)
            return grad_z

        try:
            res = minimize(fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-LBFGS stage {si+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): "
                  f"best={best_score:.6f}")

    return best_a, best_score


def peak_shave_aggressive(a0, time_limit, seed=123):
    """
    Aggressive coordinate descent: for each top peak, find the best
    single-coordinate move that reduces the score.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(20, n)]

        improved = False
        for k_idx in top_k[:10]:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            # Find best destination
            # Avoid positions that contribute to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick lowest-a candidates
            sort_idx = candidates[np.argsort(a[candidates])[:max(10, len(candidates) // 5)]]
            if len(sort_idx) == 0:
                continue

            for dst in sort_idx[:15]:
                dst = int(dst)
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            if rng.integers(5) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

            if stall > 600:
                break

    return best_a, best_score


def diverse_init(n, rng, variant=None):

    elif variant == 9:

             + np.exp(-20 * (x - 0.85) ** 2))
    else:

        a = np.ones(n)

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiscale_optimize(a0, time_limit, coarse_n=100, seed=42):
    """
    Multi-scale optimization: optimize on a coarse grid, then refine.
    """
    n = len(a0)
    t0 = time.monotonic()

    # Coarse grid
    a_coarse = interpolate_to_length(a0, coarse_n)

    # Optimize coarse
    remaining_coarse = min(time_limit * 0.3, 120.0)
    a_c, _ = adam_simplex_lse(a_coarse, remaining_coarse, T_start=50.0, T_end=1.0,
                              n_stages=10, lr_init=0.1, seed=seed, verbose=False)

    # Interpolate back
    a_refined = interpolate_to_length(a_c, n)

    # Fine optimization

                              n_stages=15, lr_init=0.03, seed=seed + 1, verbose=False)

    return a_f

            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine working length
    n_parent = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = diverse_init(n_parent, rng, variant=4)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Screen diverse initializations ===
    print(f"\n=== Phase 1: Screening at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    screen_time = min(remaining * 0.20, 300.0)
    if screen_time > 30:
        t_per_init = screen_time / 6.0
        candidates = [best_seq]
        for v in [0, 3, 4, 7, 9, 10]:
            init = diverse_init(n_parent, rng, variant=v)
            candidates.append(init)

        for idx_c, init in enumerate(candidates):
            remaining = time_budget - (time.monotonic() - t_global)
            t_trial = min(t_per_init, remaining * 0.15)
            if t_trial < 5:
                break
            a_out, _ = adam_simplex_lse(init, t_trial, T_start=100.0, T_end=5.0,
                                         n_stages=8, lr_init=0.08, seed=seed + idx_c,
                                         verbose=False)
            try_update(a_out, f"screen_v{idx_c}")

    # === Phase 2: Full Adam-LSE on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam-LSE refinement at n={len(best_seq)} ===")
    t_adam = min(remaining * 0.25, 400.0)
    if t_adam > 30:
        a_out, _ = adam_simplex_lse(best_seq, t_adam, T_start=200.0, T_end=1.0,
                                     n_stages=20, lr_init=0.04, beta1=0.9, beta2=0.999,
                                     seed=seed + 100, verbose=True)
        try_update(a_out, "adam_full")

    # === Phase 3: LSE-LBFGS polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE-LBFGS at n={len(best_seq)} ===")
    t_lse = min(remaining * 0.18, 280.0)
    if t_lse > 30:
        a_out, _ = lse_lbfgs_refined(best_seq, t_lse, T_start=80.0, T_end=0.3,
                                      n_stages=15, seed=seed + 200)
        try_update(a_out, "lse_lbfgs")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-length exploration ===")

    # Try longer length
    n_long = int(round(len(best_seq) * 1.3))
    t_long = min(remaining * 0.22, 300.0)
    if t_long > 30 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)
        # Quick Adam on longer
        t_adam_long = t_long * 0.6
        a_out, _ = adam_simplex_lse(init_long, t_adam_long, T_start=150.0, T_end=2.0,
                                     n_stages=15, lr_init=0.03, seed=seed + 300,
                                     verbose=False)
        try_update(a_out, f"adam_n{n_long}")

        # Follow up with LSE-LBFGS
        remaining = time_budget - (time.monotonic() - t_global)
        t_lse_long = min(remaining * 0.08, 100.0)
        if t_lse_long > 15 and best_seq is not None:
            init_long2 = interpolate_to_length(best_seq, n_long)
            a_out, _ = lse_lbfgs_refined(init_long2, t_lse_long, T_start=50.0, T_end=0.5,
                                          n_stages=10, seed=seed + 350)
            try_update(a_out, f"lse_n{n_long}")

    # Try slightly different length
    n_alt = int(round(len(best_seq) * 0.85))
    remaining = time_budget - (time.monotonic() - t_global)
    t_alt = min(remaining * 0.10, 120.0)
    if t_alt > 20 and best_seq is not None:
        init_alt = interpolate_to_length(best_seq, n_alt)
        a_out, _ = adam_simplex_lse(init_alt, t_alt, T_start=100.0, T_end=2.0,
                                     n_stages=12, lr_init=0.05, seed=seed + 400,
                                     verbose=False)
        try_update(a_out, f"adam_n{n_alt}")

    # === Phase 5: Aggressive peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Peak Shaving at n={len(best_seq)} ===")
    t_ps = min(remaining * 0.12, 150.0)
    if t_ps > 10 and best_seq is not None:
        a_out, _ = peak_shave_aggressive(best_seq, t_ps, seed=seed + 500)
        try_update(a_out, "peak_shave")

    # === Phase 6: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish (score={best_score:.6f}) ===")

    if best_seq is not None:
        # Short Adam with low learning rate for fine tuning
        t_f1 = min(remaining * 0.30, 100.0)
        if t_f1 > 10:
            a_out, _ = adam_simplex_lse(best_seq, t_f1, T_start=30.0, T_end=0.5,
                                         n_stages=15, lr_init=0.01, beta1=0.95,
                                         beta2=0.999, seed=seed + 600, verbose=False)
            try_update(a_out, "final_adam")

        # Short LSE-LBFGS with very low T
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.30, 80.0)
        if t_f2 > 10:
            a_out, _ = lse_lbfgs_refined(best_seq, t_f2, T_start=10.0, T_end=0.1,
                                          n_stages=12, seed=seed + 700)
            try_update(a_out, "final_lse")

        # Final peak shave
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.40, 80.0)
        if t_f3 > 5:
            a_out, _ = peak_shave_aggressive(best_seq, t_f3, seed=seed + 800)
            try_update(a_out, "final_shave")

    # Final safety evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
