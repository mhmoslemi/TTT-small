def fast_convolve(a, b=None):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def qp_refine(a0, time_limit, max_active=50):
    """QP refinement on active constraint set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a /= a.sum()
    
    best_a = a.copy()
    best_score = evaluate_quant(a)
    t0 = time.monotonic()
    
    for rnd in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_convolve(a)
        max_val = conv_a.max()
        
        # Active set: lags where conv >= 0.95 * max
        threshold = max_val * 0.95
        active = np.where(conv_a >= threshold)[0]
        if len(active) > max_active:
            # Keep the top max_active
            active = np.argsort(conv_a[active])[::-1][:max_active] + np.where(conv_a >= threshold)[0][:max_active]
            active = np.unique(active[:max_active])
        
        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k - idx
            q = cp.sum(a_var[idx] * a_var[other])
            constraints.append(q <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except:
                continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_qp = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0)
        a_qp /= a_qp.sum()
        
        score_qp = evaluate_quant(a_qp)
        if score_qp < best_score:
            best_score = score_qp
            best_a = a_qp.copy()
            a = a_qp
        else:
            # Try mixing
            improved = False
            for frac in [0.2, 0.4, 0.6, 0.8]:
                a_mix = (1 - frac) * a + frac * a_qp

                    best_a = a_mix.copy()
                    a = a_mix

            try:
                prob.solve(solver=sv)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        aq = np.maximum(np.asarray(av.value, dtype=np.float64), 0)
        aq /= aq.sum()
        sq = eval_quick(aq)
        if sq < best_sc:
            best_sc, best_a, a = sq, aq.copy(), aq
            print(f"    QP r{rnd}: {best_sc:.6f}")
        else:
            imp = False
            for fr in [0.25, 0.4, 0.6, 0.8]:
                am = np.maximum((1 - fr) * a + fr * aq, 0)
                am /= am.sum()
                sm = eval_quick(am)
                if sm < best_sc:
                    best_sc, best_a, a, imp = sm, am.copy(), am, True
                    break
            if not imp and rnd >= 4:
                break
            na = min(na * 2, n_active_max)
    return best_a, best_sc


# =========================================================
#  Constraint-Research QP  (Mathematica-style)
# =========================================================

def qp_constraint_research(a0, time_limit, max_rounds=8, max_active=25):
    """
    At each round identify the single worst lag, solve a QP that
    forces that lag below the current best peak, and add the
    complementary constraint.  Iterate with widening active set.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a /= a.sum()
    best_a, best_sc = a.copy(), eval_quick(a)
    t0 = time.monotonic()
    active_lags = set()
    cur_peak = fast_convolve(a).max()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        pk = int(c.argmax())
        active_lags.add(pk)
        # also add lags within 1% of the peak
        for k in np.where(c >= 0.99 * c.max())[0]:
            active_lags.add(int(k))
        if len(active_lags) > max_active:
            active_lags = set(list(active_lags)[:max_active])

        av = cp.Variable(n, nonneg=True)
        Mv = cp.Variable()
        cons = [cp.sum(av) == 1]
        for k in sorted(active_lags):
            js, je = max(0, k - n + 1), min(n - 1, k)
            if js > je:
                continue
            ix = np.arange(js, je + 1)
            cons.append(cp.sum(av[ix] * av[k - ix]) <= Mv)
        prob = cp.Problem(cp.Minimize(Mv), cons)
        solved = False
        for sv in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=sv)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        aq = np.maximum(np.asarray(av.value, dtype=np.float64), 0)
        aq /= aq.sum()
        sq = eval_quick(aq)
        if sq < best_sc:
            best_sc, best_a, a = sq, aq.copy(), aq
            cur_peak = fast_convolve(a).max()
            print(f"    CR-QP r{rnd}: {best_sc:.6f}  (active={len(active_lags)})")
    return best_a, best_sc


# =========================================================
#  Peak Shaving
# =========================================================

def peak_shave(a0, time_limit, seed=0):
    """Greedy peak reduction by transferring mass to low-impact positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a /= a.sum()
    best_a, best_sc = a.copy(), eval_quick(a)
    t0 = time.monotonic()
    stall = 0
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        mx = c.max()
        pk = int(c.argmax())
        js, je = max(0, pk - n + 1), min(n - 1, pk)
        if js > je:
            break
        ix = np.arange(js, je + 1)
        ct = a[ix] * a[pk - ix]
        ord_ = np.argsort(ct)[::-1]
        done = False
        for oi in ord_[:8]:
            if done:
                break
            src, src_o = int(ix[oi]), int(pk - ix[oi])
            for dst in rng.choice(n, size=40, replace=False):
                if done:
                    break
                if dst == src and dst == src_o:
                    continue
                for fr in [0.001, 0.004, 0.01, 0.03, 0.06]:
                    amt = min(a[src], a[src_o] if src != src_o else a[src]) * fr
                    if amt < 1e-14:
                        continue
                    at = a.copy()
                    at[src] -= amt
                    if src != src_o:
                        at[src_o] -= amt * 0.25
                    at[dst] += amt
                    at = np.maximum(at, 0)
                    at /= at.sum()
                    sc = eval_quick(at)
                    if sc < best_sc * (1.0 - 1e-10):
                        a, best_sc, best_a = at, sc, at.copy()
                        stall, done = 0, True
                        break
        if not done:
            stall += 1
            if stall % 25 == 0:
                at = a.copy()
                for _ in range(rng.integers(1, 4)):
                    s2, d2 = rng.integers(n), rng.integers(n)
                    am2 = at[s2] * rng.uniform(0.002, 0.02)
                    at[s2] -= am2
                    at[d2] += am2
                at = np.maximum(at, 0)
                at /= at.sum()
                sc2 = eval_quick(at)
                if sc2 < best_sc:
                    a, best_sc, best_a, stall = at, sc2, at.copy(), 0
            if stall > 600:
                break
    return best_a, best_sc


# =========================================================
#  Interpolation & Initialisation
# =========================================================

def interp_seq(f, new_n):
    """Linear-resample a normalised sequence to a new length."""
    f = np.asarray(f, dtype=np.float64)
    if len(f) == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, len(f))
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    s = fi.sum()
    return fi / s if s > 1e-15 else np.ones(new_n) / new_n


def make_laplace(n, spread):
    x = np.arange(n) - n / 2.0
    f = np.exp(-np.abs(x) / max(1.0, n * spread))
    return f / f.sum()


def make_bimodal(n, spread):
    x = np.arange(n) - n / 2.0
    f = np.exp(-np.abs(x) / max(1.0, n * spread))
    f = 1.0 - f  # U-shape
    f = np.maximum(f, 0.0)
    return f / f.sum()


def make_cosine(n, power):
    x = np.linspace(0, np.pi, n)
    f = np.cos(x / 2.0) ** power
    f = np.maximum(f, 0)
    return f / f.sum()


def make_smoothed_sinusoid(n, freq, width):
    x = np.arange(n)
    f = 1.0 + np.cos(2.0 * np.pi * freq * x / n)
    f = f ** width
    f = np.maximum(f, 0)
    return f / f.sum()

#  Main entry point

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search for the lowest-peak-autocorrelation non-negative
    sequence.  Returns a list of floats.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    TIME_BUDGET = min(float(budget_s) - 90.0, 1710.0)

    def remaining():
        return TIME_BUDGET - (time.monotonic() - t_start)

    best_seq = None
    best_score = np.inf

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_quick(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")

    # ---------- Phase 0: Gather diverse initial candidates ----------
    print("=== Phase 0: Initialisation ===")
    candidates = []

    # 0a: parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            candidates.append(('parent', parent))
            print(f"  parent: n={len(parent)}, score={eval_quick(parent):.6f}")
    except Exception:
        pass

    n_base = len(parent) if parent is not None else 1600

    # 0b: parametric families at n_base
    for label, fn in [
        ('cosine2',   lambda: make_cosine(n_base, 2.0)),
        ('cosine4',   lambda: make_cosine(n_base, 4.0)),
        ('lap-0.3',   lambda: make_laplace(n_base, 0.30)),
        ('lap-0.5',   lambda: make_laplace(n_base, 0.50)),
        ('bimo-0.3',  lambda: make_bimodal(n_base, 0.30)),
        ('sino-3-2',  lambda: make_smoothed_sinusoid(n_base, 3, 2.0)),
    ]:
        try:
            c = fn()
            candidates.append((label, c))
        except Exception:
            pass

    # 0c: random
    for ri in range(3):
        c = rng.random(n_base)
        c /= c.sum()
        candidates.append((f'rand{ri}', c))

    # Quick-evaluate and keep top-3
    scored = []
    for label, c in candidates:
        sc = eval_quick(c)
        scored.append((sc, label, c))
    scored.sort(key=lambda x: x[0])
    print(f"  top-3 inits: {[(l, f'{s:.6f}') for s, l, _ in scored[:3]]}")

    # 0d: fast coordinate polish on each top-3 candidate (3 s each)
    for sc_i, lab_i, c_i in scored[:3]:
        c_p = c_i.copy()
        for _ in range(200):  # very fast: just move mass from peak contributor to random
            ci = fast_convolve(c_p)
            pk = int(ci.argmax())
            js, je = max(0, pk - len(c_p) + 1), min(len(c_p) - 1, pk)
            if js > je:
                break
            ix = np.arange(js, je + 1)
            ct = c_p[ix] * c_p[pk - ix]
            oi = int(np.argmax(ct))
            src = int(ix[oi])
            dst = int(rng.integers(len(c_p)))
            if dst == src:
                continue
            amt = c_p[src] * 0.01
            c_p[src] -= amt
            c_p[dst] += amt
            c_p = np.maximum(c_p, 0)
            c_p /= c_p.sum()
        try_update(c_p, f'polish_{lab_i}')

    if best_seq is None:
        # Fallback
        best_seq = scored[0][2].copy()
        best_score = eval_quick(best_seq)
        print(f"  fallback init: {best_score:.6f}")

    # ---------- Phase 1: MW Nesterov  (main heavy lifting) ----------
    rem = remaining()
    print(f"\n=== Phase 1: MW Nesterov  n={len(best_seq)}  budget={rem:.0f}s ===")
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        a_out, sc_out, iters = nesterov_mw(best_seq, t1, n_inner=80, K=3)
        print(f"  NM done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, 'NM-main')

        # Second short round of NM from the current best
        rem = remaining()
        t1b = min(rem * 0.12, 180.0)
        if t1b > 20:
            a_out2, sc_out2, iters2 = nesterov_mw(best_seq, t1b, n_inner=60, K=4)
            try_update(a_out2, 'NM-round2')

    # ---------- Phase 2: QP Active-Set ----------
    rem = remaining()
    print(f"\n=== Phase 2: QP Active-Set  budget={rem:.0f}s ===")
    t2 = min(rem * 0.12, 150.0)
    if t2 > 15:
        a_out, sc_out = qp_active_set(best_seq, t2, n_active_max=40)
        try_update(a_out, 'QP-active')

    # ---------- Phase 3: Constraint-Research QP ----------
    rem = remaining()
    print(f"\n=== Phase 3: Constraint-Research QP  budget={rem:.0f}s ===")
    t3 = min(rem * 0.10, 120.0)
    if t3 > 10:
        a_out, sc_out = qp_constraint_research(best_seq, t3, max_rounds=8, max_active=30)
        try_update(a_out, 'QP-research')

    # ---------- Phase 4: Peak Shaving ----------
    rem = remaining()
    print(f"\n=== Phase 4: Peak Shaving  budget={rem:.0f}s ===")
    t4 = min(rem * 0.20, 250.0)
    if t4 > 10:
        a_out, sc_out = peak_shave(best_seq, t4, seed=1)
        try_update(a_out, 'PS-1')

    # ---------- Phase 5: Length Exploration ----------
    rem = remaining()
    n_cur = len(best_seq)
    n_try_list = sorted(set([n_cur + 100, n_cur + 200, n_cur + 400, n_cur - 100,
                              n_cur - 200, 1200, 2000, 2400]),
                        key=lambda x: abs(x - n_cur))
    n_try_list = [nt for nt in n_try_list if 400 <= nt <= 4000 and nt != n_cur][:4]

    print(f"\n=== Phase 5: Length exploration on {n_try_list}  budget={rem:.0f}s ===")
    per_len = max(30, (rem * 0.20) / max(1, len(n_try_list)))
    for n_try in n_try_list:
        if remaining() < 60:
            break
        print(f"  --- n={n_try} ---")
        init_try = interp_seq(best_seq, n_try)
        # Quick NM
        t5 = min(per_len * 0.5, 120.0)
        a_out, sc_out, _ = nesterov_mw(init_try, t5, n_inner=60, K=3)
        try_update(a_out, f'NM-n{n_try}')
        # Quick peak shave
        rem = remaining()
        t5b = min(per_len * 0.3, 60.0)
        if t5b > 8:
            a_ps, sc_ps = peak_shave(a_out, t5b, seed=n_try % 1000)
            try_update(a_ps, f'PS-n{n_try}')

    # ---------- Phase 6: Final polishing ----------
    rem = remaining()
    print(f"\n=== Phase 6: Final polishing  budget={rem:.0f}s ===")
    t6a = min(rem * 0.35, 120.0)
    if t6a > 8:
        a_out, sc_out = peak_shave(best_seq, t6a, seed=42)
        try_update(a_out, 'PS-final')

    rem = remaining()
    t6b = min(rem * 0.35, 100.0)
    if t6b > 8:
        a_out, sc_out = qp_active_set(best_seq, t6b, n_active_max=30)
        try_update(a_out, 'QP-final')

    rem = remaining()
    t6c = min(rem * 0.30, 80.0)
    if t6c > 5:
        a_out, sc_out = peak_shave(best_seq, t6c, seed=99)
        try_update(a_out, 'PS-very-final')

    # ---------- Final evaluation ----------
    if best_seq is not None:
        try:
            final_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"\nFINAL evaluate_sequence score: {final_score:.6f}  (n={len(best_seq)})")
        except Exception as e:
            print(f"\nFinal eval error: {e}; using proxy score: {best_score:.6f}")
    else:
        print("\nERROR: no valid sequence found, returning [1.0]")
        return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
