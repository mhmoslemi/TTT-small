"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with pressure-based peak weighting
2. Multi-restart LSE (log-sum-exp) smoothed optimization with L-BFGS

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def cross_corr_weighted(w, a):
    """
    Compute sum_k w[k] * a[k-i] for each i (0 <= i < n).
    This gives the gradient of the LSE-smoothed objective.
    w has length 2n-1, a has length n.
    Result has length n.
    """
    n = len(a)
    # We want result[i] = sum_k w[k] * a[k-i] where 0 <= k-i < n
    # This is a cross-correlation: result[i] = sum_j w[i+j] * a[j] (where j = k-i)
    # = sum_j a[j] * w[i+j]
    # This is the cross-correlation of a and w, evaluated at lag i.
    # cross_corr[i] = sum_j a[j] * w[i+j]
    # = (a * w_reversed)[n-1+i] ... let me think differently.
    
    # Alternatively: result[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # where w[i+j] is defined only if 0 <= i+j < 2n-1
    
    # This is equivalent to taking the convolution of a with w (reversed) and extracting the right part.
    # conv(a, w_rev)[k] = sum_i a[i] * w_rev[k-i] = sum_i a[i] * w[len(w)-1-(k-i)]
    # Hmm, this is getting confusing. Let me use a direct FFT approach.
    
    # result[i] = sum_j a[j] * w[i+j] for j=0..n-1 (where i+j < len(w))
    # = (a * r)[i] where r[m] = w[m] (but we need to handle the shift)
    
    # Let's think of it as: pad a to length 2n-1, multiply element-wise with w... no.
    
    # Actually, result[i] = sum_j a[j] * w[i+j] is the cross-correlation of a and w.
    # cross_corr_a_w[lag] = sum_j a[j] * w[j + lag]
    # So result[i] = cross_corr_a_w[i]
    
    # To compute via FFT: cross_corr = ifft(conj(FFT(a_padded)) * FFT(w_padded))
    # But we need to be careful with the padding.
    
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    fft_size = 1 << (4 * n - 2).bit_length()
    A = rfft(a_pad, fft_size)
    W = rfft(w, fft_size)
    cc = irfft(np.conj(A) * W, fft_size)
    # cc[lag] = sum_j a_pad[j] * w[j + lag]... 
    # Actually, the standard cross-correlation via FFT:
    # cross_corr[x,y][k] = sum_n x[n] * y[n+k]
    # Via FFT: cross_corr = ifft(conj(FFT(x)) * FFT(y))
    # So cc[i] = sum_j a[j] * w[j+i] which is what we want!
    return cc[:n]


def lse_obj_grad_z(z, T):
    """
    Objective and gradient in z-space where a = softmax(z).
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (where 0<=k-i<n)
    # = 2 * cross_corr(w, a)[i] where cross_corr[w,a][i] = sum_j w[i+j]*a[j]
    cc = cross_corr_weighted(w, a)
    grad_a = 2.0 * cc

    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def accelerated_mirror_descent(a0, time_limit, eta0=0.05, decay_rate=0.6,
                                pressure_sigma=0.3, seed=42, verbose=False):
    """
    Nesterov-accelerated mirror descent with pressure-based peak weighting.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Nesterov momentum
    y = a.copy()
    beta = 0.0  # will increase to 0.9
    beta_final = 0.85
    beta_ramp = 500

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        # Pressure-based weighting: weight proportional to (b[k] - b_mean)^2 for peaks
        excess = b - b_mean
        positive_excess = np.maximum(excess, 0)
        # Use a "pressure" model: weights = exp(sigma * excess) / sum
        sigma = pressure_sigma / max(b_max - b_mean, 1e-15)
        w = np.exp(sigma * positive_excess)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(len(b)) / len(b)
        else:
            w = w / w_sum

        # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
        cc = cross_corr_weighted(w, y)
        grad = 2.0 * cc

        # Learning rate
        k = max(1, min(t, beta_ramp))
        beta_t = beta_final * (1 - 1.0 / (1 + k))
        eta = eta0 / (1.0 + t / 400.0) ** decay_rate

        # Mirror descent update on y
        y_new = y * np.exp(-eta * grad)
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y + 0.01 * rng.standard_normal(n)
            y_new = np.maximum(y_new, 0)
            s_new = y_new.sum()
            y_new = y_new / max(s_new, 1e-15)
        else:
            y_new = y_new / s_new

        # Nesterov update
        beta_t = beta_final * min(1.0, t / float(beta_ramp))
        a_new = (1 + beta_t) * y_new - beta_t * a
        a_new = np.maximum(a_new, 0)
        s_new2 = a_new.sum()
        if s_new2 < 1e-15:
            a_new = y_new
            s_new2 = y_new.sum()
        else:
            a_new = a_new / s_new2

        a = a_new
        y = y_new

        if t % 20 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > 300:
            noise = 0.03 * rng.standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            stall_count = 0

    return best_a, best_score


def lse_optimize_multi(a0, time_limit, seed=42, verbose=False):
    """
    Multi-restart LSE optimization with adaptive temperature.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    best_a = a_init.copy()
    best_score = fast_score(a_init)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Multiple restarts with different temperatures
    configs = [
        (80.0, 1.0, 8),
        (40.0, 0.5, 10),
        (20.0, 0.2, 12),
        (10.0, 0.1, 10),
    ]

    for ci, (T_start, T_end, n_stages) in enumerate(configs):
        if time.monotonic() - t0 > time_limit:
            break

        # Start from current best (or perturbed version for restarts)
        if ci == 0:
            z = np.log(np.maximum(a_init, 1e-30))
        else:
            # Perturb the best solution
            pert = best_a * (1 + 0.05 * rng.standard_normal(n))
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            z = np.log(np.maximum(pert, 1e-30))

        temps = np.geomspace(T_start, T_end, n_stages)

        for si, stage_T in enumerate(temps):
            if time.monotonic() - t0 > time_limit:
                break
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 3:
                break

            iters = max(10, min(50, int(20 + 30 * si / max(1, n_stages))))

            def func(z_in):
                val, _ = lse_obj_grad_z(z_in, stage_T)
                return val

            def jac(z_in):
                _, g = lse_obj_grad_z(z_in, stage_T)
                return g

            try:
                res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                               options={'maxiter': iters, 'ftol': 1e-15,
                                        'gtol': 1e-8, 'maxfun': iters * 100,
                                        'disp': False})
                z = res.x
                z_shift = z - z.max()
                a = np.exp(z_shift)
                a = a / a.sum()
                sc = fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if verbose:
                        print(f"    LSE r{ci} s{si}: {sc:.6f}")
            except Exception:
                pass

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=123, verbose=False):
    """
    Impact-aware coordinate descent.
    For each coordinate, compute its impact on the current peaks and
    try to reduce it by moving mass to low-impact positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.mean()

        # Find top peaks
        top_k = np.argsort(b)[::-1][:min(20, n)]

        # Compute impact of each coordinate on the max peaks
        # impact[i] = sum over peak lags k of b[k] * |partial b[k]/partial a[i]|
        # = sum_k w[k] * 2 * a[k-i]
        w = np.zeros(len(b))
        for k in top_k:
            w[k] = b[k] - b_mean  # or just 1.0 for top-k

        impact = cross_corr_weighted(w, a)
        impact = np.abs(impact)

        # Sort coordinates by impact (highest first)
        order = np.argsort(impact)[::-1]

        improved = False
        for idx in order[:min(50, n)]:
            i = int(idx)
            if a[i] < 1e-10:
                continue

            # Find target: low-impact coordinate
            low_impact = np.argsort(impact)[:min(20, n // 4)]
            dst = int(low_impact[rng.integers(len(low_impact))])
            if dst == i:
                continue

            # Try moving a fraction of a[i] to a[dst]
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[dst] += amt
                if a_trial[i] < 0:
                    break
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    # Recompute impact for next iteration
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Try random perturbations
            if rng.integers(2) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

            if stall > 200:
                break

    return best_a, best_score


def diverse_init(n, rng, variant=None):

    elif variant == 2:

        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        # Sinc-squared
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        # Linear ramp
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        # Multi-harmonic
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        # V-shaped
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Asymmetric ramp (key: asymmetry reduces center peak)
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.05
    elif variant == 11:
        # Strong asymmetry
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x) + 0.01
    else:
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0, n)

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            print(f"Parent: n={n_parent}, score={fast_score(parent):.6f}")
            best_seq = parent.copy()
            best_score = fast_score(parent)
    except Exception:
        n_parent = 1600
        best_seq = diverse_init(n_parent, rng, variant=4)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
