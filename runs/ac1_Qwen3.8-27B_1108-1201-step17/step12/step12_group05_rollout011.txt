from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build Toeplitz matrix T where (T@b)_k = (a*b)_k."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_best_solver():
    """Find the fastest working LP solver."""
    n_test = 200
    rng = np.random.default_rng(0)
    a_test = rng.random(n_test)
    a_test /= a_test.sum()
    T_test = build_conv_matrix(a_test)
    
    best_solver = None
    best_time = float('inf')
    
    for solver_name in [cp.CBC, cp.GLPK, cp.CPLEX, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        try:
            b_var = cp.Variable(n_test, nonneg=True)
            M_var = cp.Variable()
            constraints = [T_test @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            t0 = time.monotonic()
            prob.solve(solver=solver_name)

                    best_time = elapsed
                    best_solver = solver_name
        except Exception:
            continue

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def compute_lse_grad(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*b)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.001, n_stages=25, lr=0.005):
    """LogSumExp-smoothed optimization with Adam, multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters = max(10, min(300, int(remaining * 120 / max(1, n_stages - si))))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_fn(z)
            grad_a, _ = compute_lse_grad(a, stage_T)
            
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 20 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.4f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=50000, lr_start=0.01, lr_end=0.0001):
    """Subgradient method for the constraint max_k (a*a)_k <= M.
    Projects onto simplex and uses averaged iterate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    lrs = np.geomspace(lr_start, lr_end, n_steps)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        k_max = b.argmax()
        
        # Subgradient of max_k (a*a)_k w.r.t. a_j:
        # d(max_k (a*a)_k)/da_j = 2 * a_{k_max - j} (if valid) or 2 * a_{j - (k_max)} (if k_max > j)
        # Actually (a*a)_k = sum_i a_i * a_{k-i}, so d/d a_j = a_{k-j} + a_{j-k} if both valid
        # But since we're at k = k_max, and (a*a)_k = sum_i a_i * a_{k-i}:
        # d(a*a)_{k_max} / d a_j = a_{k_max - j} if 0 <= k_max-j < n, plus a_{j} if k_max == 0... 
        
        # More carefully: (a*a)_k = sum_{j: 0<=k-j<n} a_j * a_{k-j}
        # d/d a_j: we get a_{k-j} if 0<=k-j<n, and a_{j'} if 0<=j'-k... 
        # Actually by symmetry (a*a)_k = sum_j a_j * a_{k-j} where a_i=0 for i<0 or i>=n
        # d(a*a)_{k_max}/d a_j = a_{k_max - j} + a_{j - k_max} if both in range
        # But since (a*a) is symmetric, d(a*a)_{k_max}/d a_j = 2 * a_{|k_max - j|} if |k_max-j| < n
        # Wait no. Let me be more careful.
        # (a*a)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_j * a_{k-j}
        # d/d a_j of this (for j in the valid range) = a_{k-j}
        # But a_j also appears in (a*a)_{k'} for other k'. However we're only differentiating
        # (a*a)_{k_max}, so:
        # d(a*a)_{k_max} / d a_j = a_{k_max - j} if 0 <= k_max-j < n, else 0
        # PLUS if we consider that a_j appears as a_{k-i} when i = j, i.e., k-j = k-j... 
        # Actually no, the derivative is just a_{k_max - j} if 0 <= k_max-j < n.
        
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        for j in range(js, je + 1):
            grad[j] = a[k_max - j]
        # Also by the symmetric part (a appears on the "right" side too)
        # (a*a)_k = sum_j a_{k-j} * a_j, so d/d a_j also gives a_{k-j} from the "left" side
        # But that's the same thing! (a*a)_k is the same whether you write sum a_j a_{k-j} or sum a_{k-j} a_j
        # So the gradient is just: grad[j] = a_{k_max - j} for valid j
        # But wait, we need to be careful. If we write (a*a)_k = sum_j a_j * a_{k-j},
        # then d/d a_j [a_j * a_{k-j}] = a_{k-j} (treating a_{k-j} as constant w.r.t. a_j)
        # AND d/d a_j [a_{j'} * a_{k-j'}] for j' = k-j gives: a_j (treating a_j as the "second" factor)
        # So total: grad[j] = a_{k-j} + a_{j-(k-j)} ... no this is getting confusing.
        
        # Let me just use the FFT-based approach:
        # The gradient of (a*a)_{k_max} w.r.t. a is 2 * (a reversed, shifted) correlated with delta at k_max
        # Simpler: just use the fact that grad = 2 * convolve_flip(a, delta_{k_max})[n-1:n-1+n]
        # which is 2 * a shifted
        
        # Actually, the cleanest way: (a*a)_k = <a, shift_k(a)> where shift_k shifts by k
        # d/d a_j (a*a)_k = (shift_k a)_j + (shift_{-k} a)_j = a_{k-j} + a_{-k+j} (with bounds)
        # = a_{k-j} if 0<=k-j<n, plus a_{j-k} if 0<=j-k<n
        
        grad = np.zeros(n)
        # First term: a_{k_max - j} for j where 0 <= k_max - j < n
        js1 = max(0, k_max - n + 1)
        je1 = min(n - 1, k_max)
        if js1 <= je1:
            idx = np.arange(js1, je1 + 1)
            grad[idx] += a[k_max - idx]
        # Second term: a_{j - k_max} for j where 0 <= j - k_max < n (i.e., j >= k_max)
        js2 = max(0, k_max)
        je2 = min(n - 1, k_max + n - 1)
        if js2 <= je2 and k_max > 0:
            idx2 = np.arange(js2, je2 + 1)
            grad[idx2] += a[idx2 - k_max]
        
        # Project: a_new = proj_simplex(a - lr * grad)
        lr_t = lrs[step]
        a_raw = a - lr_t * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_raw_sorted = np.sort(a_raw)[::-1]
        css = np.cumsum(a_raw_sorted)
        rho = np.nonzero(a_raw_sorted * np.arange(1, n + 1) > (css - 1))[0][-1]
        theta = (css[rho] - 1) / (rho + 1.0)
        a_new = np.maximum(a_raw - theta, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        avg_a += a
        n_avg += 1
        
        if step % 500 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if n_avg >= 100:
                avg_check = avg_a / n_avg
                score_avg = 2.0 * n * fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
    
    # Final check with averaged iterate
    if n_avg > 0:
        avg_final = avg_a / n_avg
        score_avg = 2.0 * n * fast_autocorr(avg_final).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_final.copy()
    
    score_final = 2.0 * n * fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()
    
    return best_a, best_score


def peak_shaving_aggressive(a0, time_limit, seed=42):
    """Aggressive multi-strategy peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        
        improved = False
        
        # Strategy 1: Move from peak contributor to low-value element
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            best_dest = -1
            best_trial_score = best_score
            
            # Try multiple random destinations
            n_tries = min(20, n // 10)
            tries = rng.choice(n, size=n_tries, replace=False)
            for dst in tries:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_trial_score:
                        best_trial_score = score
                        best_dest = dst
                        best_frac = frac
                        best_amt = amt
            
            if best_dest >= 0 and best_trial_score < best_score - 1e-14:
                a[src] -= best_amt
                a[best_dest] += best_amt
                a = a / a.sum()
                best_score = best_trial_score
                best_a = a.copy()
                improved = True
                break
        
        # Strategy 2: Two-way swap (redistribute between two elements)
        if not improved:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10 and a[j] > 1e-10:
                for mix in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    total = a[i] + a[j]
                    a_trial = a.copy()
                    a_trial[i] = total * mix
                    a_trial[j] = total * (1 - mix)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        
        # Strategy 3: Spread mass from the largest element
        if not improved:
            i_max = a.argmax()
            j = rng.integers(n)
            if j != i_max:
                amt = a[i_max] * rng.uniform(0.001, 0.02)
                a_trial = a.copy()
                a_trial[i_max] -= amt
                a_trial[j] += amt
                if a_trial[i_max] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
        
        if not improved:
            stall += 1
            if stall > 500:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1 + 0.2 * np.cos(8 * np.pi * x))
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = np.abs(np.cos(2 * x)) ** 0.5
    elif choice == 4:
        a = rng.random(n)
        a = a + 0.5 * np.convolve(a, np.ones(5) / 5, mode='same')
    else:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 0.5 - 0.4 * np.abs(x - 0.3) - 0.2 * np.abs(x - 0.7))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a_out = spectral_shaping(best_seq, n_iter=500, alpha=0.05)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [final_spectral]: {best_score:.6f}")
    
    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
