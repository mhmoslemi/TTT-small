def fast_autocorr(a):

def fast_autocorr(a):
    """Full linear autocorrelation a*a using FFT."""
    n = len(a)
    n_out = 2 * n - 1
    fft_size = 1 << (n_out - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:n_out]


def normalize(a):
    """Normalize to sum=1, ensure non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def get_solver():
    """Find a working convex solver."""
    for name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

            p.solve(solver=name, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:

    return cp.ECOS


def build_T(a):
    """Build convolution matrix T where (T @ b)[k] = (a*b)[k].
    T is (2n-1) x n."""
    n = len(a)
    n_out = 2 * n - 1
    T = np.zeros((n_out, n))
    for j in range(n):
        row_start = j
        row_end = min(j + n, n_out)
        T[row_start:row_end, j] = a[j]
    return T


def solve_best_response(a, solver):
    """Solve LP: min M s.t. (a*b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def softmax_gradient_descent(a0, time_limit, verbose=True):
    """Gradient descent in softmax parameterization."""
    n = len(a0)
    a = normalize(a0)
    # Initialize log parameters
    log_t = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = 0.5  # Learning rate
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute a from log_t (softmax)
        log_t_shifted = log_t - log_t.max()
        exp_t = np.exp(log_t_shifted)
        a = exp_t / exp_t.sum()

        # Compute objective and subgradient
        b = fast_autocorr(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Find active lags (top 5)
        top_lags = np.argsort(b)[::-1][:5]
        # Weight by proximity to max
        weights = np.exp((b[top_lags] - max_b) * 50.0)
        weights = weights / weights.sum()

        # Compute subgradient w.r.t. a
        grad_a = np.zeros(n)
        for w, k in zip(weights, top_lags):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                other = k - j
                grad_a[j] += 2.0 * w * a[other]

        # Gradient w.r.t. log_t: grad_log_t = a * (grad_a - sum(a * grad_a))
        # (chain rule through softmax)
        mean_grad = np.sum(a * grad_a)
        grad_log_t = a * (grad_a - mean_grad)

        # Adaptive learning rate
        norm_grad = np.linalg.norm(grad_log_t)
        if norm_grad > 1e-15:
            step = eta / norm_grad
            log_t = log_t - step * grad_log_t

        # Decay learning rate
        eta *= 0.9999

        if it % 1000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    SoftGD it={it} ({elapsed:.0f}s): best={best_score:.8f}, cur={cur_score:.8f}")

        if eta < 1e-6:
            break

    return best_a, best_score

                        a_try = a_try / max(a_try.sum(), 1e-15)

            if stall % 5 == 0:

def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def cosine_init(n):
    """Generate a cosine-squared initial sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2.0) ** 2
    return normalize(a)

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_norm = p / p.sum()
            p_score = eval_safe(parent_norm)
            print(f"Parent: n={len(parent_norm)}, score={p_score:.8f}")
            best_seq = parent_norm.copy()
            best_score = p_score
    except Exception as e:
        print(f"No valid parent: {e}")

    n_base = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = cosine_init(n_base)
        best_score = eval_safe(best_seq)
        print(f"Default cosine init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} ({t1:.0f}s budget) ===")
    if t1 > 60:
        a_out, sc_out, iters = nesterov_optimize(best_seq, t1, solver)
        print(f"  Done: {iters} iters, score={sc_out:.8f}")
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: KKT polishing ===
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 2: KKT Polish at n={len(best_seq)} ({t2:.0f}s budget) ===")
    if t2 > 20:
        a_out, sc_out = kkt_polish(best_seq, t2, solver)
        print(f"  Done: score={sc_out:.8f}")
        try_update(a_out, "kkt")

    # === Phase 3: Try nearby lengths ===
    nearby_lengths = []
    for dl in [-200, -100, 100, 200, 300, 400]:
        nl = n_base + dl
        if 800 <= nl <= 3000 and nl not in nearby_lengths:
            nearby_lengths.append(nl)

    for n_try in nearby_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 100:
            break
        t3 = min(remaining * 0.15, 200.0)
        if t3 < 40:
            continue
        print(f"\n=== Phase 3: LP Nesterov at n={n_try} ({t3:.0f}s budget) ===")
        init_try = interpolate_to_length(best_seq, n_try)
        a_out, sc_out, iters = nesterov_optimize(init_try, t3, solver, verbose=False)
        print(f"  Done: {iters} iters, score={sc_out:.8f}")
        try_update(a_out, f"lp_n{n_try}")

        # Quick KKT on this length
        remaining = time_budget - (time.monotonic() - t_start)
        t3b = min(remaining * 0.05, 40.0)
        if t3b > 10:
            a_out2, sc_out2 = kkt_polish(a_out, t3b, solver, verbose=False)
            try_update(a_out2, f"kkt_n{n_try}")

    # === Phase 4: Softmax gradient descent on best ===
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Softmax GD at n={len(best_seq)} ({t4:.0f}s budget) ===")
    if t4 > 20:
        a_out, sc_out = softmax_gradient_descent(best_seq, t4)
        print(f"  Done: score={sc_out:.8f}")
        try_update(a_out, "softgd")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_start)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Peak Shaving at n={len(best_seq)} ({t5:.0f}s budget) ===")
    if t5 > 10:
        a_out, sc_out = peak_shaving(best_seq, t5, seed=123)
        print(f"  Done: score={sc_out:.8f}")
        try_update(a_out, "peak_shave_1")

    # === Phase 6: Final LP Nesterov round ===
    remaining = time_budget - (time.monotonic() - t_start)
    t6 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 6: Final LP Nesterov at n={len(best_seq)} ({t6:.0f}s budget) ===")
    if t6 > 30:
        a_out, sc_out, iters = nesterov_optimize(best_seq, t6, solver, verbose=False)
        print(f"  Done: {iters} iters, score={sc_out:.8f}")
        try_update(a_out, "lp_final")

    # === Phase 7: Final polishing ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 30:
        t7 = min(remaining * 0.4, 60.0)
        if t7 > 10:
            print(f"\n=== Phase 7: Final KKT ({t7:.0f}s budget) ===")
            a_out, sc_out = kkt_polish(best_seq, t7, solver, verbose=False)
            try_update(a_out, "kkt_final")

        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 15:
            t8 = min(remaining * 0.6, 40.0)
            if t8 > 5:
                print(f"\n=== Phase 8: Final Peak Shave ({t8:.0f}s budget) ===")
                a_out, sc_out = peak_shaving(best_seq, t8, seed=999)
                try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    # Return as list of floats
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
