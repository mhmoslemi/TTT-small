        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_tapered(a_fixed, solver, top_m=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Uses tapered constraints: only top top_m entries of conv(a_fixed, 1) matter.
    """
    n = len(a_fixed)
    # Determine which constraints to keep
    if top_m is not None and top_m < 2 * n - 1:
        # Use all positions where a_fixed has significant mass
        # The important constraints are where the convolution with b could be large
        # Approximate: keep positions near where a_fixed * a_fixed is largest
        aa = fast_autocorr(a_fixed)
        sorted_idx = np.argsort(aa)[::-1]
        keep_idx = set(sorted_idx[:top_m].tolist())
        # Always include positions 0 and 2n-2
        keep_idx.add(0)
        keep_idx.add(2 * n - 2)
        keep_arr = np.array(sorted(keep_idx))
    else:
        keep_arr = np.arange(2 * n - 1)

    n_keep = len(keep_arr)

    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for k in keep_arr:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a_fixed[j])

    if len(rows) == 0:
        return None, None

    T_sp = cp.SPSD  # We'll use numpy sparse
    from scipy.sparse import csr_matrix
    T_mat = csr_matrix((vals, (rows, cols)), shape=(n_keep, n))

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Convert to dense if small enough for solver compatibility
    if n_keep * n < 2000000:
        T_dense = T_mat.toarray()
        constraints = [T_dense @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_em
a(a0, time_limit, solver, max_iters=500000, ema_decay=0.999):
    """
    Nesterov-accelerated LP best-response with exponential moving average.
    Key improvements: EMA (not simple average), tapered LP constraints.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Exponential moving average
    ema = a.copy()
    eval_count = 0

    # Tapered LP: start with all constraints, reduce over time
    top_m_current = 2 * n - 1

    stall_count = 0
    last_improve = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Taper constraints: after 200 iters, reduce to top 80%
        if t > 200:
            top_m_current = max(400, int((2 * n - 1) * 0.8))

        b_star, M_val = solve_lp_tapered(y, solver, top_m=top_m_current)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Update EMA
        ema = ema_decay * ema + (1 - ema_decay) * a
        eval_count += 1

        # Evaluate candidates periodically
        if t % 20 == 0 or t < 50:
            candidates = [a, y, ema]
            for cand in candidates:
                sc = 2.0 * n * fast_autocorr(cand).max()
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = cand.copy()
                    stall_count = 0
                    last_improve = t

        stall_count += 1

        # Adaptive restart if stalled too long
        if stall_count > 1500 and t > 500:
            a = best_a.copy()
            y = best_a.copy()
            ema = best_a.copy()
            stall_count = 0
            if t % 3000 == 0:
                print(f"      NLP-EMA restart at t={t}, best={best_score:.6f}")

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP-EMA t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    # Final check
    sc = 2.0 * n * fast_autocorr(ema).max()
    if sc < best_score:
        best_score = sc
        best_a = ema.copy()

    return best_a, best_score


def peak_sensitivity_search(a0, time_limit, seed=42):
    """
    Targeted local search using peak sensitivity.
    For each top peak, identify which a[i] contribute most,
    then move mass away from those positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_pos(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks
        top_k = min(15, len(b))
        top_indices = np.argsort(b)[::-1][:top_k]
        threshold = bmax * 0.99

        improved = False

        for k in top_indices:
            if b[k] < threshold:
                continue

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            # Contribution of each a[i] to peak at k
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # The "sensitivity" of peak k to a[i] is proportional to a[k-i]
            # Moving mass from idx[src] reduces peak by ~2*a[src]*a[k-src]/sum(a)
            src = idx[np.argmax(contrib)]

            # Find safe destinations: positions that don't contribute much to any top peak
            danger_weight = np.zeros(n)
            for k2 in top_indices[:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                if len(idx2) > 0:
                    danger_weight[idx2] += a[k2 - idx2]

            # Safe positions: low danger weight, not the source
            safe_mask = np.arange(n) != src
            safe_scores = danger_weight * a
            safe_idx = np.where(safe_mask)[0]
            if len(safe_idx) == 0:
                continue

            # Sort by safety (low danger*value first)
            safe_sorted = safe_idx[np.argsort(safe_scores[safe_idx])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 10)]
            if len(dst_pool) == 0:
                continue

            # Try multiple destinations and step sizes
            best_delta = 0
            best_move = None
            for dst in dst_pool[:10]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.035, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    delta = score - best_score
                    if delta < best_delta:
                        best_delta = delta
                        best_move = a_trial.copy()

            if best_move is not None and best_delta < -1e-14:
                a = best_move
                best_score = a.sum()  # placeholder, will recompute
                best_score = 2.0 * n * fast_autocorr(a).max()
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random exploration
            n_pert = rng.integers(2, 6)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 200:
                break

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=20, topk=12):
    """
    Log-sum-exp gradient descent on the top-k peaks.
    Different from previous: uses full LSE (not just top-k) in later stages.
    """
    n = len(a0)
    a_init = normalize_pos(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.75 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2 = 0.8, 0.999

        # Use full LSE in later stages (topk = all)
        k_use = topk if si < n_stages // 2 else min(len(fast_autocorr(a_init)), topk * 3)

        max_it = max(20, int(min(remaining, 60.0) * 2))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            a = softmax_fn(z)
            c = fast_autocorr(a)
            k_actual = min(k_use, len(c))
            top_idx = np.argsort(c)[::-1][:k_actual]
            top_vals = c[top_idx]

            Tc = stage_T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            lse_val = (max_Tc + np.log(exp_Tc.sum())) / stage_T
            w = exp_Tc / exp_Tc.sum()

            # Gradient
            grad_a = np.zeros(n)
            for idx_i in range(k_actual):
                K = top_idx[idx_i]
                wK = w[idx_i]
                js = max(0, K - (n - 1))
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize_pos(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Linear ramp
        a = x
    elif choice == 1:
        # Quadratic ramp
        a = x ** 2
    elif choice == 2:
        # Inverse ramp
        a = 1 - x
    elif choice == 3:
        # Gaussian centered
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif choice == 4:
        # Gaussian left
        a = np.exp(-6 * (x - 0.3) ** 2) + 0.3 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 5:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 6:
        # Sinc-like
        t = (x - 0.5) * n * 0.1
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Oscillatory taper
        a = (1 - x) * (0.7 + 0.3 * np.cos(4 * np.pi * x))
    elif choice == 8:
        #.Power ramp
        a = x ** 0.7
    else:
        # Random
        a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence for multi-start."""
    a = normalize_pos(a0)
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization with novel improvements over previous approach:
    1. Multi-initialization with structured sequences
    2. LP Nesterov with EMA + tapered constraints
    3. Peak-sensitivity local search
    4. Multi-length exploration
    5. LSE gradient polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 100.0, 1700.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    # Phase 1: Multi-initialization - find the best starting point
    print(f"\n=== Phase 1: Multi-init ({time_budget:.0f}s budget) ===")
    inits = []
    if parent is not None:
        inits.append(("parent", parent))
    for i in range(6):
        init = structured_init(n_base, rng)
        inits.append((f"struct_{i}", init))
    # Add perturbed variants of best
    if parent is not None:
        for i in range(3):
            pert = perturbed_init(parent, rng, scale=0.03 * (i + 1))
            inits.append((f"pert_{i}", pert))

    for name, init in inits:
        sc = 2.0 * n_base * fast_autocorr(init).max()
        if sc < best_score:
            best_score = sc
            best_seq = init.copy()
    print(f"  Best init: {best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 2: Main LP Nesterov with EMA at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: LP Nesterov EMA n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 60:
        a_out, _ = nesterov_lp_em
a(best_seq, t2, solver)
        try_update(a_out, f"lp_ema_n{n_base}")

    # Phase 3: Peak sensitivity search
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: Peak sensitivity n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, _ = peak_sensitivity_search(best_seq, t3, seed=123)
        try_update(a_out, f"peak_sens_n{n_base}")

    # Phase 4: LP Nesterov at longer length
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: LP Nesterov EMA n={n_long} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_em
a(init_long, t4, solver)
        try_update(a_out, f"lp_ema_n{n_long}")

    # Phase 5: LSE gradient at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: LSE grad n={n_base} ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.12, 120.0)
    if t5 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t5, T_start=40.0, T_end=0.3, n_stages=15, topk=10)
        try_update(a_out, f"lse_n{n_base}")

    # Phase 6: Peak sensitivity at longer length
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 6: Peak sensitivity n={n_long} ({remaining:.0f}s left) ===")
        t6 = min(remaining * 0.12, 100.0)
        if t6 > 15:
            a_out, _ = peak_sensitivity_search(best_seq, t6, seed=456)
            try_update(a_out, f"peak_sens_n{n_long}")

    # Phase 7: Multi-start with diverse inits + quick LP
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        print(f"=== Phase 7: Multi-start ({remaining:.0f}s left) ===")
        t7_total = min(remaining * 0.15, 150.0)
        n_starts = 3
        t7_each = t7_total / n_starts
        for si in range(n_starts):
            if time.monotonic() - t_global_start > time_budget - 80:
                break
            pert = perturbed_init(best_seq, rng, scale=0.04 * (si + 1))
            a_out, _ = nesterov_lp_em
a(pert, t7_each, solver)
            try_update(a_out, f"multi_start_{si}")

    # Phase 8: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8 = min(remaining * 0.15, 80.0)
        if t8 > 10:
            print(f"=== Phase 8: Final LSE polish ({remaining:.0f}s left) ===")
            a_out, _ = lse_gradient_descent(best_seq, t8, T_start=5.0, T_end=0.1, n_stages=10, topk=8)
            try_update(a_out, "final_lse")

    # Phase 9: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t9 = min(remaining * 0.3, 50.0)
        if t9 > 5:
            print(f"=== Phase 9: Final peak shave ({remaining:.0f}s left) ===")
            a_out, _ = peak_sensitivity_search(best_seq, t9, seed=789)
            try_update(a_out, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
