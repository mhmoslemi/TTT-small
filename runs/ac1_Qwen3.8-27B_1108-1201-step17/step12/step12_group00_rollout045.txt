import numpy as np

def fast_autocorr(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_corr(w, a):
    """Compute cross-correlation c[i] = sum_j w[j] * a[j-i] using FFT.
    w has length 2n-1, a has length n.
    Result c has length n (c[i] for i=0,...,n-1)."""
    n = len(a)
    size = 3 * n  # sufficient to avoid circular aliasing
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(np.asarray(w, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    # Cross-correlation: ifft(fft(w) * conj(fft(a)))
    C = irfft(W * np.conj(A), fft_size)
    # c[i] = sum_j w[j] * a[j+i]... 
    # Actually, ifft(fft(w) * conj(fft(a)))[k] = sum_j w[j+k] * conj(a[j]) (circular)
    # With zero-padding, this gives linear cross-correlation.
    # We want c[i] = sum_j w[j] * a[j-i] for i=0,...,n-1
    # = sum_j w[j+i] * a[j] (substituting j' = j+i... no)
    
    # Let me think again. The standard cross-correlation is:
    # (w ⋆ a)[k] = sum_j w[j+k] * a[j]  (this is ifft(fft(w)*conj(fft(a))) with appropriate indexing)
    
    # Actually, ifft(fft(w) * conj(fft(a)))[k] = sum_j w[(j+k) mod N] * a[j]
    # With sufficient zero-padding, this equals sum_j w[j+k] * a[j] (linear).
    
    # We want c[i] = sum_j w[j+i] * a[j] for i = 0,...,n-1
    # This is exactly (w ⋆ a)[i] = ifft(fft(w)*conj(fft(a)))[i]
    
    return C[:n]

def lse_gradient(a, T):
    """Compute the gradient of the LSE objective (1/T)*log(sum_k exp(T*b_k)).
    Returns (gradient, b, w) where b is autocorrelation, w is softmax weights."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) (with bounds)
    # = 2 * sum_k w[k] * a[k-i] (by symmetry, when both terms are valid)
    # More precisely: grad[i] = sum_k w[k] * a_{k-i} * 1_{0<=k-i<n} + sum_k w[k] * a_{i-k} * 1_{0<=i-k<n}
    # By the symmetry of the problem (w is symmetric around k=n-1), this equals:
    # 2 * sum_k w[k] * a_{k-i} * 1_{0<=k-i<n} ... no, that's not right either.
    
    # Let me use the cross-correlation approach.
    # grad[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n else 0) + sum_k w[k] * (a[i-k] if 0<=i-k<n else 0)
    # The first sum: sum_k w[k] * a[k-i] where k-i is in [0,n-1], i.e., k in [i, i+n-1]
    # This is a "sliding window" sum, which is the cross-correlation of w and a.
    # Specifically, c1[i] = sum_{k=i}^{i+n-1} w[k] * a[k-i] = sum_{j=0}^{n-1} w[j+i] * a[j]
    # This is the cross-correlation (w ⋆ a)[i] computed above.

    # This is sum_{k=max(0,i-n+1)}^{i} w[k] * a[i-k] = sum_{j=0}^{i} w[i-j] * a[j] (for i < n-1)
    # = sum_{j=0}^{min(i,n-1)} w[i-j] * a[j]
    # By symmetry of w (w[k] = w[2(n-1)-k]), this equals the first sum.
    # So grad[i] = 2 * c1[i] (when n-1 is the center of symmetry, which it is for autocorrelation)
    
    # Wait, is w symmetric? w[k] = exp(T*b[k]) / sum. And b is the autocorrelation, which IS symmetric: b[k] = b[2(n-1)-k]. So yes, w is symmetric.

    c1 = fast_cross_corr(w, a)
    grad = 2.0 * c1

Now, for the Nesterov acceleration:

Standard Nesterov (FISTA) for minimizing f(x) over a convex set C:
- y_0 = x_0
- x_1 = proj_C(y_0 - (1/L) * ∇f(y_0))
- y_{k+1} = x_{k+1} + (t_k - 1)/(t_{k+1}) * (x_{k+1} - x_k)
- where t_{k+1} = (1 + sqrt(1 + 4*t_k^2))/2, t_0 = 1
