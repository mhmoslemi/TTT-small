import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Compute full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient(a, w, n):
    """Compute gradient of sum_k w_k * b_k w.r.t. a, where b = autocorr(a)."""
    # g_i = 2 * sum_m w[i+m] * a[m]
    # = 2 * (w_rev * a)[2n-2-i]
    w_rev = w[::-1]  # w_rev[k] = w[2n-2-k]
    # Convolve w_rev (length 2n-1) with a (length n)
    conv_len = (2*n-1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(W * A, fft_size)
    # g_i = 2 * conv[2n-2-i]
    g = 2.0 * conv[2*n-2 : 2*n-2-n : -1]  # g[i] = 2*conv[2n-2-i] for i=0..n-1
    return g

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    return e / s

    Adam optimization in log-space with adaptive LogSumExp smoothing.

    if s < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based Adam optimization in log-space with LogSumExp smoothing and β-annealing
2. Fast local search using first-order convolution updates
3. Multi-length exploration with interpolation
4. Coordinate descent polish
Key innovation: efficient O(n log n) per-iteration gradient via FFT,
combined with adaptive temperature and restart strategies.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full self-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_grad_a(a, w, n):
    """
    Compute gradient of sum_k w_k * b_k w.r.t. a, where b = autocorr(a).
    g_i = 2 * sum_m w[i+m] * a[m] (dominant term, ignoring self-correction)
    """
    w_rev = w[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(W * A, fft_size)
    # g_i = 2 * conv[2n-2-i] for i = 0, ..., n-1
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * conv[idx]
    return g

def adam_logspace_optimize(a0, time_limit, beta_schedule, lr_base=0.005, momentum_beta1=0.85, momentum_beta2=0.999, eps=1e-8, check_interval=50):
    """
    Adam optimization in log-space (a = softmax(z)) with LogSumExp smoothing.
    beta_schedule: list of β values to anneal through.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(a_init + 1e-30)
    z = z - z.mean()  # Center for stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    t0 = time.monotonic()
    
    n_stages = len(beta_schedule)
    time_per_stage = time_limit / n_stages
    
    for si, beta in enumerate(beta_schedule):
        stage_t0 = time.monotonic()
        
        while time.monotonic() - t0 < time_limit:
            if time.monotonic() - stage_t0 > time_per_stage * 2.0:
                break
            
            a = softmax_fn(z)
            b = fast_autocorr(a)
            
            # LogSumExp weights
            max_b = b.max()
            w = np.exp(beta * (b - max_b))
            w_sum = w.sum()
            w = w / w_sum
            
            # Gradient in a-space
            g_a = compute_grad_a(a, w, n)
            
            # Gradient in z-space
            S = np.dot(g_a, a)
            g_z = a * (g_a - S)
            
            # Adam update
            t += 1
            lr_t = lr_base * np.sqrt(1 - momentum_beta2 ** t) / (1 - momentum_beta1 ** t)
            
            m = momentum_beta1 * m + (1 - momentum_beta1) * g_z
            v = momentum_beta2 * v + (1 - momentum_beta2) * g_z ** 2
            m_hat = m / (1 - momentum_beta1 ** t)
            v_hat = v / (1 - momentum_beta2 ** t)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            # Check score periodically
            if t % check_interval == 0:
                score = 2.0 * n * b.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    Adam-LS stage={si}/{n_stages}, beta={beta:.1f}, t={elapsed:.0f}s: best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_out, _ = adam_logspace_optimize(best_seq, t3, beta_sched_3, lr_base=0.002)
        try_update(a_out, "adam3")
    
    # === Phase 4: Local search again ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Local Search 2 (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = local_search_pair_swap(best_seq, t4, seed=99)
        try_update(a_out, "local2")
    
    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length (remaining={remaining:.0f}s) ===")
    
    n_best = len(best_seq)
    candidate_lengths = []
    for offset in [-100, -50, 50, 100, 200, 300]:
        nl = n_best + offset
        if 800 <= nl <= 4000:
            candidate_lengths.append(nl)
    
    # Deduplicate
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))
    
    # Allocate time
    n_lengths = len(candidate_lengths)
    time_per_len = max(30, (remaining * 0.6) / max(1, n_lengths))
    
    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        
        tl = min(time_per_len, remaining * 0.3)
        if tl < 30:
            continue
        
        print(f"  Exploring n={nl} (time={tl:.0f}s)")
        init = interpolate_sequence(best_seq, nl)
        
        # Quick Adam
        beta_sched = np.geomspace(100, 5, 10)
        a_out, _ = adam_logspace_optimize(init, tl * 0.6, beta_sched, lr_base=0.003, check_interval=100)
        try_update(a_out, f"adam_n{nl}")
        
        # Quick local search
        remaining_sub = time_budget - (time.monotonic() - t_global)
        tl2 = min(tl * 0.3, remaining_sub * 0.1)
        if tl2 > 15:
            a_out2, _ = local_search_pair_swap(a_out, tl2, seed=nl % 1000)
            try_update(a_out2, f"local_n{nl}")
    
    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish (remaining={remaining:.0f}s) ===")
    
    if remaining > 30:
        t6a = min(remaining * 0.4, 150.0)
        a_out, _ = local_search_pair_swap(best_seq, t6a, seed=777)
        try_update(a_out, "ps_final")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t6b = min(remaining * 0.4, 100.0)
        beta_sched = np.geomspace(50, 1, 10)
        a_out, _ = adam_logspace_optimize(best_seq, t6b, beta_sched, lr_base=0.001)
