            # For k where 0 <= k-j < n: sens[k] = 2*a[k-j] (or a[k-j] if k=2j)
            # Actually let me re-derive:
            # conv[k] = sum_i a[i]*a[k-i]
            # Change a[j] from v to v+delta:
            #   Delta conv[k] = delta * (a[k-j] + a[j-k]) + delta^2 * I[k==2j]
            #   Wait: a[k-j] appears when i=j (giving a[j]*a[k-j] = (v+delta)*a[k-j])
            #         and a[j-k] appears when k-i=j (giving a[k-j]*a[j]... wait)
            
            # Let me be very precise:
            # conv[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (where a[index] = 0 if index not in [0,n-1])
            # 
            # Terms involving a[j]:
            # 1. When i = j: a[j] * a[k-j] (if 0 <= k-j <= n-1)
            # 2. When k-i = j, i.e., i = k-j: a[k-j] * a[j] (if 0 <= k-j <= n-1)
            #
            # If k = 2j: both conditions give i = j, so it's one term: a[j]*a[j] = a[j]^2
            # If k ≠ 2j: two terms (or one if k-j is out of bounds)
            #
            # So: contrib_k = a[j]*a[k-j] if (0<=k-j<n) [one term if k=2j, else two terms = 2*a[j]*a[k-j]]
            
            # Change in conv[k] when a[j] goes from v to v+delta:
            # If k = 2j: (v+delta)^2 - v^2 = 2*v*delta + delta^2
            # If 0 <= k-j < n and k ≠ 2j: (v+delta)*a[k-j] + a[k-j]*(v+delta) - v*a[k-j] - a[k-j]*v
            #   = 2*delta*a[k-j]
            # Else: 0
            
            # So: new_conv[k] = conv[k] + delta * c_k + delta^2 * I[k==2j]
            # where c_k = 2*a[k-j] if (0 <= k-j < n and k ≠ 2j)
            #       c_k = 2*v if k = 2j
            #       c_k = 0 otherwise
            
            size = 2 * n - 1
            c = np.zeros(size)
            # a[k-j] is valid when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
            lo = max(0, j)
            hi = min(size, j + n)
            
            if 2 * j < size:
                # k = 2j gets special treatment
                c[2*j] = 2.0 * v
                # Fill the rest
                if lo < 2*j:
                    c[lo:2*j] = 2.0 * a[lo - j : 2*j - j]  # a[k-j] for k in [lo, 2j-1]
                if 2*j + 1 < hi:
                    c[2*j+1:hi] = 2.0 * a[2*j+1-j : hi-j]  # a[k-j] for k in [2j+1, hi-1]
            else:
                c[lo:hi] = 2.0 * a[lo - j : hi - j]
            
            # Now find optimal delta via golden section
            # f(delta) = 2*n * max_k (conv[k] + delta*c_k + delta^2*I[k==2j]) / (total_sum + delta)^2
            # Search range: delta in [-v, S_other] (so a[j] stays non-negative and doesn't exceed total sum)
            
            delta_lo = -v  # a[j] = 0
            delta_hi = S_other * 0.5  # reasonable upper bound
            
            if delta_hi <= delta_lo:
                delta_hi = delta_lo + max(v * 0.1, 1e-10)
            
            # Define the objective function for this coordinate
            has_quad = (2 * j < size)
            
            def eval_delta(delta):
                tmp = conv + delta * c
                if has_quad:
                    tmp = tmp.copy()  # Need to avoid modifying conv
                    tmp[2*j] += delta * delta
                # Actually, this copies the whole array each time, which is expensive
                # Let me use a pre-allocated buffer
                return tmp.max()
            
            # More efficient: use a buffer
            buf = conv + delta_hi * c  # Pre-compute at upper bound
            # Actually, let me just do it inline
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
            
            # First, do a coarse search
            n_coarse = 10
            deltas_coarse = np.linspace(delta_lo, delta_hi, n_coarse)
            best_coarse_delta = delta_lo
            best_coarse_val = np.inf
            
            base_arr = conv.copy()  # We'll modify this
            
            for dc in deltas_coarse:
                base_arr[:] = conv
                base_arr += dc * c
                if has_quad:
                    base_arr[2*j] += dc * dc
                val = base_arr.max() / (total_sum + dc) ** 2
                if val < best_coarse_val:
                    best_coarse_val = val
                    best_coarse_delta = dc
            
            # Narrow to bracket around best
            idx_best = np.argmin(deltas_coarse - best_coarse_delta)  # just in case
            # Actually find the two nearest points
            pos = int(np.searchsorted(deltas_coarse, best_coarse_delta))
            if pos == 0:
                d_lo = deltas_coarse[0]
                d_hi = deltas_coarse[1]
            elif pos >= n_coarse - 1:
                d_lo = deltas_coarse[-2]
                d_hi = deltas_coarse[-1]
            else:
                d_lo = deltas_coarse[pos - 1]
                d_hi = deltas_coarse[pos + 1]
            
            # Golden section in [d_lo, d_hi]
            for _ in range(30):
                if d_hi - d_lo < 1e-12 * max(1.0, abs(v)):
                    break
                d1 = d_hi - gr * (d_hi - d_lo)
                d2 = d_lo + gr * (d_hi - d_lo)
                
                # Evaluate at d1
                base_arr[:] = conv
                base_arr += d1 * c
                if has_quad:
                    base_arr[2*j] += d1 * d1
                f1 = base_arr.max() / (total_sum + d1) ** 2
                
                # Evaluate at d2
                base_arr[:] = conv
                base_arr += d2 * c
                if has_quad:
                    base_arr[2*j] += d2 * d2
                f2 = base_arr.max() / (total_sum + d2) ** 2
                
                if f1 < f2:
                    d_hi = d2
                else:
                    d_lo = d1
            
            delta_opt = (d_lo + d_hi) / 2.0
            
            # Evaluate the new score
            base_arr[:] = conv
            base_arr += delta_opt * c
            if has_quad:
                base_arr[2*j] += delta_opt * delta_opt
            new_max_b = base_arr.max()
            new_sum = total_sum + delta_opt
            new_score = 2.0 * n * new_max_b / (new_sum ** 2)
            
            # Annealing acceptance
            if anneal and new_score > best_score:
                delta_accept = -np.log(rng.random() + 1e-300) * T
                if new_score - best_score < delta_accept:
                    # Accept
                    pass
                else:
                    continue  # Reject
            
            if new_score <= best_score or (anneal and new_score - best_score < -np.log(rng.random() + 1e-300) * T):
                # Update
                a[j] = v + delta_opt
                total_sum = new_sum
                # Update conv in place
                conv += delta_opt * c
                if has_quad:
                    conv[2*j] += delta_opt * delta_opt
                sweep_improved = True
                
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
        
        if sweep_improved and best_score < score_normalized(best_a) - 1e-15:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        if anneal:
            T *= 0.995
        
        if (sweep + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep+1} ({elapsed:.0f}s): score={best_score:.8f} T={T:.6f}")
    
    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20):
    from scipy.optimize import minimize
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        iters = 40 if si < n_stages // 2 else 20
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': iters * 300,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass
    
    return best_a


def peak_targeted_refine(a0, time_limit, max_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.05
    last_improve_step = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 5000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.92
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)
        
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa
        
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.002, 0.15)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
