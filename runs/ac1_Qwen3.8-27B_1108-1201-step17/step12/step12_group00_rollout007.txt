    return irfft(A * A, fft_size)[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Frank-Wolfe with FOSTER-style adaptive steps and constraint reduction
2. Log-sum-exp FISTA for smooth refinement with temperature annealing
3. Exact gradient coordinate descent for final polish
4. Multi-length expansion with interpolation and refinement
Key innovations: active-set LP reduction, dual-guided constraint selection,
and exact gradient structure exploitation (grad[i] = 2*a[k*-i]).

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full linear autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    B = rfft(b.astype(np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def _find_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def lp_best_response(a, M, top_k_constraints, solver, n_max_constraints=400):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for k in top_k, b >= 0.
    Returns b or None.
    """
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    idx = top_k_constraints[:n_max_constraints]
    T_red = np.zeros((len(idx), n))
    for ii, k in enumerate(idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            T_red[ii, js:je + 1] = a[je - js::-1] if je >= js else 0

    constraints = [T_red @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def get_active_constraints(a, k=200):
    """Get indices of top-k peaks in autocorrelation of a."""
    c = fast_autocorr(a)
    n = len(a)
    return np.argsort(c)[::-1][:k]


def check_violations(a, b, M, current_constraints):
    """Check which full convolution constraints are violated."""
    conv_ab = fast_cross_corr(a, b)
    size = 2 * len(a) - 1
    violations = np.where(conv_ab[:size] > M + 1e-6)[0]
    return list(violations)

    gamma_alpha = 0.1
    gamma_rho = 0.8

    active_cons = get_active_constraints(a, k=300)

    while time.monotonic() - t0 < time_limit:
        c_y = fast_autocorr(y)
        M_curr = c_y.max()
        if M_curr < 1e-14:
            break

        # SolLP with reduced constraints
        b = lp_best_response(y, M_curr, active_cons, solver, n_max_constraints=400)
        if b is None:

            b = lp_best_response(y, M_curr, active_cons, solver, n_max_constraints=600)
            if b is None:
                t_iter += 1
                continue

        # Check for violations and update active set
        violations = check_violations(y, b, M_curr, active_cons)
        if len(violations) > 0:
            active_cons = sorted(set(list(active_cons[:300]) + list(violations)))
            active_cons = np.array(sorted(active_cons))[:600]
            b2 = lp_best_response(y, M_curr, active_cons, solver, n_max_constraints=600)
            if b2 is not None and b2.sum() > b.sum() * 0.5:
                b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            t_iter += 1
            continue
        g = b * norm / b_sum
        n_iters += 1

        # FOSTER-style step size
        y_new = (1.0 - alpha_init) * y + alpha_init * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            s_y = norm
        else:
            y_new = y_new * norm / s_y

        # Adaptive step: accept if improved, otherwise try smaller step
        if score_y_new > score_y_old * (1 + 1e-6):

            y_new_half = (1.0 - alpha_init / 2) * y + (alpha_init / 2) * g

            if s_yh > 1e-15:
                y_new_half = y_new_half * norm / s_yh
                score_half = fast_autocorr(y_new_half).max()
                if score_half < score_y_old:
                    y_new = y_new_half
                    score_y_new = score_half
                    alpha_init = max(0.01, alpha_init * 0.5)
                else:
                    alpha_init = max(0.005, alpha_init * 0.5)
                    y_new = y.copy()
                    score_y_new = score_y_old
            else:
                y_new = y.copy()
                score_y_new = score_y_old
        else:
            # Successful step, can increase alpha slightly
            alpha_init = min(1.0, alpha_init * 1.1)

        # Nesterov momentum for iterate a
        if t_iter == 1:
            a_new = g.copy()
        else:
            momentum = (t_iter - 1.0) / (t_iter + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        t_iter += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y_new < best_score:
            best_score = score_y_new
            best_a = y_new.copy()

        # Periodic averaging
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                s_avg_score = fast_autocorr(avg_check).max()
                if s_avg_score < best_score:
                    best_score = s_avg_score
                    best_a = avg_check.copy()

        # Periodically refresh active constraints
        if n_iters % 50 == 0:
            active_cons = get_active_constraints(a, k=300)

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def proximal_operator(z, target_sum):
    """Project z onto {a >= 0, sum(a) = target_sum}."""
    a = np.maximum(z, 0.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * target_sum / n
    # Find tau such that sum(max(z_i - tau, 0)) = target_sum
    z_pos = z[z > 0]
    if len(z_pos) == 0:
        return np.ones(len(z)) * target_sum / len(z)
    z_sorted = np.sort(z_pos)[::-1]
    cumsum = np.cumsum(z_sorted)
    n_pos = len(z_sorted)
    # tau = (cumsum[k] - target_sum) / (k+1) for the right k
    taus = (cumsum - target_sum) / np.arange(1, n_pos + 1)
    # Find k where z_sorted[k] >= tau_k >= z_sorted[k+1]
    valid = (taus <= z_sorted) & (taus >= np.concatenate([[0], z_sorted[:-1]]))
    if np.any(valid):
        k = np.where(valid)[0][0]
        tau = taus[k]
    else:
        tau = (s - target_sum) / len(z_pos)
    a = np.maximum(z - tau, 0.0)
    s_final = a.sum()
    if s_final < 1e-15:
        return np.ones(len(z)) * target_sum / len(z)
    return a * target_sum / s_final

def coord_descent_polish(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent using exact gradient structure."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        k_star = int(np.argmax(c))
        M_curr = c[k_star]

        # Gradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[k_star - j]

        # Find the coordinate with largest negative direction potential
        # We want to decrease a[i] where grad[i] > 0 and increase where useful
        # Actually, to decrease max(conv), we want to move a in direction -grad
        # But we must maintain a >= 0 and sum(a) = norm
        # Best coordinate: the one where reducing a[i] most decreases the peak
        # Reducing a[i] by delta changes c[k] by -2*delta*a[k-i] for the peak k
        # So we want to reduce a[i] where a[k*-i] is large (i.e., grad[i] is large)
        
        # Strategy: pick i with largest grad[i] (most contributes to peak), reduce it
        # and add to a position that contributes least to the peak
        i_reduce = int(np.argmax(grad))
        
        # Find position to add to: minimize contribution to peak
        # Adding to position j changes c[k*] by 2*delta*a[k*-j]
        # We want a[k*-j] to be small
        contrib_to_peak = np.zeros(n)
        for j in range(js, je + 1):
            contrib_to_peak[j] = a[k_star - j]
        # Positions outside [js, je] don't contribute to the peak at all
        # Prefer those
        outside = np.where((np.arange(n) < js) | (np.arange(n) > je))[0]
        if len(outside) > 0:
            # Among outside positions, pick one with low current value
            i_add = outside[np.argmin(a[outside])]
        else:
            i_add = int(np.argmin(contrib_to_peak))

        if i_reduce == i_add or a[i_reduce] < 1e-10:
            no_improve += 1
            if no_improve > 500:
                break
            continue

        # Try different fractions
        improved = False
        amts = [a[i_reduce] * f for f in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]]
        amts.sort(reverse=True)
        for amt in amts:
            if amt <= 0 or amt > a[i_reduce]:
                continue
            a_trial = a.copy()
            a_trial[i_reduce] -= amt
            a_trial[i_add] += amt
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            # Try random pair
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            if i1 != i2 and a[i1] > 1e-10:
                amt = a[i1] * rng.uniform(0.02, 0.15)
                a_trial = a.copy()
                a_trial[i1] -= amt
                a_trial[i2] += amt
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
            else:
                no_improve += 1

        if no_improve > 1000:
            break

        if verbose and no_improve % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): best={best_score:.8f}, stall={no_improve}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(10.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: finds best sequence minimizing the evaluation."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    solver = _find_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LP-FW at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lpfw1_n{n_base}")

    # Phase 2: LSE-FISTA refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LSE-FISTA n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_fista(a_norm, t2, T_start=40.0, T_end=0.5, n_stages=8, verbose=True)
        do_try_update(a_out, f"lsefista_n{n_cur}")

    # Phase 3: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: CoordDesc n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t3, seed=123, verbose=True)
        do_try_update(a_out, f"coord_n{n_cur}")

    # Phase 4: Try longer sequence
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP-FW n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 60:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lpfw2_n{n2}")

    # Phase 5: LSE-FISTA on longer
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LSE-FISTA n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 120.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_fista(a_norm, t5, T_start=10.0, T_end=0.2, n_stages=6, verbose=True)
        do_try_update(a_out, f"lse2_n{n_cur}")

    # Phase 6: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final CoordDesc n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 100.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t6, seed=777, verbose=True)
        do_try_update(a_out, f"coord2_n{n_cur}")

    # Phase 7: Second LP-FW round on best found
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 7: LP-FW round 2 n={n_cur} (budget={remaining:.0f}s) ===")
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 30:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
            do_try_update(a_out, f"lpfw3_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
