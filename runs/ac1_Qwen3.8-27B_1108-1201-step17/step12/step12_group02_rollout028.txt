    A = rfft(a, fft_size)

"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. Full-softmax subgradient mirror descent (O(n log n) per iter via FFT)
2. LSE-smoothed L-BFGS with multi-stage temperature annealing
3. Multi-length exploration with interpolation from parent
4. Adaptive Polyak-like step size with stall-based restarts
5. Coordinate descent peak shaving for final polish
Key innovations: full-subgradient (all peaks weighted by softmax),
efficient FFT-based gradient computation, adaptive multi-length search.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

# Global variables for LSE objective (to avoid closures)
_LSE_T = 1.0
_LSE_N = 1


def fast_autocorr(a):
    """Full linear autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def cross_corr_w(w, a):
    """result[j] = sum_i w[i+j]*a[i] for j=0..n-1. O(n log n) via FFT."""
    n = len(a)
    nw = len(w)
    a_rev = a[::-1].copy()
    conv_len = n + nw - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def conv_prefix(a, w):
    """(a*w)[j] for j=0..n-1. O(n log n) via FFT."""
    n = len(a)
    nw = len(w)
    conv_len = n + nw - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)
    return conv[:n]


def full_subgradient(a, w):
    """Subgradient of sum_k w[k]*b[k] w.r.t. a. O(n log n).
    grad[j] = sum_k w[k]*(a[k-j]+a[j-k]) for valid indices."""
    n = len(a)
    term1 = cross_corr_w(w, a)
    term2 = conv_prefix(a, w)
    return term1 + term2


def softmax_weights(b, T):
    """Compute softmax(T*b) safely."""
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    s = exp_Tb.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return exp_Tb / s


def subgrad_mirror_descent(a0, time_limit, T_init=80.0, T_end=5.0,
                           eta0=0.2, min_eta=0.001, decay=0.997,
                           seed=42, check_every=5, verbose=True):
    """Full-softmax subgradient mirror descent with adaptive step size."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    eta = eta0

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        cur_score = 2.0 * n * b_max

        # Anneal temperature
        ramp = min(t, 10000) / 10000.0
        T_cur = T_init * (T_end / T_init) ** ramp

        w = softmax_weights(b, T_cur)
        grad = full_subgradient(a, w)

        # Adaptive step size
        gap = max(cur_score - 1.0, 0.01)
        grad_sq_norm = np.dot(grad * grad, a)
        if grad_sq_norm < 1e-30:
            grad_sq_norm = 1e-30
        eta_polyak = gap / (4.0 * n * grad_sq_norm)
        eta = max(eta * decay, min(eta_polyak * 0.5, 0.8))
        eta = max(eta, min_eta)

        # Mirror descent update (exponential tilting)
        log_mult = -eta * grad
        log_mult = np.clip(log_mult, -30, 30)
        a_new = a * np.exp(log_mult)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a = a_new / max(s_new, 1e-15)

        if t % check_every == 0:
            sc = 2.0 * n * fast_autocorr(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                stall = 0
                eta = min(eta * 1.02, 0.8)
            else:
                stall += 1

        # Stall handling
        if stall > 300:
            scale = 0.03 * min(1.0, stall / 50.0)
            a = a + scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            eta *= 0.6
            stall = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): score={best_score:.6f}, "
                  f"bmax={b_max:.5e}, T={T_cur:.1f}, eta={eta:.4f}")

    return best_a, best_score


def _lse_func(z):
    """LSE objective for scipy minimize (uses global _LSE_T, _LSE_N)."""
    T = _LSE_T
    n = _LSE_N
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    return (Tb.max() + np.log(exp_Tb.sum() + 1e-300)) / T


def _lse_jac(z):
    """LSE gradient for scipy minimize (uses global _LSE_T, _LSE_N)."""
    T = _LSE_T
    n = _LSE_N
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    b = fast_autocorr(a)
    w = softmax_weights(b, T)
    grad_a = full_subgradient(a, w)
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lse_lbfgs(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=16,
              seed=42, verbose=True):
    """Multi-stage LSE-smoothed L-BFGS optimization."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)
    global _LSE_T, _LSE_N
    _LSE_N = n

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        _LSE_T = stage_T
        iters = max(10, min(120, int(15 + 70 * (si + 1) / n_stages)))

        try:
            res = minimize(fun=_lse_func, x0=z, jac=_lse_jac,
                           method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-7, 'maxfun': iters * 100,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if verbose and (si % 4 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE {si+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): "
                  f"best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123, verbose=True):
    """Coordinate descent: reduce top peaks by redistributing mass."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a) * (n * a.sum())  # Not quite right, let me use fast_score directly
    best_score = 2.0 * n * (a * a).max() if False else None
    # Just use normalized score
    best_bmax = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(15, 2 * n - 1)]

        improved = False
        for k_idx in top_k[:8]:
            k_idx = int(k_idx)
            if k_idx >= n:
                k_idx = 2 * n - 2 - k_idx

            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            if contrib.sum() < 1e-18:
                continue

            src = int(idx[np.argmax(contrib)])

            # Find destination: low a, not in peak area
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]
            lowest = candidates[np.argsort(a[candidates])[:max(3, len(candidates) // 5)]]
            if len(lowest) == 0:
                continue
            dst = int(lowest[rng.integers(len(lowest))])

            for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.12]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                b_trial = fast_autocorr(a_trial)
                if b_trial.max() < best_bmax - 1e-14:
                    a = a_trial
                    best_bmax = b_trial.max()
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        b_trial = fast_autocorr(a_trial)
                        if b_trial.max() < best_bmax:
                            a = a_trial
                            best_bmax = b_trial.max()
                            best_a = a.copy()
                            stall = 0

            if stall > 600:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        sc = 2.0 * n * best_bmax
        print(f"    PeakShave ({elapsed:.0f}s): score={sc:.6f}")

    return best_a, 2.0 * n * best_bmax


def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=0):

    lengths = [n_parent]
    for mult in [1.3, 1.6, 2.0, 2.5]:
        nl = int(round(n_parent * mult))
        if nl > 200 and nl < 12000 and nl not in lengths:
            lengths.append(nl)
    # Sort by length
    lengths.sort()
    print(f"\nLengths to explore: {lengths}")

    # Phase 1: Quick MD at each length (interpolated from parent)
    print(f"\n=== Phase 1: Quick MD at multiple lengths ===")
    phase1_time = min(time_budget * 0.12, 180.0)
    per_length = phase1_time / len(lengths)
    candidates = {}

    for nl in lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        t_alloc = min(per_length, remaining * 0.3)
        if t_alloc < 15:
            continue
        init = interpolate_to_length(best_seq, nl)
        a_out, _ = subgrad_mirror_descent(init, t_alloc, T_init=60.0, T_end=8.0,
                                           eta0=0.15, decay=0.998, seed=nl % 1000,
                                           check_every=10, verbose=False)
        candidates[nl] = a_out
        sc = fast_score(a_out)
        print(f"  Phase 1 n={nl}: score={sc:.6f}")
        try_update(a_out, f"p1_n{nl}")

    # Phase 2: Main MD at top 2-3 lengths
    print(f"\n=== Phase 2: Full MD optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    phase2_time = remaining * 0.38
    # Focus on the best 2-3 lengths
    scored = [(fast_score(candidates[nl]), nl) for nl in candidates]
    scored.sort()
    top_lengths = [nl for _, nl in scored[:3]]

    per_md = phase2_time / max(1, len(top_lengths))
    for nl in top_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        t_alloc = min(per_md, remaining * 0.4)
        if t_alloc < 20:
            continue
        print(f"  MD at n={nl} for {t_alloc:.0f}s...")
        a_out, _ = subgrad_mirror_descent(candidates[nl], t_alloc,
                                           T_init=80.0, T_end=3.0,
                                           eta0=0.2, decay=0.997,
                                           seed=nl % 2000 + 100,
                                           check_every=5, verbose=True)
        candidates[nl] = a_out
        try_update(a_out, f"md_n{nl}")

    # Also try diverse inits at the best length
    best_n = min(top_lengths, key=lambda nl: fast_score(candidates[nl]))
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        t_div = min(remaining * 0.08, 100.0)
        for v in [0, 4, 5]:
            if time_budget - (time.monotonic() - t_global) < 30:
                break
            init = diverse_init(best_n, rng, variant=v)
            a_out, _ = subgrad_mirror_descent(init, t_div / 3, T_init=50.0,
                                               T_end=10.0, eta0=0.25, decay=0.997,
                                               seed=v * 100 + 7, check_every=10,
                                               verbose=False)
            try_update(a_out, f"div_v{v}_n{best_n}")

    # Phase 3: LSE refinement
    print(f"\n=== Phase 3: LSE + L-BFGS refinement ===")
    remaining = time_budget - (time.monotonic() - t_global)
    phase3_time = remaining * 0.25
    for nl in top_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        t_alloc = min(phase3_time / 2, remaining * 0.4)
        if t_alloc < 20:
            continue
        print(f"  LSE at n={nl} for {t_alloc:.0f}s...")
        a_out, _ = lse_lbfgs(candidates[nl], t_alloc, T_start=150.0, T_end=0.5,
                              n_stages=16, seed=nl % 500 + 200, verbose=True)
        candidates[nl] = a_out
        try_update(a_out, f"lse_n{nl}")

    # Phase 4: Peak shaving
    print(f"\n=== Phase 4: Peak shaving ===")
    remaining = time_budget - (time.monotonic() - t_global)
    phase4_time = remaining * 0.15
    for nl in top_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        t_alloc = min(phase4_time / 2, remaining * 0.35)
        if t_alloc < 10:
            continue
        a_out, _ = peak_shaving(candidates[nl], t_alloc, seed=nl % 700 + 300,
                                 verbose=True)
        candidates[nl] = a_out
        try_update(a_out, f"ps_n{nl}")

    # Phase 5: Final polish - short MD + LSE on the best
    print(f"\n=== Phase 5: Final polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 30:
        # Short MD
        t_f1 = min(remaining * 0.25, 80.0)
        if t_f1 > 10:
            a_out, _ = subgrad_mirror_descent(best_seq, t_f1, T_init=30.0,
                                               T_end=2.0, eta0=0.05,
                                               decay=0.999, seed=999,
                                               check_every=3, verbose=False)
            try_update(a_out, "final_md")

        # Short LSE low temp
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.30, 70.0)
        if t_f2 > 10:
            a_out, _ = lse_lbfgs(best_seq, t_f2, T_start=30.0, T_end=0.1,
                                   n_stages=12, seed=777, verbose=False)
            try_update(a_out, "final_lse")

        # Final peak shave
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.40, 60.0)
        if t_f3 > 5:
            a_out, _ = peak_shaving(best_seq, t_f3, seed=555, verbose=False)
            try_update(a_out, "final_ps")

    # Final evaluation with official function
    if best_seq is not None:
        final_sc = evaluate_safe(best_seq)
        if np.isfinite(final_sc):
            best_score = final_sc
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
