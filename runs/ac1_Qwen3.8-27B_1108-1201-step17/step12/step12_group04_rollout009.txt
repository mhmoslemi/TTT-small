def fft_convolve(a, b):

    na = len(a)
    nb = len(b)
    size = na + nb - 1

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    return fft_convolve(a, a)

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

def fast_mirror_descent_br(y, b_init, T, n_steps=3, eta=1.0):

    Approximate best response: min_b max_k (y*b)_k
    Using n_steps of mirror descent with entropy.

    nb = len(b)
    
    for _ in range(n_steps):
        g = cross_conv_gradient(y, b, T)
        # Mirror descent: b *= exp(-eta * g), then normalize
        b = b * np.exp(-eta * (g - g.mean()))
        b = np.maximum(b, 1e-30)
        b = b / b.sum()
    
    return b

def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15,
                      lr_start=0.1, lr_decay=0.95):
    """
    LSE-smoothed optimization using Adam in softmax parameterization.
    """
    n = len(a0)
    a_init = normalize(a0.copy())
    z = np.log(np.maximum(a_init, 1e-30))
    # Center z for numerical stability
    z = z - z.max()
    
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = score_fn(a_init)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    lr = lr_start
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        beta1 = 0.8
        beta2 = 0.999
        eps = 1e-8
        t_adam = 0
        
        # How many iterations for this stage (time-based)
        stage_time = remaining / (n_stages - si) * 0.8
        max_stage_time = min(stage_time, remaining * 0.5)
        
        stage_start = time.monotonic()
        n_stage_iters = 0
        
        while time.monotonic() - stage_start < max_stage_time:
            if time.monotonic() - t0 > time_limit:
                break
            
            t_adam += 1
            n_stage_iters += 1
            
            # Compute a from z
            a = softmax_fn(z)
            
            # Compute LSE gradient w.r.t. a
            grad_a, w = lse_gradient(a, stage_T)
            
            # Transform to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic check
            if t_adam % 50 == 0:
                sc = score_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        # Decay learning rate
        lr *= lr_decay
        
        elapsed = time.monotonic() - t0
        print(f"    LSA stage={si}/{n_stages} T={stage_T:.3f} ({elapsed:.0f}s): best={best_score:.8f}, iters={n_stage_iters}")
    
    return best_a, best_score

def peak_shaving_descent(a0, time_limit, seed=42):
    """
    Targeted peak-shaving: identify top peaks and try to reduce them
    by moving mass from peak-contributing positions to low-impact positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0
    K = 5  # number of top peaks to consider
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]
        
        improved = False
        for k in top_k:
            # Positions contributing to peak k: i and k-i where 0<=i<n, 0<=k-i<n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Find the largest contributor
            top_contrib_idx = np.argsort(contrib)[::-1][:5]
            
            for tc in top_contrib_idx:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Find target: position that contributes least to all top peaks
                # Compute "pressure" at each position
                pressure = np.zeros(n)
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for j2 in range(js2, je2 + 1):
                        partner = k2 - j2
                        if 0 <= partner < n:
                            pressure[j2] += a[j2] * a[partner]
                
                # Target positions: low pressure, not src
                pressure[src] = np.inf
                # Get low-pressure candidates
                n_candidates = max(10, n // 10)
                low_pressure_idx = np.argsort(pressure)[:n_candidates]
                
                for dst in low_pressure_idx[:10]:
                    if dst == src:
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_fn(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbations
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_fn(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                        break
            
            if stall > 500:
                break
        else:
            stall = 0
        
        if (stall // 100) > 0 and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PSD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. (a_fixed * b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def nesterov_lp_optimize(a0, time_limit, solver, max_iters=5000):
    """
    Nesterov-accelerated LP best-response iteration (original approach).
    """
    n = len(a0)
    a = normalize(a0.copy())
    y = a.copy()
    a_prev = a.copy()
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star = lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = normalize(y_new)
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize(a_new)
        
        y = y_new
        a_prev = a
        a = a_new
        
        if t % 20 == 0:
            sc_a = score_fn(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            
            sc_y = score_fn(y)
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 30 and n_avg % 60 == 0:
            avg_check = avg_a / n_avg
            sc_avg = score_fn(avg_check)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    sc_final = score_fn(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
