def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1

    # So (T@b)[k] = sum_j T[k,j] * b[j] = sum_j a_fixed[k-j] * b[j] = (a_fixed*b)[k]

        # k-j is valid when 0 <= k-j < n, i.e., j <= k < j+n
        T[j:j+n, j] = a_fixed
    return T

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, use_acceleration=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    
    # Skip first few iterations to let things stabilize
    skip_prob = max(0, 1.0 - (time_limit - (time.monotonic() - t0)) / time_limit)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Every 50 iterations, skip one LP to save time (use current a as approximation)
        if it % 50 == 0 and it > 100:
            b_new = a.copy()  # Use current as approximate best response
        else:
            b_new, M_lp = solve_lp_best_response(y, solver)
            if b_new is None:
                continue

        if use_acceleration:
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
            alpha = 2.0 / (t_k + 1.0)
            momentum = (t_k - 1.0) / t_k
        else:
            alpha = 2.0 / (it + 3.0)
            momentum = 0.0

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        if use_acceleration:
            y_new = a_new + momentum * (a_new - a_old)
            y_new = np.maximum(y_new, 0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / sy
        else:
            y_new = a_new

        a = a_new
        y = y_new

        # Check score every 10 iterations
        if it % 10 == 0:

    """Peak-targeted subgradient with moving average of active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Track running average of peak locations
    peak_history = np.zeros(n)
    peak_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) > n_active:
            active = active[np.argsort(b[active])[-n_active:]]
        elif len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Update peak history (exponential moving average)
        peak_history[active] += 1.0
        peak_count += 1

        # Compute subgradient using active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Also use peak history to add a small regularization
        # This encourages spreading mass away from historically peak-contributing positions
        if step > 100:
            hist_normalized = peak_history / (peak_history.sum() + 1e-15)
            g = 0.9 * g + 0.1 * hist_normalized * 2.0 * n * max_b

        # Project to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.5)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Line search if needed
        if sc_new > cur_score * 1.05:
            for factor in [0.5, 0.25, 0.1]:
                z_try = z - eta * factor * grad_z
                a_try = softmax_from_z(z_try)
                if score_normalized(a_try) < cur_score:
                    z_new = z_try
                    a_new = a_try
                    break
            else:
                continue  # Skip this step

        z = z_new
        a = a_new

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        # Decay peak history
        if step % 1000 == 0:
            peak_history *= 0.9

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-2 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n)
    elif choice == 5:
        a = 0.5 + 0.25 * np.cos(3 * 2 * np.pi * x) + 0.15 * np.cos(7 * 2 * np.pi * x)
    elif choice == 6:
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    elif choice == 8:
        # Barker-like
        a = 0.5 + 0.5 * np.sign(np.sin(2 * np.pi * 5 * x))
    elif choice == 9:
        # Zadoff-Chu inspired
        a = 1.0 + 0.3 * np.cos(2 * np.pi * (np.arange(n) ** 2) / n)
    elif choice == 10:

        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(2 * np.pi * 5 * x) + 0.05 * np.cos(2 * np.pi * 11 * x)

        freqs = rng.integers(1, 20, size=5)

        a = np.ones(n)

    a = np.maximum(a, 0.1)  # Keep minimum to avoid zeros

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetric_constraint(a):
    """Enforce palindromic symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a[i]
        a_sym[n - 1 - i] = a[i]
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start

    for stage in range(n_stages):

        if remaining < 3:

        max_b = b.max()

            a_val = softmax_from_z(z_val)
            b_val = fast_convolve(a_val)
            shifted = b_val / T
            shifted = shifted - shifted.max()
            exp_val = np.exp(shifted)
            lse = T * (shifted.max() + np.log(exp_val.sum()))
            return (2.0 * n) * lse / (a_val.sum() ** 2)

        def jac_staged(z_val):
            a_val = softmax_from_z(z_val)
            b_val = fast_convolve(a_val)
            shifted = b_val / T
            shifted = shifted - shifted.max()
            exp_val = np.exp(shifted)

            for k in range(len(b_val)):
                if w[k] < 1e-18:
                    continue

                je = min(n - 1, k)

                grad_a[idx] += 2.0 * w[k] * a_val[k - idx]

            sum_a = a_val.sum()

            # d/dz: need chain rule through softmax
            # dLSE/dz = a * (2/T * w_conv - 2/T * LSE_grad_dot_a / sum_a)
            # where w_conv[k] gives the gradient of b_k w.r.t. a
            
            # Actually, let's compute gradient of the objective w.r.t. a first:
            # f(a) = 2n * LSE(b/T) / (sum a)^2
            # df/da_i = 2n * (dLSE/da_i * (sum a)^2 - LSE * 2 * (sum a)) / (sum a)^4
            # = 2n * (dLSE/da_i / (sum a)^2 - 2 * LSE / (sum a)^3)

            # db_k/da_i = a_{k-i} (when 0 <= k-i < n)
            # dLSE/da_i = (1/T) * sum_k w_k * a_{k-i}
            # This is what grad_a computes (with the factor 2 removed)
            
            # Let me redo: grad_a as computed is sum_k w_k * a_{k-i} (the 2 is from the objective)
            # Actually in my code, grad_a[idx] += 2.0 * w[k] * a_val[k - idx]
            # So grad_a = 2 * (gradient of LSE w.r.t. a)
            # The objective is 2n * LSE / sum^2
            # d(obj)/da = 2n * (grad_LSE / sum^2 - 2 * LSE / sum^3)
            #           = 2n/sum^2 * (grad_LSE - 2*LSE/sum)
            # where grad_LSE = grad_a / 2
            
            LSE_val = T * (shifted.max() + np.log(exp_val.sum()))
            grad_LSE = grad_a / 2.0  # Remove the factor of 2
            
            sum_a_sq = sum_a ** 2
            df_da = (2.0 * n / sum_a_sq) * (grad_LSE - 2.0 * LSE_val / sum_a)
            
            # Now d/da = d/dz * dz/da
            # z = log(a), so dz_i = da_i / a_i
            # df/dz_i = df/da_i * a_i
            grad_z = a_val * df_da

            return grad_z

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            T = max(T_min, T * 0.7)

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score

Key innovations: adaptive LP call frequency, peak history regularization,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, lp_frequency=1):
    """Nesterov-accelerated LP best-response with adaptive LP call frequency."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    # Adaptive LP frequency: skip LP calls when time is tight
    lp_skip_interval = lp_frequency
    no_lp_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        remaining_frac = (time_limit - (time.monotonic() - t0)) / time_limit

        # In the last 20% of time, skip more LP calls
        effective_interval = lp_skip_interval
        if remaining_frac < 0.2:
            effective_interval = max(lp_skip_interval, 5)
        elif remaining_frac < 0.5:
            effective_interval = max(lp_skip_interval, 3)

        if it % effective_interval == 0:
            b_new, M_lp = solve_lp_best_response(y, solver)
            if b_new is None:
                b_new = a.copy()
            no_lp_count = 0
        else:
            # Use a simple heuristic: perturb current solution toward uniform
            # This simulates a weak best response
            b_new = 0.95 * a + 0.05 * np.ones(n) / n
            no_lp_count += 1
            if no_lp_count > effective_interval * 3:
                # Force an LP call if stuck too long without one
                b_new, M_lp = solve_lp_best_response(y, solver)
                if b_new is not None:
                    no_lp_count = 0

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        # Check score periodically
        if it % 20 == 0:
            sc_a = score_normalized(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                sc = score_normalized(a)
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): cur={sc:.8f} best={best_score:.8f}")
                last_print = elapsed

    # Final check
    sc = score_normalized(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = b / T_val
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    active = np.where(w > 1e-16)[0]
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsbfgs_lse_adaptive(a0, time_limit, T_start=80.0, T_min=0.3, n_stages=25, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0
    prev_score = best_score

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature update
            if prev_score > best_score + 1e-10:
                # We made progress
                stagnant = max(0, stagnant - 1)
                T = max(T_min, T * 0.7)
            elif sc > prev_score * 0.999:
                # Stagnant
                stagnant += 1
                if stagnant >= 3:
                    T = min(T_start * 0.5, T * 1.5)
                    stagnant = 0
            else:
                stagnant = max(0, stagnant - 1)
                T = max(T_min, T * 0.85)

            prev_score = best_score

            if (stage + 1) % 5 == 0:
                print(f"    LSE-Adaptive stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_subgradient(a0, time_limit, n_steps=200000, n_active=15):
    """Peak-targeted subgradient with Polyak step, peak history, and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0

    # Peak history for regularization
    peak_history = np.zeros(n)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks
        threshold = max_b * 0.97
        active = np.where(b >= threshold)[0]
        if len(active) > n_active:
            active = active[np.argsort(b[active])[-n_active:]]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Update peak history
        peak_history[active] += 1.0

        # Compute subgradient: average of gradients at active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Add peak history regularization (push away from historically peak-contributing positions)
        if step > 200:
            hist_norm = peak_history / (peak_history.sum() + 1e-15)
            # Scale history contribution by the current peak value
            g = 0.85 * g + 0.15 * (2.0 * n * max_b) * hist_norm

        # Project gradient to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.5)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Bisection line search if step too large
        if sc_new > cur_score * 1.02:
            lo_e, hi_e = 0.0, eta
            for _ in range(6):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        # Maintain running average (exponentially weighted)
        avg_a = 0.98 * avg_a + 0.02 * a
        avg_count += 1

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        # Check average periodically
        if step % 100 == 0 and avg_count > 500:
            sc_avg = score_normalized(avg_a)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_a.copy()

        # Decay peak history slowly
        if step % 500 == 0:
            peak_history *= 0.95

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-2 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n)
    elif choice == 5:
        a = 0.5 + 0.25 * np.cos(3 * 2 * np.pi * x) + 0.15 * np.cos(7 * 2 * np.pi * x)
    elif choice == 6:
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    elif choice == 8:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(2 * np.pi * 5 * x) + 0.05 * np.cos(2 * np.pi * 11 * x)
    elif choice == 9:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * (np.arange(n) ** 2) / n)
    elif choice == 10:
        freqs = rng.integers(1, 20, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.1 * np.cos(2 * np.pi * f * x + p)
    else:
        a = np.ones(n) * (1.0 + 0.1 * np.random.default_rng(choice).normal(0, 0.1, n))
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetric_constraint(a):
    """Enforce palindromic symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a[i]
        a_sym[n - 1 - i] = a[i]
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def nesterov_lp_symmetric(a0, time_limit, solver=None):
    """Nesterov LP with palindromic symmetry constraint."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = symmetric_constraint(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        a = symmetric_constraint(a)
    a = a / a.sum()

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(symmetric_constraint(y), solver)
        if b_new is None:
            continue
        b_new = symmetric_constraint(b_new)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        a_new = symmetric_constraint(a_new)

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        y_new = symmetric_constraint(y_new)

        a = a_new
        y = y_new

        if it % 20 == 0:
            sc_a = score_normalized(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()

    return best_a, best_score


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_seq_local = best_seq
    best_score_local = best_score

    def try_update(a, label):
        nonlocal best_seq_local, best_score_local
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score_local:
            best_score_local = sc
            best_seq_local = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP (main) - 30% time ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score_local:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t1 = min(0.30 * (budget_s - 40), remaining * 0.45)
    if t1 > 30:
        a_out, _ = nesterov_lp(best_seq_local, t1, solver=solver, lp_frequency=1)
        try_update(a_out, "p1_nesterov")
    print(f"  Phase 1 done: best_score={best_score_local:.8f}")

    # === Phase 2: LSE L-BFGS - 25% time ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score_local:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.25 * (budget_s - 40), remaining * 0.45)
    if t2 > 25:
        a_lse1, _ = lsbfgs_lse_adaptive(best_seq_local, t2 * 0.5, T_start=100.0, T_min=0.2, n_stages=20, iters_per_stage=35)
        try_update(a_lse1, "p2_lse1")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lsbfgs_lse_adaptive(best_seq_local, remaining * 0.4, T_start=20.0, T_min=0.1, n_stages=15, iters_per_stage=30)
            try_update(a_lse2, "p2_lse2")

        # Perturbed LSE
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            perturbed = best_seq_local + 0.05 * rng.normal(0, 1, len(best_seq_local))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_lse3, _ = lsbfgs_lse_adaptive(perturbed, remaining * 0.4, T_start=50.0, T_min=0.2, n_stages=12, iters_per_stage=25)
            try_update(a_lse3, "p2_lse_perturbed")
    print(f"  Phase 2 done: best_score={best_score_local:.8f}")

    # === Phase 3: Peak Subgradient - 20% time ===
    print(f"\n=== Phase 3: Peak Subgradient (score={best_score_local:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 40), remaining * 0.5)
    if t3 > 20:
        a_sg, _ = peak_subgradient(best_seq_local, t3 * 0.6, n_steps=500000, n_active=15)
        try_update(a_sg, "p3_sg1")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_sg2, _ = peak_subgradient(best_seq_local, remaining * 0.5, n_steps=300000, n_active=8)
            try_update(a_sg2, "p3_sg2")
    print(f"  Phase 3 done: best_score={best_score_local:.8f}")

    # === Phase 4: Symmetric Nesterov - 10% time ===
    print(f"\n=== Phase 4: Symmetric Nesterov (score={best_score_local:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 40), remaining * 0.5)
    if t4 > 20:
        a_sym = symmetric_constraint(best_seq_local)
        a_out_sym, _ = nesterov_lp_symmetric(a_sym, t4 * 0.6, solver=solver)
        try_update(a_out_sym, "p4_sym")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Unsymmetrize and continue
            a_out_sym2, _ = nesterov_lp(a_out_sym, remaining * 0.4, solver=solver, lp_frequency=2)
            try_update(a_out_sym2, "p4_from_sym")
    print(f"  Phase 4 done: best_score={best_score_local:.8f}")

    # === Phase 5: Multi-length exploration - 10% time ===
    print(f"\n=== Phase 5: Multi-length (score={best_score_local:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.10 * (budget_s - 40), remaining * 0.6)
    if t5_total > 30:
        n_base_cur = len(best_seq_local)
        lengths_to_try = [n_base_cur + 200, n_base_cur + 500, n_base_cur - 200]
        lengths_to_try = [l for l in lengths_to_try if l >= 100]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 25:
                break
            t5 = min(t5_total / max(1, len(lengths_to_try)), remaining * 0.45)
            if t5 < 15:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq_local, n_try)
            a_try, _ = nesterov_lp(init_try, t5 * 0.5, solver=solver, lp_frequency=2)
            try_update(a_try, f"p5_len{n_try}")

            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = lsbfgs_lse_adaptive(a_try, remaining * 0.4, T_start=50.0, T_min=0.3, n_stages=10, iters_per_stage=25)
                try_update(a_try2, f"p5_len{n_try}_lse")
    print(f"  Phase 5 done: best_score={best_score_local:.8f}")

    # === Phase 6: Final polish - 5% time ===
    print(f"\n=== Phase 6: Final polish (score={best_score_local:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        # Quick Nesterov
        a_f1, _ = nesterov_lp(best_seq_local, remaining * 0.3, solver=solver, lp_frequency=1)
        try_update(a_f1, "f1_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Quick LSE from low T
            a_f2, _ = lsbfgs_lse_adaptive(best_seq_local, remaining * 0.4, T_start=10.0, T_min=0.1, n_stages=10, iters_per_stage=25)
            try_update(a_f2, "f2_lse")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Quick subgradient
            a_f3, _ = peak_subgradient(best_seq_local, remaining * 0.5, n_steps=200000, n_active=20)
            try_update(a_f3, "f3_sg")

    # Final verification
    fs = evaluate_safe(best_seq_local)
    if np.isfinite(fs):
        best_score_local = fs

    print(f"\nFINAL SCORE: {best_score_local:.8f} (n={len(best_seq_local)})")
    return [float(x) for x in best_seq_local]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
