def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def exp_grad_best_response(a, b_init=None, n_inner=30, eta_scale=10.0):
    """
    Approximate best response using exponentiated gradient.
    Minimizes (1/eta)*log(sum_m exp(eta * conv(a,b)[m])) over b in simplex.
    """
    n = len(a)
    # Precompute FFT of a (for conv with b) and reversed_a (for gradient)
    conv_size = 2 * n - 1
    conv_fft = 1 << (conv_size - 1).bit_length()
    A_fft = rfft(a, conv_fft)
    
    a_rev = a[::-1]
    grad_size = n + (2*n-1) - 1  # = 3n-2
    grad_fft = 1 << (grad_size - 1).bit_length()
    A_rev_fft = rfft(a_rev, grad_fft)
    
    if b_init is None:
        b = np.ones(n) / n
    else:
        b = b_init.copy()
        b = np.maximum(b, 1e-30)
        b = b / b.sum()
    
    for it in range(n_inner):
        # Compute c = conv(a, b)
        B_fft = rfft(b, conv_fft)
        c = irfft(A_fft * B_fft, conv_fft)[:conv_size]
        
        # Compute softmax weights
        c_max = c.max()
        eta = eta_scale / max(c_max, 1e-15)
        w = softmax_fn(eta * c)
        
        # Compute grad_b = conv(a_rev, w)[n-1 : 2n-1]
        W_fft = rfft(w, grad_fft)
        grad_full = irfft(A_rev_fft * W_fft, grad_fft)[:grad_size]
        grad_b = grad_full[n-1 : 2*n-1]
        
        # Exponentiated gradient update (stable version)
        # b_new ∝ b * exp(-eta * grad_b)
        # Use shifted exponent to avoid overflow/underflow
        shifted = -eta * grad_b
        shifted -= shifted.max()
        b = b * np.exp(shifted)
        b_sum = b.sum()
        if b_sum < 1e-300:
            b = np.ones(n) / n
        else:
            b = b / b_sum
        b = np.maximum(b, 1e-30)
        b = b / b.sum()
    
    return b
