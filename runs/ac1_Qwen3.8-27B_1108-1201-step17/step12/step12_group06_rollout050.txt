def project_simplex(v):
    """Project onto probability simplex (sum=1, v>=0)."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated
approximate best-response with fast FFT-based convolution.
Key innovations: approximate LP via projected subgradient (much faster than
exact LP), adaptive Nesterov schedule, multi-length exploration, and
frequency-domain initialization for diverse starting points.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a, b=None):
    """Fast linear convolution via FFT. If b is None, computes a*a."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project onto probability simplex (sum=1, v>=0)."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v, kind='mergesort')[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def approx_best_response(a_fixed, n_inner=80):
    """
    Approximate best response: minimize max_k conv(a_fixed, b)[k]
    subject to sum(b)=1, b>=0, using projected subgradient.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A_fft = rfft(a_fixed, fft_size)
    
    b = np.ones(n) / n
    
    # Track active peaks for averaging
    active_peaks = [np.argmax(fast_convolve(a_fixed, b))]
    
    for it in range(n_inner):
        # Compute convolution
        B_fft = rfft(b, fft_size)
        c = irfft(A_fft * B_fft, fft_size)[:size]
        
        max_c = c.max()
        
        # Find top peaks (for better subgradient)
        threshold = max_c * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) > 10:
            # Keep top 10
            top_vals = c[active]
            top_idx = np.argsort(top_vals)[-10:]
            active = active[top_idx]
        
        # Compute averaged subgradient
        subgrad = np.zeros(n)
        for k in active:
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            subgrad[m_lo:m_hi+1] += a_fixed[k - m_hi:k - m_lo+1][::-1]
        subgrad /= len(active)
        
        # Step size (Polyak-style with decreasing)
        sg_max = subgrad.max()
        if sg_max < 1e-15:
            break
        eta = 1.0 / ((it + 1) * sg_max)
        eta = min(eta, 0.5 / sg_max)
        
        # Update
        b_new = b - eta * subgrad
        b = project_simplex(b_new)
    
    return b


def nesterov_approx_lp(a0, time_limit, n_inner=80, max_iters=100000):
    """
    Nesterov-accelerated approximate LP best-response iteration.
    Uses subgradient-based approximate best response.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = approx_best_response(y, n_inner=n_inner)
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = project_simplex(a_new)

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    # Gradient
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_polish(a0, time_limit, T_start=50.0, T_min=0.1, n_stages=20, iters_per_stage=30):
    """LogSumExp L-BFGS polishing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    T = T_start
    stagnant = 0
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            # Adaptive temperature
            if stagnant >= 3:
                T = max(T_min, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)
            
            stagnant = max(0, stagnant - 1) if sc < best_score + 0.001 else stagnant + 1
            
        except Exception:
            T = max(T_min, T * 0.8)
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 3:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_lp_exact(a0, time_limit, max_iters=100000):
    """
    Nesterov with exact LP best response using scipy linprog.
    """
    from scipy.optimize import linprog
    
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Pre-build A_eq and bounds (these don't change)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(None, None)]
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build A_ub for current y
        A_ub = np.zeros((size, n + 1))
        for j in range(n):
            A_ub[j:j+n, j] = y
        A_ub[:, n] = -1.0
        b_ub = np.zeros(size)
        
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'disp': False})
        
        if result.status != 0:
            # Fallback to approximate
            b_new = approx_best_response(y, n_inner=50)
        else:
            b_new = result.x[:n]
            b_new = np.maximum(b_new, 0)
            s_b = b_new.sum()
            if s_b < 1e-15:
                b_new = approx_best_response(y, n_inner=50)
            else:
                b_new = b_new / s_b
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = project_simplex(a_new)
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = project_simplex(y_new)
        
        a = a_new
        y = y_new
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
