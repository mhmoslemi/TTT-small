import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgrad(a, k_star):
    """Compute subgradient of max_k conv(a,a)_k at index k_star."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        j1 = k_star - i
        if 0 <= j1 < n:
            val += a[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            val += a[j2]
        grad[i] = 2.0 * val
    return grad

def mwu_optimize(a0, time_limit, verbose=True):
    """Multiplicative Weights Update optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive learning rate
    eta = 1.0 / (2.0 * n)  # initial
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = b[k_star]

    A = rfft(a, fft_size)

    """Compute subgradient of max_k conv(a,a)_k at index k_star."""
    n = len(a)

            val += a[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            val += a[j2]

def proj_simplex(v, S):
    """Project vector v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - S)
    if not np.any(cond):
        return np.ones(n) * S / n
    rho_val = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_val - 1] - S) / rho_val
    return np.maximum(v - theta, 0.0)

        i_lo = max(0, k_star - n + 1)

        contrib = np.zeros(n)

        # a[i] * a[k_star - i] (if k_star - i is valid)
