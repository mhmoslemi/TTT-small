def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), size)
    return irfft(A * A, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Compute conv(a, b) using FFT."""
    na, nb = len(a), len(b)
    size = 1 << (na + nb - 1).bit_length()
    A = rfft(a, size)
    B = rfft(b, size)
    return irfft(A * B, size)[:na + nb - 1]

def compute_subgradient_fft(b, y):
    """
    Compute g[i] = sum_k y[k] * b[k-i] for i=0,...,n-1 using FFT.
    y has length 2n-1, b has length n.
    """
    n = len(b)
    b_rev = b[::-1].copy()
    size = 1 << (3 * n - 2).bit_length()
    Y = rfft(y, size)
    Brev = rfft(b_rev, size)
    g_full = irfft(Y * Brev, size)[:3 * n - 2]
    g = g_full[n - 1:2 * n - 1]
    return g


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten the power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    a_avg = a.copy()
    n_avg = 1
    
    # Exponential moving average (faster convergence in practice)
    a_ema = a.copy()
    ema_alpha = 0.1

                    a = a_new

            sc_avg = 2.0 * n * fast_autocorr(a_avg).max()
            sc_ema = 2.0 * n * fast_autocorr(a_ema).max()
            print(f"      ABR t={t} ({elapsed:.0f}s): M_b={M_t:.8f}, M_a={M_br:.8f}, "
                  f"score_a={sc_a:.6f}, score_avg={sc_avg:.6f}, score_ema={sc_ema:.6f}, best={best_score:.6f}")
    
    # Final check
    for candidate in [a, a_avg, a_ema]:
        sc = 2.0 * n * fast_autocorr(candidate).max()
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver):
    """Nesterov-accelerated LP best-response iteration (standard approach)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val, y_dual = solve_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")
    
    return best_a, best_score
