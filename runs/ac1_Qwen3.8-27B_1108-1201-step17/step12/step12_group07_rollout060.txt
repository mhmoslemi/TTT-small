Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Nesterov accelerated iterations for coarse optimization
2. Active-set QP for precise peak minimization (main innovation)
3. LogSumExp L-BFGS for smooth polishing
4. Coordinate descent fine-tuning
5. Length exploration

The key innovation is the active-set QP phase, which identifies the
binding constraints (peaks of the autocorrelation) and directly solves
a convex QP to minimize the maximum peak, iteratively updating the
active set.

def fast_convolve(a):
    """Fast convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T*b)_k = (a*b)_k."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. (a_fixed * b)_k <= M, b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)

    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s

    # Actually, let's just build it properly
    all_rows = []

def active_set_qp(a0, active_indices, solver=None):
    """
    Solve QP: min M s.t. (a*a)_k <= M for k in active_indices,
    a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_indices:
        P_k = build_P_k(k, n)
        constraints.append(cp.quad_form(a_var, P_k, assume_PSD=True) <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver is None:
        solver = get_lp_solver()
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    return normalize(a_val)

def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov LP with adaptive restarts."""
    n = len(a0)
    a = normalize(a0.copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = normalize(y)
        else:
            y = a
        
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = normalize(a_new)
        
        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new
        
        a = a_new
        a = normalize(a)
    
    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    """LogSumExp smoothed objective and gradient."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=30):
    """LSE L-BFGS polishing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        iters = 50 if si < n_stages // 2 else 25
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            z = z_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, max_rounds=10000):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            # Try a several-step line search on coordinate i
            a_i_orig = a[i]
            if a_i_orig < 1e-10:
                continue
            # The objective as a function of a_i (with others fixed) is quadratic
            # (a*a)_k as function of a_i: for each k, contribution is 2*a_i*sum_{j != i, j valid} a_j + a_i^2 (if 2i valid)
            # Actually: (a*a)_k = a_i * a_{k-i} + sum_{j != i, j valid} a_j * a_{k-j}
            # So fixing all except a_i: (a*a)_k = a_i * a_{k-i} + c_k (where c_k doesn't involve a_i)
            # Wait, that's not right either. Let me think again.
            # (a*a)_k = sum_j a_j * a_{k-j}
            # The terms involving a_i are: a_i * a_{k-i} (when i is in range) and a_{k-i} * a_i (when k-i is in range)
            # If i != k-i: both terms are a_i * a_{k-i}, so total contribution is 2 * a_i * a_{k-i}
            # If i == k-i: the term is a_i^2
            # So: (a*a)_k = a_i^2 * [i == k-i] + 2 * a_i * a_{k-i} * [i != k-i, both valid] + rest_k
            # where rest_k = sum_{j != i, k-j != i} a_j * a_{k-j}
            
            # This means the objective max_k (a*a)_k is the max of quadratic functions of a_i.
            # Each is of the form: alpha_k * a_i^2 + beta_k * a_i + gamma_k
            # where alpha_k = 1 if 2i=k, 0 otherwise
            #       beta_k = 2*a_{k-i} if i != k-i and i != 0 and k-i valid, a_{k-i} if i == k-i... 
            # Actually let me just do a simple grid search / golden section
            
            # Simple approach: try a few values
            for factor in [0.5, 0.8, 1.2, 1.5, 2.0, 0.3]:
                a[i] = a_i_orig * factor
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = a_i_orig
                    break  # Go back and try next direction
            a[i] = a_i_orig
        
        if not improved and rnd > 50:
            break
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initialization."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.8 * (2.0 * x - 1.0)
    elif choice == 1:
        a = x + 0.1
    elif choice == 2:
        a = 2.0 * np.exp(-3.0 * x) + 0.3
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.2
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (x + 0.1)) + 0.15 * np.cos(4 * np.pi * x)
    elif choice == 5:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.7) ** 2) + 0.1
    elif choice == 6:
        a = (x + 0.05) ** 0.5
    else:
        a = 0.8 + 0.4 * np.sin(2 * np.pi * x + 0.5)
    a = np.maximum(a, 0.01)
    return normalize(a)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

def active_set_qp_iteration(a0, time_limit, solver=None, K_max=50, tol=1e-8):
    """
    Main active set QP iteration:
    1. Compute b = a*a, find top K peaks
    2. Solve QP: min M s.t. (a*a)_k <= M for k in active set
    3. Update active set
    4. Repeat until convergence
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    iteration = 0
    
    K = min(15, n)  # Start with small active set
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_convolve(a)
        M_current = b.max()
        
        # Find active set: peaks close to the maximum
        threshold = M_current * (1.0 - 1e-4)
        active_candidates = np.where(b >= threshold)[0]
        
        # Expand if needed
        if len(active_candidates) == 0:
            active_candidates = np.array([int(np.argmax(b))])
        
        # Also include the top K peaks by value
        top_k = np.argsort(b)[-K:]
        active_set = np.unique(np.concatenate([active_candidates, top_k]))
        active_set = active_set[np.argsort(-b[active_set])]
        K_use = min(len(active_set), K_max)
        active_set = active_set[:K_use]
        
        # Solve QP
        a_new = active_set_qp(a, active_set, solver=solver)
        
        if a_new is not None:
            a_new = normalize(a_new)
            sc_new = score_normalized(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
            
            # Check convergence
            b_new = fast_convolve(a)
            if abs(b_new.max() - M_current) < tol * M_current:
                # Converged; try with larger active set
                K = min(K * 2, K_max)
                if K > K_max:
                    break
            else:
                # Update K adaptively
                # If we made progress, keep same K; if not, increase
                new_M = b_new.max()
                if new_M >= M_current * (1.0 - 1e-6):
                    K = min(K + 5, K_max)
        else:
            # QP failed, try smaller active set
            K = max(K // 2, 3)
            if K < 3:
                break
    
    return best_a, best_score

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a_fixed)

def build_P_k_sparse(k, n):
    """Build sparse P_k where a^T P_k a = (a*a)_k."""
    rows = []
    cols = []
    vals = []
    for i in range(max(0, k - n + 1), min(n, k + 1)):
        j = k - i
        if 0 <= j < n:
            if i == j:
                rows.append(i)
                cols.append(j)
                vals.append(1.0)
            else:
                rows.append(i)
                cols.append(j)
                vals.append(0.5)
                rows.append(j)
                cols.append(i)
                vals.append(0.5)
    P = csr_matrix((vals, (rows, cols)), shape=(n, n))
    return P

def active_set_cutting_plane(a0, time_limit, K_max=40, solver=None):

    Iteratively adds violated constraints and solves the QP subproblem.

    S = set(int(k) for k in range(len(b)) if b[k] >= M_cur * 0.995)
    if not S:
        S = {int(np.argmax(b))}

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        it_count += 1
        # Limit active set size
        if len(S) > K_max:
            b_cur = fast_convolve(a)
            top_k = np.argsort(b_cur)[-K_max:]
            S = set(int(k) for k in top_k)

        # Solve QP
        a_new, M_qp = solve_active_qp(list(S), n, solver=solver)

        if a_new is None:
            # Try with smaller active set
            if len(S) > 5:
                b_cur = fast_convolve(a)
                top_k = np.argsort(b_cur)[-5:]
                S = set(int(k) for k in top_k)
                continue
            break

        a_new = normalize(a_new)
        b_new = fast_convolve(a_new)
        M_new = b_new.max()

        # Check if improved
        if M_new < M_cur * (1.0 - 1e-9):
            M_cur = M_new
            a = a_new
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Find violated constraints (peaks exceeding M_qp significantly)
        if M_qp is not None:
            violated = set(int(k) for k in range(len(b_new))
                           if b_new[k] > M_qp + 1e-10 and k not in S)
        else:
            violated = set(int(k) for k in range(len(b_new))
                           if b_new[k] >= M_new * 0.999 and k not in S)

        if len(violated) == 0:
            # All constraints satisfied - check convergence
            if M_new >= M_cur * (1.0 - 1e-10):
                break
            # Slight progress possible, add near-max peaks
            near_max = set(int(k) for k in range(len(b_new))
                           if b_new[k] >= M_new * 0.999)
            S = S | near_max
        else:
            # Add violated constraints, prioritize most violated
            if len(violated) > K_max - len(S):
                # Keep only top violators
                viol_list = sorted(violated, key=lambda k: -b_new[k])
                room = max(0, K_max - len(S))
                S = S | set(viol_list[:room])
            else:
                S = S | violated

        if (it + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ACS iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} |S|={len(S)}")

    return best_a, best_score


def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov LP with adaptive restarts."""
    n = len(a0)
    a = normalize(a0.copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = normalize(y)
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = normalize(a_new)

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_func_wrapper(z_val, T_val, n):
    obj, _ = lse_objective_grad(z_val, T_val, n)
    return obj


def lse_jac_wrapper(z_val, T_val, n):

    return grad

    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = 40 if si < n_stages // 2 else 20
        try:
            res = minimize(
                fun=lse_func_wrapper, x0=z, args=(stage_T, n),
                jac=lse_jac_wrapper, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            z = z_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def mult_weight_refine(a0, time_limit, max_steps=50000):
    """Multiplicative weights targeting top peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.04
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.93
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.003, 0.12)
        else:
            eta *= 0.88
            if eta < 1e-10:
                break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.8 * (2.0 * x - 1.0)
    elif choice == 1:
        a = x + 0.1
    elif choice == 2:
        a = 2.0 * np.exp(-3.0 * x) + 0.3
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.2
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (x + 0.1)) + 0.15 * np.cos(4 * np.pi * x)
    elif choice == 5:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.7) ** 2) + 0.1
    elif choice == 6:
        a = (x + 0.05) ** 0.5
    else:
        a = 0.8 + 0.4 * np.sin(2 * np.pi * x + 0.5)
    a = np.maximum(a, 0.01)
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using solver: {solver}")

    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = normalize(arr)
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 1: LP Nesterov (200s) ===
    print(f"\n=== Phase 1: LP Nesterov (score={best_score:.8f}) ===")
    t1 = min(200.0, (_global_deadline - time.monotonic()) * 0.12)
    if t1 > 10:
        a_out, _ = nesterov_lp_adaptive(best_seq, t1, solver=solver)
        try_update(a_out, "lp_nesterov_p1")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Active Set Cutting Plane QP (600s) ===
    print(f"\n=== Phase 2: Active Set CP-QP (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(600.0, remaining * 0.40)
    if t2 > 60:
        a_out, _ = active_set_cutting_plane(best_seq, t2, K_max=40, solver=solver)
        try_update(a_out, "acs_qp")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: LSE L-BFGS (300s) ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(300.0, remaining * 0.22)
    if t3 > 20:
        a_lse = lbgfs_logsumexp(best_seq, t3, T_start=30.0, T_end=0.01, n_stages=30)
        try_update(a_lse, "lse_polish")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Active Set CP-QP round 2 (250s) ===
    print(f"\n=== Phase 4: ACS-QP round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.20)
    if t4 > 30:
        a_out2, _ = active_set_cutting_plane(best_seq, t4, K_max=40, solver=solver)
        try_update(a_out2, "acs_qp_r2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multiplicative refinement (150s) ===
    print(f"\n=== Phase 5: Multi-weight refine (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(150.0, remaining * 0.12)
    if t5 > 10:
        a_mw, _ = mult_weight_refine(best_seq, t5, max_steps=30000)
        try_update(a_mw, "mult_weight")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final LSE + LP (150s) ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        t6a = min(80.0, remaining * 0.5)
        if t6a > 10:
            a_lse2 = lbgfs_logsumexp(best_seq, t6a, T_start=10.0, T_end=0.005, n_stages=20)
            try_update(a_lse2, "lse_final")
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            t6b = min(70.0, remaining * 0.7)
            if t6b > 10:
                a_lp2, _ = nesterov_lp_adaptive(best_seq, t6b, solver=solver)
                try_update(a_lp2, "lp_final")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Length exploration (80s) ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        for n_try in [best_n + 50, best_n - 50]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try < 100:
                continue
            t7 = min(35.0, remaining * 0.4)
            if t7 < 10:
                break
            print(f"  Trying n={n_try}")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_adaptive(init_try, t7, solver=solver)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 7 done: best_score={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
