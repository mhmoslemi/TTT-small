"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based subgradient with Polyak step and iterate averaging
2. Active-set LSE polish (minimize sum of top peaks with L-BFGS on log-param)
3. Nesterov LP best-response for exact Nash equilibrium
4. Multi-length exploration and final polishing
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_grad_convolve(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a.
    grad_i = 2 * a[k-i] if 0 <= k-i < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * a[k - i]
    return grad

def softmax_from_z(z):
    """Stable softmax (returns probability vector)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))

def fast_subgradient(a0, time_limit, n_steps=1000000):
    """Fast O(n log n) subgradient method for the max-autocorrelation problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0
    
    # Adaptive step size tracking
    step_sizes = []
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        # Find peak
        k_star = int(np.argmax(b))
        
        # Compute gradient
        grad = compute_grad_convolve(a, k_star)
        
        # Project gradient to tangent space of simplex
        S = np.dot(grad, a)
        grad_tangent = grad - S  # Tangent to sum=1 constraint
        
        # For the a >= 0 constraint, we'll handle via projection
        grad_norm_sq = np.dot(grad_tangent, grad_tangent)
        if grad_norm_sq < 1e-20:
            break
        
        # Polyak step size
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / grad_norm_sq
        eta = min(eta, 1.0)  # Cap
        
        # Take step and project
        a_new = a - eta * grad_tangent
        a_new = project_simplex(a_new)
        
        # Evaluate
        sc_new = score_normalized(a_new)
        
        # Line search if needed
        if sc_new > cur_score * 1.05:
            # Bisection
            for _ in range(8):
                eta = eta * 0.5
                a_new = a - eta * grad_tangent
                a_new = project_simplex(a_new)
                sc_new = score_normalized(a_new)
                if sc_new <= cur_score:
                    break
        
        a = a_new
        sc_new = score_normalized(a)
        
        # Update average (exponential moving average)
        avg_a = 0.999 * avg_a + 0.001 * a
        avg_count += 1
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
        
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score

def active_set_lse(a0, time_limit, n_peaks=10, max_rounds=50):
    """Active-set LSE polish: minimize sum of top-n_peaks autocorrelation values."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    
    t0 = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Identify active peaks
        b = fast_convolve(a)
        top_indices = np.argsort(b)[-n_peaks:]
        
        # Define the smooth objective: sum of (a*a)[k] for k in top_indices
        active_k = np.sort(top_indices)
        
        def obj_and_grad(z_val):
            a_val = softmax_from_z(z_val)
            # Compute sum of (a*a)[k] for active k
            val = 0.0
            grad_a = np.zeros(n)
            for k in active_k:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                # (a*a)[k] = sum_{j=lo}^{hi} a_j * a_{k-j}
                js = np.arange(lo, hi + 1)
                val += np.dot(a_val[js], a_val[k - js])
                # Gradient: 2 * a[k-i] for i in [lo, hi]
                grad_a[js] += 2.0 * a_val[k - js]
            
            # Convert to z-gradient
            S = np.dot(grad_a, a_val)
            grad_z = a_val * (grad_a - S)
            return val, grad_z
        
        # Run L-BFGS
        remaining = time_limit - (time.monotonic() - t0)
        max_iter = max(10, int(remaining * 10))  # Scale iterations with time
        max_iter = min(max_iter, 200)
        
        try:
            res = minimize(
                fun=lambda z_val: obj_and_grad(z_val)[0],
                x0=z,
                jac=lambda z_val: obj_and_grad(z_val)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16, 'disp': False}
            )
            z = res.x
            a = softmax_from_z(z)
            sc = score_normalized(a)
            
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                
        except Exception:
            pass
        
        # Check convergence
        b_new = fast_convolve(a)
        if abs(b_new.max() - b.max()) < 1e-12:
            break
        
        b = b_new
        elapsed = time.monotonic() - t0
        if (rnd + 1) % 5 == 0:
            print(f"    ActiveSet-LSE round={rnd+1} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f}")
    
    return best_a, best_score

def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b_val, M_val) or (None, None)."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp(a0, time_limit, max_iters=100000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_lp = solve_lp_best_response(y)
        if b_new is None:
            continue
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed
    
    return best_a, best_score

def lse_objective_grad(z_val, T_val, n, active_k=None):
    """LogSumExp objective and gradient, optionally restricted to active peaks."""
    a = softmax_from_z(z_val)
    b = fast_convolve(a)
    
    if active_k is not None:
        Tb = T_val * b[active_k]
    else:
        Tb = T_val * b
    
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    
    if active_k is not None:
        obj = (2.0 * n / T_val) * log_sum_exp
        # Gradient restricted to active peaks
        grad_a = np.zeros(n)
        for idx, k in enumerate(active_k):
            if w[idx] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad_a[idx_arr] += 2.0 * w[idx] * a[k - idx_arr]
    else:
        obj = (2.0 * n / T_val) * log_sum_exp
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad_a[idx_arr] += 2.0 * w[k] * a[k - idx_arr]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lse_lbfgs(a0, time_limit, T_start=100.0, T_min=0.1, n_stages=30, iters_per_stage=50):
    """Adaptive LogSumExp L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    T = T_start
    stagnant = 0
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_val):
            return lse_objective_grad(z_val, T, n)[0]
        
        def jac(z_val):
            return lse_objective_grad(z_val, T, n)[1]
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            # Temperature update
            if stagnant >= 3:
                T = max(T, T * 1.1)
                stagnant = 0
            else:
                T = max(T_min, T * 0.8)
            
            # Check for stagnation
            if sc > best_score * 1.001:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 1)
                
        except Exception:
            T = max(T_min, T * 0.8)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_grad_convolve(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def softmax_from_z(z):
    """Stable softmax (returns probability vector)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def fast_subgradient(a0, time_limit, n_steps=1000000):
    """Fast O(n log n) subgradient method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:

    a = a / s

    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    avg_a = a.copy()

    for step in range(n_steps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        k_star = int(np.argmax(b))
        grad = compute_grad_convolve(a, k_star)

        S = np.dot(grad, a)
        grad_tangent = grad - S

        grad_norm_sq = np.dot(grad_tangent, grad_tangent)
        if grad_norm_sq < 1e-20:
            break

        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / grad_norm_sq
        eta = min(eta, 1.0)

        a_new = a - eta * grad_tangent
        a_new = project_simplex(a_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.1:
            for _ in range(10):
                eta *= 0.5
                a_new = a - eta * grad_tangent
                a_new = project_simplex(a_new)
                sc_new = score_normalized(a_new)
                if sc_new <= cur_score:
                    break

        a = a_new
        avg_a = 0.9995 * avg_a + 0.0005 * a

        sc_new = score_normalized(a)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        if (step + 1) % 2000 == 0:

def active_set_func(z_val, active_k, n):
    """Sum of (a*a)[k] for k in active_k, as function of z."""
    a = softmax_from_z(z_val)
    val = 0.0
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        val += np.dot(a[js], a[k - js])
    return val


def active_set_jac(z_val, active_k, n):
    """Gradient of sum of (a*a)[k] for k in active_k, as function of z."""
    a = softmax_from_z(z_val)
    grad_a = np.zeros(n)
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad_a[js] += 2.0 * a[k - js]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

                best_a = a_check.copy()

            if stagnant >= 3:
                T = max(T, T * 1.1)
                stagnant = 0
            else:
                T = max(T_min, T * 0.75)

            if sc > best_score * 1.002:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 1)

        except Exception:
            T = max(T_min, T * 0.8)

        if (stage + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, max_iters=50000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def fourier_coeffs_to_seq(coeffs, n):
    """Convert Fourier coefficients to a non-negative sequence.
    coeffs[0] is DC, coeffs[k] for k>=1 is amplitude of mode k."""
    j = np.arange(n)
    a = np.ones(n) * coeffs[0]
    for k in range(1, len(coeffs)):
        a = a + coeffs[k] * np.cos(2.0 * np.pi * k * j / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fourier_optimize(a0, time_limit, n_harmonics=8):
    """Optimize Fourier coefficients of the sequence."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    # Extract initial Fourier coefficients
    j = np.arange(n)
    coeffs = np.zeros(n_harmonics + 1)
    coeffs[0] = np.mean(a)
    for k in range(1, n_harmonics + 1):
        coeffs[k] = 2.0 / n * np.dot(a, np.cos(2.0 * np.pi * k * j / n))

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    def _fourier_obj(c):
        seq = fourier_coeffs_to_seq(c, n)
        return score_normalized(seq)

    try:
        res = minimize(
            fun=_fourier_obj, x0=coeffs,
            method='Nelder-Mead',
            options={'maxiter': 5000, 'fatol': 1e-14, 'xatol': 1e-14, 'disp': False}
        )
        a_new = fourier_coeffs_to_seq(res.x, n)
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
    except Exception:
        pass

    # Also try from uniform
    if time.monotonic() - t0 < time_limit * 0.5:
        uniform_coeffs = np.zeros(n_harmonics + 1)
        uniform_coeffs[0] = 1.0 / n
        try:
            res2 = minimize(
                fun=_fourier_obj, x0=uniform_coeffs,
                method='Nelder-Mead',
                options={'maxiter': 5000, 'fatol': 1e-14, 'xatol': 1e-14, 'disp': False}
            )
            a_new2 = fourier_coeffs_to_seq(res2.x, n)
            sc2 = score_normalized(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
        except Exception:
            pass

    print(f"    Fourier opt: best={best_score:.8f}")
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
