def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

        return _lp_solver_name

        try:

            prob.solve(solver=solver_name)

                return solver_name

    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def spectral_init(a_parent, rng):
    """Create a new initialization by randomizing phases in Fourier domain."""
    n = len(a_parent)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a_parent, fft_size)
    # Randomize phases
    phases = rng.uniform(0, 2 * np.pi, size=fft_size // 2 + 1)
    phases[0] = 0  # Keep DC component real
    A_new = np.abs(A) * np.exp(1j * phases)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    return normalize(a_new)


def u_shaped_init(n, rng, c=1.0):
    """Create a U-shaped (edge-weighted) initialization."""
    x = np.arange(n) / (n - 1)
    a = 1.0 + c * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.01)
    return normalize(a)


def bimodal_init(n, rng):
    """Create a bimodal initialization."""
    x = np.arange(n) / (n - 1)
    center1 = 0.2 + 0.1 * rng.uniform(-1, 1)
    center2 = 0.8 + 0.1 * rng.uniform(-1, 1)
    a = np.exp(-20.0 * (x - center1) ** 2) + np.exp(-20.0 * (x - center2) ** 2) + 0.05
    return normalize(a)

def nesterov_lp_diversified(a0, time_limit, solver=None, max_iters=100000, div_interval=200):
    """LP Nesterov with periodic spectral diversification."""
    n = len(a0)
    rng = np.random.default_rng(42)
    a = normalize(np.asarray(a0, dtype=np.float64))
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    div_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Diversity injection
        if it > 0 and it % div_interval == 0 and it % (div_interval * 2) == 0:
            # Every other diversity step, do spectral perturbation
            idx = rng.integers(0, n)
            bandwidth = max(1, n // 50)
            start = max(0, idx - bandwidth)
            end = min(n, idx + bandwidth)
            noise = rng.normal(0, 0.1 * a[start:end].mean(), size=end - start)
            a[start:end] = np.maximum(a[start:end] + noise, 0)
            a = normalize(a)
            t_nest = 1.0
            a_prev = a.copy()
            div_count += 1
            continue

        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = normalize(y)
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = normalize(a_new)

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Adaptive restart
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 10.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLDP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} restarts={restart_count} div={div_count}")

    return best_a, best_score

        je = min(n - 1, k)

        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    k_idx = np.arange(size)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    T = np.zeros((size, n))
    mask = (diff >= 0) & (diff < n)
    T[mask] = a_fixed[diff[mask].astype(np.int32)]
    return T


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def spectral_init(a_parent, rng):
    n = len(a_parent)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a_parent, fft_size)
    phases = rng.uniform(0, 2 * np.pi, size=fft_size // 2 + 1)
    phases[0] = 0.0
    A_new = np.abs(A) * np.exp(1j * phases)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    return normalize(a_new)


def u_shaped_init(n, c=1.5):
    x = np.arange(n, dtype=np.float64) / (n - 1)

def bimodal_init(n, rng):

def peak_shaving_refine(a0, time_limit, max_steps=30000):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coordinate_polish(a0, time_limit, n_factors=5):

            a_ref, _ = subsgradient_quick(init, t_left, max_steps=20000)

    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Multi-start LP Nesterov (600s) ===
    print(f"\n=== Phase 2: LP Nesterov main (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2_total = min(600.0, remaining * 0.4)
    if t2_total > 30:
        # Run from best current
        t2a = min(t2_total * 0.6, _global_deadline - time.monotonic())
        if t2a > 20:
            a_out, _ = nesterov_lp_diversified(best_seq, t2a, solver=solver,
                                               max_iters=500000, div_interval=200, seed=1)
            try_update(a_out, "nesterov_main_a")

        # Run from a different init (spectral or U-shaped)
        t2b = min(t2_total * 0.4, _global_deadline - time.monotonic())
        if t2b > 20:
            alt_init = u_shaped_init(best_n, 2.5)
            alt_init = interpolate_sequence(alt_init, best_n)
            a_out2, _ = nesterov_lp_diversified(alt_init, t2b, solver=solver,
                                                max_iters=300000, div_interval=150, seed=2)
            try_update(a_out2, "nesterov_main_b")

    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: LSE L-BFGS multi-stage (350s) ===
    print(f"\n=== Phase 3: LSE L-BFGS polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(350.0, remaining * 0.25)
    if t3 > 20:
        a_lse = lbgfs_logsumexp(best_seq, t3, T_start=200.0, T_end=0.01, n_stages=30)
        try_update(a_lse, "lse_polish_1")

    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Peak shaving + second Nesterov round (350s) ===
    print(f"\n=== Phase 4: Peak shaving + Nesterov r2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4a = min(150.0, remaining * 0.25)
    if t4a > 10:
        a_ps, _ = peak_shaving_refine(best_seq, t4a, max_steps=30000)
        try_update(a_ps, "peak_shaving_1")

    remaining = _global_deadline - time.monotonic()
    t4b = min(200.0, remaining * 0.3)
    if t4b > 15:
        a_out3, _ = nesterov_lp_diversified(best_seq, t4b, solver=solver,
                                            max_iters=300000, div_interval=100, seed=3)
        try_update(a_out3, "nesterov_r2")

    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: LSE round 2 + coordinate polish (250s) ===
    print(f"\n=== Phase 5: LSE r2 + coord polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5a = min(150.0, remaining * 0.35)
    if t5a > 15:
        a_lse2 = lbgfs_logsumexp(best_seq, t5a, T_start=50.0, T_end=0.005, n_stages=20)
        try_update(a_lse2, "lse_polish_2")

    remaining = _global_deadline - time.monotonic()
    t5b = min(100.0, remaining * 0.3)
    if t5b > 10:
        a_cd, _ = coordinate_polish(best_seq, t5b, n_factors=5)
        try_update(a_cd, "coord_polish")

    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Length exploration + final LSE (200s) ===
    print(f"\n=== Phase 6: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 40:
        for n_try in [best_n + 200, best_n + 400]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 25:
                break
            t6 = min(60.0, remaining * 0.4)
            if t6 < 10:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_diversified(init_try, t6, solver=solver,
                                               max_iters=100000, div_interval=100, seed=5)
            try_update(a_try, f"len_{n_try}")

    # Final LSE polish
    remaining = _global_deadline - time.monotonic()
    if remaining > 15:
        t_final = min(remaining * 0.5, 60.0)
        if t_final > 10:
            a_final = lbgfs_logsumexp(best_seq, t_final, T_start=20.0, T_end=0.005, n_stages=10)
            try_update(a_final, "lse_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
