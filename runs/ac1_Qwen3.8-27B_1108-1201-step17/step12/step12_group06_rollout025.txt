"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation with adaptive log-sum-exp
2. L-BFGS-B with geometric temperature cooling
3. Adam optimizer with adaptive learning rate

import numpy as np

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp(x, T):
    """Compute log-sum-exp with temperature T. Approximates max as T->0."""
    x_scaled = x / T
    x_max = x_scaled.max()
    return T * (x_max + np.log(np.exp(x_scaled - x_max).sum()))


def lse_obj_grad(z, T, n):
    """
    Compute log-sum-exp objective and gradient w.r.t. z (log-parameter).
    a = softmax(z), obj = 2n * LSE(conv(a,a), T)
    Gradient computed via FFT-based cross-correlation.
    """
    a = softmax(z)
    b = fast_convolve(a)
    
    # LSE of b with temperature T
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()  # softmax weights over b
    lse_val = T * (b_max + np.log(exp_b.sum()))
    obj = 2.0 * n * lse_val
    
    # Gradient w.r.t. a:
    # grad_a[i] = 2 * c_i - w[2i] * a[i]
    # where c_i = sum_{m=0}^{n-1} w[i+m] * a[m] (cross-correlation)
    
    # Compute c via FFT: c is the cross-correlation of w (padded to length 2n-1) and a
    # c_i = sum_m w[i+m] * a[m]
    # This is like a "anti-convolution": (w * a)[i] but with w reversed
    # Actually: c_i = <w[i:i+n], a> = (cross-corr of w and a at lag i)
    # We can compute this as: if we define wpad = w (length 2n-1), 
    # then c_i = sum_{m=0}^{n-1} wpad[i+m] * a[m]
    # = (w_reversed_conv_a)[...] 
    
    # Simpler: use direct FFT approach
    # c_i = sum_m w[i+m] * a[m] for i = 0..n-1
    # This is the cross-correlation: c = ifft(fft(w, M) * conj(fft(a, M))) ... no
    # 
    # Actually, let me think of it as: 
    # Define w_ext = w (length 2n-1)
    # c_i = sum_{m=0}^{n-1} w_ext[i+m] * a[m]
    # This is the (i)-th element of the "sliding dot product" of w_ext with a
    # = (w_ext cross-correlated with a)[i]
    #
    # Cross-correlation can be computed as: 
    # corr[i] = sum_m x[m] * y[i+m] = ifft(conj(fft(x)) * fft(y)) ... 
    # Hmm, let me just use a different approach.
    #
    # c_i = sum_{m=0}^{n-1} w[i+m] * a[m]
    # Let's define v = w reversed: v[k] = w[2n-2-k] for k in 0..2n-2
    # Then c_i = sum_m w[i+m] * a[m] = sum_m v[2n-2-i-m] * a[m]
    # = (v * a)[2n-2-i] (convolution)
    # 
    # So c = reverse(conv(v, a))[n-1 : 2n-1] ... this is getting confusing.
    #
    # Let me just compute it directly. For n=1200, O(n^2) = 1.44M which is fine.
    # But let me try to be smarter.
    #
    # c_i = sum_{m=0}^{n-1} w[i+m] * a[m]
    # This is the inner product of w[i:i+n] with a.
    # If we pad a to length 2n-1 (with zeros), then:
    # c = np.correlate(w, a, mode='valid') ... no, that's not quite right.
    #
    # Actually: np.correlate(w, a, mode='valid') gives:
    # result[i] = sum_k w[k+i] * a[k] for valid i
    # But w has length 2n-1 and a has length n.
    # 'valid' mode: the result has length max(2n-1, n) - min(2n-1, n) + 1 = 2n-1-n+1 = n
    # result[i] = sum_{k=0}^{n-1} w[k+i] * a[k] = c_i
    # 
    # So c = np.correlate(w, a, mode='valid')
    # But np.correlate might be slow for large arrays. Let's use FFT.
    #
    # Using FFT: correlate(w, a) = ifft(fft(w, M) * conj(fft(a, M)))
    # Actually, the cross-correlation theorem says:
    # (x ⋆ y)[n] = sum_k x[k] y[k+n] = ifft(FFT(x) * conj(FFT(y)))
    # where we need FFT size >= len(x) + len(y) - 1
    #
    # So c = ifft(fft(w, M) * conj(fft(a, M)))[:n] where M >= 2n-1+n-1 = 3n-2
    
    M = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, M)
    A = rfft(a, M)
    c_full = irfft(W * np.conj(A), M)  # Hmm, I need to be careful with real FFT
    # Actually for real signals: ifft(FFT(w) * conj(FFT(a))) gives the correlation
    # With rfft/irfft: irfft(rfft(w, M) * conj(rfft(a, M)), M) should work
    c = c_full[:n]
    
    # Correction for the diagonal term
    # grad_a[i] = 2*c_i - w[2i]*a[i] (where w[2i] = 0 if 2i >= 2n-1)
    two_i = 2 * np.arange(n)
    w_diag = np.where(two_i < len(w), w[two_i], 0.0)
    grad_a = 2.0 * c - w_diag * a
    
    # Convert to gradient w.r.t. z via softmax Jacobian:
    # grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def coordinate_descent_polish(a0, time_limit, n_passes=100):
    """Coordinate-wise descent to polish the solution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)
    
    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Try to reduce the peak by adjusting elements that contribute to it
        peak_idx = np.argmax(b)
        
        # The peak at position k involves a[j]*a[k-j] for valid j
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(n - 1, peak_idx)
        
        # Try reducing the largest contributors
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            
            contrib = a[j_lo:j_hi+1] * a[peak_idx - (j_lo:j_hi+1)]
            max_j = j_lo + np.argmax(contrib)
            
            # Reduce a[max_j] slightly and redistribute
            old_val = a[max_j]
            a[max_j] *= 0.95
            # Redistribute to random other positions
            redistribution = old_val * 0.05
            targets = rng_choice = np.random.choice(n, size=min(10, n), replace=False)
            a[targets] += redistribution / len(targets)
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            else:
                a[max_j] = old_val  # revert
                a = a / a.sum()
        
        if (p + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDescent pass={p+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_lp(a0, time_limit, max_iters=100000):
    """
    Nesterov-accelerated LP best-response (simplified, no cvxpy).
    Uses a projected gradient approach instead of full LP solve.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute "best response" direction: minimize max(T(a) @ b) s.t. sum(b)=1, b>=0
        # Approximate with softmax of negative T(a)^T @ peak_indicators
        b_conv = fast_convolve(a)
        max_b = b_conv.max()
        
        # Find active constraints (where b_conv is close to max)
        threshold = max_b * 0.95
        active = np.where(b_conv >= threshold)[0]
        
        # Build gradient for best response
        # We want to push b away from positions where conv is high
        # The "cost" of b at position j is sum_k 1[conv_a[k] >= threshold] * T(k,j)
        # where T(k,j) = a[k-j] if 0<=k-j<n else 0
        
        cost = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            cost[j_lo:j_hi+1] += a[k - (j_lo:j_hi+1)]
        
        # Best response: b proportional to exp(-cost/T_br)
        T_br = max_b * 0.5
        if T_br < 1e-10:
            b_new = np.ones(n) / n
        else:
            b_new = np.exp(-cost / T_br)
            b_new = b_new / b_new.sum()
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation with adaptive log-sum-exp (LSE)
2. L-BFGS-B with geometric temperature cooling
3. Adam optimizer as complementary method
4. Nesterov-accelerated soft best-response
5. Multi-restart with diverse structured initializations
6. Coordinate-wise and local search polishing

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_func(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad(z, T, n):
    """
    Compute log-sum-exp objective and gradient w.r.t. z (log-parameter).
    a = softmax(z), obj = 2n * LSE(conv(a,a), T)
    Gradient computed via FFT-based cross-correlation in O(n log n).
    """
    a = softmax_func(z)
    b = fast_convolve(a)

    # LSE of b with temperature T
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()
    lse_val = T * (b_max + np.log(exp_b.sum()))
    obj = 2.0 * n * lse_val

    # Compute c_i = sum_{m=0}^{n-1} w[i+m] * a[m] via FFT cross-correlation
    # c = (a *cross-corr* w) at lags 0..n-1
    M = 1 << (3 * n - 1).bit_length()
    A = rfft(a, M)
    W = rfft(w, M)
    # Cross-correlation: ifft(conj(FFT(a, M)) * FFT(w, M))
    c_full = irfft(np.conj(A) * W, M)
    c = c_full[:n]

    # grad_a[i] = 2*c_i - w[2i]*a[i] (with w[2i]=0 if 2i >= 2n-1)
    two_i = 2 * np.arange(n)
    w_diag = np.where(two_i < len(w), w[two_i], 0.0)
    grad_a = 2.0 * c - w_diag * a

    # Softmax reparametrization: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lnbfgs_lse(a0, time_limit, T_start=50.0, T_min=0.05, cooldown=0.85):
    """L-BFGS-B with log-sum-exp and geometric temperature cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    T = T_start
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    stall = 0

    stage = 0
    while time.monotonic() - t0 < time_limit:
        if T < T_min:
            break

        def func(z_val):
            return lse_obj_grad(z_val, T, n)[0]

        def jac(z_val):
            return lse_obj_grad(z_val, T, n)[1]

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': 80, 'ftol': 1e-18, 'gtol': 1e-14, 'disp': False})
            z = res.x
        except Exception:
            pass

        a_now = softmax_func(z)
        sc_now = score_normalized(a_now)

        if sc_now < best_score:
            best_score = sc_now
            best_a = a_now.copy()
            stall = 0
        else:
            stall += 1

        if (stage + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-LBFGS stage={stage+1} T={T:.3f}: score={sc_now:.8f} best={best_score:.8f}")

        if stall >= 6:
            T = max(T * 1.15, T_min)
            stall = 2
        else:
            T *= cooldown

        stage += 1
        if time.monotonic() - t0 > time_limit - 3:
            break

    return best_a, best_score


def adam_lse(a0, time_limit, T_start=40.0, T_min=0.05, lr=0.005):
    """Adam optimizer with log-sum-exp."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    T = T_start
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if T < T_min:
            T = T_min

        obj, grad = lse_obj_grad(z, T, n)
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_now = softmax_func(z)
        sc_now = score_normalized(a_now)

        if sc_now < best_score:
            best_score = sc_now
            best_a = a_now.copy()
            stall = 0
            T *= 0.88
        else:
            stall += 1
            if stall % 30 == 0:
                T *= 0.95

        if step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam step={step} T={T:.3f}: score={sc_now:.8f} best={best_score:.8f}")

    return best_a, best_score


def nesterov_soft_lr(a0, time_limit, max_iters=50000):
    """
    Nesterov-accelerated soft best-response iteration.
    Uses softmax-based best response instead of full LP solve.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute best response direction
        b_conv = fast_convolve(a)
        max_b = b_conv.max()

        # Active constraints: where b_conv is in top range
        threshold = max_b * 0.9
        active = np.where(b_conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b_conv))])
        if len(active) > 80:
            top_idx = np.argsort(b_conv[active])[-80:]
            active = active[top_idx]

        # Compute cost vector for best response
        cost = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            cost[j_lo:j_hi + 1] += a[k - (j_lo:j_hi + 1)]

        # Best response via softmax with adaptive temperature
        T_br = max_b * 0.5
        if T_br < 1e-10:
            b_new = np.ones(n) / n
        else:
            b_new = np.exp(-cost / T_br)
            ss = b_new.sum()
            if ss < 1e-30:
                b_new = np.ones(n) / n
            else:
                b_new = b_new / ss

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

    return best_a, best_score

def local_search_polish(a0, time_limit, n_trials=5000, rng=None):
    """Local search: try random perturbations and keep improvements."""
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)
    trial = 0
    accepted = 0

    while time.monotonic() - t0 < time_limit and trial < n_trials:
        # Random perturbation: modify a small random subset of elements
        n_mod = rng.integers(max(1, n // 50), max(2, n // 10))
        idx = rng.choice(n, size=n_mod, replace=False)

        a_trial = a.copy()
        scale = rng.uniform(0.5, 1.5, size=n_mod)
        a_trial[idx] *= scale
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            trial += 1
            continue
        a_trial = a_trial / ss

        sc = score_normalized(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
            accepted += 1

        trial += 1

        if trial % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalSearch trial={trial} ({elapsed:.0f}s): best={best_score:.8f} accepted={accepted}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    """
    Main search function. Combines multiple optimization strategies:
    - L-BFGS-B with adaptive LSE temperature (primary)
    - Adam with LSE (complementary)
    - Nesterov soft best-response (diverse exploration)
    - Multi-restart with structured inits
    - Local search polishing
    """
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic()
            remain = global_deadline - elapsed
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    total_budget = budget_s - 30

    # === Phase 1: L-BFGS-B with LSE (main workhorse) - 35% ===
    print(f"\n=== Phase 1: L-BFGS-B LSE at n={n_base} (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t1 = min(0.35 * total_budget, remaining * 0.5)
    a_out, _ = lnbfgs_lse(best_seq, t1, T_start=50.0, T_min=0.05, cooldown=0.85)
    try_update(a_out, "lbfgs_lse_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Adam LSE - 15% ===
    print(f"\n=== Phase 2: Adam LSE (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.15 * total_budget, remaining * 0.4)
    if t2 > 15:
        a_out2, _ = adam_lse(best_seq, t2, T_start=30.0, T_min=0.05, lr=0.003)
        try_update(a_out2, "adam_lse")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Nesterov soft best-response - 15% ===
    print(f"\n=== Phase 3: Nesterov soft LR (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * total_budget, remaining * 0.4)
    if t3 > 15:
        a_out3, _ = nesterov_soft_lr(best_seq, t3, max_iters=50000)
        try_update(a_out3, "nesterov_soft")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Multi-restart with diverse inits - 20% ===
    print(f"\n=== Phase 4: Multi-restart (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4_total = min(0.20 * total_budget, remaining * 0.5)

    if t4_total > 30:
        n_restarts = 4
        t_per = t4_total / n_restarts
        for r in range(n_restarts):
            remaining = global_deadline - time.monotonic()
            if remaining < 15:
                break
            t_r = min(t_per, remaining * 0.5)
            if t_r < 10:
                break

            # Diverse initialization: blend structured init with current best
            init = structured_init(n_base, rng)
            blend = 0.3 * init + 0.7 * best_seq
            blend = np.maximum(blend, 0)
            ss = blend.sum()
            if ss > 1e-15:
                blend = blend / ss

            print(f"  Restart {r+1}/{n_restarts}, budget={t_r:.0f}s")
            a_r, _ = lnbfgs_lse(blend, t_r, T_start=80.0, T_min=0.1, cooldown=0.88)
            try_update(a_r, f"restart_{r+1}")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Local search polish - 10% ===
    print(f"\n=== Phase 5: Local search polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.10 * total_budget, remaining * 0.5)
    if t5 > 10:
        a_pol, sc_pol = local_search_polish(best_seq, t5, n_trials=20000, rng=rng)
        try_update(a_pol, "local_search")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final low-T LSE polish - 5% ===
    print(f"\n=== Phase 6: Final low-T LSE polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 10:
        a_fin, _ = lnbfgs_lse(best_seq, remaining * 0.9, T_start=5.0, T_min=0.02, cooldown=0.9)
        try_update(a_fin, "final_lowT")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
