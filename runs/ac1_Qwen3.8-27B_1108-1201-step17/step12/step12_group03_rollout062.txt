    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Fast local evaluation (no clipping to 1000, assumes valid input)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

    """Call the external evaluate_sequence safely."""
    try:

        return float(sc)
    except Exception:

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b_star, M_val)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

        # Nesterov update

        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:

            if a_s < 1e-15:

        a = a_new

    # Wait: d(1/s^2)/da_j = -2/s^3 (for the sum, d(s)/da_j = 1)
    # So df/da_j = 2n * [-2*smoothed_max/s^3 + grad_smoothed[j]/s^2]
    # = 2n/s^2 * grad_smoothed[j] - 4n*smoothed_max/s^3

    grad = 2.0 * n / (s * s) * grad_smoothed - 4.0 * n * smoothed_max / (s ** 3)

    return base, grad


def lse_smoothing_refine(a0, time_limit, beta_start=50.0, beta_end=500.0, n_total_iters=2000):
    """
    Refine using log-sum-exp smoothed objective with L-BFGS-B.
    Decreasing beta anneals toward the true max.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()

    # Use parameterization: a = softmax(z) to ensure a >= 0 and sum(a) = 1
    z = np.log(np.maximum(a, 1e-30))

    for phase in range(4):
        beta = beta_start * (beta_end / beta_start) ** (phase / 4.0)
        phase_time = time_limit / 4.0

        def obj_and_grad(z_flat):
            z_arr = z_flat.copy()
            z_arr -= z_arr.max()
            e = np.exp(z_arr)
            a_cur = e / e.sum()
            val, grad_a = logsumexp_smoothed_objective(a_cur, beta, n)

            # Convert gradient from a-space to z-space (softmax Jacobian)
            # da_j/dz_k = a_j*(delta_jk - a_k)
            # grad_z_k = sum_j grad_a_j * a_j * (delta_jk - a_k)
            # = a_k * (sum_j grad_a_j * a_j - (sum_j grad_a_j * a_j)) ...
            # = a_k * (sum_j grad_a_j * a_j) - a_k * sum_j grad_a_j * a_j
            # Actually: grad_z_k = sum_j grad_a_j * da_j/dz_k
            # = sum_j grad_a_j * a_j * (delta_jk - a_k)
            # = a_k * grad_a_k - a_k * sum_j grad_a_j * a_j
            # = a_k * (grad_a_k - sum_j grad_a_j * a_j)
            dot = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - dot)
            return val, grad_z

        # Run L-BFGS-B
        result = minimize(
            obj_and_grad, z, method='L-BFGS-B',
            options={'maxiter': n_total_iters // 4, 'ftol': 1e-10, 'gtol': 1e-8,
                     'maxfun': 5000}
        )

        z_new = result.x
        z_new -= z_new.max()
        e = np.exp(z_new)
        a_new = e / e.sum()

        score = eval_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            z = z_new.copy()
            print(f"    LSE phase {phase} (beta={beta:.1f}): improved to {best_score:.6f}")
        else:
            # Try mixing
            for frac in [0.3, 0.5, 0.7]:
                a_mix = (1 - frac) * best_a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                ms = a_mix.sum()
                if ms < 1e-15:
                    continue
                a_mix = a_mix / ms
                score_mix = eval_local(a_mix)
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    z = np.log(np.maximum(best_a, 1e-30))
                    print(f"    LSE phase {phase} mix {frac}: improved to {best_score:.6f}")
                    break

        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks_max=10):
    """
    Peak equalization: iteratively identify top peaks and solve QP to equalize them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()

    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Get top peaks
        n_peaks = min(n_peaks_max, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        # Only keep peaks that are close to max
        threshold = max_b * 0.95
        top_peaks = top_peaks[b[top_peaks] >= threshold]
        if len(top_peaks) < 2:
            break

        # Build QP: min M s.t. (a*a)_k <= M for k in top_peaks, sum(a)=1, a>=0
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iter=20000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False, max_iter=20000)
            except Exception:
                try:
                    prob.solve(solver=cp.PDLP, verbose=False)
                except Exception:
                    break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        score = eval_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            print(f"    PEQ round {round_num}: {best_score:.6f} (n_peaks={len(top_peaks)})")
        else:
            # Try mixing
            improved = False
            for frac in [0.2, 0.5, 0.8]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                ms = a_mix.sum()
                if ms < 1e-15:
                    continue
                a_mix = a_mix / ms
                score_mix = eval_local(a_mix)
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    improved = True
                    print(f"    PEQ round {round_num} mix {frac}: {best_score:.6f}")
                    break
            if not improved:
                # Widen peaks or stop
                if round_num >= 5:
                    break
                # Add one more peak
                n_peaks_max = min(n_peaks_max + 1, 15)
                continue

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_iter_per_peak=50):
    """
    Coordinate descent: for each coordinate, find the best value by 1D search.
    Focus on coordinates that contribute most to the peak lags.
    """
    n = len(a0)
    rng = np.random.default_rng(42)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        b = fast_autocorr(a)
        max_b = b.max()
        # Top peaks
        n_peaks = min(5, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        # Random coordinate order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Try moving mass from j to other positions (and vice versa)
            # Simple: try changing a[j] and redistributing to maintain sum
            # Or: try swapping mass between j and a random other coordinate
            k = rng.integers(0, n)
            if k == j:
                continue

            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = min(a[j], a[k]) * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[j] -= amt
                a_trial[k] += amt
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-15:
                    continue
                a_trial = a_trial / ts
                score = eval_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                if time.monotonic() - t0 > time_limit:
                    break

        if not improved:
            stall += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def ramp_init(n, p=1.0):
    """Create a monotonically increasing initialization: a_j = (j+1)^p / sum."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** p
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining LP Nesterov,
    log-sum-exp smoothing, peak equalization, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = find_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_local(
                np.maximum([float(max(0.0, min(1000.0, x))) for x in parent], 0)
            )
            print(f"Parent: n={len(parent)}, local_score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # If no parent, create initial sequence
    if best_seq is None:
        n_init = 1600
        best_seq = ramp_init(n_init, p=1.0)
        best_score = eval_local(best_seq)
        print(f"Default init (ramp): n={n_init}, score={best_score:.6f}")

    n_base = len(best_seq)

    # ============ PHASE 1: LP Nesterov (main workhorse) ============
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.50, 800.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t1, solver)
        print(f"  LP Nesterov done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")

    # ============ PHASE 2: Log-sum-exp smoothing ============
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LSE Smoothing at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, sc_out = lse_smoothing_refine(best_seq, t2, beta_start=30.0, beta_end=300.0)
        print(f"  LSE done: score={sc_out:.6f}")
        try_update(a_out, "lse_refine")

    # ============ PHASE 3: Peak equalization ============
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Peak Equalization (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_out, sc_out = peak_equalization(best_seq, t3, n_peaks_max=8)
        print(f"  PEQ done: score={sc_out:.6f}")
        try_update(a_out, "peq")

    # ============ PHASE 4: Second LP Nesterov round ============
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LP Nesterov Round 2 (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t4, solver)
        print(f"  LP Round 2: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "lp_round2")

    # ============ PHASE 5: Coordinate descent polishing ============
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Coordinate Descent (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.08, 80.0)
    if t5 > 10:
        a_out, sc_out = coordinate_descent_refine(best_seq, t5)
        print(f"  CD done: score={sc_out:.6f}")
        try_update(a_out, "coord_desc")

    # ============ PHASE 6: Multi-length exploration ============
    for n_try in [n_base + 100, n_base - 100, n_base + 200]:
        if n_try < 500:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 6: LP Nesterov at n={n_try} (remaining={remaining:.0f}s) ===")
        t6 = min(remaining * 0.15, 120.0)
        if t6 < 30:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp_optimize(init_try, t6, solver)
        print(f"  LP at n={n_try}: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_try}")

        # Quick peak equalization
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 40:
            t_peq = min(remaining * 0.05, 40.0)
            if t_peq > 10:
                a_out2, sc_out2 = peak_equalization(a_out, t_peq, n_peaks_max=6)
                try_update(a_out2, f"peq_n{n_try}")

    # ============ PHASE 7: Final LSE refinement ============
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LSE Refinement (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.3, 100.0)
    if t7 > 20:
        a_out, sc_out = lse_smoothing_refine(best_seq, t7, beta_start=50.0, beta_end=500.0)
        print(f"  Final LSE: score={sc_out:.6f}")
        try_update(a_out, "final_lse")

    # ============ PHASE 8: Final peak equalization ============
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t8 = min(remaining * 0.4, 60.0)
        if t8 > 15:
            a_out, sc_out = peak_equalization(best_seq, t8, n_peaks_max=10)
            print(f"  Final PEQ: score={sc_out:.6f}")
            try_update(a_out, "final_peq")

    # Final verification with the actual evaluator
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
