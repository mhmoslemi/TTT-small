"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Asymmetric exponential-decay initialization (exploits convolution asymmetry)
2. FFT-based mirror descent with adaptive all-peaks subgradient
3. LogSumExp-smoothed L-BFGS with multi-stage annealing
4. Aggressive peak shaving with sub-pixel line search

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full linear autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def cross_corr_fft(w, a):
    """
    Compute result[j] = sum_i w[j+i]*a[i] for j=0..n-1.
    w has length 2n-1, a has length n.
    Uses FFT.
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_obj_grad(z, T):
    """
    LogSumExp-smoothed objective and gradient in z-space (a = softmax(z)).
    Returns (objective, gradient).
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()
    cc = cross_corr_fft(w, a)
    grad_a = 2.0 * cc
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def mirror_descent_adaptive(a0, time_limit, eta0=0.04, decay_rate=0.6,
                            k_start=1, k_max=20, k_ramp=500,
                            threshold_mode='adaptive', seed=42, verbose=False):
    """
    Mirror descent with adaptive all-peaks-above-threshold subgradient.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        # Determine k (number of peaks to target)
        frac = min(1.0, t / float(k_ramp))
        k = max(k_start, min(k_max, int(k_start + frac * (k_max - k_start))))

        # Select peaks: top-k OR all above threshold
        if threshold_mode == 'adaptive':
            threshold = b_max - 0.1 * (b_max - b_mean)
            peak_mask = b >= threshold
            n_peaks = peak_mask.sum()
            if n_peaks < k:
                # Fall back to top-k
                top_idx = np.argsort(b)[::-1][:k]
                peak_indices = top_idx
            else:
                peak_indices = np.where(peak_mask)[0]
                if len(peak_indices) > k_max * 2:
                    # Cap the number of peaks
                    peak_sorted = peak_indices[np.argsort(b[peak_indices])[::-1]][:k_max * 2]
                    peak_indices = peak_sorted
        else:
            peak_indices = np.argsort(b)[::-1][:k]

        # Compute weights for each peak
        n_sel = len(peak_indices)
        if n_sel == 0:
            break
        excess = b[peak_indices] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(n_sel) / n_sel
        else:
            # Use a sharp weighting to focus on the very top
            temp = max(0.05, 1.0 / (n_sel + 1))
            excess_shift = excess / max(excess.max(), 1e-15)
            weights = np.exp(excess_shift / temp)
            weights = weights / weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for ii in range(n_sel):
            m = int(peak_indices[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]

        # Learning rate
        eta = eta0 / (1.0 + t / 500.0) ** decay_rate

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / max(s_new, 1e-15)
        else:
            a_new = a_new / s_new

        a = a_new

        # Periodic score check
        if t < 200:
            check_every = 5
        elif t < 2000:
            check_every = 15
        else:
            check_every = 40

        if t % check_every == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Stall recovery
        if stall > 800:
            # Structured perturbation: shift the "center of mass"
            noise = 0.02 * rng.standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall = 0

        if verbose and t % 5000 == 0:
            print(f"    MD t={t}: best={best_score:.6f}, n_peaks={n_sel}")

    return best_a, best_score


def lse_refine(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=15, seed=42):
    """LogSumExp L-BFGS refinement with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        iters = max(20, min(100, int(20 + 60 * si / max(1, n_stages - 1))))

        def func(z_in):
            val, _ = lse_obj_grad(z_in, T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad(z_in, T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score

def exp_decay_init(n, gamma, rng=None):
    """Exponential decay initialization: a[i] = exp(-gamma * i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * i / n)
    if rng is not None and rng.random() < 0.3:
        # Add small perturbation
        a += rng.uniform(0, 0.1) * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def step_init(n, step_pos, height_ratio, rng=None):
    """Step function: high before step_pos, low after."""
    a = np.ones(n)
    a[:int(step_pos * n)] = height_ratio
    if rng is not None:
        a += rng.uniform(0, 0.05) * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Asymmetric exponential-decay initialization (exploits linear convolution asymmetry)
2. FFT-based mirror descent with adaptive all-peaks subgradient
3. LogSumExp-smoothed L-BFGS with multi-stage annealing
4. Aggressive peak shaving with targeted mass transfer
5. Multi-start with diverse structured inits (exp-decay, step, ramp variants)
Key insight: linear autocorrelation is asymmetric, allowing scores < 2 via
concentrating mass at one end so center-peak pairs high with low values.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full linear autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def cross_corr_fft(w, a):
    """Compute result[j] = sum_i w[j+i]*a[i] for j=0..n-1 via FFT."""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_obj_grad(z, T):
    """LogSumExp-smoothed objective and gradient in z-space (a = softmax(z))."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    cc = cross_corr_fft(w, a)
    grad_a = 2.0 * cc
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_func_wrapper(z, T):
    """Wrapper for scipy minimize fun."""
    val, _ = lse_obj_grad(z, T)
    return val


def lse_jac_wrapper(z, T):
    """Wrapper for scipy minimize jac."""
    _, g = lse_obj_grad(z, T)
    return g


def mirror_descent_adaptive(a0, time_limit, eta0=0.04, decay_rate=0.6,
                            k_start=1, k_max=20, k_ramp=500,
                            threshold_mode='adaptive', seed=42, verbose=False):
    """Mirror descent with adaptive all-peaks-above-threshold subgradient."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        frac = min(1.0, t / float(k_ramp))
        k = max(k_start, min(k_max, int(k_start + frac * (k_max - k_start))))

        if threshold_mode == 'adaptive':
            threshold = b_max - 0.15 * (b_max - b_mean)
            peak_mask = b >= threshold
            n_peaks_available = int(peak_mask.sum())
            if n_peaks_available < k:
                top_idx = np.argsort(b)[::-1]
                peak_indices = top_idx[:k]
            else:
                peak_indices = np.where(peak_mask)[0]
                if len(peak_indices) > k_max * 3:
                    scores = b[peak_indices]
                    top_within = peak_indices[np.argsort(scores)[::-1]][:k_max * 3]
                    peak_indices = top_within
        else:
            peak_indices = np.argsort(b)[::-1][:k]

        n_sel = len(peak_indices)
        if n_sel == 0:
            break

        excess = b[peak_indices] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(n_sel) / n_sel
        else:
            temp = max(0.08, 0.8 / (n_sel + 1.0))
            excess_norm = excess / max(excess.max(), 1e-15)
            weights = np.exp(excess_norm / temp)
            weights = weights / weights.sum()

        grad = np.zeros(n)
        for ii in range(n_sel):
            m = int(peak_indices[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]

        eta = eta0 / (1.0 + t / 600.0) ** decay_rate
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / max(s_new, 1e-15)
        else:
            a_new = a_new / s_new
        a = a_new

        if t < 300:
            check_every = 5
        elif t < 3000:
            check_every = 15
        else:
            check_every = 40

        if t % check_every == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 1000:
            noise = 0.02 * rng.standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall = 0

        if verbose and t % 10000 == 0:
            print(f"    MD t={t}: best={best_score:.6f}, n_peaks={n_sel}, eta={eta:.5f}")

    return best_a, best_score


def lse_refine(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=15, seed=42):
    """LogSumExp L-BFGS refinement with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        T_val = temps[si]
        iters = max(20, min(120, int(20 + 80 * si / max(1, n_stages - 1))))

        try:
            res = minimize(fun=lse_func_wrapper, x0=z, jac=lse_jac_wrapper,
                           args=(T_val,), method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 400,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_shaving_aggressive(a0, time_limit, seed=123, verbose=False):
    """Aggressive peak shaving: shift mass from peak contributors to low regions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]

        if contrib.sum() < 1e-15:
            break

        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])

        peak_region = np.zeros(n, dtype=bool)
        top5 = np.argsort(b)[::-1][:5]
        for pk in top5:
            pk = int(pk)
            s2 = max(0, pk - n + 1)
            e2 = min(n - 1, pk)
            peak_region[s2:e2 + 1] = True

        candidates = np.where(~peak_region)[0]
        if len(candidates) < 3:
            candidates = np.where(np.arange(n) != src)[0]

        if len(candidates) > 0:
            n_pick = max(5, len(candidates) // 5)
            bottom = candidates[np.argsort(a[candidates])[:n_pick]]
            dst = int(bottom[rng.integers(len(bottom))])

            improved = False
            for frac in [0.001, 0.003, 0.006, 0.01, 0.02, 0.04, 0.08]:
                amt = a[src] * frac
                if amt < 1e-15 or a[src] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if not improved:
                if rng.integers(4) == 0:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.002, 0.015)
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        if a_trial[i] >= 0:
                            a_trial = a_trial / a_trial.sum()
                            sc = fast_score(a_trial)
                            if sc < best_score:
                                a = a_trial.copy()
                                best_score = sc
                                best_a = a.copy()
                            else:
                                stall += 1
                else:
                    stall += 1
        else:
            stall += 1

        if stall > 800:
            break

    return best_a, best_score


def exp_decay_init(n, gamma, rng=None):
    """Exponential decay: a[i] = exp(-gamma * i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    denom = max(n - 1, 1)
    a = np.exp(-gamma * i / denom)
    if rng is not None and rng.random() < 0.3:
        a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    return f_interp

def fast_autocorr(a):
    """Full linear autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def cross_corr_fft(w, a):
    """Compute result[j] = sum_i w[j+i]*a[i] for j=0..n-1 via FFT."""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_obj_grad(z, T):
    """LogSumExp-smoothed objective and gradient in z-space (a = softmax(z))."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    cc = cross_corr_fft(w, a)
    grad_a = 2.0 * cc
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z
