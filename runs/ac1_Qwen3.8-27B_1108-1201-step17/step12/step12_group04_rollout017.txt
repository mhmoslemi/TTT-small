"""

2. Incremental coordinate descent with first-order peak tracking

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    Uses FFT for O(n log n).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    a_rev = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_objective_val(a, T):
    """Compute the LSE objective value (approximation of max)."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (max_Tb + np.log(exp_Tb.sum())) / T


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def logsumexp_lbfgs_optimize(a0, time_limit, T_start, T_end, n_stages, max_iter_per_stage=100):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage temperature annealing.
    Uses FFT-based gradients for O(n log n) per iteration.
    Temperature increases from T_start to T_end (sharper max approximation).
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-300))
    t0 = time.monotonic()

    if T_start > T_end:
        temps = np.geomspace(T_start, T_end, n_stages)
    else:
        temps = np.geomspace(T_start, T_end, n_stages)

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = min(max_iter_per_stage, max(20, int(20 * remaining / max(1.0, (n_stages - si) * 2.0))))

        def func_grad(z_in):
            a = softmax_fn(z_in)
            n_local = len(a)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            obj_val = (max_Tb + np.log(exp_Tb.sum()))
            # Gradient w.r.t. z
            grad_a, w = compute_lse_grad(a, stage_T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            return obj_val, grad_z

        def func_only(z_in):
            val, _ = func_grad(z_in)
            return val

        def jac_only(z_in):
            _, g = func_grad(z_in)
            return g

        try:
            res = minimize(
                fun=func_only, x0=z, jac=jac_only, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxfun': iters * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si}/{n_stages} T={stage_T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def incremental_coord_descent(a0, time_limit, seed=42, n_peaks=5, n_trials_per_peak=20):

    Coordinate descent with first-order incremental convolution updates.
    Focuses on reducing the top peaks by moving mass from peak-contributing
    positions to positions that minimize impact on all peaks.
    """

    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_autocorr(a)
    b_max = b.max()
    best_a = a.copy()
    best_bmax = b_max
    best_score = 2.0 * n * b_max
    t0 = time.monotonic()
    stall = 0
    iter_count = 0

    # Precompute peak positions
    peak_indices = np.argsort(b)[::-1][:n_peaks]

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Get current top peaks
        if iter_count % 50 == 0:
            peak_indices = np.argsort(b)[::-1][:n_peaks]
            b_max = b.max()

        improved = False

        # Try to reduce each top peak
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                    break

            pk = int(pk)
            # Find contributing pairs: i + j = pk, both in [0, n-1]
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[pk - idx]
            total_contrib = contrib.sum()

            if total_contrib < 1e-15:
                continue

            # Find top contributors
            top_c = np.argsort(contrib)[::-1][:5]

            for tc in top_c:
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-12:

                # Try moving a small fraction to various targets
                for trial in range(min(n_trials_per_peak, 20)):
                    # Choose target: prefer positions far from all peak-contributing positions
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue

                            continue

                        a_trial[src] -= delta
                        a_trial[dst] += delta
                        a_trial = np.maximum(a_trial, 0)
                        s_trial = a_trial.sum()
                        if s_trial < 1e-15:
                            continue
                        a_trial = a_trial / s_trial

                        b_trial = fast_autocorr(a_trial)
                        bm_trial = b_trial.max()

                        if bm_trial < best_bmax - 1e-14:
                            a = a_trial
                            b = b_trial
                            best_bmax = bm_trial
                            best_score = 2.0 * n * bm_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1

            # Random exploration when stuck

                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))

                        b_trial = fast_autocorr(a_trial)
                        bm_trial = b_trial.max()
                        if bm_trial < best_bmax:
                            a = a_trial
                            b = b_trial
                            best_bmax = bm_trial
                            best_score = 2.0 * n * bm_trial

            if stall > 500:

        if iter_count % 500 == 0:

                # Move from src to dst: delta_b[k] = 2*delta*(a[k-dst] - a[k-src])
                # We want delta_b[k] <= 0 for all peak k
                # i.e., a[k-dst] <= a[k-src] for all peak k

                # Precompute a[k-src] and a[k-dst] for all peaks
                def dest_score(dst):
                    pen = 0.0
                    for pk2 in peak_set[:10]:
                        pk2 = int(pk2)
                        val_src = a[pk2 - src] if 0 <= pk2 - src < n else 0.0
                        val_dst = a[pk2 - dst] if 0 <= pk2 - dst < n else 0.0
                        if val_dst > val_src:
                            pen += (val_dst - val_src)
                    return pen

                # Find best destination
                # Sample a subset of destinations for efficiency
                sample_size = min(100, n)
                candidates = rng.choice(n, size=sample_size, replace=False)
                scores = np.array([dest_score(int(d)) for d in candidates])
                best_dst_idx = int(np.argmin(scores))
                best_dst = int(candidates[best_dst_idx])
                best_score_dst = scores[best_dst_idx]

                if best_score_dst < 0 or (best_score_dst == 0 and best_dst != src):
                    # Good destination found!
                    for frac in [0.02, 0.01, 0.005, 0.002, 0.001]:
                        delta = a[src] * frac
                        if delta < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= delta
                        a_trial[best_dst] += delta
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        b_trial = fast_autocorr(a_trial)
                        bm = b_trial.max()
                        if bm < best_bmax - 1e-14:
                            a = a_trial
                            best_bmax = bm
                            best_score = 2.0 * n * bm
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

                    if improved:
                        break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 100:
                # Random perturbation to escape
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.05)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    b_trial = fast_autocorr(a_trial)
                    if b_trial.max() < best_bmax:
                        a = a_trial
                        best_bmax = b_trial.max()
                        best_score = 2.0 * n * best_bmax
                        best_a = a.copy()
                        stall = 0
                if stall > 300:
                    break

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PFR it={it} stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic-like method."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.arange(old_n) / old_n
    new_x = np.arange(new_n) / new_n
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations (asymmetric!)."""
    choice = rng.integers(0, 10)
    if choice == 0:
        # Geometric decay (asymmetric)
        r = rng.uniform(0.95, 0.999)
        x = np.arange(n)
        a = r ** x
    elif choice == 1:
        # Exponential decay
        rate = rng.uniform(2, 8)
        x = np.linspace(0, 1, n)
        a = np.exp(-rate * x)
    elif choice == 2:
        # One-sided bump
        center = rng.uniform(0.1, 0.4)
        width = rng.uniform(0.1, 0.3)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - center) / width) ** 2)
    elif choice == 3:
        # Cosine window (tapered at both ends, but asymmetric)
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2 * np.exp(-0.5 * x)
    elif choice == 4:
        # Power law decay
        p = rng.uniform(1.5, 4.0)
        x = np.arange(1, n + 1)
        a = 1.0 / x ** p
    elif choice == 5:
        # Two-bump asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.2) ** 2) + 0.5 * np.exp(-20 * (x - 0.6) ** 2)
    elif choice == 6:
        # Sine-modulated decay
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x) * (0.5 + 0.5 * np.cos(8 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 7:
        # Polynomial taper
        x = np.linspace(0, 1, n)
        a = (1 - x) ** rng.uniform(1, 3)
    elif choice == 8:
        # Bell with skew
        x = np.linspace(0, 1, n)
        mu = rng.uniform(0.2, 0.5)
        sig = rng.uniform(0.1, 0.25)
        skew = rng.uniform(0.5, 3.0)
        a = np.exp(-((x - mu) / sig) ** 2 * (1 + skew * (x - mu)))
    else:
        # Random smooth
        n_modes = rng.integers(2, 6)
        a = np.ones(n)
        for _ in range(n_modes):
            freq = rng.uniform(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.5)
            x = np.linspace(0, 1, n)
            a += amp * np.abs(np.sin(freq * np.pi * x + phase))
        a = np.maximum(a, 0)

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fine_polish(a0, time_limit):
    """
    Very fine coordinate descent for final polishing.
    Uses extremely small steps and evaluates many candidates per iteration.
    """
    n = len(a0)
    rng = np.random.default_rng(999)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_autocorr(a)
    best_a = a.copy()
    best_bmax = b.max()
    best_score = 2.0 * n * best_bmax
    t0 = time.monotonic()
    stall = 0

    # Find top peaks
    top_peaks = np.argsort(b)[::-1][:8]

    while time.monotonic() - t0 < time_limit:
        if stall > 1000:
            break

        improved = False
        # Refresh peak list every 100 iterations
        if (stall // 100) > 0 and stall % 200 == 0:
            top_peaks = np.argsort(b)[::-1][:8]

        for pk in top_peaks:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[pk - idx]
            top_c = np.argsort(contrib)[::-1][:3]

            for tc in top_c:
                src = int(idx[tc])
                if a[src] < 1e-14:
                    continue

                # Try moving very small amounts
                for frac in [0.0005, 0.001, 0.002, 0.003]:
                    delta = a[src] * frac
                    if delta < 1e-16:
                        continue

                    # Try several destinations
                    # Prefer positions not in any peak's contributing range
                    in_peak = np.zeros(n, dtype=bool)
                    for pk2 in top_peaks:
                        pk2 = int(pk2)
                        js2 = max(0, pk2 - n + 1)
                        je2 = min(n - 1, pk2)
                        in_peak[js2:je2 + 1] = True
                    in_peak[src] = False

                    free_positions = np.where(~in_peak)[0]
                    if len(free_positions) > 0:
                        dst_candidates = free_positions[rng.choice(len(free_positions), size=min(10, len(free_positions)), replace=False)]
                    else:
                        dst_candidates = rng.choice(n, size=10, replace=False)
                        dst_candidates = dst_candidates[dst_candidates != src]

                    for dst in dst_candidates:
                        dst = int(dst)
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= delta
                        a_trial[dst] += delta
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        b_trial = fast_autocorr(a_trial)
                        bm = b_trial.max()
                        if bm < best_bmax - 1e-15:
                            a = a_trial
                            b = b_trial
                            best_bmax = bm
                            best_score = 2.0 * n * bm
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Tiny random perturbation
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * rng.uniform(0.0001, 0.002)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    b_trial = fast_autocorr(a_trial)
                    if b_trial.max() < best_bmax:
                        a = a_trial
                        b = b_trial
                        best_bmax = b_trial.max()
                        best_score = 2.0 * n * best_bmax
                        best_a = a.copy()
                        stall = 0

        if stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"      FP stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mixed_method_optimize(a0, time_limit, solver=None):
    """
    Combined optimization: alternate between LogSumExp gradient steps
    and coordinate descent, with periodic restarts.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Phase A: LogSumExp optimization
    t_a = min(time_limit * 0.4, 200.0)
    a_out, _ = logsumexp_lbfgs_optimize(a, t_a, T_start=1000.0, T_end=100000.0, n_stages=15, max_iter_per_stage=80)
    sc = 2.0 * n * fast_autocorr(a_out).max()
    if sc < best_score:
        best_score = sc
        best_a = a_out.copy()
        a = a_out.copy()
    elapsed = time.monotonic() - t0
    print(f"      MM Phase A ({elapsed:.0f}s): score={best_score:.8f}")

    # Phase B: Coordinate descent
    remaining = time_limit - (time.monotonic() - t0)
    t_b = min(remaining * 0.35, 200.0)
    a_out, _ = incremental_coord_descent(a, t_b, seed=77)
    sc = 2.0 * n * fast_autocorr(a_out).max()
    if sc < best_score:
        best_score = sc
        best_a = a_out.copy()
        a = a_out.copy()
    elapsed = time.monotonic() - t0
    print(f"      MM Phase B ({elapsed:.0f}s): score={best_score:.8f}")

    # Phase C: More LogSumExp at lower T
    remaining = time_limit - (time.monotonic() - t0)
    t_c = min(remaining * 0.25, 150.0)
    a_out, _ = logsumexp_lbfgs_optimize(a, t_c, T_start=50000.0, T_end=300000.0, n_stages=8, max_iter_per_stage=100)
    sc = 2.0 * n * fast_autocorr(a_out).max()
    if sc < best_score:
        best_score = sc
        best_a = a_out.copy()
        a = a_out.copy()
    elapsed = time.monotonic() - t0
    print(f"      MM Phase C ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

        if np.isfinite(sc_eval) and sc_eval < best_score:

            elapsed = time.monotonic() - t_global_start

            # Keep tracking even if evaluate_safe gives slightly different result
