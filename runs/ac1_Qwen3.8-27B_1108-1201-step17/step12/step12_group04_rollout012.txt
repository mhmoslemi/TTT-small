import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def compute_grad_at_peaks(a, top_k_indices):
    """
    Compute the subgradient of max_k (a*a)_k at a.
    If top_k_indices are the indices achieving (or near) the max,
    the subgradient is the average of the individual gradients.
    grad_i = (1/|S|) * sum_{k in S} 2*a[k-i] (for valid indices)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in top_k_indices:
        # (a*a)_k = sum_i a_i * a[k-i], grad wrt a_j = 2*a[k-j] (if 0 <= k-j < n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(top_k_indices)
    return grad
```

In log-space: $\log|A|_{\text{new}} = (1-\beta)\log|A| + \beta \log(\text{const})$

This is a geometric mean that pulls the spectrum toward flat.

    new_mag = np.power(np.maximum(mag, 1e-30), 1.0 - alpha) * np.power(mean_mag, alpha)
    phase = A / np.maximum(mag, 1e-30)
    A_new = new_mag * phase
    a_new = np.fft.ifft(A_new).real
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
```

Now, let me think about the multiplicative update:

The idea: $a_i \leftarrow a_i \cdot \exp(-\eta \cdot g_i)$ where $g_i$ is the $i$-th component of the subgradient.

This is like a mirror descent step with the negative entropy as the mirror map. The step size $\eta$ should be small (e.g., $\eta = c / \max|g|$).

```python
def multiplicative_step(a, grad, eta=None):
    if eta is None:
        max_g = np.abs(grad).max()
        eta = 0.1 / max(max_g, 1e-30)
    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s
```

def optimize_fast(a0, time_limit, n_top=5):

Hmm, this has a problem: the score computation calls `fast_autocorr` twice per iteration (once in `compute_grad_peaks` and once in the score check). Let me optimize this by reusing the convolution.

Also, I realize that the BB step size formula $\eta = s^T s / s^T y$ can give very large values if $s^T y$ is small. I should clip it.

Let me also add a "restart" mechanism: if the score doesn't improve for a long time, perturb the sequence randomly.

def optimize_combined(a0, time_limit, n_top=5, seed=42):

    eta = 0.3
    stall = 0
    it = 0

        it += 1
        b = fast_autocorr(a)
        peak_val = b.max()
        idx = np.argsort(b)[::-1][:n_top]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je+1] += 2.0 * a[k-je : k-js+1][::-1]

            ss = np.dot(s_k, s_k)

                eta_bb = ss / sy
                eta = 0.7 * eta + 0.3 * np.clip(eta_bb, 0.01, 2.0)

        a_old = a.copy()
        if it % 100 == 0:
            a = ift_step(a, alpha=0.2 + 0.3 * (it % 1000) / 1000.0)
        elif it % 50 == 0:

        else:
            a = project_simplex(a_old - eta * grad)

        b_new = fast_autocorr(a)

            best_a = a.copy()

        elif score_new < 2.0 * n * peak_val:
            stall = 0  # Progress (not best yet)

            stall += 1

        g_prev = grad.copy()

```python
def local_search(a0, time_limit, n_top_peaks=5, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:n_top_peaks]
        
        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # Contribution of each pair (i, k-i) to peak k
            i_vals = np.arange(js, je + 1)
            contrib = a[i_vals] * a[k - i_vals]
            # Find the largest contributor
            max_contrib_idx = np.argmax(contrib)
            src = i_vals[max_contrib_idx]
            src_partner = k - src
            
            # Try moving mass from src (or src_partner) to a random position
            # that doesn't contribute to the top peaks
            peak_positions = set()
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_positions.update(range(js2, je2 + 1))
            
            # Candidate destinations: positions NOT in any peak area
            candidates = [j for j in range(n) if j not in peak_positions]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Pick a few random candidates
            n_try = min(5, len(candidates))
            for _ in range(n_try):
                dst = candidates[rng.integers(len(candidates))]
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial /= ss
                    b_trial = fast_autocorr(a_trial)
                    sc = 2.0 * n * b_trial.max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
        
        if (stall // 50) > 0 and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LS stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
```

Now for the multi-length exploration:

```python
def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new /= ss
    return a_new
```
