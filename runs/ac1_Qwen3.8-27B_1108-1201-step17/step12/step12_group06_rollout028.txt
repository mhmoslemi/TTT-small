from scipy.fft import rfft, irfft, fft, ifft
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    N = 1 << (size-1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]

def lse_obj_grad(a, T):
    """Compute LSE objective and gradient.
    Returns (obj, grad) where obj approximates max(conv(a,a)) and grad is w.r.t. a.
    """
    n = len(a)
    conv = fast_convolve(a)
    Tconv = T * conv
    max_Tconv = Tconv.max()
    exp_Tconv = np.exp(Tconv - max_Tconv)
    sum_exp = exp_Tconv.sum()
    w = exp_Tconv / sum_exp  # weights
    
    # LSE value (approximation to max)
    lse_val = max_Tconv / T + np.log(sum_exp) / T  # = (max + log(sum(exp(T*conv - max))))/T
    
    # Gradient: grad_a[j] = T * sum_k w[k] * a[k-j]
    # Using FFT cross-correlation
    N = 1 << (2*n-1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    cross = ifft(W * np.conj(A)).real[:n]
    
    grad = T * cross
    return lse_val, grad

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def fast_convolve(a):
    """Fast linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    if a.sum() < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_lse_grad(a, T):
    """Compute LSE objective value and gradient w.r.t. a.
    
    Returns (lse_value, grad) where lse_value ≈ max(conv(a,a))
    and grad[j] = T * sum_k w[k] * a[k-j].
    """
    n = len(a)
    conv = fast_convolve(a)
    Tconv = T * conv
    max_Tconv = Tconv.max()
    exp_Tconv = np.exp(Tconv - max_Tconv)
    sum_exp = exp_Tconv.sum()
    w = exp_Tconv / sum_exp
    lse_val = (max_Tconv + np.log(sum_exp)) / T
    
    # Cross-correlation: cross[j] = sum_k w[k] * a[k-j] for j=0..n-1
    N = 1 << (2 * n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    cross = ifft(W * np.conj(A)).real[:n]
    
    grad = T * cross
    return lse_val, grad, w, conv

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    while time.monotonic() - t0 < time_budget:

        lse_val, grad, w, conv = compute_lse_grad(a, T)
        
        # For simplex constraint: remove the mean component
        # (this ensures we stay on the simplex to first order)
        grad = grad - np.dot(grad, a)  # since sum(a)=1
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Check score

            stagnant = 0
        else:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

_lp_solver = None


def get_lp_solver():

    if _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:

        except Exception:

def fast_convolve(a):
    """Fast linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:

        if len(a_list) == 0:

        return float(evaluate_sequence(a_list))
    except Exception:

def softmax_stable(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_lse_grad(a, T):
    """Compute LSE objective and gradient w.r.t. a using FFT.
    
    Returns (lse_value, grad) where lse_value approximates max(conv(a,a))
    and grad[j] = T * sum_k w[k] * a[k-j] (cross-correlation).
    """
    n = len(a)
    conv = fast_convolve(a)
    Tconv = T * conv
    max_Tconv = Tconv.max()
    exp_Tconv = np.exp(Tconv - max_Tconv)
    sum_exp = exp_Tconv.sum()
    w = exp_Tconv / sum_exp
    lse_val = (max_Tconv + np.log(sum_exp)) / T

    # Cross-correlation via FFT: cross[j] = sum_k w[k] * a[k-j]
    N = 1 << (2 * n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    cross = ifft(W * np.conj(A)).real[:n]

    grad = T * cross
    return lse_val, grad, w, conv


def adam_lse_optimize(a_init, time_budget, T_start=40.0, T_max=300.0,
                      T_increase=1.25, lr=None, beta1=0.9, beta2=0.999,
                      eps=1e-8, T_patience=300, verbose=False):
    """Adam-based LSE optimization on the probability simplex with progressive temperature."""
    n = len(a_init)
    if lr is None:
        lr = 1.5 / n

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    T = T_start
    stagnant = 0
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)
    last_print = t0

    while time.monotonic() - t0 < time_budget:
        t += 1

        lse_val, grad, w, conv = compute_lse_grad(a, T)

        # Project gradient to tangent space of simplex (sum = 0)
        grad = grad - np.mean(grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        sc = score_normalized(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new

        # Adaptive temperature: increase when stagnant
        if stagnant >= T_patience:
            if T < T_max:
                T = min(T_max, T * T_increase)
                # Reset Adam state partially when T changes (avoid stale momentum)
                m *= 0.5
                v *= 0.5
            stagnant = 0

        # Also slowly increase T even when not stagnant
        if t % 10000 == 0 and T < T_max:
            T = min(T_max, T * 1.05)

        if verbose and (t % 5000 == 0 or time.monotonic() - last_print > 30):
            elapsed = time.monotonic() - t0
            print(f"        Adam-LSE iter={t} T={T:.1f}: cur={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")
            last_print = time.monotonic()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):

    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    k = np.arange(n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:

    elif choice == 2:

    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(10 * np.pi * x)
    elif choice == 4:
        a = 1.0 + 0.15 * np.cos(6 * np.pi * k / n) + 0.1 * np.cos(14 * np.pi * k / n)
    elif choice == 5:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.5
    elif choice == 6:

def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=100000):
    """Nesterov-accelerated LP best-response for final polishing."""
    import cvxpy as cp
    n = len(a0)
    if n > 2500:
        # Too large for efficient LP matrix construction
        return a0, score_normalized(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Build convolution matrix for current y
        size = 2 * n - 1
        T_mat = np.zeros((size, n))
        for j in range(n):
            T_mat[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        sb = b_new.sum()
        if sb < 1e-15:
            continue
        b_new = b_new / sb

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"        NesterovLP iter={it+1}: best={best_score:.8f} ({elapsed:.0f}s)")
                last_print = elapsed

    return best_a, best_score


def peak_focused_gradient(a, top_k=10):
    """Compute gradient focusing only on the top-k peaks of the convolution."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Find top k peaks
    top_indices = np.argsort(conv)[-top_k:]
    top_indices = np.sort(top_indices)
    
    # Build sparse weight vector
    w = np.zeros(2 * n - 1)
    for k in top_indices:
        w[k] = 1.0 / len(top_indices)
    
    # Cross-correlation
    N = 1 << (2 * n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    cross = ifft(W * np.conj(A)).real[:n]
    
    return cross, conv

    """Adam optimization using peak-focused gradient (exact max, no LSE approximation)."""

                                         lr=4.0/len(best_seq), verbose=False)
            try_update(a_p2, "polish_peak3")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Random restarts
            n_restarts = 3
            t4c = remaining * 0.5 / n_restarts
            for r in range(n_restarts):
                if time.monotonic() > global_deadline:
                    break
                noise = 0.05 * rng.normal(0, 1, len(best_seq))
                perturbed = np.maximum(best_seq + noise, 0)
                sp = perturbed.sum()
                if sp > 1e-10:
                    perturbed = perturbed / sp
                    a_r, _ = adam_lse_optimize(perturbed, t4c, T_start=80.0, T_max=200.0,
                                                lr=2.0/len(best_seq), T_patience=50, verbose=False)
                    try_update(a_r, f"restart_{r}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
