    """Build basis matrix Phi of shape (n, P)."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1) if n > 1 else np.zeros(1)
    cols = [np.ones(n)]
    for k in range(1, n_harm + 1):
        cols.append(np.cos(2 * np.pi * k * x))
        cols.append(np.sin(2 * np.pi * k * x))
    # Edge terms
    cols.append(x)
    cols.append(1 - x)
    cols.append(x * x)
    cols.append((1 - x) ** 2)
    cols.append(x * (1 - x))
    # Gaussians
    for c in [0.25, 0.5, 0.75]:
        cols.append(np.exp(-50 * (x - c) ** 2))
    Phi = np.column_stack(cols)
    return Phi

1. Low-dimensional basis function parameterization (Fourier + edge + Gaussian) with L-BFGS
2. Full-space LSE continuation optimization with adaptive temperature
3. Simulated annealing with subgradient (multiscale: global shape + local refinement)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)

def cross_corr(w, a):

    # Or: = conv(a_rev, w)[n-1+k] where a_rev = a[::-1]

def lse_obj_and_grad(a, T):
    """
    Compute LSE objective and gradient w.r.t. a.
    Returns (obj, grad_a) where grad_a has same shape as a.
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    # grad_a[k] = 2 * sum_j w[j] * a[j-k] (valid) = 2 * cross_corr(w, a)[k]
    cc = cross_corr(w, a)
    grad_a = 2.0 * cc
    return obj, grad_a


def build_basis(n, n_harm=20, n_gauss=5):
    """Build basis matrix Phi of shape (n, P) for parameterization."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(1, n - 1)
    cols = [np.ones(n)]
    for k in range(1, n_harm + 1):
        cols.append(np.cos(2 * np.pi * k * x))
        cols.append(np.sin(2 * np.pi * k * x))
    # Edge tapering terms
    cols.append(x)
    cols.append(1.0 - x)
    cols.append(x * x)
    cols.append((1.0 - x) ** 2)
    cols.append(4.0 * x * (1.0 - x))
    # Gaussian bumps at various positions
    for c in [0.2, 0.35, 0.5, 0.65, 0.8]:
        cols.append(np.exp(-80.0 * (x - c) ** 2))
    # Wider Gaussians
    for c in [0.3, 0.7]:
        cols.append(np.exp(-15.0 * (x - c) ** 2))
    Phi = np.column_stack(cols)
    return Phi


def basis_optimize(n, time_limit, n_harm=20, seed=42, T=50.0):
    """Optimize over basis coefficients using L-BFGS with LSE objective."""
    Phi = build_basis(n, n_harm)
    P = Phi.shape[1]
    rng = np.random.default_rng(seed)
    
    # Precompute Phi.T @ a for gradient
    t0 = time.monotonic()
    
    def obj_grad(c):
        z = Phi @ c
        z_shift = z - z.max()
        e = np.exp(z_shift)
        a = e / e.sum()
        
        b = fast_conv(a)
        Tb = T * b
        Tb_shift = Tb - Tb.max()
        exp_Tb = np.exp(Tb_shift)
        obj = (Tb.max() + np.log(exp_Tb.sum())) / T
        w = exp_Tb / exp_Tb.sum()
        
        cc = cross_corr(w, a)
        grad_a = 2.0 * cc
        S = np.dot(grad_a, a)
        grad_c = Phi.T @ (a * grad_a) - S * (Phi.T @ a)
        
        return obj, grad_c
    
    # Try multiple random initializations
    best_a = None
    best_score = float('inf')
    
    n_attempts = min(5, max(2, int(time_limit / 40)))
    attempt_time = time_limit / n_attempts
    
    for attempt in range(n_attempts):
        c0 = rng.standard_normal(P) * 0.3
        c0[0] = 0.0  # No DC offset initially
        
        try:
            res = minimize(fun=obj_grad, x0=c0, jac=True, method='L-BFGS-B',
                          options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxfun': 2000, 'disp': False})
            c_opt = res.x
            z = Phi @ c_opt
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_score


def lse_full_optimize(a0, time_limit, T_stages=None, seed=42):
    """Full-space LSE optimization with multi-stage annealing using L-BFGS."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if T_stages is None:
        T_stages = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_score(a)
    
    # Work in log-space: a = softmax(z), z = log(a) initially
    z = np.log(np.maximum(a, 1e-30))
    
    for si, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        iters = max(20, min(100, int(30 + 50 * si / len(T_stages))))
        
        def func_jac(z_in):
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a_loc = e / e.sum()
            
            b = fast_conv(a_loc)
            Tb = stage_T * b
            Tb_shift = Tb - Tb.max()
            exp_Tb = np.exp(Tb_shift)
            obj = (Tb.max() + np.log(exp_Tb.sum())) / stage_T
            w = exp_Tb / exp_Tb.sum()
            
            cc = cross_corr(w, a_loc)
            grad_a = 2.0 * cc
            S = np.dot(grad_a, a_loc)
            grad_z = a_loc * (grad_a - S)
            
            return obj, grad_z
        
        try:
            res = minimize(fun=func_jac, x0=z, jac=True, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10,
                                   'maxfun': iters * 50, 'disp': False})
            z = res.x
            z_shift = z - z.max()
            a_new = np.exp(z_shift)
            a_new = a_new / a_new.sum()
            sc = fast_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
        except Exception:
            pass
        
        elapsed = time.monotonic() - t0
        if si % 2 == 0:
            print(f"    LSE stage {si+1}/{len(T_stages)} T={stage_T:.0f} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def sa_subgradient(a0, time_limit, seed=42, T_init=1.0, T_min=0.001, cooling=0.9999):
    """
    Simulated annealing with subgradient method.
    Alternates between:
    - Global moves: perturb many coordinates simultaneously
    - Local moves: adjust a few coordinates
    Uses LSE smooth approximation for gradient.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    current_T = T_init
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Compute LSE gradient
        T_ls = max(20.0, 50.0 * current_T + 10.0)
        b = fast_conv(a)
        Tb = T_ls * b
        Tb_shift = Tb - Tb.max()
        exp_Tb = np.exp(Tb_shift)
        w = exp_Tb / exp_Tb.sum()
        
        cc = cross_corr(w, a)
        grad = 2.0 * cc
        
        # Choose move type
        r = rng.random()
        if r < 0.3:
            # Global move: subtract scaled gradient from all (with noise)
            step = 0.02 * current_T + 0.001
            noise = rng.standard_normal(n) * 0.01 * current_T
            a_trial = a - step * grad + noise
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
        elif r < 0.6:
            # Local move: adjust top-10 gradient positions
            top_k = int(min(10, n * 0.01))
            idx = np.argsort(-np.abs(grad))[:top_k]
            a_trial = a.copy()
            for i in idx:
                delta = -0.05 * current_T * grad[i] + rng.standard_normal() * 0.02 * current_T
                a_trial[i] = max(0, a[i] + delta)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
        elif r < 0.8:
            # Mass transfer: move mass from high-grad to low-grad positions
            top_pos = int(np.argmax(grad))
            bot_candidates = np.argsort(a)[:n // 4]
            bot_pos = int(bot_candidates[rng.integers(len(bot_candidates))])
            frac = rng.uniform(0.005, 0.05)
            a_trial = a.copy()
            amt = a[top_pos] * frac
            a_trial[top_pos] -= amt
            a_trial[bot_pos] += amt
            if a_trial[top_pos] < 0:
                continue
            a_trial = a_trial / a_trial.sum()
        else:
            # Random exploration
            n_perturb = int(rng.integers(5, min(50, n // 10)))
            idx = rng.integers(0, n, size=n_perturb)
            a_trial = a.copy()
            for i in idx:
                a_trial[i] = max(0, a[i] * (1 + rng.standard_normal() * 0.3))
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
        
        sc_trial = fast_score(a_trial)
        sc_current = fast_score(a)
        
        # Accept/reject
        if sc_trial < sc_current or rng.random() < np.exp(-(sc_trial - sc_current) / max(current_T, 1e-8)):
            a = a_trial
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a.copy()
        
        # Cooling
        current_T = max(T_min, current_T * cooling)
        
        # Periodic check
        if iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={iter_count} ({elapsed:.0f}s): T={current_T:.4f}, best={best_score:.6f}")
    
    return best_a, best_score


def coordinate_polish(a0, time_limit, seed=42):
    """
    Coordinate descent: for each coordinate, find the optimal value
    (given others fixed) that minimizes the LSE objective.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    # Precompute: for each k, conv[k] = sum_i a[i]*a[k-i]
    # If we change a[j] to a[j]+delta:
    #   conv[k] changes by delta * (a[k-j] + a[k-j] if valid) = 2*delta*a[k-j] (with boundary)
    # Actually: d conv[k] / d a[j] = 2*a[k-j] (valid) as derived earlier
    
    order = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        rng.shuffle(order)
        improved_epoch = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                    break
            
            # Optimize a[j] given others fixed
            # conv[k] as function of a[j]:
            # conv(k) = base_k + a[j] * (a[k-j] if valid) + a[k-j] * a[j] if k-j valid
            # Actually: conv(k) = sum_i a[i]*a[k-i]
            # Terms involving a[j]: a[j]*a[k-j] (when k-j valid) + a[k-j]*a[j] (same term if k-j != j)
            # More carefully: a[j] appears when i=j (gives a[j]*a[k-j]) or when k-i=j i.e. i=k-j (gives a[k-j]*a[j])
            # If k != 2j: two distinct terms, each contributes a[j]*a[k-j], total 2*a[j]*a[k-j]
            # If k == 2j: one term a[j]*a[j] = a[j]^2
            
            # So conv(k) = base(k) + a[j] * c_k where:
            # c_k = a[k-j] (if 0<=k-j<n and k-j != j) + a[k-j] (if 0<=k-j<n and k-j != j) + (a[j] if k==2j else 0)
            # Simpler: conv(k) = base_no_j(k) + a[j] * 2 * a[max(0,k-j)] * I(0<=k-j<n, k!=2j)
            #                                  + a[j]^2 * I(k==2j)
            
            # For the max over k, we want a[j] that minimizes max_k [base(k) + contribution(k, a[j])]
            # This is a 1D problem! Use ternary search or golden section.
            
            # Compute base convolution (without a[j]'s contribution)
            a_save = a[j]
            a[j] = 0
            b_base = fast_conv(a)
            a[j] = a_save
            
            # Contribution of a[j]: for each k, add a[j]*a[k-j] + a[k-j]*a[j] (handling k=2j)
            # = 2*a[j]*a[k-j] for k != 2j, and a[j]^2 for k = 2j
            # Build the contribution vector
            contrib = np.zeros(2 * n - 1)
            if a_save > 1e-15:
                for i in range(n):
                    k = i + j  # when first a is at position j, second at i: conv position j+i
                    # Actually conv[j+i] gets contribution a[j]*a[i]
                    # And conv[i+j] gets a[i]*a[j] (same position)
                    # So conv[j+i] += a[j]*a[i] (from i=j in sum) 
                    # And conv[j+i] += a[i]*a[j] (from k-i=j, i.e., the term where position j is in second factor)
                    # Wait, I need to be more careful.
                    pass
            
            # Simpler approach: just compute contribution directly
            # conv(k) with a[j] = x: 
            # = conv_base(k) + x * [a[k-j] * I(0<=k-j<n) + a[k-j] * I(0<=k-j<n)]
            #   - x * x * I(k==2j)  [to correct for double counting at k=2j, since base was computed with a[j]=0]
            # Actually, with a[j]=0 in base:
            # conv_base(k) = sum_{i != j} a[i] * a[k-i]
            # conv with a[j]=x:
            # = sum_{i != j} a[i]*a[k-i] + x*a[k-j]*I(0<=k-j<n) + a[k-j]*x*I(0<=k-j<n) - x*x*I(k==2j)
            # Wait no. If a[j] was 0 in base, then:
            # conv(k) = conv_base(k) + x * a[k-j] * I(0<=k-j<n) + a[j_new] ... 
            # This is getting confusing. Let me just do it directly.
            
            # Let's just do: for a trial value x of a[j], compute the change in max conv.
            # conv_new(k) = conv_base(k) + delta_conv(k, x)
            # where delta_conv(k, x) = x*a[k-j] (if valid) + a[k-j]*x (if valid) - x^2 (if k==2j)
            # = 2*x*a[k-j]*I(valid) - x^2*I(k==2j)
            
            # For efficiency, precompute the relevant a values
            aj = a_save
            # The max of (b_base[k] + 2*x*a[k-j] - x^2*I(k==2j)) over k
            # We want to find x >= 0 that minimizes this max, then renormalize.
            
            # Since this is a 1D optimization, use a few trial values
            # x ranges from 0 to 2*aj (can't be too large)
            best_x = aj
            best_max_val = float('inf')
            
            # Try a grid + refinement
            x_vals = np.concatenate([
                np.linspace(0.0, aj * 1.5, 20),
                np.array([aj * 0.5, aj * 0.7, aj * 1.3])
            ])
            
            for x in x_vals:
                if x < 0:
                    x = 0
                # Compute max of b_base + 2*x*a[k-j] - x^2*I(k==2j)
                contrib = np.zeros(2 * n - 1)
                # 2*x*a[k-j] for valid k-j
                lo = max(0, j)  # k-j >= 0 => k >= j
                hi = min(2 * n - 2, j + n - 1)  # k-j <= n-1 => k <= j+n-1
                ks = np.arange(lo, hi + 1)
                contrib[ks] = 2 * x * a[ks - j]
                if 2 * j < 2 * n - 1:
                    contrib[2 * j] -= x * x  # correction for double count
                total = b_base + contrib
                max_val = total.max()
                if max_val < best_max_val:
                    best_max_val = max_val
                    best_x = x
            
            # Update
            if best_x != aj:
                a_trial = a.copy()
                a_trial[j] = best_x
                s_t = a_trial.sum()
                if s_t > 0.01:
                    a_trial = a_trial / s_t
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved_epoch = True
        
        if not improved_epoch:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
    
    elapsed = time.monotonic() - t0
    print(f"    CoordPolish ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, seed=42, variant=0):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        # Water-shaped: dip in center
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(np.pi * x)
    elif variant == 8:
        # Multi-peak
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for c in [0.2, 0.4, 0.6, 0.8]:
            a += np.exp(-100 * (x - c) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def mirror_descent_fast(a0, time_limit, eta0=0.05, decay_rate=0.6,
                        k_min=1, k_max=15, k_ramp=300, seed=42):
    """Fast mirror descent with adaptive top-k subgradient."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        
        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))
        
        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx] - b.mean()
        if excess.max() < 1e-15:
            weights = np.ones(k) / k
        else:
            excess_shift = excess - excess.max()
            weights = np.exp(excess_shift) / np.exp(excess_shift).sum()
        
        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]
        
        eta = eta0 / (1.0 + t / 300.0) ** decay_rate
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.02 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / max(s_new, 1e-15)
        else:
            a_new = a_new / s_new
        
        a = a_new
        
        check_interval = 3 if t < 500 else (10 if t < 5000 else 30)
        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
        
        if stall_count > 500:
            noise_scale = min(0.05, 0.01 * (stall_count / 100.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
    
    return best_a, best_score
