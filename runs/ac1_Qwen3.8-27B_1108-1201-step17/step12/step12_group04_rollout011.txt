"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast iterative methods:
1. Nesterov-accelerated approximation of LP best-response (MWU-based, no LP solver)
2. LogSumExp smooth approximation with Adam optimizer and multi-stage annealing
3. Direct subgradient method with heavy-ball momentum on the exact max
4. Coordinate descent for final polish
All convolution operations use FFT for O(n log n) complexity.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft


def fast_full_conv(a, b):
    """Compute full convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Compute full autocorrelation (self-convolution) of a."""
    return fast_full_conv(a, a)


def fast_cross_corr(w, a):
    """
    Compute cross-correlation: cost[j] = sum_i w[i+j] * a[i] for i=0..n-1
    where w has length 2n-1 and a has length n.
    Returns array of length n.
    """
    n = len(a)
    # cost[j] = sum_{i=0}^{n-1} w[i+j] * a[i]
    # This is a "sliding dot product"
    # = conv(w, a_rev)[n-1+j] where a_rev = a[::-1]
    # Let's verify: conv(w, a_rev)[k] = sum_i w[i] * a_rev[k-i] = sum_i w[i] * a[n-1-k+i]
    # At k = n-1+j: = sum_i w[i] * a[n-1-j-1+i+1] = sum_i w[i] * a[i-j]
    # That gives sum_i w[i] * a[i-j] which is cost at a different index.
    # 
    # Alternative: cost[j] = sum_i w[i+j] * a[i]
    # Let m = i+j: = sum_m w[m] * a[m-j] (where m-j is in [0,n-1])
    # = conv(w, a_rev)[2n-2-j] ... still confusing
    #
    # Let me just use: cost = irfft(rfft(w, N) * conj(rfft(a, N))) at appropriate offset
    # The cross-correlation (w ⋆ a)[l] = sum_i w[i+l] * conj(a[i]) = sum_i w[i+l] * a[i]
    # = cost[l]
    # Using FFT: (w ⋆ a) = ifft(conj(FFT(w)) * FFT(a)) -- for real signals
    # Wait: correlation(w, a) = ifft(FFT(w) * conj(FFT(a)))
    # Let me verify with a small example...
    # 
    # For real vectors: corr(l) = sum_i w[i+l] * a[i]
    # FFT(corr) = conj(FFT(w)) * FFT(a) ... no
    # 
    # Standard result: if c[l] = sum_i x[i] * y[i+l], then FFT(c) = conj(FFT(x)) * FFT(y)
    # (with appropriate zero-padding)
    # So: c = cost, x = a, y = w
    # FFT(cost) = conj(FFT(a)) * FFT(w)
    # cost = ifft(conj(A) * W)
    
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    cost_full = irfft(np.conj(A) * W, N)
    # cost_full[l] = sum_i a[i] * w[i+l] (with wraparound if not enough padding)
    # We want cost[j] for j = 0, ..., n-1
    return cost_full[:n]


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    Returns (gradient, weights).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid k)
    # = 2 * cross_correlation(w, a)[i]
    grad = 2.0 * fast_cross_corr(w, a)
    return grad, w

        else:

def mwu_best_response(a_fixed, n_iters, T_param, eta):

        if w_sum < 1e-300:

        b_sum = b.sum()
        if b_sum < 1e-300:

        else:

        # These are two different terms in the sum (different i values), both contributing.
        # d/da_j: 
        #   From i=j term: d/da_j (a_j * a_{k-j}) = a_{k-j}
        #   From i=k-j term: d/da_j (a_{k-j} * a_j) = a_{k-j}
        #   Total: 2 * a_{k-j}
        #
        # If j = k-j (k=2j):
        #   Only i=j term exists (since i=k-j = j, same term)
        #   d/da_j (a_j * a_j) = 2 * a_j
        #   Total: 2 * a_j = 2 * a_{k-j} ✓
        #
        # So in ALL cases: d/da_j [(a*a)_k] = 2 * a_{k-j} (when 0 <= k-j < n), else 0.
        # 
        # Great! So the gradient of (a*a)_k w.r.t. a is: 2 * (a shifted and reversed)
        # Specifically: grad[j] = 2 * a[k-j] if 0 <= k-j < n, else 0
        # = 2 * (a flipped)[k-j] if valid
        # = 2 * (a[::-1])[n-1-(k-j)] if valid
        # = 2 * a_rev[n-1-k+j] if 0 <= n-1-k+j < n
        
        # For multiple peaks, use weighted average
        n_peaks = len(peak_indices)
        grad = np.zeros(n)
        for k in peak_indices:
            # grad[j] += 2 * a[k-j] (valid)
            # a[k-j] is valid when 0 <= j <= k and j < n and k-j < n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * a[k - j]
        
        grad /= n_peaks
        
        # Momentum update
        vel = beta * vel - lr * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0)
        a_new = project_simplex(a_new)
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        lr *= lr_decay
        
        if stall > 500:
            # Restart momentum
            vel = np.zeros(n)
            lr = lr_start * 0.5
            stall = 0
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG t={it} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Coordinate descent: try transferring small amounts between pairs of coordinates.
    Focus on reducing the maximum autocorrelation peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    # Precompute peak locations
    b = fast_autocorr(a)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Get top peaks
        top_k = np.argsort(b)[::-1][:10]
        
        improved = False
        for k in top_k:
            # For peak at lag k, the contributing pairs are (i, k-i) for valid i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Find the largest contributor
            if je < js:
                continue
            contrib = a[js:je+1] * a[k-js:k-je+1][::-1]
            top_contrib_idx = np.argsort(contrib)[::-1][:5]
            
            for tc in top_contrib_idx:
                src = js + tc
                src_val = a[src]
                if src_val < 1e-14:
                    continue
                
                # Try to move mass from src to a "good" destination
                # Good destination: one that doesn't contribute much to any top peak
                # Score each potential destination
                peak_set = set(top_k.tolist())
                
                # Compute contribution of each position to all top peaks
                worst_contrib = np.zeros(n)
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for j2 in range(js2, je2 + 1):
                        partner = k2 - j2
                        if 0 <= partner < n:
                            worst_contrib[j2] += a[j2] * a[partner]
                
                # Good destinations: low worst_contrib, not src
                worst_contrib[src] = np.inf
                # Sort by worst_contrib (ascending) and try the best few
                dst_candidates = np.argsort(worst_contrib)[:20]
                
                for dst in dst_candidates:
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.01, 0.03, 0.05]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = project_simplex(a_trial)
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_trials = 3
            for _ in range(n_trials):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = project_simplex(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            
            if stall > 500:
                break
        
        if stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"      CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.sin(10 * np.pi * x))
    elif choice == 7:
        # Multi-frequency
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.4 + 0.25 * np.cos(2*x) + 0.15 * np.cos(5*x) + 0.1 * np.cos(11*x) + 0.05 * np.cos(23*x)
    elif choice == 8:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) + 0.3 * np.sin(7 * np.pi * x) ** 2
    else:
        # Flat with noise
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
