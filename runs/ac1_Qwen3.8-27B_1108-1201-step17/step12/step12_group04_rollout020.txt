"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response iteration (primary method)
2. L_p norm gradient descent with Adam optimizer (polish)
3. Peak-aware coordinate descent (fine tuning)
4. Multi-length exploration
5. Simulated annealing (final escape from local minima)

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * fast_autocorr(a).max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b)_k = sum_j a_fixed[j]*b[k-j]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. (a_fixed * b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

    avg_a = np.zeros(n)

        if b_star is None:

        alpha_t = 2.0 / (t + 2.0)

        if ss < 1e-15:

        else:

        else:

                a_new = y_new.copy()

        a = a_new

            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, LP_M={M_val:.10f}")

def lp_norm_gradient(a, p):
    """
    Compute gradient of (sum_k c_k^p)^(1/p) where c = autocorr(a).
    O(n log n) via FFT.
    Returns (value, gradient).
    """
    n = len(a)
    c = fast_autocorr(a)
    c = np.maximum(c, 0)  # numerical safety
    
    if p == float('inf'):
        # Just return max and subgradient
        k_max = np.argmax(c)
        val = c[k_max]
        # Subgradient: 2 * a_{k_max - i}
        grad = np.zeros(n)
        for i in range(n):
            idx = k_max - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]
        return val, grad
    
    c_p = c ** (p - 1)
    sum_cp = c_p.sum()
    if sum_cp < 1e-300:
        return 0.0, np.zeros(n)
    
    val = (c ** p).sum() ** (1.0 / p)
    # Actually: (sum c_k^p)^(1/p)
    sum_cp_p = (c ** p).sum()
    val = sum_cp_p ** (1.0 / p)
    
    # grad_i = (1/p) * (sum c^p)^(1/p - 1) * sum_k p * c_k^(p-1) * 2 * a_{k-i}
    #        = (sum c^p)^(1/p - 1) * sum_k c_k^(p-1) * 2 * a_{k-i}
    
    # Compute sum_k c_k^(p-1) * a_{k-i} for all i via FFT
    w = c_p  # w_k = c_k^(p-1)
    a_rev = a[::-1].copy()
    
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    
    grad = 2.0 * conv[n - 1:n - 1 + n]
    scale = sum_cp_p ** (1.0 / p - 1.0)
    grad = scale * grad
    
    return val, grad


def adam_lp_optimize(a0, time_limit, p_start=4, p_end=256, n_stages=12):
    """
    L_p norm optimization with Adam optimizer.
    Multi-stage: p goes from p_start to p_end.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    weights = np.geomspace(p_start, p_end, n_stages)
    
    for si, p in enumerate(weights):
        if time.monotonic() - t0 > time_limit:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Learning rate schedule: start higher, decay
        lr_base = 0.02 * (n_stages - si) / n_stages + 0.003
        iters_per_stage = max(20, min(200, int(remaining / max(0.01, (time.monotonic() - t0) / max(1, si + 1) if si > 0 else 1))))
        # Simpler: just run until time runs out or max iters
        max_iters = 300
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            t_adam += 1
            val, grad = lp_norm_gradient(a, p)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            a_new = a - lr_base * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project to simplex (simple version)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / ss
            
            a = a_new
            
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Adam stage={si}/p={p:.1f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 2:
        # Bimodal
        a = np.exp(-6 * (x - 0.15) ** 2) + 0.8 * np.exp(-6 * (x - 0.85) ** 2)
    elif choice == 3:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 4:
        # Flat top with tapered edges
        a = np.where((x > 0.1) & (x < 0.9), 1.0, 0.5 * (1 + np.cos(np.pi * x / 0.1)))
        a = np.where(x <= 0.1, a, a)
        a = np.where(x >= 0.9, 0.5 * (1 + np.cos(np.pi * (x - 1) / 0.1)), a)
    else:
        # Sine with modulation
        a = (0.5 + 0.3 * np.sin(4 * np.pi * x)) * np.sin(np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
