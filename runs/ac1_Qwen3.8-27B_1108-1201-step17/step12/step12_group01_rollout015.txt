2. Mirror descent (exponentiated gradient) subgradient method

4. Multi-scale peak shaving with greedy mass transfer

Key improvements over previous: faster LP via HiGHS, new mirror descent,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_objective(a):
    """Compute 2*n*max_conv/sum^2 without calling evaluate_sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, clip non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        length = min(n, 2 * n - 1 - j)
        T[start:start + length, j] = a[:length]
    return T


def solve_lp_best_response(a, T_matrix):
    """
    Solve LP: min M s.t. T @ b <= M, sum(b)=1, b>=0.
    Returns (b_val, M_val) or (None, None).
    """
    n = len(a)
    num_con = 2 * n - 1
    # Variables: [b_0, ..., b_{n-1}, M]
    # Minimize: [0,...,0, 1] @ x
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality: T @ b - M <= 0  =>  [T, -1] @ x <= 0
    A_ub = np.zeros((num_con, n + 1))
    A_ub[:, :n] = T_matrix
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(num_con)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: b_i >= 0, M free
    bounds = [(0, None)] * n + [(-np.inf, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status != 0:
        return None, None

    b_val = np.maximum(res.x[:n], 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(res.x[-1])

    best_score = get_objective(a)
    t0 = time.monotonic()

    avg_count = 0

    SHORT_W = 50

        elapsed = time.monotonic() - t0

        b_star, M_val = solve_lp_best_response(y, T_mat_y)
        if b_star is None:

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b_star.copy()
        else:

        else:

            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Update averages
        avg_counter += a
        avg_count += 1
        short_buf.append(a.copy())
        if len(short_buf) > SHORT_W:
            short_buf.pop(0)

        # Check scores
        score_a = get_objective(a)
        if score_a < best_score:

            best_a = a.copy()
            stall = 0
        else:

        # Check y
        score_y = get_objective(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        # Check running average periodically
        if avg_count % 50 == 0 and avg_count >= 100:
            avg_val = avg_counter / avg_count
            sc_avg = get_objective(avg_val)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_val.copy()
                stall = 0

        # Check short window average

            avg_short = np.mean(short_buf, axis=0)

def mirror_descent_subgrad(a0, time_limit, topk=20, T_start=50.0, T_end=0.5,
                           lr_start=0.05, lr_end=0.001, n_stages=15):
    """
    Mirror descent (exponentiated gradient) with softmax-weighted subgradient.
    a_i <- a_i * exp(-eta * g_i) / Z
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = get_objective(a)
    t0 = time.monotonic()

    # Annealing schedule
    stage_len = max(10, int((time_limit - 5) * 3 / n_stages))
    temps = np.geomspace(T_start, T_end, n_stages)
    lrs = np.geomspace(lr_start, lr_end, n_stages)

    global_it = 0
    for stage in range(n_stages):
        T_curr = temps[stage]
        lr_curr = lrs[stage]

        for it in range(stage_len):
            if time.monotonic() - t0 > time_limit - 5:
                break

            # Compute autocorrelation
            c = fast_autocorr(a)
            cmax = c.max()

            # Softmax weights over top-k
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]

            Tc = T_curr * top_vals
            Tc_shifted = Tc - Tc.max()
            exp_Tc = np.exp(Tc_shifted)
            w = exp_Tc / exp_Tc.sum()

            # Compute subgradient
            g = np.zeros(n)
            for idx in range(k_use):
                k = top_idx[idx]
                w_k = w[idx]
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                g[i_start:i_end + 1] += 2.0 * w_k * a[k - np.arange(i_start, i_end + 1)]

            # Normalize subgradient
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                continue
            g = g / g_norm

            # Mirror descent update
            log_a = np.log(np.maximum(a, 1e-30))
            log_a -= lr_curr * g
            # Numerical stability
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = a_new / a_new.sum()
            a = a_new
            global_it += 1

            # Track best
            sc = get_objective(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

            # Also try the "flat" version (average of recent iterates)
            if global_it % 20 == 0 and global_it > 500:
                # Can't easily average in log space, skip for now
                pass

        if stage % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD stage {stage}/{n_stages} (T={T_curr:.1f}, lr={lr_curr:.4f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def spectral_peak_reduction(a0, time_limit, n_iters=500, alpha=0.3):
    """
    FFT-based spectral peak reduction.
    Modifies the power spectrum to flatten the autocorrelation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = get_objective(a)
    t0 = time.monotonic()

    # FFT size (power of 2, >= 2n-1)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Compute current autocorrelation
        c = fast_autocorr(a)
        cmax = c.max()
        cmean = c.mean()

        # Compute FFT
        A = np.fft.fft(a, N)
        pow_spec = np.abs(A) ** 2

        # Compute the "error" spectrum: how much each frequency contributes to peaks
        # Peaks are where c > threshold
        threshold = cmax * 0.95
        peak_mask = c >= threshold

        # The contribution of frequency f to lag k is |A(f)|^2 * exp(2*pi*i*f*k/N)
        # To reduce peaks, reduce |A(f)| where it constructively interferes at peak lags
        # Simple approach: subtract a fraction of the "peak excess" from the spectrum
        # Compute IFFT of pow_spec to verify = autocorrelation (circular)
        c_circ = np.real(np.fft.ifft(pow_spec))[:size]

        # Excess at peaks
        excess = np.maximum(0, c_circ - cmean)
        excess_spec = np.fft.fft(excess, N)

        # Modify power spectrum
        new_pow = pow_spec - alpha * alpha * np.abs(excess_spec) ** 2 / N
        new_pow = np.maximum(new_pow, 0)

        # Recover signal
        new_mag = np.sqrt(new_pow)
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new))[:n]

        # Clip and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = get_objective(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()

        # Decay alpha
        alpha *= 0.999

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Spec it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Greedy peak shaving: identify top peaks, move mass from contributing
    positions to safe positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = get_objective(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 5:
        c = fast_autocorr(a)
        cmax = c.max()
        threshold = cmax * 0.99
        top_peaks = np.where(c >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:5]

        improved = False
        for k in top_peaks[:15]:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            # Find safe destination: position that doesn't contribute to any top peak
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js = max(0, k2 - n + 1)
                je = min(n - 1, k2)
                danger[js:je + 1] = True
            danger[src] = True

            safe = np.where(~danger)[0]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]

            # Prefer positions with low a values
            safe_sorted = safe[np.argsort(a[safe])]
            candidates = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(candidates) == 0:
                continue
            dst = candidates[rng.integers(len(candidates))]

            # Try multiple step sizes (log-spaced)
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.12, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-15:
                    continue
                a_trial = a_trial / ts
                sc = get_objective(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts > 1e-10:
                    a_trial = a_trial / ts
                    sc = get_objective(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0

            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    else:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LP Nesterov, mirror descent,
    spectral methods, and peak shaving at multiple lengths.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)
    print(f"Time budget: {time_budget:.0f}s")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600
    if parent is None:
        best_seq = structured_init(n_base, rng)
        best_score = get_objective(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t1)
        try_update(a_out, f"lp_n{n_base}_1")

    # === Phase 2: Mirror descent at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: Mirror descent n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 30:
        a_out, _ = mirror_descent_subgrad(best_seq, t2, topk=15,
                                          T_start=40.0, T_end=0.3,
                                          lr_start=0.03, lr_end=0.001,
                                          n_stages=12)
        try_update(a_out, f"md_n{n_base}")

    # === Phase 3: LP Nesterov at longer length ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: LP Nesterov n={n_long} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.22, 320.0)
    if t3 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t3)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 4: Spectral method ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Spectral n={n_base} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 20:
        a_out, _ = spectral_peak_reduction(best_seq, t4, n_iters=300, alpha=0.2)
        try_update(a_out, f"spec_n{n_base}")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Peak shaving n={n_base} ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 15:
        a_out, _ = peak_shaving(best_seq, t5, seed=777)
        try_update(a_out, f"peak_n{n_base}")

    # === Phase 6: LP Nesterov at even longer length ===
    n_long2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 6: LP Nesterov n={n_long2} ({remaining:.0f}s left) ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 60:
        init_long2 = interpolate_sequence(best_seq, n_long2)
        a_out, _ = nesterov_lp_optimize(init_long2, t6)
        try_update(a_out, f"lp_n{n_long2}")

    # === Phase 7: Mirror descent on best so far ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 7: Mirror descent final ({remaining:.0f}s left) ===")
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 15:
        a_out, _ = mirror_descent_subgrad(best_seq, t7, topk=10,
                                          T_start=20.0, T_end=0.2,
                                          lr_start=0.02, lr_end=0.0005,
                                          n_stages=10)
        try_update(a_out, "md_final")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 8: Final peak shaving ({remaining:.0f}s left) ===")
    t8 = min(remaining * 0.35, 120.0)
    if t8 > 10:
        a_out, _ = peak_shaving(best_seq, t8, seed=999)
        try_update(a_out, "peak_final")

    # === Phase 9: Very short LP run for polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 9: LP polish ({remaining:.0f}s left) ===")
        t9 = min(remaining * 0.4, 80.0)
        if t9 > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, t9)
            try_update(a_out, "lp_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
