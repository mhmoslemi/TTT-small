import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    """Full linear convolution a*b using FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def proj_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1.0 + u * np.arange(1, n + 1) > cssv)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_mirriskdescent(a0, time_limit, T=20.0, eta=0.08, momentum=0.9, n_stages=5):
    """
    LSE Mirror Descent with Nesterov momentum in log-space.
    Works with KL geometry: a = softmax(log_a), update log_a -= eta * grad.
    Each iteration is O(n log n) via FFT.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    log_a = np.log(np.maximum(a, 1e-300))
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    temps = np.geomspace(T, max(1.0, T * 0.1), n_stages)
    tout = 0

    for si, cur_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        iters = max(50, min(50000, int(remaining * 300 / (n_stages - si))))

        for it in range(iters):
            tout += 1
            if it % 2000 == 1999 and time.monotonic() - t0 > time_limit:
                break

            a = np.exp(log_a - log_a.max())
            a = a / a.sum()

            # Compute b = conv(a, a)
            A = rfft(a, fft_size)
            b = irfft(A * A, fft_size)[:2 * n - 1]

            # LSE gradient: w = softmax(T * b), grad = 2*conv(a, w)
            Tb = cur_T * b
            Tb_shifted = Tb - Tb.max()
            exp_Tb = np.exp(Tb_shifted)
            w = exp_Tb / exp_Tb.sum()

            # grad = 2 * conv(a, w) - we absorb the 2 into eta
            W = rfft(w, fft_size)
            grad = 2.0 * irfft(A * W, fft_size)[:n]

            # Nesterov momentum
            v = momentum * v + (1.0 - momentum) * grad

            # Mirror descent update in log-space
            log_a = log_a - eta * v
            # Periodic renormalization for numerical stability
            if it % 500 == 499:
                log_a = log_a - log_a.max()

            # Track best
            if it % 200 == 199:
                score = 2.0 * n * b.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        # End of stage: evaluate
        a_final = np.exp(log_a - log_a.max())
        a_final = a_final / a_final.sum()
        score = 2.0 * n * fast_autocorr(a_final).max()
        if score < best_score:
            best_score = score
            best_a = a_final.copy()

        elapsed = time.monotonic() - t0
        print(f"      LSE-MD stage {si}/{n_stages} (T={cur_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_conv_row(a, k):
    """Build the k-th row of the convolution matrix T(a).
    Row k: (T@a)_k = sum_j a_{j} * a_{j+k} for j=0..n-1-k (when k>=0)
    Actually T is (2n-1) x n where T[i,j] = a_{i-j} if 0<=i-j<n else 0
    """
    n = len(a)
    row = np.zeros(n)
    if k < n:
        # T[k, j] = a[k-j] if 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(k, n-1)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        row[js:je + 1] = a[k - je:k - js + 1][::-1]
    else:
        # k >= n: T[k, j] = a[k-j] if 0 <= k-j < n
        # j ranges from k-n+1 to n-1
        js = k - n + 1
        if js < n:
            je = n - 1
            row[js:je + 1] = a[k - je:k - js + 1][::-1]
    return row


def solve_lp_simple(a_fixed, active_indices=None, max_active=50):
    """
    Solve LP: min M s.t. (conv matrix) @ b <= M, sum(b)=1, b>=0.
    Uses active set reduction: only enforce constraints at active_indices.
    Returns (b, M) or (None, None).
    """
    import cvxpy as cp
    n = len(a_fixed)

    if active_indices is None or len(active_indices) == 0:
        active_indices = list(range(2 * n - 1))

    # Start with a limited set of constraints
    # Always include lag 0 (the diagonal)
    if 0 not in active_indices:
        active_indices = [0] + list(active_indices)

    # Limit the number of constraints
    if len(active_indices) > max_active:
        # Keep the first max_active
        active_indices = active_indices[:max_active]

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraints for active indices
    constraints = [cp.sum(b_var) == 1]
    for k in active_indices:
        row = build_conv_row(a_fixed, k)
        constraints.append(row @ b_var <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    solved = False
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None, None
    b_val = b_val / sv
    M_val = float(M_var.value)

    # Get tight constraints for next iteration
    T_b = np.array([row @ b_val for row_idx in range(2 * n - 1)
                    for row in [build_conv_row(a_fixed, row_idx)]])

    # Actually, let's compute all (T*b)_k efficiently
    Tb_all = fast_conv(a_fixed, b_val)
    tight_mask = Tb_all >= M_val - 1e-8
    tight_indices = np.where(tight_mask)[0].tolist()

    return b_val, M_val, tight_indices


def solve_lp_simple_v2(a_fixed, active_indices=None, max_active=50):
    """Fixed version that always returns 3 values or 3 Nones."""
    n = len(a_fixed)
    if active_indices is None or len(active_indices) == 0:
        active_indices = list(range(0, 2 * n - 1, max(1, (2 * n - 1) // max_active)))

    if 0 not in active_indices:
        active_indices = [0] + list(active_indices)

    if len(active_indices) > max_active:
        # Prioritize: always keep 0, then spread out the rest
        if 0 in active_indices:
            active_indices = [0] + sorted(active_indices)[1:max_active]
        else:
            active_indices = sorted(active_indices)[:max_active]

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(b_var) == 1]
    for k in active_indices:
        row = build_conv_row(a_fixed, k)
        constraints.append(row @ b_var <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    solved = False
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None, None, None
    b_val = b_val / sv
    M_val = float(M_var.value)

    Tb_all = fast_conv(a_fixed, b_val)
    tight_mask = Tb_all >= M_val - 1e-6
    tight_indices = np.where(tight_mask)[0].tolist()

    return b_val, M_val, tight_indices


def lp_nesterov_active(a0, time_limit, max_active=40, verify_period=50):
    """
    LP Nesterov with active set reduction.
    Each LP only enforces the currently active (tight) constraints.
    Periodically verifies against the full constraint set.
    """
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Initialize active set
    b_init = fast_autocorr(a)
    active_set = list(range(0, 2 * n - 1, max(1, (2 * n - 1) // max_active)))
    if 0 not in active_set:
        active_set.insert(0, 0)

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 1000001):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP with active set
        result = solve_lp_simple_v2(y, active_set, max_active)
        if result[0] is None:
            continue
        b_star, M_val, tight_indices = result

        # Update active set
        if tight_indices:
            # Merge: keep existing + new tight ones, limit size
            new_set = set(active_set) | set(tight_indices)
            if 0 not in new_set:
                new_set.add(0)
            sorted_set = sorted(new_set)
            if len(sorted_set) > max_active:
                # Keep 0 and spread out
                others = [x for x in sorted_set if x != 0]
                step = max(1, len(others) // (max_active - 1))
                kept = others[::step]
                if len(kept) > max_active - 1:
                    kept = kept[:max_active - 1]
                active_set = [0] + kept
            else:
                active_set = sorted_set

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Evaluate
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 40 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        # Periodic full verification
        if t % verify_period == 0:
            full_b = fast_conv(y, b_star)
            full_M = full_b.max()
            if full_M > M_val + 1e-4:
                # Solution violates some dropped constraint
                # Add the violated constraints to active set
                violated = np.where(full_b > M_val + 1e-6)[0].tolist()
                active_set = list(set(active_set) | set(violated[:10]))
                if 0 not in active_set:
                    active_set.append(0)
                active_set = sorted(active_set)[:max_active]

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-Active t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}, |A|={len(active_set)}")

    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, eta_start=0.1, decay=0.9999):
    """
    Nesterov-accelerated subgradient method for max(conv(a,a)).
    Very fast per iteration: O(n) for subgradient computation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta_start
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Compute subgradient for b_{k_star}
        g = np.zeros(n)
        if k_star == 0:
            g = 2.0 * a
        else:
            g[:n - k_star] += a[k_star:]
            g[k_star:] += a[:n - k_star]

        # Nesterov step
        if t == 0:
            a_candidate = a - eta * g
        else:
            momentum = (t) / (t + 3.0)
            a_candidate = a + momentum * (a - a_prev) - eta * g

        a_candidate = proj_simplex(a_candidate)
        a_prev = a
        a = a_candidate

        # Decay step size
        eta *= decay

        avg_a += a
        n_avg += 1

        if t % 1000 == 999:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if t % 10000 == 9999:
                avg_check = avg_a / n_avg
                score_avg = 2.0 * n * fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

                elapsed = time.monotonic() - t0
                print(f"      SubGrad-N t={t//1000}k ({elapsed:.0f}s): best={best_score:.6f}, eta={eta:.4f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss

        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
