def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_T(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_full(a_fixed):
    """Solve: min M s.t. T@b <= M, sum(b)=1, b>=0 using HiGHS."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_T(a_fixed)
    A_ub = np.column_stack([T, -np.ones(size)])
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    c = np.zeros(n + 1)
    c[n] = 1.0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(res.x[n])


def solve_lp_cg(a_fixed, max_rounds=200):
    """Solve LP via constraint generation (cutting plane). Much faster for large n."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Start with a few constraints
    active_k = list(range(0, size, max(1, size // 10)))
    
    b_val = None
    M_val = np.inf
    
    for round_idx in range(max_rounds):
        k_arr = np.array(active_k)
        T_sub = np.zeros((len(k_arr), n))
        for idx, k in enumerate(k_arr):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            T_sub[idx, js:je + 1] = a_fixed[k - np.arange(js, je + 1)]
        
        A_ub = np.column_stack([T_sub, -np.ones(len(k_arr))])
        b_ub = np.zeros(len(k_arr))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            break
        
        b_new = res.x[:n]
        b_new = np.maximum(b_new, 0)
        s = b_new.sum()
        if s < 1e-15:
            break
        b_new = b_new / s
        M_new = float(res.x[n])
        
        # Check all constraints
        T_full = build_conv_T(a_fixed)
        violations = T_full @ b_new - M_new
        violation_indices = np.where(violations > 1e-10)[0]
        
        if len(violation_indices) == 0:
            return b_new, M_new
        
        # Add most violated constraints
        if len(violation_indices) > len(active_k):
            order = np.argsort(violations[violation_indices])[::-1]
            new_k = violation_indices[order[:len(active_k)]]
        else:
            new_k = violation_indices
        
        active_k = list(set(active_k) | set(new_k.tolist()))
        b_val = b_new
        M_val = M_new
    
    if b_val is not None:
        return b_val, M_val
    return None, None


def nesterov_lp_fast(a0, time_limit, use_cg=True, max_iters=500000):
    """Nesterov-accelerated LP iteration with HiGHS."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    solver = solve_lp_cg if use_cg else solve_lp_full
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a
        
        b_new, M_lp = solver(y)
        if b_new is None:
            b_new, M_lp = solve_lp_full(y)
            if b_new is None:
                continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        
        a = a_new
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed
    
    return best_a, best_score


def subgrad_nesterov_fast(a0, time_limit, max_steps=500000, eta_init=0.5):
    """Fast projected subgradient Nesterov with exact convolution gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = eta_init
    last_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000 and step > 500:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Use near-argmax set for stability
        threshold = max_b * 0.99
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad *= 2.0 / len(topk)
        
        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
        else:
            y = a
        
        # Mirror descent update (exponential)
        z_curr = z_from_a(np.maximum(y, 1e-300))
        z_new = z_curr - eta * grad
        a_new = softmax_from_z(z_new)
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
            eta = min(eta * 1.001, 1.0)
        else:
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.95
            else:
                a_prev = a
                a = a_new
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            eta = max(eta * 0.999, 0.05)
    
    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def coordinate_descent(a0, time_limit, max_rounds=200000):
    """Coordinate descent: adjust one element at a time to reduce peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    last_improve = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if rnd - last_improve > 5000 and rnd > 1000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Try adjusting a random element
        i = int(rng.integers(0, n))
        
        # Compute effect of increasing a_i on the peak
        # (a*a)_k increases by delta * (a_{k-i} + a_{i-k}) for valid k
        contributions = np.zeros(2 * n - 1)
        # contribution to lag k from a_i: a_{k-i} if 0<=k-i<n
        js = max(0, i)
        je = min(2 * n - 2, i + n - 1)
        for k in range(i, i + n):
            if k < 2 * n - 1:
                contributions[k] += a[k - i]
        for k in range(max(0, i - n + 1), i):
            contributions[k] += a[i - k]
        
        # If contributions at peak are high, decrease a_i
        peak_idx = int(np.argmax(b))
        if contributions[peak_idx] > 0:
            # Decrease a_i
            delta = a[i] * 0.1
            a[i] = max(0, a[i] - delta)
        else:
            # Increase a_i
            a[i] += delta if 'delta' in dir() else a[i] * 0.1
        
        s = a.sum()
        if s < 1e-15:
            a[i] = 1.0 / n
            continue
        a = a / s
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            last_improve = rnd
    
    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) * 0.1 + rng.random(n) * 0.9
    elif mode == 1:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.random() * x + rng.random() * 2 * np.pi)
        a = np.abs(a)
    elif mode == 2:
        a = np.exp(-2.0 * np.abs(x - rng.random()) * 3) + 0.1
    elif mode == 3:
        a = np.ones(n)
        n_bumps = rng.integers(2, 5)
        for _ in range(n_bumps):
            center = rng.random()
            width = rng.uniform(0.05, 0.2)
            a += np.exp(-((x - center) / width) ** 2)
    elif mode == 4:
        # Block structure
        n_blocks = rng.integers(3, 8)
        a = np.zeros(n)
        for bi in range(n_blocks):
            start = int(bi * n / n_blocks)
            end = int((bi + 1) * n / n_blocks)
            a[start:end] = rng.uniform(0.1, 1.0)
    elif mode == 5:
        # Periodic with noise
        period = rng.integers(3, 10)
        a = np.tile(rng.uniform(0.1, 1.0, period), n // period + 1)[:n]
        a += 0.1 * rng.random(n)
    elif mode == 6:
        # Power law
        a = (x + 0.01) ** rng.uniform(-1, 1)
        a = np.abs(a) + 0.05
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x * rng.integers(2, 6)) + 0.2 * rng.random(n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    if best_seq is None:
        best_seq = structured_init(n_base, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    best_n = n_base
    
    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")
    
    # === Phase 0: Quick diverse initialization (30s) ===
    print(f"\n=== Phase 0: Diverse init (score={best_score:.8f}) ===")
    t0_phase = time.monotonic()
    for mode in range(6):
        if time.monotonic() > _deadline - 150:
            break
        a_try = structured_init(n_base, rng, mode=mode)
        # Quick subgradient polish (3s each)
        a_out, _ = subgrad_nesterov_fast(a_try, 3.0, max_steps=5000, eta_init=0.3)
        try_update(a_out, f"init_mode{mode}")
    print(f"  Phase 0 done: best_score={best_score:.8f}")
    
    # === Phase 1: Fast LP-based Nesterov (main workhorse, 700s) ===
    print(f"\n=== Phase 1: Nesterov LP (HiGHS) at n={best_n} (score={best_score:.8f}) ===")
    remaining = _deadline - time.monotonic()
    t1 = min(700.0, remaining * 0.55)
    if t1 > 30:
        a_out, _ = nesterov_lp_fast(best_seq, t1, use_cg=True, max_iters=500000)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")
    
    # === Phase 2: Subgradient Nesterov polish (300s) ===
    print(f"\n=== Phase 2: Subgrad Nesterov (score={best_score:.8f}) ===")
    remaining = _deadline - time.monotonic()
    t2 = min(300.0, remaining * 0.25)
    if t2 > 20:
        a_out, _ = subgrad_nesterov_fast(best_seq, t2, max_steps=500000, eta_init=0.3)
        try_update(a_out, "subgrad_nesterov")
    print(f"  Phase 2 done: best_score={best_score:.8f}")
    
    # === Phase 3: LSE L-BFGS (300s) ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    remaining = _deadline - time.monotonic()
    t3 = min(300.0, remaining * 0.25)
    if t3 > 20:
        a_out = lbgfs_logsumexp(best_seq, t3, T_start=60.0, T_end=0.02, n_stages=30)
        try_update(a_out, "lse_polish")
    print(f"  Phase 3 done: best_score={best_score:.8f}")
    
    # === Phase 4: Second LP round (200s) ===
    print(f"\n=== Phase 4: Nesterov LP round 2 (score={best_score:.8f}) ===")
    remaining = _deadline - time.monotonic()
    t4 = min(200.0, remaining * 0.2)
    if t4 > 15:
        a_out, _ = nesterov_lp_fast(best_seq, t4, use_cg=True, max_iters=300000)
        try_update(a_out, "nesterov_lp_r2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")
    
    # === Phase 5: Final fine polish (150s) ===
    print(f"\n=== Phase 5: Final polish (score={best_score:.8f}) ===")
    remaining = _deadline - time.monotonic()
    if remaining > 30:
        # Subgradient with very small steps
        t5a = min(70.0, remaining * 0.4)
        if t5a > 10:
            a_out, _ = subgrad_nesterov_fast(best_seq, t5a, max_steps=300000, eta_init=0.1)
            try_update(a_out, "subgrad_fine")
        
        remaining = _deadline - time.monotonic()
        if remaining > 15:
            t5b = min(60.0, remaining * 0.5)
            if t5b > 10:
                a_out = lbgfs_logsumexp(best_seq, t5b, T_start=20.0, T_end=0.005, n_stages=20)
                try_update(a_out, "lse_fine")
    print(f"  Phase 5 done: best_score={best_score:.8f}")
    
    # === Phase 6: Length exploration (remaining) ===
    print(f"\n=== Phase 6: Length exploration (score={best_score:.8f}) ===")
    remaining = _deadline - time.monotonic()
    if remaining > 60:
        for n_try in [best_n + 100, best_n - 100, best_n + 200, best_n + 50]:
            remaining = _deadline - time.monotonic()
            if remaining < 30:
                break
            if n_try < 200:
                continue
            t6 = min(40.0, remaining * 0.3)
            if t6 < 10:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_fast(init_try, t6, use_cg=True, max_iters=100000)
            try_update(a_try, f"len_{n_try}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
