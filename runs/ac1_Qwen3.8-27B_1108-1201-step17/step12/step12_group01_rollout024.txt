"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using power-law asymmetric sequences.
Key insight: asymmetric sequences (ramps) achieve much lower objectives than
symmetric ones because they avoid the Cauchy-Schwarz lower bound on the
anti-diagonal convolution sum.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute the objective 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * np.max(c) / (s * s)


def power_law(n, p, max_val=1000.0):
    """Generate a power law sequence scaled to have max value = max_val."""
    if p == 0:
        a = np.ones(n)
    else:
        a = np.arange(1, n + 1, dtype=np.float64) ** p
    a = a * max_val / np.max(a)
    return a


def local_refine(a, time_limit, seed=42):
    """Local search: try moving mass from positions contributing to peaks."""

        cmax = np.max(c)
        # Find the peak location
        peak_k = np.argmax(c)
        
        # Find positions that contribute to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        if np.max(contrib) < 1e-15:
            break
            
        # Source: position in the peak with highest contribution
        src_offset = np.argmax(contrib)
        src = idx[src_offset]

        improved = False

        # So "safe" positions are those outside this range for the top peaks
