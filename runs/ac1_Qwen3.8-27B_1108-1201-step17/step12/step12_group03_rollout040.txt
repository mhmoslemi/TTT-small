2. Nesterov LP best-response (global zero-sum game optimization)

import numpy as np

def _get_idx(n):
    if n not in _IDX_CACHE:
        _IDX_CACHE[n] = np.arange(n)[:, None] + np.arange(n)[None, :]
    return _IDX_CACHE[n]


def fast_autocorr(a):
    """Full linear autocorrelation a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

        return float('inf')


def expand_symmetric(half):
    """Expand half-sequence to full symmetric sequence."""
    m = len(half)
    if m % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, half[-2::-1]])

    """Normalize sequence to sum 1."""

def hac_multiply(w, a):
    """Compute (W @ a) where W[i,j] = w[i+j]. w has length 2n-1, a has length n."""
    n = len(a)
    idx = _get_idx(n)
    return w[idx] @ a


def get_half_gradient(grad_full, n):
    """Project full gradient to symmetric half."""
    m = len(grad_full) // 2 if n % 2 == 0 else (n + 1) // 2
    sym_idx = np.array([n - 1 - i for i in range(m)])
    grad_half = grad_full[:m].copy() + grad_full[sym_idx]
    if n % 2 == 1:
        mid = (n - 1) // 2
        if mid < m:
            grad_half[mid] = grad_full[mid]
    return grad_half


def smooth_max_obj_grad_sym(half, tau):
    """
    Objective and gradient for symmetric smoothed max autocorrelation.
    Minimizes smoothed_max(autocorr(a/s)) where a = expand_symmetric(half), s = sum(a).
    """
    half = np.maximum(half, 0)
    a = expand_symmetric(half)
    n = len(a)
    s = a.sum()
    if s < 1e-15:

    a_norm = a / s
    b = fast_autocorr(a_norm)

    b_shifted = tau * (b - b.max())
    exp_b = np.exp(b_shifted)

    w = exp_b / exp_sum
    obj = float(w @ b)

    grad_u = 2.0 * hac_multiply(w, a_norm)
    grad_a = grad_u / s - 2.0 * obj * a / (s ** 3)

    grad_half = get_half_gradient(grad_a, n)
    return obj, grad_half


def lbfgs_smoothed_max(half_init, tau_seq, max_iter=500, tol=1e-12, time_limit=None, t_start=None):
    """
    L-BFGS optimization with increasing temperature.
    half_init: initial symmetric half-sequence
    tau_seq: list of temperature values (should be increasing)
    """
    if t_start is None:
        t_start = time.monotonic()
    half = np.maximum(half_init.copy(), 0.0)
    best_half = half.copy()
    best_obj = float('inf')
    m = len(half)

    for tau in tau_seq:
        time_remaining = (time_limit - (time.monotonic() - t_start)) if time_limit else float('inf')
        if time_remaining <= 0:

        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_new_s = a_new.sum()
            if a_new_s < 1e-15:
                a_new = y_new.copy()
                a_new_s = 1.0
            a_new = a_new / a_new_s

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            if score_a > best_score * 0.999999:
                stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, iters={iter_count}")

        if stall_count > 5000 and t > 1000:
            break

    return best_a, best_score, iter_count


def _find_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def _build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def _solve_lp_best_response(a_fixed, n, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = _build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def qp_active_set_refine(a0, time_limit, n_active=30):
    """QP refinement on active constraint set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), float('inf')
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b

    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        threshold = max_b * 0.995
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) > n_active:
            active_lags = np.argsort(b[active_lags])[::-1][:n_active] + np.sort(active_lags)[:n_active]
            active_lags = np.unique(active_lags[:n_active])
        if len(active_lags) == 0:
            active_lags = np.argsort(b)[::-1][:5]

        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=20000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iter=20000)
            except Exception:
                break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                break
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        improved = False
        if score < best_score * (1 - 1e-10):
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
            improved = True
            print(f"      QP round {round_num}: {best_score:.8f}")
        else:
            for frac in [0.2, 0.4, 0.6, 0.8]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score * (1 - 1e-10):
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    improved = True
                    print(f"      QP mix {frac}: {best_score:.8f}")
                    break
        if not improved and round_num >= 5:
            break

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Peak shaving with random perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(5, 2 * n - 1)]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            top_ci = np.argsort(contrib)[::-1][:3]
            for ci in top_ci:
                src = idx[ci]
                src_o = k - src
                for _ in range(5):
                    dst = rng.integers(0, n)
                    for frac in [0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        if src != src_o:
                            a_t[src_o] -= amt * 0.5
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0)
                        a_t_s = a_t.sum()
                        if a_t_s < 1e-15:
                            continue
                        a_t = a_t / a_t_s
                        sc = 2.0 * n * fast_autocorr(a_t).max()
                        if sc < best_score * (1 - 1e-12):
                            a = a_t
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
            if improved:
                break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break
            if stall % 5 == 0:
                n_p = rng.integers(1, 4)
                sources = rng.choice(n, size=n_p, replace=False)
                targets = rng.choice(n, size=n_p, replace=False)
                a_t = a.copy()
                for src, dst in zip(sources, targets):
                    frac = rng.uniform(0.005, 0.03)
                    amt = a_t[src] * frac
                    a_t[src] -= amt
                    a_t[dst] += amt
                a_t = np.maximum(a_t, 0)
                a_t_s = a_t.sum()
                if a_t_s < 1e-15:
                    continue
                a_t = a_t / a_t_s
                sc = 2.0 * n * fast_autocorr(a_t).max()
                if sc < best_score:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    stall = 0

    return best_a, best_score


def init_cosine(n, offset=0.0):
    """Cosine initialization (symmetric)."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2.0 + offset) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def init_gaussian(n, sigma_frac=0.3):
    """Gaussian initialization (symmetric)."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_half(a):
    """Get symmetric half of a sequence."""
    n = len(a)
    if n % 2 == 0:
        return a[:n // 2].copy()
    else:
        return a[:n // 2 + 1].copy()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search for optimal sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses L-BFGS smoothed max, Nesterov LP, QP, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    solver = _find_lp_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        n_init = 1600
        parent = init_cosine(n_init)
        best_seq = parent.copy()
        best_score = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_base = len(best_seq)

    # Phase 1: L-BFGS with multiple initializations at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 350.0)

    if t1 > 30:
        t1_start = time.monotonic()
        inits = [best_seq.copy()]
        # Add alternative initializations
        inits.append(init_cosine(n_base, offset=0.2))
        inits.append(init_gaussian(n_base, sigma_frac=0.25))
        inits.append(init_gaussian(n_base, sigma_frac=0.4))
        # Random symmetric
        r = rng.random(n_base // 2 + 1)
        r_sym = expand_symmetric(r)
        r_sym = r_sym / r_sym.sum()
        inits.append(r_sym)

        for ii, init_a in enumerate(inits):
            t_rem = t1 - (time.monotonic() - t1_start)
            if t_rem < 10:
                break
            half = get_half(init_a)
            tau_seq = [5, 15, 40, 100, 200]
            half_out, obj_out = lbfgs_smoothed_max(half, tau_seq, max_iter=300, time_limit=t_rem, t_start=t1_start)
            a_out = expand_symmetric(half_out)
            a_out = a_out / a_out.sum()
            sc = 2.0 * n_base * fast_autocorr(a_out).max()
            try_update(a_out, f"lbfgs_init{ii}")

    # Phase 2: Nesterov LP at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t2, solver)
        try_update(a_out, f"nesterov_n{n_base}")

    # Phase 3: L-BFGS refinement on Nesterov result
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: L-BFGS refinement (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        n_cur = len(best_seq)
        half = get_half(best_seq)
        tau_seq = [20, 50, 100, 200, 400, 800]
        t3_start = time.monotonic()
        half_out, obj_out = lbfgs_smoothed_max(half, tau_seq, max_iter=400, time_limit=t3, t_start=t3_start)
        a_out = expand_symmetric(half_out)
        a_out = a_out / a_out.sum()
        try_update(a_out, "lbfgs_refine1")

    # Phase 4: Try larger lengths
    for n_try in [n_base + 400, n_base + 800, n_base + 1600]:
        if n_try > 6000:
            break
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 4: L-BFGS at n={n_try} (remaining={remaining:.0f}s) ===")
        t4 = min(remaining * 0.15, 180.0)
        if t4 < 30:
            continue
        init_try = interpolate_seq(best_seq, n_try)
        half = get_half(init_try)
        t4_start = time.monotonic()
        tau_seq = [10, 30, 80, 200]
        half_out, obj_out = lbfgs_smoothed_max(half, tau_seq, max_iter=250, time_limit=t4, t_start=t4_start)
        a_out = expand_symmetric(half_out)
        a_out = a_out / a_out.sum()
        try_update(a_out, f"lbfgs_n{n_try}")

        # Quick Nesterov at this length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t4b = min(remaining * 0.10, 120.0)
            if t4b > 40:
                a_nest, sc_nest, _ = nesterov_lp_optimize(a_out, t4b, solver)
                try_update(a_nest, f"nesterov_n{n_try}")

    # Phase 5: QP refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: QP refinement at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_out, sc_out = qp_active_set_refine(best_seq, t5, n_active=25)
        try_update(a_out, "qp_refine")

    # Phase 6: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Peak shaving (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        a_out, sc_out = peak_shaving(best_seq, t6, seed=123)
        try_update(a_out, "peak_shave1")

    # Phase 7: Final L-BFGS + Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final refinement at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        t7a = min(remaining * 0.4, 200.0)
        n_cur = len(best_seq)
        half = get_half(best_seq)
        tau_seq = [50, 100, 200, 500, 1000]
        t7_start = time.monotonic()
        half_out, obj_out = lbfgs_smoothed_max(half, tau_seq, max_iter=500, time_limit=t7a, t_start=t7_start)
        a_out = expand_symmetric(half_out)
        a_out = a_out / a_out.sum()
        try_update(a_out, "lbfgs_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.4, 180.0)
            if t7b > 40:
                a_nest, sc_nest, _ = nesterov_lp_optimize(best_seq, t7b, solver)
                try_update(a_nest, "nesterov_final")

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.5, 60.0)
        a_out, sc_out = peak_shaving(best_seq, t8, seed=789)
        try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
