import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def project_simplex(v):
    """Project vector v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def lse_objective_grad_z(z, T, n):
    """LSE-smoothed objective and gradient in z-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    # LSE
    shifted = b / T
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    w = exp_b / exp_b.sum()
    lse = T * (shifted.max() + np.log(exp_b.sum()))
    
    # Gradient with respect to a: grad_a[i] = sum_k w[k] * 2 * a[k-i] (with bounds)
    grad_a = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        # a_i contributes to (a*a)[k] with coefficient 2*a_{k-i} if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += w[k] * 2.0 * a[k - idx]
    
    obj = (2.0 * n) * lse  # scaling doesn't matter for optimization direction
    
    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def nesterov_fw(a0, time_limit, max_iters=500000):
    """Nesterov-accelerated Frank-Wolfe method on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at current point
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Subgradient: g_i = 2*a_{k*-i} if 0 <= k*-i < n, else 0
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k_star - idx]
        
        # FW step: s = e_{argmin g_i}
        j = int(np.argmin(g))
        s_vec = np.zeros(n)
        s_vec[j] = 1.0
        
        # Nesterov step size
        gamma = 2.0 / (it + 2)
        
        # Update
        a_new = (1 - gamma) * a + gamma * s_vec
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = s_vec.copy()
        else:
            a_new = a_new / sa
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        if (it + 1) % 10000 == 0:
            print(f"    FW iter={it+1}: score={best_score:.8f}")
    
    return best_a, best_score

def nesterov_fw_smoothed(a0, time_limit, T=0.01, max_iters=200000):
    """Nesterov-accelerated FW with LSE-smoothed subgradient."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    a = softmax_from_z(z)
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # We'll use the smoothed gradient in z-space with mirror descent
    # Actually, let's use projected gradient in a-space
    a_prev = a.copy()
    t_nest = 1.0
    eta = 0.5  # step size
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = project_simplex(y)
        else:
            y = a
        
        # Compute smoothed gradient at y
        b = fast_convolve(y)
        shifted = b / T
        shifted -= shifted.max()
        exp_b = np.exp(shifted)
        w = exp_b / exp_b.sum()
        
        # Gradient
        grad = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w[k] * 2.0 * y[k - idx]
        
        # Projected gradient step
        a_new = project_simplex(y - eta * grad)
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        t_nest_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        
        # Adaptive restart
        if sc > best_score * 1.01:
            t_nest = 1.0
            eta *= 0.9
        else:
            t_nest = t_nest_new
        
        a_prev = a.copy()
        a = a_new
        
        # Decay step size
        eta = max(eta * 0.9999, 0.01)
        
        if (it + 1) % 5000 == 0:
            print(f"    FW-smooth iter={it+1}: score={best_score:.8f} T={T:.4f} eta={eta:.4f}")
    
    return best_a, best_score

def lse_lbgfs(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=20):
    """LSE L-BFGS with adaptive temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Adaptive number of iterations based on remaining time
        iters = max(5, min(50, int(remaining / (time_limit / n_stages) * 10)))
        
        def fun(z_val):
            obj, _ = lse_objective_grad_z(z_val, T_val, n)
            return obj
        
        def jac(z_val):
            _, g = lse_objective_grad_z(z_val, T_val, n)
            return g
        
        try:
            res = minimize(
                fun=fun, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-16, 'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new / sa
            
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            
            if (si + 1) % 4 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={T_val:.4f}: score={best_score:.8f}")
        except Exception as e:
            print(f"    LSE stage {si+1} failed: {e}")
    
    return best_a, best_score

                if remaining < 0:
                    continue
                others_sum = a_test.sum() - c
                if others_sum < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_test[mask] *= remaining / others_sum
                
                sc = score_normalized(a_test)
                if sc < best_sc:
                    best_sc = sc
                    best_cand = c
            
            if best_cand != a_i_old:
                a[i] = best_cand
                remaining = 1.0 - best_cand
                others_sum = a.sum() - best_cand
                if others_sum > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a[mask] *= remaining / others_sum
                improved = True
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_score

def multiplicative_weights_smoothed(a0, time_limit, T_start=1.0, T_end=0.01, max_iters=100000):
    """Multiplicative weights with LSE smoothing and adaptive step size."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    
    best_a = softmax_from_z(z)
    best_score = score_normalized(best_a)
    t0 = time.monotonic()
    
    T = T_start
    eta = 1.0
    n_stages = 30
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max_iters // n_stages
    
    for si, T_val in enumerate(temps):
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                    break
            
            a = softmax_from_z(z)
            b = fast_convolve(a)
            
            # Smoothed gradient
            shifted = b / T_val
            shifted -= shifted.max()
            exp_b = np.exp(shifted)
            w = exp_b / exp_b.sum()
            
            grad = np.zeros(n)
            for k in range(2*n - 1):
                if w[k] < 1e-15:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                grad[idx] += w[k] * 2.0 * a[k - idx]
            
            # Multiplicative update in z-space
            z = z - eta * grad
            a_new = softmax_from_z(z)
            
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                eta = min(eta * 1.01, 5.0)
            else:
                eta *= 0.95
                # Revert
                z = z + eta * grad  # undo
        
        if (si + 1) % 5 == 0:
            print(f"    MW stage={si+1}/{n_stages} T={T_val:.4f}: score={best_score:.8f} eta={eta:.3f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 1:
        a = np.exp(-0.5 * ((x - 0.3)/0.2)**2) + 0.3 * np.exp(-0.5 * ((x - 0.7)/0.15)**2) + 0.1
    elif choice == 2:
        a = 1.0 / (0.5 + x)
    elif choice == 3:
        a = np.abs(np.sin(np.pi * x)) ** 0.5
    elif choice == 4:
        # Random smooth
        freqs = rng.integers(1, 5, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        a = 1.0 + 0.3 * sum(np.sin(2*np.pi*f*x + p) for f, p in zip(freqs, phases))
    elif choice == 5:
        # Wigner-like
        a = np.exp(-2.0 * x * (1-x)) + 0.2
    elif choice == 6:
        # Peaks at rational positions
        a = 0.5 + 0.5 * np.exp(-20*(x-0.25)**2) + 0.3 * np.exp(-20*(x-0.5)**2) + 0.2 * np.exp(-20*(x-0.75)**2)
    elif choice == 7:
        # Golden ratio spacing
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.sqrt(2) * x)
    elif choice == 8:
        # Logarithmic
        a = np.log(1 + 10*x) + 0.1
    else:
        a = 1.0 + 0.4 * np.sin(4*np.pi*x) * np.cos(2*np.pi*x)
    
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
