    """
    Iteratively flatten the power spectrum of a.
    1. Compute FFT of a
    2. Modify |FFT(a)| to be flatter (toward geometric mean)
    3. Inverse FFT
    4. Project to non-negative
    5. Repeat
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    
    for it in range(n_iters):
        # Compute FFT
        A = np.fft.rfft(a, n=2*n)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: move toward geometric mean (keeping DC fixed)
        # DC component is fixed (it's the sum)
        mag_non_dc = mag[1:].copy()
        if mag_non_dc.max() > 0:
            # Target: geometric mean of non-DC magnitudes
            geom_mean = np.exp(np.mean(np.log(mag_non_dc + 1e-30)))
            # Blend toward flat
            mag_new = mag.copy()
            mag_new[1:] = (1 - 0.1) * mag[1:] + 0.1 * geom_mean
            # Keep DC

        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a = a_new
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L2 gradient descent warm-up (smooth quartic objective via FFT)
2. LP Nesterov acceleration (best-response iterations)
3. Spectral equalization (iterative power spectrum flattening)
4. Multi-length exploration with interpolation
5. QP active-set polishing and mirror descent refinement
6. Peak shaving with adaptive line search
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv)/sum^2 without calling evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def grad_l2_conv(a):
    """
    Gradient of sum_k(conv(a,a)_k)^2 with respect to a.
    Uses FFT: grad = 4 * IFFT(|FFT(a)|^2 * conj(FFT(a)))[:n]
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    # |A|^2 * conj(A) in real form:
    # A * conj(A)^2 in complex, but we use rfft/irfft
    # Full complex FFT approach:
    Af = np.fft.fft(a, N)
    R = np.fft.ifft(np.abs(Af) ** 2 * np.conj(Af), N).real
    return 4.0 * R[:n]


def l2_gradient_descent(a0, time_limit, lr=0.01, verbose=False):
    """
    Projected gradient descent on sum(conv(a,a)^2) s.t. sum(a)=1, a>=0.
    Uses backtracking line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        g = grad_l2_conv(a)
        # Project gradient onto tangent space of simplex
        g = g - g.mean()

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(20):
            a_new = a - step * g
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                break
            a_new = a_new / sn
            sc_new = compute_score(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
            a = a_new
            improved = True
            break
        if not improved:
            step *= 0.5
            if step < 1e-10:
                break
        else:
            # Occasionally increase step
            if it % 100 == 0:
                lr = min(lr * 1.1, 0.1)

        it += 1
        if verbose and it % 200 == 0:
            print(f"      L2-GD it={it}: score={best_score:.6f}, lr={lr:.6f}")

    return best_a, best_score


def spectral_equalization(a0, time_limit, n_iters=200, alpha=0.05):
    """
    Iteratively flatten the power spectrum while maintaining non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    N = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flatten: pull magnitudes toward their geometric mean (keep DC)
        mag_nc = mag[1:].copy()
        if mag_nc.max() > 1e-15:
            log_mag = np.log(mag_nc + 1e-30)
            target = np.exp(np.mean(log_mag))
            mag_new = mag.copy()
            mag_new[1:] = (1.0 - alpha) * mag[1:] + alpha * target
        else:
            break

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new, N).real[:n]

        # Soft projection: clip and renormalize
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            if alpha > 0.005:
                alpha *= 0.9
            # Try mixing
            for frac in [0.3, 0.5]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                sm = a_mix.sum()
                if sm < 1e-15:
                    continue
                a_mix = a_mix / sm
                sc_mix = compute_score(a_mix)
                if sc_mix < best_score:
                    best_score = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    break

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with adaptive schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        # Nesterov schedule
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        yns = y_new.sum()
        if yns < 1e-15:
            y_new = y.copy()
            yns = 1.0
        y_new = y_new / yns

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            asn = a_new.sum()
            if asn < 1e-15:
                a_new = y_new.copy()
                asn = 1.0
            a_new = a_new / asn

        y = y_new
        a = a_new

        # Track best
        sc_a = compute_score(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        else:
            if sc_a > best_score * 1.00001:
                stall += 1

        sc_y = compute_score(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        # Running average
        avg_a += a_new
        n_avg += 1
        if n_avg >= 10 and n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            sc_avg = compute_score(avg_check)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()
                stall = 0

        if t % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")

        if stall > 5000 and t > 500:
            break

    return best_a, best_score


def qp_refinement(a0, time_limit, n_max_active=40):
    """QP on active constraints for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for rnd in range(15):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        max_c = c.max()
        threshold = max_c * 0.995
        active = np.where(c >= threshold)[0]
        if len(active) > n_max_active:
            active = np.argsort(c)[::-1][:n_max_active]

        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            oth = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[oth])
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, max_iter=5000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            print(f"      QP r{rnd}: {best_score:.6f}")
        else:
            for frac in [0.3, 0.5, 0.7]:
                am = (1 - frac) * a + frac * a_new
                am = np.maximum(am, 0)
                sm = am.sum()
                if sm < 1e-15:
                    continue
                am = am / sm
                scm = compute_score(am)
                if scm < best_score:
                    best_score = scm
                    best_a = am.copy()
                    a = am
                    print(f"      QP mix {frac}: {best_score:.6f}")
                    break
            else:
                if rnd >= 3:
                    break

    return best_a, best_score


def mirror_descent(a0, time_limit, n_iters=3000):
    """Mirror descent (exponential gradient) on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        max_c = c.max()
        top_k = np.argsort(c)[::-1][:5]
        weights = np.exp((c[top_k] - max_c) * 10)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            oth = k - idx
            grad[idx] += 2 * w * a[oth]
            grad[oth] += 2 * w * a[idx]

        lr = 0.05 * (0.99 ** (it / 100.0))
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        sn = a_new.sum()
        if sn < 1e-30:
            break
        a_new = a_new / sn

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Adaptive peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        max_c = c.max()
        top_peaks = np.argsort(c)[::-1][:8]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            cs = np.argsort(contrib)[::-1][:3]

            for ci in cs:
                src = idx[ci]
                src_oth = k - src
                amt_base = min(a[src], a[src_oth] if src != src_oth else a[src])

                for frac in [0.005, 0.01, 0.02, 0.04]:
                    amt = amt_base * frac
                    if amt < 1e-15:
                        continue
                    # Try moving to several random positions
                    for _ in range(5):
                        dst = rng.integers(0, n)
                        a_tr = a.copy()
                        a_tr[src] = max(0, a_tr[src] - amt)
                        if src != src_oth:
                            a_tr[src_oth] = max(0, a_tr[src_oth] - amt * 0.5)
                        a_tr[dst] += amt
                        sr = a_tr.sum()
                        if sr < 1e-15:
                            continue
                        a_tr = a_tr / sr
                        sc = compute_score(a_tr)
                        if sc < best_score:
                            best_score = sc
                            best_a = a_tr.copy()
                            a = a_tr
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                # Random perturbation
                a_tr = a + rng.normal(0, 0.01, n)
                a_tr = np.maximum(a_tr, 0)
                sr = a_tr.sum()
                if sr > 1e-15:
                    a_tr = a_tr / sr
                    sc = compute_score(a_tr)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_tr.copy()
                        a = a_tr
                        stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi


def cosine_init(n, shape='raised'):
    """Generate a cosine-based initialization."""
    x = np.linspace(0, np.pi, n)
    if shape == 'basic':
        a = np.cos(x / 2) ** 2
    elif shape == 'raised':
        a = 0.5 * (1 + np.cos(x))
    elif shape == 'blackman':
        a = 0.42 + 0.5 * np.cos(x) + 0.08 * np.cos(2 * x)
    else:
        a = np.cos(x / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    1. Multi-start initialization
    2. L2 gradient descent warm-up
    3. LP Nesterov (main workhorse)
    4. Spectral equalization
    5. Multi-length exploration
    6. Final polishing (QP + mirror descent + peak shaving)
    """
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_start
            print(f"  *** BEST [{label}]: {sc:.6f} (n={len(a)}, t={el:.0f}s)")

    # === Phase 1: Multi-start initialization ===
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    n_parent = 1600

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            sc_p = evaluate_safe(parent)
            print(f"  Parent: n={n_parent}, score={sc_p:.6f}")
            candidates.append((parent, sc_p, 'parent'))
    except Exception:
        pass

    # Generate alternatives at parent length
    for shape in ['basic', 'raised', 'blackman']:
        a = cosine_init(n_parent, shape)
        sc = evaluate_safe(a)
        candidates.append((a, sc, f'cos_{shape}'))

    # A couple of other lengths
    for n_try in [n_parent - 200, n_parent + 200]:
        if n_try < 400:
            continue
        a = cosine_init(n_try, 'raised')
        sc = evaluate_safe(a)
        candidates.append((a, sc, f'cos_n{n_try}'))

    candidates.sort(key=lambda x: x[1])
    print(f"  Top candidates: {[(c[2], f'{c[1]:.4f}') for c in candidates[:4]]}")

    # Start with best candidate
    start_seq = candidates[0][0].copy()
    start_n = len(start_seq)
    try_update(start_seq, 'init')

    # === Phase 2: L2 Gradient Descent warm-up ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: L2 GD warm-up ({min(remaining*0.1, 150):.0f}s) ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 30:
        a_out, sc_out = l2_gradient_descent(start_seq, t2)
        try_update(a_out, 'l2_gd')
        if sc_out < compute_score(start_seq):
            start_seq = a_out.copy()

    # === Phase 3: LP Nesterov (main) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: LP Nesterov ({min(remaining*0.35, 600):.0f}s) ===")
    t3 = min(remaining * 0.35, 600.0)
    if t3 > 60:
        a_out, sc_out = nesterov_lp(start_seq, t3, solver, verbose=True)
        try_update(a_out, 'lp_main')

    # === Phase 4: Spectral Equalization ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: Spectral Equalization ({min(remaining*0.08, 100):.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 20:
        a_out, sc_out = spectral_equalization(best_seq, t4)
        try_update(a_out, 'spectral')

    # Second LP Nesterov round (shorter, from improved init)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4b: LP Nesterov round 2 ({min(remaining*0.15, 200):.0f}s) ===")
    t4b = min(remaining * 0.15, 200.0)
    if t4b > 40:
        a_out, sc_out = nesterov_lp(best_seq, t4b, solver, verbose=False)
        try_update(a_out, 'lp_r2')

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: Multi-length exploration ===")
    base_n = len(best_seq)
    lengths_to_try = [base_n - 100, base_n + 100, base_n + 200, base_n - 200]
    lengths_to_try = [l for l in lengths_to_try if l >= 400]

    for n_try in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 80:
            break
        t5 = min(remaining * 0.12, 120.0)
        if t5 < 30:
            continue
        print(f"  Trying n={n_try} ({t5:.0f}s)")
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out = nesterov_lp(init_try, t5, solver, verbose=False)
        try_update(a_out, f'lp_n{n_try}')

    # === Phase 6: Final polishing ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 6: Final polishing ({remaining:.0f}s) ===")

    # QP
    t6a = min(remaining * 0.2, 80.0)
    if t6a > 15:
        a_out, sc_out = qp_refinement(best_seq, t6a, n_max_active=30)
        try_update(a_out, 'qp_final')

    # Mirror descent
    remaining = time_budget - (time.monotonic() - t_start)
    t6b = min(remaining * 0.15, 60.0)
    if t6b > 10:
        a_out, sc_out = mirror_descent(best_seq, t6b)
        try_update(a_out, 'md_final')

    # Peak shaving
    remaining = time_budget - (time.monotonic() - t_start)
    t6c = min(remaining * 0.2, 80.0)
    if t6c > 10:
        a_out, sc_out = peak_shaving(best_seq, t6c, seed=123)
        try_update(a_out, 'peak_final')

    # Another spectral pass
    remaining = time_budget - (time.monotonic() - t_start)
    t6d = min(remaining * 0.1, 40.0)
    if t6d > 10:
        a_out, sc_out = spectral_equalization(best_seq, t6d, alpha=0.02)
        try_update(a_out, 'spectral_final')

    # Final LP Nesterov
    remaining = time_budget - (time.monotonic() - t_start)
    t6e = min(remaining * 0.2, 60.0)
    if t6e > 20:
        a_out, sc_out = nesterov_lp(best_seq, t6e, solver, verbose=False)
        try_update(a_out, 'lp_final')

    # Final peak shave
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 10:
        a_out, sc_out = peak_shaving(best_seq, min(remaining * 0.5, 30.0), seed=999)
        try_update(a_out, 'peak_very_final')

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)})")
    else:
        # Fallback
        best_seq = cosine_init(1000, 'raised')
        best_score = evaluate_safe(best_seq)
        print(f"\nFALLBACK: {best_score:.6f}")

    return [float(x) for x in best_seq]
