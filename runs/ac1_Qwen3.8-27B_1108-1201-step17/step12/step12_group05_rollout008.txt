```python
    # Start with uniform
    a = np.ones(n) / n
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Precompute the circulant FFT
    circ = np.zeros(fft_size)
    circ[0] = p[0]
    if n > 1:
        circ[1:n] = p[1:n]
        circ[n:] = p[1:n][::-1]
    C_fft = rfft(circ, fft_size)
    
    for it in range(n_inner):
        # Gradient: 2 * Q @ a
        Qa = irfft(C_fft * rfft(a, fft_size), fft_size)[:n]
        grad = 2.0 * Qa
        
        # Gradient step
        a = a - eta * grad
        
        # Project onto simplex
        a = proj_simplex(a)
    
    return a

def proj_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum a = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def proj_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum a = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    cond = u * arange > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


class ToeplitzOp:
    """Efficient Toeplitz matrix-vector product using FFT circulant embedding."""
    __slots__ = ('n', 'fft_size', 'circ_fft')

    def __init__(self, p):
        self.n = len(p)
        size = 2 * self.n - 1
        self.fft_size = 1 << (size - 1).bit_length()
        self._set_p(p)

    def _set_p(self, p):
        n = self.n
        if n == 1:
            self.circ_fft = np.array([p[0], 0.0])
            self.fft_size = 2
            return
        circ = np.zeros(self.fft_size, dtype=np.float64)
        circ[0] = p[0]
        circ[1:n] = p[1:n]
        circ[n:2*n-1] = p[1:n][::-1]
        self.circ_fft = rfft(circ, self.fft_size)

    def matvec(self, a):
        n = self.n
        a_padded = np.zeros(self.fft_size, dtype=np.float64)
        a_padded[:n] = a
        return irfft(self.circ_fft * rfft(a_padded, self.fft_size), self.fft_size)[:n]

        a_new = proj_simplex(a_new)

        a_y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_k = t_new
    return a

        peak_score = b_star.max()
        full_score = 2.0 * n * peak_score
        if full_score < best_score:
            best_score = full_score
            best_a = a_star.copy()

        peak_idx = int(np.argmax(b_star))
        lag = abs(peak_idx - (n - 1))
        e_k = np.zeros(n)
        e_k[lag] = 1.0

        alpha_t = 2.0 / (t + 2.0)
        p = (1.0 - alpha_t) * p + alpha_t * e_k

        if t == 1:
            a_new = a_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = a_star + momentum * (a_star - a)
        a_new = proj_simplex(a_new)

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a_prev = a
        a = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      QP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten power spectrum to reduce peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=20, lr=0.01):
    """LogSumExp-smoothed optimization with Adam optimizer and annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(20, min(200, int(remaining * 100 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = lse_obj_grad_z(z, stage_T)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            lr_t = lr / (1.0 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            if it % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def uniform_weighted_init(n, rng, alpha=1.0):
    """Create a smooth non-uniform initial distribution."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** alpha
    noise = rng.normal(0, 0.05, n)
    a = a + noise
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")

            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)

        print(f"Default init: n={n_base}, score={best_score:.8f}")

        n_base = len(best_seq)

    # Phase 1: Spectral shaping (quick, ~30s)

            best_seq = a_out.copy()

    if t2 > 30:

            print(f"  *** NEW BEST [qp_nesterov]: {best_score:.8f} (n={len(best_seq)})")

    print(f"\n=== Phase 8: Final peak shaving (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 80.0)
    if t8 > 5:
        a_out, _ = peak_shaving(best_seq, t8, seed=789)
        sc = evaluate_safe(a_out)

            best_seq = a_out.copy()

                best_seq = a_out.copy()
                print(f"  *** NEW BEST [alt_init_{alt_seed}]: {best_score:.8f} (n={len(best_seq)})")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0.0)

        print(f"\nFINAL SCORE: {final_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
