```python
def find_descent_direction(p, top_indices):
    n = len(p)
    K = len(top_indices)
    # Build gradient matrix: G[i, j] = p[top_indices[i] - j] (for valid j)
    G = np.zeros((K, n))
    for i, k in enumerate(top_indices):
        for j in range(n):
            if 0 <= k - j < n:
                G[i, j] = p[k - j]
    
    # LP: min t s.t. G @ d <= t, sum(d) = 0, -1 <= d <= 1
    # Using cvxpy
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        G @ d <= t,
        cp.sum(d) == 0,
        d >= -1,
        d <= 1
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve()
    if prob.status == 'optimal':
        return d.value, t.value
    return None, None
```

If t* < 0, we have a descent direction. We then do a line search.

```python
"""
```

```python
2. Active-set LP direction finding (finds optimal descent direction for top-peak max)
3. Nesterov-accelerated subgradient with simplex projection
4. Peak equalization via dual-weighted averaging
5. Multi-length exploration and interpolation-based transfer
6. uses parent sequence as primary starting point for fine-tuning

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_objective_grad(a, T, topk=0):
    """
    Compute LSE objective and gradient.
    If topk > 0, only use top-k peaks (faster but approximate).
    If topk = 0, use all peaks (exact).
    Returns (objective, grad_a) where grad_a is gradient w.r.t. a.
    """
    n = len(a)
    c = fast_autocorr(a)
    total_len = len(c)

    if topk > 0 and topk < total_len:
        top_idx = np.argsort(c)[::-1][:topk]
        work_idx = top_idx
        work_vals = c[top_idx]
    else:
        work_idx = np.arange(total_len)
        work_vals = c

    Tc = T * work_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    # Gradient: grad_a[j] = 2 * sum_i w[i] * a[work_idx[i] - j] for valid indices
    grad_a = np.zeros(n)
    for idx_i in range(len(work_idx)):
        K = work_idx[idx_i]
        wi = w[idx_i]
        if wi < 1e-15:
            continue
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wi * a[K - idx_range]

    return lse_val, grad_a


def full_lse_grad_z(z, T, topk=0):
    """LSE objective and gradient in z-space (log-parameterization)."""
    a = softmax_fn(z)
    obj, grad_a = lse_objective_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_lse_optimize(a0, time_limit, T_list, topk=0, lr_base=0.005):
    """
    Adam-optimized LSE with multi-stage temperature annealing.
    Uses z-space parameterization for simplex constraint handling.
    """
    n = len(a0)
    a_init = normalize_simplex(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, T_cur in enumerate(T_list):
        if time.monotonic() - t0 > time_limit - 10:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Learning rate: start moderate, decay across stages
        lr = lr_base * np.sqrt(n / 1600.0) * (0.85 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        # Iterations for this stage
        max_it = max(50, int(remaining * 2.5))
        stall = 0
        best_stage = best_score

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            obj, grad_z = full_lse_grad_z(z, T_cur, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Periodic evaluation
            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    best_stage = score
                    stall = 0
                else:
                    stall += 1

                # Reset to best if stalled too long
                if stall > 100:
                    z = np.log(np.maximum(best_a, 1e-30))
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    stall = 0

                if stall > 200:
                    break  # Move to next temperature stage

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        elapsed = time.monotonic() - t0
        print(f"    LSE stage {si}/{len(T_list)} (T={T_cur:.2f}, {elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def active_set_lp_refine(a0, time_limit, K=15, nesterov=True, seed=42):
    """
    Active-set LP direction finding for minimax fine-tuning.
    At each step, solve an LP to find the best direction that reduces
    all top-K peaks simultaneously. Uses Nesterov acceleration.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_simplex(a0)
    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_a = a.copy()

    # Find LP solver
    solver = None
    for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS

    iter_count = 0
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit - 10:
            break

        iter_count += 1

        # Compute autocorrelation and find top K peaks
        c = fast_autocorr(y)
        sorted_idx = np.argsort(c)[::-1]
        top_K = sorted_idx[:K]
        max_val = c[top_K[0]]

        # Build gradient matrix for top K peaks
        # G[i, j] = contribution of a[j] to peak at position top_K[i]
        # Gradient of c[k] w.r.t. a[j] = 2 * a[k-j] (for valid k-j)
        # For direction finding, we use the linearized version:
        # c[k] ≈ c[k]_current + 2 * sum_j a[k-j] * d[j]
        G = np.zeros((K, n))
        for i, k in enumerate(top_K):
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                G[i, idx_range] = a[k - idx_range]  # linearized contribution

        # LP: min t s.t. G @ d <= t, sum(d) = 0, -1 <= d <= 1
        try:
            d_var = cp.Variable(n)
            t_var = cp.Variable()
            constraints = [
                G @ d_var <= t_var,
                cp.sum(d_var) == 0,
                d_var >= -0.5,
                d_var <= 0.5
            ]
            prob = cp.Problem(cp.Minimize(t_var), constraints)
            prob.solve(solver=solver, max_iters=5000)

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                # Fall back to subgradient
                d_val = None
                t_val = 0.0
            else:
                d_val = np.asarray(d_var.value, dtype=np.float64)
                t_val = float(t_var.value)
        except Exception:
            d_val = None
            t_val = 0.0

        if d_val is None or t_val >= -1e-10:
            # No good descent direction found; try subgradient fallback
            if nesterov:
                alpha_t = 2.0 / (t + 2.0)
                y_new = (1 - alpha_t) * y + alpha_t * a
            # Subgradient step
            g = np.zeros(n)
            for k in top_K[:min(5, K)]:
                js = max(0, k - (n - 1))
                je = min(n - 1, k)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    g[idx_range] += 2.0 * a[k - idx_range]
            g = g / len(top_K[:min(5, K)])
            # Project onto tangent space
            g = g - np.mean(g)
            a = a - 0.01 * g
            a = normalize_simplex(a)
        else:
            # Use LP direction with line search
            step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            improved = False
            for alpha in step_sizes:
                a_trial = y + alpha * d_val
                a_trial = normalize_simplex(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if not improved:
                # Try smaller steps
                for alpha in [0.0005, 0.0002]:
                    a_trial = y + alpha * d_val
                    a_trial = normalize_simplex(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        # Nesterov update
        if nesterov:
            alpha_t = 2.0 / (t + 2.0)
            y_new = (1 - alpha_t) * y + alpha_t * a
            momentum = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
            y_trial = y_new + momentum * (y_new - y)
            y_trial = normalize_simplex(y_trial)
            y = y_trial
        else:
            y = a.copy()

        # Stall detection
        if not improved:
            stall += 1
        else:
            stall = 0

        if stall > 200:
            # Restart from best
            a = best_a.copy()
            y = best_a.copy()
            prev_a = a.copy()
            stall = 0
            # Perturb slightly
            noise = rng.normal(0, 0.001, n)
            a_pert = normalize_simplex(a + noise)
            y = a_pert.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, t_val={t_val:.6f}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, K=10, alpha_init=0.05):
    """
    Nesterov-accelerated subgradient method for minimax.
    Uses average of top-K peak gradients as subgradient direction.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    prev_a = a.copy()
    stall = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Compute top K peaks
        c = fast_autocorr(y)
        sorted_idx = np.argsort(c)[::-1][:K]
        max_val = c[sorted_idx[0]]

        # Subgradient: average of top-K gradient directions
        g = np.zeros(n)
        for k in sorted_idx:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                g[idx_range] += 2.0 * y[k - idx_range]
        g = g / K

        # Project onto tangent space of simplex
        g = g - np.mean(g)

        # Nesterov step
        alpha_t = alpha_init / (1 + 0.01 * t)
        y_new = (1 - alpha_t) * y + alpha_t * a
        momentum = max(0.0, (t - 1.0) / (t + 2.0))
        a_new = y_new - alpha_t * momentum * g
        a_new = normalize_simplex(a_new)

        y = y_new
        a = a_new

        # Evaluate
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Also check y
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if stall > 500:
            a = best_a.copy()
            y = best_a.copy()
            prev_a = a.copy()
            stall = 0

    return best_a, best_score


def peak_equalization(a0, time_limit, K=10, n_outer=100):
    """
    Iteratively equalize the top K peaks.
    At each step, compute the average of top-K peak values, then
    find a direction that reduces peaks above average and increases
    peaks below average (within the top K set).
    """
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit - 5:
            break

        c = fast_autocorr(a)
        sorted_idx = np.argsort(c)[::-1][:K]
        top_vals = c[sorted_idx]
        avg_val = top_vals.mean()
        max_val = top_vals[0]

        # If peaks are already well-equalized, stop
        if (max_val - top_vals.min()) / max(max_val, 1e-15) < 0.01:
            break

        # For peaks above average, compute "excess" gradient directions
        # For peaks below average, compute "deficit" gradient directions
        g_reduce = np.zeros(n)  # direction to reduce high peaks
        g_increase = np.zeros(n)  # direction to increase low peaks
        n_reduce = 0
        n_increase = 0

        for k in sorted_idx:
            if c[k] > avg_val:
                js = max(0, k - (n - 1))
                je = min(n - 1, k)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    g_reduce[idx_range] += 2.0 * a[k - idx_range]
                n_reduce += 1
            else:
                js = max(0, k - (n - 1))
                je = min(n - 1, k)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    g_increase[idx_range] -= 2.0 * a[k - idx_range]
                n_increase += 1

        if n_reduce == 0 or n_increase == 0:
            break

        g_reduce /= n_reduce
        g_increase /= n_increase

        # Combined direction: reduce high peaks, increase low peaks
        g = g_reduce + 0.5 * g_increase
        g = g - np.mean(g)  # Maintain sum constraint

        # Line search
        best_trial_score = best_score
        best_trial_a = a.copy()
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
            a_trial = a + alpha * (-g)  # Negative gradient = descent
            a_trial = normalize_simplex(a_trial)
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_trial_score:
                best_trial_score = score
                best_trial_a = a_trial

        if best_trial_score < best_score - 1e-14:
            a = best_trial_a
            best_score = best_trial_score
            best_a = a.copy()
        else:
            # No improvement; try smaller perturbation
            break

        if outer % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PEQ outer={outer} ({elapsed:.0f}s): score={best_score:.6f}, spread={max_val-top_vals.min():.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_nesterov_fast(a0, time_limit, solver, max_iters=20000):
    """
    Fast LP-based Nesterov: solves smaller LPs (top peaks only) for direction.
    This is a hybrid between the full LP approach and subgradient.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 10:
            break

        c = fast_autocorr(y)
        max_c = c.max()

        # Best response: find b that maximizes sum(b) s.t. conv(y, b) <= M
        # But we only need to consider the top peaks for the constraint
        top_m = min(30, 2 * n - 1)
        top_positions = np.argsort(c)[::-1][:top_m]
        M = max_c * 1.001  # Slightly above current max

        try:
            b_var = cp.Variable(n, nonneg=True)
            # conv(y, b)[k] = sum_j y[k-j] * b[j] for valid j
            # We only enforce constraints at top positions
            constraints = [cp.sum(b_var) == 1, b_var >= 0]
            for k in top_positions:
                js = max(0, k - (n - 1))
                je = min(n - 1, k)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    row = np.zeros(n)
                    row[idx_range] = y[k - idx_range]
                    constraints.append(row @ b_var <= M)

            prob = cp.Problem(cp.Minimize(M), [cp.sum(b_var) == 1, b_var >= 0])
            # Actually, let's just find max sum(b) subject to constraints
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            prob.solve(solver=solver, max_iters=3000)

            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                bs = b_val.sum()
                if bs > 1e-10:
                    b_val = b_val / bs
                else:
                    b_val = None
            else:
                b_val = None
        except Exception:
            b_val = None

        if b_val is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_val
        y_new = normalize_simplex(y_new)

        momentum = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        a_new = y_new + momentum * (y_new - y)
        a_new = normalize_simplex(a_new)

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if stall > 500:
            a = best_a.copy()
            y = best_a.copy()
            stall = 0

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPN t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LSE Adam, active-set LP,
    subgradient Nesterov, and peak equalization.
    Starts from parent sequence for fine-tuning.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1700.0)

    # Find LP solver
    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LSE Adam fine-tuning at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LSE Adam n={n_base} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        T_list = np.geomspace(50.0, 0.5, 15)
        a_out, _ = adam_lse_optimize(best_seq, t1, T_list, topk=0, lr_base=0.004)
        try_update(a_out, f"lse_full_n{n_base}")

    # === Phase 2: Active-set LP refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: Active-set LP n={n_base} ({remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_out, _ = active_set_lp_refine(best_seq, t2, K=12, nesterov=True, seed=123)
        try_update(a_out, f"aslp_n{n_base}")

    # === Phase 3: Subgradient Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: Subgrad Nesterov n={n_base} ({remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = subgradient_nesterov(best_seq, t3, K=10, alpha_init=0.03)
        try_update(a_out, f"subgrad_n{n_base}")

    # === Phase 4: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Peak Equilization n={n_base} ({remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = peak_equalization(best_seq, t4, K=12, n_outer=200)
        try_update(a_out, f"peq_n{n_base}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Multi-length ({remaining:.0f}s) ===")
    for n_try in [n_base - 50, n_base + 50, n_base + 100]:
        if time.monotonic() - t_global_start > time_budget - 120:
            break
        t_try = min(remaining * 0.15, 150.0)
        if t_try < 20:
            break
        init_try = interpolate_sequence(best_seq, n_try)
        # Quick LSE
        T_list_short = np.geomspace(30.0, 1.0, 8)
        a_out, _ = adam_lse_optimize(init_try, t_try * 0.6, T_list_short, topk=20, lr_base=0.004)
        try_update(a_out, f"lse_n{n_try}")
        # Quick ASLP
        t_aslp = min(remaining * 0.08, 60.0)
        if t_aslp > 15:
            a_out, _ = active_set_lp_refine(best_seq, t_aslp, K=10, nesterov=True, seed=456)
            # This refines at original length but from different state
            # Instead, let's do ASLP on the new length
            init_try2 = interpolate_sequence(a_out, n_try) if len(a_out) != n_try else a_out
            a_out2, _ = active_set_lp_refine(init_try2, t_aslp, K=10, nesterov=True, seed=789)
            try_update(a_out2, f"aslp_n{n_try}")

    # === Phase 6: Final LSE polish on best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 6: Final LSE polish ({remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 15:
        T_list_final = np.geomspace(10.0, 0.2, 10)
        a_out, _ = adam_lse_optimize(best_seq, t6, T_list_final, topk=0, lr_base=0.002)
        try_update(a_out, "final_lse")

    # === Phase 7: Final ASLP polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 7: Final ASLP polish ({remaining:.0f}s) ===")
    t7 = min(remaining * 0.35, 200.0)
    if t7 > 15:
        a_out, _ = active_set_lp_refine(best_seq, t7, K=8, nesterov=True, seed=1011)
        try_update(a_out, "final_aslp")

    # === Phase 8: Final peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8 = min(remaining * 0.6, 100.0)
        if t8 > 10:
            print(f"=== Phase 8: Final PEQ ({remaining:.0f}s) ===")
            a_out, _ = peak_equalization(best_seq, t8, K=8, n_outer=300)
            try_update(a_out, "final_peq")

    # === Phase 9: One more LSE with very low T ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t9 = min(remaining * 0.8, 80.0)
        if t9 > 10:
            T_list_ultra = np.geomspace(5.0, 0.1, 8)
            a_out, _ = adam_lse_optimize(best_seq, t9, T_list_ultra, topk=0, lr_base=0.001)
            try_update(a_out, "ultra_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
```
