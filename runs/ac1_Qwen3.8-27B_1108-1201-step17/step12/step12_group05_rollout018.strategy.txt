def fast_autocorr(a):
    """Calculate the autocorrelation using Fourier transforms."""
    n = len(a)
    # Ensure real-valued and length appropriate for full convolution
    if isinstance(a, np.ndarray):
        a_real = a
    else:
        a_real = np.array(a, dtype=np.float64)
    # Extend to proper FFT size
    size_needed = 2 * n - 1
    fft_size = 1 << (size_needed - 1).bit_length()
    # Pad and FFT
    A = np.fft.rfft(np.concatenate([a_real, np.zeros(fft_size - n)]), fft_size)
    B = np.conj(A)
    conv_result = np.fft.irfft(A * B, fft_size)[:size_needed]
    return conv_result
