"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start structured initialization
2. Nesterov-accelerated subgradient on LogSumExp (FFT-based, O(n log n) per iter)
3. LP best-response for global structure
4. Multi-scale coordinate descent with adaptive step sizes
5. Final dense adjacent-pair polish
Key innovation: analytical gradient of conv(k) as 2*shifted_reversed(a), enabling
very fast subgradient iterations; multi-scale systematic coordinate descent for final polish.
"""

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full self-convolution a*a_a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # First clip negatives
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) * target_sum / n
    # Scale to target
    return v * (target_sum / s)


def conv_gradient(a, k):
    """
    Gradient of conv(a,a)[k] w.r.t. a.
    Result: grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # For each j, grad[j] = 2 * a[k-j] if 0 <= k-j < n
    js = np.arange(n)
    idx = k - js
    valid = (idx >= 0) & (idx < n)
    grad[valid] = 2.0 * a[idx[valid]]
    return grad


def logsumexp_grad(a, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*conv_k(a))) w.r.t. a.
    Uses FFT for efficiency.
    Returns (objective_value, gradient).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    # grad = sum_k w[k] * grad_conv(a, k)
    # grad[j] = 2 * sum_k w[k] * a[k-j] (valid)
    # = 2 * (w * a_rev)[j + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv_result = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return obj_val, grad


def nesterov_subgradient_logsumexp(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=25):
    """
    Nesterov-accelerated subgradient descent on LogSumExp objective.
    Very fast: O(n log n) per iteration via FFT.
    """
    n = len(a0)
    a = normalize(a0)
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    y = a.copy()  # Nesterov "lookahead" point
    a_prev = a.copy()
    step_size_base = 0.01

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        T = T_start * (T_end / T_start) ** (stage / max(1, n_stages - 1))
        step_size = step_size_base / (1.0 + 0.1 * stage)
        n_iters = max(50, int(200 / (1.0 + 0.05 * stage)))

        for it in range(1, n_iters + 1):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute gradient at y
            _, grad_y = logsumexp_grad(y, T)
            # Project gradient to sum-0 subspace
            grad_y = grad_y - grad_y.mean()

            # Nesterov update
            alpha = 2.0 / (it + 2.0)
            y_new = y - alpha * step_size * grad_y
            y_new = project_simplex(y_new)

            # Extrapolation
            momentum = (it - 1.0) / (it + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

            # Track best
            score_new = 2.0 * n * fast_autocorr(a_new).max()
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()

            score_y = 2.0 * n * fast_autocorr(y_new).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y_new.copy()

            y = y_new
            a = a_new

        if stage % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NSL stage={stage} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)

def multisccale_cd(a0, time_limit, seed=42):
    """
    Multi-scale coordinate descent:
    - Large: random distant moves (1-5% mass)
    - Medium: random nearby moves (0.1-1% mass)
    - Small: systematic nearby passes
    - Fine: dense adjacent-pair optimization
    """

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac_time = elapsed / time_limit

        # Phase transitions based on time fraction
        if frac_time < 0.3:
            scale = 'large'
        elif frac_time < 0.6:
            scale = 'medium'
        elif frac_time < 0.85:
            scale = 'small'
        else:
            scale = 'fine'

        improved = False

        if scale == 'large':
            # Random distant moves
            n_trials = 20
            for _ in range(n_trials):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-8:
                    continue
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = project_simplex(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        elif scale == 'medium':
            # Random nearby moves (within 50 positions)
            n_trials = 50
            for _ in range(n_trials):
                i = rng.integers(n)
                offset = rng.integers(1, 50) * rng.choice([-1, 1])
                j = i + offset
                if j < 0 or j >= n or a[i] < 1e-9:
                    continue
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = project_simplex(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        elif scale == 'small':
            # Systematic: try every 10th position, move to nearby
            step = max(1, n // 200)
            for i in range(0, n, step):
                if a[i] < 1e-10:
                    continue
                for offset in [-3, -2, -1, 1, 2, 3, 5, 10]:
                    j = i + offset
                    if j < 0 or j >= n:
                        continue
                    for frac in [0.001, 0.005, 0.01]:
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = project_simplex(a_trial)
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break

        else:
            # Fine: dense adjacent-pair optimization (full pass)
            # Try moving small amounts between adjacent pairs
            for direction in [1, -1]:
                for i in range(0, n - 1, max(1, n // 500)):
                    j = i + 1
                    src, dst = (i, j) if direction == 1 else (j, i)
                    if a[src] < 1e-12:
                        continue
                    for frac in [0.0005, 0.001, 0.002, 0.005, 0.01]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = project_simplex(a_trial)
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            stall += 1
            # Try a random "kick" to escape

                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-6:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt

                    # Accept with probability based on how much worse it is
                    if score < best_score * 1.001 or rng.random() < 0.01:
                        if score < best_score * 1.005:
                            a = a_trial
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()
                            stall = 0
            if stall > 200 and scale == 'fine':
                break

        if improved:
            stall = 0

        if int(elapsed) % 30 < 1:
            print(f"      MSCD {elapsed:.0f}s ({frac_time:.0%}): best={best_score:.8f} scale={scale} stall={stall}")

    return best_a, best_score


def logsumexp_lbfgs(a0, time_limit, T_start=10.0, T_end=0.02, n_stages=15):
    """LogSumExp optimization with L-BFGS and annealing."""
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters_per_stage = max(10, min(40, int(15 + 25 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            a_tmp = softmax_fn(z_in)
            b_tmp = fast_autocorr(a_tmp)
            Tb_tmp = stage_T * b_tmp
            max_Tb = Tb_tmp.max()
            return (max_Tb + np.log(np.exp(Tb_tmp - max_Tb).sum())) / stage_T

        def jac(z_in):
            a_tmp = softmax_fn(z_in)
            _, grad_a = logsumexp_grad(a_tmp, stage_T)
            S = np.dot(grad_a, a_tmp)
            return a_tmp * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si} T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-x ** 2 * 5) + 0.1
    elif choice == 2:
        a = (1 - x) ** 2 + 0.01
    elif choice == 3:
        a = (1 - x) * (1 - x) * (1 + 0.3 * np.sin(10 * np.pi * x))
    elif choice == 4:
        a = np.exp(-3 * (x - 0.1) ** 2) + 0.3 * np.exp(-3 * (x - 0.5) ** 2) + 0.3 * np.exp(-3 * (x - 0.9) ** 2)
    elif choice == 5:
        a = x ** 0.5 + 0.1
    elif choice == 6:
        a = (1 + x) ** 2 + 0.01
    elif choice == 7:
        a = np.exp(-2 * (1 - x)) + 0.05
    elif choice == 8:
        a = (1 - 2 * x) ** 2 + 0.05
    elif choice == 9:
        a = np.sin(np.pi * x ** 0.7) ** 2 + 0.01
    elif choice == 10:
        a = np.exp(-4 * (x - 0.2) ** 2) + 0.5 * np.exp(-4 * (x - 0.8) ** 2) + 0.05
    else:
        a = 1.0 / (0.1 + x ** 1.5)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def dense_adjacent_polish(a0, time_limit, seed=0):
    """
    Dense adjacent-pair optimization: iterate through all adjacent pairs
    and find the optimal small transfer between each pair.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = 0
        # Multiple passes with decreasing step sizes
        for step_frac in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            if time.monotonic() - t0 > time_limit:
                break
            for i in range(n - 1):
                if a[i] < 1e-12 or a[i+1] < 1e-12:
                    continue
                # Try moving from i to i+1
                amt = a[i] * step_frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[i+1] += amt
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    pass_improved += 1
                    continue
                # Try moving from i+1 to i
                amt = a[i+1] * step_frac
                a_trial = a.copy()
                a_trial[i+1] -= amt
                a_trial[i] += amt
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    pass_improved += 1

        n_passes += 1
        if n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DAP pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f} improved={pass_improved}")
        if pass_improved == 0:
            break

    return best_a, best_score


def peak_shaving_targeted(a0, time_limit, seed=42):
    """
    Targeted peak shaving: identify the top peaks and move mass
    from their contributors to positions that minimize impact on all peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    K = 5  # Number of top peaks

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]
        b_max = b[top_k[0]]

        improved = False
        for k in top_k:
            # Find contributors to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            top_contrib = np.argsort(contrib)[::-1][:5]

            for tc in top_contrib:
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-10:
                    continue

                # Find positions with minimal contribution to all top peaks
                def impact_score(j):
                    total = 0.0
                    for k2 in top_k:
                        lo = max(0, k2 - n + 1)
                        hi = min(n - 1, k2)
                        if lo <= j <= hi:
                            partner = k2 - j
                            if 0 <= partner < n:
                                total += a[j] * a[partner]
                    return total + 0.001 * a[j]

                # Sample candidates
                candidates = rng.integers(n, size=min(50, n))
                candidates = np.array([c for c in candidates if c != src])
                if len(candidates) == 0:
                    continue
                candidates = candidates[np.argsort([impact_score(c) for c in candidates])[:10]]

                for dst in candidates[:5]:
                    dst = int(dst)
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                        amt = src_val * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = project_simplex(a_trial)
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            # Random exploration
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = project_simplex(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        break
            else:
                # No improvement even with random - try bigger kick
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-6:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = project_simplex(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score * 1.002:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                # Check if we should stop
                if time.monotonic() - t0 > time_limit * 0.9:
                    break

        if (int(time.monotonic() - t0)) % 30 < 1:
            print(f"      PST {time.monotonic()-t0:.0f}s: best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function. Multi-phase optimization exploiting the structure
    of the self-convolution: gradient of conv[k] is 2*shifted_reversed(a).
    Uses Nesterov subgradient (fast, O(n log n)), LP for global structure,
    and multi-scale coordinate descent for fine polish.
    """
    rnm = np.random.default_rng(12345)
    t_global_start = time.monotonic()
    time_budget = 1740.0  # Leave 60s buffer

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        # Try multiple structured initializations
        best_init = None
        best_init_score = float('inf')
        for i in range(10):
            init = structured_init(n_base, rnm)
            sc = 2.0 * n_base * fast_autocorr(init).max()
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init
        best_score = evaluate_safe(best_seq)
        print(f"Best init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov subgradient on LogSumExp (fast!) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov Subgradient LogSumExp at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, _ = nesterov_subgradient_logsumexp(best_seq, t1, T_start=50.0, T_end=0.1, n_stages=25)
        try_update(a_out, "nsl_p1")

    # === Phase 2: LP best-response for global structure ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LP Best-Response at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, max_iters=200)
        try_update(a_out, "nlp_p2")

    # === Phase 3: Another round of subgradient (lower T) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Subgradient refined at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = nesterov_subgradient_logsumexp(best_seq, t3, T_start=10.0, T_end=0.05, n_stages=20)
        try_update(a_out, "nsl_p3")

    # === Phase 4: Multi-scale coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Multi-scale CD at n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 30:
        a_out, _ = multisccale_cd(best_seq, t4, seed=77)
        try_update(a_out, "mscd_p4")

    # === Phase 5: Targeted peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Targeted peak shaving at n={n_base} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_out, _ = peak_shaving_targeted(best_seq, t5, seed=123)
        try_update(a_out, "pst_p5")

    # === Phase 6: LogSumExp L-BFGS polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: LogSumExp L-BFGS at n={n_base} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 10:
        a_out, _ = logsumexp_lbfgs(best_seq, t6, T_start=8.0, T_end=0.02, n_stages=12)
        try_update(a_out, "lse_p6")

    # === Phase 7: Dense adjacent polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Dense adjacent polish at n={n_base} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 250.0)
    if t7 > 10:
        a_out, _ = dense_adjacent_polish(best_seq, t7, seed=456)
        try_update(a_out, "dap_p7")

    # === Phase 8: Final multi-scale + peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final refinement (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t8a = min(remaining * 0.4, 100.0)
        if t8a > 10:
            a_out, _ = multisccale_cd(best_seq, t8a, seed=789)
            try_update(a_out, "mscd_final")

        remaining = time_budget - (time.monotonic() - t_global_start)
        t8b = min(remaining * 0.4, 80.0)
        if t8b > 10:
            a_out, _ = peak_shaving_targeted(best_seq, t8b, seed=321)
            try_update(a_out, "pst_final")

        remaining = time_budget - (time.monotonic() - t_global_start)
        t8c = min(remaining * 0.5, 60.0)
        if t8c > 5:
            a_out, _ = dense_adjacent_polish(best_seq, t8c, seed=654)
            try_update(a_out, "dap_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
