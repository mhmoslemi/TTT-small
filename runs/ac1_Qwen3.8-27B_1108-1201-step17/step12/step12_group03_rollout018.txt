import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, warm_b=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solve_kwargs = {}
    if warm_b is not None:
        b_var.value = warm_b
    
    try:
        prob.solve(solver=solver, **solve_kwargs)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def subgradient_descent(a0, time_limit, n_iters=20000, verbose=True):
    """
    Subgradient descent on the simplex for min max_k (a*a)_k.
    Uses the structure: grad of (a*a)_k at a is shift_k(a) + shift_-k(a).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    lr_base = 0.5
    decay = 0.9999
    lr = lr_base

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Compute weighted subgradient from top peaks
        top_k = min(8, 2 * n - 1)
        peak_indices = np.argsort(b)[::-1][:top_k]
        # Weights: exponential of (peak - max) / scale
        scale = max(max_b * 0.01, 1e-10)
        weights = np.exp((b[peak_indices] - max_b) / scale)
        weights = weights / weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            for i in range(n):
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    grad[i] += w * a[j1]
                if 0 <= j2 < n:
                    grad[i] += w * a[j2]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Step
        x = a - lr * grad / (n * max_b + 1e-15)

        # Project onto simplex {x: sum=1, x>=0}
        # Use bisection for tau
        x_sorted = np.sort(x)[::-1]
        cssv = np.cumsum(x_sorted)
        rho = np.nonzero(x_sorted * np.arange(1, n + 1) > (cssv - 1))[0][-1]
        tau = (cssv[rho] - 1) / (rho + 1.0)
        a_new = np.maximum(x - tau, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_s

        a = a_new
        lr *= decay

        if it % 2000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      SGD it={it} ({elapsed:.0f}s): best={best_score:.6f}, cur={score:.6f}")

        if it > 100 and abs(score - best_score) < 1e-10:
            # Stuck, try a kick
            if it % 100 == 0:
                rng = np.random.default_rng(it)
                a = best_a.copy()
                perturb_idx = rng.choice(n, size=max(1, n // 100), replace=False)
                for idx in perturb_idx:
                    a[idx] *= (1 + rng.uniform(-0.1, 0.1))
                a = np.maximum(a, 0)
                a_s = a.sum()
                if a_s > 1e-15:
                    a = a / a_s

    return best_a, best_score


def qp_refinement(a0, time_limit, n_active=40):
    """
    QP: min M s.t. (a*a)_k <= M for k in active_set, sum(a)=1, a>=0.
    Note: This is a non-convex QP in general, but cvxpy may handle it.
    As a fallback, we use the S-lemma / SOCP reformulation when possible.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b

    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        # Identify active lags (peaks near max)
        threshold = max_b * (1 - 0.005)
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(b)])
        if len(active_lags) > n_active:
            # Keep top n_active
            active_lags = active_lags[np.argsort(b[active_lags])[::-1]][:n_active]

        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(cp.multiply(a_var[idx], a_var[other_idx]))
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        solved = False
        for sol in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=sol, max_iter=20000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
            if round_num % 5 == 0:
                print(f"      QP round {round_num}: score={best_score:.6f}")
        else:
            # Try mixing
            improved = False
            for frac in [0.2, 0.4, 0.6]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    improved = True
                    break
            if not improved:
                if round_num >= 5:
                    break
                n_active = min(n_active * 2, 100)

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Coordinate descent peak shaving with line search.
    Move mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 300
    n_peaks = min(6, 2 * n - 1)

    iterations = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        # For each top peak, find the best mass reallocation
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            
            # Find the highest contributor
            ci = np.argmax(contrib)
            src = idx[ci]
            src_other = k - src
            
            # Find target: position that minimizes increase in max peak
            # Simple heuristic: move to a position with low current value
            # that's far from the peak indices
            candidates = np.argsort(a)[:min(20, n)]
            
            for dst in candidates:
                if dst == src and dst == src_other:
                    continue
                
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = min(a[src], a[src_other] if src != src_other else a[src]) * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    if src != src_other:
                        a_trial[src_other] -= amt * 0.5
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation every few stalls
            if stall % 20 == 0:
                n_perturb = rng.integers(2, max(3, n // 100))
                sources = rng.choice(n, size=n_perturb, replace=False)
                targets = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for src_i, dst_i in zip(sources, targets):
                    frac = rng.uniform(0.01, 0.05)
                    amt = a_trial[src_i] * frac
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s < 1e-15:
                    continue
                a_trial = a_trial / a_trial_s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > max_stall:
                break

        iterations += 1

    return best_a, best_score, iterations


def generate_asymmetric_exp(n, alpha, rng=None):
    """Generate an asymmetric exponential-like sequence."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.arange(n) / n
    a = np.exp(-alpha * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def generate_cosine_variant(n, phase, power, rng=None):
    """Generate a cosine-based variant."""
    x = np.arange(n) / n
    a = np.cos(np.pi * (x + phase)) ** 2
    a = np.maximum(a, 0) ** power
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization with LP Nesterov, subgradient descent,
    QP refinement, and peak shaving. Uses adaptive restarts and multi-start.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get initial best from parent
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # If no parent, create initial
    if best_seq is None:
        n_base = 1600
        best_seq = generate_asymmetric_exp(n_base, alpha=1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 0: Try a few alternative initializations (quick)
    print(f"\n=== Phase 0: Alternative initializations ===")
    alphas = [0.5, 1.0, 1.5, 2.0, 3.0]
    for alpha in alphas:
        a_init = generate_asymmetric_exp(n_base, alpha)
        sc = evaluate_safe(a_init)
        if sc < best_score:
            best_score = sc
            best_seq = a_init.copy()
            print(f"  Better init: alpha={alpha}, score={sc:.6f}")
    
    # Try cosine variants
    for phase in [0.0, 0.1, -0.1, 0.2]:
        for power in [1.0, 2.0, 0.5]:
            a_init = generate_cosine_variant(n_base, phase, power)
            sc = evaluate_safe(a_init)
            if sc < best_score:
                best_score = sc
                best_seq = a_init.copy()
                print(f"  Better init: phase={phase}, power={power}, score={sc:.6f}")

    # Phase 1: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t1, solver)
        print(f"  LP Nesterov done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Subgradient descent (new technique)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Subgradient Descent at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, sc_out = subgradient_descent(best_seq, t2, n_iters=30000)
        print(f"  SGD done: score={sc_out:.6f}")
        try_update(a_out, f"sgd_n{n_base}")

    # Phase 3: QP Refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: QP Refinement at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, sc_out = qp_refinement(best_seq, t3, n_active=30)
        print(f"  QP done: score={sc_out:.6f}")
        try_update(a_out, f"qp_n{n_base}")

    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak Shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_out, sc_out, _ = peak_shaving(best_seq, t4, seed=123)
        print(f"  Peak shave 1: score={sc_out:.6f}")
        try_update(a_out, "peak_shave_1")

    # Phase 5: Second round of LP Nesterov (exploit improvements)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: LP Nesterov Round 2 (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 30:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t5, solver, verbose=False)
        print(f"  LP round 2: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "lp_round2")

    # Phase 6: Try nearby lengths
    for n_try in [n_base - 200, n_base + 200]:
        if n_try < 200:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 90:
            break
        print(f"\n=== Phase 6: LP Nesterov at n={n_try} (remaining={remaining:.0f}s) ===")
        t6 = min(remaining * 0.20, 150.0)
        if t6 < 40:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp_optimize(init_try, t6, solver, verbose=False)
        print(f"  LP at n={n_try}: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_try}")

        # Quick QP
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 40:
            t_qp = min(remaining * 0.10, 60.0)
            if t_qp > 10:
                a_qp, sc_qp = qp_refinement(a_out, t_qp, n_active=20)
                try_update(a_qp, f"qp_n{n_try}")

    # Phase 7: Final polish - subgradient + peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final Polish (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t7a = min(remaining * 0.3, 60.0)
        if t7a > 10:
            a_out, sc_out = subgradient_descent(best_seq, t7a, n_iters=10000, verbose=False)
            try_update(a_out, "final_sgd")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t7b = min(remaining * 0.4, 60.0)
        if t7b > 10:
            a_out, sc_out, _ = peak_shaving(best_seq, t7b, seed=789)
            print(f"  Final peak shave: score={sc_out:.6f}")
            try_update(a_out, "peak_shave_final")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t7c = min(remaining * 0.5, 60.0)
        if t7c > 10:
            a_out, sc_out = qp_refinement(best_seq, t7c, n_active=20)
            try_update(a_out, "final_qp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
