def solve_lp_reduced(a_fixed, active_idx, solver):
    """Solve LP: min M s.t. (a*b)_k <= M for k in active_idx, sum(b)=1, b>=0"""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(b_var) == 1]
    for k in active_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # (a*b)_k = sum_{j=js}^{je} a[j] * b[k-j]
        # b index is k-j for j in [js, je]
        b_indices = k - np.arange(js, je + 1)
        a_coeffs = a_fixed[js:je + 1]
        constraints.append(a_coeffs @ b_var[b_indices] <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp_active(a0, time_limit, solver, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a
        
        b_new, M_lp = lp_best_response_active(y, solver)
        if b_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        # Adaptive restart
        if sc_new > best_score * 1.001 and t_nest > 10.0:
            t_nest = 1.0
            restart_count += 1
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new
        
        a = a_new
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_reduced(a_fixed, active_idx, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(b_var) == 1]
    for k in active_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        b_indices = k - np.arange(js, je + 1)
        a_coeffs = a_fixed[js:je + 1]
        constraints.append(a_coeffs @ b_var[b_indices] <= M_var)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_best_response_active(a_fixed, solver, max_cut_iter=25, max_active=200):
    n = len(a_fixed)
    b = np.ones(n) / n
    conv = fft_convolve(a_fixed, b)
    max_val = conv.max()
    threshold = max_val * 0.90
    active_idx = list(np.where(conv >= threshold)[0])
    if len(active_idx) < 10:
        threshold = max_val * 0.7
        active_idx = list(np.where(conv >= threshold)[0])
    active_set = set(active_idx)

    b_best = b.copy()
    M_best = float(max_val)

    for cut_iter in range(max_cut_iter):
        b_new, M_new = solve_lp_reduced(a_fixed, sorted(active_set), solver)
        if b_new is None:
            break
        full_conv = fft_convolve(a_fixed, b_new)
        max_full = float(full_conv.max())
        if max_full <= M_new + 1e-5:
            return b_new, M_new
        b_best = b_new
        M_best = max_full
        violated = np.where(full_conv > M_new + 1e-6)[0]
        for k in violated:
            active_set.add(int(k))
        if len(active_set) > max_active:
            active_list = sorted(active_set)
            conv_vals = full_conv[active_list]
            order = np.argsort(conv_vals)[::-1][:max_active - 50]
            active_set = set([active_list[i] for i in order])
            active_set.add(int(np.argmax(full_conv)))
    return b_best, M_best


def subgradient_nesterov(a0, time_limit, max_steps=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.5
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 8000 and step > 200:
            break
        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.98)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z
        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc = score_normalized(a_new)
        if sc < best_score * 0.9999:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.95
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.9995, 0.05)
    return best_a, best_score


def nesterov_lp_active(a0, time_limit, solver, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    last_print = t0
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if stagnant > 30:
            break
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a
        b_new, M_lp = lp_best_response_active(y, solver)
        if b_new is None:
            continue
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            if sc_new < best_score * 0.99999:
                stagnant = 0
            else:
                stagnant += 1
            best_score = sc_new
            best_a = a_new.copy()
        else:
            stagnant += 1
        if sc_new > best_score * 1.0005 and t_nest > 8.0:
            t_nest = 1.0
            restart_count += 1
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new
        a = a_new
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} restarts={restart_count} stag={stagnant}")
            last_print = elapsed
    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=35):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 80 if stage_T > 1.0 else 40
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc <= best_score + 1e-12:
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                z = z_new
            else:
                z = z_new
            if (si + 1) % 10 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={best_score:.8f}")
        except Exception:
            pass
    return best_a, best_score


def peak_targeted_refine(a0, time_limit, max_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve_step = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 8000:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.95
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.001, 0.1)
        else:
            eta *= 0.92
            if eta < 1e-12:
                break
    return best_a, best_score
