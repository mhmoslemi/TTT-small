"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive dual subgradient method with FFT-accelerated gradients
2. Adam optimizer with simplex projection
3. Temperature annealing for dual weight distribution
4. Periodic coordinate descent for local refinement
5. Multi-start with perturbation and length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_correlate(a, w):
    """Compute cross-correlation: result[j] = sum_m a[m] * w[m+j] for j=0..n-1"""
    n = len(a)
    total = n + len(w) - 1
    fft_size = 1 << (total - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    corr = np.fft.ifft(np.conj(A) * W).real
    return corr[:n]

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax_fn(x, T=1.0):
    x = x / T
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def score_func(a):
    """Compute 2*n*max(a*a)/(sum(a))^2"""
    n = len(a)
    c = fast_autocorr(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)

def dual_subgradient_optimize(a0, time_limit, seed=42):
    """
    Dual subgradient method with Adam and temperature annealing.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initial w: softmax of autocorrelation
    c = fast_autocorr(a)
    T_init = 1.0 / c.max() if c.max() > 0 else 1.0
    w = softmax_fn(c, T_init)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 1.0 / n
    
    best_a = a.copy()
    best_score = score_func(a)
    t0 = time.monotonic()
    
    # Temperature schedule: from high (flat) to low (peaked)
    T_start = 2.0 * c.max()  # initial: roughly flat
    T_end = 0.1 * c.max()
    total_iters = int(time_limit * 2000)  # estimate
    stall = 0
    stall_threshold = 500
    
    for it in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        # Temperature annealing
        progress = it / max(total_iters, 1)
        T_cur = T_start * (T_end / T_start) ** progress
        
        # Compute w = softmax(c / T_cur)
        c = fast_autocorr(a)
        w = softmax_fn(c, T_cur)
        
        # Compute gradient: grad = 2 * correlate(a, w)
        grad = 2.0 * fast_correlate(a, w)
        
        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)
        
        # Check score periodically (every 10 iterations to save time)
        if it % 10 == 0:
            sc = 2.0 * n * c.max()  # c is current autocorrelation, sum(a)=1
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        
        # Periodic restart if stalled
        if stall > stall_threshold and it > 100:
            # Perturb best and restart
            a = best_a.copy()
            noise = rng.normal(0, 0.01, n) * a
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            stall = 0
            # Increase learning rate slightly
            lr = lr * 1.1
            
        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DSG it={it} ({elapsed:.0f}s): best={best_score:.6f}, T={T_cur:.4f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, rng, max_rounds=50):
    """
    Coordinate descent: for each coordinate, try to move mass to reduce the max peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_func(a)
    t0 = time.monotonic()
    
    for round_idx in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        c = fast_autocorr(a)
        cmax = c.max()
        # Find peaks within 1% of max
        threshold = cmax * 0.99
        peak_indices = np.where(c >= threshold)[0]
        
        improved = False
        # For each peak, find the coordinate that contributes most
        for k in peak_indices[:10]:
            # Contribution of a[j] to c[k]: 2*a[j]*a[k-j] (or a[j]^2 if j=k/2)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            js_max = max(1, je - js + 1)
            idx = np.arange(js, min(je + 1, js + js_max))
            
            if len(idx) == 0:
                continue
            
            contrib = np.zeros(len(idx))
            for ii, j in enumerate(idx):
                k_minus_j = k - j
                if 0 <= k_minus_j < n:
                    if j == k_minus_j:
                        contrib[ii] = 2 * a[j] * a[j]
                    else:
                        contrib[ii] = 2 * a[j] * a[k_minus_j]
            
            if contrib.max() < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find safe destination: a coordinate that doesn't contribute much to any peak
            # Compute "danger" for each coordinate
            danger = np.zeros(n)
            for k2 in peak_indices[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j2 in range(js2, je2 + 1):
                    if 0 <= k2 - j2 < n:
                        danger[j2] += a[k2 - j2]
            
            # Safe positions: low danger and not the source
            safe = np.where((danger < danger.max() * 0.3) & (np.arange(n) != src))[0]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            
            # Pick destination with lowest a value among safe
            safe_sorted = safe[np.argsort(a[safe])]
            dst = safe_sorted[rng.integers(0, max(1, len(safe_sorted) // 3))]
            
            # Try moving some mass
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = score_func(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Random perturbation
            n_pert = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    amt = a_trial[i] * rng.uniform(0.001, 0.01)
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                sc = score_func(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of:
1. Nesterov-accelerated mirror descent with adaptive dual temperature (fast subgradient)
2. LP-based Nesterov best-response for exact refinement (warm-started)
3. Coordinate descent with peak-directed mass transfer for local polish
4. Multi-start diversity with structured perturbations and length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_correlate(a, w):
    """Cross-correlation: result[j] = sum_m a[m] * w[m+j] for j=0..n-1."""
    n = len(a)
    total = n + len(w) - 1
    fft_size = 1 << (total - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    corr = np.fft.ifft(np.conj(A) * W).real
    return corr[:n]

def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(x, T=1.0):
    """Numerically stable softmax."""
    if T <= 0:
        T = 1e-10
    x = np.asarray(x, dtype=np.float64) / T
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def score_func(a):
    """Compute 2*n*max(a*a)/(sum(a))^2 efficiently."""
    n = len(a)
    c = fast_autocorr(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def nesterov_mirror_descent(a0, time_limit, seed=42, lr_scale=3.0,
                            T_ratio_start=3.0, T_ratio_end=0.05,
                            stall_threshold=800, momentum_boost=1.5):
    """
    Nesterov-accelerated mirror descent for min_a max_k (a*a)_k on the simplex.
    
    Uses adaptive dual temperature and momentum for fast convergence.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c0 = fast_autocorr(a)
    c0max = c0.max()
    if c0max < 1e-15:
        c0max = 1.0

    # Initialize dual weights
    T_init = T_ratio_start * c0max
    w = softmax_fn(c0, T_init)

    # Nesterov state
    y = a.copy()
    a_prev = a.copy()

    # Adam state for a
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps_adam = 0.85, 0.999, 1e-8
    lr = lr_scale / n

    best_a = a.copy()
    best_score = 2.0 * n * c0max  # sum(a)=1 so score = 2*n*max(c)
    t0 = time.monotonic()

    # Estimate total iterations
    est_iters_per_sec = 3000.0
    total_est = int(time_limit * est_iters_per_sec)
    total_est = max(total_est, 1000)

    stall = 0
    check_interval = max(1, n // 200)

    for it in range(1, total_est + 1):
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        # Temperature annealing (with a floor to avoid numerical issues)
        progress = min(1.0, it / total_est)
        T_cur = c0max * T_ratio_start * (T_ratio_end / T_ratio_start) ** progress
        T_cur = max(T_cur, 1e-6 * c0max)

        # Compute autocorrelation of current y
        c_y = fast_autocorr(y)

        # Update dual weights w = softmax(c_y / T_cur)
        w = softmax_fn(c_y, T_cur)

        # Nesterov extrapolation
        if it > 1:
            momentum_coeff = (it - 1.0) / (it + 2.0)
            y_new = y + momentum_coeff * (y - a_prev)
            y_new = project_simplex(y_new)
        else:
            y_new = y.copy()

        # Compute gradient at y_new: grad = 2 * correlate(y_new, w)
        grad = 2.0 * fast_correlate(y_new, w)

        # Adam update
        t_adam += 1
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * grad * grad
        m_hat = m_adam / (1.0 - beta1 ** t_adam)
        v_hat = v_adam / (1.0 - beta2 ** t_adam)

        a_new = y_new - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = project_simplex(a_new)

        a_prev = y
        y = y_new
        a = a_new

        # Periodic score check
        if it % check_interval == 0:
            c_a = fast_autocorr(a)
            sc = 2.0 * n * c_a.max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                stall = 0
            elif it % (check_interval * 10) == 0:
                stall += 1

            # Also check y
            sc_y = 2.0 * n * c_y.max()
            if sc_y < best_score - 1e-14:
                best_score = sc_y
                best_a = y.copy()
                stall = 0

        # Adaptive restart
        if stall > stall_threshold // check_interval and it > total_est // 20:
            # Perturb and restart
            a = best_a.copy()
            y = best_a.copy()
            a_prev = best_a.copy()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            stall = 0
            lr = lr * momentum_boost
            # Slightly perturb
            noise = rng.normal(0, 0.005, n) * best_a
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            y = a.copy()
            a_prev = best_a.copy()

        # Progress logging
        if it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NMD it={it} ({elapsed:.0f}s): best={best_score:.6f}, T={T_cur:.6f}, lr={lr:.4f}")

    # Final check
    c_final = fast_autocorr(a)
    sc_final = 2.0 * n * c_final.max()
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def lp_nesterov_optimize(a0, time_limit, solver, max_iters=50000):
    """
    Nesterov-accelerated LP best-response. Each iteration solves an LP.
    Warm-started from a0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            break

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = project_simplex(y_new)

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Check scores
        sc_a = 2.0 * n * fast_autocorr(a).max()
        sc_y = 2.0 * n * fast_autocorr(y).max()
        if sc_a < best_score - 1e-14:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        elif sc_y < best_score - 1e-14:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restart
        if stall > 500 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            stall = 0

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LPN t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score
