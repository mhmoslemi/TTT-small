def fast_autocorr(a):

            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, LP_M={M_val:.10f}")

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_power_grad(a, p):
    """
    Compute gradient of (sum_k ((a*a)[k])^p)^(1/p) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    b = fast_autocorr(a)
    bp = b ** (p - 1)
    f_val = (b ** p).sum() ** (1.0 / p)
    if f_val < 1e-300:
        return np.zeros(n), 0.0

    w = bp / (bp.sum() + 1e-300)
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    grad = grad * f_val ** (1.0 - p)
    return grad, f_val


def power_smooth_optimize(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=15):
    """
    Power-smoothed optimization with L-BFGS and multi-stage p-increase.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    ps = np.geomspace(p_start, p_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_p in enumerate(ps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters_per_stage = max(15, min(80, int(20 + 50 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            a_s = softmax_fn(z_in)
            b = fast_autocorr(a_s)
            return (b ** stage_p).sum() ** (1.0 / stage_p)

        def jac(z_in):
            a_s = softmax_fn(z_in)
            grad_a, _ = compute_power_grad(a_s, stage_p)
            S = np.dot(grad_a, a_s)
            grad_z = a_s * (grad_a - S)
            return grad_z

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      POW stage={si}/p={stage_p:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization with LogSumExp."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.005, n_stages=25):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters_per_stage = max(10, min(100, int(15 + 60 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving_direct(a0, time_limit, seed=42, K=10):
    """
    Direct peak shaving: iteratively move mass from peak contributors to low-contribution positions.
    Uses top-K peaks and smart candidate scoring.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            top_contrib = np.argsort(contrib)[::-1][:5]
            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                peak_areas = set()
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_areas.update(range(js2, je2 + 1))

                candidates = [j for j in range(n) if j not in peak_areas and j != src]
                if not candidates:
                    candidates = [j for j in range(n) if j != src]
                if not candidates:
                    break

                def cand_score(j):
                    total = 0.0
                    for k2 in top_k:
                        js2 = max(0, k2 - n + 1)
                        je2 = min(n - 1, k2)
                        if js2 <= j <= je2:
                            partner = k2 - j
                            if 0 <= partner < n:
                                total += a[j] * a[partner]
                    return total + 0.005 * a[j]

                candidates.sort(key=cand_score)
                dst_pool = candidates[:max(1, min(15, len(candidates) // 4))]

                for dst in dst_pool[:5]:
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            n_trials = 8
            for _ in range(n_trials):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.001, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 500:
                break

        if stall > 0 and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PS stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mirror_descent_lift(a0, time_limit, seed=123):
    """
    Mirror descent (exponential weights) to escape local minima.
    Uses subgradient of max(a*a) with adaptive step size.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = 0.05
    T_smooth = 50.0

    for it in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient via smoothed max
        b = fast_autocorr(a)
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        a_rev = a[::-1]
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        if conv_len > fft_size:
            fft_size *= 2
        W = rfft(w, fft_size)
        A_rev = rfft(a_rev, fft_size)
        conv = irfft(W * A_rev, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Mirror descent step: a_new[i] proportional to a[i] * exp(-eta * grad[i])
        log_a = np.log(np.maximum(a, 1e-30))
        new_log_a = log_a - eta * grad
        new_log_a -= new_log_a.max()
        a_new = np.exp(new_log_a)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            eta = eta * 0.8
            a_trial = a.copy()
            new_log_a2 = np.log(np.maximum(a_trial, 1e-30)) - eta * grad
            new_log_a2 -= new_log_a2.max()
            a_trial = np.exp(new_log_a2)
            a_trial = a_trial / a_trial.sum()
            score2 = 2.0 * n * fast_autocorr(a_trial).max()
            if score2 < best_score:
                best_score = score2
                best_a = a_trial.copy()
                a = a_trial
            else:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score3 = 2.0 * n * fast_autocorr(a_trial).max()
                    if score3 < best_score:
                        best_score = score3
                        best_a = a_trial.copy()
                        a = a_trial

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD it={it} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    # Smooth slightly
    kernel = np.array([0.25, 0.5, 0.25])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth = f_smooth / ss
    else:
        f_smooth = f_interp / ss if ss > 0 else f_interp
    return f_smooth


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.sin(10 * np.pi * x))
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * abs(x - 0.5)) + 0.2 * np.sin(5 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    else:
        a = np.random.default_rng(42).dirichlet(np.ones(n) * 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def coordinate_descent_polish(a0, time_limit, seed=77):
    """
    Fast coordinate descent: for each coordinate, find optimal value.
    Cycles through coordinates, adjusting one at a time.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    # Precompute which lags each index participates in
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        improved = False

        # Try modifying random coordinates
        n_trials = 50
        for _ in range(n_trials):
            i = rng.integers(n)
            if a[i] < 1e-10:
                continue

            # Try different amounts to move from i to j
            j = rng.integers(n)
            if i == j:
                continue

            fracs = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]
            for frac in fracs:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    fracs.clear()
                    break
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 100:
                break
        else:
            stall = 0

        if stall > 0 and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function. Multi-phase optimization combining:
    - Nesterov LP best-response
    - LogSumExp and Power-smoothed gradient optimization
    - Direct peak shaving and coordinate descent
    - Mirror descent for escape
    - Multi-length exploration
    """
    rng = np.random.default_rng(2024)
    t_global_start = time.monotonic()
    time_budget = min(1800.0 - 60.0, 1720.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None
    n_base = 1600

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov LP at parent length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"nlp_n{n_base}")

    # === Phase 2: LogSumExp refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LSE at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_out, _ = logsumexp_optimize(best_seq, t2, T_start=80.0, T_end=0.005, n_stages=25)
        try_update(a_out, f"lse_n{len(best_seq)}")

    # === Phase 3: Power-smoothed optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Power-smooth at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 160.0)
    if t3 > 20:
        a_out, _ = power_smooth_optimize(best_seq, t3, p_start=8.0, p_end=400.0, n_stages=15)
        try_update(a_out, f"pow_n{len(best_seq)}")

    # === Phase 4: Direct peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_out, _ = peak_shaving_direct(best_seq, t4, seed=999, K=10)
        try_update(a_out, f"ps_n{len(best_seq)}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Multi-length exploration (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)

    candidate_lengths = []
    for offset in [-200, -100, 100, 200, 400, 600]:
        nl = n_best + offset
        if nl >= 600 and nl <= 4000:
            candidate_lengths.append(nl)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    time_per_length = max(20, min(remaining * 0.6 / max(1, len(candidate_lengths)), 200.0))

    for nl in candidate_lengths[:5]:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 40:
            break
        tl = min(time_per_length, remaining * 0.35)
        if tl < 20:
            continue

        print(f"  Exploring n={nl} (time={tl:.0f}s)")
        init_long = interpolate_sequence(best_seq, nl)

        # LSE at new length
        t_sub = tl * 0.6
        a_out, _ = logsumexp_optimize(init_long, t_sub, T_start=50.0, T_end=0.01, n_stages=15)
        try_update(a_out, f"lse_n{nl}")

        # Quick peak shaving
        remaining_sub = time_budget - (time.monotonic() - t_global_start)
        t_ps = max(5, min(tl * 0.3, remaining_sub * 0.1))
        if t_ps > 5:
            a_out2, _ = peak_shaving_direct(a_out, t_ps, seed=nl, K=8)
            try_update(a_out2, f"ps_n{nl}")

    # === Phase 6: Mirror descent escape ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Mirror descent at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.1, 80.0)
    if t6 > 10:
        a_out, _ = mirror_descent_lift(best_seq, t6, seed=456)
        try_update(a_out, f"md_n{len(best_seq)}")

    # === Phase 7: Final intensive polishing ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final polish (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)

    # Coordinate descent
    t7a = min(remaining * 0.25, 120.0)
    if t7a > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t7a, seed=111)
        try_update(a_out, f"cd_n{n_best}")

    # Another LSE pass with very low temperature
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7b = min(remaining * 0.25, 100.0)
    if t7b > 10:
        a_out, _ = logsumexp_optimize(best_seq, t7b, T_start=10.0, T_end=0.001, n_stages=20)
        try_update(a_out, f"lse2_n{n_best}")

    # Peak shaving again
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7c = min(remaining * 0.25, 100.0)
    if t7c > 10:
        a_out, _ = peak_shaving_direct(best_seq, t7c, seed=888, K=12)
        try_update(a_out, f"ps2_n{n_best}")

    # Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7d = min(remaining * 0.2, 80.0)
    if t7d > 5:
        a_out, _ = coordinate_descent_polish(best_seq, t7d, seed=222)
        try_update(a_out, f"cd2_n{len(best_seq)}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
