Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov subgradient pre-optimization in log-space
2. Nesterov-accelerated LP block-coordinate descent (main workhorse)
3. LSE-smoothed L-BFGS multi-stage annealing for fine polishing
4. Iterated local search with perturbation for escaping local optima
5. Length exploration and final verification
Key improvements: adaptive restart strategies, efficient Toeplitz construction,
smarter peak-targeting, diverse initializations, and better time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_global_deadline = 0.0
_lp_solver = None


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    # Build Toeplitz matrix efficiently
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 1.2 * (2.0 * x - 1.0)
    elif choice == 1:
        a = x**0.3 + 0.05
    elif choice == 2:
        a = 2.0 * np.exp(-4.0 * x) + 0.3
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.2) ** 2) + 0.1
    elif choice == 4:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * (x + 0.15)) + 0.2 * np.cos(5 * np.pi * x)
    elif choice == 5:
        a = np.exp(-10.0 * (x - 0.2) ** 2) + 0.7 * np.exp(-10.0 * (x - 0.8) ** 2) + 0.05
    elif choice == 6:
        a = (1.0 - x + 0.05) ** 0.7
    elif choice == 7:
        a = 0.6 + 0.5 * np.sin(3 * np.pi * x + 1.0)
    elif choice == 8:
        # Strongly asymmetric: high on left, low on right
        a = np.exp(-2.0 * x) + 0.1
    else:
        # Oscillating asymmetric
        a = 0.7 + 0.3 * np.cos(4 * np.pi * x) * (1.0 - x) + 0.1
    a = np.maximum(a, 0.005)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def subgradient_nesterov_fast(a0, time_limit, max_steps=500000):
    """Fast Nesterov subgradient in log-space with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 1.0 / np.sqrt(2.0 * n)
    last_improve = 0
    no_improve_count = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step > 100 and no_improve_count > 3000:
            break

        # Compute subgradient
        b = fast_convolve(a)
        max_b = b.max()
        # Use all peaks above 90% of max for better gradient
        threshold = max_b * 0.90
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Gradient in a-space
        grad_a = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(active)

        # Convert to z-space gradient (for softmax parameterization)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Nesterov extrapolation in z-space
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z

        # Adaptive step size
        eta_t = eta / np.sqrt(1.0 + 0.01 * step)
        z_new = z_extrap - eta_t * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
            no_improve_count = 0
            eta = min(eta * 1.001, 0.5)
        else:
            no_improve_count += 1
            if sc > best_score * 1.01 and t_nest > 5.0:
                t_nest = 1.0
                a_prev = a
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                a_prev = a
                a = a_new
                z = z_new
            eta = max(eta * 0.995, 0.01)

    return best_a, best_score


def nesterov_lp_main(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov LP with adaptive restarts and look-back."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    score_history = []
    last_window_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Adaptive restart with look-back window
        score_history.append(sc_new)
        if len(score_history) > 200:
            score_history.pop(0)
        window_best = min(score_history)

        if sc_new > last_window_best * 1.001 and t_nest > 10.0:
            # Stalled - restart with best found
            t_nest = 1.0
            a = best_a.copy()
            a_prev = best_a.copy()
            restart_count += 1
            score_history = []
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new
        if len(score_history) >= 50:
            last_window_best = min(score_history)

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LSE-smoothed objective and gradient in z-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    exp_sum = exp_Tb.sum()
    w = exp_Tb / exp_sum
    log_sum_exp = max_Tb + np.log(exp_sum)
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient with respect to a
    grad_a = np.zeros(n)
    # Only process significant weights
    significant = w > 1e-15
    for k in np.where(significant)[0]:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs_polish(a0, time_limit, T_start=80.0, T_end=0.02, n_stages=25):
    """Multi-stage LSE annealing with L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        # More iterations for early stages (higher T = smoother = easier)
        iters = 60 if si < n_stages // 3 else (30 if si < 2 * n_stages // 3 else 15)
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a, best_score


def peak_targeted_refine(a0, time_limit, max_steps=50000):
    """Multiplicative weights targeting top peaks with adaptive threshold."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.08
    threshold_frac = 0.88
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step > 100 and step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * threshold_frac
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Gradient toward reducing peaks
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.003, 0.2)
            threshold_frac = max(threshold_frac - 0.0005, 0.80)
        else:
            eta *= 0.9
            threshold_frac = min(threshold_frac + 0.001, 0.95)
            if eta < 1e-10:
                break

    return best_a, best_score


def iterated_local_search(a0, time_limit, solver, n_perturbations=20):
    """Perturb the best solution and re-optimize to escape local optima."""
    n = len(a0)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    time_per_pert = time_limit / n_perturbations

    for pi in range(n_perturbations):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        t_alloc = min(time_per_pert, remaining * 0.5)

        # Generate perturbation
        perturbation_type = rng.integers(0, 3)
        a_pert = best_a.copy()
        if perturbation_type == 0:
            # Add noise
            noise = rng.normal(0, 0.05, n)
            a_pert = a_pert + noise
        elif perturbation_type == 1:
            # Multiplicative perturbation
            a_pert = a_pert * np.exp(rng.normal(0, 0.1, n))
        else:
            # Mix with a random structured sequence
            a_struct = structured_init(n, rng)
            a_pert = 0.7 * a_pert + 0.3 * a_struct

        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert / s

        # Short LP optimization
        a_out, sc_out = nesterov_lp_main(a_pert, t_alloc, solver=solver, max_iters=100000)
        if a_out is not None:
            sc = score_normalized(a_out)
            if sc < best_score:
                best_score = sc
                best_a = a_out.copy()
                print(f"    ILS improve: {sc:.8f} (pert {pi+1})")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"LP solver: {solver}, deadline in {budget_s-20:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)}) at {time.monotonic() - (_global_deadline - (budget_s-20)):.0f}s")

    # === Phase 1: Fast subgradient Nesterov (60s) ===
    print(f"\n=== Phase 1: Subgradient Nesterov (score={best_score:.8f}) ===")
    t1 = min(60.0, (_global_deadline - time.monotonic()) * 0.05)
    if t1 > 5:
        a_out, _ = subgradient_nesterov_fast(best_seq, t1)
        try_update(a_out, "subgrad_p1")
    print(f"  After P1: {best_score:.8f}")

    # === Phase 2: Nesterov LP main (700s) ===
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(700.0, remaining * 0.45)
    if t2 > 30:
        a_out, _ = nesterov_lp_main(best_seq, t2, solver=solver)
        try_update(a_out, "nest_lp_main")
    print(f"  After P2: {best_score:.8f}")

    # === Phase 3: LSE L-BFGS polish (200s) ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(200.0, remaining * 0.15)
    if t3 > 20:
        a_lse = lse_lbfgs_polish(best_seq, t3, T_start=60.0, T_end=0.02, n_stages=20)
        try_update(a_lse, "lse_p1")
    print(f"  After P3: {best_score:.8f}")

    # === Phase 4: Nesterov LP round 2 (250s) ===
    print(f"\n=== Phase 4: Nesterov LP r2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.18)
    if t4 > 20:
        a_out2, _ = nesterov_lp_main(best_seq, t4, solver=solver)
        try_update(a_out2, "nest_lp_r2")
    print(f"  After P4: {best_score:.8f}")

    # === Phase 5: Iterated local search (250s) ===
    print(f"\n=== Phase 5: ILS (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(250.0, remaining * 0.2)
    if t5 > 30:
        a_ils, _ = iterated_local_search(best_seq, t5, solver, n_perturbations=15)
        try_update(a_ils, "ils")
    print(f"  After P5: {best_score:.8f}")

    # === Phase 6: Peak-targeted refinement (100s) ===
    print(f"\n=== Phase 6: Peak-targeted (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6 = min(100.0, remaining * 0.1)
    if t6 > 10:
        a_pt, _ = peak_targeted_refine(best_seq, t6)
        try_update(a_pt, "peak_target")
    print(f"  After P6: {best_score:.8f}")

    # === Phase 7: Final LSE + LP (120s) ===
    print(f"\n=== Phase 7: Final refinement (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        t7a = min(60.0, remaining * 0.5)
        if t7a > 10:
            a_lse2 = lse_lbfgs_polish(best_seq, t7a, T_start=30.0, T_end=0.01, n_stages=15)
            try_update(a_lse2, "lse_final")
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            t7b = min(50.0, remaining * 0.6)
            if t7b > 10:
                a_out3, _ = nesterov_lp_main(best_seq, t7b, solver=solver, max_iters=50000)
                try_update(a_out3, "lp_final")
    print(f"  After P7: {best_score:.8f}")

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 40:
        for n_try in [best_n - 50, best_n + 50, best_n + 100, best_n - 100]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try < 200:
                continue
            t8 = min(30.0, remaining * 0.3)
            if t8 < 8:
                break
            print(f"  Trying n={n_try} (budget={t8:.0f}s)")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_main(init_try, t8, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")
    print(f"  After P8: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
