            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cvxpy_solver = name
                return name
        except Exception:
            continue
    _cvxpy_solver = 'scipy'
    return 'scipy'


def build_conv_matrix_dense(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_cvxpy(a, solver=None):
    if solver is None or solver == 'scipy':
        return solve_lp_scipy(a)
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_dense(a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return solve_lp_scipy(a)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return solve_lp_scipy(a)
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver='scipy', max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    stale_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
