            no_improve_count += 1
        if sc_new > best_score * 1.001 and t_nest > 5.0:
            t_nest = 1.0
        elif dist < 1e-9 * n:
            t_nest = 1.0
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        a = a_new
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count} no_improve={no_improve_count}")
    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve_auto(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.03, n_stages=25):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        stage_T = temps[si]

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 50 if si < n_stages // 3 else 30
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={best_score:.8f}")
        except Exception:
            pass
    return best_a, best_score


def coordinate_descent(a0, time_limit, max_rounds=5000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    b = fast_convolve_auto(a)
    best_round = 0
    for rd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if rd - best_round > 50:
            break
        any_improve = False
        for i in range(n):
            if a[i] < 1e-15:
                continue
            for frac in [-0.03, -0.01, 0.01, 0.03]:
                delta = frac * a[i]
                if a[i] + delta < 0:
                    continue
                new_b = b + 2.0 * delta * np.zeros(2 * n - 1)
                ks = np.arange(max(0, i), min(2 * n - 1, i + n))
                new_b[ks] = b[ks] + 2.0 * delta * a[ks - i]
                if 2 * i < 2 * n - 1:
                    new_b[2 * i] += delta * delta
                new_max = new_b.max()
                new_score = 2.0 * n * new_max
                if new_score < best_score - 1e-12:
                    a[i] += delta
                    b = new_b
                    best_score = new_score
                    best_a = a.copy()
                    any_improve = True
                    best_round = rd
                    break
        if not any_improve and rd > 0:
            break
        if (rd + 1) % 100 == 0:
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
                b = fast_convolve_auto(a)
    return best_a


def subgradient_fast(a0, time_limit, max_steps=200000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.5
    last_improve = 0
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break
        b = fast_convolve_auto(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.90)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z
        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.85
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.9995, 0.01)
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.8 * (2.0 * x - 1.0)
    elif choice == 1:
        a = x + 0.1
    elif choice == 2:
        a = 2.0 * np.exp(-3.0 * x) + 0.3
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.2
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (x + 0.1)) + 0.15 * np.cos(4 * np.pi * x)
    else:
        a = 0.8 + 0.4 * np.sin(2 * np.pi * x + 0.5)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    best_seq = None
    best_score = float('inf')
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            elapsed = time.monotonic()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # Phase 1: Fast subgradient warmup
    print(f"\n=== Phase 1: Subgradient warmup n={n_base} score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t1 = min(60.0, remaining * 0.05)
    if t1 > 5:
        a_out, _ = subgradient_fast(best_seq, t1, max_steps=300000)
        try_update(a_out, "subgrad_p1")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # Phase 2: Main LP-Nesterov with separation oracle
    print(f"\n=== Phase 2: LP-Nesterov (separation) n={len(best_seq)} score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(750.0, remaining * 0.50)
    if t2 > 30:
        a_out, _ = nesterov_lp_sep(best_seq, t2, solver=solver, max_iters=500000)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # Phase 3: LSE L-BFGS
    print(f"\n=== Phase 3: LSE L-BFGS score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(300.0, remaining * 0.20)
    if t3 > 20:
        a_lse = lbgfs_logsumexp(best_seq, t3, T_start=80.0, T_end=0.02, n_stages=25)
        try_update(a_lse, "lse_p3")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # Phase 4: Coordinate descent
    print(f"\n=== Phase 4: Coordinate descent score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(200.0, remaining * 0.15)
    if t4 > 10:
        a_cd = coordinate_descent(best_seq, t4, max_rounds=10000)
        try_update(a_cd, "coord_descent")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # Phase 5: Second LP-Nesterov round
    print(f"\n=== Phase 5: LP-Nesterov round 2 score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(200.0, remaining * 0.15)
    if t5 > 15:
        a_out2, _ = nesterov_lp_sep(best_seq, t5, solver=solver, max_iters=200000)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # Phase 6: Final LSE + coordinate descent
    print(f"\n=== Phase 6: Final refinement score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        t6a = min(80.0, remaining * 0.4)
        if t6a > 10:
            a_lse2 = lbgfs_logsumexp(best_seq, t6a, T_start=30.0, T_end=0.01, n_stages=15)
            try_update(a_lse2, "lse_final")
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            t6b = min(80.0, remaining * 0.5)
            if t6b > 5:
                a_cd2 = coordinate_descent(best_seq, t6b, max_rounds=5000)
                try_update(a_cd2, "cd_final")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # Phase 7: Multi-length exploration
    print(f"\n=== Phase 7: Length exploration score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        candidates = [n_best - 50, n_best + 50, n_best + 100, n_best - 100]
        for n_try in candidates:
            remaining = _global_deadline - time.monotonic()
            if remaining < 25:
                break
            if n_try < 100:
                continue
            t7 = min(35.0, remaining * 0.35)
            if t7 < 10:
                break
            print(f"  Trying n={n_try}, budget={t7:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_sep(init_try, t7, solver=solver, max_iters=100000)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 7 done: best={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
