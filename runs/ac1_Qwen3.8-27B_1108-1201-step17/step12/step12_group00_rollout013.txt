def fast_autocorr(a):
    """Compute linear autocorrelation (full convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_autocorr(b, n):
    """Compute the score given the autocorrelation and sequence length."""
    return 2.0 * n * b.max()


def normalize(a, target_sum=1.0):
    """Normalize array to have the target sum, keeping non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s

def lp_best_response(a, M, solver, warm_start=None):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build convolution matrix T (2n-1) x n
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = j + n
        T[k_start:k_end, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64), 1.0)
    
    best_a = a.copy()
    best_score = score_from_autocorr(fast_autocorr(a), n)
    
    y = a.copy()
    x = a.copy()  # Nesterov extrapolated point
    avg = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    iter_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute M from current y
        b_y = fast_autocorr(y)
        M = b_y.max()
        if M < 1e-15:
            break
        
        # LP best response
        b = lp_best_response(y, M, solver)
        if b is None:
            continue
        
        # Normalize b to sum 1
        g = normalize(b, 1.0)
        iter_count += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, 1.0)
        
        if t == 1:
            x_new = g
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            x_new = normalize(x_new, 1.0)
        
        y = y_new
        x = x_new
        
        # Check scores
        for candidate in [y, x]:
            s = score_from_autocorr(fast_autocorr(candidate), n)
            if s < best_score:
                best_score = s
                best_a = candidate.copy()
        
        # Running average
        avg += x
        n_avg += 1
        if n_avg % 50 == 0:
            avg_candidate = normalize(avg / n_avg, 1.0)
            s = score_from_autocorr(fast_autocorr(avg_candidate), n)
            if s < best_score:
                best_score = s
                best_a = avg_candidate.copy()
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={iter_count} ({elapsed:.0f}s): best_score={best_score:.8f}")
    
    return best_a, best_score
