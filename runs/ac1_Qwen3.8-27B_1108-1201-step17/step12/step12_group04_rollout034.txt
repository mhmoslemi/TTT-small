2. Multi-stage temperature annealing

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full self-convolution using FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_grad_a(a, T):
    """Compute gradient of soft-max objective w.r.t. a, plus the objective value."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # = 3n-2

    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    
    return obj_val, grad_a, w, b

def adam_step(z, grad_z, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """One step of Adam optimizer. In-place updates to m and v."""
    m[:] = beta1 * m + (1 - beta1) * grad_z
    v[:] = beta2 * v + (1 - beta2) * (grad_z ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    z[:] -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v

def interpolate_seq(a, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result

def structured_init(n, rng):

    elif choice == 3:

        a = np.maximum(a, 0)
    elif choice == 4:

        a = np.ones(n)
        s = n

To reduce it, we want to make the sequence "anti-symmetric" around the center (i.e., `a_i` and `a_{n-1-i}` should be uncorrelated). But since `a >= 0`, we can't make them negatively correlated. The best we can do is make one side small and the other side large, which reduces the paired product.

This suggests a "sawtooth" or "ramp" shape might be good: `a_i ∝ (i+1)` or `a_i ∝ (n-i)`. Let's check:

For `a_i = (i+1)/S` where `S = n(n+1)/2`:
`(a*a)_{n-1} = sum_i a_i * a_{n-1-i} = sum_i [(i+1)/(n(n+1)/2)] * [(n-i)/(n(n+1)/2)]`

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on soft-max (LogSumExp) objective with softmax parameterization
2. Multi-stage temperature annealing for gradual peak isolation
3. Greedy coordinate descent with peak-aware mass transfer
4. Fine-grained micro-step polishing
5. Multi-length exploration with interpolation transfer
6. Direct subgradient projected gradient descent as alternative
Key: Direct optimization of the true max-convolution objective (not game relaxation).

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Full self-convolution using FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_grad_a(a, T):
    """Compute gradient of soft-max objective w.r.t. a, plus objective value."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    # = 2 * (w * a_rev)[n-1+j]
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    return obj_val, grad_a, w, b


def adam_update(z, grad_z, m, v, t, lr, beta1, beta2, eps):
    """Adam optimizer step, in-place."""
    m[:] = beta1 * m + (1.0 - beta1) * grad_z
    v[:] = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z[:] -= lr * m_hat / (np.sqrt(v_hat) + eps)


def optimize_adam_soft(a0, time_budget, T_start, T_end, n_stages, lr_base, seed):
    """Adam optimization of soft-max objective on softmax parameterization."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    temps = np.geomspace(T_start, T_end, n_stages)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for si in range(n_stages):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2:
            break

        stage_T = temps[si]
        # More iterations at lower T (sharper peak, need more refinement)
        progress = si / max(1, n_stages - 1)
        iters = max(30, int(80 + 120 * progress))

        # Learning rate: larger early, smaller late
        lr = lr_base * (1.0 - 0.6 * progress)

        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0

        for it in range(iters):
            if time.monotonic() - t0 > time_budget:
                break

            a = softmax_fn(z)
            obj_val, grad_a, w, b = compute_grad_a(a, stage_T)
            grad_z = a * (grad_a - np.dot(grad_a, a))

            t_adam += 1
            adam_update(z, grad_z, m, v, t_adam, lr, beta1, beta2, eps)

            if (it + 1) % 20 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        # End of stage check
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam stage={si}/{n_stages} T={stage_T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_projected(a0, time_budget, lr_start, n_iters_max):
    """
    Subgradient method with projection onto simplex.
    Minimizes max(a*a) directly using active subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    best_b_val = fast_autocorr(a).max()

    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        # Subgradient: 2*a[k_max - j] for valid j
        k_max = int(np.argmax(b))
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        for j in range(js, je + 1):
            partner = k_max - j
            if 0 <= partner < n:
                grad[j] = 2.0 * a[partner]

        # Polyak step size approximation
        eta = lr_start * best_b_val / (max(best_b_val - 1.0 / (2 * n - 1), 1e-10) + 1e-10)
        eta = min(eta, 0.1)

        a_new = a - eta * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            best_b_val = fast_autocorr(a_new).max()
        else:
            # Decrease learning rate
            lr_start *= 0.95

        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad it={it+1} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def greedy_cd(a0, time_budget, seed, n_top_peaks=5, max_stall=300):
    """
    Greedy coordinate descent: find best single-pair mass transfer to reduce max autocorr.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_budget and stall < max_stall:
        b = fast_autocorr(a)
        b_max = b.max()
        top_k = np.argsort(b)[::-1][:n_top_peaks]

        # Identify source candidates (positions contributing to top peaks)
        src_set = set()
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                p = k - j
                if 0 <= p < n and a[j] > 1e-7:
                    src_set.add(j)

        src_list = list(src_set)
        if not src_list:
            stall += 1
            continue

        # Score sources by their contribution to max peak
        def src_score(j):
            sc = 0.0
            for k in top_k:
                if max(0, k - n + 1) <= j <= min(n - 1, k):
                    p = k - j
                    if 0 <= p < n:
                        sc += a[j] * a[p]
            return sc

        src_list.sort(key=src_score, reverse=True)
        src_list = src_list[:20]

        # Identify good targets (low contribution to peaks)
        def tgt_score(j):
            sc = 0.0
            for k in top_k:
                if max(0, k - n + 1) <= j <= min(n - 1, k):
                    p = k - j
                    if 0 <= p < n:
                        sc += a[j] * a[p]
            return sc + 0.005 * a[j]

        all_positions = list(range(n))
        tgt_scores = np.array([tgt_score(j) for j in all_positions])
        top_targets = all_positions[np.argsort(tgt_scores)[:15]]

        improved = False
        for src in src_list:
            if a[src] < 1e-8:
                continue
            for dst in top_targets:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    b_trial = fast_autocorr(a_trial)
                    score = 2.0 * n * b_trial.max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Random kicks
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-6:
                        frac = float(rng.uniform(0.01, 0.15))
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss > 1e-15:
                            a_trial = a_trial / ss
                            score = 2.0 * n * fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break
            if stall > max_stall:
                break

        if stall > 0 and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(a, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 12))

            try_update(a_out, f"multi_n{nl}")

    # === Phase 6: Fine tuning ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Fine tuning at n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t6 = remaining * 0.25
    a_out, _ = fine_tune_cd(best_seq, t6, seed=777)
    try_update(a_out, "fine1")

    # === Phase 7: Final Adam with very low T ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Adam low-T at n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.25
    a_out, _ = optimize_adam_soft(best_seq, t7, T_start=3.0, T_end=0.005,
                                   n_stages=15, lr_base=0.01, seed=3)
    try_update(a_out, "adam_final")

    # === Phase 8: Final fine tune ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 8: Final fine tune (budget={remaining:.0f}s) ===")
        t8 = remaining * 0.5
        a_out, _ = fine_tune_cd(best_seq, t8, seed=999)
        try_update(a_out, "fine_final")

    # Final verification
    if best_seq is not None:
        final_check = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE: {final_check:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
