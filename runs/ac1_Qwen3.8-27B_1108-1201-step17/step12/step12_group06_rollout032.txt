import numpy as np

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def max_conv_value(a):
    """Return max of convolution (without the 2n factor)."""
    b = fast_convolve(a)
    return b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:

        if len(a_list) == 0:

        if sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None

def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:

    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)

    last_print = t0

        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, score_normalized(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution once per sweep for coefficient extraction
        b_cur = fast_convolve(a)
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[i]
            if x_old < 1e-12:
                # Try a small value
                x_try = 1e-10
                a[i] = x_try
                a /= a.sum()
                sc = score_normalized(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = x_try
                    a /= a.sum()
                continue

            S = 1.0 - x_old  # sum of other elements

            # Compute c, d, e vectors for the 1D problem
            # (a*a)[k] = c_k + d_k*x + e_k*x^2 (unnormalized, with a[i]=x, others at current values)
            k_range = np.arange(2 * n - 1)
            ki = k_range - i

            e_vec = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                e_vec[2 * i] = 1.0

            valid = (ki >= 0) & (ki < n) & (ki != i)
            d_vec = np.zeros(2 * n - 1)
            d_vec[valid] = 2.0 * a[ki[valid]]

            c_vec = b_cur.copy()
            c_vec[valid] -= d_vec[valid] * x_old
            if 2 * i < 2 * n - 1:
                c_vec[2 * i] -= x_old * x_old

            # Golden section search on g(x) = max_k (c_k + d_k*x + e_k*x^2) / (x+S)^2
            # Search range: [0, x_hi]

            def g_eval(x_val):
                num = c_vec + d_vec * x_val + e_vec * x_val * x_val
                den = (x_val + S) ** 2
                return num.max() / den

            g_old = g_eval(x_old)

            # Determine search bounds
            x_lo = 0.0
            x_hi = min(x_old * 3.0, 1.0)
            g_hi = g_eval(x_hi)
            g_lo = g_eval(x_lo)

            # Expand if needed
            for _ in range(8):
                x_try = min(x_hi * 1.5, 1.0)
                if x_try - x_hi < 1e-10:
                    break
                g_try = g_eval(x_try)
                if g_try < g_hi:
                    x_lo = x_hi
                    g_lo = g_hi
                    x_hi = x_try
                    g_hi = g_try
                else:
                    break

            # Also check shrinking
            for _ in range(8):
                x_try = x_lo * 1.5
                if x_try >= x_hi:
                    break
                g_try = g_eval(x_try)
                if g_try < g_lo:
                    x_hi = x_lo
                    g_hi = g_lo
                    x_lo = x_try
                    g_lo = g_try
                else:
                    break

            # Golden section search
            if x_hi - x_lo < 1e-12:
                x_opt = (x_lo + x_hi) * 0.5
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                x1 = x_hi - gr * (x_hi - x_lo)
                x2 = x_lo + gr * (x_hi - x_lo)
                g1 = g_eval(x1)
                g2 = g_eval(x2)

                for _ in range(50):
                    if g1 < g2:
                        x_hi = x2
                        x2 = x1
                        g2 = g1
                        x1 = x_hi - gr * (x_hi - x_lo)
                        g1 = g_eval(x1)
                    else:
                        x_lo = x1
                        g1 = g2
                        x1 = x_hi - gr * (x_hi - x_lo)
                        g1 = g_eval(x1)
                        x2 = x_lo + gr * (x_hi - x_lo)
                        g2 = g_eval(x2)

                x_opt = (x_lo + x_hi) * 0.5

            # Apply update
            a[i] = x_opt
            a /= a.sum()
            sc = score_normalized(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                improved = True

        # Periodic perturbation to escape local minima
        if perturbed_needed(sweep, perturb_every) and best_score > 1.502:
            if time.monotonic() - t0 < time_limit - 5:
                idx = rng.integers(0, n)
                a[idx] = max(0, a[idx] + perturb_scale * rng.normal())
                a = np.maximum(a, 0)
                a /= a.sum()
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if not improved:
            # Try one more sweep with tighter tolerance, then stop
            if sweep >= 5:
                break

        if (sweep + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep={sweep+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def perturbed_needed(sweep, period):
    """Helper for perturbation scheduling."""
    return (sweep + 1) % period == 0 and sweep > 0


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(T_val * (b - max_b))
    w = exp_b / exp_b.sum()
    log_sum_exp = max_b + np.log(exp_b.sum()) / T_val
    obj = 2.0 * n * log_sum_exp

    # Gradient of w.r.t. a: for each active k, grad_a gets contribution from a[k-j]
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k] * T_val
        if wk < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]
    # Divide by sum(exp) for the LSE gradient
    grad_a /= exp_b.sum() * T_val

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsbfgs_lse(a0, time_limit, T_start=50.0, T_min=0.5, n_stages=20, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            if best_score < 2.0:
                rel_gap = (sc - best_score) / (best_score + 1e-15)
                if rel_gap > 0.002:
                    stagnant += 1
                else:
                    stagnant = max(0, stagnant - 1)
            else:
                stagnant = 0

            if stagnant >= 4:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.85)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.1 * np.cos(11 * x)
    elif choice == 3:
        a = np.ones(n) * (1.0 + 0.15 * np.cos(2 * np.pi * 5 * np.arange(n) / n))
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.08) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.03) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.03)
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.25 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def peak_subgradient(a0, time_limit, n_steps=300000):
    """Peak-targeted subgradient with Polyak step and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.005:
            lo_e, hi_e = 0.0, eta
            for _ in range(8):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 3000:
            avg_count = 3000
            avg_a = (avg_a * 3000 + a) / 3001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s")

    # Get initial sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP (35% time) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_base} score={best_score:.8f} ===")
    remaining = global_deadline - time.monotonic()
    t1 = 0.35 * remaining
    a_out, _ = nesterov_lp(best_seq, t1, solver=solver)
    try_update(a_out, "nesterov_main")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: Coordinate Descent Polish (30% time) ===
    print(f"\n=== Phase 2: CoordDesc Polish score={best_score:.8f} ===")
    remaining = global_deadline - time.monotonic()
    t2 = 0.30 * remaining
    a_cd, _ = coordinate_descent_polish(best_seq, t2, max_sweeps=5000, perturb_every=30, perturb_scale=0.03)
    try_update(a_cd, "coord_descent")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: LSE refinement (15% time) ===
    print(f"\n=== Phase 3: LSE score={best_score:.8f} ===")
    remaining = global_deadline - time.monotonic()
    t3 = 0.15 * remaining
    if t3 > 20:
        a_lse, _ = lsbfgs_lse(best_seq, t3 * 0.6, T_start=30.0, T_min=0.5, n_stages=15, iters_per_stage=35)
        try_update(a_lse, "lse")
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_lse2, _ = lsbfgs_lse(best_seq, remaining * 0.6, T_start=15.0, T_min=0.3, n_stages=10, iters_per_stage=30)
            try_update(a_lse2, "lse2")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Peak Subgradient (10% time) ===
    print(f"\n=== Phase 4: PeakSG score={best_score:.8f} ===")
    remaining = global_deadline - time.monotonic()
    t4 = 0.10 * remaining
    if t4 > 15:
        a_sg, _ = peak_subgradient(best_seq, t4, n_steps=300000)
        try_update(a_sg, "subgrad")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Multi-length exploration (5% time) ===
    print(f"\n=== Phase 5: Multi-length score={best_score:.8f} ===")
    remaining = global_deadline - time.monotonic()
    t5_total = 0.05 * remaining
    if t5_total > 20:
        for n_try in [n_base + 200, n_base - 200, n_base + 400]:
            remaining = global_deadline - time.monotonic()
            if remaining < 15:
                break
            t5 = min(t5_total / 3.0, remaining * 0.45)
            if t5 < 10:
                break
            print(f"  Trying n={n_try}")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t5 * 0.5, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")
            if remaining - t5 > 10:
                a_try2, _ = coordinate_descent_polish(a_try, (remaining - t5) * 0.6, max_sweeps=500)
                try_update(a_try2, f"len_{n_try}_cd")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Final combined polish (5% time) ===
    print(f"\n=== Phase 6: Final polish score={best_score:.8f} ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 10:
        # Short coordinate descent
        t6a = remaining * 0.5
        a_f1, _ = coordinate_descent_polish(best_seq, t6a, max_sweeps=2000, perturb_every=0, perturb_scale=0)
        try_update(a_f1, "final_cd")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Short Nesterov
            a_f2, _ = nesterov_lp(best_seq, remaining * 0.6, solver=solver, max_iters=30000)
            try_update(a_f2, "final_nesterov")

            remaining = global_deadline - time.monotonic()
            if remaining > 8:
                # Final quick CD
                a_f3, _ = coordinate_descent_polish(best_seq, remaining * 0.7, max_sweeps=1000, perturb_every=0, perturb_scale=0)
                try_update(a_f3, "final_cd2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
