def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T @ b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS

def solve_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:

                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def solve_multi_peak_lp(a, solver, K=10):

    b = fast_autocorr(a)
    top_k = np.argsort(b)[::-1][:K]

    # = sum_i delta[i] * a[n-1-(k-i)] for valid indices

    T_top = T[top_k, :]  # K x n
    
    delta_var = cp.Variable(n)
    t_var = cp.Variable()
    b_top = b[top_k]
    
    constraints = [
        cp.sum(delta_var) == 0,
        delta_var >= -a,
        b_top + 2.0 * (T_top @ delta_var) <= t_var
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):

                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):

    delta = np.asarray(delta_var.value, dtype=np.float64)

        if b_star is None:

        if t == 1:
            a = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a = y + momentum * (y - (y - alpha_t * (y - b_star)))

            # x_new = y_new + momentum*(y_new - y)

            a = y + momentum * (y - ((1-alpha_t)*y + alpha_t*(y - b_star)))

            # Simplify: y - (y - alpha_t*(y-b_star)) = alpha_t*(y-b_star)

            a = normalize(a)
        
        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()

            best_a = a.copy()

def lse_mirror_descent(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15):
    """
    LogSumExp optimization using mirror descent (exponentiated gradient)
    with Nesterov momentum.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t_start = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t_start > time_limit:
            break
        remaining_stage = time_limit - (time.monotonic() - t_start)
        if remaining_stage < 1:
            break
        
        iters = int(min(200, max(20, remaining_stage / 0.05)))
        
        # Learn a good step size
        eta = 1.0 / T  # initial guess
        
        a_mom = a.copy()
        a_prev = a.copy()
        
        for it in range(iters):
            if time.monotonic() - t_start > time_limit:
                break
            
            # Compute LSE gradient
            b = fast_autocorr(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # grad = 2 * (w * a_rev)
            a_rev = a[::-1]
            fft_size = 1 << (n + 2*n - 2 - 1).bit_length()
            W = rfft(w, fft_size)
            Ar = rfft(a_rev, fft_size)
            conv = irfft(W * Ar, fft_size)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(max(0.0, min(1000.0, x))) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T @ b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

    T = build_conv_matrix(a_fixed)

        cp.sum(delta_var) == 0,

            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    delta = np.asarray(delta_var.value, dtype=np.float64)
    return delta


def phase_nesterov_lp(a0, time_limit, solver):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t_start = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    fail_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t_start > time_limit:
            break

        b_star = solve_best_response(y, solver)
        if b_star is None:
            fail_count += 1
            if fail_count > 10:
                break
            continue
        fail_count = 0

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = normalize(y_new)

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = normalize(x_new)

        y = y_new
        x_prev = x_new

        # Track best
        score_x = 2.0 * n * fast_autocorr(x_new).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x_new.copy()
        score_y = 2.0 * n * fast_autocorr(y_new).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()

        # Running average
        avg_a += x_new
        n_avg += 1
        if n_avg >= 30 and n_avg % 60 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 300 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    NLP iter={t} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lse_gradient(a, T):
    """Compute gradient of LSE objective (1/T)*log(sum exp(T*b_k)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # grad = 2 * (w * a_rev) sliced appropriately
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def phase_lse_gradient(a0, time_limit, T_start=80.0, T_end=0.3, n_stages=20):
    """LogSumExp optimization with projected gradient + Nesterov momentum."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t_start = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t_start > time_limit:
            break
        remaining_stage = time_limit - (time.monotonic() - t_start)
        if remaining_stage < 1:
            break

        n_iters = int(min(300, max(30, remaining_stage / 0.03)))
        eta = 0.5 / T

        y = a.copy()
        x_prev = a.copy()

        for it in range(n_iters):
            if time.monotonic() - t_start > time_limit:
                break

            grad, w = lse_gradient(y, T)

            # Gradient step
            x_new = y - eta * grad
            x_new = simplex_project(x_new)

            # Nesterov extrapolation
            if it == 0:
                y_new = x_new.copy()
            else:
                beta_t = (it) / (it + 2.0)
                y_new = x_new + beta_t * (x_new - x_prev)
                y_new = simplex_project(y_new)

            x_prev = x_new
            y = y_new

            # Adapt eta based on progress
            if it > 0 and it % 20 == 0:
                score_check = 2.0 * n * fast_autocorr(x_new).max()
                if score_check > best_score * 1.01:
                    eta *= 0.7

            score = 2.0 * n * fast_autocorr(x_new).max()
            if score < best_score:
                best_score = score
                best_a = x_new.copy()

        elapsed = time.monotonic() - t_start
        if si % 4 == 0:
            print(f"    LSE stage={si}/T={T:.3f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def phase_multi_peak(a0, time_limit, solver):
    """Iteratively shave top K peaks using LP with adaptive K."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t_start = time.monotonic()
    stall = 0

    for it in range(500):
        if time.monotonic() - t_start > time_limit:
            break

        # Adaptive K: start large, reduce as we converge
        improvement_gap = best_score - 2.0 * n * (fast_autocorr(a).sum() / (2 * n - 1))
        K = min(20, max(3, int(20 * min(1.0, improvement_gap / 0.5))))

        delta = solve_multi_peak_lp(a, solver, K=K)

        if delta is not None:
            a_new = a + delta
            a_new = normalize(a_new)
        else:
            # Fallback: best response
            b_star = solve_best_response(a, solver)
            if b_star is None:
                stall += 1
                if stall > 20:
                    break
                continue
            a_new = 0.3 * a + 0.7 * b_star
            a_new = normalize(a_new)

        score = 2.0 * n * fast_autocorr(a_new).max()

        if score < best_score - 1e-14:
            a = a_new
            best_score = score
            best_a = a.copy()
            stall = 0
            if it % 15 == 0:
                elapsed = time.monotonic() - t_start
                print(f"    MPS iter={it} ({elapsed:.0f}s): score={best_score:.8f}")
        else:
            stall += 1
            if stall > 40:
                break

    return best_a, best_score


def phase_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent with peak-aware moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t_start = time.monotonic()
    stall = 0

    while time.monotonic() - t_start < time_limit:
        b = fast_autocorr(a)
        # Get top peaks for awareness
        top_peaks = np.argsort(b)[::-1][:5]

        improved = False
        n_trials = 200

        for _ in range(n_trials):
            if time.monotonic() - t_start > time_limit:
                break

            # Pick source from a peak-contributing position
            k_peak = top_peaks[rng.integers(len(top_peaks))]
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            if je < js:
                continue
            src = int(rng.integers(js, je + 1))

            if a[src] < 1e-10:
                continue

            # Pick destination: prefer positions that contribute less to peaks
            dst = int(rng.integers(n))
            if dst == src:
                continue

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Try larger random moves
            for _ in range(10):
                if time.monotonic() - t_start > time_limit:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = float(rng.uniform(0.01, 0.08))
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = normalize(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 100:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, seed):
    """Generate a structured initialization."""
    rng = np.random.default_rng(seed)
    choice = seed % 6
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.sin(8 * np.pi * x))
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Main search function. Multi-phase optimization combining:
    - Nesterov LP best-response (workhorse)
    - LogSumExp projected gradient
    - Multi-peak LP shaving
    - Peak-aware coordinate descent
    - Multi-length exploration
    """
    t_global = time.monotonic()
    time_budget = 1730.0  # Safety margin

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        print("Using structured initialization")
        best_seq = structured_init(n_base, 42)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Nesterov LP (workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: Nesterov LP (time={t1:.0f}s) ===")
    a_out, _ = phase_nesterov_lp(best_seq, t1, solver)
    try_update(a_out, "nlp")

    # === Phase 2: LSE Gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.22
    print(f"\n=== Phase 2: LSE Gradient (time={t2:.0f}s) ===")
    a_out, _ = phase_lse_gradient(best_seq, t2, T_start=60.0, T_end=0.3, n_stages=18)
    try_update(a_out, "lse_grad")

    # === Phase 3: Multi-peak LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.18
    print(f"\n=== Phase 3: Multi-peak LP (time={t3:.0f}s) ===")
    a_out, _ = phase_multi_peak(best_seq, t3, solver)
    try_update(a_out, "mps")

    # === Phase 4: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.12
    print(f"\n=== Phase 4: Coordinate Descent (time={t4:.0f}s) ===")
    a_out, _ = phase_coord_descent(best_seq, t4, seed=42)
    try_update(a_out, "cd1")

    # Round 2 of refinement on current best
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.15
    print(f"\n=== Phase 5: Refinement round 2 (time={t5:.0f}s) ===")

    # LSE with lower temperature
    sub_t = t5 * 0.4
    a_out, _ = phase_lse_gradient(best_seq, sub_t, T_start=15.0, T_end=0.1, n_stages=10)
    try_update(a_out, "lse_r2")

    # Multi-peak again
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_out, _ = phase_multi_peak(best_seq, remaining * 0.4, solver)
        try_update(a_out, "mps_r2")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length (remaining={remaining:.0f}s) ===")

    n_best = len(best_seq)
    offsets = [200, -200, 400]

    for offset in offsets:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        nl = n_best + offset
        if nl < 1000 or nl > 2600:
            continue
        t_sub = remaining * 0.35
        print(f"  Trying n={nl} (time={t_sub:.0f}s)")
        init = interpolate_sequence(best_seq, nl)
        a_out, _ = phase_nesterov_lp(init, t_sub * 0.6, solver)
        if a_out is not None:
            # Quick LSE polish
            rem_sub = t_sub * 0.35
            if rem_sub > 5:
                a_out2, _ = phase_lse_gradient(a_out, rem_sub, T_start=30.0, T_end=0.5, n_stages=8)
                try_update(a_out2, f"lse_n{nl}")
            try_update(a_out, f"nlp_n{nl}")

    # === Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Final Polish (remaining={remaining:.0f}s) ===")
    if remaining > 15:
        a_out, _ = phase_coord_descent(best_seq, remaining * 0.5, seed=777)
        try_update(a_out, "final_cd")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = phase_multi_peak(best_seq, remaining * 0.7, solver)
            try_update(a_out, "final_mps")

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_safe(result)
    print(f"\nFINAL: score={final_score:.8f}, n={len(result)}")
    print(f"Total time: {time.monotonic() - t_global:.0f}s")

    return result
