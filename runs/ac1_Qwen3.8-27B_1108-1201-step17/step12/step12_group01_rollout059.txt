from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Full convolution of a and b (same length) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def cross_corr_weights(w, x):
    """
    Compute (w * x)[j] = sum_k w[k] * x[k-j] for j = 0..len(x)-1.
    w has length 2n-1, x has length n. This is the cross-correlation.
    """
    n = len(x)
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = rfft(np.asarray(w, dtype=np.float64), fft_size)
    X = rfft(np.asarray(x, dtype=np.float64), fft_size)
    # Cross-correlation: ifft(fft(w) * conj(fft(x)))
    corr = irfft(W * np.conj(X), fft_size)
    return corr[:n]


def softmax_weights(values, T):
    """Compute softmax(T * values) normalized."""
    Tvals = T * np.asarray(values, dtype=np.float64)
    Tvals = Tvals - Tvals.max()
    exps = np.exp(Tvals)
    s = exps.sum()
    if s < 1e-300:
        return np.ones(len(values)) / len(values)
    return exps / s


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def mirror_descent_lse(a0, time_limit, T_start=200.0, T_end=0.5, n_stages=30, topk=0):
    """
    Mirror descent (multiplicative weights) on LSE-smoothed objective.
    Adapts temperature and step size.
    topk=0 means use all peaks (full LSE).
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 2:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Adaptive learning rate
        c = fast_autocorr(a)
        base_eta = 1.0 / max(c.max(), 1e-10)
        lr = base_eta * (1.0 / (1.0 + 0.1 * si))
        
        # Number of iterations for this stage
        max_it = int(min(remaining, 60.0) * 20)
        max_it = max(max_it, 20)
        
        m = np.zeros(n)  # momentum
        beta = 0.9
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            c = fast_autocorr(a)
            
            if topk > 0 and topk < len(c):
                # Top-k LSE
                top_idx = np.argpartition(c, -topk)[-topk:]
                top_vals = c[top_idx]
                Tt = T * top_vals
                Tt = Tt - Tt.max()
                exps = np.exp(Tt)
                w_top = exps / exps.sum()
                # Full weight vector (sparse)
                w = np.zeros(len(c))
                w[top_idx] = w_top
            else:
                w = softmax_weights(c, T)
            
            # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
            grad = 2.0 * cross_corr_weights(w, a)
            
            # Momentum
            m = beta * m + (1.0 - beta) * grad
            
            # Multiplicative weights update
            a = a * np.exp(-lr * m)
            a = normalize_nonneg(a)
            
            # Periodic evaluation
            if (it + 1) % 10 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        # Evaluate at end of stage
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD-LSE stage {si}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, max_passes=1000):
    """
    Exact coordinate descent: for each coordinate j, find the optimal value
    that minimizes max_k conv(a,a)_k (keeping others fixed, then normalize).
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for passed in range(max_passes):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        c = fast_autocorr(a)
        cmax = c.max()
        improved = False
        
        # Process coordinates in order of potential improvement
        # Focus on coordinates that contribute to the peak
        peak_idx = np.where(c >= 0.95 * cmax)[0]
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            # Check if coordinate j contributes to any top peak
            affected = (j <= peak_idx.max()) and (j + n - 1 >= peak_idx.min())
            if not affected and passed > 5:
                # Still try it but with lower priority
                if j % 3 != passed % 3:
                    continue
            
            v = a[j]
            if v < 1e-15:
                continue
            
            # Find optimal v' by bisection
            # c_k changes by: delta_c_k = (v' - v) * coeff_k where coeff depends on j
            # For k in valid range: the change in c_k is (v'-v)*a[k-j] + (v'-v)*a[j-(k-j)] crossed terms...
            # Actually simpler: just try a few candidates
            
            # Compute the "coefficient" of a_j in each c_k
            # c_k contains a_j * a_{k-j} (if 0 <= k-j < n) and a_{k-j} * a_j (if 0 <= k-j < n and k-j != j)
            # So the linear coefficient of a_j in c_k is:
            #   2 * a[k-j] if k-j != j and 0 <= k-j < n
            #   a[j] (= a[k-j] when k=2j) if k = 2j
            #   0 otherwise
            
            # For simplicity, try: reduce a_j to bring the max down
            # The max is cmax. We want c_k + (v'-v)*coef_k <= cmax for all k.
            
            # Try several fractions of v to remove
            found_better = False
            for frac in [0.9, 0.8, 0.7, 0.5, 0.3, 0.1, 0.0]:
                v_trial = v * frac
                delta = v_trial - v
                if abs(delta) < 1e-15:
                    continue
                
                # Compute new max quickly using the fact that only O(n) values change
                # c_k_new = c_k + delta * (a[k-j] if valid) + delta * (a[k-j] if valid and k-j != j)
                # Actually: c_k = sum_i a_i * a_{k-i}
                # The terms involving a_j: when i=j: a_j * a_{k-j}; when i=k-j: a_{k-j} * a_j
                # If j != k-j: two terms, each a_j * a_{k-j}, total 2*a_j*a_{k-j}
                # If j == k-j (i.e. k=2j): one term a_j^2
                # Change when a_j -> a_j + delta:
                #   If k=2j: change = (a_j+delta)^2 - a_j^2 = 2*a_j*delta + delta^2
                #   If k!=2j and 0<=k-j<n: change = 2*delta*a_{k-j}
                
                # Check new max
                new_cmax = cmax
                for k in range(max(0, j), min(2*n-1, j + n) + 1):
                    kj = k - j
                    if 0 <= kj < n:
                        if k == 2 * j:
                            new_ck = c[k] + 2 * v * delta + delta * delta
                        else:
                            new_ck = c[k] + 2 * delta * a[kj]
                        if new_ck > new_cmax:
                            new_cmax = new_ck
                            # No need to check more, already worse
                            break
                
                if new_cmax < cmax - 1e-14:
                    a[j] = v_trial
                    cmax = new_cmax
                    found_better = True
                    improved = True
                    # Recompute c for next coordinates
                    pass
            
            if found_better:
                # Recompute full c (expensive but necessary for correctness)
                c = fast_autocorr(a)
                cmax = c.max()
        
        # Normalize and check
        a = normalize_nonneg(a)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        if not improved:
            # Try moving mass between pairs
            rng_pair = np.random.default_rng(passed * 1000 + 42)
            n_pairs = min(20, n)
            for _ in range(n_pairs):
                if time.monotonic() - t0 > time_limit - 1:
                    break
                i = rng_pair.integers(n)
                jj = rng_pair.integers(n)
                if i == jj or a[i] < 1e-12:
                    continue
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[jj] += amt
                    a_trial = normalize_nonneg(a_trial)
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < score - 1e-14:
                        a = a_trial
                        score = sc
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        improved = True
                        break
        
        if not improved and passed > 20:
            break
        
        if passed % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDescent pass={passed} ({elapsed:.0f}s): score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x * 3)) * np.sin(np.pi * x)
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(4 * np.pi * x))
    else:
        a = np.ones(n) * (0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining:
    1. Nesterov with multiplicative weights interior solver (fast, no LP)
    2. Mirror descent with adaptive LSE (polishing)
    3. Coordinate descent (exact 1D improvements)
    4. Multi-start and multi-length exploration
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1740.0)
    
    print(f"Total time budget: {time_budget:.0f}s")
    
    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0.0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr)}, t={elapsed:.0f}s)")
    
    n_base = len(best_seq) if best_seq is not None else 1600
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    # === Phase 1: Nesterov with MWU (main workhorse, 45% time) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t1 = remaining * 0.45
    print(f"\n=== Phase 1: Nesterov-MWU n={n_base} ({t1:.0f}s) ===")
    if t1 > 30:
        a_out, sc = nesterov_mwu(best_seq, t1, n_inner=3, T_inner=80.0)
        try_update(a_out, "nesterov_mwu")
    
    # === Phase 2: Mirror descent with LSE (25% time) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t2 = remaining * 0.25
    print(f"=== Phase 2: Mirror Descent LSE n={n_base} ({t2:.0f}s) ===")
    if t2 > 20:
        a_out, sc = mirror_descent_lse(best_seq, t2, T_start=150.0, T_end=0.3, n_stages=25)
        try_update(a_out, "md_lse")
    
    # === Phase 3: Coordinate descent (15% time) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t3 = remaining * 0.15
    print(f"=== Phase 3: Coordinate Descent n={n_base} ({t3:.0f}s) ===")
    if t3 > 15:
        a_out, sc = coordinate_descent(best_seq, t3)
        try_update(a_out, "coord_desc")
    
    # === Phase 4: Second Nesterov round with perturbation (10% time) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t4 = remaining * 0.10
    print(f"=== Phase 4: Nesterov round 2 ({t4:.0f}s) ===")
    if t4 > 20:
        noise = rng.normal(0, 0.02, len(best_seq))
        perturbed = normalize_nonneg(best_seq + noise * best_seq)
        a_out, sc = nesterov_mwu(perturbed, t4, n_inner=4, T_inner=40.0)
        try_update(a_out, "nesterov_r2")
    
    # === Phase 5: Longer sequence exploration (5% time) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t5 = remaining * 0.05
    print(f"=== Phase 5: Longer sequence ({t5:.0f}s) ===")
    if t5 > 15:
        n_long = n_base + 200
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, sc = nesterov_mwu(init_long, t5, n_inner=3, T_inner=50.0)
        try_update(a_out, f"long_n{n_long}")
    
    # === Phase 6: Final MD polish (remaining time) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t6 = max(0, remaining - 10)
    print(f"=== Phase 6: Final MD polish ({t6:.0f}s) ===")
    if t6 > 10:
        a_out, sc = mirror_descent_lse(best_seq, t6 * 0.5, T_start=5.0, T_end=0.1, n_stages=10)
        try_update(a_out, "final_md")
        t6b = t6 * 0.4
        if t6b > 5:
            a_out, sc = coordinate_descent(best_seq, t6b)
            try_update(a_out, "final_coord")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
