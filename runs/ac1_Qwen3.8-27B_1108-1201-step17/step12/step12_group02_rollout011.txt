import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))

def lse_obj_grad(a, T):
    """Compute LogSumExp objective and gradient."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    obj = (1.0 / T) * (Tb_max + np.log(Z))
    w = exp_Tb / Z
    
    # Gradient: grad[j] = 2 * sum_m w[m] * a[m-j] (with correction for m=2j)
    # Use FFT for the main term
    L = len(w)  # 2n-1
    fft_size = 1 << (n + L - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    
    grad = 2.0 * corr
    
    # Correction for the diagonal term (m = 2j)
    diag_idx = np.arange(0, 2*n-1, 2)  # m = 0, 2, 4, ..., 2n-2
    j_idx = diag_idx // 2  # j = 0, 1, 2, ..., n-1
    grad[j_idx] += 2.0 * w[diag_idx] * a[j_idx]
    
    return obj, grad

    best_a = a.copy()
    best_score = fast_score(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    t0 = time.monotonic()
    total_iters = 5000

        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature schedule
        frac = (t - 1) / total_iters

        beta1, beta2 = 0.9, 0.999
        bias1 = 1.0 - beta1 ** t_adj

                others_sum = 1.0 - old_val
                if others_sum > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= remaining / others_sum
                a = np.maximum(a, 0)
                a /= a.sum()
                best_score = best_s
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Stuck - try random perturbation
            if rng.integers(5) == 0:
                idx = rng.integers(n)
                a[idx] = max(0, a[idx] + rng.normal(0, 0.1/n))
                a = np.maximum(a, 0)
                a /= a.sum()
            else:
                break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDescent ({elapsed:.0f}s, {sweep} sweeps): best={best_score:.6f}")
    
    return best_a, best_score

def peak_shaving(a0, time_limit, seed=123):
    """Targeted peak shaving via mass transfer."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(5, n)]
        
        improved = False
        for k_idx in top_k:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            
            if contrib.sum() < 1e-15:
                continue
            
            src = int(idx[np.argmax(contrib)])
            
            # Find good destination: low a value, not contributing to peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:3]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            candidates = np.where(~peak_mask & (np.arange(n) != src))[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]
            
            sorted_cand = candidates[np.argsort(a[candidates])[:max(3, len(candidates)//5)]]
            if len(sorted_cand) == 0:
                continue
            dst = int(sorted_cand[rng.integers(len(sorted_cand))])
            
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
