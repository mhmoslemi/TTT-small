import numpy as np

def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def cross_corr_w_a(w, a):
    """Compute cross-correlation for gradient: result[j] = sum_i w[j+i]*a[i].”
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]

def lse_obj_grad_z(z, T):
    """Objective and gradient in z-space where a = softmax(z)."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    
    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T
    
    w = exp_Tb / exp_Tb.sum()
    cc = cross_corr_w_a(w, a)
    grad_a = 2.0 * cc
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def chirp_init(n, rate=1.0, alpha=0.5, rng=None):
    """Chirp-based initialization: a[i] = 1 + alpha*cos(2*pi*rate*(i-n/2)^2/n)."""
    x = np.arange(n, dtype=np.float64) - n/2.0
    a = 1.0 + alpha * np.cos(2.0 * np.pi * rate * x**2 / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def spectral_reweight(a, n_iters=100, target='flat', rng=None):
    """
    Iterative spectral reweighting: adjust DFT magnitude to be more uniform.
    This pushes the autocorrelation toward flatness.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    
    for it in range(n_iters):
        A = rfft(a)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        
        if target == 'flat':
            target_mag = np.ones_like(mag)
        else:
            # Water-filling: reduce high magnitudes, increase low
            target_mag = mag ** 0.5  # compress dynamic range
        
        # Blend: don't fully commit to target
        blend = 0.1 + 0.3 * (it / n_iters)  # increase commitment over time
        new_mag = (1 - blend) * mag + blend * target_mag * (mag.mean() / target_mag.mean())
        
        A_new = new_mag * phase
        a_new = irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = fast_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

def mirror_descent_v2(a0, time_limit, eta0=0.05, decay_rate=0.6,
                      k_min=1, k_max=20, k_ramp=300, seed=42):
    """
    Improved mirror descent with:
    - Softmax peak weighting (not just top-k)
    - Adaptive k based on peak structure
    - Line search for step size
    - Periodic noise injection
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()

        a = np.ones(n) / n
    else:
        a = a / s

    stall = 0

        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Adaptive k: more peaks when they're more spread out
        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))
        
        # Top-k with softmax weighting
        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx]
        # Use temperature-based softmax
        T_peak = max(1e-6, (excess.max() - excess.min()) * 0.5)
        w_exp = np.exp((excess - excess.max()) / T_peak)
        weights = w_exp / w_exp.sum()

        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]
        
        # Learning rate with decay
        eta = eta0 / (1.0 + t / 200.0) ** decay_rate
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        # Check score
        if t < 200:
            check = 2

            check = 5
        else:

            a = a + noise_scale * rng.standard_normal(n)

            stall = 0

def lse_optimize_v2(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=15, seed=42):

        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)

    best_score = fast_score(a_init)
    
    for si, stage_T in enumerate(temps):

            sc = fast_score(a)
            if sc < best_score:

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def cross_corr_w_a(w, a):
    """Compute gradient cross-correlation: result[j] = sum_i w[j+i]*a[i]."""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_obj_grad_z(z, T):
    """Objective and gradient in z-space where a = softmax(z)."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()
    cc = cross_corr_w_a(w, a)
    grad_a = 2.0 * cc
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        progress = si / max(1, n_stages - 1)
        iters = max(20, min(120, int(20 + 100 * progress ** 2)))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': iters * 300,
                                   'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_surgery_v2(a0, time_limit, seed=123):
    """
    Systematic peak surgery with pressure-based targeting.
    Tries single moves and small paired moves to reduce top peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(5, n)]

        # Compute pressure map from top peaks
        pressure = np.zeros(n)
        for pk in top_k[:3]:
            pk = int(pk)
            pk_js = max(0, pk - n + 1)
            pk_je = min(n - 1, pk)
            pk_idx = np.arange(pk_js, pk_je + 1)
            pressure[pk_idx] += a[pk - pk_idx]

        improved = False
        for k_idx in top_k[:3]:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            # Target: low pressure, low a
            target_score = pressure + 0.05 * a
            target_score[src] = np.inf
            candidates = np.argsort(target_score)[:15]

            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.008, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-10:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(3) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

            if stall > 300:
                break

    return best_a, best_score


def paired_surgery(a0, time_limit, seed=555):
    """
    Paired coordinate surgery: reduce two source positions simultaneously
    and distribute to two low-pressure targets.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_idx = np.argsort(b)[::-1][:min(3, n)]

        pressure = np.zeros(n)
        for pk in top_idx:
            pk = int(pk)
            pk_js = max(0, pk - n + 1)
            pk_je = min(n - 1, pk)
            pk_idx = np.arange(pk_js, pk_je + 1)
            pressure[pk_idx] += a[pk - pk_idx]

        improved = False
        for k_idx in top_idx:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            top_contrib = np.argsort(contrib)[::-1][:3]
            for tc in top_contrib:
                src = int(idx[tc])
                partner = k_idx - src
                if partner < 0 or partner >= n or partner == src:
                    continue

                for out_frac in [0.003, 0.008, 0.015]:
                    amt1 = a[src] * out_frac
                    amt2 = a[partner] * out_frac

                    # Find two low-pressure targets
                    ts = pressure + 0.05 * a
                    ts[src] = np.inf
                    ts[partner] = np.inf
                    targets = np.argsort(ts)[:10]
                    d1 = int(targets[0])
                    d2 = int(targets[1])

                    a_trial = a.copy()
                    a_trial[src] -= amt1
                    a_trial[partner] -= amt2
                    a_trial[d1] += amt1
                    a_trial[d2] += amt2

                    if a_trial[src] < -1e-10 or a_trial[partner] < -1e-10:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function - multi-phase optimization.
    Phase 1: Spectral reweighting (global shaping via DFT magnitude compression)
    Phase 2: Mirror descent (fast local optimization)
    Phase 3: LogSumExp L-BFGS (precise refinement with annealing)
    Phase 4: Peak surgery (coordinate-level polish)
    Phase 5: Multi-length exploration
    Phase 6: Final combined polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1740.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # Try parent
    n_parent = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            n_parent = len(p)
            parent = p / p.sum()
            ps = fast_score(parent)
            print(f"Parent: n={n_parent}, score={ps:.6f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(n_parent, variant=0)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Spectral reweighting ===
    print(f"\n=== Phase 1: Spectral Reweighting (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_sr = min(remaining * 0.10, 120.0)
    if t_sr > 10:
        # Run spectral reweight with aggressive compression schedule
        n_iters_sr = max(100, int(t_sr * 5))
        compress_sched = np.geomspace(0.7, 0.05, min(n_iters_sr, 500))
        a_out, sc_out = spectral_reweight(best_seq, n_iters=min(n_iters_sr, 500),
                                           compress_schedule=compress_sched)
        try_update(a_out, "spectral_aggr")

        # Run with gentler compression
        compress_sched2 = np.geomspace(0.9, 0.3, 200)
        a_out2, _ = spectral_reweight(best_seq, n_iters=200, compress_schedule=compress_sched2)
        try_update(a_out2, "spectral_gentle")

    # === Phase 2: Mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_md = min(remaining * 0.18, 280.0)
    if t_md > 20:
        a_out, _ = mirror_descent_v2(best_seq, t_md, eta0=0.04, decay_rate=0.5,
                                      k_min=2, k_max=15, k_ramp=300, seed=1)
        try_update(a_out, "md_main")

    # Chirp-based mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    t_md2 = min(remaining * 0.10, 120.0)
    if t_md2 > 20:
        chirp_configs = [(1.0, 0.6), (2.0, 0.4), (3.0, 0.3), (0.5, 0.8)]
        for i, (rate, alpha) in enumerate(chirp_configs):
            if time.monotonic() - t_global > time_budget - 100:
                break
            chirp_a = chirp_init(n_parent, rate=rate, alpha=alpha)
            t_chunk = t_md2 / len(chirp_configs)
            a_out, _ = mirror_descent_v2(chirp_a, t_chunk, eta0=0.06, decay_rate=0.6,
                                          k_min=2, k_max=12, k_ramp=150, seed=10 + i)
            try_update(a_out, f"chirp_r{rate}_a{alpha}")

    # === Phase 3: LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (n={n_parent}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse = min(remaining * 0.25, 400.0)
    if t_lse > 30:
        a_out, _ = lse_optimize_v2(best_seq, t_lse, T_start=50.0, T_end=0.05,
                                    n_stages=15, seed=7)
        try_update(a_out, "lse_main")

    # === Phase 4: Peak surgery ===
    print(f"\n=== Phase 4: Peak Surgery (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_ps = min(remaining * 0.12, 150.0)
    if t_ps > 15:
        a_out, _ = peak_surgery_v2(best_seq, t_ps, seed=99)
        try_update(a_out, "surgery1")

        remaining = time_budget - (time.monotonic() - t_global)
        t_ps2 = min(remaining * 0.06, 80.0)
        if t_ps2 > 10:
            a_out2, _ = paired_surgery(best_seq, t_ps2, seed=222)
            try_update(a_out2, "paired_surgery")

    # === Phase 5: Multi-length exploration ===
    n_long = int(round(n_parent * 1.3))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-Length (n_long={n_long}) ===")

    if best_seq is not None and remaining > 150:
        init_long = interpolate_to_length(best_seq, n_long)

        # Spectral at longer length
        t_sr_l = min(remaining * 0.08, 80.0)
        if t_sr_l > 10:
            cs = np.geomspace(0.7, 0.1, 300)
            a_out, _ = spectral_reweight(init_long, n_iters=300, compress_schedule=cs)
            try_update(a_out, f"spectral_n{n_long}")

        # MD at longer length
        remaining = time_budget - (time.monotonic() - t_global)
        t_md_l = min(remaining * 0.18, 220.0)
        if t_md_l > 20:
            init_l2 = interpolate_to_length(best_seq, n_long)
            a_out, _ = mirror_descent_v2(init_l2, t_md_l, eta0=0.03, decay_rate=0.5,
                                          k_min=1, k_max=15, k_ramp=400, seed=3)
            try_update(a_out, f"md_n{n_long}")

        # LSE at longer length
        remaining = time_budget - (time.monotonic() - t_global)
        t_lse_l = min(remaining * 0.18, 200.0)
        if t_lse_l > 20:
            init_l3 = interpolate_to_length(best_seq, n_long)
            a_out, _ = lse_optimize_v2(init_l3, t_lse_l, T_start=30.0, T_end=0.05,
                                        n_stages=12, seed=13)
            try_update(a_out, f"lse_n{n_long}")

        # Also try shorter length
        n_short = int(round(n_parent * 0.7))
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            init_short = interpolate_to_length(best_seq, n_short)
            a_out, _ = mirror_descent_v2(init_short, min(remaining * 0.3, 80.0),
                                          eta0=0.05, decay_rate=0.6,
                                          k_min=1, k_max=10, k_ramp=200, seed=5)
            try_update(a_out, f"md_n{n_short}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 50:
                a_out2, _ = lse_optimize_v2(init_short, min(remaining * 0.5, 80.0),
                                             T_start=30.0, T_end=0.1, n_stages=10, seed=15)
                try_update(a_out2, f"lse_n{n_short}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None:
        # Very low T LSE for precise convergence
        t_f1 = min(remaining * 0.30, 120.0)
        if t_f1 > 15:
            a_out, _ = lse_optimize_v2(best_seq, t_f1, T_start=8.0, T_end=0.02,
                                        n_stages=10, seed=44)
            try_update(a_out, "final_lse_lowT")

        # Spectral reweight on best (different compression)
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.20, 60.0)
        if t_f2 > 5:
            cs_final = np.geomspace(0.5, 0.01, 200)
            a_out2, _ = spectral_reweight(best_seq, n_iters=200, compress_schedule=cs_final)
            try_update(a_out2, "final_spectral")

        # Peak surgery
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.25, 80.0)
        if t_f3 > 10:
            a_out3, _ = peak_surgery_v2(best_seq, t_f3, seed=777)
            try_update(a_out3, "final_surgery")

        # Short MD with high k for smoothing
        remaining = time_budget - (time.monotonic() - t_global)
        t_f4 = min(remaining * 0.20, 50.0)
        if t_f4 > 5:
            a_out4, _ = mirror_descent_v2(best_seq, t_f4, eta0=0.01, decay_rate=0.8,
                                           k_min=10, k_max=30, k_ramp=50, seed=888)
            try_update(a_out4, "final_md_smooth")

    # Final
    if best_seq is not None:
        final_score = fast_score(best_seq)
        print(f"\nFINAL: score={final_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
