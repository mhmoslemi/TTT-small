    result = np.maximum(v - theta, 0)
    
    # Ensure exact sum
    s = result.sum()
    if s > 0:
        result = result * target_sum / s
    return result

def nesterov_subgradient(a0, time_limit, k=5, verbose=True):
    """
    Nesterov-accelerated subgradient descent on the simplex.
    Minimizes f(a) = max_j conv(a,a)[j] subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    y = a.copy()
    
    best_a = a.copy()
    b = fast_autocorr(a)
    best_score = 2.0 * n * b.max()
    
    t0 = time.monotonic()
    t = 0
    eta = 1.0 / n  # Initial step size
    momentum = 0.0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute subgradient at y
        grad, b_y, max_b_y = subgradient_topk(y, k=k)
        L = 2.0 * np.dot(y, y)  # Lipschitz constant estimate
        
        # Backtracking line search (simplified: use fixed decay)
        eta_t = eta / (1.0 + 0.01 * t)
        
        # Gradient step
        y_new = y - eta_t * grad / L
        y_new = simplex_project(y_new)
        
        # Nesterov momentum update
        if t == 1:
            a_new = y_new
        else:
            beta = (t - 1.0) / (t + 1.0 + 2.0 * np.sqrt(max(0, t - 100)))
            beta = min(beta, 0.9)
            a_new = y_new + beta * (y_new - y)
            a_new = simplex_project(a_new)
        
        y = y_new
        a = a_new
        
        # Score
        b_a = fast_autocorr(a)
        score_a = 2.0 * n * b_a.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        b_y_score = 2.0 * n * max_b_y
        if b_y_score < best_score:
            best_score = b_y_score
            best_a = y.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS t={t} ({elapsed:.0f}s): best={best_score:.6f}, cur_a={score_a:.6f}")
    
    return best_a, best_score

def spectral_flatten(a0, time_limit, alpha=0.7, n_iters=5000, verbose=True):
    """
    Gerchberg-Saxton variant: iteratively flatten the spectrum.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Forward FFT
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Compress dynamic range (flatten spectrum)
        P_max = P.max()
        P_min = P.min()
        if P_max > 0:
            P_norm = P / P_max
            P_new = P_max * (P_norm ** alpha + (1 - alpha) * 0.01)
            # Actually, let's use a softer flattening
            # Target: make P more uniform
            P_target = P_max * (P_norm ** alpha)
            P_blend = (1 - 0.1) * P + 0.1 * P_target
        else:
            P_blend = P
        
        # New amplitude, keep phase
        A_new = np.sqrt(np.maximum(P_blend, 0)) * np.exp(1j * np.angle(A))
        
        # Inverse FFT
        a_new = irfft(A_new, fft_size)[:n]
        a_new = simplex_project(a_new)
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SF it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def lp_best_response_fast(a_fixed, n_constraints=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b)[k] <= M, sum(b)=1, b>=0.
    Uses only a subset of constraints for speed.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    
    if n_constraints is None or n_constraints >= size:
        # Use all constraints (expensive)
        # Build constraint matrix: (size x n) where row k has a_fixed at appropriate positions
        # A_ub[b_k] = sum_i a_fixed[i] * b[k-i] for valid i
        A_ub = np.zeros((size, n))
        for j in range(n):
            A_ub[j:j+n, j] = a_fixed
    else:
        # Use top constraints only
        # First compute which constraints are likely binding
        # Use a heuristic: constraints near the center are more likely binding
        center = n - 1
        indices = list(range(max(0, center - n_constraints//2), min(size, center + n_constraints//2 + 1)))
        A_ub = np.zeros((len(indices), n))
        for i, k in enumerate(indices):
            for j in range(max(0, k-n+1), min(n, k+1)):
                A_ub[i, j] = a_fixed[k-j]
    
    # min M s.t. A_ub @ b <= M, sum(b) = 1, b >= 0
    # Variables: x = [b_0, ..., b_{n-1}, M]
    # Objective: minimize M => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ b - M <= 0 => [A_ub, -1] @ x <= 0
    A_ineq = np.zeros((A_ub.shape[0], n + 1))
    A_ineq[:, :n] = A_ub
    A_ineq[:, -1] = -1.0
    b_ineq = np.zeros(A_ub.shape[0])
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        b_val = res.x[:n]
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None
        return b_val / s
    return None

def nesterov_lp(a0, time_limit, solver='linprog', verbose=True):
    """
    Nesterov-accelerated LP best response method.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    t = 0
    max_iters = 5000
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response
        b_star = lp_best_response_fast(y)
        if b_star is None:
            continue
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = simplex_project(y_new)
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = simplex_project(a_new)
        
        y = y_new
        a = a_new
        
        # Score
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL t={t} ({elapsed:.0f}s): best={best_score:.6f}")
        
        # Early stopping
        if t > 500 and t % 200 == 0:
            # Check for stagnation (simplified)
            if score_a > best_score * 1.001:
                pass  # Continue for now
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent: try pairwise mass transfers to reduce max autocorrelation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    b = fast_autocorr(a)
    
    t0 = time.monotonic()
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit:
        # Identify top peaks
        n_peaks = min(10, 2*n-1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        max_b = b.max()
        
        improved = False
        
        for peak in top_peaks[:5]:
            # For this peak, find the top contributors
            js = max(0, peak - n + 1)
            je = min(n - 1, peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[peak - idx]
            
            # Find top contributor pair
            top_contrib = np.argsort(contrib)[::-1][:3]
            
            for ci in top_contrib:
                src = idx[ci]
                src_other = peak - src
                
                # Try moving mass from src (or src_other) to a random target
                for _ in range(20):
                    dst = rng.integers(0, n)
                    if dst == src or dst == src_other:
                        continue
                    
                    for frac in [0.001, 0.005, 0.01, 0.02]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        if src != src_other:
                            a_trial[src_other] = max(0, a_trial[src_other] - amt * 0.5)
                        a_trial[dst] += amt
                        a_trial = simplex_project(a_trial)
                        
                        b_trial = fast_autocorr(a_trial)
                        score = 2.0 * n * b_trial.max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            b = b_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            
            # Random perturbation to escape local minima
            if stall % 10 == 0:
                n_perturb = rng.integers(1, 5)
                sources = rng.choice(n, size=n_perturb, replace=False)
                targets = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for src, dst in zip(sources, targets):
                    frac = rng.uniform(0.002, 0.01)
                    amt = a_trial[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                a_trial = simplex_project(a_trial)
                b_trial = fast_autocorr(a_trial)
                score = 2.0 * n * b_trial.max()
                if score < best_score:
                    a = a_trial
                    b = b_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            
            if stall > max_stall:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
