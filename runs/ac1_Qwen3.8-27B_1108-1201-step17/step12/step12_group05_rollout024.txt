def fast_conv_matrix_apply(a, b):
    """Compute T_a @ b where T_a is the convolution matrix."""
    # (T_a @ b)_k = sum_j a_{k-j} * b_j = convolve(a, b)_k
    return np.convolve(a, b)  # or use FFT for speed

def multiplicative_weights_best_response(a, n_iter=10, eta=1.0):
    """Approximate the LP best response using multiplicative weights."""
    n = len(a)
    b = np.ones(n) / n
    for _ in range(n_iter):
        conv = fast_autocorr_cross(a, b)  # convolve(a, b)
        k_star = np.argmax(conv)
        grad = np.zeros(n)
        # (T_a @ b)_k = sum_j a[k-j] * b[j], so d/d b_j (T_a @ b)_k* = a[k*-j]
        js = np.arange(n)
        idx = k_star - js
        valid = (idx >= 0) & (idx < n)
        grad[valid] = a[idx[valid]]
        # Multiplicative weights update
        b = b * np.exp(-eta * grad)
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-300:
            b = np.ones(n) / n
        else:
            b = b / s
    return b

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Full convolution a*b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_warmup(a0, time_limit, lr_init=0.1):
    """
    Fast subgradient method using the exact subgradient of max_k (a*a)_k.
    The subgradient at lag k* is 2*a_{k*-j} (reversed and shifted copy of a).
    Uses Nesterov acceleration for faster convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Nesterov momentum variables
    a_curr = a.copy()
    a_prev = a.copy()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a_curr)
        k_star = np.argmax(b)

        # Subgradient: grad_j = 2 * a_curr[k_star - j] (where valid)
        js = np.arange(n)
        idx = k_star - js
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a_curr[idx[valid]]

        # Step size: diminishing
        lr = lr_init / (1.0 + 0.01 * t)

        # Gradient step
        a_new = a_curr - lr * grad

        # Project onto simplex
        a_new = project_simplex(a_new, 1.0)

        # Nesterov momentum
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        a_next = a_new + ((t - 1.0) / beta_t) * (a_new - a_prev)
        a_next = np.maximum(a_next, 0)
        a_next_sum = a_next.sum()
        if a_next_sum < 1e-15:
            a_next = a_new.copy()
        else:
            a_next = a_next / a_next_sum

        a_prev = a_curr
        a_curr = a_next

        # Track best
        if t % 10 == 0:
            score = 2.0 * n * fast_autocorr(a_curr).max()
            if score < best_score:
                best_score = score
                best_a = a_curr.copy()

            score_avg = 2.0 * n * fast_autocorr(a_new).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_new.copy()

        if t % 500 == 0:
            print(f"      SubGrad t={t}: best={best_score:.6f}")

    return best_a, best_score


def compute_lse_grad_a(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / max_Tb.sum()

    fft_size = 1 << (2 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=20, lr=0.005):
    """
    LogSumExp-smoothed optimization with Adam optimizer and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(200, int(remaining * 100 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            obj, grad = lse_obj_grad_z(z, stage_T)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.003 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 6 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def mw_best_response(a_fixed, n_iter=20, eta_base=0.5):
    """
    Approximate LP best response using multiplicative weights.
    Solves: min_b max_k conv(a,b)_k s.t. b>=0, sum(b)=1
    using exponential weighting on the most-violated constraint.
    """
    n = len(a_fixed)
    b = np.ones(n) / n

    for it in range(n_iter):
        conv = fast_cross_conv(a_fixed, b)
        k_star = np.argmax(conv)
        max_val = conv[k_star]

        if max_val < 1e-15:
            break

        # Subgradient w.r.t. b: grad_j = a_fixed[k_star - j] (where valid)
        js = np.arange(n)
        idx = k_star - js
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = a_fixed[idx[valid]]

        # Multiplicative weights update
        eta = eta_base / (1.0 + 0.1 * it)
        b = b * np.exp(-eta * grad / max_val)
        b = np.maximum(b, 1e-30)
        b = b / b.sum()

    return b

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """
    Spectral shaping: iteratively flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving(a0, time_limit, seed=123):
    """
    Advanced coordinate-descent peak shaving with multi-scale moves
    and simultaneous consideration of multiple peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Look at more peaks
        top_k = np.argsort(b)[::-1][:12]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            # Top contributing pairs
            src_rank = np.argsort(contrib)[::-1][:3]
            for sr in src_rank:
                src = idx[sr]

                # Find destination: positions with low a that don't create new peaks
                peak_areas = set()
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_areas.update(range(js2, je2 + 1))

                candidates = [j for j in range(n) if j not in peak_areas and j != src]
                if not candidates:
                    candidates = [j for j in range(n) if j != src]
                candidates_sorted = sorted(candidates, key=(lambda j: a[j]))
                dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]

                for _ in range(3):
                    if not dst_pool:
                        break
                    dst = dst_pool[rng.integers(len(dst_pool))]

                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 200:
                # Larger random restart within the simplex
                if stall > 500:
                    break
                a_pert = a + rng.normal(0, 0.001, n)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                score = 2.0 * n * fast_autocorr(a_pert).max()
                if score < best_score:
                    a = a_pert
                    best_score = score
                    best_a = a.copy()
                    stall = 100  # Don't fully reset stall

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    elif choice == 7:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 8:
        # Rudin-Shapiro inspired
        rs = [1]
        for _ in range(int(np.log2(n)) - 1):
            rs = rs + [-1] * len(rs)
        rs = rs[:n]
        a = 0.5 + 0.3 * np.array(rs, dtype=float)
        a = np.maximum(a, 0)
    else:
        # Asymmetric bump
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.25) ** 2) + 0.7 * np.exp(-4 * (x - 0.75) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Subgradient warmup (fast, O(n log n) per step)
    Phase 2: Spectral shaping
    Phase 3: Nesterov LP (main workhorse)
    Phase 4: Adam LSE refinement
    Phase 5: Nesterov LP continued
    Phase 6: Multi-length exploration
    Phase 7: Final peak shaving + polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        # Try multiple initializations and pick best
        best_init = None
        best_init_score = float('inf')
        for trial in range(5):
            candidate = structured_init(n_base, np.random.default_rng(seed + trial * 100))
            sc = 2.0 * n_base * fast_autocorr(candidate).max()
            if sc < best_init_score:
                best_init_score = sc
                best_init = candidate
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Subgradient warmup (fast) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Subgradient warmup at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.02, 25.0)
    if t1 > 3:
        a_out, sc_out = subgradient_warmup(best_seq, t1, lr_init=0.05)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [subgrad]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 2: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Spectral shaping at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.03, 30.0)
    if t2 > 3:
        a_out = spectral_shaping(best_seq, n_iter=800, alpha=0.25)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [spectral]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 3: Nesterov LP (main workhorse, part 1) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Nesterov LP at n={n_base} part 1 (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lp_part1]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 4: Adam LSE refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Adam LSE at n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_out, _ = adam_lse_optimize(best_seq, t4, T_start=30.0, T_end=0.03, n_stages=15, lr=0.004)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [adam_lse]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 5: Nesterov LP continued ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Nesterov LP continued at n={n_base} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 350.0)
    if t5 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lp_part2]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 6: Try longer length ===
    n_long = int(round(n_base * 1.3))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral + subgrad at new length
        a_spec = spectral_shaping(init_long, n_iter=300, alpha=0.2)
        sc = evaluate_safe(a_spec)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_spec, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [spectral_n{n_long}]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

        # LP at longer length
        t6b = min(t6 * 0.6, 120.0)
        if t6b > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, t6b, solver)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                elapsed = time.monotonic() - t_global_start
                print(f"  *** NEW BEST [lp_n{n_long}]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 7: Try shorter length ===
    n_short = int(round(n_base * 0.8))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Shorter n={n_short} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.08, 80.0)
    if t7 > 20:
        init_short = interpolate_sequence(best_seq, n_short)
        a_out, _ = nesterov_lp_optimize(init_short, t7, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lp_n{n_short}]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Peak shaving (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 100.0)
    if t8 > 5:
        a_out, _ = peak_shaving(best_seq, t8, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [peak_shave]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 9: Final Adam LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 9: Final Adam LSE polish (remaining={remaining:.0f}s) ===")
    t9 = min(remaining * 0.12, 80.0)
    if t9 > 10:
        a_out, _ = adam_lse_optimize(best_seq, t9, T_start=5.0, T_end=0.01, n_stages=10, lr=0.002)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [final_adam]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 10: Final Nesterov LP burst ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"\n=== Phase 10: Final Nesterov LP burst (remaining={remaining:.0f}s) ===")
        t10 = min(remaining * 0.35, 120.0)
        a_out, _ = nesterov_lp_optimize(best_seq, t10, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [final_lp]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # === Phase 11: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        print(f"\n=== Phase 11: Final peak shaving (remaining={remaining:.0f}s) ===")
        t11 = min(remaining * 0.5, 60.0)
        a_out, _ = peak_shaving(best_seq, t11, seed=789)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:

            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:

            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [final_peak]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
