def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if target_sum <= 0:
        return np.zeros(n)
    # Scale to sum=1 then project
    v_scaled = v * (target_sum / max(v.sum(), 1e-300))
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v_scaled - theta, 0)
    s = w.sum()
    if s > 1e-15:
        w = w * target_sum / s
    return w


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        if k_start <= k_end:
            T[k_start:k_end + 1, j] = a_fixed[:k_end - k_start + 1]
    return T


def get_lp_solver():
    """Find a working LP/QP solver, prefer interior point."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def smoothed_subgradient_refine(a0, time_limit, n_top=5, temperature=50.0):
    """Refine using smoothed subgradient with top-k peak weighting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    lr = 0.05
    decay = 0.999

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        max_c = c.max()

        # Softmax weights over top peaks
        top_k = np.argsort(c)[::-1][:n_top]
        weights = np.exp((c[top_k] - max_c) * temperature)
        weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += 2.0 * w * a[other]

        # Gradient descent step
        a_new = a - lr * grad
        a_new = project_simplex(a_new, 1.0)

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a - 0.5 * lr * grad
            a_try = project_simplex(a_try, 1.0)
            score_try = 2.0 * n * fast_autocorr(a_try).max()
            if score_try < best_score:
                best_score = score_try
                best_a = a_try.copy()
                a = a_try
            else:
                lr *= 0.9

        lr = max(lr * decay, 0.001)

        if it % 1000 == 0:
            print(f"      SS it={it}: best={best_score:.6f}, lr={lr:.5f}")

    return best_a, best_score

def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_ramp_init(n, power=1.0):
    """Create an asymmetric ramp initialization."""
    i = np.arange(n)
    a = (i + 1) ** power
    a = a / a.sum()
    return a


def make_exp_init(n, alpha=0.5):
    """Create an exponential taper initialization."""
    i = np.arange(n)
    a = np.exp(alpha * i / n)
    a = a / a.sum()
    return a


def make_cos_init(n, shift=0.3):
    """Create a cosine-bump initialization, asymmetric via shift."""
    i = np.arange(n) / n
    center = 0.5 + shift
    a = np.exp(-((i - center) ** 2) / 0.05)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Also try a few alternative initializations to see if they're better
    for init_func, label in [(lambda n_: make_ramp_init(n_, 1.5), "ramp1.5"),
                             (lambda n_: make_exp_init(n_, 0.8), "exp0.8"),
                             (lambda n_: make_cos_init(n_, 0.2), "cos0.2")]:
        alt = init_func(n_base)
        alt_score = evaluate_safe(alt)
        if alt_score < best_score:
            best_score = alt_score
            best_seq = alt.copy()
            print(f"  Better init [{label}]: {best_score:.6f}")

    # Phase 1: LP Nesterov at main length
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp(best_seq, t1, solver)
        print(f"  LP Nesterov done: {iters} iters, internal_score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Smoothed subgradient refinement
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    print(f"\n=== Phase 2: Smoothed Subgradient at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 20:
        a_out, sc_out = smoothed_subgradient_refine(best_seq, t2, n_top=5, temperature=30.0)
        print(f"  SS done: internal_score={sc_out:.6f}")
        try_update(a_out, "ss_nesterov")

    # Phase 3: Coordinate descent
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    print(f"\n=== Phase 3: Coordinate Descent at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 10:
        a_out, sc_out = coordinate_descent_refine(best_seq, t3, seed=123)
        print(f"  CD done: internal_score={sc_out:.6f}")
        try_update(a_out, "cd_nesterov")

    # Phase 4: Second LP Nesterov round (shorter, from improved point)
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    print(f"\n=== Phase 4: LP Nesterov Round 2 at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        a_out, sc_out, iters = nesterov_lp(best_seq, t4, solver)
        print(f"  LP R2 done: {iters} iters, internal_score={sc_out:.6f}")
        try_update(a_out, "lp_round2")

    # Phase 5: Multi-length exploration
    for n_try in [n_base - 100, n_base + 100, n_base + 200, n_base - 200, n_base + 300]:
        if n_try < 400:
            continue
        elapsed = time.monotonic() - t_global_start
        remaining = time_budget - elapsed
        if remaining < 120:
            break
        print(f"\n=== Phase 5: Explore n={n_try} (remaining={remaining:.0f}s) ===")
        t5 = min(remaining * 0.12, 120.0)
        if t5 < 30:
            continue
        init_try = interpolate_to_length(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp(init_try, t5, solver)
        print(f"  LP at n={n_try}: {iters} iters, internal_score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_try}")

        # Quick refinement at this length
        elapsed = time.monotonic() - t_global_start
        remaining = time_budget - elapsed
        if remaining > 60:
            t5b = min(remaining * 0.05, 50.0)
            if t5b > 10:
                a_ss, sc_ss = smoothed_subgradient_refine(a_out, t5b, n_top=5, temperature=30.0)
                try_update(a_ss, f"ss_n{n_try}")

    # Phase 6: Final polish - coordinate descent on global best
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    print(f"\n=== Phase 6: Final Polish at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 120.0)
    if t6 > 10:
        a_out, sc_out = coordinate_descent_refine(best_seq, t6, seed=789)
        try_update(a_out, "cd_final")

        a_out2, sc_out2 = smoothed_subgradient_refine(best_seq, t6 * 0.5, n_top=8, temperature=20.0)
        try_update(a_out2, "ss_final")

    # Phase 7: Try a fresh ramp at the best length found
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    if remaining > 100:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Fresh ramp at n={n_best} (remaining={remaining:.0f}s) ===")
        fresh = make_ramp_init(n_best, power=rng.uniform(0.8, 1.5))
        t7 = min(remaining * 0.15, 100.0)
        a_out, sc_out, iters = nesterov_lp(fresh, t7, solver)
        try_update(a_out, "fresh_ramp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
