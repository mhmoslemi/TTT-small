def lp_best_response(a, c=None):
    """Solve: min M s.t. T(a)@b <= M, sum(b)=1, b>=0"""
    n = len(a)
    size = 2*n - 1
    # Variables: b[0..n-1], M (index n)
    # Objective: minimize M (index n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: T@b - M <= 0, i.e., T@b + (-1)*M <= 0
    # T is (2n-1) x n, and M has coefficient -1
    A_ub = np.zeros((size, n + 1))
    # Build T matrix
    for j in range(n):
        A_ub[j:j+n, j] = a
    A_ub[:, n] = -1.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]  # b >= 0, M free
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(size), A_eq=A_eq, b_eq=[1.0],
                  bounds=bounds, method='highs')
    if res.status == 0:
        b_val = res.x[:n]
        b_val = np.maximum(b_val, 0)
        b_val = b_val / b_val.sum()
        return b_val, res.x[n]
    return None, None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov-accelerated LP best-response with scipy HiGHS (replacing cvxpy for speed)
2. Exponential iterate averaging for improved convergence

4. Local peak-directed search
5. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Fast linear autocovolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Fast linear convolution of a and b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_T_dense(a):
    """Build dense matrix T where (T@b)[k] = convolve(a, b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0 using HiGHS.
    Returns (b_opt, M_opt) or (None, None)."""
    n = len(a)
    size = 2 * n - 1

    # Variables: [b_0, ..., b_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # A_ub: T @ b - M <= 0
    T = build_T_dense(a)
    A_ub = np.empty((size, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)

    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None, None

    if res.status == 0:
        b_val = np.maximum(res.x[:n], 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, float(res.x[n])
    return None, None


def nesterov_lp_fast(a0, time_limit, max_iters=1000000):
    """Nesterov-accelerated LP best-response with exponential averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)

    # Exponential moving average
    avg_a = a.copy()
    avg_decay = 0.999

    t0 = time.monotonic()
    last_print = t0
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = lp_best_response(y)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        # Update exponential average
        avg_a = avg_decay * avg_a + (1.0 - avg_decay) * a_new
        avg_a = np.maximum(avg_a, 0)
        avg_a = avg_a / max(avg_a.sum(), 1e-15)

        a = a_new
        y = y_new

        # Check scores
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve_count = 0

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
            no_improve_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.6f} no_improve={no_improve_count}")
                last_print = elapsed

        # Early stop if stuck for very long
        if no_improve_count > 50000 and it > 5000:
            break

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    max_b = b.max()
    # LogSumExp of b
    shifted = b - max_b
    exp_b = np.exp(shifted)

    log_sum_exp = max_b + np.log(exp_b.sum())

    #           = a_j * db_k/da_j - a_j * sum_j' a_j' * db_k/da_j'

    # db_k/dz_j = a_j * (a[k-j] - b_k) ... that's not right either
    # Let me redo: b_k = sum_j a[j] * a[k-j]
    # db_k/da_j = a[k-j] + a_j * d(k-j)/da_j ... no
    # For fixed k: b_k = sum_j a[j]*a[k-j]. If I vary a[j0]:
    # db_k/da[j0] = a[k-j0] (from the a[j] term) + a[j0] * (if k-j0 is a valid index, from the a[k-j] term)
    # Actually b_k = sum_j a[j]*a[k-j]. The terms involving a[j0] are:
    #   j = j0: a[j0]*a[k-j0]
    #   k-j = j0, i.e., j = k-j0: a[k-j0]*a[j0] (if k-j0 is valid)

    # Wait no. Let me be more careful.
    # b_k = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]

    #            = a[k-m] (if m valid and k-m valid) + a[m] (if k-m valid and m valid)

    # Actually for the convolution: c[k] = sum_j a[j]*b[k-j], here a=b so:
    # c[k] = sum_j a[j]*a[k-j]
    # dc[k]/da[m] = a[k-m] (from j=m term) + [a[m] if k-m is a valid index for the second factor]
    # For the second factor a[k-j] to equal a[m], we need k-j=m, i.e., j=k-m.
    # So dc[k]/da[m] = a[k-m] (if 0<=k-m<n) + a[m] (if 0<=k-m<n)
    # Wait that gives 2*a[k-m] when k-m is valid. That's because a appears twice in the product.
    # Actually no: dc[k]/da[m] = (d/da[m]) sum_j a[j]*a[k-j]
    # = (d/da[m]) a[m]*a[k-m] + (d/da[m]) sum_{j!=m} a[j]*a[k-j]
    # = a[k-m] + a[m]*d(k-m)/d... no, a[k-j] depends on a[m] only when j = k-m
    # = a[k-m] (from j=m) + a[k-m] (from the a[k-j] term when j=k-m) [if k-m != m]
    # Hmm, I'm overcomplicating this. Let me just use:
    # dc[k]/da[m] = a[k-m] if 0<=k-m<n, plus a[m] if 0<=k-m<n (from the other occurrence)
    # Actually the cleanest way: c[k] = sum_j a[j]*a[k-j]
    # dc[k]/da[m] = a[k-m] (from j=m) + [if k-m is in [0,n), then a[k-(k-m)] = a[m] from j=k-m]

    # = a[k-m] if only m is in [0,n) but k-m is not
    # = a[m] if only k-m is in [0,n) but m is not (but m is always in [0,n) since it's an index of a)

    # Hmm, this is still complex. Let me just use the simpler subgradient approach.
    
    # Simpler gradient: use the subgradient of max(b) which is a shifted version of a
    # At the peak k*, the subgradient of b[k*] w.r.t. a is: a shifted by k*
    # For LogSumExp, the "soft" subgradient is the weighted sum over all peaks
    grad_a = np.zeros(n)
    for k in range(len(b)):

            continue

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * ws[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=30.0, T_min=0.1, n_stages=30, iters_per_stage=50):
    """Log-sum-exp L-BFGS with aggressive temperature decay."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    T = T_start

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Decrease temperature
            T = max(T_min, T * 0.7)

            if (stage + 1) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={stage+1} T={T:.3f}: score={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_directed_search(a0, time_limit, n_trials=5000):
    """Local search: perturb elements near peaks to reduce them."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_idx = np.argmax(b)

        # Perturb elements that contribute to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        # Random perturbation of a few elements near the contributing range
        n_perturb = rng.integers(2, min(10, n // 2))
        perturb_idx = rng.choice(np.arange(lo, hi + 1), size=min(n_perturb, hi - lo + 1), replace=False)

        a_trial = a.copy()
        scale = rng.uniform(0.01, 0.1)
        for idx in perturb_idx:
            direction = rng.choice([-1, 1])
            a_trial[idx] += direction * scale * a[idx]

        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s

        sc = score_normalized(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

        if (trial + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSearch trial={trial+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def eg_best_response(a, n_iters=50, eta=0.1):
    """Approximate LP best response using exponentiated gradient."""
    n = len(a)
    b = np.ones(n) / n
    for _ in range(n_iters):
        conv = fast_cross_conv(a, b)
        max_conv = conv.max()
        # Softmax distribution over constraints
        z = (conv - max_conv) / 0.5
        exp_z = np.exp(z)
        p = exp_z / exp_z.sum()
        # Gradient for b
        grad = np.zeros(n)
        for k in range(len(conv)):
            if p[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += p[k] * a[idx]
        b = b * np.exp(-eta * grad)
        b = b / b.sum()
    return b


def nesterov_eg(a0, time_limit, max_iters=500000, n_eg_iters=20, eta=0.1):
    """Nesterov with EG-approximated best response (faster per iteration)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    avg_a = a.copy()
    avg_decay = 0.9995

    t0 = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = eg_best_response(y, n_iters=n_eg_iters, eta=eta)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0)
        y_new = y_new / max(y_new.sum(), 1e-15)

        a = a_new
        y = y_new
        avg_a = avg_decay * avg_a + (1.0 - avg_decay) * a_new
        avg_a = np.maximum(avg_a, 0)
        avg_a = avg_a / max(avg_a.sum(), 1e-15)

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NE iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 3:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02)
    else:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: Nesterov LP + LSE polish + local search + multi-length."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Get starting sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP (main) - 55% time ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} ===")
    t1 = 0.55 * (budget_s - 30)
    remaining = global_deadline - time.monotonic()
    t1 = min(t1, remaining)
    a_out, _ = nesterov_lp_fast(best_seq, t1)
    try_update(a_out, "nesterov_lp")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Nesterov EG (faster inner loop) - 15% time ===
    print(f"\n=== Phase 2: Nesterov EG ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t2 > 30:
        a_eg, _ = nesterov_eg(best_seq, t2, n_eg_iters=15, eta=0.15)
        try_update(a_eg, "nesterov_eg")
        # Also try with different eta
        remaining = global_deadline - time.monotonic()
        if remaining > 30:
            a_eg2, _ = nesterov_eg(best_seq, remaining * 0.5, n_eg_iters=10, eta=0.08)
            try_update(a_eg2, "nesterov_eg2")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: LSE L-BFGS polishing - 15% time ===
    print(f"\n=== Phase 3: LSE Polish ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 30), remaining * 0.6)
    if t3 > 30:
        a_lse, _ = lse_lbfgs(best_seq, t3 * 0.6, T_start=20.0, T_min=0.05, n_stages=25, iters_per_stage=40)
        try_update(a_lse, "lse_pass1")
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lse_lbfgs(best_seq, remaining * 0.5, T_start=5.0, T_min=0.02, n_stages=20, iters_per_stage=35)
            try_update(a_lse2, "lse_pass2")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Peak-directed local search - 7% time ===
    print(f"\n=== Phase 4: Peak Search ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.07 * (budget_s - 30), remaining * 0.5)
    if t4 > 15:
        a_ps, _ = peak_directed_search(best_seq, t4, n_trials=10000)
        try_update(a_ps, "peak_search")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multi-length exploration - 8% time ===
    print(f"\n=== Phase 5: Multi-length ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.08 * (budget_s - 30), remaining * 0.7)
    if t5_total > 40:
        for n_try in [n_base + 400, n_base - 200]:
            remaining = global_deadline - time.monotonic()
            if remaining < 30:
                break
            t5 = min(t5_total / 2.0, remaining * 0.6)
            if t5 < 20:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_fast(init_try, t5 * 0.7)
            try_update(a_try, f"len_{n_try}")
            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = lse_lbfgs(a_try, remaining * 0.4, T_start=10.0, T_min=0.1, n_stages=10, iters_per_stage=25)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final combined polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 30:
        a_f1, _ = nesterov_lp_fast(best_seq, remaining * 0.4)
        try_update(a_f1, "final_nlp")
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_f2, _ = lse_lbfgs(best_seq, remaining * 0.5, T_start=5.0, T_min=0.02, n_stages=15, iters_per_stage=30)
            try_update(a_f2, "final_lse")
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_f3, _ = peak_directed_search(best_seq, remaining * 0.6, n_trials=5000)
            try_update(a_f3, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
