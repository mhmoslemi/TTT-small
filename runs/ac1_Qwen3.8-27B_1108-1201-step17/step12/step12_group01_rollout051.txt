def fast_conv_full(a, b):

    n = len(a)

    fft_size = 1 << (size - 1).bit_length()

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def get_lp_solver():
    """Find a working LP solver."""

            prob.solve(solver=solver_name)

def build_T_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

def solve_lp_br(a_fixed, solver, T_cache=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    if T_cache is not None:
        T = T_cache
    else:
        T = build_T_matrix(a_fixed)

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver]
    for alt in [cp.CBC, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GLPK]:
        if alt != solver:
            solvers_to_try.append(alt)

    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def soft_grad_obj_z(z, T_temp, topk):
    """
    Compute top-k LSE objective and gradient in z-space.
    z -> a = softmax(z), then compute top-k LSE of autocorr(a).
    """
    a = softmax_fn(z)
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T_temp * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T_temp
    w = exp_Tc / exp_Tc.sum()

    # Gradient of conv_k(a) w.r.t. a: grad_k[i] = a[k-i] if 0<=k-i<n
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    # Chain rule through softmax
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return lse_val, grad_z


def adam_soft_grad(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=40, topk=20):
    """
    Adam-optimized top-k LSE with aggressive multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Learning rate: start high, decay
        base_lr = 0.01 * min(1.0, 100.0 / n)
        lr = base_lr * (0.85 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 60.0) * 8))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = soft_grad_obj_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 60 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Grad stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def subgrad_nesterov(a0, time_limit, T_peak=50.0, topk=10):
    """
    Subgradient method with Nesterov momentum on the soft-max objective.
    Works in the original a-space (simplex constrained).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Polyak-style step size
    step = 0.01
    decay = 0.9999

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(y)
        k_use = min(topk, len(c))
        top_idx = np.argsort(c)[::-1][:k_use]
        top_vals = c[top_idx]

        Tc = T_peak * top_vals
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()

        # Compute gradient
        grad = np.zeros(n)
        for idx in range(k_use):
            K = top_idx[idx]
            wK = w[idx]
            js = max(0, K - (n - 1))
            je = min(n - 1, K)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                grad[idx_range] += 2.0 * wK * y[K - idx_range]

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = a - step * grad
        y_new = np.maximum(y_new, 0)
        sn = y_new.sum()
        if sn > 1e-15:
            y_new = y_new / sn

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - a)
        a_new = np.maximum(a_new, 0)
        sn2 = a_new.sum()
        if sn2 > 1e-15:
            a_new = a_new / sn2

        a = a_new
        y = y_new
        step *= decay

        if t % 100 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_optimized(a0, time_limit, seed=42):
    """
    Focused peak shaving: directly reduce the top peaks by identifying
    the most contributing pairs and redistributing mass.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:20]
        top_peaks = top_peaks[:30]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) > 0:
                safe_sorted = safe[np.argsort(a[safe])]
                dst_pool = safe_sorted[:max(3, len(safe_sorted) // 4)]
            else:
                dst_pool = np.where(np.arange(n) != src)[0]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.0005, 0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.04)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    else:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Combines LP Nesterov, soft gradient Adam, multi-length search, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length (main workhorse, ~40% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.40, 650.0)
    if t1 > 50:
        a_out, _ = nesterov_lp(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Soft gradient Adam at base length (~25% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Soft Grad Adam n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_out, _ = adam_soft_grad(best_seq, t2, T_start=200.0, T_end=0.05, n_stages=40, topk=25)
        try_update(a_out, f"grad_n{n_base}")

    # === Phase 3: Peak shaving at base length (~10% time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Peak Shaving n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_out, _ = peak_shaving_optimized(best_seq, t3, seed=456)
        try_update(a_out, f"peak_n{n_base}")

    # === Phase 4: LP Nesterov at longer length (~15% time) ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LP Nesterov n={n_long} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 50:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp(init_long, t4, solver)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 5: Soft gradient at longer length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        n_long2 = n_base + 400
        t5 = min(remaining * 0.20, 200.0)
        if t5 > 30:
            print(f"\n=== Phase 5: Soft Grad n={n_long2} ({remaining:.0f}s left) ===")
            init_long2 = interpolate_sequence(best_seq, n_long2)
            a_out, _ = adam_soft_grad(init_long2, t5, T_start=100.0, T_end=0.1, n_stages=30, topk=20)
            try_update(a_out, f"grad_n{n_long2}")

    # === Phase 6: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t6 = min(remaining * 0.30, 100.0)
        if t6 > 10:
            print(f"\n=== Phase 6: Final Peak Shaving ({remaining:.0f}s left) ===")
            a_out, _ = peak_shaving_optimized(best_seq, t6, seed=789)
            try_update(a_out, "final_peak")

    # === Phase 7: Subgradient Nesterov on best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t7 = min(remaining * 0.40, 120.0)
        if t7 > 15:
            print(f"\n=== Phase 7: SubGrad Nesterov ({remaining:.0f}s left) ===")
            a_out, _ = subgrad_nesterov(best_seq, t7, T_peak=50.0, topk=12)
            try_update(a_out, "subgrad")

    # === Final: one more peak shave with whatever time remains ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = remaining * 0.7
        if t8 > 5:
            print(f"\n=== Final Polish ({t8:.0f}s) ===")
            a_out, _ = peak_shaving_optimized(best_seq, t8, seed=101)
            try_update(a_out, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
