```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on softmax parameterization with LogSumExp-smoothed full gradient
2. Spectral (flat-spectrum) initialization for diverse starting points
3. Multi-temperature annealing with adaptive learning rate
4. Cross-length transfer and diverse initializations
5. Final polish with coordinate descent
Key: O(n log n) per step via FFT, Adam for adaptive per-coordinate learning,
full gradient (not just top-k subgradient) for smoother optimization.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def compute_grad_b(w, a):
    """
    Compute grad_b[j] = 2 * sum_m w[j+m] * a_m for j=0,...,n-1
    where w has length 2n-1 and a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # B = reverse(w), then c[j] = (B * A)[n-1+j]
    B = w[::-1]
    conv_len = len(B) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Bf = rfft(B, fft_size)
    Af = rfft(a, fft_size)
    conv = irfft(Bf * Af, fft_size)
    return conv[n-1 : n-1+n] * 2.0

def softmax_grad(F, a, T=10.0):
    """
    Compute gradient of F = 2*n*max(b)/S^2 using LogSumExp smoothing.
    Returns gradient with respect to a.
    """
    n = len(a)
    S = a.sum()
    b = fast_autocorr(a)
    
    # Softmax weights on b
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    w = exp_Tb / exp_Tb.sum()
    
    # max_b_approx = (Tb.max() + log(sum(exp_Tb))) / T
    max_b_approx = (Tb.max() + np.log(exp_Tb.sum())) / T
    
    # grad_b[j] = 2 * sum_m w[j+m] * a_m
    grad_b = compute_grad_b(w, a)
    
    # Total gradient of F w.r.t. a
    # F = 2*n*max_b / S^2
    # dF/da[j] = 2*n*grad_b[j]/S^2 - 4*n*max_b/S^3
    grad = 2.0 * n * grad_b / (S * S) - 4.0 * n * max_b_approx / (S ** 3)
    
    return grad, max_b_approx

def adam_optimize(a0, time_limit, T_start=20.0, T_end=1.0, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, seed=42, verbose=True):
    """
    Adam optimizer on softmax parameterization.
    a = softmax(z), optimize z.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.max()  # numerical stability
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    t0 = time.monotonic()
    
    # Warmup and total iterations estimate
    # Assume ~5ms per iteration for n=1600
    est_iter_time = 0.005 * (n / 1600.0) ** 1.2
    max_iters = int(time_limit / est_iter_time)
    
    best_a = a_init.copy()
    best_score = fast_score(a_init)
    
    # Temperature schedule: linear from T_start to T_end
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current temperature
        T = T_start + (T_end - T_start) * min(1.0, t / (max_iters * 0.7))
        
        # Current learning rate with warmup + decay
        warmup_iters = max(50, max_iters // 50)
        if t < warmup_iters:
            lr_t = lr * t / warmup_iters
        else:
            progress = (t - warmup_iters) / max(1, max_iters - warmup_iters)
            lr_t = lr * 0.5 * (1 + np.cos(np.pi * progress))
        
        # Compute a from z
        z_shift = z - z.max()
        e = np.exp(z_shift)
        a = e / e.sum()
        
        # Compute gradient in a-space
        grad_a, max_b_approx = softmax_grad(a, T=T)
        
        # Convert to z-space gradient
        # da_j/dz_k = a_j * (delta_jk - a_k)
        # grad_z_k = sum_j grad_a_j * da_j/dz_k = a_k * (grad_a_k - sum_j grad_a_j * a_j)
        S_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - S_grad)
        
        # Adam update
        t_float = float(t)
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t_float)
        v_hat = v / (1 - beta2 ** t_float)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check score periodically
        check_interval = max(1, min(50, t // 100))
        if t % check_interval == 0:
            z_shift = z - z.max()
            a_new = np.exp(z_shift)
            a_new = a_new / a_new.sum()
            sc = fast_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.2f}, lr={lr_t:.6f}")
    
    # Final check
    z_shift = z - z.max()
    a_final = np.exp(z_shift)
    a_final = a_final / a_final.sum()
    sc = fast_score(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    if verbose:
        print(f"    Adam done: best={best_score:.6f}")
    
    return best_a, best_score

def spectral_init(n, rng, n_peaks=1):
    """
    Generate a sequence with approximately flat |DFT|^2 spectrum.
    Start with random phases in frequency domain, take IDFT, project to non-negative.
    """
    # For real sequence, DFT has conjugate symmetry
    # A[0] is real (DC), A[n/2] is real (Nyquist) if n even
    # A[k] and A[n-k] are conjugates
    
    # Strategy: set |A[k]| to be approximately constant, random phases
    A = np.zeros(n, dtype=complex)
    target_power = 1.0 / n  # normalize so sum(|A|^2) = 1
    
    A[0] = np.sqrt(n * target_power)  # DC component (real, positive)
    
    for k in range(1, n // 2 + 1):
        if k == n - k:  # Nyquist (n even)
            phase = rng.uniform(0, 2 * np.pi)
            A[k] = np.sqrt(target_power) * np.exp(1j * phase)
            if A[k].imag != 0:  # should be real
                A[k] = np.abs(A[k])
        else:
            phase = rng.uniform(0, 2 * np.pi)
            A[k] = np.sqrt(target_power) * np.exp(1j * phase)
            A[n - k] = np.conj(A[k])
    
    # IDFT to get time domain
    a = np.real(np.fft.ifft(A)) * n  # scale appropriately
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))
    
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0) + 0.01
    elif variant == 1:

        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
    elif variant == 2:

        a = np.exp(-3 * x ** 2) + 0.01
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2) + 0.01
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2 + 0.01
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Spectral initialization
        return spectral_init(n, rng)
    elif variant == 11:
        # Multi-bump
        a = np.ones(n) * 0.1
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(0.5, 2.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def mirror_descent_v2(a0, time_limit, eta0=0.05, decay_rate=0.6,
                      k_min=1, k_max=20, k_ramp=300, seed=42, verbose=True):
    """
    Improved mirror descent with:
    - Full gradient option (not just top-k)
    - Adaptive k based on convergence
    - Better stall handling
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0
    last_score = best_score
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Determine k with ramp
        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))
        
        # Top-k with Gaussian weights
        top_idx = np.argsort(b)[::-1][k]
        top_k_idx = np.argsort(b)[::-1][:k]
        
        # Gaussian weights based on peak values
        peak_vals = b[top_k_idx]
        peak_range = peak_vals.max() - peak_vals.min()
        if peak_range < 1e-15:
            weights = np.ones(k) / k
        else:
            # Subtract the mean of all b to emphasize peaks
            excess = peak_vals - b.mean()
            sigma = max(peak_range * 0.3, 1e-10)
            weights = np.exp(-0.5 * ((peak_vals - peak_vals.mean()) / sigma) ** 2)
            weights = weights / weights.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_k_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]
        
        # Learning rate
        eta = eta0 / (1.0 + t / 200.0) ** decay_rate
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / max(a_new.sum(), 1e-15)
        else:
            a_new = a_new / s_new
        
        a = a_new
        
        # Check score
        if t < 200:
            check_interval = 2
        elif t < 2000:
            check_interval = 5
        else:
            check_interval = 20
        
        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                if sc > last_score * 1.001:
                    stall_count += check_interval
                else:
                    stall_count += 1
            last_score = sc
        
        # Stall handling
        if stall_count > 300:
            # Try different noise scales
            noise_scale = [0.02, 0.05, 0.1][min(2, stall_count // 100)]
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD2 t={t} ({elapsed:.0f}s): best={best_score:.6f}, k={k}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent: try to reduce the max autocorrelation peak by
    adjusting individual elements.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_idx = int(np.argmax(b))
        
        # Which elements contribute to this peak?
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        
        if contrib.sum() < 1e-15:
            break
        
        # Try reducing the top contributor
        src = int(idx[np.argmax(contrib)])
        
        # Try moving a small amount to a random other position
        for _ in range(5):
            dst = int(rng.integers(n))
            if dst == src:
                continue
            for frac in [0.001, 0.003, 0.01, 0.03]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    break
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                    break
                else:
                    break
        
        if stall > 200:
            break
        else:
            stall += 1
    
    if verbose:
        print(f"    CoordDesc ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")
    return best_a, best_score

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on softmax parameterization with LogSumExp-smoothed full gradient (O(n log n) per step via FFT)
2. Improved mirror descent with adaptive top-k Gaussian-weighted subgradient
3. Spectral (flat-spectrum) initialization for diverse starting points
4. Multi-temperature annealing with adaptive learning rate scheduling
5. Cross-length transfer (interpolate to longer, re-optimize)
6. Coordinate descent polish
Key: Full gradient computation via FFT cross-correlation, Adam for adaptive per-coordinate learning rates,
multiple diverse initializations (spectral, structured, parent-based).
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def compute_grad_b(w, a):
    """
    Compute grad_b[j] = 2 * sum_m w[j+m] * a_m for j=0,...,n-1
    where w has length 2n-1 and a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    B = w[::-1]
    conv_len = len(B) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Bf = rfft(B, fft_size)
    Af = rfft(a, fft_size)
    conv = irfft(Bf * Af, fft_size)
    return conv[n - 1:2 * n - 1][::-1] * 2.0


def lse_gradient(a, T=10.0):
    """
    Compute gradient of F = 2*n*max(b)/S^2 using LogSumExp smoothing.
    Returns (grad_a, max_b_approx, b).
    """
    n = len(a)
    S = a.sum()
    b = fast_autocorr(a)

    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    w = exp_Tb / exp_Tb.sum()
    max_b_approx = (Tb.max() + np.log(exp_Tb.sum())) / T

    grad_b = compute_grad_b(w, a)
    grad = 2.0 * n * grad_b / (S * S) - 4.0 * n * max_b_approx / (S ** 3)

    return grad, max_b_approx, b


def adam_optimize(a0, time_limit, T_start=20.0, T_end=1.0, lr=0.005,
                  beta1=0.9, beta2=0.999, eps=1e-8, seed=42, verbose=True):
    """
    Adam optimizer on softmax parameterization with LogSumExp-smoothed objective.
    a = softmax(z), optimize z with Adam.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.max()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()

    # Estimate iterations
    est_iter_time = 0.004 * (n / 1600.0) ** 1.3
    max_iters = max(100, int(time_limit / est_iter_time))

    best_a = a_init.copy()
    best_score = fast_score(a_init)
    warmup_iters = max(50, max_iters // 50)

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        T = T_start + (T_end - T_start) * min(1.0, t / max(1, int(max_iters * 0.7)))

        if t < warmup_iters:
            lr_t = lr * t / warmup_iters
        else:
            progress = (t - warmup_iters) / max(1, max_iters - warmup_iters)
            lr_t = lr * 0.5 * (1 + np.cos(np.pi * progress))

        z_shift = z - z.max()
        e = np.exp(z_shift)
        a = e / e.sum()

        grad_a, max_b_approx, b = lse_gradient(a, T=T)

        S_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - S_grad)

        tf = float(t)
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** tf)
        v_hat = v / (1 - beta2 ** tf)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        check_interval = max(1, min(30, t // 50))
        if t % check_interval == 0:
            z_shift = z - z.max()
            a_new = np.exp(z_shift)
            a_new = a_new / a_new.sum()
            sc = fast_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if verbose and t % max(1, max_iters // 20) == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t}/{max_iters} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.2f}, lr={lr_t:.6f}")

    z_shift = z - z.max()
    a_final = np.exp(z_shift)
    a_final = a_final / a_final.sum()
    sc = fast_score(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    if verbose:
        print(f"    Adam done: best={best_score:.6f}")

    return best_a, best_score


def mirror_descent_v2(a0, time_limit, eta0=0.04, decay_rate=0.65,
                      k_min=1, k_max=20, k_ramp=300, seed=42, verbose=True):
    """
    Improved mirror descent with adaptive top-k Gaussian-weighted subgradient.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)

        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        top_k_idx = np.argsort(b)[::-1][:k]
        peak_vals = b[top_k_idx]
        peak_range = peak_vals.max() - peak_vals.min()
        if peak_range < 1e-15:
            weights = np.ones(k) / k
        else:
            sigma = max(peak_range * 0.3, 1e-10)
            weights = np.exp(-0.5 * ((peak_vals - peak_vals.mean()) / sigma) ** 2)
            weights = weights / weights.sum()

        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_k_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]

        eta = eta0 / (1.0 + t / 200.0) ** decay_rate

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / max(a_new.sum(), 1e-15)
        else:
            a_new = a_new

        a = a

        if t < 200:
            check_interval = 2
        elif t < 2000:
            check_interval = 5
        else:
            check_interval = 20

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > 400:
            noise_scale = [0.02, 0.05, 0.1][min(2, stall_count // 100)]
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / max(a.sum(), 1e-15)
            stall_count = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD2 t={t} ({elapsed:.0f}s): best={best_score:.6f}, k={k}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent: reduce max autocorrelation peak by moving mass
    from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_autocorr(a)
        peak_idx = int(np.argmax(b))

        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]

        if contrib.sum() < 1e-15:
            break

        src = int(idx[np.argmax(contrib)])
        improved = False

        peak_mask = np.zeros(n, dtype=bool)
        top5 = np.argsort(b)[::-1][:5]
        for k2 in top5:
            k2 = int(k2)
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        candidates = np.where(~peak_mask)[0]
        if len(candidates) < 5:
            candidates = np.where(np.arange(n) != src)[0]

        for _ in range(3):
            dst = int(candidates[rng.integers(len(candidates))]) if len(candidates) > 0 else int(rng.integers(n))
            if dst == src:
                continue
            for frac in [0.001, 0.003, 0.01, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    break
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break
                else:
                    break

        if not improved:
            stall += 1
            if rng.integers(5) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

            if stall > 300:
                break

    if verbose:
        print(f"    CoordDesc ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def spectral_init(n, rng):
    """
    Generate a sequence with approximately flat |DFT|^2 spectrum.
    Random phases in frequency domain, IDFT, project to non-negative.
    """
    A = np.zeros(n, dtype=complex)
    target_power = 1.0 / n

    A[0] = np.sqrt(n * target_power)

    for k in range(1, (n // 2) + 1):
        phase = rng.uniform(0, 2 * np.pi)
        if k == n - k:
            A[k] = np.sqrt(target_power) * (1 if rng.random() > 0.5 else -1)
        else:
            A[k] = np.sqrt(target_power) * np.exp(1j * phase)
            A[n - k] = np.conj(A[k])

    a = np.real(np.fft.ifft(A)) * n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        # Fallback: try different random seed
        a = np.abs(np.real(np.fft.ifft(A * np.exp(1j * rng.uniform(0, 2*np.pi, n)))))
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
    return a / s


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0) + 0.01
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2) + 0.01
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2) + 0.01
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2 + 0.01
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        return spectral_init(n, rng)
    elif variant == 11:
        a = np.ones(n) * 0.1
        for _ in range(int(rng.integers(2, 6))):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(0.5, 2.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def multi_start_adam(a0, time_limit, n_starts=3, seed=42, verbose=True):
    """
    Run Adam from multiple perturbations of a0, keep the best result.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    best_a = a0.copy()
    best_score = fast_score(a0)
    t_total = time_limit
    t_per = t_total / (n_starts + 1)

    # First: run from a0 directly
    t0 = time.monotonic()
    a_out, sc_out = adam_optimize(a0, min(t_per, t_total - (time.monotonic() - t0)),
                                   T_start=20.0, T_end=1.0, lr=0.004, seed=seed, verbose=False)
    if sc_out < best_score:
        best_score = sc_out
        best_a = a_out.copy()

    # Then: run from perturbed versions
    for i in range(1, n_starts + 1):
        remaining = t_total - (time.monotonic() - t0)
        if remaining < 30:
            break
        t_this = min(t_per, remaining - 10)
        if t_this < 20:
            break

        noise_scale = 0.03 * i
        a_perturb = a0 + noise_scale * rng.standard_normal(n)
        a_perturb = np.maximum(a_perturb, 0)
        a_perturb = a_perturb / a_perturb.sum()

        a_out, sc_out = adam_optimize(a_perturb, t_this,
                                       T_start=15.0, T_end=0.5, lr=0.003,
                                       seed=seed + i * 100, verbose=False)
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"    MultiStart: improvement at start {i}: {best_score:.6f}")

    if verbose:
        print(f"    MultiStart done: best={best_score:.6f}")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Adam optimizer at parent length (full gradient, softmax parameterization)
    Phase 2: Improved Mirror Descent
    Phase 3: Multi-start Adam (diversity)
    Phase 4: Spectral initialization + Adam
    Phase 5: Cross-length transfer (longer sequences)
    Phase 6: Even longer sequences
    Phase 7: Coordinate descent polish
    Phase 8: Final combined polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1760.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fast_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_parent = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = diverse_init(n_parent, rng, variant=0)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Adam at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam at n={n_parent} (remaining={remaining:.0f}s) ===")
    t_adam1 = min(remaining * 0.18, 280.0)
    if t_adam1 > 30:
        a_out, _ = adam_optimize(best_seq, t_adam1, T_start=30.0, T_end=2.0,
                                  lr=0.005, beta1=0.9, beta2=0.999, seed=1, verbose=True)
        try_update(a_out, f"adam1_n{n_parent}")

    # === Phase 2: Mirror Descent v2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror Descent v2 at n={n_parent} (remaining={remaining:.0f}s) ===")
    t_md1 = min(remaining * 0.15, 250.0)
    if t_md1 > 30:
        a_out, _ = mirror_descent_v2(best_seq, t_md1, eta0=0.04, decay_rate=0.65,
                                      k_min=1, k_max=20, k_ramp=300, seed=2, verbose=True)
        try_update(a_out, f"md2_n{n_parent}")

    # === Phase 3: Multi-start Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Multi-start Adam at n={n_parent} (remaining={remaining:.0f}s) ===")
    t_multi = min(remaining * 0.12, 200.0)
    if t_multi > 30 and best_seq is not None:
        a_out, _ = multi_start_adam(best_seq, t_multi, n_starts=3, seed=3, verbose=True)
        try_update(a_out, f"multi_n{n_parent}")

    # === Phase 4: Spectral init + Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Spectral init + Adam at n={n_parent} (remaining={remaining:.0f}s) ===")
    t_spec = min(remaining * 0.10, 150.0)
    if t_spec > 20:
        spec_init = spectral_init(n_parent, rng)
        spec_score = fast_score(spec_init)
        print(f"    Spectral init score: {spec_score:.6f}")
        a_out, _ = adam_optimize(spec_init, t_spec, T_start=20.0, T_end=1.0,
                                  lr=0.004, seed=4, verbose=True)
        try_update(a_out, f"spec_n{n_parent}")

        # Also try diverse structured init
        remaining = time_budget - (time.monotonic() - t_global)
        t_div = min(remaining * 0.05, 80.0)
        if t_div > 15:
            div_init = diverse_init(n_parent, rng, variant=4)  # U-shaped
            a_out, _ = adam_optimize(div_init, t_div, T_start=15.0, T_end=1.0,
                                      lr=0.004, seed=5, verbose=False)
            try_update(a_out, f"div_u_n{n_parent}")

    # === Phase 5: Cross-length transfer (1.5x) ===
    n_long = int(round(n_parent * 1.5))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Longer length n={n_long} (remaining={remaining:.0f}s) ===")
    t_long = min(remaining * 0.18, 280.0)
    if t_long > 30 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)
        a_out, _ = adam_optimize(init_long, t_long, T_start=25.0, T_end=1.5,
                                  lr=0.003, seed=6, verbose=True)
        try_update(a_out, f"adam_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        t_md_long = min(remaining * 0.08, 120.0)
        if t_md_long > 20 and best_seq is not None:
            init_long2 = interpolate_to_length(best_seq, n_long)
            a_out, _ = mirror_descent_v2(init_long2, t_md_long, eta0=0.03,
                                          decay_rate=0.55, k_min=1, k_max=15,
                                          k_ramp=400, seed=7, verbose=False)
            try_update(a_out, f"md2_n{n_long}")

    # === Phase 6: Even longer (2x) ===
    n_longer = int(round(n_parent * 2.0))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Even longer n={n_longer} (remaining={remaining:.0f}s) ===")
    t_longer = min(remaining * 0.12, 180.0)
    if t_longer > 30 and best_seq is not None:
        init_longer = interpolate_to_length(best_seq, n_longer)
        a_out, _ = adam_optimize(init_longer, t_longer, T_start=20.0, T_end=1.0,
                                  lr=0.002, seed=8, verbose=True)
        try_update(a_out, f"adam_n{n_longer}")

    # === Phase 7: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Coordinate Descent (remaining={remaining:.0f}s) ===")
    t_cd = min(remaining * 0.08, 100.0)
    if t_cd > 10 and best_seq is not None:
        a_out, _ = coordinate_descent(best_seq, t_cd, seed=9, verbose=True)
        try_update(a_out, f"cd_n{len(best_seq)}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final Polish (remaining={remaining:.0f}s) ===")
    if best_seq is not None and remaining > 20:
        n_final = len(best_seq)

        # Adam with very low temperature (sharp max approximation)
        t_f1 = min(remaining * 0.35, 100.0)
        if t_f1 > 10:
            a_out, _ = adam_optimize(best_seq, t_f1, T_start=5.0, T_end=0.2,
                                      lr=0.001, beta1=0.95, beta2=0.9999, seed=10, verbose=True)
            try_update(a_out, f"final_adam_n{n_final}")

        # MD with high k (explore more of the autocorrelation)
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.35, 80.0)
        if t_f2 > 10:
            a_out, _ = mirror_descent_v2(best_seq, t_f2, eta0=0.01,
                                          decay_rate=0.8, k_min=5, k_max=35,
                                          k_ramp=100, seed=11, verbose=False)
            try_update(a_out, f"final_md_n{n_final}")

        # Final CD
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.30, 60.0)
        if t_f3 > 5:
            a_out, _ = coordinate_descent(best_seq, t_f3, seed=12, verbose=False)
            try_update(a_out, f"final_cd_n{len(best_seq)}")

    # Final evaluation with official function
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
