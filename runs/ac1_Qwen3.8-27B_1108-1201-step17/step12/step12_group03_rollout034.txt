import numpy as np

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""

def proj_simplex(v):
    """Project vector v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, b, n_peaks=5, temperature=1.0):
    """Compute weighted subgradient of max_k (a*a)_k using top peaks."""
    n = len(a)
    max_b = b.max()
    # Weight peaks by exponential of (b_k - max_b) / temperature
    logits = (b - max_b) / temperature
    logits -= logits.max()
    weights = np.exp(logits)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        # Only the max peak
        k_star = np.argmax(b)
        grad = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                grad[j] += 2.0 * a[k_j]
        return grad
    weights = weights / w_sum
    # Focus on top peaks for efficiency
    top_idx = np.argsort(weights)[::-1][:n_peaks]
    weights = weights[top_idx]
    w_sum = weights.sum()
    weights = weights / w_sum

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        for j in range(n):
            k_j = k - j
            if 0 <= k_j < n:
                grad[j] += 2.0 * w * a[k_j]
    return grad


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, verbose=False):
    """Nesterov-accelerated LP best-response method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    stall_count = 0

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if t % 200 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")

        if stall_count > 5000 and t > 500:
            if verbose:
                print(f"    Early stop at t={t}")
            break

    return best_a, best_score, iter_count


def subgradient_descent(a0, time_limit, n_peaks=5, temperature=0.01):
    """Projected subgradient descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    rng = np.random.default_rng(12345)
    y = a.copy()  # Nesterov momentum

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        grad = compute_subgradient(y, b, n_peaks=n_peaks, temperature=temperature)

        # Adaptive step size: decrease if no improvement recently
        progress = it / 50000.0
        eta_t = max(0.001, eta * (1.0 - 0.5 * progress))

        # Nesterov momentum
        t_k = 2.0 / (it + 2.0)
        y_new = (1 - t_k) * y + t_k * a
        # Actually for NAG: y_{k+1} = x_{k+1} + (t_k/(t_{k+1}+1))(x_{k+1} - x_k)
        # Simplified: just use heavy ball
        y_new = a + 0.1 * (a - y)
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s > 1e-15:
            y_new = y_new / y_new_s

        # Compute gradient at y_new
        b_y = fast_autocorr(y_new)
        grad_y = compute_subgradient(y_new, b_y, n_peaks=n_peaks, temperature=temperature)

        # Step
        a_raw = y_new - eta_t * grad_y
        a_new = proj_simplex(a_raw)

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            # Increase step size slightly on improvement
            eta = min(0.1, eta * 1.01)
        else:
            # Decrease step size
            eta = max(0.0005, eta * 0.95)

        a = a_new
        y = y_new

            n_pert = rng.integers(1, min(10, n))

            a_pert = a.copy()
            for si, di in zip(src, dst):
                amt = a_pert[si] * 0.01
                a_pert[si] -= amt
                a_pert[di] += amt
            a_pert = np.maximum(a_pert, 0)
            a_pert_s = a_pert.sum()

                    best_a = a_pert.copy()
                    a = a_pert

        if it % 1000 == 0:
            print(f"    SG it={it}: best={best_score:.6f}, eta={eta_t:.6f}")

    return best_a, best_score


def pairwise_descent(a0, time_limit, seed=42):
    """
    Systematic pairwise coordinate descent.
    At each step, identify the peak, find positions contributing to it,
    and move mass to positions that reduce the peak without increasing others too much.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 2000

    # Pre-compute the "contribution matrix" structure
    # For peak k, position i contributes a[i] * a[k-i]
    # Moving delta from i to j changes peak k by:
    #   delta * (a[k-j] - a[k-i]) + delta^2 * (correction terms)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        # Top peaks
        n_top = min(8, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_top]

        # Compute pressure: for each position, how much it contributes to top peaks
        pressure = np.zeros(n)
        for k in top_peaks[:4]:
            for j in range(n):
                k_j = k - j
                if 0 <= k_j < n:
                    pressure[j] += a[k_j]

        # Find high-pressure sources and low-pressure targets
        src_candidates = np.argsort(pressure)[::-1][:50]
        dst_candidates = np.argsort(pressure)[:50]

        improved = False
        # Try moves from high-pressure to low-pressure positions
        for src in src_candidates:
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                # Try moving a fraction of mass
                for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        break
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        break
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Try random moves when stalled
            if stall % 5 == 0:
                n_moves = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_moves):
                    src = rng.integers(0, n)
                    dst = rng.integers(0, n)
                    if src == dst:
                        continue
                    frac = rng.uniform(0.002, 0.02)
                    amt = a_trial[src] * frac
                    if amt < 1e-12:
                        continue
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s > 1e-15:
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > max_stall:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fine_polish(a0, time_limit, seed=99):
    """Very fine local optimization using small random steps with simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    current_score = best_score
    T = 0.001  # initial temperature for annealing
    T_min = 1e-6
    cooling = 0.995

    while time.monotonic() - t0 < time_limit:
        # Generate a random move
        n_moves = rng.integers(1, 4)
        a_trial = a.copy()
        for _ in range(n_moves):
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue
            frac = rng.uniform(0.001, 0.01)
            amt = a_trial[src] * frac
            if amt < 1e-12:
                continue
            a_trial[src] -= amt
            a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0)
        a_trial_s = a_trial.sum()
        if a_trial_s < 1e-15:
            continue
        a_trial = a_trial / a_trial_s
        score = 2.0 * n * fast_autocorr(a_trial).max()

        # Acceptance criterion (simulated annealing)
        delta = score - current_score
        if delta < 0 or (T > T_min and rng.uniform() < np.exp(-delta / T)):
            a = a_trial
            current_score = score
            if score < best_score:
                best_score = score
                best_a = a.copy()
        T *= cooling

        if int((time.monotonic() - t0) * 10) % 100 == 0:
            if time.monotonic() - t0 > 30 and int((time.monotonic() - t0) * 10) % 500 == 0:
                print(f"    Polish: best={best_score:.6f}, T={T:.2e}")

    return best_a, best_score

        print(f"\n=== Phase 7: Final polish ({t7:.0f}s) ===")
        a_out, sc_out = fine_polish(best_seq, t7, seed=271)
        try_update(a_out, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(max(0.0, x)) for x in best_seq]
