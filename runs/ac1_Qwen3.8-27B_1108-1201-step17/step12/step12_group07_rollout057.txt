This is a zero-sum game where:

- Player 2 (maximizer) chooses `p` (probability distribution over lags)
- Payoff: `a^T A(p) a`

The value of the game is the optimal score (times 2n).

By von Neumann's minimax theorem, `min_a max_p a^T A(p) a = max_p min_a a^T A(p) a`.

The inner minimization `min_a a^T A(p) a` subject to `a >= 0, sum(a) = 1` is a quadratic program. If `A(p)` is positive semi-definite, this is convex and can be solved efficiently.

For `A(p) = sum_j p_j T_j`, the matrix `T_j` has eigenvalues in `[-1, 1]` (it's a "reversed identity" or "anti-diagonal" matrix). So `A(p)` has eigenvalues in `[-1, 1]` and is NOT necessarily positive semi-definite.

However, for the specific structure of the convolution problem, the relevant eigenvalues might be positive. Let me check: `T_j` for lag `j` has 1s on the anti-diagonal (offset by `j`). The eigenvalues of such a matrix are `e^{iπk/n}` for `k = 0, ..., n-1` (approximately). So the eigenvalues are on the unit circle, meaning they can be negative.

This means the inner QP is non-convex, and the game-theoretic approach doesn't directly give a convex algorithm.

However, we can still use **fictitious play** or **mirror descent** in the probability space `p`:
1. Maintain a distribution `p_t` over lags.

4. Update `p_{t+1}` based on the response of `a` (e.g., put more weight on the lag that achieved the maximum).

This is similar to the LP best-response approach but in the dual space.

Actually, the LP best-response is exactly the "best response" of Player 2 (the maximizer) given Player 1's strategy `a`:
`best_response(a) = argmax_{p} a^T A(p) a = the lag `j` that maximizes `(a*a)_j``

def fast_convolve(a):

    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):

        # Hmm, let me just do: d/da_m sum_k p_k conv(a,a)_k
        # = sum_k p_k (a_{k-m} + a_{m-k} * [m != k-m])
        # For m = k-m (i.e., k = 2m), we get 2*a_0... no, a_{k-m} = a_m, so it's just a_m appearing twice = 2*a_m? No.
        # conv(a,a)_k = sum_i a_i * a_{k-i}
        # If we differentiate w.r.t. a_m:
        #   term where i=m: a_{k-m}
        #   term where k-i=m (i.e., i=k-m): a_{k-m} [but only if k-m != m, i.e., k != 2m]
        #   if k = 2m: only one term a_m * a_m, derivative is 2*a_m
        # So: d/da_m = a_{k-m} + a_{k-m} [if k-m != m] = 2*a_{k-m} [if k=2m], else a_{k-m} + a_{m-k}... 
        # Wait no. Let me be very careful.
        # f = sum_k p_k * sum_i a_i * a_{k-i}
        # df/da_m = sum_k p_k * [d/da_m (sum_i a_i * a_{k-i})]
        #          = sum_k p_k * [a_{k-m} + sum indicator(i = k-m, i != m) * a_i]
        #          = sum_k p_k * [a_{k-m} + (a_m if k-m == m and k-m valid else (a_{k-m} if k-m valid and k-m != m else 0))]
        # Hmm this is getting too complicated. Let me just use the simple approximation:
        # df/da_m ≈ 2 * sum_k p_k * a_{|k-m|} (for the symmetric part)
        # Actually the cleanest: conv(a,a)_k = sum_i a_i a_{k-i}
        # df/da_m = sum_k p_k [a_{k-m} (from i=m term) + a_{k-m} (from k-i=m term, i.e., i=k-m)]
        # But if i=m AND k-i=m (i.e., k=2m), these are the same term, so we get 2*a_m.
        # Otherwise if k-m != m, we get a_{k-m} + a_{k-m} = 2*a_{k-m}? No!
        # From i=m: contributes a_{k-m}
        # From k-i=m (i=k-m): contributes a_i = a_{k-m}... wait no, the i=k-m term is a_{k-m} * a_m, and d/da_m of that is a_{k-m}.
        # So total: a_{k-m} + a_{k-m} = 2*a_{k-m} when k-m != m, and 2*a_m when k=2m.
        # So df/da_m = 2 * sum_k p_k * a_{k-m} (summed over valid k-m)
        # This is just 2 * (conv(p, a))_m (with appropriate padding)!

        # So grad_a = 2 * conv(p_padded, a) where p_padded is p placed in the middle
        # p is of length 2n-1, a is of length n
        # conv(p, a) has length 3n-2, and we want the central n elements
        # Actually: (conv(p,a))_m = sum_i p_i * a_{m-i}... 
        # Let me just compute it directly for the active lags:
        grad_a = np.zeros(n)
        for j in range(size):
            if p[j] < 1e-10:
                continue
            coeff = 2.0 * p[j]
            # For each m, the contribution is coeff * a_{j-m} if 0 <= j-m < n
            valid_m_start = max(0, j - (n - 1))
            valid_m_end = min(n - 1, j)
            if valid_m_start <= valid_m_end:
                ms = np.arange(valid_m_start, valid_m_end + 1)
                grad_a[ms] += coeff * a[j - ms]

        # Projected gradient step in z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        step_size = 1.0 / (1.0 + 0.1 * it)
        z_new = z - step_size * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            z = z_new
        else:
            # Revert z, keep a
            step_size *= 0.9
            if step_size < 1e-6:
                break

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=30):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 60 if si < n_stages // 3 else 30
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def simulated_annealing_seq(a0, time_limit, max_iters=200000, T_start=0.01, T_end=0.0001):
    """Simulated annealing with structured perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    current_score = best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    n_temp_steps = max(1, max_iters // 10)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / max_iters
        T = T_start * (T_end / T_start) ** progress

        # Random perturbation
        r = rng.random()
        a_pert = a.copy()
        if r < 0.3:
            # Flip a few elements
            for _ in range(rng.integers(1, 6)):
                idx = rng.integers(n)
                a_pert[idx] = max(0, a[idx] + rng.normal(0, 0.01))
        elif r < 0.5:
            # Block shift
            block_size = rng.integers(5, min(50, n // 4))
            start = rng.integers(0, n - block_size)
            a_pert[start:start + block_size] *= 0.8
            target = rng.integers(0, n - block_size)
            a_pert[target:target + block_size] += a[start:start + block_size] * 0.2
        elif r < 0.7:
            # Smooth perturbation
            kernel_size = 3
            A = rng.chisquare(kernel_size)
            A /= A.sum()
            a_pert = np.convolve(a, A, mode='same')
            a_pert = np.maximum(a_pert, 0)
        elif r < 0.85:
            # Global scale perturbation
            a_pert *= rng.uniform(0.95, 1.05)
            # Add asymmetric tilt
            x = np.linspace(0, 1, n)
            a_pert *= (1.0 + rng.uniform(-0.1, 0.1) * (2 * x - 1))
            a_pert = np.maximum(a_pert, 0)
        else:
            # Two-point swap
            i1, i2 = rng.choice(n, 2, replace=False)
            delta = (a[i1] - a[i2]) * rng.uniform(0.1, 0.5)
            a_pert[i1] = max(0, a[i1] - delta)
            a_pert[i2] = max(0, a[i2] + delta)

        sp = a_pert.sum()
        if sp < 1e-15:
            continue
        a_pert = a_pert / sp
        sc = score_normalized(a_pert)

        # Accept/reject
        if sc < current_score:
            a = a_pert
            current_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            accept_prob = np.exp(-(sc - current_score) / (T + 1e-15))
            if rng.random() < accept_prob:
                a = a_pert
                current_score = sc

    return best_a, best_score


def peak_targeted_refine(a0, time_limit, max_steps=50000):
    """Multiplicative weights update targeting top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve_step = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.005, 0.12)
        else:
            eta *= 0.9
            if eta < 1e-12:
                break

    return best_a, best_score


def structured_init(n, rng, style=0):
    x = np.linspace(0, 1, n)
    if style == 0:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (x - 0.3))
    elif style == 1:
        a = np.exp(-2.0 * (x - 0.5) ** 2) + 0.3
    elif style == 2:
        a = 0.5 + 0.4 * np.sin(4 * np.pi * x + 0.5) + 0.1
    elif style == 3:
        a = (x + 0.1) ** 0.3 * (1.1 - x) ** 0.3
    elif style == 4:
        a = 1.0 + 0.3 * np.cos(6 * np.pi * x)
    elif style == 5:
        a = np.exp(-5.0 * (x - 0.35) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.65) ** 2) + 0.15
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x + 1.0)
    else:
        a = 1.0 + rng.normal(0, 0.1, n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    best_n = n_base

        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()

            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 1: Multi-start quick LP (120s) ===
    print(f"\n=== Phase 1: Multi-start LP exploration (120s) ===")
    t1_total = min(120.0, (_global_deadline - time.monotonic()) * 0.08)
    n_starts = 4
    t1_per = t1_total / n_starts

    # Start 1: parent
    if t1_per > 10:
        a_out, _ = nesterov_lp_diverse(best_seq, t1_per, solver=solver, diversity_interval=100)
        try_update(a_out, "lp_parent")

    # Starts 2-4: structured inits
