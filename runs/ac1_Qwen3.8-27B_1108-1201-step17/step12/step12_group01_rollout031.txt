"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_grad(a, T, k_use=None):
    """
    Compute LSE objective and gradient.
    If k_use is None, use all positions (full LSE).
    If k_use is int, use top-k positions.
    """
    n = len(a)
    c = fast_autocorr(a)
    if k_use is not None and k_use < len(c):
        top_idx = np.argsort(c)[::-1][:k_use]
        top_vals = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        top_vals = c
    
    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    
    grad_a = np.zeros(n)
    for idx in range(len(top_idx)):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    return lse_val, grad_a


def lse_grad_z(z, T, k_use=None):
    """Objective and gradient in z-space."""
    a = softmax_fn(z)
    obj, grad_a = lse_grad(a, T, k_use)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, M_override=None):
    """Solve LP: maximize sum(b) s.t. conv(a_fixed, b) <= M, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    if M_override is not None:
        # Fix M and maximize sum(b)
        T = build_conv_matrix(a_fixed)
        constraints = [T @ b_var <= M_override, b_var >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    else:
        # Original: min M s.t. conv <= M, sum(b)=1
        T = build_conv_matrix(a_fixed)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    if M_override is not None:
        return b_val, M_override
    return b_val, float(M_var.value)


def nesterov_lp_enhanced(a0, time_limit, solver, max_iters=200000):
    """
    Enhanced Nesterov LP with:
    - Exponential moving averages
    - More frequent checking
    - Adaptive step size
    - Perturbation-based restart
    """
    n = len(a0)
    rng = np.random.default_rng(12345)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Multiple EMA with different rates
    ema_fast = a.copy()  # rate 0.1
    ema_med = a.copy()   # rate 0.03
    ema_slow = a.copy()  # rate 0.01

    # Circular buffer for windowed average
    WINDOW = 50
    buf = [a.copy()] * WINDOW
    buf_idx = 0

    stall_count = 0
    base_alpha = 2.0  # base for alpha_t = base_alpha / (t + 2)
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = base_alpha / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Update EMAs
        ema_fast = 0.1 * a + 0.9 * ema_fast
        ema_med = 0.03 * a + 0.97 * ema_med
        ema_slow = 0.01 * a + 0.99 * ema_slow
        
        # Update circular buffer
        buf[buf_idx] = a.copy()
        buf_idx = (buf_idx + 1) % WINDOW

        # Check multiple candidates every 10 iterations
        if t % 10 == 0 or t <= 20:
            candidates = [a, y]
            if t > 5:
                candidates.append(ema_fast)
            if t > 20:
                candidates.append(ema_med)
            if t > 100:
                candidates.append(ema_slow)
                candidates.append(np.mean(buf, axis=0))
            
            for cand in candidates:
                cand = np.maximum(cand, 0)
                cs = cand.sum()
                if cs < 1e-15:
                    continue
                cand = cand / cs
                score = 2.0 * n * fast_autocorr(cand).max()
                if score < best_score:
                    best_score = score
                    best_a = cand.copy()
                    stall_count = 0
                    break
            else:
                stall_count += 1
        
        # Adaptive restart with perturbation
        if stall_count > 500 and t > 100:
            # Perturb the best solution
            a = best_a.copy()
            y = best_a.copy()
            # Small random perturbation
            noise = rng.normal(0, 0.01, n)
            a = a + noise * (a + 1e-6)
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
            y = a.copy()
            ema_fast = a.copy()
            ema_med = a.copy()
            ema_slow = a.copy()
            buf = [a.copy()] * WINDOW
            buf_idx = 0
            stall_count = 0
            if t % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      NLP restart at t={t}: best={best_score:.6f}")

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def full_lse_adam(a0, time_limit, T_start=50.0, T_end=0.2, n_stages=30, k_use=None):
    """
    Full (or top-k) LSE with Adam optimizer.
    If k_use is None, uses full LSE over all 2n-1 positions.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.85 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(20, int(remaining * 4))
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = lse_grad_z(z, stage_T, k_use)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 5 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 25 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_flattening(a0, time_limit, seed=42):

    Aggressive peak flattening with exact line search.

    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def ftrl_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """
    FTRL (Follow The Regularized Leader) with LP best response.
    Uses entropy regularization for natural non-negativity.
    """
    n = len(a0)
    rng = np.random.default_rng(999)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # FTRL state
    z = np.zeros(n)  # cumulative "gradient"
    eta_base = 0.5  # learning rate base

    # Running average
    avg = a.copy()
    n_avg = 0

    stall_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(a, solver)
        if b_star is None:
            continue

        # Compute gradient: "loss" for player 1 is the max of conv(a, b_star)
        # Subgradient w.r.t. a[i] is: the positions k where conv(a,b)[k] = max
        # For FTRL: z += eta * gradient
        conv_ab = np.convolve(a, b_star, 'full')
        max_val = conv_ab.max()
        # Find positions at max
        top_positions = np.where(conv_ab >= max_val * 0.999)[0]
        
        # Gradient: for each position k at max, d/d a[i] conv[a,b][k] = b[k-i]
        grad = np.zeros(n)
        for k in top_positions:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += b_star[k - idx]
        grad /= len(top_positions)
        
        eta_t = eta_base / (1.0 + 0.01 * t)
        z = z + eta_t * grad

        # FTRL update: a[i] = exp(-z[i]) / sum(exp(-z))
        z_shift = z - z.max()
        exp_z = np.exp(z_shift)
        a_new = exp_z / exp_z.sum()
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Running average
        avg = (n_avg * avg + a_new) / (n_avg + 1)
        n_avg += 1

        a = a_new

        # Check
        if t % 20 == 0 or t <= 10:
            for cand in [a, avg]:
                score = 2.0 * n * fast_autocorr(cand).max()
                if score < best_score:
                    best_score = score
                    best_a = cand.copy()
                    stall_count = 0
                    break
            else:
                stall_count += 1

        if stall_count > 300 and t > 50:
            a = best_a.copy()
            z = np.zeros(n)
            avg = a.copy()
            n_avg = 0
            stall_count = 0

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def fast_autocorr(a):

    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def softmax_fn(z):
    """Numerically stable softmax."""
    z_arr = np.asarray(z, dtype=np.float64)
    z_arr = z_arr - z_arr.max()
    e = np.exp(z_arr)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z_arr)) / len(z_arr)
    return e / s


def lse_obj_grad_z(z, T, k_use=None):
    """Objective and gradient in z-space for LSE."""
    a = softmax_fn(z)
    n = len(a)
    c = fast_autocorr(a)
    if k_use is not None and k_use < len(c):
        top_idx = np.argsort(c)[::-1][:k_use]
        top_vals = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        top_vals = c

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return lse_val, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    return b_val, float(M_var.value), prob


def nesterov_lp_enhanced(a0, time_limit, solver):
    """
    Enhanced Nesterov LP with multi-rate EMA and adaptive restart.
    """
    n = len(a0)
    rng = np.random.default_rng(12345)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Multi-rate EMAs
    ema_r1 = a.copy()  # rate 0.15 (fast)
    ema_r2 = a.copy()  # rate 0.05 (medium)
    ema_r3 = a.copy()  # rate 0.01 (slow)
    ema_r4 = a.copy()  # rate 0.003 (very slow)

    # Windowed average buffer
    WINDOW = 40
    buf = [a.copy()] * WINDOW
    buf_idx = 0

    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val, _ = solve_lp(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Update EMAs
        ema_r1 = 0.15 * a + 0.85 * ema_r1
        ema_r2 = 0.05 * a + 0.95 * ema_r2
        ema_r3 = 0.01 * a + 0.99 * ema_r3
        ema_r4 = 0.003 * a + 0.997 * ema_r4

        # Update buffer
        buf[buf_idx] = a.copy()
        buf_idx = (buf_idx + 1) % WINDOW

        # Check candidates every 8 iterations
        if t % 8 == 0 or t <= 16:
            best_local = best_score
            best_local_a = None
            if t > 10:
                for cand in [a, y, ema_r1, ema_r2]:
                    cand = np.maximum(cand, 0)
                    cs = cand.sum()
                    if cs < 1e-15:
                        continue
                    cand = cand / cs
                    score = 2.0 * n * fast_autocorr(cand).max()
                    if score < best_local:
                        best_local = score
                        best_local_a = cand.copy()
            if t > 50:
                for cand in [ema_r3, ema_r4, np.mean(buf, axis=0)]:
                    cand = np.maximum(cand, 0)
                    cs = cand.sum()
                    if cs < 1e-15:
                        continue
                    cand = cand / cs
                    score = 2.0 * n * fast_autocorr(cand).max()
                    if score < best_local:
                        best_local = score
                        best_local_a = cand.copy()

            if best_local_a is not None and best_local < best_score:
                best_score = best_local
                best_a = best_local_a
                stall_count = 0
            else:
                stall_count += 1

        # Adaptive restart
        if stall_count > 400 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            noise = rng.normal(0, 0.005, n)
            a_pert = a + noise * (a + 1e-6)
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
                y = a_pert.copy()
                a = a_pert.copy()
            ema_r1 = a.copy()
            ema_r2 = a.copy()
            ema_r3 = a.copy()
            ema_r4 = a.copy()
            buf = [a.copy()] * WINDOW
            buf_idx = 0
            stall_count = 0

        if t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def ftrl_lp_optimize(a0, time_limit, solver):
    """
    FTRL with LP best response. Uses multiplicative weight updates.
    """
    n = len(a0)
    rng = np.random.default_rng(777)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # FTRL state
    z = np.zeros(n)
    eta_schedule_start = 1.0
    eta_schedule_end = 0.1
    total_est = int(time_limit * 5)  # rough estimate of iterations

    ema_fast = a.copy()
    ema_slow = a.copy()
    stall_count = 0

    for t in range(1, max(100, total_est)):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val, _ = solve_lp(a, solver)
        if b_star is None:
            continue

        # Compute subgradient of max_k conv(a, b*) w.r.t. a
        conv_ab = np.convolve(a, b_star, 'full')
        max_val = conv_ab.max()
        # Top positions (within 1% of max)
        threshold = max_val * 0.995
        top_positions = np.where(conv_ab >= threshold)[0]

        # Subgradient
        grad = np.zeros(n)
        weights = np.zeros(len(top_positions))
        for idx, k in enumerate(top_positions):
            # weight proportional to how close to max
            weights[idx] = conv_ab[k]
        weights = weights / weights.sum()

        for idx, k in enumerate(top_positions):
            w = weights[idx]
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            idx_range = np.arange(js, je + 1)
            grad[idx_range] += w * b_star[k - idx_range]

        # Adaptive eta
        progress = min(1.0, t / max(1, total_est))
        eta_t = eta_schedule_start * (eta_schedule_end / eta_schedule_start) ** progress

        z = z + eta_t * grad

        # FTRL update (mirror descent with entropy)
        z_shift = z - z.max()
        exp_neg = np.exp(-z_shift)
        a_new = exp_neg / exp_neg.sum()
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a = a_new

        # EMAs
        ema_fast = 0.1 * a + 0.9 * ema_fast
        ema_slow = 0.02 * a + 0.98 * ema_slow

        # Check
        if t % 15 == 0 or t <= 10:
            best_local = best_score
            best_local_a = None
            for cand in [a, ema_fast, ema_slow]:
                score = 2.0 * n * fast_autocorr(cand).max()
                if score < best_local:
                    best_local = score
                    best_local_a = cand.copy()
            if best_local_a is not None:
                best_score = best_local
                best_a = best_local_a
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > 250 and t > 50:
            a = best_a.copy()
            z = np.zeros(n)
            ema_fast = a.copy()
            ema_slow = a.copy()
            stall_count = 0

        if t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=0.2, n_stages=30, k_use=None):
    """
    LSE (full or top-k) with Adam optimizer in z-space.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_T = temps[si]

        base_lr = 4e-3 * min(1.0, 120.0 / n)
        lr = base_lr * (0.82 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(10, int(remaining * 3))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = lse_obj_grad_z(z, stage_T, k_use)

            t_adam += 1
            m_z = beta1 * m_z + (1.0 - beta1) * grad_z
            v_z = beta2 * v_z + (1.0 - beta2) * grad_z ** 2
            m_hat = m_z / (1.0 - beta1 ** t_adam)
            v_hat = v_z / (1.0 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 6 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 30 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_flattening(a0, time_limit, seed=42):
    """
    Aggressive peak flattening with multi-source transfer.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()

        top_k = min(25, len(b))
        top_idx = np.argsort(b)[::-1][:top_k]

        improved = False
        for k in top_idx:
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-18:
                continue

            src = int(idx[np.argmax(contrib)])

            # Build danger mask
            danger = np.zeros(n, dtype=bool)
            for k2 in top_idx[:12]:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                danger[js2:je2 + 1] = True

            # Find safe destinations
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:min(8, max(3, len(safe_sorted) // 5))]
            if len(dst_pool) == 0:
                continue

            for dst in dst_pool[:4]:
                for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[int(dst)] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Multi-point random perturbation
            n_pert = int(rng.integers(1, 5))
            a_trial = a.copy()
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.04)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 150:
                break

    return best_a, best_score


def dual_guided_descent(a0, time_limit, solver, seed=555):
    """
    Dual-guided multiplicative weight updates.
    Uses LP dual info to get exact subgradient of the value function.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    ema = a.copy()
    stall_count = 0
    eta = 0.3

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val, _ = solve_lp(a, solver)
        if b_star is None:
            continue

        # Compute subgradient: a[i] appears in conv(a,b)[k] as a[i]*b[k-i]
        # The "risk" for position i is: sum over top-k of a[i]*b_star[k-i]
        conv_ab = np.convolve(a, b_star, 'full')
        max_val = conv_ab.max()
        threshold = max_val * 0.99

        # For each position i, compute its "exposure" to the top peaks
        exposure = np.zeros(n)
        top_count = 0
        for k in range(len(conv_ab)):
            if conv_ab[k] >= threshold:
                top_count += 1
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx_range = np.arange(js, je + 1)
                exposure[idx_range] += b_star[k - idx_range]
        
        if top_count > 0:
            exposure /= top_count

        # Multiplicative weight update: reduce high-exposure positions
        # a_new[i] ∝ a[i] * (1 - eta * exposure[i] / max_exposure)
        max_exp = exposure.max()
        if max_exp > 1e-15:
            factor = 1.0 - eta * exposure / max_exp
            factor = np.maximum(factor, 0.01)  # prevent zeroing out
            a_new = a * factor
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
        else:
            a_new = a.copy()

        a = a_new
        ema = 0.05 * a + 0.95 * ema

        # Check
        if t % 10 == 0 or t <= 10:
            best_local = best_score
            best_local_a = None
            for cand in [a, ema]:
                score = 2.0 * n * fast_autocorr(cand).max()
                if score < best_local:
                    best_local = score
                    best_local_a = cand.copy()
            if best_local_a is not None:
                best_score = best_local
                best_a = best_local_a
                stall_count = 0
                # Decay eta on improvement
                eta = max(0.05, eta * 0.98)
            else:
                stall_count += 1

        if stall_count > 200 and t > 50:
            a = best_a.copy()
            ema = a.copy()
            eta = 0.3
            stall_count = 0

        if t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DGD t={t} ({elapsed:.0f}s): best={best_score:.6f}, eta={eta:.3f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 10))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3.0)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1.0 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3.0) ** 2
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1.0 + np.cos(np.pi * x))
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    else:
        a = np.ones(n)
        a = a + 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update_state(best_seq, best_score, a, label, t_global_start):
    """Try to update the best sequence. Returns (best_seq, best_score, improved)."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_seq = a.copy()
        best_score = sc
        elapsed = time.monotonic() - t_global_start
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining:
    1. Enhanced LP Nesterov (zero-sum game with multi-rate EMA)

    print(f"Time budget: {time_budget:.0f}s")

    try:

        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Enhanced LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.38, 580.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_enhanced(best_seq, t1, solver)
        best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, f"lp_n{n_base}", t_global_start)

    # === Phase 2: FTRL variant ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: FTRL n={n_base} ({remaining:.0f}s) ===")
    t2 = min(remaining * 0.14, 180.0)
    if t2 > 30:
        a_out, _ = ftrl_lp_optimize(best_seq, t2, solver)
        best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, f"ftrl_n{n_base}", t_global_start)

    # === Phase 3: Dual-guided descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: Dual-guided n={n_base} ({remaining:.0f}s) ===")
    t3 = min(remaining * 0.14, 180.0)
    if t3 > 30:
        a_out, _ = dual_guided_descent(best_seq, t3, solver)
        best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, f"dgd_n{n_base}", t_global_start)

    # === Phase 4: Full LSE with Adam ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Full LSE n={n_base} ({remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, _ = lse_adam_optimize(best_seq, t4, T_start=50.0, T_end=0.2, n_stages=30, k_use=None)
        best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, f"full_lse_n{n_base}", t_global_start)

    # === Phase 5: Top-k LSE (different params) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Top-k LSE n={n_base} ({remaining:.0f}s) ===")
    t5 = min(remaining * 0.08, 80.0)
    if t5 > 15:
        a_out, _ = lse_adam_optimize(best_seq, t5, T_start=120.0, T_end=0.5, n_stages=20, k_use=25)
        best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, f"topk25_lse_n{n_base}", t_global_start)

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 6: Multi-length ({remaining:.0f}s) ===")
    t6_total = min(remaining * 0.12, 120.0)
    if t6_total > 30:
        for delta in [100, -100, 200]:
            n_try = n_base + delta
            if n_try < 200:
                continue
            t_each = t6_total / 3.0
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            init_try = interpolate_sequence(best_seq, n_try)
            a_out, _ = lse_adam_optimize(init_try, t_each, T_start=40.0, T_end=1.0, n_stages=12, k_use=15)
            best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, f"lse_n{n_try}", t_global_start)

    # === Phase 7: Peak flattening ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 7: Peak flattening ({remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 10:
        a_out, _ = peak_flattening(best_seq, t7, seed=456)
        best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, "peak_flat_1", t_global_start)

    # === Phase 8: Final short LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t8 = min(remaining * 0.18, 100.0)
        if t8 > 20:
            print(f"=== Phase 8: Final LP Nesterov ({remaining:.0f}s) ===")
            a_out, _ = nesterov_lp_enhanced(best_seq, t8, solver)
            best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, "final_lp", t_global_start)

    # === Phase 9: Final peak flatten ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t9 = min(remaining * 0.5, 60.0)
        if t9 > 5:
            print(f"=== Phase 9: Final peak flatten ({remaining:.0f}s) ===")
            a_out, _ = peak_flattening(best_seq, t9, seed=789)
            best_seq, best_score, _ = try_update_state(best_seq, best_score, a_out, "peak_flat_2", t_global_start)

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.array([1.0])

    return [float(x) for x in best_seq]
