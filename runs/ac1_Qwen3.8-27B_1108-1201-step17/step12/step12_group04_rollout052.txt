def fast_autocorr(a):

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_objective_and_grad(z, T):

    LogSumExp-smoothed objective in log-space.
    z: log of the sequence (before softmax normalization)
    Returns (objective_in_convolution_space, gradient_wrt_z)

    # The actual score is 2*n*max(b), but we minimize max(b) since sum(a)=1

def logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=500.0, n_stages=25, 
                       iters_per_stage=50, seed=42):
    """
    Multi-stage LogSumExp optimization.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-15))
    # Make z sum-to-zero for identifiability (softmax is invariant to shifts)
    z = z - z.sum() / n
    
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        stage_T = temps[si]
        
        def func(z_in):
            val, _ = lse_objective_and_grad(z_in, stage_T)
            return val
        
        def jac(z_in):
            _, g = lse_objective_and_grad(z_in, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception as e:
            print(f"    LSE stage {si} error: {e}")
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            a_cur = softmax_fn(z)
            cur_score = 2.0 * n * fast_autocorr(a_cur).max()
            print(f"    LSE stage={si} T={stage_T:.1f} ({elapsed:.0f}s): cur={cur_score:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

def active_set_optimize(a0, time_limit, n_peaks=5, max_iters=500):
    """
    Active-set method: identify top peaks, then minimize max of those peaks
    using projected gradient on the probability simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    lr = 0.1
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        # Get top peaks
        top_idx = np.argsort(b)[::-1][:n_peaks]
        b_top = b[top_idx]
        b_max = b_top.max()
        
        # Subgradient of max of top peaks w.r.t. a
        # The w_j for the LogSumExp-like weighting
        T_active = 100.0
        w = np.exp(T_active * (b_top - b_max))
        w = w / w.sum()
        
        # Gradient: sum_j w_j * db_{top[j]}/da
        grad = np.zeros(n)
        for jj, k in enumerate(top_idx):
            if w[jj] < 1e-10:
                continue
            # db_k/da_i = a[k-i] for valid i (times 2 approximately)
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += w[jj] * 2.0 * a[ki]
                ik = i - k
                if 0 <= ik < n:
                    grad[i] += w[jj] * 2.0 * a[ik]
        
        # Projected gradient step on simplex
        a_new = a - lr * grad
        # Project onto simplex (per algorithm of Duchi et al.)
        a_new = project_simplex(a_new)
        
        if a_new.sum() > 1e-15:
            a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce learning rate
            lr *= 0.95
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS it={it} ({elapsed:.0f}s): score={best_score:.8f}, lr={lr:.6f}")
    
    return best_a, best_score

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1
    rho = np.where(u - cv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Pairwise coordinate descent: try moving mass from one position to another.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    # Precompute which positions contribute to which peaks
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Find positions that contribute most to the max peak(s)
        top_peaks = np.where(np.isclose(b, b_max, atol=b_max * 0.01))[0]
        
        improved = False
        for k in top_peaks[:5]:
            # Positions contributing to peak k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            
            # Find the largest contributor
            max_c_idx = np.argmax(contributions)
            src = idx[max_c_idx]
            
            if a[src] < 1e-10:
                continue
            
            # Try moving small amounts to various destinations
            # Score each destination by its "damage" to all top peaks
            src_val = a[src]
            
            # Use FFT to compute the effect of moving mass
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                # Try moving to the least "dangerous" position
                # Positions at the edges are safest
                candidates = [0, n-1, 1, n-2, 2, n-3]
                rng.shuffle(candidates)
                
                for dst in candidates:
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Random perturbation
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            no_improve = 0
                            break
            
            if no_improve > 500:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)

    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)  # cosine window
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.5) ** 2)  # Gaussian
    else:
        x = np.linspace(0, 1, n)
        a = x * (1 - x)  # parabola
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
