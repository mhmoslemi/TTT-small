"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Multiplicative multi-peak gradient (fast, no LP) - iteratively reduces top convolution peaks
2. Nesterov-accelerated LP best-response (powerful zero-sum game solver) - main workhorse

4. Coordinate descent peak shaving (final polish)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def score_fn(a):
    """Compute 2n * max(a*a) for normalized a."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def multiplicative_multipeak(a, time_limit, top_k=5, eta_start=0.05, eta_end=0.001):
    """
    Multiplicative weights update targeting top-k convolution peaks.
    a_new[j] proportional to a[j] * exp(-eta * sum_{i in top_k} w_i * a[k_i - j])
    This is mirror descent with KL divergence on the simplex.
    """
    n = len(a)
    a = normalize_nonneg(a)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    iters_in_window = 0
    window_start = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_autocorr(a)
        top_idx = np.argsort(b)[::-1][:top_k]
        top_vals = b[top_idx]
        
        # Compute weights (normalized softmax of top values)
        w = np.exp(top_vals - top_vals.max())
        w = w / w.sum()
        
        # Compute gradient: g[j] = 2 * sum_i w_i * a[k_i - j]
        # a[k_i - j] is valid when 0 <= k_i - j < n
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            # g[j] += 2 * w[i] * a[k - j] for 0 <= k-j < n
            # j ranges from max(0, k-n+1) to min(n-1, k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                g[j_start:j_end+1] += 2.0 * w[i] * a[k - j_start:k - j_start + (j_end - j_start + 1)][::-1]
        
        # Multiplicative update
        log_update = -eta * g
        # Clip for numerical stability
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = normalize_nonneg(a_new)
        
        score = score_fn(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        iters_in_window += 1
    
    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """
    Spectral shaping: iteratively flatten the power spectrum to reduce
    autocorrelation peaks.
    """
    n = len(a)
    a = normalize_nonneg(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = score_fn(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = score_fn(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. (a_fixed * b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver):
    """
    Nesterov-accelerated LP best-response iteration for the zero-sum game.
    Time-averaged iterates give the approximate equilibrium.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    # Time averaging (exponentially weighted)
    avg_a = a.copy()
    avg_weight = 1.0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        # Nesterov step size
        alpha_t = 2.0 / (t + 2.0)
        
        # Update y (the "ghost" point for Nesterov)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        # Nesterov extrapolation for a
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s
        
        y = y_new
        a = a_new
        
        # Check current iterate
        score_a = score_fn(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Time-average (uniform)
        avg_a = (avg_weight * avg_a + a) / (avg_weight + 1)
        avg_weight += 1
        
        # Check time-average every 50 iterations
        if t % 50 == 0:
            score_avg = score_fn(avg_a)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving: directly reduce top autocorrelation peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Find destination: low-contribution coordinate
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Pick from lowest-valued candidates
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = score_fn(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = score_fn(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def gradient_descent_subgrad(a0, time_limit, top_k=3, eta0=0.1):
    """
    Projected subgradient descent with top-k peak averaging.
    After each step, project onto the simplex.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        eta = eta0 / np.sqrt(1.0 + 0.01 * it)
        
        b = fast_autocorr(a)
        top_idx = np.argsort(b)[::-1][:top_k]
        
        # Subgradient: average of gradients at top peaks
        g = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # g[j] += 2 * a[k - j]
            for j in range(js, je + 1):
                g[j] += 2.0 * a[k - j]
        g /= top_k
        
        # Gradient step
        a_new = a - eta * g
        # Project onto simplex (a >= 0, sum = 1)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        score = score_fn(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
