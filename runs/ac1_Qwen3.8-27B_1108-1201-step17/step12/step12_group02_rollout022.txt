    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(c)/(sum(a))^2 where c = convolve(a,a)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_self_conv(a)
    return 2.0 * n * c.max() / (s * s)


def subgradient_topk(a, c, k, T):
    """Compute subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    vals = c[top_idx]
    shift = T * (vals - vals.max())
    w = np.exp(shift)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for ii in range(k):
        kpos = int(top_idx[ii])
        wi = w[ii] * 2.0
        # a[kpos - j] for j = 0..min(n-1, kpos)
        j_end = min(n, kpos + 1)
        if j_end > 0:
            grad[:j_end] += wi * a[kpos - np.arange(j_end)]
        # a[j - kpos] for j = max(0,kpos)..n-1
        j_start = max(0, kpos)
        if j_start < n:
            grad[j_start:] += wi * a[j_start - kpos:]
    
    return grad


def annealed_subgrad(a0, time_limit, seed=42, T0=300.0, T1=5.0,
                     k_min=3, k_max=12, lr0=0.04, momentum=0.8):
    """
    Subgradient descent with annealed temperature and adaptive k.
    Uses momentum for faster convergence.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    vel = np.zeros(n)  # Momentum
    stall = 0
    
    total_budget = 50000  # Max iterations
    
    while time.monotonic() - t0 < time_limit:
        # Annealing parameters
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T0 * (T1 / T0) ** frac
        k = int(k_min + (k_max - k_min) * min(1.0, (time.monotonic() - t0) / (time_limit * 0.3)))
        lr = lr0 / (1.0 + (time.monotonic() - t0) / (time_limit * 0.5)) ** 0.6
        
        c = fast_self_conv(a)
        c_max = c.max()
        
        # Subgradient
        grad = subgradient_topk(a, c, k, T)
        
        # Momentum update
        vel = momentum * vel + (1 - momentum) * grad
        a = a * np.exp(-lr * vel)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        # Check score
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Stall handling
        if stall > 200:
            noise = 0.03 * rng.standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            vel = np.zeros(n)
            stall = 0
    
    return best_a, best_score


def l2_flatten(a0, time_limit, seed=42):
    """
    Minimize L2 variance of self-convolution (proxy for min-max).
    Uses gradient of sum(c^2) = sum_k (sum_i a[i]*a[k-i])^2
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    vel = np.zeros(n)
    lr = 0.02
    
    while time.monotonic() - t0 < time_limit:
        c = fast_self_conv(a)
        
        # Gradient of sum(c^2) with respect to a
        # d(sum c^2)/da[j] = 2 * sum_k c[k] * dc[k]/da[j]
        # = 2 * sum_k c[k] * (a[k-j] + a[j-k])
        # = 2 * (c * a)[j] + 2 * (c_rev * a)[j]  -- can use FFT
        
        # For efficiency, top-k approach:
        # Weight by c[k] itself (richer peaks get more gradient)
        top_k = min(20, n)
        top_idx = np.argsort(c)[::-1][:top_k]
        vals = c[top_idx]
        w = vals / vals.sum() if vals.sum() > 0 else np.ones(top_k) / top_k
        
        grad = np.zeros(n)
        for ii in range(top_k):
            kpos = int(top_idx[ii])
            wi = w[ii] * 2.0
            j_end = min(n, kpos + 1)
            if j_end > 0:
                grad[:j_end] += wi * a[kpos - np.arange(j_end)]
            j_start = max(0, kpos)
            if j_start < n:
                grad[j_start:] += wi * a[j_start - kpos:]
        
        # Gradient step with momentum
        vel = 0.9 * vel + 0.1 * grad
        a = a * np.exp(-lr * vel)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        # Decay learning rate
        lr *= 0.9995
        
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def peak_shave(a0, time_limit, seed=123):
    """Aggressive peak shaving with multiple transfer attempts."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_self_conv(a)
        top_k = np.argsort(c)[::-1][:8]
        
        improved = False
        for k_idx in top_k:
            k_idx = int(k_idx)
            # Find contributing positions
            j_start = max(0, k_idx - n + 1)
            j_end = min(n, k_idx + 1)
            js = np.arange(j_start, j_end)
            contrib = a[js] * a[k_idx - js]
            
            if contrib.sum() < 1e-15:
                continue
            
            # Source: highest contributor
            src = int(js[np.argmax(contrib)])
            
            # Destination: random positions not contributing to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n, k2 + 1)
                peak_mask[js2:je2] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Try multiple destinations and amounts
            for _ in range(5):
                dst = int(candidates[rng.integers(len(candidates))])
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if rng.integers(4) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=0):
    """Generate diverse structured initializations."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Double bump with specific separation
        x = np.linspace(0, 1, n)
        sep = 0.3
        a = np.exp(-50 * (x - (0.5 - sep)) ** 2) + np.exp(-50 * (x - (0.5 + sep)) ** 2) + 0.05
    elif variant == 11:
        # Triple bump
        x = np.linspace(0, 1, n)
        a = (np.exp(-40 * (x - 0.2) ** 2) + np.exp(-40 * (x - 0.5) ** 2) + 
             np.exp(-40 * (x - 0.8) ** 2) + 0.05)
    else:
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.1, n * 0.4)
            amp = rng.uniform(-0.5, 1.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lse_lbfgs(a0, time_limit, T_start=100.0, T_end=1.0, n_stages=10, seed=42):
    """LogSumExp-smoothed optimization with L-BFGS."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        iters = max(10, min(50, int(20 + 30 * si / n_stages)))
        
        def func_grad(z_in):
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a = e / e.sum()
            c = fast_self_conv(a)
            Tc = stage_T * c
            Tc_shift = Tc - Tc.max()
            exp_Tc = np.exp(Tc_shift)
            obj = (Tc.max() + np.log(exp_Tc.sum())) / stage_T
            
            w = exp_Tc / exp_Tc.sum()
            # grad_a[j] = 2 * sum_k w[k] * (a[k-j] + a[j-k])
            grad_a = np.zeros(n)
            top_k = min(30, 2 * n - 1)
            top_idx = np.argsort(c)[::-1][:top_k]
            for ii in range(top_k):
                kpos = int(top_idx[ii])
                wi = 2.0 * w[kpos]
                j_end = min(n, kpos + 1)
                if j_end > 0:
                    grad_a[:j_end] += wi * a[kpos - np.arange(j_end)]
                j_start = max(0, kpos)
                if j_start < n:
                    grad_a[j_start:] += wi * a[j_start - kpos:]
            
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            return obj, grad_z
        
        def func(z_in):
            val, _ = func_grad(z_in)
            return val
        
        def jac(z_in):
            _, g = func_grad(z_in)
            return g
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-16,
                                    'gtol': 1e-10, 'maxfun': iters * 100,
                                    'disp': False})
            z = res.x
            a = np.exp(z - z.max())
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_score
