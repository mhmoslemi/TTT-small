    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def full_lse_obj_grad_z(z, T, topk=None):
    """
    Full (or top-k) LSE objective and gradient in z-space.
    obj = (1/T)*log(sum_{k in K} exp(T*c_k)) where c = autocorr(a)
    a = softmax(z), normalized so sum(a) = 1

    a = softmax_fn(z)
    n = len(a)
    c = fast_autocorr(a)
    
    if topk is not None and topk < len(c):
        top_idx = np.argsort(c)[::-1][:topk]
        top_vals = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        top_vals = c
    
    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)

            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    S = np.dot(grad_a, a)

    return lse_val, grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated full LSE gradient descent with Adam + Nesterov momentum
2. LP Nesterov (zero-sum game) for diverse solutions

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n <= 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_grad_fft(a, T, topk=None):
    """
    Compute LSE objective and gradient (in a-space) using FFT.
    a is normalized (sum=1, a>=0).
    Returns (lse_value, grad_a) where grad_a is the gradient of LSE wrt a.
    """
    n = len(a)
    c = fast_autocorr(a)
    m = len(c)  # 2n-1
    
    if topk is not None and topk < m:
        top_idx = np.argpartition(c, -topk)[-topk:]
        top_vals = c[top_idx]
    else:
        top_idx = np.arange(m)
        top_vals = c
    
    Tc = T * top_vals
    max_Tc = np.max(Tc)
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = np.sum(exp_Tc)
    lse_val = (max_Tc + np.log(sum_exp)) / T
    w = exp_Tc / sum_exp
    
    # Build w_at of length m (2n-1)
    w_at = np.zeros(m)
    w_at[top_idx] = w
    
    # grad_a[i] = 2 * sum_k w_at[k] * a[k-i] (for valid k-i in [0, n-1])
    # This is 2 * cross_correlation(w_at, a) [0:n]
    # cross_corr(x, y)[i] = sum_j x[j] * y[j-i] (with bounds)
    # = IFFT(FFT(x) * conj(FFT(y)))[i] (with sufficient zero-padding)
    
    fft_size = 1 << (2 * m - 1).bit_length()  # Ensure no circular aliasing
    # Actually, we need fft_size >= m + n - 1 for the cross-correlation
    # m = 2n-1, so we need fft_size >= 3n - 2
    fft_size = 1 << (3 * n).bit_length()
    
    W = rfft(w_at, fft_size)
    A = rfft(a, fft_size)
    # cross_correlation(w_at, a)[i] = sum_j w_at[j] * a[j-i]
    # This is IFFT(conj(FFT(w_at)) * FFT(a))[i] ... let me verify
    # Actually: IFFT(FFT(x) * conj(FFT(y)))[n] = sum_k x[k] * y[(k-n) mod N]
    # For linear (non-circular): if N >= len(x) + len(y) - 1, this gives cross-correlation
    
    # Let me use a different formulation:
    # conv(w_at_reversed, a)[i + m - 1] = sum_j w_at[m-1-j] * a[i-j] ... this is getting confusing
    
    # Simplest: use the relation grad_a = 2 * (w_at conv a) appropriately indexed
    # Actually, let me just verify: 
    # sum_k w_at[k] * a[k-i] for 0<=k-i<n
    # = sum_{j=0}^{n-1} w_at[i+j] * a[j] (substituting j=k-i)
    # This is the cross-correlation of w_at and a at lag i = (a cross w_at)[i]
    # = IFFT(conj(FFT(a)) * FFT(w_at))[i]
    # = IFFT(FFT(w_at) * conj(FFT(a)))[i]
    
    # Let me just use: result = IFFT(FFT(w_at) * conj(FFT(a)))
    # And check that result[i] = sum_j w_at[(i+j) mod N] * a[j]
    # For N large enough (>= m + n - 1 = 3n-2), this equals sum_j w_at[i+j] * a[j] (no wrap-around)
    
    cross = irfft(W * np.conj(A), fft_size)[0:n]
    grad_a = 2.0 * cross
    
    return lse_val, grad_a


def lse_grad_z(z, T, topk=None):
    """LSE objective and gradient in z-space (where a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = lse_grad_fft(a, T, topk)
    # Convert to z-space: grad_z = a * (grad_a - a^T grad_a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_rhs(a_fixed):
    """For LP: conv(a_fixed, b) = T @ b where T is (2n-1) x n Toeplitz-like."""
    n = len(a_fixed)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # So conv(a_fixed, b)[k] = sum_j a_fixed[k-j] * b[j] (for valid k-j)
    # = sum_j T[k,j] * b[j] where T[k,j] = a_fixed[k-j] if 0<=k-j<n else 0
    # This is a Toeplitz matrix (but transposed from the usual convention)
    # T has (2n-1) rows and n columns
    # T[k, j] = a_fixed[k-j] for max(0, k-n+1) <= j <= min(n-1, k)
    
    # For efficiency, we can represent this implicitly or build it as a sparse matrix
    # For n up to ~2000, a dense (2n-1) x n matrix has ~2n^2 entries, which is fine
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_rhs(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg = np.zeros(n)
    count = 0
    stall = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ys
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        avg += a
        count += 1
        
        # Check various candidates
        for candidate in [a, y]:
            sc = 2.0 * n * fast_autocorr(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                stall = 0
        if count > 50 and count % 50 == 0:
            avg_c = avg / count
            sc = 2.0 * n * fast_autocorr(avg_c).max()
            if sc < best_score:
                best_score = sc
                best_a = avg_c.copy()
                stall = 0
                # Reset average
                avg = np.zeros(n)
                count = 0
        
        # Run stall counter
        if count % 100 == 0:
            sc = 2.0 * n * fast_autocorr(avg / count if count > 0 else a).max()
            if sc >= best_score:
                stall += 1
        if stall > 100 and t > 500:
            y = best_a.copy()
            a = best_a.copy()
            avg = np.zeros(n)
            count = 0
            stall = 0
        
        if t % 5000 == 0:
            print(f"    NLP t={t}: best={best_score:.6f}")
    
    return best_a, best_score


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=30, 
                      topk=None, lr_scale=5e-3, nesterov_beta=0.9):
    """
    Adam + Nesterov momentum optimization of full/top-k LSE in z-space.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    base_lr = lr_scale * min(1.0, 100.0 / n)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        lr = base_lr * (0.85 ** si)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        
        # Nesterov state
        v_nest = np.zeros(n)
        
        # Adaptive T within stage (slowly decrease)
        T_cur = stage_T
        
        max_it = 50000
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            # Compute gradient
            obj, grad_z = lse_grad_z(z, T_cur, topk)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            
            # Adam step
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Nesterov acceleration
            v_nest = nesterov_beta * v_nest - step
            z = z + v_nest
            
            # Adaptive T: decrease within stage
            if (it + 1) % 100 == 0:
                T_cur *= 0.95
                T_cur = max(T_cur, stage_T * 0.3)
            
            # Periodic evaluation
            if (it + 1) % 20 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        # End of stage: update best
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        if si % 5 == 0:
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.1f}): best={best_score:.6f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated full LSE gradient descent with Adam + Nesterov momentum
2. LP Nesterov (zero-sum game) for diverse exploration
3. Coarse-to-fine multi-resolution optimization
4. Adaptive temperature with peak-concentration feedback
5. Final coordinate-level peak flattening

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n <= 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_grad_fft(a, T, topk=None):
    """
    Compute LSE objective and gradient in a-space using FFT.
    a is normalized (sum=1, a>=0).
    Returns (lse_value, grad_a).
    """
    n = len(a)
    c = fast_autocorr(a)
    m = len(c)  # 2n-1

    if topk is not None and topk < m:
        top_idx = np.argpartition(c, -topk)[-topk:]
        top_vals = c[top_idx]
    else:
        top_idx = np.arange(m)
        top_vals = c

    Tc = T * top_vals
    max_Tc = np.max(Tc)
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = np.sum(exp_Tc)
    lse_val = (max_Tc + np.log(sum_exp)) / T
    w = exp_Tc / sum_exp

    w_at = np.zeros(m)
    w_at[top_idx] = w

    # grad_a[i] = 2 * sum_{k} w_at[k] * a[k-i] for valid k-i in [0, n-1]
    # = 2 * IFFT(rfft(w_at) * conj(rfft(a)))[0:n] (with sufficient zero-padding)
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(w_at, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[0:n]
    grad_a = 2.0 * cross

    return lse_val, grad_a


def lse_grad_z(z, T, topk=None):
    """LSE objective and gradient in z-space (a = softmax(z))."""
    a = softmax_fn(z)
    obj, grad_a = lse_grad_fft(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK, cp.CBC, cp.SCIP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build matrix T where T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver):
    """Nesterov-accelerated LP best-response with adaptive restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg = np.zeros(n)
    count = 0
    stall = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ys

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new
        avg += a
        count += 1

        # Check candidates
        for candidate in [a, y]:
            sc = 2.0 * n * fast_autocorr(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                stall = 0

        if count > 100 and count % 50 == 0:
            avg_c = avg / count
            sc = 2.0 * n * fast_autocorr(avg_c).max()
            if sc < best_score:
                best_score = sc
                best_a = avg_c.copy()
                stall = 0
                avg = np.zeros(n)
                count = 0

        if count % 200 == 0:
            sc = 2.0 * n * fast_autocorr(avg / max(count, 1)).max()
            if sc >= best_score:
                stall += 1

        if stall > 80 and t > 500:
            y = best_a.copy()
            a = best_a.copy()
            avg = np.zeros(n)
            count = 0
            stall = 0

        if t % 5000 == 0:
            print(f"    NLP t={t}: best={best_score:.6f}")

    return best_a, best_score


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=35,
                      topk=None, lr_scale=8e-3, nest_beta=0.9, restart_every=2000):
    """
    Adam + Nesterov momentum optimization of LSE in z-space.
    Uses adaptive T within each stage.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-25))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    base_lr = lr_scale * min(1.0, 120.0 / n)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break

        lr = base_lr * (0.88 ** si)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        v_nest = np.zeros(n)
        T_cur = float(stage_T)

        # Use top-k for very low T stages to focus gradient
        stage_topk = topk
        if stage_T < 1.0 and topk is None:
            stage_topk = max(5, n // 40)

        max_it = 100000
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            obj, grad_z = lse_grad_z(z, T_cur, stage_topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            v_nest = nest_beta * v_nest - step
            z = z + v_nest
            z = np.clip(z, -50, 50)

            # Adaptive T: slow exponential decay within stage
            if (it + 1) % 50 == 0:
                T_cur *= 0.97
                T_cur = max(T_cur, stage_T * 0.2)

            # Periodic evaluation and restart
            if (it + 1) % 30 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

            # Restart from best periodically to prevent drift
            if restart_every is not None and (it + 1) % restart_every == 0:
                z = np.log(np.maximum(best_a, 1e-25))
                v_nest = np.zeros(n)
                m_z = np.zeros(n)
                v_z = np.zeros(n)
                t_adam = 0

        # End of stage
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_flatten(a0, time_limit, seed=42):
    """
    Direct peak-flattening: iteratively moves mass from high-contributing
    positions to low-impact positions to reduce the max peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        bmax = c.max()
        # Find near-maximal peaks
        thresh = bmax * 0.998
        top_peaks = np.where(c >= thresh)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:5]
        top_peaks = top_peaks[:15]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destination: low a[j] and not contributing to top peaks
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.array([j for j in range(n) if j != src])

            # Prefer low a[j]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(2, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            # Try several step sizes
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15 or a[src] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 200:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    elif choice == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = np.abs(np.sin(3 * x) * np.cos(7 * x)) + 0.1
    else:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining gradient-based and LP approaches:
    Phase 1: Fast Adam LSE refinement of parent (quick local improvements)
    Phase 2: LP Nesterov (global zero-sum game search)
    Phase 3: Adam LSE at longer length (more freedom)
    Phase 4: Fine-grained LSE with very low T (precision)
    Phase 5: Peak flattening (direct minimax greedy)
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)

    solver = get_lp_solver()
    print(f"Solver: {solver}, Budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Fast Adam LSE refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam LSE n={n_base} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_out, _ = adam_lse_optimize(best_seq, t1, T_start=40.0, T_end=0.3,
                                     n_stages=25, topk=None, lr_scale=8e-3,
                                     nest_beta=0.9, restart_every=3000)
        try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: LP Nesterov n={n_base} ({remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40:
        a_out, _ = nesterov_lp(best_seq, t2, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 3: Adam LSE at longer length ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Adam LSE n={n_long} ({remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = adam_lse_optimize(init_long, t3, T_start=30.0, T_end=0.2,
                                     n_stages=25, topk=None, lr_scale=6e-3,
                                     nest_beta=0.9, restart_every=3000)
        try_update(a_out, f"lse2_n{n_long}")

    # === Phase 4: LP Nesterov at longer length ===
    n_long2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: LP Nesterov n={n_long2} ({remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40:
        init_lp = interpolate_seq(best_seq, n_long2)
        a_out, _ = nesterov_lp(init_lp, t4, solver)
        try_update(a_out, f"lp_n{n_long2}")

    # === Phase 5: Fine-grained LSE (very low T) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Fine LSE n={len(best_seq)} ({remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_out, _ = adam_lse_optimize(best_seq, t5, T_start=5.0, T_end=0.05,
                                     n_stages=20, topk=10, lr_scale=3e-3,
                                     nest_beta=0.85, restart_every=2000)
        try_update(a_out, "fine_lse")

    # === Phase 6: Peak flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 6: Peak Flatten ({remaining:.0f}s) ===")
    t6 = min(remaining * 0.30, 150.0)
    if t6 > 10:
        a_out, _ = peak_flatten(best_seq, t6, seed=777)
        try_update(a_out, "peak_flat_1")

    # === Phase 7: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7 = min(remaining * 0.40, 100.0)
        if t7 > 10:
            print(f"=== Phase 7: Final Polish ({remaining:.0f}s) ===")
            a_out, _ = adam_lse_optimize(best_seq, t7, T_start=3.0, T_end=0.02,
                                         n_stages=15, topk=8, lr_scale=2e-3,
                                         nest_beta=0.8, restart_every=1500)
            try_update(a_out, "final_polish")

    # === Phase 8: Final peak flatten ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t8 = min(remaining * 0.50, 60.0)
        if t8 > 5:
            a_out, _ = peak_flatten(best_seq, t8, seed=999)
            try_update(a_out, "peak_flat_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)}, total_time={time.monotonic()-t_global:.0f}s)")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
