"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Multi-stage LogSumExp optimization with Nesterov acceleration in log-space
2. LP-based best response (Nesterov game)
3. Aggressive randomized coordinate descent with pair swaps
4. Multi-length exploration
5. Final polish
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

# --- Core utilities ---

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_grad(a, T):
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    neg_max = -Tb.max()
    exp_Tb = np.exp(Tb + neg_max)
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1]
    conv_len = (2*n-1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)
    return 2.0 * conv[n-1 : n-1+n]

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def safe_evaluate(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_nesterov_logspace(a0, time_limit, T_start=100.0, T_end=0.5, n_stages=15):
    """Nesterov-accelerated mirror descent on LSE objective in log-space."""
    n = len(a0)
    a = normalize(a0)
    z = np.log(a + 1e-30)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        stage_time = remaining / (n_stages - si)
        
        # Step size heuristic
        eta = 1.0 / (T * 3.0)
        
        z_prev = z.copy()
        z_curr = z.copy()
        iters = 0
        
        stage_t0 = time.monotonic()
        while time.monotonic() - stage_t0 < stage_time:
            if time.monotonic() - t0 > time_limit:
                break
            iters += 1
            a_curr = softmax_fn(z_curr)
            grad = lse_grad(a_curr, T)
            
            if iters <= 2:
                z_new = z_curr - eta * grad
            else:
                momentum = (iters - 2.0) / (iters + 1.0)
                z_new = z_curr - eta * grad + momentum * (z_curr - z_prev)
            
            z_prev = z_curr
            z_curr = z_new
            a = softmax_fn(z_curr)
            
            if iters % 500 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}, iters={iters}")
    
    return best_a, best_score

def lse_lbfgs(a0, time_limit, T_start=100.0, T_end=0.5, n_stages=12):
    """L-BFGS on LSE objective with softmax parameterization."""
    from scipy.optimize import minimize
    
    n = len(a0)
    a = normalize(a0)
    z = np.log(a + 1e-30)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_in):
            a_in = softmax_fn(z_in)
            b = fast_autocorr(a_in)
            Tb = T * b
            return (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum())) / T
        
        def jac(z_in):
            a_in = softmax_fn(z_in)
            g = lse_grad(a_in, T)
            S = np.dot(g, a_in)
            return a_in * (g - S)
        
        iters = max(15, min(80, int(20 + 60 * (si / max(1, n_stages - 1)))))
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': iters * 200, 'disp': False})
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
        
        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-LBFGS stage {si}/{n_stages} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=42):
    """Aggressive randomized coordinate descent with pair swaps."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        # Find peaks
        peaks = np.where(b > 0.9 * b_max)[0]
        
        improved = False
        n_trials = 20
        for trial in range(n_trials):
            # Pick a source from a peak-contributing position
            if len(peaks) > 0:
                k = peaks[rng.integers(len(peaks))]
            else:
                k = rng.integers(2 * n - 1)
            
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Pick source as top contributor
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            if a[src] < 1e-12:
                continue
            
            # Pick destination: random position that's not in the top peaks
            dst = rng.integers(n)
            if dst == src:
                continue
            
            # Try different swap fractions
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
        
        if stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0"""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def nesterov_game(a0, time_limit, solver, max_iters=5000):
    """Nesterov-accelerated LP best response (zero-sum game)."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star = lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss
        
        if t <= 2:
            a_new = y_new
        else:
            momentum = (t - 2.0) / (t + 1.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        
        y = y_new
        a_prev = a
        a = a_new
        
        if t % 50 == 0:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            
            elapsed = time.monotonic() - t0
            print(f"    GAME t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
