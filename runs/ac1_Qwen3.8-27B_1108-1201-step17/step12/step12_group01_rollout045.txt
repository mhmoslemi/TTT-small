import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_grad_fft(a, T, topk=None):
    """
    Compute LSE objective and gradient using FFT for efficiency.
    If topk is None, use full LSE over all 2n-1 coefficients.
    If topk is specified, use top-k LSE.
    Returns (obj, grad_a) where grad_a is d(obj)/d(a).
    """
    n = len(a)
    c = fast_autocorr(a)
    L = len(c)  # 2n-1
    
    if topk is not None and topk < L:
        top_idx = np.argsort(c)[::-1][:topk]
        c_use = c[top_idx]
        # We'll compute gradient only for these peaks (exact top-k LSE gradient)
        Tc = T * c_use
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()
        obj = (max_Tc + np.log(exp_Tc.sum())) / T
        
        # Gradient: grad_a[i] = (2/T) * sum_k w_k * a[k-i] for valid k-i
        grad_a = np.zeros(n)
        for idx in range(topk):
            K = top_idx[idx]
            wK = w[idx]
            js = max(0, K - (n - 1))
            je = min(n - 1, K)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    else:
        # Full LSE
        Tc = T * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()
        obj = (max_Tc + np.log(exp_Tc.sum())) / T
        
        # FFT-based gradient computation
        # grad_a[i] = (2/T) * sum_{m=0}^{n-1} w[i+m] * a[m]
        # This is a cross-correlation that can be computed via FFT
        L_fft = 1 << (L + n - 1 - 1).bit_length()
        # W is w (length L=2n-1), A is a (length n)
        # We want: D[i] = sum_{m=0}^{n-1} a[m] * w[i+m] for i=0..n-1
        # This is (a cross-correlate w)[i]
        # Using FFT: IFFT(conj(FFT(a)) * FFT(w)) gives the circular cross-correlation
        A_fft = rfft(a, L_fft)
        W_fft = rfft(w, L_fft)
        D = irfft(np.conj(A_fft) * W_fft, L_fft)
        grad_a = (2.0 / T) * D[:n]
    
    return obj, grad_a


def full_lse_grad_z(z, T, topk=None):
    """Objective and gradient in z-space (softmax parameterization)."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = lse_grad_fft(a, T, topk=topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: max sum(b) s.t. conv(a_fixed, b) <= M, b>=0, sum(b)=1."""
    import cvxpy as cp
    n = len(a_fixed)
    c = fast_autocorr(a_fixed)
    M = c.max()
    
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    solvers_to_try = []
    if solver:
        solvers_to_try.append(solver)
    solvers_to_try.extend([cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK])
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None
    b_val = b_val / s_val
    return b_val


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

        if n_long % 20 == 0 and n_long >= 40:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

                best_a = avg_s.copy()

            avg_short = np.zeros(n)

            short_buf = [None] * WINDOW_SHORT
            short_idx = 0
            stall_count = 0

            print(f"      NLP t={t} ({elapsed:.0f}s, it_time={iter_time_avg:.3f}s): best={best_score:.6f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42):
    """
    Coordinate descent: iteratively move mass from peak-contributing
    positions to low-impact positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        c = fast_autocorr(a)
        cmax = c.max()
        
        # Find top peaks
        top_k = min(10, len(c))
        top_peaks = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for k in top_peaks:
            # Find the pair (i, k-i) that contributes most to c[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            # Compute danger zone
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True
            
            # Find safe destinations
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            
            # Prefer low-value safe positions
            safe_sorted = safe[np.argsort(a[safe])][:20]
            if len(safe_sorted) == 0:
                continue
            
            # Try moving mass from src to each safe destination
            best_frac = 0
            best_score_try = best_score
            best_dst = None
            
            for dst in safe_sorted[:10]:
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score_try:
                        best_score_try = score
                        best_frac = frac
                        best_dst = dst
            
            if best_dst is not None and best_score_try < best_score - 1e-14:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = a / a.sum()
                best_score = best_score_try
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                a_trial = a.copy()
                for _ in range(rng.integers(1, 5)):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 5)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 3
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.8 + 0.15 * np.cos(2 * x) + 0.05 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.3 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 2) ** 2
    elif choice == 7:
        # Random with smooth structure
        a = np.abs(np.random.normal(0, 1, n))
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif choice == 8:
        # Exponential with oscillation
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x) * (1 + 0.3 * np.cos(20 * np.pi * x))
    else:
        # Symmetric bell
        x = np.linspace(-1, 1, n)
        a = (1 - x**2)**2
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence for multi-start."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn
