from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid multi-strategy approach:
1. Pressure-based coordinate descent (fast O(n) moves guided by convolution peaks)
2. Mirror descent with soft multi-peak subgradient (FFT-accelerated)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)


def normalize(a):
    """Normalize a to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:

def cross_corr_weights(w, a):
    """
    Compute: result[j] = sum_k w[k] * a[k-j] for valid indices.
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def soft_subgradient(a, b, beta=50.0):
    """
    Compute a soft subgradient: weight each peak by exp(beta*(b[k]-b_max)),
    then compute the aggregate pressure on each position.
    Returns gradient vector of length n.
    """
    n = len(a)
    b_max = b.max()
    if b_max < 1e-15:
        return np.zeros(n)
    # Soft weights: how much each peak "matters"
    shift = b - b_max
    w = np.exp(beta * shift)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    # Aggregate pressure on each position
    grad = cross_corr_weights(2.0 * w, a)
    return grad


def mirror_descent_fast(a0, time_limit, eta0=0.04, beta0=30.0, beta_max=200.0,
                        decay=0.5, seed=42):
    """
    Mirror descent with temperature-annealed soft subgradient.
    Beta increases over time (from broad subgradient to sharp peak-focusing).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    stall_threshold = 200

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_conv(a)
        # Anneal beta: start broad, end sharp
        progress = min(1.0, step / 3000.0)
        beta = beta0 + (beta_max - beta0) * progress
        eta = eta0 / (1.0 + step / 500.0) ** decay

        grad = soft_subgradient(a, b, beta=beta)
        # Center the gradient (for simplex constraint)
        S = np.dot(grad, a)
        grad_centered = grad - S

        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / s_new
        a = a_new

        # Check score periodically
        if step < 100:
            check_every = 2
        elif step < 1000:
            check_every = 5
        else:
            check_every = 10

        if step % check_every == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > stall_threshold:
            # Perturb to escape
            noise = 0.02 * rng.standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            stall_threshold = min(stall_threshold + 100, 800)

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42, verbose=False):
    """
    Pressure-based coordinate descent.
    For each peak, compute pressure on each position and move mass
    from high-pressure to low-pressure positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    stall_limit = 500

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        b_max = b.max()

        # Find top peaks
        n_peaks = min(8, len(b))
        peak_indices = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for pk in peak_indices:
            pk = int(pk)
            # Pressure on position i: a[pk - i] for valid i
            # (pk - i) must be in [0, n-1], so i in [pk-(n-1), pk]
            i_min = max(0, pk - n + 1)
            i_max = min(n - 1, pk)
            idx = np.arange(i_min, i_max + 1)
            pressure = a[pk - idx]  # pressure[i] = a[pk - i]

            if pressure.max() < 1e-12:
                continue

            # Source: highest pressure (and has mass)
            src_local = int(np.argmax(pressure * a[idx]))
            src = int(idx[src_local])
            if a[src] < 1e-10:
                continue

            # Target: lowest pressure, prefer positions with low a value
            # Also prefer positions not contributing to other top peaks
            low_mask = pressure < np.median(pressure)
            candidates_local = np.where(low_mask)[0]
            if len(candidates_local) == 0:
                candidates_local = np.where(pressure == pressure.min())[0]
            # Among candidates, pick one with lowest a value
            cand_a = a[idx[candidates_local]]
            dst_local = int(candidates_local[np.argmin(cand_a)])
            dst = int(idx[dst_local])
            if dst == src:
                continue

            # Try several step sizes
            for frac in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                if a_trial[src] < 0:
                    continue
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Try random 2D moves
            if rng.integers(5) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    for frac in [0.002, 0.005, 0.01, 0.03]:
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        if a_trial[i] < 0:
                            break
                        a_trial[j] += amt
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                            break

            if stall > stall_limit:
                # Larger perturbation
                noise_scale = min(0.05, 0.005 * (stall / 100.0))
                a = a + noise_scale * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                stall_limit = min(stall_limit + 200, 2000)

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.6f}, stall={stall}")

    return best_a, best_score


def lse_smoothed_optimize(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=15, seed=42):
    """
    LSE-smoothed objective optimization with L-BFGS and multi-stage temperature.
    """
    n = len(a0)
    a_init = normalize(np.asarray(a0, dtype=np.float64))
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = fast_score(a_init)
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(10, min(60, int(20 + 30 * si / n_stages)))

        def lse_func(z_in):
            z_s = z_in - z_in.max()
            e = np.exp(z_s)
            a = e / e.sum()
            b = fast_conv(a)
            Tb = T * b
            Tb_s = Tb - Tb.max()
            exp_Tb = np.exp(Tb_s)
            obj = (Tb.max() + np.log(exp_Tb.sum())) / T
            w = exp_Tb / exp_Tb.sum()
            cc = cross_corr_weights(2.0 * w, a)
            S = np.dot(cc, a)
            grad = a * (cc - S)
            return obj, grad

        try:
            def func_wrapper(z_in):
                val, _ = lse_func(z_in)
                return val

            def jac_wrapper(z_in):
                _, g = lse_func(z_in)
                return g

            res = minimize(fun=func_wrapper, x0=z, jac=jac_wrapper,
                           method='L-BFGS-B',
                           options={'maxiter': iters, 'maxfun': iters * 500,
                                    'ftol': 1e-18, 'gtol': 1e-12, 'disp': False})
            z = res.x
            z_s = z - z.max()
            a = np.exp(z_s)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lp_project_step(f, M_max_constr=100, time_limit=5.0):
    """
    Single LP projection step: find g >= 0, sum(g)=1, (f*g)[k] <= M for top-K constraints.
    Returns rescaled g.
    """
    n = len(f)
    b = fast_conv(f)
    M = b.max()
    # Use top K constraints for speed
    K = min(M_max_constr, len(b))
    top_k = np.argsort(b)[::-1][:K]

    try:
        g = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(g) == 1.0]

        # Build constraint rows for top K lags
        for k in top_k:
            k = int(k)
            # (f*g)[k] = sum_i f[k-i] * g[i]
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            coeffs = np.zeros(n)
            coeffs[i_min:i_max + 1] = f[k - np.arange(i_min, i_max + 1)]
            constraints.append(coeffs @ g <= M)

        prob = cp.Problem(cp.Minimize(cp.sum_squares(g)), constraints)
        start = time.monotonic()
        prob.solve(solver=cp.CLARABEL)
        if time.monotonic() - start > time_limit:
            return None

        if g.value is None:
            return None
        res = np.maximum(g.value, 0)
        s = res.sum()
        if s < 1e-10:
            return None
        return res / s
    except Exception:
        return None


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 8))
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = 1 + 0.5 * np.cos(2 * np.pi * x)
    elif variant == 1:
        a = x ** 2 + (1 - x) ** 2 + 0.05
    elif variant == 2:
        a = 1 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x)
    elif variant == 3:
        a = np.abs(x - 0.5) + 0.1
    elif variant == 4:
        a = np.sinc(2 * x - 1) ** 2 + 0.01
    elif variant == 5:
        a = 1.0 + 0.4 * np.sin(np.pi * x) ** 2
    elif variant == 6:
        # Random Gaussian bumps
        a = np.ones(n) * 0.5
        for _ in range(4):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.05, 0.3)
            amp = rng.uniform(-0.5, 0.8)
            a += amp * np.exp(-((x - c) / w) ** 2)
    else:
        a = np.ones(n) * 0.3 + 0.7 * (x < 0.5).astype(float)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function: multi-phase hybrid optimization.
    Phase 1: Mirror descent for global shaping
    Phase 2: Coordinate descent for precise peak shaving
    Phase 3: LSE refinement for smooth landscape navigation
    Phase 4: LP projection for basin jumps
    Phase 5: Intensive final polishing (alternating CD + MD)
    """
    t_global = time.monotonic()
    time_budget = float(budget_s) - 20.0  # Safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = fast_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return False
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Default initialization if no parent
    if best_seq is None:
        n_init = 1600
        best_seq = diverse_init(n_init, rng, variant=1)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n = len(best_seq)
    print(f"Starting optimization at n={n}, score={best_score:.6f}")

    # === Phase 1: Mirror Descent (global shaping) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_md = min(remaining * 0.15, 250.0)
    if t_md > 30:
        print(f"\n=== Phase 1: Mirror Descent (budget={t_md:.0f}s) ===")
        a_out, _ = mirror_descent_fast(best_seq, t_md, eta0=0.04, beta0=20.0,
                                        beta_max=150.0, decay=0.5, seed=1)
        try_update(a_out, "MD_global")
        # Also try from a diverse init
        a_div = diverse_init(n, rng, variant=5)
        a_out2, _ = mirror_descent_fast(a_div, t_md / 3, eta0=0.06, beta0=15.0,
                                         beta_max=100.0, decay=0.6, seed=77)
        try_update(a_out2, "MD_diverse")

    # === Phase 2: Coordinate Descent (precise peak shaving) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_cd = min(remaining * 0.30, 500.0)
    if t_cd > 30:
        print(f"\n=== Phase 2: Coordinate Descent (budget={t_cd:.0f}s) ===")
        a_out, _ = coordinate_descent(best_seq, t_cd, seed=13, verbose=True)
        try_update(a_out, "CD_main")

    # === Phase 3: LSE Refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse = min(remaining * 0.20, 300.0)
    if t_lse > 30:
        print(f"\n=== Phase 3: LSE Refinement (budget={t_lse:.0f}s) ===")
        a_out, _ = lse_smoothed_optimize(best_seq, t_lse, T_start=40.0, T_end=0.2,
                                          n_stages=15, seed=11)
        try_update(a_out, "LSE_refine")

    # === Phase 4: LP Projection (basin jumps) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_lp = min(remaining * 0.10, 120.0)
    if t_lp > 10:
        print(f"\n=== Phase 4: LP Projection (budget={t_lp:.0f}s) ===")
        n_lp_iters = int(t_lp / 3.0)
        current = best_seq.copy()
        current_score = best_score
        for it in range(max(1, n_lp_iters)):
            if time.monotonic() - t_global > time_budget - 100:
                break
            g = lp_project_step(current, M_max_constr=80, time_limit=2.5)
            if g is not None:
                # Interpolate between current and g
                for t_mix in [0.3, 0.5, 0.7, 1.0]:
                    trial = (1 - t_mix) * current + t_mix * g
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()
                    sc = fast_score(trial)
                    if sc < current_score:
                        current = trial.copy()
                        current_score = sc
                        break
        try_update(current, "LP_projections")

    # === Phase 5: Intensive Final Polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Intensive Final Polishing (budget={remaining:.0f}s) ===")

    # Alternate between CD and short MD bursts
    n_polish_cycles = max(3, int(remaining / 60))
    for cyc in range(n_polish_cycles):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break

        # CD burst
        t_cd_burst = min(remaining * 0.4, 80.0)
        if t_cd_burst > 10:
            a_out, _ = coordinate_descent(best_seq, t_cd_burst, seed=100 + cyc, verbose=False)
            try_update(a_out, f"polish_CD_{cyc}")

        # Short MD burst
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t_md_burst = min(remaining * 0.3, 50.0)
        if t_md_burst > 10:
            a_out, _ = mirror_descent_fast(best_seq, t_md_burst,
                                            eta0=0.015, beta0=100.0,
                                            beta_max=300.0, decay=0.7, seed=200 + cyc)
            try_update(a_out, f"polish_MD_{cyc}")

        # Very short LSE
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 15:
            break
        t_lse_burst = min(remaining * 0.2, 30.0)
        if t_lse_burst > 8:
            a_out, _ = lse_smoothed_optimize(best_seq, t_lse_burst,
                                              T_start=10.0, T_end=0.1,
                                              n_stages=8, seed=300 + cyc)
            try_update(a_out, f"polish_LSE_{cyc}")

    # === Phase 6: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 6: Length Exploration (budget={remaining:.0f}s) ===")
        for factor in [0.85, 1.15, 1.3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            n_new = int(round(n * factor))
            if n_new < 100 or n_new > 5000:
                continue
            init_new = interpolate_to_length(best_seq, n_new)
            t_try = min(remaining * 0.3, 60.0)
            if t_try > 15:
                a_out, _ = mirror_descent_fast(init_new, t_try * 0.5,
                                                eta0=0.03, beta0=30.0,
                                                beta_max=120.0, decay=0.5,
                                                seed=500 + int(factor * 100))
                try_update(a_out, f"len_{n_new}_MD")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 15:
                    a_out2, _ = coordinate_descent(init_new, min(t_try * 0.5, remaining * 0.3),
                                                    seed=600 + int(factor * 100), verbose=False)
                    try_update(a_out2, f"len_{n_new}_CD")

    # === Final: One more intensive CD on the best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Final Polish (budget={remaining:.0f}s) ===")
        a_out, _ = coordinate_descent(best_seq, min(remaining * 0.7, 100.0),
                                       seed=999, verbose=False)
        try_update(a_out, "final_CD")

    # Final evaluation with official evaluator
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
