def fast_autocorr(a):

def softmax_fn(z):

def compute_subgradient(b, k_star, a):
    """
    Compute the subgradient of max(conv(a,a)) at a, given the active peak index k_star.
    grad[j] = 2 * a[k_star - j] if 0 <= k_star - j < n, else 0

    if lo <= hi:

        src_idx = k_star - idx

def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)[:conv_len]
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20, lr=0.01):

    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(200, int(remaining * 100 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            obj, grad = lse_obj_grad_z(z, stage_T)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.003 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, warm_start=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j2 in range(js2, je2 + 1):
                    peak_areas.add(j2)

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > 300:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function with multi-phase optimization:
    Phase 1: Load parent + quick spectral shaping
    Phase 2: Adam LSE at parent length
    Phase 3: LP Nesterov at parent length (primary workhorse)
    Phase 4: Subgradient peak shaving
    Phase 5: Adaptive length extension + LP Nesterov
    Phase 6: Final polishing
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    # Phase 1: Quick spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (rem={remaining:.0f}s) ===")
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 5:
        a_out = spectral_shaping(best_seq, n_iter=800, alpha=0.25)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [spectral]: {best_score:.6f}")

    # Phase 2: Adam LSE
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Adam LSE n={n_base} (rem={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_out, _ = adam_lse_optimize(best_seq, t2, T_start=50.0, T_end=0.05, n_stages=15, lr=0.005)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [adam_lse]: {best_score:.6f}")

    # Phase 3: LP Nesterov (main workhorse - give it the most time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov n={n_base} (rem={remaining:.0f}s) ===")
    t3 = min(remaining * 0.35, 500.0)
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [lp_nesterov]: {best_score:.6f}")

    # Phase 4: Subgradient peak shaving (new method)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Subgradient peak shave n={n_base} (rem={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        a_out, _ = subgradient_peak_shave(best_seq, t4, lr_start=0.003, lr_end=0.0002, seed=77)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [subgrad]: {best_score:.6f}")

    # Phase 5: Try longer length
    n_long = int(round(n_base * 1.3))
    n_long = max(n_long, n_base + 100)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Longer n={n_long} (rem={remaining:.0f}s) ===")
    t5 = min(remaining * 0.30, 400.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral at longer length
        t5a = t5 * 0.15
        a_spec = spectral_shaping(init_long, n_iter=300, alpha=0.2)
        sc = evaluate_safe(a_spec)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_spec
            print(f"  *** NEW BEST [spectral_long]: {best_score:.6f}")
        # LP Nesterov at longer length
        t5b = t5 * 0.75
        a_out, _ = nesterov_lp_optimize(best_seq, t5b, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [lp_long]: {best_score:.6f}")

    # Phase 6: Subgradient at current best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Subgradient polish (rem={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        a_out, _ = subgradient_peak_shave(best_seq, t6, lr_start=0.002, lr_end=0.0001, seed=99)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [subgrad2]: {best_score:.6f}")

    # Phase 7: Direct peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Direct peak shave (rem={remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 80.0)
    if t7 > 5:
        a_out, _ = peak_shaving_direct(best_seq, t7, seed=321)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [peak_shave]: {best_score:.6f}")

    # Phase 8: Final Adam LSE with low temperature
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final Adam LSE (rem={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 80.0)
    if t8 > 10:
        a_out, _ = adam_lse_optimize(best_seq, t8, T_start=10.0, T_end=0.01, n_stages=10, lr=0.003)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [final_adam]: {best_score:.6f}")

    # Phase 9: Final spectral polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 5:
        t9 = min(remaining * 0.3, 30.0)
        if t9 > 3:
            a_out = spectral_shaping(best_seq, n_iter=200, alpha=0.05)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  *** NEW BEST [final_spectral]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        final_score = evaluate_safe(best_seq)
        print(f"\nFINAL SCORE: {final_score:.6f} (n={len(best_seq)})")
        best_score = final_score
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
