2. Adam-based LogSumExp gradient optimization
3. Nesterov-accelerated LP zero-sum game (primary method, ~55% of time)
4. Constraint-reduced LP for faster iterations in later stages
5. Peak-shaving coordinate descent for final polish

import numpy as np

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad_a(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20, lr=0.01):
    """LogSumExp-smoothed optimization with Adam optimizer and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(200, int(remaining * 60 / max(1, (n_stages - si)))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            obj, grad = lse_obj_grad_z(z, stage_T)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.003 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working and fast LP solver."""
    results = []
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL, cp.SCS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                results.append((solver_name, elapsed))
        except Exception:
            continue
    if not results:
        return cp.ECOS
    results.sort(key=lambda x: x[1])
    return results[0][0]


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    m = 2 * n - 1
    T = np.zeros((m, n))
    for j in range(n):
        lo = j
        hi = j + n
        T[lo:hi, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, constraint_indices=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    if constraint_indices is not None:
        T_full = build_conv_matrix(a_fixed)
        T = T_full[constraint_indices, :]
    else:
        T = build_conv_matrix(a_fixed)

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, use_reduced_lp=True, reduction_start=50):
    """
    Nesterov-accelerated LP best-response iteration for the zero-sum game.
    Uses constraint reduction after initial iterations for speed.
    Tracks exponentially-weighted running average.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Virtual strategy for LP
    a_prev = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Exponentially weighted running average
    ew_avg = a.copy()
    ew_decay = 1.0  # Will be set adaptively
    ew_weight_sum = 1.0

    # Uniform running average
    avg_a = np.zeros(n)
    n_avg = 0

    active_constraint_indices = None
    last_constraint_update = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Periodically update active constraints (top 50 most binding)
        if use_reduced_lp and t >= reduction_start and (t - last_constraint_update) >= 200:
            b_current = fast_autocorr(y)
            M_current = b_current.max()
            threshold = M_current * 0.95
            active_constraint_indices = np.where(b_current >= threshold)[0]
            if len(active_constraint_indices) < 10:
                active_constraint_indices = np.argsort(b_current)[-50:]
            last_constraint_update = t

        b_star, M_val = solve_lp_best_response(y, solver, active_constraint_indices)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Update running averages
        ew_decay = 1.0 / (1.0 + 0.005 * t ** 0.5)
        ew_avg = ew_decay * ew_avg + (1 - ew_decay) * a
        ew_avg = ew_avg / ew_avg.sum()

        avg_a += a
        n_avg += 1

        # Periodic evaluation
        if t % 100 == 0 or t <= 10:
            score_a = 2.0 * n * fast_autocorr(a).max()
            score_ew = 2.0 * n * fast_autocorr(ew_avg).max()
            score_avg = 2.0 * n * fast_autocorr(avg_a / n_avg).max() if n_avg > 0 else score_a
            score_y = 2.0 * n * fast_autocorr(y).max()

            candidates = [
                (a, score_a),
                (ew_avg, score_ew),
                (avg_a / n_avg, score_avg) if n_avg > 0 else (a, score_a),
                (y, score_y),
            ]

            for cand, sc in candidates:
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()

            # Also try blends
            if n_avg > 50:
                for blend in [0.3, 0.5, 0.7]:
                    mixed = blend * (avg_a / n_avg) + (1 - blend) * a
                    mixed = np.maximum(mixed, 0)
                    ms = mixed.sum()
                    if ms > 1e-15:
                        mixed = mixed / ms
                        sc = 2.0 * n * fast_autocorr(mixed).max()
                        if sc < best_score:
                            best_score = sc
                            best_a = mixed.copy()

            if t % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving: directly reduce top autocorrelation peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for jj in range(js2, je2 + 1):
                    peak_areas.add(jj)

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 500:
                break

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=5000):
    """Subgradient method on the max function with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    eta_0 = 1.0 / np.sqrt(n * 5)
    b = fast_autocorr(a)
    k_star = int(np.argmax(b))

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        eta = eta_0 / np.sqrt(1 + 0.1 * step)

        # Subgradient of max_k c[k] = 2 * (shifted a)
        # c[k] = sum_j a[j] * a[k-j], dc[0]/da[j] = a[k-j] (for valid k-j)
        # So subgrad[j] = 2 * a[k_star - j] for valid indices
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        # Update: a - eta * grad, project onto simplex
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

            # Update k_star
            b = fast_autocorr(a)
            k_star = int(np.argmax(b))
        else:
            # Try smaller step
            a_new = a - 0.1 * eta * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                score = 2.0 * n * fast_autocorr(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
                    b = fast_autocorr(a)
                    k_star = int(np.argmax(b))

        if step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad step {step} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization with LP game as primary method.
    Phase 1: Spectral shaping (fast init)
    Phase 2: Adam LSE (gradient-based refinement)
    Phase 3: LP Nesterov game (main optimization, ~55% of budget)
    Phase 4: Subgradient polish
    Phase 5: Peak shaving
    Phase 6: Low-T Adam final polish
    Phase 7: Try longer length if time permits
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = 1600
    if parent is not None and len(parent) > 0:
        n_base = len(parent)
    else:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Spectral shaping (fast, ~25s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 25.0)
    if t1 > 5:
        t1_start = time.monotonic()
        a_out = spectral_shaping(best_seq, n_iter=2000, alpha=0.25)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [spectral]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")
        print(f"  Phase 1 done ({time.monotonic()-t1_start:.1f}s)")

    # Phase 2: Adam LSE at base length (~60s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Adam LSE at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.05, 60.0)
    if t2 > 10:
        t2_start = time.monotonic()
        a_out, _ = adam_lse_optimize(best_seq, t2, T_start=80.0, T_end=0.1, n_stages=12, lr=0.004)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [adam_lse]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")
        print(f"  Phase 2 done ({time.monotonic()-t2_start:.1f}s)")

    # Phase 3: LP Nesterov game - MAIN PHASE (~55% of budget)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov game at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = remaining * 0.55
    if t3 > 30:
        t3_start = time.monotonic()
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, use_reduced_lp=True, reduction_start=30)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lp_game]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")
        print(f"  Phase 3 done ({time.monotonic()-t3_start:.1f}s)")

    # Phase 4: Subgradient polish (~40s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Subgradient polish (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.04, 40.0)
    if t4 > 5:
        t4_start = time.monotonic()
        a_out, _ = subgradient_polish(best_seq, t4, n_steps=10000)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [subgrad]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")
        print(f"  Phase 4 done ({time.monotonic()-t4_start:.1f}s)")

    # Phase 5: Peak shaving (~60s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak shaving (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.06, 60.0)
    if t5 > 5:
        t5_start = time.monotonic()
        a_out, _ = peak_shaving(best_seq, t5, seed=789)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [peak_shave]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")
        print(f"  Phase 5 done ({time.monotonic()-t5_start:.1f}s)")

    # Phase 6: Low-T Adam final polish (~50s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Low-T Adam polish (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.05, 50.0)
    if t6 > 10:
        t6_start = time.monotonic()
        a_out, _ = adam_lse_optimize(best_seq, t6, T_start=10.0, T_end=0.01, n_stages=10, lr=0.001)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [final_adam]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")
        print(f"  Phase 6 done ({time.monotonic()-t6_start:.1f}s)")

    # Phase 7: One more LP round with reduced constraints if time permits
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        n_try2 = int(round(n_base * 1.15))
        print(f"\n=== Phase 7: LP at n={n_try2} (remaining={remaining:.0f}s) ===")
        t7 = remaining * 0.5
        t7_start = time.monotonic()
        init7 = interpolate_sequence(best_seq, n_try2)
        a_spec7 = spectral_shaping(init7, n_iter=300, alpha=0.2)
        a_out7, _ = nesterov_lp_optimize(a_spec7, t7, solver, use_reduced_lp=True, reduction_start=30)
        sc = evaluate_safe(a_out7)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out7, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [lp_n{n_try2}]: {best_score:.6f} (n={len(best_seq)}, t={elapsed:.0f}s)")
        print(f"  Phase 7 done ({time.monotonic()-t7_start:.1f}s)")

    # Phase 8: Final peak shaving + subgradient
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t8 = remaining * 0.5
        print(f"\n=== Phase 8: Final polish (remaining={remaining:.0f}s) ===")
        t8_start = time.monotonic()
        a_out, _ = peak_shaving(best_seq, t8 * 0.5, seed=101)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [final_shave]: {best_score:.6f} (n={len(best_seq)})")

        remaining2 = time_budget - (time.monotonic() - t_global_start)
        if remaining2 > 10:
            a_out2, _ = subgradient_polish(best_seq, remaining2 * 0.8, n_steps=5000)
            sc2 = evaluate_safe(a_out2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = np.maximum(a_out2, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [final_subgrad]: {best_score:.6f} (n={len(best_seq)})")
        print(f"  Phase 8 done ({time.monotonic()-t8_start:.1f}s)")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        # Final evaluate after normalization
        fs2 = evaluate_safe(best_seq)
        if np.isfinite(fs2):
            best_score = fs2
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
