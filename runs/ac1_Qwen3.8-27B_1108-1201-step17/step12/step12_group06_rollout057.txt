        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsbfgs_lse(a0, time_limit, T_start=80.0, T_min=0.3, n_stages=30, iters_per_stage=35):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            obj_ratio = (sc - best_score) / (best_score + 1e-15)
            if obj_ratio > 0.003:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 1)

            if stagnant >= 2:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            if (stage + 1) % 8 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def exact_subgradient_optimize(a0, time_limit, n_steps=500000):
    """Exact active-set subgradient with Polyak step and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0
    patience = 0
    patience_best = best_score

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find ALL positions achieving the maximum (or near-maximum)
        threshold = max_b - 1e-12 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute exact subgradient: average gradient at active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Project to tangent space of simplex
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Polyak step size with capping
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Backtracking line search
        if sc_new > cur_score * 1.005:
            lo_e, hi_e = 0.0, eta
            for _ in range(6):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            if eta < 1e-12:
                patience += 1
                if patience > 200:
                    break
                # Add small random perturbation to escape
                z += 0.01 * np.random.randn(n)
                a = softmax_from_z(z)
                continue
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        # Running average (exponential, window ~3000)
        beta = 0.999
        avg_a = beta * avg_a + (1 - beta) * a
        avg_count += 1

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
            patience = 0
            patience_best = best_score
        else:
            if sc_new < patience_best:
                patience = 0
            else:
                patience += 1

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
            patience = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_passes=50):
    """Coordinate descent: optimize one coordinate at a time via ternary search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Work in log-space for positivity
    z = z_from_a(a).copy()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break

        # Cycle through coordinates
        order = np.arange(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                    break

            # For coordinate i, find optimal value
            # The score as function of z[i] is smooth, use golden section
            z_i_old = z[i]
            # Search range
            z_lo = z_i_old - 2.0
            z_hi = z_i_old + 2.0

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = z_lo + (1.0 - gr) * (z_hi - z_lo)
            b_g = z_lo + gr * (z_hi - z_lo)

            z[i] = a_g
            a_g_test = softmax_from_z(z)
            f_a = score_normalized(a_g_test)

            z[i] = b_g
            a_b_test = softmax_from_z(z)
            f_b = score_normalized(a_b_test)

            for _ in range(12):
                if f_a < f_b:
                    z_hi = b_g
                    b_g = a_g
                    f_b = f_a
                    a_g = z_lo + (1.0 - gr) * (z_hi - z_lo)
                    z[i] = a_g
                    a_g_test = softmax_from_z(z)
                    f_a = score_normalized(a_g_test)
                else:
                    z_lo = a_g
                    a_g = b_g
                    f_a = f_b
                    b_g = z_lo + gr * (z_hi - z_lo)
                    z[i] = b_g
                    a_b_test = softmax_from_z(z)
                    f_b = score_normalized(a_b_test)

            # Set to better of the two
            if f_a < f_b:
                z[i] = a_g
            else:
                z[i] = b_g

        # Check improvement
        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)
        if sc < best_score:
            best_score = sc
            best_a = a_check.copy()

        if (p + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDescent pass={p+1}/{n_passes}: score={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def window_init(n, window_type='hann'):
    """Generate window-function based initializations."""
    x = np.linspace(0, 1, n)
    if window_type == 'hann':
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif window_type == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif window_type == 'raised_cosine':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
    elif window_type == 'sinc_like':
        t = (x - 0.5) * n * 0.1
        a = np.sinc(t) ** 2
        a = np.maximum(a, 0)
    elif window_type == 'double_bump':
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif window_type == 'optimal_shape':
        # Shape inspired by minimum-peak autocorrelation windows
        a = (1 - x) ** 0.5 * x ** 0.5 * (1 + 0.3 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif window_type == 'asymmetric_bump':
        a = x ** 0.8 * (1 - x) ** 1.2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_random_init(n, rng):
    """Generate a random smooth initialization."""
    # Random Fourier series with limited bandwidth
    n_freq = rng.integers(2, min(8, n // 4))
    coeffs = rng.normal(0, 1, n_freq + 1)
    phases = rng.uniform(0, 2 * np.pi, n_freq + 1)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for k in range(1, n_freq + 1):
        a += (coeffs[k] / k) * np.cos(2 * np.pi * k * x + phases[k])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_simple(a0, time_limit, max_iters=200000):
    """Simplified Nesterov LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Best response: find b that minimizes max_k (y*b)[k] subject to sum(b)=1, b>=0
        # Use simple iterative approach: b_new ∝ exp(-T * (y*b_old))
        # Or use the LP approach with cvxpy
        b_prod = fast_convolve(y)
        T_br = 5.0
        b_new = np.exp(-T_br * (b_prod - b_prod.max()))
        b_new = np.maximum(b_new, 0)
        sb = b_new.sum()
        if sb < 1e-15:
            b_new = np.ones(n) / n
        else:
            b_new = b_new / sb

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Nesterov iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function with multi-phase optimization."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Get parent sequence if available
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = window_init(1200, 'hann')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Multi-initialization exploration (15% time) ===
    print(f"\n=== Phase 1: Multi-init exploration (score={best_score:.8f}) ===")
    t1_total = 0.15 * (budget_s - 30)
    n_base = len(best_seq)
    inits_to_try = ['hann', 'blackman', 'gaussian', 'optimal_shape', 'asymmetric_bump', 'double_bump']
    t_per_init = t1_total / (len(inits_to_try) + 2)

    for wtype in inits_to_try:
        remaining = global_deadline - time.monotonic()
        if remaining < t_per_init:
            break
        t_init = min(t_per_init, remaining * 0.8)
        a_init = window_init(n_base, wtype)
        # Quick LSE
        a_out, _ = lsbfgs_lse(a_init, t_init, T_start=50.0, T_min=1.0, n_stages=8, iters_per_stage=25)
        try_update(a_out, f"init_{wtype}")

    # Random init
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        a_rand = structured_random_init(n_base, rng)
        a_out, _ = lsbfgs_lse(a_rand, min(30, remaining * 0.5), T_start=50.0, T_min=1.0, n_stages=8, iters_per_stage=25)
        try_update(a_out, "init_random")

    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Main LSE optimization (30% time) ===
    print(f"\n=== Phase 2: Main LSE optimization (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.30 * (budget_s - 30), remaining * 0.6)
    if t2 > 30:
        # Multi-temperature LSE from best
        a_lse, _ = lsbfgs_lse(best_seq, t2 * 0.5, T_start=100.0, T_min=0.5, n_stages=15, iters_per_stage=30)
        try_update(a_lse, "lse_main_1")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lsbfgs_lse(best_seq, min(t2 * 0.3, remaining * 0.7), T_start=20.0, T_min=0.2, n_stages=12, iters_per_stage=30)
            try_update(a_lse2, "lse_main_2")

    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Exact subgradient (25% time) ===
    print(f"\n=== Phase 3: Exact subgradient (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.25 * (budget_s - 30), remaining * 0.6)
    if t3 > 20:
        a_sg, _ = exact_subgradient_optimize(best_seq, t3 * 0.6, n_steps=300000)
        try_update(a_sg, "subgrad_main")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_sg2, _ = exact_subgradient_optimize(best_seq, remaining * 0.5, n_steps=200000)
            try_update(a_sg2, "subgrad_2")

    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Coordinate descent polish (15% time) ===
    print(f"\n=== Phase 4: Coordinate descent (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t4 > 15:
        a_cd, _ = coordinate_descent_polish(best_seq, t4, n_passes=30)
        try_update(a_cd, "coord_descent")

        # Another round from the result
        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_cd2, _ = coordinate_descent_polish(best_seq, remaining * 0.5, n_passes=20)
            try_update(a_cd2, "coord_descent_2")

    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.10 * (budget_s - 30), remaining * 0.7)
    if t5_total > 30:
        lengths_to_try = [n_base + 200, n_base + 500, n_base + 1000]
        t_per_len = t5_total / len(lengths_to_try)
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < t_per_len:
                break
            t5 = min(t_per_len, remaining * 0.8)
            if t5 < 15:
                break
            print(f"    Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = lsbfgs_lse(init_try, t5 * 0.5, T_start=50.0, T_min=1.0, n_stages=8, iters_per_stage=25)
            try_update(a_try, f"len_{n_try}")

            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = exact_subgradient_optimize(a_try, remaining * 0.5, n_steps=100000)
                try_update(a_try2, f"len_{n_try}_sg")

    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final combined polish (5% time) ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        # Quick LSE from low T
        a_f1, _ = lsbfgs_lse(best_seq, remaining * 0.3, T_start=10.0, T_min=0.1, n_stages=8, iters_per_stage=20)
        try_update(a_f1, "final_lse")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Quick subgradient
            a_f2, _ = exact_subgradient_optimize(best_seq, remaining * 0.4, n_steps=50000)
            try_update(a_f2, "final_sg")

        remaining = global_deadline - time.monotonic()
        if remaining > 8:
            # Quick coordinate descent
            a_f3, _ = coordinate_descent_polish(best_seq, remaining * 0.5, n_passes=10)
            try_update(a_f3, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
