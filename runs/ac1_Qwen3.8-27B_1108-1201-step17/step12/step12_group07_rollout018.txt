def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None, warm_start=None, time_limit=None):

    T = build_conv_matrix(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:

        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break

                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

def mirror_descent_subgrad(a0, time_limit, max_steps=500000, verbose=True):
    """Mirror descent with exact subgradient of max(conv(a,a)) in log-softmax parameterization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.5
    last_improve = 0
    last_print = t0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 8000 and eta < 0.01:
            break

        b = fast_convolve(a)
        max_b = b.max()
        argmax_idx = int(np.argmax(b))

        # Subgradient of max conv: gradient of conv(k*) w.r.t. a
        k_star = argmax_idx
        grad_a = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad_a[i] = 2.0 * a[j]
            # Handle the case where i appears as both positions
            # (already accounted for since we're summing over the convolution formula)

        # Project gradient to simplex in log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            z = z_new
            last_improve = step
            eta = min(eta * 1.001, 2.0)
        else:
            eta *= 0.95
            if eta < 0.001:
                break
            # Still update for continuity if improvement is marginal
            if sc < best_score * 1.001:
                z = z_new
                a = a_new

        if verbose and (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    MD iter={step+1}: score={best_score:.8f} eta={eta:.4f}")
                last_print = elapsed

    return best_a, best_score


def nesterov_lp(a0, time_limit, solver=None, max_iters=200000, verbose=True):
    """Nesterov-accelerated LP with adaptive momentum and restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    last_print = t0
    warm_start = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        lp_time = min(5.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        b_new, M_lp = solve_lp_best_response(y, solver, warm_start=warm_start, time_limit=lp_time)
        if b_new is None:
            # Fallback: use gradient step
            b_conv = fast_convolve(a)
            k_star = int(np.argmax(b_conv))
            grad = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] = 2.0 * a[j]
            a_new = a - 0.01 * (grad - np.dot(grad, a))
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new / sa
                a_prev = a.copy()
                a = a_new
            continue

        warm_start = b_new.copy()
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Adaptive restart
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=30, verbose=True):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        # More iterations for early stages (smoother landscape)
        iters = 60 if si < n_stages // 3 else 30

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            if verbose and (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: best={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def peak_refine(a0, time_limit, threshold_frac=0.90, max_steps=100000, verbose=True):
    """Multiplicative weights targeting top peaks with adaptive threshold."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 10000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * threshold_frac

        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Weighted gradient: more weight on higher peaks
        weights = b[topk]
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for idx, k in enumerate(topk):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                grad[i] += weights[idx] * a[k - i]

        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.005, 0.1)
            # Tighten threshold as we converge
            threshold_frac = min(threshold_frac * 1.001, 0.99)
        else:
            eta *= 0.8
            if eta < 1e-8:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_start_phase(base_seq, n, time_budget, solver, verbose=True):
    """Try multiple diverse initializations and return the best."""
    rng = np.random.default_rng(12345)
    best_a = base_seq.copy()
    best_score = score_normalized(base_seq)
    
    t0 = time.monotonic()
    
    # Try several structured inits
    for mode in range(6):
        if time.monotonic() - t0 > time_budget:
            break
        a_try = structured_init(n, rng, mode=mode)
        sc = score_normalized(a_try)
        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
    
    # Blend with parent
    for t_blend in [0.3, 0.5, 0.7]:
        if time.monotonic() - t0 > time_budget:
            break
        parent_interp = interpolate_sequence(base_seq, n)
        a_try = (1 - t_blend) * parent_interp + t_blend * structured_init(n, rng, mode=mode)
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s > 1e-15:
            a_try = a_try / s
            sc = score_normalized(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 0: Quick diverse start (20s) ===
    print(f"\n=== Phase 0: Diverse start at n={n_base} ===")
    t0 = min(20.0, (_global_deadline - time.monotonic()) * 0.02)
    if t0 > 5:
        a_div, _ = diverse_start_phase(best_seq, n_base, t0, solver)
        try_update(a_div, "diverse_start")
    print(f"  Phase 0 done: best={best_score:.8f}")

    # === Phase 1: Mirror descent pre-optimization (60s) ===
    print(f"\n=== Phase 1: Mirror descent n={best_n} ===")
    remaining = _global_deadline - time.monotonic()
    t1 = min(60.0, remaining * 0.04)
    if t1 > 5:
        a_out, _ = mirror_descent_subgrad(best_seq, t1, max_steps=300000)
        try_update(a_out, "mirror_descent")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: LP Nesterov main (500s) ===
    print(f"\n=== Phase 2: LP Nesterov n={best_n} ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(500.0, remaining * 0.35)
    if t2 > 30:
        a_out, _ = nesterov_lp(best_seq, t2, solver=solver)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Peak-targeted refinement (150s) ===
    print(f"\n=== Phase 3: Peak refinement ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(150.0, remaining * 0.12)
    if t3 > 10:
        a_pt, _ = peak_refine(best_seq, t3, threshold_frac=0.90)
        try_update(a_pt, "peak_refine")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: LSE L-BFGS polish (250s) ===
    print(f"\n=== Phase 4: LSE L-BFGS ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.18)
    if t4 > 20:
        a_lse = lse_lbgfs(best_seq, t4, T_start=80.0, T_end=0.01, n_stages=30)
        try_update(a_lse, "lse_polish")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 5: Second LP round (200s) ===
    print(f"\n=== Phase 5: LP Nesterov round 2 ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(200.0, remaining * 0.15)
    if t5 > 20:
        a_out2, _ = nesterov_lp(best_seq, t5, solver=solver)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Peak refine + LSE (100s) ===
    print(f"\n=== Phase 6: Combined refinement ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        t6a = min(50.0, remaining * 0.3)
        if t6a > 5:
            a_pt2, _ = peak_refine(best_seq, t6a, threshold_frac=0.95)
            try_update(a_pt2, "peak_refine_r2")
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            t6b = min(50.0, remaining * 0.4)
            if t6b > 5:
                a_lse2 = lse_lbgfs(best_seq, t6b, T_start=30.0, T_end=0.005, n_stages=20)
                try_update(a_lse2, "lse_r2")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: Length exploration (150s) ===
    print(f"\n=== Phase 7: Length exploration ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 60:
        n_candidates = sorted(set([
            best_n - 50, best_n + 50,
            best_n - 100, best_n + 100,
            best_n + 150, best_n - 200,
            best_n + 200, 1000, 1500, 2000
        ]))
        n_candidates = [x for x in n_candidates if x >= 100 and x <= 5000]
        
        for n_try in n_candidates:
            remaining = _global_deadline - time.monotonic()
            if remaining < 20:
                break
            if n_try == best_n:
                continue
            t7 = min(30.0, remaining * 0.25)
            if t7 < 8:
                continue
            print(f"  Trying n={n_try} (budget={t7:.0f}s)")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick optimization at new length
            a_try, _ = nesterov_lp(init_try, t7, solver=solver, max_iters=50000, verbose=False)
            try_update(a_try, f"len_{n_try}")
    
    # === Phase 8: Final mirror descent on best (remaining) ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 8: Final mirror descent (budget={remaining:.0f}s) ===")
        a_final, _ = mirror_descent_subgrad(best_seq, remaining * 0.6, max_steps=200000)
        try_update(a_final, "final_md")
    
    remaining = _global_deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Phase 9: Final peak refine (budget={remaining:.0f}s) ===")
        a_final2, _ = peak_refine(best_seq, remaining * 0.7, threshold_frac=0.97)
        try_update(a_final2, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
