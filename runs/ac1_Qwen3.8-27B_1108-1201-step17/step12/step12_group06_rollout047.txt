import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

_lp_solver = None

def get_lp_solver():

    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:

def solve_lp_best_response(a_fixed, solver=None):

    b_var = cp.Variable(n, nonneg=True)

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Extrapolation (Nesterov)
    y = a.copy()
    # Polyak average
    avg_a = a.copy()
    avg_count = 0
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        # Polyak averaging
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count = min(avg_count + 1, 100000)
        if avg_count >= 100000:
            avg_a = a.copy()  # reset to current to avoid drift
        
        # Track best
        for candidate in [a, y, avg_a]:
            sc = score_normalized(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NL iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed
    
    return best_a, best_score

def lse_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = np.clip(Tb.max(), None, 700)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    # Gradient: 4n * sum_k w_k * a_{k-j} for each j
    # This is the cross-correlation of w and a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    grad_a *= (2.0 * n)
    
    # Convert to z-gradient (softmax Jacobian)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lse_lbfgs(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=None, iters_per_stage=None):
    n = len(a0)
    if n_stages is None:
        n_stages = 40
    if iters_per_stage is None:
        iters_per_stage = 50
    
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    # Geometric temperature schedule
    if T_start / T_end > 1:
        ratio = (T_end / T_start) ** (1.0 / n_stages)
    else:
        ratio = 1.0
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        T = T_start * (ratio ** stage)
        
        # Adaptive iterations based on remaining time
        iters = iters_per_stage
        if remaining < n_stages * 5:  # running low on time
            iters = max(10, iters_per_stage // 2)
        
        def func(z_val):
            obj, _ = lse_grad(z_val, T, n)
            return obj
        
        def jac(z_val):
            _, grad = lse_grad(z_val, T, n)
            return grad
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            if (stage + 1) % 8 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.4f}: cur={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")
                
        except Exception:
            pass
    
    return best_a, best_score

def exact_subgradient_polish(a0, time_limit, n_steps=1000000):
    """Subgradient method with Polyak step for the exact (non-smooth) objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    z = z_from_a(a).copy()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    
    # Running average for convergence
    avg_a = a.copy()
    avg_count = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        # Active peaks (within 0.5% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) > 100:
            top_idx = np.argsort(b[active])[-100:]
            active = active[top_idx]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Convert to z-gradient
        S = np.dot(g, a)
        grad_z = a * (g - S)
        
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-16:
            break
        
        # Polyak step size
        gap = cur_score - lower_bound
        if gap < 1e-12:
            break
        eta = 2.0 * gap / (grad_norm * grad_norm)
        eta = min(eta, 1.0)
        
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)
        
        # Line search if needed
        if sc_new > cur_score * 1.005:
            lo_e, hi_e = 0.0, eta
            for _ in range(8):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            if eta < 1e-15:
                break
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)
        
        z = z_new
        a = a_new
        
        # Average
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count = min(avg_count + 1, 20000)
        if avg_count >= 20000:
            avg_a = a.copy()
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
        
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
        
        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ES step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-2 * (x - 0.5) ** 2 / 0.1)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    elif choice == 4:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n)
    else:
        a = np.exp(-((x - 0.5) ** 2) / 0.08)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
