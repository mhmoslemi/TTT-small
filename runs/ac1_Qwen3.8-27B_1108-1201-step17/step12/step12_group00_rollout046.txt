def build_conv_matrix(a):

def lp_best_response(a, M, solver):

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu - (cv - S) / rho > 0
    if not np.any(cond):
        return np.full(n, S / n)
    rho_star = int(np.max(rho[cond]))
    theta = (cv[rho_star - 1] - S) / rho_star
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, w):
    """Compute gradient of LSE objective w.r.t. a.
    w: weights over lags (length 2n-1)
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # conv(w, a) gives sum_k w[k] * a[j+k] at position j
    conv1 = irfft(W * A, fft_size)
    # For the second term, we need w flipped
    w_flip = w[::-1].copy()
    Wf = rfft(w_flip, fft_size)
    conv2 = irfft(Wf * A, fft_size)
    # The gradient at position j is 2*(conv1[j] + conv2[j-n+1] or similar)
    # Actually let me just do it directly
    grad = np.zeros(n)
    # Term 1: sum_k w[k] * a[j+k] for valid j+k
    # This is (w convolved with a) at position j, but w has length 2n-1
    # Let's use a different approach: for each j, grad[j] = 2 * sum_k w[k] * (a[j+k]*1{valid} + a[j-k]*1{valid})
    # Use FFT: define extended a (length 2n-1, zero-padded to fft_size)
    A_ext = np.zeros(fft_size)
    A_ext[:n] = a
    A_ext_f = rfft(A_ext, fft_size)
    # conv(w, A_ext)[j] = sum_k w[k] * A_ext[j+k]
    conv_forward = irfft(W * A_ext_f, fft_size)
    # For backward: sum_k w[k] * A_ext[j-k] = sum_k w[k] * A_ext_sym[j+k] where A_ext_sym is reversed
    A_ext_sym = np.zeros(fft_size)
    A_ext_sym[:n] = a[::-1]
    A_ext_sym_f = rfft(A_ext_sym, fft_size)
    w_rev = w[::-1].copy()
    Wr = rfft(w_rev, fft_size)
    conv_backward = irfft(Wr * A_ext_f, fft_size)
    grad = 2.0 * (conv_forward[:n] + conv_backward[:n])
    return grad


def lse_smoothed_objective(a, w, tau):
    """Compute LSE approximation of max_k c_k(a)."""
    c = fast_autocorr(a)

def semi_smooth_newton_polish(a0, time_limit, tau_start=1.0, tau_end=0.01):
    """Semi-smooth Newton via LSE + backtracking projected gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adaptive tau schedule
    n_phases = 8
    taus = np.geomspace(tau_start, tau_end, n_phases)
    tau_idx = 0

    for tau in taus:
        if time.monotonic() - t0 > time_limit:
            break
        phase_time = min(120.0, time_limit / n_phases)
        phase_t0 = time.monotonic()
        step = norm / (n * tau)  # Initial step size

        while time.monotonic() - phase_t0 < phase_time:
            c = fast_autocorr(a)
            c_max = c.max()
            exp_c = np.exp(c / tau - c_max / tau)
            w = exp_c / exp_c.sum()

            grad = lse_gradient(a, w)
            # Projected gradient step with backtracking
            for backtrack in range(20):
                a_trial = a - step * grad
                a_trial = project_simplex(a_trial, norm)
                score_trial = fast_autocorr(a_trial).max()
                if score_trial < best_score + 0.01 * c_max:
                    break
                step *= 0.5
            else:
                break

            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()
            a = a_trial
            step = step * 0.9  # Slightly decay step

    return best_a, best_score


def nesterov_lp_longstep(a0, time_limit, solver, verbose=True):

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)

def coordinate_descent_polish(a0, time_limit, seed=123):
    """Coordinate descent with spectral peak identification."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            # Find the pair (i, j) with i+j=k (or |i-j|=k for linear conv)
            # For linear convolution: c[k] = sum_i a[i]*a[k-i] for valid i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Move mass from src to a position that contributes least to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            if len(candidates) == 0:
                continue

            # Try a few destinations
            for _ in range(min(3, len(candidates))):
                dst = candidates[rng.integers(len(candidates))]
                for frac in [0.001, 0.003, 0.008, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-6:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    else:
        # Multi-scale oscillation
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * x) + 0.1 * np.sin(2 * np.pi * 7 * x) + 0.05 * np.sin(2 * np.pi * 13 * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def dual_guided_descent(a0, time_limit, solver, verbose=False):
    """Primal-dual approach: track dual variables (lag distribution) explicitly."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    # Dual: distribution over lags
    c = fast_autocorr(a)
    c_max = c.max()
    q = np.exp(c / max(c_max, 1e-10) - 1.0)
    q = q / q.sum()

    best_a = a.copy()
    best_score = c_max
    t0 = time.monotonic()
    tau = 0.1

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)

        # Dual update: softmax of convex combination
        tau_cur = tau / (1.0 + 0.01 * t)
        c_shifted = c - c.max()
        q_new = np.exp(c_shifted / max(tau_cur, 1e-6) - np.max(c_shifted / max(tau_cur, 1e-6)))
        q_sum = q_new.sum()
        if q_sum < 1e-30:
            q_new = np.zeros(len(c))
            q_new[np.argmax(c)] = 1.0
        else:
            q_new = q_new / q_sum
        q = 0.5 * q + 0.5 * q_new

        # Primal update: gradient of sum_k q_k * c_k(a)
        # grad = 2 * sum_k q_k * (a shifted by k)
        grad = lse_gradient(a, q)

        # Projected gradient step
        step = 1.0 / (np.linalg.norm(grad) + 1e-10)
        a_trial = a - step * grad
        a_trial = project_simplex(a_trial, norm)

        score = fast_autocorr(a_trial).max()
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
        a = a_trial

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Dual t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    print(f"\n=== Phase 1: LP Nesterov (standard) n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_standard(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_std_n{n_base}")

    # Phase 2: Long-step FOM+ at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Long-step FOM+ n={n_base} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_longstep(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_ls_n{n_base}")

    # Phase 3: Semi-smooth Newton polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Semi-smooth Newton n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = semi_smooth_newton_polish(a_norm, t3, tau_start=2.0, tau_end=0.05)
        do_try_update(a_out, f"ssn_n{n_best}")

    # Phase 4: Dual-guided descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Dual-guided n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = dual_guided_descent(a_norm, t4, solver, verbose=False)
        do_try_update(a_out, f"dual_n{n_best}")

    # Phase 5: Try longer length
    n_long = n_base + 448  # 2048 if base is 1600
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n_long} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_longstep(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_long}")

    # Phase 6: SSN polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: SSN polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = semi_smooth_newton_polish(a_norm, t6, tau_start=0.5, tau_end=0.02)
        do_try_update(a_out, f"ssn_polish_n{n_best}")

    # Phase 7: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=555)
        do_try_update(a_out, f"cd_n{n_best}")

    # Phase 8: Final LP push if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.40, 200.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_longstep(a_norm, t8, solver, verbose=True)
        do_try_update(a_out, f"final_lp_n{n_best}")

    # Phase 9: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 5:
            n_best = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coordinate_descent_polish(a_norm, t9, seed=999)
            do_try_update(a_out, "final_cd")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
