def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Monotone (minimizes center peak by rearrangement)
        a = 1.0 + 2.0 * x
    elif choice == 1:
        # Monotone with curvature
        a = (1.0 - x) ** 2 + 0.1
    elif choice == 2:
        # Anti-monotone
        a = 1.0 + 2.0 * (1.0 - x)
    elif choice == 3:
        # Asymmetric bump
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.1 * np.exp(-4.0 * (x - 0.7) ** 2) + 0.05
    elif choice == 4:
        # Two bumps
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.75) ** 2) + 0.1
    elif choice == 5:
        # Power law
        a = (x + 0.1) ** 0.3
    elif choice == 6:
        # Cosine taper
        a = 0.5 + 0.5 * np.cos(np.pi * x)
    elif choice == 7:
        # Asymmetric cosine
        a = 0.5 + 0.4 * np.cos(2 * np.pi * (x - 0.2)) + 0.1 * np.cos(4 * np.pi * x)
    elif choice == 8:
        # Exponential decay
        a = 2.0 * np.exp(-3.0 * x) + 0.2
    else:
        # Linear with noise
        a = 1.0 + x + 0.1 * rng.standard_normal(n)
    
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive restart
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 3.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        elif no_improve > 200:
            # Force restart after prolonged no-improvement
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
            no_improve = 0
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def coordinate_descent(a0, time_limit, max_rounds=500):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Precompute which k values are affected by changing a[i]
    # Changing a[i] affects (a*a)_k for k in [0, 2*i] (from a[i]*a[k-i]) and [i, i+2*(n-1-i)] 
    # Actually: (a*a)_k includes a[i]*a[k-i] and a[k-i]*a[i] (same term if i == k-i)
    # So changing a[i] to x changes (a*a)_k by:
    #   For k where both i and k-i are valid: delta = (x - a[i]) * a[k-i] + a[i] * (x - a[i]) - (x-a[i])^2 if i == k-i
    #   More precisely: new (a*a)_k = old (a*a)_k - a[i]*a[k-i] - a[k-i]*a[i] + x*a[k-i] + a[k-i]*x (adjusting for double count)
    # Actually let's just recompute the affected convolution values directly.
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to a small positive value
                candidates = [0.0, a[i] + 1e-4]
            else:
                # Try a range of values around current
                delta = max(1e-4, old_val * 0.2)
                candidates = [max(0, old_val - delta), old_val, min(2*old_val + 1e-4, 1.0)]
            
            best_x = old_val
            best_local_score = best_score
            
            for x in candidates:
                a[i] = x
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                sc = score_normalized(a_norm)
                if sc < best_local_score - 1e-10:
                    best_local_score = sc
                    best_x = x
            
            if best_x != old_val:
                a[i] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                    improved = True
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
            
            a[i] = old_val  # Restore for next iteration? No, keep the change.
        
        if not improved:
            break
    
    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 30 if si < n_stages // 3 else 15
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-20, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a, best_score


def multiplicative_weights_fast(a0, time_limit, eta_init=0.1, max_steps=100000):
    """Fast multiplicative weights with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_init
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Weight by proximity to max
        threshold = max_b * 0.85
        weights = np.where(b >= threshold, 1.0, 0.0)
        if weights.sum() == 0:
            weights[int(np.argmax(b))] = 1.0
        
        # Gradient: for each position i, how much does it contribute to weighted peaks?
        grad = np.zeros(n)
        for k in np.where(weights > 0)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += weights[k] * a[k - j]
        grad /= weights.sum()
        
        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.005, 0.3)
        else:
            eta *= 0.9
            if eta < 1e-8:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def build_T_rows(a_fixed, k_indices):
    """Build constraint matrix rows for given lag indices."""
    n = len(a_fixed)
    T = np.zeros((len(k_indices), n))
    for row, k in enumerate(k_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[row, j] = a_fixed[k - j]
    return T


def solve_lp_best_response(a_fixed, solver=None, top_k=None):
    """Solve LP: min M s.t. (a*b)_k <= M for top_k lags, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_conv = fast_convolve(a_fixed)
    size = len(b_conv)
    
    if top_k is not None and top_k < size:
        top_indices = np.argsort(b_conv)[-top_k:]
        top_indices = np.sort(top_indices)
    else:
        top_indices = np.arange(size)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_T_rows(a_fixed, top_indices)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, top_k_init=60, top_k_max=200):
    """Nesterov LP with adaptive top-K and restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    no_improve = 0
    last_print = t0
    top_k = top_k_init

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Gradually increase top_k as we converge
        progress = min(1.0, it / (max_iters * 0.5))
        top_k_cur = int(top_k_init + (top_k_max - top_k_init) * progress)

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver, top_k=top_k_cur)
        if b_new is None:
            b_new, M_lp = solve_lp_best_response(y, solver, top_k=30)
            if b_new is None:
                continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive restart
        sc_cur = score_normalized(a)
        if (sc_new > sc_cur * 1.0008 and t_nest > 4.0) or no_improve > 150:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
            no_improve = 0
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NL iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} kr={restart_count} tk={top_k_cur}")
                last_print = elapsed

    return best_a, best_score


def multiplicative_weights_fast(a0, time_limit, eta_init=0.1, max_steps=200000, threshold_frac=0.88):
    """Fast multiplicative weights with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_init
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * threshold_frac
        peak_lags = np.where(b >= threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([int(np.argmax(b))])

        # Compute gradient
        grad = np.zeros(n)
        for k in peak_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(peak_lags)

        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.003, 0.25)
            threshold_frac = max(threshold_frac - 0.0005, 0.6)
        else:
            eta *= 0.92
            if eta < 1e-9:
                break

    return best_a, best_score


def peak_deflation(a0, time_limit, max_rounds=200):
    """Targeted peak reduction by adjusting coordinates that contribute to peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Find the top peaks
        num_peaks = min(10, n)
        peak_indices = np.argsort(b)[-num_peaks:]
        peak_indices = np.sort(peak_indices)

        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # For peak at lag k: (a*a)_k = sum_j a_j * a_{k-j}
            # Find the pair (j, k-j) with maximum product
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            best_reduction = 0
            best_j = -1
            for j in range(js, je + 1):
                other = k - j
                if other < 0 or other >= n or other == j:
                    prod = a[j] * a[j]
                else:
                    prod = a[j] * a[other]
                if prod > best_reduction:
                    best_reduction = prod
                    best_j = j

            if best_j < 0:
                continue

            other = k - best_j
            # Try reducing a[best_j] (and maybe a[other])
            old_a = a.copy()
            
            # Strategy: reduce the larger of the pair
            if other >= 0 and other < n and other != best_j:
                if a[best_j] >= a[other]:
                    target = best_j
                    recomp = other
                else:
                    target = other
                    recomp = best_j
            else:
                target = best_j
                recomp = -1

            # Reduce target, redistribute to recomp (if valid) or spread
            reduction_frac = 0.15  # reduce by 15%
            amount = a[target] * reduction_frac
            a[target] -= amount
            if recomp >= 0 and recomp != target:
                a[recomp] += amount
            else:
                # Spread to neighbors
                spread = amount / 5.0
                for d in range(-2, 3):
                    ni = target + d
                    if 0 <= ni < n:
                        a[ni] += spread

            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = old_a.copy()
                continue
            a = a / s_a

            sc = score_normalized(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                improved = True
            else:
                a = old_a.copy()

        if not improved:
            break

    return best_a, best_score


def coordinate_descent(a0, time_limit, max_rounds=300):
    """Coordinate descent with golden-section-like search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        # Randomized coordinate order for diversity
        order = np.random.default_rng(rnd * 7 + 42).permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Try a few candidate values
            candidates = []
            delta1 = old_val * 0.15
            delta2 = old_val * 0.35
            candidates.extend([
                max(0.0, old_val - delta2),
                max(0.0, old_val - delta1),
                old_val + delta1,
                old_val + delta2,
            ])

            best_x = old_val
            best_sc = best_score

            for x in candidates:
                a[i] = x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                sc = score_normalized(a_norm)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_x = x

            if best_x != old_val:
                a[i] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                if best_sc < best_score - 1e-12:
                    best_score = best_sc
                    best_a = a.copy()
                    improved = True

            a[i] = old_val if best_x == old_val else best_x
            # Actually we already set it above, just make sure consistency
            if best_x != old_val:
                a[i] = best_x

        if not improved:
            break

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = max(5, int(30 * (1.0 - si / n_stages)))
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-20, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Strong monotone (minimizes center peak)
        a = 1.0 + 3.0 * x
    elif choice == 1:
        # Curved monotone
        a = (1.0 - x) ** 2 + 0.05
    elif choice == 2:
        # Anti-monotone
        a = 1.0 + 3.0 * (1.0 - x)
    elif choice == 3:
        # Power law
        a = (x + 0.05) ** 0.4
    elif choice == 4:
        # Exponential
        a = 3.0 * np.exp(-2.5 * x) + 0.2
    elif choice == 5:
        # Two asymmetric bumps
        a = np.exp(-10.0 * (x - 0.2) ** 2) + 0.6 * np.exp(-10.0 * (x - 0.75) ** 2) + 0.1
    elif choice == 6:
        # Cosine taper
        a = 0.3 + 0.7 * np.cos(np.pi * x) ** 2
    elif choice == 7:
        # Asymmetric cosine
        a = 0.5 + 0.4 * np.cos(2 * np.pi * (x - 0.15)) + 0.15 * np.cos(6 * np.pi * x)
    elif choice == 8:
        # S-shaped
        a = 1.0 / (1.0 + np.exp(-8.0 * (x - 0.4))) + 0.1
    elif choice == 9:
        # Linear with curvature
        a = 1.0 + 1.5 * x + 0.5 * x ** 2
    elif choice == 10:
        # Gaussian-like with offset
        a = np.exp(-2.0 * (x - 0.35) ** 2) * 2.0 + 0.3
    else:
        # Noisy monotone
        a = 1.0 + 2.0 * x + 0.15 * rng.standard_normal(n)

    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                try_update(a_def2, "deflate_r2")
    print(f"  Phase 7 done: best_score={best_score:.8f}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final polish score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 15:
        t8 = min(80.0, remaining * 0.4)
        if t8 > 10:
            a_lse2 = lse_lbgfs(best_seq, t8, T_start=15.0, T_end=0.005, n_stages=12)
            try_update(a_lse2, "lse_final")
        
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            t8b = min(40.0, remaining * 0.3)
            if t8b > 5:
                a_cd2, _ = coordinate_descent(best_seq, t8b, max_rounds=100)
                try_update(a_cd2, "cd_final")
    print(f"  Phase 8 done: best_score={best_score:.8f}")

    # === Phase 9: Length exploration ===
    print(f"\n=== Phase 9: Length exploration score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        cur_n = len(best_seq)
        for n_try in [cur_n + 100, cur_n - 100, cur_n + 200]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try < 100:
                continue
            t9 = min(30.0, remaining * 0.3)
            if t9 < 8:
                break
            print(f"  Trying n={n_try}, budget={t9:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = multiplicative_weights_fast(init_try, t9, eta_init=0.1, max_steps=5000)
            try_update(a_try, f"len_{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        # Fallback
        fallback = np.ones(100) / 100.0
        print("FALLBACK to uniform")
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
