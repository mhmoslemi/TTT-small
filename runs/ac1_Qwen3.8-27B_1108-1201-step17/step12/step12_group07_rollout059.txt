def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None, warm_start=None):
    """Solve LP: min M s.t. (a*b)_k <= M, sum(b)=1, b>=0"""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if warm_start is not None and solver == cp.PDLP:
            prob.solve(solver=solver, warm_start=True)
        else:
            prob.solve(solver=solver)
    except Exception:
        # Retry without warm start
        try:
            prob.solve(solver=solver)
        except Exception:
            return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

    else:

def subgradient_nesterov(a0, time_limit, max_steps=200000):
    """Fast subgradient Nesterov using exact convolution gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.3
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.95)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z.copy()

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.001:
                t_nest = 1.0
                eta *= 0.95
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.999, 0.01)

    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def peak_equalize(a0, time_limit, max_steps=20000):
    """Directly try to equalize the top peaks by adjusting a."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.02

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        mean_b = b.mean()
        
        # Find peaks that are above average
        above = b > max_b * 0.85
        if not above.any():
            break
        
        peak_indices = np.where(above)[0]
        
        # Compute which a_i contribute to each peak
        # For peak at position k: (a*a)_k = sum_i a_i * a_{k-i}
        # Gradient w.r.t. a_i from peak k: a_{k-i} for valid i
        grad = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        # Project gradient onto tangent space (sum a = 0)
        S = np.dot(grad, a)
        grad_tangent = grad - S * a
        
        # Update
        a_new = a - eta * grad_tangent
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.1)
        else:
            eta *= 0.9
            
            if eta < 1e-8:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):

def softmax_from_z(z):

def build_conv_matrix(a_fixed):

def solve_lp_l1(a_fixed, solver=None):

def subgradient_nesterov(a0, time_limit, max_steps=200000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.3
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break
        b = fast_convolve(a)
        max_b = b.max()
        topk = np.where(b >= max_b * 0.95)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z.copy()
        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.001:
                t_nest = 1.0
                eta *= 0.95
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.999, 0.01)
    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def peak_shave(a0, time_limit, max_steps=30000):
    """Targeted peak shaving: reduce specific peaks by adjusting contributing a_i."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.01

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Find all peaks above 95% of max
        thresh = max_b * 0.93
        peak_idx = np.where(b >= thresh)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([int(np.argmax(b))])
        # Build gradient: for each peak, identify contributing a_i and push them down
        grad = np.zeros(n)
        for k in peak_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # The contribution to C[k] from a[i] is a[k-i]
            # To reduce C[k], we want to decrease a[i] where a[k-i] is large
            contrib = a[k - idx]
            grad[idx] += contrib
        # Also add a "flatness" term: push up values that are below average
        mean_a = a.mean()
        flatness = (mean_a - a) * 0.1
        grad = grad + flatness
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        # Project to tangent space
        S = np.dot(grad, a)
        grad_t = grad - S * a
        a_new = a - eta * grad_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.05)
        else:
            eta *= 0.92
            if eta < 1e-9:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    best_n = n_base
    t_global_start = time.monotonic()

    # Phase 1: Subgradient Nesterov warm-up (50s)
    print(f"\n=== Phase 1: Subgrad Nesterov n={n_base} score={best_score:.8f} ===")
    t1 = min(50.0, (_global_deadline - time.monotonic()) * 0.06)
    if t1 > 5:
        a_out, _ = subgradient_nesterov(best_seq, t1, max_steps=200000)
        if a_out is not None:
            a_out = np.maximum(a_out, 0)
            s = a_out.sum()
            if s > 0.01:
                a_out = a_out / s
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    best_n = len(a_out)
                    print(f"  NEW BEST [subgrad_p1]: {sc:.8f}")
    print(f"  P1 done: score={best_score:.8f}")

    # Phase 2: Main LP Nesterov (750s) with L1 mixing
    print(f"\n=== Phase 2: Nesterov LP n={best_n} score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(750.0, remaining * 0.5)
    if t2 > 30:
        a_out, _ = nesterov_lp_improved(best_seq, t2, solver=solver, use_l1_mix=True)
        if a_out is not None:
            a_out = np.maximum(a_out, 0)
            s = a_out.sum()
            if s > 0.01:
                a_out = a_out / s
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    best_n = len(a_out)
                    print(f"  NEW BEST [nesterov_lp_p2]: {sc:.8f}")
    print(f"  P2 done: score={best_score:.8f}")

    # Phase 3: Peak shaving (120s)
    print(f"\n=== Phase 3: Peak shave score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(120.0, remaining * 0.12)
    if t3 > 10:
        a_ps, _ = peak_shave(best_seq, t3)
        if a_ps is not None:
            a_ps = np.maximum(a_ps, 0)
            s = a_ps.sum()
            if s > 0.01:
                a_ps = a_ps / s
                sc = evaluate_safe(a_ps)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_ps.copy()
                    best_n = len(a_ps)
                    print(f"  NEW BEST [peak_shave_p3]: {sc:.8f}")
    print(f"  P3 done: score={best_score:.8f}")

    # Phase 4: LSE L-BFGS (300s)
    print(f"\n=== Phase 4: LSE L-BFGS score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(300.0, remaining * 0.25)
    if t4 > 20:
        a_lse = lbgfs_logsumexp(best_seq, t4, T_start=50.0, T_end=0.015, n_stages=40)
        if a_lse is not None:
            a_lse = np.maximum(a_lse, 0)
            s = a_lse.sum()
            if s > 0.01:
                a_lse = a_lse / s
                sc = evaluate_safe(a_lse)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_lse.copy()
                    best_n = len(a_lse)
                    print(f"  NEW BEST [lse_p4]: {sc:.8f}")
    print(f"  P4 done: score={best_score:.8f}")

    # Phase 5: Second LP round (200s)
    print(f"\n=== Phase 5: Nesterov LP r2 score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(200.0, remaining * 0.2)
    if t5 > 15:
        a_out2, _ = nesterov_lp_improved(best_seq, t5, solver=solver, use_l1_mix=False)
        if a_out2 is not None:
            a_out2 = np.maximum(a_out2, 0)
            s = a_out2.sum()
            if s > 0.01:
                a_out2 = a_out2 / s
                sc = evaluate_safe(a_out2)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out2.copy()
                    best_n = len(a_out2)
                    print(f"  NEW BEST [nesterov_lp_p5]: {sc:.8f}")
    print(f"  P5 done: score={best_score:.8f}")

    # Phase 6: Final peak shave + LSE (80s)
    print(f"\n=== Phase 6: Final polish score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        t6a = min(40.0, remaining * 0.4)
        if t6a > 5:
            a_ps2, _ = peak_shave(best_seq, t6a)
            if a_ps2 is not None:
                a_ps2 = np.maximum(a_ps2, 0)
                s = a_ps2.sum()
                if s > 0.01:
                    a_ps2 = a_ps2 / s
                    sc = evaluate_safe(a_ps2)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_ps2.copy()
                        print(f"  NEW BEST [peak_shave_p6]: {sc:.8f}")
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            t6b = min(40.0, remaining * 0.5)
            if t6b > 5:
                a_lse2 = lbgfs_logsumexp(best_seq, t6b, T_start=25.0, T_end=0.008, n_stages=20)
                if a_lse2 is not None:
                    a_lse2 = np.maximum(a_lse2, 0)
                    s = a_lse2.sum()
                    if s > 0.01:
                        a_lse2 = a_lse2 / s
                        sc = evaluate_safe(a_lse2)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            best_seq = a_lse2.copy()
                            print(f"  NEW BEST [lse_p6]: {sc:.8f}")
    print(f"  P6 done: score={best_score:.8f}")

    # Phase 7: Length exploration
    print(f"\n=== Phase 7: Length exploration score={best_score:.8f} ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 80:
        for n_try in [best_n + 150, best_n - 150, best_n + 300, best_n - 300]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 25:
                break
            if n_try < 100 or n_try > 5000:
                continue
            t7 = min(30.0, remaining * 0.3)
            if t7 < 8:
                break
            print(f"  Trying n={n_try}, budget={t7:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_improved(init_try, t7, solver=solver, max_iters=30000)
            if a_try is not None:
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s > 0.01:
                    a_try = a_try / s
                    sc = evaluate_safe(a_try)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_try.copy()
                        best_n = len(a_try)
                        print(f"  NEW BEST [len_{n_try}]: {sc:.8f}")

    # Phase 8: Multi-start (if time remains)
    remaining = _global_deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 8: Multi-start score={best_score:.8f} ===")
        for ms in range(4):
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            t_ms = min(12.0, remaining * 0.25)
            if t_ms < 4:
                break
            init_ms = structured_init(best_n, rng)
            a_ms, _ = subgradient_nesterov(init_ms, t_ms, max_steps=30000)
            if a_ms is not None:
                a_ms = np.maximum(a_ms, 0)
                s = a_ms.sum()
                if s > 0.01:
                    a_ms = a_ms / s
                    sc = evaluate_safe(a_ms)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_ms.copy()
                        print(f"  NEW BEST [multistart_{ms}]: {sc:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    total_time = time.monotonic() - t_global_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.1f}s)")
    return [float(x) for x in best_seq]
