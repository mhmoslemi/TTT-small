"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def topk_lse_obj_grad(a, T, topk):
    """
    Compute top-k LSE objective and gradient w.r.t. a.
    Returns (lse_value, grad_a) where grad_a is the gradient of 
    (1/T)*log(sum_{top_k} exp(T*b_k)) w.r.t. a.
    """
    n = len(a)
    b = fast_autocorr(a)
    k_use = min(topk, len(b))
    top_idx = np.argsort(b)[::-1][:k_use]
    top_vals = b[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    # grad_a[i] = 2 * sum_K w_K * a[K-i] for valid K-i
    # This is 2 * (w-conweighted autocorrelation gradient)
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        if wK < 1e-15:
            continue
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space (log-space) for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = topk_lse_obj_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def cosine_schedule(progress, start, end):
    """Cosine annealing schedule."""
    return end + 0.5 * (start - end) * (1 + np.cos(np.pi * progress))


def adam_topk_lse_v2(a0, time_limit, T_start=200.0, T_end=0.5, n_stages=30, topk=20):
    """

    beta1, beta2, eps = 0.85, 0.999, 1e-9

def peak_shaving_v2(a0, time_limit, seed=123):
    """
    Improved peak shaving: 
    - Uses FFT-based gradient for efficient peak identification
    - Tries to move mass from contributors to the top peak to "safe" positions
    - Uses a more aggressive exploration when stalled
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find all peaks within 0.1% of max (more aggressive than before)
        threshold = bmax * 0.999
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        improved = False
        
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # The peak value is total_contrib; we want to reduce it
            # Find the contributor with the largest individual contribution
            src = idx[np.argmax(contrib)]
            
            # Compute danger: positions that contribute to ANY top peak
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            # Safe destinations
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) > 0:
                # Prefer positions with low a value
                safe_sorted = safe[np.argsort(a[safe])]
                dst_pool = safe_sorted[:max(2, len(safe_sorted) // 4)]
            else:
                dst_pool = np.array([j for j in range(n) if j != src])

            if len(dst_pool) == 0:
                continue

            # Try the best destination with multiple step sizes
            best_frac = None
            best_trial = None
            best_trial_score = best_score
            
            for dst in dst_pool[:5]:
                for frac in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_trial_score:
                        best_trial_score = score
                        best_trial = a_trial
                        best_frac = frac
                        best_dst = dst

            if best_trial is not None and best_trial_score < best_score - 1e-15:
                a = best_trial
                best_score = best_trial_score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random exploration
            n_pert = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > max_stall:
                break

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    try:
        import cvxpy as cp
        for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
    except ImportError:
        pass
    return None


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_quick(a0, time_limit, solver, max_iters=50000):
    """
    Quick Nesterov LP for initialization. Fewer iterations but still effective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng):

        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)

        a = (1 + np.cos(2 * np.pi * x)) * (1 + 0.5 * np.cos(4 * np.pi * x))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def topk_lse_obj_grad(a, T, topk):

        wK = w[idx]
        if wK < 1e-15:
            continue
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space (log-space) for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = topk_lse_obj_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse_v2(a0, time_limit, T_start=200.0, T_end=0.5, n_stages=30, topk=20):
    """
    Improved Adam-optimized top-k LSE with:
    - Geometric temperature annealing
    - Adaptive top-k
    - Gradient clipping
    - Periodic restart from best
    - More aggressive learning in early stages
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    beta1, beta2, eps = 0.85, 0.999, 1e-9
    max_grad_norm = 5.0

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        progress = si / max(1, n_stages - 1)
        stage_T = T_start * (T_end / T_start) ** progress

        base_lr = 0.03 * min(1.0, 500.0 / n)
        lr = base_lr * (0.88 ** si)

        k_stage = max(5, int(topk * (0.4 + 0.6 * progress)))

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0

        if progress < 0.3:
            max_it = int(min(remaining, 80.0) * 10)
        elif progress < 0.7:
            max_it = int(min(remaining, 50.0) * 6)
        else:
            max_it = int(min(remaining, 40.0) * 4)
        max_it = max(max_it, 30)

        stall_count = 0
        best_a_stage = best_a.copy()
        best_score_stage = best_score

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, k_stage)

            grad_norm = np.linalg.norm(grad_z)
            if grad_norm > max_grad_norm:
                grad_z = grad_z * (max_grad_norm / grad_norm)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 4 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score_stage:
                    best_score_stage = score
                    best_a_stage = a.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                    if stall_count % 30 == 0 and stall_count > 60:
                        z = np.log(np.maximum(best_a_stage, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0
                        stall_count = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        if best_score_stage < best_score:
            best_score = best_score_stage
            best_a = best_a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSEv2 stage {si}/{n_stages} (T={stage_T:.1f}, k={k_stage}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lcgd_optimize(a0, time_limit):
    """
    Locally Coupled Gradient Descent in primal space.
    Uses smoothed max with projection onto simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    T = 80.0
    step = 0.02
    decay = 0.997

    for it in range(int(time_limit * 25)):
        if time.monotonic() - t0 > time_limit - 3:
            break

        b = fast_autocorr(a)
        Tc = T * b
        Tc_max = Tc.max()
        exp_Tc = np.exp(Tc - Tc_max)
        w = exp_Tc / exp_Tc.sum()

        grad = np.zeros(n)
        top_k = 40
        top_idx = np.argsort(w)[::-1][:top_k]
        for K in top_idx:
            wK = w[K]
            if wK < 1e-12:
                continue
            js = max(0, K - (n - 1))
            je = min(n - 1, K)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                grad[idx_range] += 2.0 * wK * a[K - idx_range]

        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = a.copy()
        else:
            a_new = a.copy()

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        elif it % 50 == 0:
            a_pert = a + rng.normal(0, 0.002, n) * (a + 1e-6)
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-10:
                a_pert = a_pert / s_p
                sc_p = 2.0 * n * fast_autocorr(a_pert).max()
                if sc_p < best_score:
                    best_score = sc_p
                    best_a = a_pert.copy()
                    a = a_pert

        step *= decay
        if it % 100 == 0:
            T *= 0.97

    return best_a, best_score


def peak_shaving_v2(a0, time_limit, seed=123):
    """
    Improved peak shaving with more aggressive exploration.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 600

    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.999
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        improved = False

        for k in top_peaks[:12]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.sum() < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:12]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) > 0:
                safe_sorted = safe[np.argsort(a[safe])]
                dst_pool = safe_sorted[:max(3, len(safe_sorted) // 3)]
            else:
                dst_pool = np.array([j for j in range(n) if j != src])

            if len(dst_pool) == 0:
                continue

            best_trial = None
            best_trial_score = best_score

            for dst in dst_pool[:6]:
                for frac in [0.0003, 0.001, 0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_trial_score:
                        best_trial_score = score
                        best_trial = a_trial

            if best_trial is not None and best_trial_score < best_score - 1e-15:
                a = best_trial
                best_score = best_trial_score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            n_pert = rng.integers(1, 6)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.04)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > max_stall:
                break

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    try:
        import cvxpy as cp
        for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
    except ImportError:
        pass
    return None


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_quick(a0, time_limit, solver, max_iters=30000):
    """Quick Nesterov LP for initialization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x)) * (1 + 0.5 * np.cos(4 * np.pi * x))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_scale_optimize(a_short, n_target, time_limit):
    """Optimize at short length, then upsample to target."""
    a_opt_short, _ = adam_topk_lse_v2(a_short, time_limit * 0.4,
                                       T_start=50.0, T_end=1.0, n_stages=12, topk=10)
    n_short = len(a_opt_short)
    x_old = np.linspace(0, 1, n_short)
    x_new = np.linspace(0, 1, n_target)
    a_interp = np.interp(x_new, x_old, a_opt_short)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp = a_interp / s
    return a_interp


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    Phase 1: LP Nesterov initialization (if solver available)
    Phase 2: Main aggressive LSE optimization (largest time budget)
    Phase 3: LSE variant with perturbed start
    Phase 4: LCGD complementary optimization
    Phase 5: Multi-scale approach
    Phase 6: Multi-start diversity
    Phase 7-9: Peak shaving and final polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 150.0, 1650.0)

    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Helper logic inlined (no closures)
    def _do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")
    if solver is not None and remaining > 120:
        print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
        t1 = min(remaining * 0.12, 180.0)
        if t1 > 30:
            a_out, _ = nesterov_lp_quick(best_seq, t1, solver)
            _do_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Main LSE (heaviest) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Top-k LSE n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.38, 500.0)
    if t2 > 30:
        a_out, _ = adam_topk_lse_v2(best_seq, t2,
                                     T_start=400.0, T_end=0.2,
                                     n_stages=35, topk=30)
        _do_update(a_out, f"lse_main_n{n_base}")

    # === Phase 3: LSE variant (perturbed start, different params) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LSE variant n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.14, 180.0)
    if t3 > 20:
        a_pert = best_seq + rng.normal(0, 0.025, n_base) * (best_seq + 1e-6)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_out, _ = adam_topk_lse_v2(a_pert, t3,
                                     T_start=150.0, T_end=0.5,
                                     n_stages=22, topk=20)
        _do_update(a_out, f"lse_var_n{n_base}")

    # === Phase 4: LCGD ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LCGD n={n_base} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.10, 100.0)
    if t4 > 15:
        a_out, _ = lcgd_optimize(best_seq, t4)
        _do_update(a_out, f"lcgd_n{n_base}")

    # === Phase 5: Multi-scale ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        n_short = max(200, n_base // 4)
        print(f"\n=== Phase 5: Multi-scale {n_short}->{n_base} ({remaining:.0f}s left) ===")
        a_short = np.interp(np.linspace(0, 1, n_short),
                            np.linspace(0, 1, n_base), best_seq)
        a_short = a_short / (a_short.sum() + 1e-30)
        t5 = min(remaining * 0.12, 120.0)
        if t5 > 20:
            a_long, _ = multi_scale_optimize(a_short, n_base, t5)
            _do_update(a_long, f"ms_{n_short}_{n_base}")

    # === Phase 6: Multi-start ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        print(f"\n=== Phase 6: Multi-start ({remaining:.0f}s left) ===")
        t6_each = min(remaining * 0.06, 50.0)
        for si in range(3):
            if time.monotonic() - t_global_start > time_budget - 80:
                break
            a_struct = structured_init(n_base, rng)
            blend = 0.25 * a_struct + 0.75 * best_seq
            blend = blend / (blend.sum() + 1e-30)
            a_out, _ = adam_topk_lse_v2(blend, t6_each,
                                         T_start=80.0, T_end=0.5,
                                         n_stages=15, topk=15)
            _do_update(a_out, f"ms_{si}")

    # === Phase 7: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Peak shaving ({remaining:.0f}s left) ===")
    t7 = min(remaining * 0.14, 140.0)
    if t7 > 10:
        a_out, _ = peak_shaving_v2(best_seq, t7, seed=456)
        _do_update(a_out, "peak_shave_1")

    # === Phase 8: Final LSE polish (low T) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8 = min(remaining * 0.18, 100.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final LSE polish ({remaining:.0f}s left) ===")
            a_out, _ = adam_topk_lse_v2(best_seq, t8,
                                         T_start=15.0, T_end=0.05,
                                         n_stages=12, topk=8)
            _do_update(a_out, "final_lse")

    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t9 = min(remaining * 0.3, 50.0)
        if t9 > 5:
            print(f"\n=== Phase 9: Final peak shave ({remaining:.0f}s left) ===")
            a_out, _ = peak_shaving_v2(best_seq, t9, seed=789)
            _do_update(a_out, "peak_shave_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
