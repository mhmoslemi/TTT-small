import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_conv(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s * s)

def spectral_smooth(a, sigma=0.5):
    """Apply Gaussian smoothing in frequency domain."""
    n = len(a)

def subgradient_step(a, k_peaks, weights, eta):
    """
    Compute subgradient and apply multiplicative (mirror) update.
    k_peaks: indices of top peaks in convolution
    weights: weights for each peak
    eta: step size
    """
    n = len(a)
    grad = np.zeros(n)
    for m, w in zip(k_peaks, weights):
        m = int(m)
        # grad[j] += 2 * w * a[m-j] for 0 <= m-j < n
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        js = np.arange(j_start, j_end + 1)
        grad[js] += 2.0 * w * a[m - js]
    
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def subgradient_step_euclidean(a, k_peaks, weights, eta):
    """Additive (Euclidean) update."""
    n = len(a)
    grad = np.zeros(n)
    for m, w in zip(k_peaks, weights):
        m = int(m)
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        js = np.arange(j_start, j_end + 1)
        grad[js] += 2.0 * w * a[m - js]
    
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

        w_shift = (peak_vals - peak_vals.max()) / temp
        weights = np.exp(w_shift)
        weights = weights / weights.sum()
        
        # Choose update type
        if use_mirror:
            a_new = subgradient_step(a, top_idx, weights, eta)
        else:
            a_new = subgradient_step_euclidean(a, top_idx, weights, eta)

                trial = subgradient_step_euclidean(a, top_idx, weights, eta_trial)
            trial_score = fast_score(trial)

                best_trial = trial

def init_gap(n, gap_frac=0.15, rng=None):
    """Initialize with two-blob structure with a gap in the middle."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    gap = int(n * gap_frac)
    n_each = (n - gap) // 2
    
    a = np.zeros(n)
    # Two raised-cosine blobs
    x1 = np.linspace(0, 1, n_each)
    a[:n_each] = 0.5 * (1 + np.cos(2 * np.pi * x1))
    
    x2 = np.linspace(0, 1, n - gap - n_each)
    a[n - (n - gap - n_each):] = 0.5 * (1 + np.cos(2 * np.pi * x2))
    
    # Add small noise
    a += 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

def init_from_parent(parent, n_target, gap_frac=0.1):
    """Create initialization by combining parent with gap structure."""
    n_p = len(parent)
    # Interpolate parent to target length
    x_old = np.linspace(0, 1, n_p)
    x_new = np.linspace(0, 1, n_target)
    p_interp = np.interp(x_new, x_old, parent)
    
    # Apply gap mask
    gap = int(n_target * gap_frac)
    center = n_target // 2
    gap_start = center - gap // 2
    gap_end = gap_start + gap
    
    a = p_interp.copy()
    # Reduce values in gap region
    a[gap_start:gap_end] *= 0.1
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

2. Both multiplicative (mirror) and additive (Euclidean) updates, selecting the better

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_conv(a):
    """Full linear convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def compute_subgradient(a, top_idx, weights):
    """
    Compute subgradient of max_k conv(a,a)[k] at the given peaks.
    top_idx: array of peak indices
    weights: array of weights (should sum to 1)
    Returns gradient array of same length as a.
    """
    n = len(a)
    grad = np.zeros(n)
    for m, w in zip(top_idx, weights):
        m = int(m)
        w = float(w)
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        grad[js] += 2.0 * w * a[m - js]
    return grad


def mirror_update(a, grad, eta):
    """Multiplicative (mirror) update on simplex."""
    log_step = -eta * grad
    log_step = np.clip(log_step, -10, 10)
    a_new = a * np.exp(log_step)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a + 1e-6
    return a_new / s


def euclidean_update(a, grad, eta):
    """Additive (Euclidean) update with projection."""
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a + 1e-6
    return a_new / s

def init_gap(n, gap_frac=0.12, rng=None):

def init_from_parent(parent, n_target, gap_frac=0.08):

    a[gap_start:gap_end] *= 0.3

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

def spectral_nudge(a, rng, strength=0.02):

        a_out, _ = optimize_subgradient(best_seq, t2, eta0=0.001, k_start=5, k_end=35,
                                        use_mirror=False, seed=3, verbose=True)
        try_update(a_out, "eucl_subgrad")

    # Phase 3: Local moves
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Local Moves ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 10:
        a_out, _ = local_moves(best_seq, t3, seed=99, verbose=True)
        try_update(a_out, "local_base")

    # Phase 4: Gap init + subgradient (different structure)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Gap Init + Subgradient ===")
    t4 = min(remaining * 0.10, 140.0)
    if t4 > 20:
        for gap_f in [0.10, 0.15]:
            remaining = time_budget - (time.monotonic() - t_global)
            t4s = min(remaining * 0.45, 80.0)
            if t4s < 15:
                break
            a_gap = init_gap(n_base, gap_f, rng)
            a_out, _ = optimize_subgradient(a_gap, t4s, eta0=0.04, k_start=3, k_end=20,
                                            use_mirror=True, seed=int(gap_f * 100), verbose=False)
            try_update(a_out, f"gap_{gap_f}")

    # Phase 5: U/V shaped init
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Alternative Inits ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 20:
        for init_fn, label in [(init_u_shaped, "u_shaped"), (init_v_shaped, "v_shaped")]:
            remaining = time_budget - (time.monotonic() - t_global)
            t5s = min(remaining * 0.45, 60.0)
            if t5s < 10:
                break
            a_init = init_fn(n_base, rng)
            a_out, _ = optimize_subgradient(a_init, t5s, eta0=0.04, k_start=3, k_end=20,
                                            use_mirror=True, seed=hash(label) % 100, verbose=False)
            try_update(a_out, label)

    # Phase 6: Longer length n=2000
    n_long = 2000
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Longer length n={n_long} ===")
    if best_seq is not None and remaining > 80:
        a_long = init_from_parent(best_seq, n_long, gap_frac=0.08)
        t6 = min(remaining * 0.18, 250.0)
        if t6 > 20:
            a_out, _ = optimize_subgradient(a_long, t6, eta0=0.03, k_start=3, k_end=20,
                                            use_mirror=True, seed=7, verbose=True)
            try_update(a_out, f"subgrad_n{n_long}")

        # Local at longer length
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.07, 80.0)
        if t6b > 10 and best_seq is not None:
            a_out, _ = local_moves(best_seq, t6b, seed=77, verbose=False)
            try_update(a_out, f"local_n{len(best_seq)}")

    # Phase 7: Spectral nudges + subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Spectral Nudges + Polish ===")
    if best_seq is not None and remaining > 40:
        a_current = best_seq.copy()
        n_nudges = max(3, int(remaining / 30))
        for ni in range(n_nudges):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 10:
                break
            a_nudged = spectral_nudge(a_current, rng, strength=0.03)
            t_nudge = min(remaining * 0.3, 30.0)
            if t_nudge < 5:
                break
            a_out, _ = optimize_subgradient(a_nudged, t_nudge, eta0=0.02, k_start=5, k_end=20,
                                            use_mirror=True, seed=100 + ni, verbose=False)
            if fast_score(a_out) < fast_score(a_current):
                a_current = a_out
            try_update(a_out, f"nudge_{ni}")

    # Phase 8: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None and remaining > 20:
        # Short subgradient with high k
        t8 = min(remaining * 0.35, 80.0)
        if t8 > 10:
            a_out, _ = optimize_subgradient(best_seq, t8, eta0=0.008, k_start=10, k_end=50,
                                            use_mirror=True, seed=55, verbose=True)
            try_update(a_out, "final_subgrad")

        # Final local moves
        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.55, 100.0)
        if t8b > 10:
            a_out, _ = local_moves(best_seq, t8b, seed=444, verbose=True)
            try_update(a_out, "final_local")

        # One more subgradient after local
        remaining = time_budget - (time.monotonic() - t_global)
        t8c = min(remaining * 0.3, 40.0)
        if t8c > 5:
            a_out, _ = optimize_subgradient(best_seq, t8c, eta0=0.005, k_start=15, k_end=40,
                                            use_mirror=True, seed=66, verbose=False)
            try_update(a_out, "final_subgrad2")

    # Final evaluation with official function
    if best_seq is not None:
        try:
            fs = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(fs):
                best_score = min(best_score, fs)
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
