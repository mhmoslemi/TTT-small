Actually, I want to add one more idea: **FISTA (Fast Iterative Shrinkage-Thresholding Algorithm)** style optimization.

The objective is f(a) = 2n * max_k (a*a)[k] (assuming sum(a) = 1).

We can write this as:
f(a) = max_k g_k(a) where g_k(a) = 2n * sum_i a[i]*a[k-i]

Each g_k is a quadratic form: g_k(a) = 2n * a^T C_k a where C_k is a symmetric matrix.

The subgradient of f at a is in the convex hull of {C_k a : k in argmax f}.

FISTA would be:
1. y = a + momentum * (a - a_prev)
2. Compute subgradient g of f at y
3. a_new = proj_simplex(y - step * g) (project onto the simplex: a >= 0, sum(a) = 1)
4. Update momentum

The projection onto the simplex can be done in O(n log n) time.

But computing the subgradient requires computing the convolution, which is O(n log n) with FFT. So each iteration is O(n log n).

For n=1200, this is very fast. We can do hundreds of thousands of iterations.

However, the subgradient method converges slowly (O(1/sqrt(T))). FISTA improves this to O(1/T), but it's still slow for non-smooth objectives.

The LP-based approach converges faster because it uses the exact best response (not just a subgradient). So I'll stick with LP-based Nesterov as the main approach.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Nesterov acceleration,

random perturbation restarts, and aggressive low-temperature polishing.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_global_deadline = 0.0

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def solve_lp_best_response(a_fixed, b_init=None, solver=None):
    """Solve the LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    # Warm start
    if b_init is not None:
        b_var.value = b_init
        M_val = float((T @ b_init).max())
        M_var.value = M_val
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)
