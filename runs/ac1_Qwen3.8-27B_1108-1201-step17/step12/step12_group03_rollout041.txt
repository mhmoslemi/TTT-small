"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast soft-max gradient descent with multiplicative weights (mirror descent) on simplex
2. Nesterov-accelerated LP best-response for refinement
3. Peak-shaving coordinate descent for final polish
4. Multi-length exploration
Key innovation: FFT-based soft-max gradient is O(n log n) per step vs O(n^2) for LP,
enabling 100x more iterations in the same time budget.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution using FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_grad_softmax(a, epsilon):
    """
    Compute the soft-max gradient of max_k conv(a,a)_k.
    Returns (grad, b, max_b, p) where grad is the gradient vector of length n.
    grad[m] = 2 * sum_j a[j] * p[m+j] for m=0..n-1
    Computed via FFT as 2 * conv(a_rev, p)[n-1 : 2n-1]
    """
    n = len(a)
    b = fft_convolve(a, a)  # length 2n-1
    max_b = b.max()
    
    # Soft-max weights (numerically stable)
    b_scaled = b / epsilon
    b_shifted = b_scaled - b_scaled.max()
    exp_b = np.exp(b_shifted)
    p = exp_b / exp_b.sum()  # length 2n-1
    
    # Gradient via: grad = 2 * conv(a[::-1], p)[n-1 : 2*n-1]
    a_rev = a[::-1]
    conv_ap = fft_convolve(a_rev, p)  # length n + (2n-1) - 1 = 3n-2
    grad = 2.0 * conv_ap[n - 1:2 * n - 1]  # length n
    
    return grad, b, max_b, p


def soft_max_mirror_descent(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated soft-max mirror descent (multiplicative weights) on the simplex.
    Minimizes max_k conv(a,a)_k subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a, a).max()
    t0 = time.monotonic()
    
    # Nesterov state
    a_prev = a.copy()
    
    # Adaptive parameters
    b_init = fft_convolve(a, a)
    max_b_init = b_init.max()
    eps = max_b_init * 0.5  # Start with moderate temperature
    eps_decay = 0.9995  # Per-step decay factor
    lr = 0.1  # Initial learning rate
    lr_decay = 0.9999
    
    iter_count = 0
    stall = 0
    stall_patience = 5000
    
    # Track for Nesterov momentum
    y = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0.0)
            y_s = y.sum()
            if y_s < 1e-15:
                y = a.copy()
                y_s = 1.0
            y = y / y_s
        else:
            y = a.copy()
        
        # Compute gradient at y
        grad, b, max_b, p = compute_grad_softmax(y, eps)
        
        # Mirror descent step: multiplicative weights
        # a_new = y * exp(-lr * grad) / Z
        log_update = -lr * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -50.0, 50.0)
        a_new = y * np.exp(log_update)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            a_new = y * (1.0 - lr * grad)
            a_new = np.maximum(a_new, 0.0)
            a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s
        
        # Score
        score = 2.0 * n * fft_convolve(a_new, a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a_prev = a
        a = a_new
        iter_count += 1
        
        # Adaptive parameters
        eps *= eps_decay
        eps = max(eps, max_b_init * 0.001)  # Floor for epsilon
        lr *= lr_decay
        lr = max(lr, 0.001)
        
        # If stalling, boost learning rate temporarily
        if stall > stall_patience and stall % stall_patience == 0:
            lr = min(lr * 2.0, 0.5)
            eps = max(eps * 2.0, max_b_init * 0.1)
            if verbose:
                print(f"      SMD: stall at iter {iter_count}, boosting lr={lr:.4f}, eps={eps:.6f}")
        
        if stall > stall_patience * 3:
            if verbose:
                print(f"      SMD: early stop at iter {iter_count}")
            break
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SMD iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}, max_b={max_b:.6f}, eps={eps:.4f}, lr={lr:.4f}")
    
    return best_a, best_score, iter_count


def nesterov_lp_optimize(a0, time_limit, verbose=True, max_iters=500):
    """
    LP-based Nesterov best-response iteration.
    min M s.t. T(a_fixed) @ b <= M, sum(b)=1, b>=0
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    y = a.copy()
    a_prev = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a, a).max()
    t0 = time.monotonic()
    
    # Build constraint matrix once
    # T has shape (2n-1, n), T[k,j] = a_fixed[k-j] if valid
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = j + n
        T[start:end, j] = a  # This will be updated each iteration
    
    solver = None
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = solver_name
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    iter_count = 0
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: min M s.t. T @ b_var <= M, sum(b_var)=1, b_var >= 0
        # T depends on y (the current "fixed" sequence)
        T_new = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_new[j:j+n, j] = y
        
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_new @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            prob.solve(solver=cp.SCS)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_star = np.asarray(b_var.value, dtype=np.float64)
        b_star = np.maximum(b_star, 0)
        b_star_s = b_star.sum()
        if b_star_s < 1e-15:
            continue
        b_star = b_star / b_star_s
        M_val = float(M_var.value)
        iter_count += 1
        
        # Nesterov update
        if t == 1:
            a_new = b_star.copy()
            y_new = b_star.copy()
        else:
            alpha_t = 2.0 / (t + 2.0)
            y_new = (1 - alpha_t) * y + alpha_t * b_star
            y_new = np.maximum(y_new, 0)
            y_new_s = y_new.sum()
            if y_new_s < 1e-15:
                y_new = y.copy()
                y_new_s = 1.0
            y_new = y_new / y_new_s
            
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_new_s = a_new.sum()
            if a_new_s < 1e-15:
                a_new = y_new.copy()
                a_new_s = 1.0
            a_new = a_new / a_new_s
        
        y = y_new
        a_prev = a
        a = a_new
        
        # Score
        score_a = 2.0 * n * fft_convolve(a, a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fft_convolve(y, y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
    
    return best_a, best_score, iter_count


def peak_shaving_refine(a0, time_limit, seed=123):
    """
    Coordinate descent peak shaving: identify top peaks, reduce contributing elements,
    redistribute mass to low-contribution positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a, a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fft_convolve(a, a)
        max_b = b.max()
        
        # Find top peaks
        n_peaks = min(5, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        for k in top_peaks:
            # Elements contributing to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.sum() < 1e-15:
                continue
            
            # Find the element with highest contribution
            ci = np.argmax(contrib)
            src = idx[ci]
            src_other = k - src
            
            # Find best target: position that minimizes impact on top peaks
            # Score each position by its total contribution to top peaks
            contrib_to_peaks = np.zeros(n)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                contrib_to_peaks[idx2] += a[idx2] * a[k2 - idx2]
                contrib_to_peaks[k2 - idx2] += a[idx2] * a[k2 - idx2]
            
            # Choose target with minimum contribution to peaks
            # Avoid the source positions
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            if src_other != src:
                mask[src_other] = False
            
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                continue
            
            # Pick the few with lowest peak contribution
            c_scores = contrib_to_peaks[candidates]
            # Also prefer positions with currently low value (less risk of creating new peaks)
            c_scores += 0.1 * a[candidates] * n
            top_targets = candidates[np.argsort(c_scores)[:10]]
            
            for dst in top_targets:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if src != src_other:
                        amt = min(amt, a[src_other] * frac)
                    if amt < 1e-15:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    if src != src_other:
                        a_trial[src_other] -= amt * 0.5
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    
                    score = 2.0 * n * fft_convolve(a_trial, a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation every 50 stalls
            if stall % 50 == 0:
                n_pert = rng.integers(2, 6)
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for src, dst in zip(sources, targets):
                    frac = rng.uniform(0.002, 0.01)
                    amt = a_trial[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s > 1e-15:
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fft_convolve(a_trial, a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 2000:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    if best_seq is None:
        # Create a ramp initialization
        n_base = 1600
        x = np.arange(n_base, dtype=np.float64)
        best_seq = x + 1.0  # Ramp
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Ramp init: n={n_base}, score={best_score:.6f}")
    
    # Phase 1: Soft-max mirror descent (primary workhorse - 55% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Soft-max Mirror Descent at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = remaining * 0.55
    if t1 > 30:
        a_out, sc_out, iters = soft_max_mirror_descent(best_seq, t1, verbose=True)
        print(f"  SMD done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"smd_n{n_base}")
    
    # Phase 1b: Continue with lower epsilon for sharper peak targeting
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        print(f"\n=== Phase 1b: SMD round 2 (lower eps) (remaining={remaining:.0f}s) ===")
        t1b = remaining * 0.15
        if t1b > 30:
            a_out, sc_out, iters = soft_max_mirror_descent(best_seq, t1b, verbose=False)
            print(f"  SMD2 done: {iters} iters, score={sc_out:.6f}")
            try_update(a_out, f"smd2_n{n_base}")
    
    # Phase 2: LP Nesterov refinement (20% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = remaining * 0.25
    if t2 > 30:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t2, verbose=True)
        print(f"  LP done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")
    
    # Phase 3: Another SMD round after LP (exploit LP improvement)
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        print(f"\n=== Phase 3: SMD round 3 (remaining={remaining:.0f}s) ===")
        t3 = remaining * 0.2
        if t3 > 20:
            a_out, sc_out, iters = soft_max_mirror_descent(best_seq, t3, verbose=False)
            print(f"  SMD3 done: {iters} iters, score={sc_out:.6f}")
            try_update(a_out, f"smd3_n{n_base}")
    
    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak Shaving (remaining={remaining:.0f}s) ===")
    t4 = remaining * 0.25
    if t4 > 10:
        a_out, sc_out = peak_shaving_refine(best_seq, t4, seed=456)
        print(f"  Peak shave: score={sc_out:.6f}")
        try_update(a_out, "peak_shave")
    
    # Phase 5: Try nearby lengths
    for n_try in [n_base + 200, n_base - 200, n_base + 400]:
        if n_try < 500:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 90:
            break
        print(f"\n=== Phase 5: SMD at n={n_try} (remaining={remaining:.0f}s) ===")
        t5 = remaining * 0.25
        if t5 < 30:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = soft_max_mirror_descent(init_try, t5, verbose=False)
        print(f"  SMD n={n_try}: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"smd_n{n_try}")
        
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 30:
            a_out2, sc_out2, _ = nesterov_lp_optimize(best_seq, min(remaining * 0.3, 60.0), verbose=False)
            try_update(a_out2, f"lp_n{n_try}")
    
    # Phase 6: Final SMD + peak shave on global best
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"\n=== Phase 6: Final refinement (remaining={remaining:.0f}s) ===")
        t6 = remaining * 0.4
        if t6 > 15:
            a_out, sc_out, _ = soft_max_mirror_descent(best_seq, t6, verbose=False)
            try_update(a_out, "final_smd")
        
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 10:
            a_out, sc_out = peak_shaving_refine(best_seq, remaining * 0.6, seed=789)
            try_update(a_out, "final_peak")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 1600
    
    return [float(x) for x in best_seq]
