The dual variable $y_k$ tells us the "weight" of constraint $k$. At the optimum, the complementary slackness says:
- If $(Tb)_k < M$, then $y_k = 0$
- If $b_j > 0$, then $\sum_k T_{kj} y_k + \nu = 0$

So the "active" constraints are those with $y_k > 0$, which is a subset of the constraints where $(Tb)_k = M$.

2. Computing a "weighted" subgradient for the outer optimization

    if s < 1e-15:
        return None, None, None
    b_val = b_val / s

    dual_vals = None
    try:
        if active_indices is not None:
            # Can't easily get partial duals, skip
            dual_vals = None
        else:
            # The dual of the inequality T@b <= M*1 gives us y
            pass
    except Exception:
        dual_vals = None

    return b_val, float(M_var.value), dual_vals


def multiplicative_weights_opt(a0, time_limit, eta_init=0.1, decay=0.9995, max_steps=500000):
    """Fast multiplicative weights without LP."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_init
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 10000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a
            a = new_a
            last_improve = step
            eta = min(eta * 1.001, 0.2)
        else:
            eta *= decay
            if eta < 1e-5:
                break

    return best_a, best_score


def nesterov_lp_opt(a0, time_limit, solver, max_iters=500000, use_reduced_lp=False, top_k=300):
    """LP-based Nesterov with adaptive restart and optional reduced LP."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    score_history = []

    full_lp_interval = 200 if use_reduced_lp else 1

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Determine which constraints to use
        if use_reduced_lp:
            b_curr = fast_convolve(a)
            K = min(top_k, len(b_curr))
            active_indices = np.argsort(b_curr)[-K:]
            active_indices = np.sort(active_indices)
        else:
            active_indices = None

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp, _ = solve_lp_best_response(y, solver, active_indices)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        score_history.append(sc_new)
        if len(score_history) > 50:
            score_history.pop(0)

        # Adaptive restart: check if we're making progress
        if len(score_history) >= 30:
            recent_avg = np.mean(score_history[-10:])
            older_avg = np.mean(score_history[:10])
            if recent_avg > older_avg * 1.001 and t_nest > 10.0:
                t_nest = 1.0
                a_prev = a_new.copy()
                restart_count += 1
            else:
                a_prev = a.copy()
                t_nest = t_new

            # Additional restart if stuck
            if t_nest > 50.0 and best_score - sc_new > 0.001:
                t_nest = 1.0
                a_prev = a_new.copy()
                restart_count += 1
        else:
            if sc_new > score_normalized(a) * 1.001 and t_nest > 5.0:
                t_nest = 1.0
                a_prev = a_new.copy()
                restart_count += 1
            else:
                a_prev = a.copy()
                t_nest = t_new

        a = a_new

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
        Tb = np.clip(Tb, None, max_Tb)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs_opt(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=30):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        if si < n_stages * 0.5:
            iters = 30
        elif si < n_stages * 0.8:
            iters = 20
        else:
            iters = 15

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-16, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: best={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def peak_shave_opt(a0, time_limit, top_k_frac=0.05, eta_init=0.03, max_steps=100000):
    """Peak-shaving local search with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_init
    last_improve = 0

    K = max(5, int(top_k_frac * (2 * n - 1)))

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        sorted_idx = np.argsort(b)[-K:]

        grad = np.zeros(n)
        for k in sorted_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= K

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a
            a = new_a
            last_improve = step
            eta = min(eta * 1.003, 0.1)
        else:
            eta *= 0.90
            if eta < 1e-8:
                break

    return best_a, best_score


def nesterov_lp_no_extrap(a0, time_limit, solver, max_iters=200000, top_k=200):
    """Simpler Nesterov-LP without extrapolation, with constraint generation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    alpha = 1.0 / 5.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_curr = fast_convolve(a)
        K = min(top_k, len(b_curr))
        active_indices = np.sort(np.argsort(b_curr)[-K:])

        b_new, M_lp, _ = solve_lp_best_response(a, solver, active_indices)
        if b_new is None:
            continue

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.002, 0.3)
        else:

                break

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.5 * (2.0 * x - 1.0)
    elif choice == 1:
        a = np.exp(-3.0 * x) + 0.5 * np.exp(-3.0 * (1.0 - x)) + 0.1
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.abs(a) + 0.1
    elif choice == 3:
        a = np.exp(-5.0 * (x - 0.3)**2) + 0.7 * np.exp(-5.0 * (x - 0.7)**2) + 0.05
    elif choice == 4:
        a = (x + 0.1) ** 0.3 * ((1.0 - x) + 0.1) ** 0.3
    else:
        a = 1.0 + 0.3 * np.sin(3.0 * np.pi * x)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {budget_s:.0f}s, deadline set to {_global_deadline - time.monotonic():.0f}s from now")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            elapsed = time.monotonic()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)}) at t={elapsed - (elapsed - elapsed):.0f}s")

    # === Phase 1: Fast multiplicative weights (80s) ===
    remaining = _global_deadline - time.monotonic()
    t1 = min(80.0, remaining * 0.05)
    if t1 > 10:
        print(f"\n=== Phase 1: Multiplicative weights ({t1:.0f}s) ===")
        a_out, _ = multiplicative_weights_opt(best_seq, t1, eta_init=0.08)
        try_update(a_out, "MW_p1")
    print(f"  After Phase 1: best={best_score:.8f}")

    # === Phase 2: LP-based Nesterov main (600s) ===
    remaining = _global_deadline - time.monotonic()
    t2 = min(600.0, remaining * 0.4)
    if t2 > 30:
        print(f"\n=== Phase 2: Nesterov LP main ({t2:.0f}s) ===")
        a_out, _ = nesterov_lp_opt(best_seq, t2, solver, max_iters=500000, use_reduced_lp=True, top_k=300)
        try_update(a_out, "NLP_p2")
    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: LSE L-BFGS polish (280s) ===
    remaining = _global_deadline - time.monotonic()
    t3 = min(280.0, remaining * 0.18)
    if t3 > 20:
        print(f"\n=== Phase 3: LSE L-BFGS ({t3:.0f}s) ===")
        a_lse = lse_lbgfs_opt(best_seq, t3, T_start=40.0, T_end=0.01, n_stages=30)
        try_update(a_lse, "LSE_p3")
    print(f"  After Phase 3: best={best_score:.8f}")

    # === Phase 4: Second LP-based Nesterov (350s) ===
    remaining = _global_deadline - time.monotonic()
    t4 = min(350.0, remaining * 0.22)
    if t4 > 30:
        print(f"\n=== Phase 4: Nesterov LP round 2 ({t4:.0f}s) ===")
        a_out2, _ = nesterov_lp_opt(best_seq, t4, solver, max_iters=500000, use_reduced_lp=True, top_k=250)
        try_update(a_out2, "NLP_p4")
    print(f"  After Phase 4: best={best_score:.8f}")

    # === Phase 5: Peak shaving + final LSE (200s) ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        t5a = min(100.0, remaining * 0.35)
        t5b = min(100.0, remaining * 0.35)
        if t5a > 10:
            print(f"\n=== Phase 5a: Peak shaving ({t5a:.0f}s) ===")
            a_ps, _ = peak_shave_opt(best_seq, t5a, top_k_frac=0.03, eta_init=0.02)
            try_update(a_ps, "PS_p5")
        remaining = _global_deadline - time.monotonic()
        if t5b > 10 and remaining > 10:
            print(f"=== Phase 5b: Final LSE ({t5b:.0f}s) ===")
            a_lse2 = lse_lbgfs_opt(best_seq, t5b, T_start=15.0, T_end=0.005, n_stages=20)
            try_update(a_lse2, "LSE_p5")
    print(f"  After Phase 5: best={best_score:.8f}")

    # === Phase 6: Perturbation experiments (remaining time) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 6: Perturbations (budget={remaining:.0f}s) ===")
    if remaining > 60:
        n_perturbations = min(4, int(remaining / 80))
        for pi in range(n_perturbations):
            remaining = _global_deadline - time.monotonic()
            if remaining < 60:
                break
            t_pert = min(80.0, remaining * 0.4)
            if t_pert < 20:
                break

            # Generate perturbation
            scale = 0.05 * (1 - pi * 0.1)
            noise = rng.standard_normal(len(best_seq)) * scale
            perturbed = best_seq * (1.0 + noise)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-10:
                continue
            perturbed = perturbed / ps

            # Quick optimize the perturbation
            a_pert, _ = nesterov_lp_opt(perturbed, t_pert, solver, max_iters=200000, use_reduced_lp=True, top_k=200)
            try_update(a_pert, f"pert_{pi}")

    # === Phase 7: Final spot check with different LP size ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 40:
        t7 = min(60.0, remaining * 0.5)
        print(f"\n=== Phase 7: Full-constraint refinement ({t7:.0f}s) ===")
        a_full, _ = nesterov_lp_opt(best_seq, t7, solver, max_iters=100000, use_reduced_lp=False)
        try_update(a_full, "full_lp_p7")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
