    W_fft = np.fft.rfft(w, N)
    out = np.fft.irfft(np.conj(A_fft) * W_fft, N)[:n]
    
    return 2.0 * out, w, c

def project_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c}."""
    n = len(v)
    # Bisection on lambda: sum(max(v - lambda, 0)) = c
    lo = v.min() - c / n
    hi = v.max()
    
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > c:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    return np.maximum(v - lam, 0)

def lse_nesterov(a0, time_limit, tau_start=10.0, tau_end=1.0, verbose=True):
    """LSE-based Nesterov accelerated projected gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    a_prev = a.copy()
    n_iters = 0
    
    # Learning rate schedule
    # The LSE gradient has Lipschitz constant roughly tau * 2 * max_a^2 * n
    # We'll use an adaptive approach
    
    # Tau annealing schedule
    total_planned = 200000  # rough estimate of total iterations
    
    for t in range(1, total_planned + 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal tau
        progress = min(1.0, t / total_planned)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Nesterov momentum
        if t == 1:
            y = a.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a.copy()
            else:
                y = y * norm / s_y
        
        a_prev = a
        
        # Compute gradient at y
        grad, w, c = lse_gradient(y, tau)
        
        # Learning rate: use tau-based estimate
        # The effective smoothness of LSE is about tau * (max component of grad)^2 / (something)
        # Simple approach: lr = 1 / (tau * ||y||_inf * n) * safety
        # Or: lr = norm / (tau * c.max() + 1)
        
        max_c = c.max()
        if max_c < 1e-12:
            break
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Step size: we want ||lr * grad|| to be comparable to the spread of y
        spread = y.max() - y.min() + 1e-10
        lr = 0.5 * spread / (grad_norm + 1e-10)
        lr = min(lr, 1.0 / (tau * y.max() + 1e-10))
        
        # Projected gradient step
        a_trial = y - lr * grad
        a_trial = project_simplex(a_trial, norm)
        
        a = a_trial
        n_iters += 1
        
        # Check improvement
        if n_iters % 100 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if verbose and n_iters % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE t={t} tau={tau:.2f} ({elapsed:.0f}s): best={best_score:.8f}, cur={score:.8f}")
    
    return best_a, best_score

def active_set_lp(a, M, solver, n_active=50):
    """Solve LP with only the top-n_active tightest constraints."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    # Identify active constraints: where T @ (ones/n) is largest
    b_ref = np.ones(n) / n
    conv_vals = T @ b_ref
    idx = np.argsort(conv_vals)[::-1][:n_active]
    T_active = T[idx]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_active @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov(a0, time_limit, solver, verbose=True, n_active=None):
    """LP-based Nesterov with optional active set reduction."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    a_prev = a.copy()
    n_iters = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = active_set_lp(y, M_curr, solver, n_active)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        if t == 1:
            y_new = g
            a_new = g
        else:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def spectral_peak_shave(a, time_limit, seed=42):
    """Peak shaving via targeted mass redistribution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:20]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find positions that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            if len(candidates) == 0:
                continue
            
            candidates_sorted = candidates[np.argsort(a[candidates])]
            
            for dst in candidates_sorted[:10]:
                for frac in [0.001, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial * norm / a_trial.sum()
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                break
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * norm / a_trial.sum()
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng, variant=0):
    """Generate structured initializations."""
    if variant == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif variant == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.2
    elif variant == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(13.0 * x)
    elif variant == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif variant == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif variant == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.1 * np.cos(4 * np.pi * x) + 0.05 * np.cos(10 * np.pi * x)
    elif variant == 6:
        # Flat with small ripple
        a = np.ones(n)
        a += 0.05 * np.random.default_rng(rng.integers(1e6)).standard_normal(n)
        a = np.maximum(a, 0.1)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*c_k)) where c = conv(a,a).
    Returns (grad, weights, autocorr)."""
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    exp_vals = np.exp(tau * (c - max_c))
    w = exp_vals / exp_vals.sum()
    # grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * corr(w, a)[i]
    # corr(w,a) via FFT: ifft(conj(FFT(w,N)) * FFT(a,N))
    N = 1 << (3 * n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = ifft(np.conj(W) * A, N)[:n].real
    grad = 2.0 * corr
    return grad, w, c


def project_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c} using bisection."""
    n = len(v)
    lo = v.min() - c / n - 1e-10
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.maximum(v - lam, 0.0)


def lse_nesterov(a0, time_limit, tau_start, tau_end, verbose):
    """LSE-based Nesterov accelerated projected gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    a_prev = a.copy()
    n_iters = 0
    total_planned = 300000

    for t in range(1, total_planned + 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if t % 50 == 0 and time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, t / total_planned)
        tau = tau_start * (tau_end / tau_start) ** progress

        if t == 1:
            y = a.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a.copy()
            else:
                y = y * norm / s_y

        a_prev = a

        grad, w, c = lse_grad(y, tau)

        max_c = c.max()
        if max_c < 1e-12:
            break

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        spread = y.max() - y.min() + 1e-10
        lr = 0.8 * spread / (grad_norm + 1e-10)
        lr = min(lr, 2.0 / (tau * y.max() + 1e-10))
        lr = min(lr, 1.0)

        a_trial = y - lr * grad
        a_trial = project_simplex(a_trial, norm)
        a = a_trial
        n_iters += 1

        if n_iters % 200 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if verbose and n_iters % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE iter={n_iters} tau={tau:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    score_final = fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()

    return best_a, best_score


def build_conv_matrix_active(a, n_active):
    """Build convolution matrix with only top-n_active rows (most binding)."""
    n = len(a)
    T_full = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_full[j:j + n, j] = a
    b_ref = np.ones(n) / n
    conv_vals = T_full @ b_ref
    idx = np.argsort(conv_vals)[::-1][:n_active]
    return T_full[idx]


def active_set_lp(a, M, solver, n_active):
    """Solve LP best response with only top-n_active constraints."""
    n = len(a)
    T = build_conv_matrix_active(a, n_active)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def full_lp(a, M, solver):
    """Solve full LP best response."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_val_ensure(b_var), dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def b_val_ensure(b_var):
    return b_var.value

        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        if t == 1:
            y_new = g.copy()
            a_new = g.copy()
        else:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    score_final = fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()

    return best_a, best_score


def spectral_peak_shave(a, time_limit, seed):
    """Peak shaving via targeted mass redistribution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:20]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue

            candidates_sorted = candidates[np.argsort(a[candidates])]

            for dst_idx in range(min(5, len(candidates_sorted))):
                dst = candidates_sorted[dst_idx]
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng, variant):
    """Generate diverse structured initializations."""
    if variant == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif variant == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.2
    elif variant == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(13.0 * x)
    elif variant == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif variant == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif variant == 5:
        a = np.ones(n)
        a += 0.05 * rng.standard_normal(n)
        a = np.maximum(a, 0.05)
    elif variant == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.1 * np.cos(4.0 * np.pi * x) + 0.05 * np.cos(10.0 * np.pi * x)
        a = np.maximum(a, 0.1)
    elif variant == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2) + 0.5
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - 0.2 * np.cos(2.0 * np.pi * x * 3)
        a = np.maximum(a, 0.1)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
