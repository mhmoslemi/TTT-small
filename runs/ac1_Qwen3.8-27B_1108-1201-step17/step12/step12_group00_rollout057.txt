def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    s = a.sum()
    if s < 0.01:

    b = fast_autocorr(a)

    n = len(a)

def coordinate_local_search(a0, time_limit, seed=42):
    """Local search that moves mass between coordinates to reduce the peak."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize(np.maximum(np.asarray(a0, dtype=np.float64), 0.0), 1.0)
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        
        # Find coordinates contributing to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]
        
        if contrib.max() < 1e-15:
            break
        
        # Source: coordinate with highest contribution to the peak
        src = idx[np.argmax(contrib)]
        
        # Find a good destination: low value, low contribution to all top peaks
        top_peaks = np.argsort(b)[::-1][:min(5, 2*n-1)]
        
        # Score each coordinate by its contribution to top peaks
        peak_contrib = np.zeros(n)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            idx2 = np.arange(js2, je2 + 1)
            peak_contrib[idx2] += a[idx2] * a[k2 - idx2]
        
        # Exclude the source
        peak_contrib[src] = float('inf')
        
        # Find the best destination (low peak contribution, and not too correlated with source)
        # We want a destination that doesn't increase other peaks
        candidates = np.where(peak_contrib < np.median(peak_contrib))[0]
        if len(candidates) == 0:
            candidates = np.argsort(peak_contrib)[:n//10]
        
        # Among candidates, prefer those with low current value (spreading mass)
        candidates = candidates[np.argsort(a[candidates])]
        
        dst = candidates[rng.integers(min(len(candidates), max(1, len(candidates)//4)))]
        
        # Try different transfer fractions
        improved = False
        for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            a_trial = normalize(a_trial, 1.0)
            obj = objective_value(a_trial)
            if obj < best_obj - 1e-12:
                a = a_trial
                best_obj = obj
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try random moves
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                for frac in [0.001, 0.005, 0.01]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < 0:
                        continue
                    a_trial = normalize(a_trial, 1.0)
                    obj = objective_value(a_trial)
                    if obj < best_obj:
                        a = a_trial
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
    
    return best_a, best_obj

Now, for the optimization, I'll use Nesterov-accelerated gradient descent on the smoothed objective:

subject to a >= 0, sum(a) = 1.

The constraint sum(a) = 1 can be handled by projecting onto the simplex (or just renormalizing after each step, which is a good approximation).

The non-negativity constraint can be handled by projecting (clipping to 0) after each step.

So the algorithm is:
1. Initialize a (non-negative, sum = 1)
2. Compute b = conv(a, a)
3. Compute p = softmax(τ * b)
4. Compute grad via FFT
5. Update: a_new = a - η * grad (with Nesterov momentum)
6. Project: a_new = max(a_new, 0), a_new = a_new / sum(a_new)
7. Anneal τ (start large, decrease over time)
8. Repeat

This should be much faster than the LP approach since each iteration is O(n log n) instead of O(n^2) or more.

Let me also add a **Frank-Wolfe** step as an alternative: instead of gradient descent, use:
- v = e_{argmin(grad)} (the "linear oracle" for the simplex)
- a_new = (1-α) * a + α * v

This is the Frank-Wolfe (conditional gradient) method, which might converge faster in some cases.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-smoothed Nesterov gradient descent (fast, O(n log n) per iteration)
2. Frank-Wolfe with softmax gradient (alternative fast method)
3. LP-based Nesterov (precise, for refinement)
4. Coordinate local search (final polish)
5. Multi-length exploration
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def objective_value(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def normalize(a, target_sum=1.0):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a * target_sum / s

def softmax(x, tau):
    x = np.asarray(x, dtype=np.float64)
    x_max = x.max()
    ex = np.exp(x / tau - x_max / tau)
    return ex / ex.sum()

def compute_gradient(a, p, fft_size):
    """Compute gradient of smoothed max autocorrelation w.r.t. a."""
    n = len(a)
    # pad
    p_pad = np.zeros(fft_size)
    p_pad[:len(p)] = p
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    P = rfft(p_pad, fft_size)
    A = rfft(a_pad, fft_size)
    corr = irfft(P * np.conj(A), fft_size)
    return 2.0 * corr[:n]

def nesterov_smoothed(a0, time_limit, tau_start=10.0, tau_end=1.0, verbose=True):
    """Nesterov-accelerated gradient descent on softmax-smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, 1.0)
    
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (4 * n - 1).bit_length()
    
    y = a.copy()
    a_nest = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    
    n_iters = 0
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal tau
        progress = min(t / 5000.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        b = fast_autocorr(y)
        p = softmax(b, tau)
        grad = compute_gradient(y, p, fft_size)
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        # Step size: adaptive
        step = 1.0 / (np.linalg.norm(grad) + 1e-10)
        
        y_new = y - alpha * step * grad
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, 1.0)
        
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, 1.0)
        
        y = y_new
        a = a_new
        n_iters += 1
        
        obj = objective_value(y)
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()
        
        obj_a = objective_value(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = normalize(avg_a / n_avg, 1.0)
            obj_avg = objective_value(avg_check)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = avg_check.copy()
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth t={n_iters} (tau={tau:.2f}, {elapsed:.0f}s): best={best_obj:.6f}")
    
    return best_a, best_obj
