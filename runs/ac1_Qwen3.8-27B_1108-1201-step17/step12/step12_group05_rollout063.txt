def fast_autocorr(a):
    """Full convolution a*a (length 2n-1) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    a = a / s
    n = len(a)

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def peak_shaving(a0, time_limit, seed=42):
    """Aggressive coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find the peak location
        k_peak = int(np.argmax(b))
        
        # Identify contributing pairs
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[k_peak - idx]
        
        # Find the largest contributor
        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])
        
        # Try moving mass from src to a random location
        improved = False
        for _ in range(20):
            dst = int(rng.integers(n))
            if dst == src:
                continue
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = score_of(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt

                    sc = score_of(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def spectral_shaping(a, n_iter=200, alpha=0.3):
    """Flatten the power spectrum to reduce convolution peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = score_of(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_conv(a, b):

def spectral_shaping(a, n_iter=200, alpha=0.3):
    """Iteratively flatten the power spectrum to reduce convolution peaks."""
    n = len(a)

        return a

        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        a_new = irfft(A_new, fft_size)[:n]

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

            # d/d a_i (a*a)_k = a_{k-i} (from j=i term) + a_{k-i} (from k-j=i term... no)
            # Let me be careful: (a*a)_k = sum_j a_j * a_{k-j}
            # The term involving a_i is: when j=i: a_i * a_{k-i}; when k-j=i (j=k-i): a_{k-i} * a_i
            # If i != k-i: both terms are a_i * a_{k-i}, total = 2 * a_i * a_{k-i}
            # If i = k-i (k=2i): term is a_i * a_i, total = 2 * a_i^2 (derivative is 2*a_i)
            # In all cases: d/d a_i (a*a)_k = 2 * a_{k-i} (if 0 <= k-i < n, else 0)
            # Wait that's not right either. Let me redo.
            # (a*a)_k = sum_j a_j * a_{k-j}
            # d/d a_i = sum_j [delta_{ji} * a_{k-j} + a_j * delta_{k-j,i}]
            # = a_{k-i} + a_{k-i} = 2 * a_{k-i} (if 0 <= k-i < n)
            # But wait, the first term requires 0 <= k-i < n (for a_{k-i} to be valid)
            # The second term: k-j = i means j = k-i, and we need 0 <= k-i < n
            # So both give the same condition. Total: d/d a_i (a*a)_k = 2 * a_{k-i} (if valid)
            #
            # So the gradient of sum_k w_k * (a*a)_k w.r.t. a_i is:
            # 2 * sum_k w_k * a_{k-i} (for valid k-i, i.e., 0 <= k-i < n)
            # = 2 * sum_{j=0}^{n-1} w_{i+j} * a_j (where j = k-i, k = i+j)
            # = 2 * (w_shifted * a) where w_shifted[j] = w[j+i]
            #
            # This is essentially a cross-correlation: 2 * (a * w)[n-1] ... 
            # Actually, let us just compute it via FFT:
            # sum_k w_k * a_{k-i} = (w * a_reversed)[i] where a_reversed = a[::-1]
            # No wait. Let's define: for fixed i, sum_k w_k * a_{k-i}
            # Let j = k-i, then k = i+j: sum_j w_{i+j} * a_j
            # This is the cross-correlation of w and a at lag i: (w ⊛ a)[i]
            # = (w * a_flipped)[n-1+i] where a_flipped = a[::-1] and * is convolution
            # Actually: (w * a_flipped)[n-1+i] = sum_j w_j * a_flipped[n-1+i-j]
            # = sum_j w_j * a[n-1-(n-1+i-j)] = sum_j w_j * a[j-i]
            # Hmm, let me just use the direct formula with FFT.
            #
            # grad = 2 * corr(w, a) where corr(w,a)[i] = sum_k w[k] * a[k-i]
            # = 2 * ifft(fft(w) * conj(fft(a)))  ... this is getting complicated.
            # Let me just use the convolution matrix approach.

            fft_size = 1 << (2 * n - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv_result = irfft(A_rev * W, fft_size)
            grad = 2.0 * conv_result[n - 1:n - 1 + n]

            # Projected gradient step on the simplex
            # a_new = proj_simplex(a - lr * grad)
            lr = 1.0 / (T * n)
            a_trial = a - lr * grad
            # Project onto simplex: {x: sum(x)=1, x>=0}
            a_trial = simplex_project(a_trial)
            sc = score_of(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
            a = a_trial

        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth-grad stage {si}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def simplex_project(v):
    """Project a vector onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    csv = sv.cumsum()
    rho = np.where(sv * np.arange(1, n + 1) > (csv - 1))[0][-1]
    theta = (sv[rho] - 1) / (rho + 1.0)
    mu = np.maximum(v - theta, 0)
    return mu


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization focused on LP Nesterov:
    Phase 1: Quick spectral shaping (~3% time)
    Phase 2: LP Nesterov at base length (~50% time)
    Phase 3: Smoothed gradient descent (~10% time)
    Phase 4: Peak shaving (~15% time)
    Phase 5: LP Nesterov at alternative length (~15% time)
    Phase 6: Final polishing (~7% time)
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    print(f"Total time budget: {time_budget:.0f}s")

    # Get the LP solver
    solver = get_lp_solver()

    # Initialize from parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_of(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = score_of(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Quick spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 5:
        print(f"\n=== Phase 1: Spectral shaping (t={t1:.0f}s) ===")
        a_spec = spectral_shaping(best_seq, n_iter=400, alpha=0.3)
        sc = score_of(a_spec)
        if sc < best_score:
            best_score = sc
            best_seq = a_spec.copy()
            print(f"  Improved: {best_score:.6f}")

    # Phase 2: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2 = remaining * 0.48
    print(f"\n=== Phase 2: LP Nesterov at n={n_base} (t={t2:.0f}s) ===")
    a_out, sc = nesterov_lp_optimize(best_seq, t2, solver, restart_patience=400)
    if sc < best_score:
        best_score = sc
        best_seq = a_out.copy()
        print(f"  LP improved: {best_score:.6f}")

    # Phase 3: Smoothed gradient descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    t3 = remaining * 0.10
    if t3 > 10:
        print(f"\n=== Phase 3: Smooth gradient (t={t3:.0f}s) ===")
        a_out, sc = smooth_gradient_optimize(best_seq, t3, T_start=30.0, T_end=0.5)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Smooth-grad improved: {best_score:.6f}")

    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    t4 = remaining * 0.15
    if t4 > 10:
        print(f"\n=== Phase 4: Peak shaving (t={t4:.0f}s) ===")
        a_out, sc = peak_shaving(best_seq, t4, seed=42)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Peak shaving improved: {best_score:.6f}")

    # Phase 5: LP at alternative length
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 120:
        # Try a longer sequence
        n_alt = n_base + 200
        t5 = remaining * 0.25
        print(f"\n=== Phase 5: LP at n={n_alt} (t={t5:.0f}s) ===")
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_out, sc = nesterov_lp_optimize(init_alt, t5, solver, restart_patience=300)
        sc_real = evaluate_safe(a_out)
        if np.isfinite(sc_real) and sc_real < best_score:
            best_score = sc_real
            best_seq = a_out.copy()
            print(f"  Alt length (n={n_alt}) improved: {best_score:.6f}")

    # Phase 6: Final polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        # Peak shaving on the current best
        t6a = remaining * 0.4
        print(f"\n=== Phase 6a: Final peak shaving (t={t6a:.0f}s) ===")
        a_out, sc = peak_shaving(best_seq, t6a, seed=999)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Final shaving improved: {best_score:.6f}")

        # One more LP round if time remains
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 60:
            t6b = remaining * 0.6
            n_final = len(best_seq)
            print(f"\n=== Phase 6b: Final LP at n={n_final} (t={t6b:.0f}s) ===")
            a_out, sc = nesterov_lp_optimize(best_seq, t6b, solver, restart_patience=200)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Final LP improved: {best_score:.6f}")

    # Final verification with the real evaluator
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={time.monotonic()-t_global_start:.0f}s)")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
