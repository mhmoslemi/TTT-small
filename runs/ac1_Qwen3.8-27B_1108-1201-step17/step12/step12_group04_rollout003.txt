"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weight updates (LogSumExp gradient)
2. Nesterov-accelerated LP best-response
3. Multi-stage LogSumExp with L-BFGS
4. Targeted coordinate descent on peaks
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_grad_and_weights(a, T):
    """
    Compute gradient and mixture weights for LogSumExp(T * (a*a)).
    Returns (grad, w) where grad = d/dt [1/T * log(sum exp(T*b_k))] and w = softmax(T*b).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad_j = 2 * sum_k w_k * a[k-j] (when 0 <= k-j < n)
    # = 2 * (w * a_rev)[j + n - 1]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T @ b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, n_max_iters=500):
    """
    Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0.
    Returns (b, M) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build the LP in standard form for linprog (HiGHS)
    # Variables: x = (M, b_0, ..., b_{n-1})
    # Minimize: x[0] = M
    # Constraints: T @ b - M <= 0  =>  [-1, T] @ x <= 0
    #              sum(b) = 1     =>  [0, 1,...,1] @ x = 1
    # Bounds: M free, b >= 0
    
    c = np.zeros(n + 1)
    c[0] = 1.0  # minimize M
    
    # Inequality: T @ b - M <= 0
    # Row k: -1 * M + sum_j T[k,j] * b_j <= 0
    A = np.zeros((size, n + 1))
    A[:, 0] = -1.0
    A[:, 1:] = build_conv_matrix(a_fixed)
    b_ineq = np.zeros(size)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(-np.inf, np.inf)] + [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': n_max_iters})
        if res.success:
            M_val = res.x[0]
            b_val = np.maximum(res.x[1:], 0.0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
            return b_val, float(M_val)
    except Exception:
        pass
    
    return None, None


def nesterov_lp_optimize(a0, time_limit, max_iters=10000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}")

def logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=12):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    def softmax_fn(z_in):
        z_in = z_in - z_in.max()
        e = np.exp(z_in)
        s_e = e.sum()
        if s_e < 1e-300:
            return np.ones(len(z_in)) / len(z_in)
        return e / s_e
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters_per_stage = max(5, min(30, int(10 + 20 * (si / max(1, n_stages - 1)))))
        
        def func(z_in):
            a_tmp = softmax_fn(z_in)
            b_tmp = fast_autocorr(a_tmp)
            Tb_tmp = stage_T * b_tmp
            max_Tb = Tb_tmp.max()
            return (max_Tb + np.log(np.exp(Tb_tmp - max_Tb).sum())) / stage_T
        
        def jac(z_in):
            a_tmp = softmax_fn(z_in)
            grad_a, _ = lse_grad_and_weights(a_tmp, stage_T)
            S = np.dot(grad_a, a_tmp)
            return a_tmp * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
        
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si} T={stage_T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
