import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Compute gradient of sum_k w[k]*conv(f,f)[k] w.r.t. f using FFT."""
    n = len(f)
    # Cross-correlation: g[j] = sum_k w[k]*f[k-j]
    N = 1 << (3 * n - 3).bit_length()  # N >= 3n-2
    W = rfft(w, N)
    F = rfft(f, N)
    g = irfft(W * np.conj(F), N)
    grad = 2.0 * g[:n].copy()
    # Correction for 2j=k terms
    for j in range(n):
        k = 2 * j
        if k < 2 * n - 1:
            grad[j] -= w[k] * f[j]
    return grad

                   label="md", seed=42):

    best_f = f.copy()

    eta = eta_init

    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        scaled = b / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute gradient
        grad = compute_grad_fft(f, w)
        
        # Multiplicative update
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - eta * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:

        f_new /= s
        f = f_new

"""
FFT-accelerated mirror descent for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses adaptive temperature annealing, multiplicative weight updates,
coordinate descent polishing, and multi-length exploration.
No external LP solver required.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_fft(f, w):
    """
    Compute gradient of sum_k w[k]*conv(f,f)[k] w.r.t. f using FFT.
    w has length 2n-1 (the convolution support).
    Returns array of length n.
    """
    n = len(f)
    # Cross-correlation via FFT: g[j] = sum_k w[k]*f[k-j]
    total_len = (2 * n - 1) + n - 1  # = 3n - 2
    N = 1 << (total_len - 1).bit_length()
    W = rfft(w, N)
    F = rfft(f, N)
    g = irfft(W * np.conj(F), N)
    grad = 2.0 * g[:n].copy()
    # Correction for diagonal terms (2j = k)
    for j in range(n):
        k = 2 * j
        if k < 2 * n - 1:
            grad[j] -= w[k] * f[j]
    return grad


def mirror_descent_step(f, eta, tau):
    """Single mirror descent step. Returns new f and the max convolution."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    
    # Softmax weights on convolution
    scaled = b / (tau * max(M, 1e-10))
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return f, M
    w /= w_sum
    
    # Compute gradient
    grad = compute_grad_fft(f, w)
    
    # Multiplicative (mirror) update
    log_f = np.log(f + 1e-30)
    log_f_new = log_f - eta * grad
    log_f_new -= log_f_new.max()
    f_new = np.exp(log_f_new)
    s = f_new.sum()
    if s < 1e-30:
        return f, M
    f_new /= s
    return f_new, M


def run_mirror_descent(f_init, time_limit, eta_init=0.05, tau_init=0.005,
                       label="md", seed=42):
    """Run mirror descent with adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    eta = eta_init
    tau = tau_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f, _ = mirror_descent_step(f, eta, tau)
        
        if it % 5 == 0:
            b = fast_convolve(f)
            M = b.max()
            if M < best_b - 1e-15:
                best_b = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve % 5 == 0:
                    eta *= 0.8
                if no_improve > 20:
                    # Perturb and reset
                    f = f + rng.standard_normal(n) * 0.002 * f
                    f = np.maximum(f, 1e-15)
                    f /= f.sum()
                    eta = eta_init
                    no_improve = 0
            
            if it % 50 == 0:
                score = 2.0 * n * best_b
                print(f"  {label} it={it} obj={best_b:.12f} score={score:.6f} "
                      f"eta={eta:.5f} tau={tau:.5f} t={time.monotonic()-t0:.0f}s")
            
            # Anneal temperature
            if it % 100 == 0:
                tau = min(tau * 1.05, 0.05)
        
        if no_improve > 50:
            break
    
    # Final evaluation
    best_b = fast_convolve(best_f).max()
    return best_f, best_b
