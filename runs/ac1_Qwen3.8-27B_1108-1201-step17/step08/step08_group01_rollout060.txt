def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_hankel_symmetric(K, lambdas, n):
    """Build symmetric Hankel matrix M_s where (M_s)[i,j] = lambda[i+j] if i+j in K."""
    M = np.zeros((n, n), dtype=np.float64)
    lag_dict = dict(zip(K, lambdas))
    for k, lam in lag_dict.items():
        if k < 0 or k >= 2 * n - 1:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            if 0 <= j < n:
                M[i, j] += lam * 0.5
                M[j, i] += lam * 0.5
    # Ensure symmetry
    M = 0.5 * (M + M.T)
    return M

def kkt_solve_active(n, active_lags, lambdas, reg=1e-12):
    """
    Solve KKT system for active set method.
    Returns the optimal a (or None if infeasible).
    """
    M = build_hankel_symmetric(active_lags, lambdas, n)
    M = M + reg * np.eye(n)
    ones = np.ones(n)
    try:
        x = np.linalg.solve(M, ones)
    except np.linalg.LinAlgError:
        M += 1e-8 * np.eye(n)
        try:
            x = np.linalg.solve(M, ones)
        except np.linalg.LinAlgError:
            return None
    s = np.sum(x)
    if s <= 0:
        return None
    a_sol = x / s
    if np.min(a_sol) < -0.01:
        return None
    a_sol = np.maximum(a_sol, 0.0)
    a_sol = a_sol / a_sol.sum()
    return a_sol

def active_set_polish(a0, time_limit, tol=1e-10, max_iter=200, verbose=False):

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1

        temp = 0.01 * max(M_cur, 1e-10)

            a = a_new

def nesterov_subgradient(a0, time_limit, alpha0=0.05, verbose=False, label=""):
    """Nesterov-accelerated subgradient with mirror geometry (multiplicative)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    alpha = alpha0
    y = a.copy()  # Nesterov momentum variable
    a_prev = a.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_y = fast_convolve(y)
        M_y = float(conv_y.max())

        # Active set for subgradient
        k_act = min(30, len(conv_y))
        peaks = np.argpartition(conv_y, -k_act)[-k_act:]
        pv = conv_y[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        # Subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            jj = np.arange(i_min, i_max + 1)
            grad[jj] += 2.0 * w[idx] * y[k - jj]

        # Multiplicative update
        a_new = y * np.exp(-alpha * grad)
        a_new = simplex_project(a_new)

        # Nesterov update
        t_k = 0.5 + np.sqrt(1.0 + 2.0 * it * it) / 2.0
        t_prev = 0.5 + np.sqrt(1.0 + 2.0 * (it - 1) * (it - 1)) / 2.0 if it > 1 else 1.0
        y = a_new + (t_prev - 1.0) / t_k * (a_new - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        a_prev = a_new.copy()
        a = a_new

        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            alpha *= 0.8
            no_improve = 0
            if alpha < 1e-8:
                break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} alpha={alpha:.6f}")

    return best_a, best_score

def lp_best_response_fast(a_fixed, n_constraints=None, solver=None):
    """
    Given fixed a (sum=1, a>=0), find b minimizing max_k (a*b)[k] s.t. b>=0, sum(b)=1.
    Uses constraint selection for speed.
    """
    import cvxpy as cp
    if solver is None:
        # Try to find a fast solver
        for s in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solver = s
                    break
            except Exception:
                continue
        if solver is None:
            solver = cp.ECOS

    n = len(a_fixed)
    # Compute current max for constraint selection
    # (a*b)_k for b = uniform is (a*a)_k / n ... no
    # Let's just use all constraints if n is small, or select top ones
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Constraint selection: use all constraints (they're only 2n-1)
    if n_constraints is not None and n_constraints < 2 * n - 1:
        # Select constraints most likely to be active
        # Use the uniform b as a proxy
        b_unif = T @ (np.ones(n) / n)
        top_idx = np.argsort(b_unif)[-n_constraints:]
        constraints = [T[top_idx] @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[max(0, 0):n]  # a[k-i] for k in [k_start, k_end]
            # For k = i+j, the coefficient is a[j] where j = k - i, so j ranges from 0 to n-1
            # a_shifted[j] = a[j] (since k-i = j)

            best_delta = 0.0
            best_new_M = M

            factors = [-0.3, -0.15, -0.08, -0.04, 0.0, 0.04, 0.08, 0.15, 0.3, 0.5][:n_trials]
            for f in factors:
                delta = f * a[i]
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue

                # New affected values
                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                # Fix the diagonal term at k=2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                # Update conv
                delta = best_delta
                a_shifted_old = a[i] - delta  # old value of a[i]
                changes = 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * a_shifted_old + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[0:n]
            best_delta = 0.0
            best_new_M = M

            for factor in [0.995, 0.99, 0.985, 0.98, 0.9925, 1.005, 1.01, 1.015, 1.02]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue

                new_aff = (conv[k_start:k_end + 1] + 2.0 * delta * a_shifted).copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-17:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-17:
                a[i] = max(a[i] + best_delta, 0.0)
                delta = best_delta
                a_old = a[i] - delta
                changes = (2.0 * delta * a_shifted).copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes[idx] = 2.0 * delta * a_old + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_score, label=""):
    """Evaluate and return score."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return None, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return a_norm, best_score
    return None, best_score

def perturbed_restart(a0, rng, mag=0.02):
    """Create a perturbed version to escape local patterns."""
    n = len(a0)
    a = a0.copy() + rng.standard_normal(n) * mag
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s

def propose_candidate():
