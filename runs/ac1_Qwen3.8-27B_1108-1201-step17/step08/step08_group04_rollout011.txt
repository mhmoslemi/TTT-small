def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

def solve_lp_direction(f, indices, M, n, prev_g=None):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M"""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n, initial_value=prev_g if prev_g is not None else None)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(objective, constraints)
    
    try:
        if n <= 1500:
            prob.solve(solver=cp.ECOS, max_iters=50000, abs_acceleration=1e-10)
        else:
            prob.solve(solver=cp.SCS, max_iters=500000, eps_abs=1e-10, eps_rel=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-9, eps_rel=1e-9)
        except Exception:
            return None, None, None
    
    if g.value is None:
        return None, None, None
    g_val = np.maximum(g.value, 0.0)
    
    # Try to get dual values
    dual_vals = None
    try:
        if prob.status in ['optimal', 'optimal_inaccurate']:
            # constraints[1] is A @ g <= M
            dual_vals = constraints[1].dual_value
    except Exception:
        pass
    
    return g_val, dual_vals, prob.status

"""
Optimized step function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based direction finding with warm-starting, dual-weighted adaptive peak
selection, Nesterov-momentum subgradient, pair coordinate descent, and multi-length
exploration. Key improvements over baseline: LP warm-start, better line search,
pair CD updates, and more strategic length sweeping.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def build_A_matrix(f, indices, n):
    """Build K x n constraint matrix for given peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, indices, M, n, prev_g=None):
    """
    Solve: max sum(g) s.t. g >= 0, A(g) <= M.
    Returns (g_val, dual_vals, status).
    """
    K = len(indices)
    A = build_A_matrix(f, indices, n)

    if prev_g is not None:
        g = cp.Variable(n, initial_value=prev_g)
    else:
        g = cp.Variable(n)

    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(objective, constraints)

    status = None
    try:
        if n <= 1200 and K <= 600:
            prob.solve(solver=cp.ECOS, max_iters=80000, abs_acceleration=1e-11)
        elif n <= 2000:
            prob.solve(solver=cp.SCS, max_iters=300000, eps_abs=1e-10, eps_rel=1e-10)
        else:
            prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None, None, 'error'

    status = prob.status
    if g.value is None:
        return None, None, status
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, None, status

    dual_vals = None
    try:
        if status in ('optimal', 'optimal_inaccurate'):
            dual_vals = np.asarray(constraints[1].dual_value, dtype=np.float64)
            if dual_vals.shape[0] != K:
                dual_vals = None
    except Exception:
        dual_vals = None

    return g_val, dual_vals, status

            stuck += 1

        if it % 25 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.8f} "
                  f"k={len(indices)} t={elapsed:.0f}s stuck={stuck}")

    return best_f, best_score


def subgradient_refine(f_init, time_limit, lr_init=0.004, n_top=120):
    """
    Subgradient optimization with adaptive temperature and Nesterov momentum.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    v = np.zeros(n)
    lr = lr_init
    temp = 0.002
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(55555)
    prev_f = f.copy()
    prev_g = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Smoothed max gradient with adaptive temperature
        scaled = (b - M) / temp
        # Clip for numerical stability
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / ws

        # Restrict to top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            wss = w_sel.sum()
            if wss > 1e-30:
                w_sel = w_sel / wss
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sel
            w = w_new

        # Compute subgradient
        g = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * f[k - idx]

        # Barzilai-Borwein step size adaptation
        if it > 1:
            s_bb = f - prev_f
            y_bb = g - prev_g
            sy = np.dot(s_bb, y_bb)
            yy = np.dot(y_bb, y_bb)
            ss = np.dot(s_bb, s_bb)
            if yy > 1e-20:
                lr_bb1 = sy / yy if sy > 0 else lr
                lr = 0.5 * lr + 0.5 * max(1e-6, min(lr_bb1, 0.05))
            prev_g = g.copy()
            prev_f = f.copy()

        v = 0.82 * v - lr * g
        f_new = project_simplex(f + v)
        f = f_new

        if it % 15 == 0:
            obj_cur = fast_convolve(f).max()
            if obj_cur < best_obj - 1e-15:
                best_obj = obj_cur
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 4:
                    lr *= 0.92
                if no_improve > 25:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    v = np.zeros(n)
                    no_improve = 0
                    lr = lr_init * 0.5

            temp = min(temp * 1.08, 0.4)

            if it % 75 == 0:
                print(f"    sg it={it} obj={best_obj:.12f} score={2*n*best_obj:.8f} "
                      f"lr={lr:.6f} temp={temp:.4f} t={time.monotonic()-t0:.0f}s")

    return best_f, 2.0 * n * best_obj


def pair_coordinate_descent(f_init, time_limit):
    """
    Pair coordinate descent: update two coordinates simultaneously.
    More efficient than single-coordinate for this problem.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    rng = np.random.default_rng(777)

    mults = [0.5, 0.65, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7, 2.2]
    no_pass = 0
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        pass_count += 1

        # Create random pair ordering
        indices = np.arange(n)
        rng.shuffle(indices)
        pairs = []
        for i in range(0, n - 1, 2):
            pairs.append((int(indices[i]), int(indices[i + 1])))
        if n % 2 == 1:
            pairs.append((int(indices[-1]), int(indices[0])))

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if f[i] < 1e-14 and f[j] < 1e-14:
                continue

            best_i_val = f[i]
            best_j_val = f[j]
            best_pair_obj = best_obj

            # Try a subset of multipliers for speed
            for mi in mults:
                f_trial = f.copy()
                f_trial[i] = f[i] * mi
                for mj in mults:
                    f_trial[j] = f[j] * mj
                    s = f_trial.sum()
                    if s < 1e-12:
                        continue
                    f_trial = f_trial / s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_pair_obj - 1e-14:
                        best_pair_obj = obj
                        best_i_val = f_trial[i]
                        best_j_val = f_trial[j]

            if best_i_val != f[i] or best_j_val != f[j]:
                f[i] = best_i_val
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_pair_obj < best_obj - 1e-14:
                    best_obj = best_pair_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True

        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, best_score


def random_local_search(f_init, time_limit, num_trials=200):
    """
    Random perturbation local search: try small random perturbations
    and keep improvements. Cheap way to escape very shallow local minima.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    rng = np.random.default_rng(31415)

    trial = 0
    while time.monotonic() - t0 < time_limit and trial < num_trials:
        trial += 1
        # Random perturbation
        scale = 0.001 + 0.005 * (trial / num_trials)  # Slightly increasing
        perturbation = rng.standard_normal(n) * scale
        f_trial = project_simplex(f + perturbation)

        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_score = 2.0 * n * obj
            best_f = f_trial.copy()
            f = f_trial.copy()

        if trial % 50 == 0:
            print(f"    rls trial={trial} obj={best_obj:.12f} score={best_score:.8f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1 (35%): LP at parent length with warm-starting and dual-weighted peaks
    Phase 2 (20%): LP at nearby lengths for exploration
    Phase 3 (12%): Subgradient with BB step size
    Phase 4 (8%):  Random local search
    Phase 5 (10%): Pair coordinate descent
    Phase 6 (15%): Final LP burst with high constraint count
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    else:
        parent = np.ones(1152) / 1152.0
        p_score = 2.0
        print("No parent found, using uniform n=1152")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # ============================================================
    # Phase 1: LP at parent length
    # ============================================================
    rem = deadline - time.monotonic()
    t1 = rem * 0.35
    print(f"\n{'='*60}")
    print(f"Phase 1: LP n={n_parent}, budget={t1:.0f}s, target={best_score:.8f}")
    print(f"{'='*60}")

    if t1 > 30:
        # First pass: moderate constraints
        alloc1a = t1 * 0.55
        f_res, sc = run_lp_optimization(parent, time_limit=alloc1a,
                                         top_k_init=80, top_k_max=500, label="p1a")
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  IMPROVED: {best_score:.8f}")

        # Second pass: higher constraints on improved solution
        rem = deadline - time.monotonic()
        if rem > 20:
            alloc1b = min(t1 * 0.45, rem * 0.7)
            if alloc1b > 15:
                f_res2, sc2 = run_lp_optimization(best_seq, time_limit=alloc1b,
                                                   top_k_init=150, top_k_max=700,
                                                   label="p1b", momentum_beta=0.4)
                real_sc2 = float(evaluate_sequence(f_res2.tolist()))
                if np.isfinite(real_sc2) and real_sc2 < best_score:
                    best_score = real_sc2
                    best_seq = f_res2.copy()
                    print(f"  IMPROVED: {best_score:.8f}")

    # ============================================================
    # Phase 2: Nearby length exploration
    # ============================================================
    rem = deadline - time.monotonic()
    t2 = rem * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 2: Length exploration, budget={t2:.0f}s, best={best_score:.8f}")
    print(f"{'='*60}")

    if t2 > 30:
        n_b = len(best_seq)
        candidates = []
        for delta in [-150, -100, -50, 50, 100, 150, 200]:
            n_try = n_b + delta
            if 300 <= n_try <= 2500:
                candidates.append(n_try)
        # Add some fixed "nice" lengths
        for nice in [1024, 1100, 1200, 1280, 1350, 1400, 1500]:
            if 300 <= nice <= 2500 and nice not in candidates:
                candidates.append(nice)

        candidates = sorted(set(candidates))
        # Limit to at most 6 to ensure time per length
        if len(candidates) > 6:
            # Prioritize lengths closer to current best
            candidates.sort(key=lambda x: abs(x - n_b))
            candidates = candidates[:6]
            candidates = sorted(candidates)

        per_n = t2 / max(1, len(candidates))

        for n_try in candidates:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            alloc = min(per_n * 1.3, rem * 0.8)
            if alloc < 15:
                break

            f_init = interp_sequence(best_seq, n_try)
            print(f"  LP n={n_try}, budget={alloc:.0f}s")
            f_res, sc = run_lp_optimization(f_init, time_limit=alloc * 0.75,
                                             top_k_init=60, top_k_max=400,
                                             label=f"n{n_try}")
            real_sc = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_sc) and real_sc < best_score:
                best_score = real_sc
                best_seq = f_res.copy()
                print(f"  NEW BEST [n={n_try}]: {best_score:.8f}")

    # ============================================================
    # Phase 3: Subgradient refinement
    # ============================================================
    rem = deadline - time.monotonic()
    t3 = rem * 0.12
    print(f"\n{'='*60}")
    print(f"Phase 3: Subgradient, budget={t3:.0f}s, best={best_score:.8f}, n={len(best_seq)}")
    print(f"{'='*60}")

    if t3 > 15:
        n_b = len(best_seq)
        f_res, sc = subgradient_refine(best_seq, time_limit=t3,
                                        lr_init=0.003, n_top=max(60, n_b // 8))
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  IMPROVED [sg]: {best_score:.8f}")

    # ============================================================
    # Phase 4: Random local search
    # ============================================================
    rem = deadline - time.monotonic()
    t4 = rem * 0.08
    print(f"\n{'='*60}")
    print(f"Phase 4: Random LS, budget={t4:.0f}s, best={best_score:.8f}")
    print(f"{'='*60}")

    if t4 > 10:
        f_res, sc = random_local_search(best_seq, time_limit=t4, num_trials=300)
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  IMPROVED [rls]: {best_score:.8f}")

    # ============================================================
    # Phase 5: Pair coordinate descent
    # ============================================================
    rem = deadline - time.monotonic()
    t5 = rem * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 5: Pair CD, budget={t5:.0f}s, best={best_score:.8f}")
    print(f"{'='*60}")

    if t5 > 10:
        f_res, sc = pair_coordinate_descent(best_seq, time_limit=t5)
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  IMPROVED [cd]: {best_score:.8f}")

    # ============================================================
    # Phase 6: Final LP burst
    # ============================================================
    rem = deadline - time.monotonic()
    t6 = rem * 0.80
    print(f"\n{'='*60}")
    print(f"Phase 6: Final LP, budget={t6:.0f}s, best={best_score:.8f}, n={len(best_seq)}")
    print(f"{'='*60}")

    if t6 > 15:
        n_b = len(best_seq)
        f_res, sc = run_lp_optimization(best_seq, time_limit=t6,
                                         top_k_init=200, top_k_max=900,
                                         label="final", momentum_beta=0.3)
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  IMPROVED [final]: {best_score:.8f}")

    # Final safety check
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = parent.copy()
        best_score = p_score

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
