"""
FFT-accelerated multiplicative weight (mirror descent) optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses cross-correlation via FFT for O(n log n) gradient
computation, adaptive temperature, multiple diverse initializations, coordinate descent
polishing, and multi-length exploration. Significantly faster per-iteration than LP-based
methods, enabling far more iterations and restarts within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def mw_optimize(f_init, time_limit, label="mw", T_frac_init=0.03, T_frac_min=0.003,
                T_decay=0.9985, eta_init=0.08, peak_frac_init=0.6, peak_frac_end=0.9,
                restart_interval=500, seed=42):
    """
    FFT-accelerated multiplicative weight / mirror descent optimization.
    Minimizes max(conv(f, f)) subject to f >= 0, sum(f) = 1.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    f = np.maximum(f, 1e-30)
    f /= f.sum()

    best_f = f.copy()
    best_b = fast_convolve(f).max()

    T_frac = T_frac_init
    eta = eta_init
    peak_frac = peak_frac_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    conv_size = 2 * n - 1
    fft_size_conv = 1 << (conv_size - 1).bit_length()
    # Cross-correlation: w has length conv_size, f has length n
    corr_size = conv_size + n - 1
    fft_size_corr = 1 << (corr_size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Step 1: Compute convolution
        A = rfft(f, fft_size_conv)
        b = irfft(A * A, fft_size_conv)[:conv_size]
        M = b.max()

        # Step 2: Compute softmax weights
        T = max(T_frac * M, 1e-30)
        n_peaks = max(20, int(peak_frac * conv_size))
        if n_peaks < conv_size:
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
            b_focused = np.zeros(conv_size)
            b_focused[top_idx] = b[top_idx]
        else:
            b_focused = b

        scaled = b_focused / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(conv_size) / conv_size
        else:
            w /= w_sum

        # Step 3: Compute gradient via cross-correlation
        W = rfft(w, fft_size_corr)
        F = rfft(f, fft_size_corr)
        corr = irfft(np.conj(W) * F, fft_size_corr)
        grad = 2.0 * corr[:n]

        # Step 4: Multiplicative update
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-30)
        s = f.sum()
        if s < 1e-30:
            f = best_f.copy()
        else:
            f /= s

        # Periodic evaluation
        if it % 10 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-15:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive parameters
        T_frac = max(T_frac_min, T_frac * T_decay)
        peak_frac = min(1.0, peak_frac + 0.0001)

        if no_improve > 30:
            eta = max(0.005, eta * 0.93)
            no_improve = 0

        if it % restart_interval == 0 and no_improve > restart_interval // 2:
            # Perturbation restart
            f = best_f + rng.standard_normal(n) * 0.005 * np.median(best_f)
            f = np.maximum(f, 1e-30)
            f /= f.sum()
            eta = eta_init
            T_frac = max(T_frac_min, T_frac_init * 0.5)
            peak_frac = peak_frac_init
            no_improve = 0

        if it % 1000 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                  f"T_frac={T_frac:.6f} eta={eta:.6f} {time.monotonic()-t0:.0f}s")

    # Final check
    obj = fast_convolve(f).max()
    if obj < best_b:
        best_b = obj
        best_f = f.copy()

    return best_f, best_b

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def make_parabolic(n):
    """Parabolic window (high in middle, low at ends)."""
    x = np.linspace(0, 1, n)
    f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_sine(n):
    """Sine window."""
    x = np.linspace(0, 1, n)
    f = np.sin(np.pi * x)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_gaussian(n, sigma_frac=0.35):
    """Gaussian window."""
    x = np.linspace(0, 1, n)
    center = 0.5
    f = np.exp(-0.5 * ((x - center) / sigma_frac) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_random_smooth(n, seed=0, smooth=5):
    """Smooth random sequence."""
    rng = np.random.default_rng(seed)
    f = np.abs(rng.standard_normal(n))
    # Smooth
    kernel = np.ones(smooth) / smooth
    f = np.convolve(f, kernel, mode='same')
    f = np.maximum(f, 1e-10)
    return f / f.sum()
