import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n Toeplitz matrix C such that C @ g = f * g."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

    return best_a, best_score


def lp_fictitious_tight(a0, time_limit, verbose=True, label="LP-Tight"):
    """LP-FP with tightening constraints on hot lags."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    stall_threshold = 150
    epsilon = 0.005
    step = 0.3

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        use_tight = (iters > 50)
        if use_tight:
            g = solve_tight_response(a, M, prev_g=prev_g, epsilon=epsilon)
        else:
            g = solve_best_response(a, M, prev_g=prev_g)

        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > stall_threshold:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters * 104729)
            a = best_a + rng.standard_normal(n) * 0.0008
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            prev_g = None
            stall_count = 0
            epsilon = min(0.05, epsilon * 1.5)

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, "
                  f"eps={epsilon:.4f}, step={step:.4f}, time={elapsed:.0f}s")

    return best_a, best_score


def bussgang_iteration(a0, time_limit, verbose=True):
    """Bussgang-type spectral flattening with adaptive tau."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0
    tau = 0.3

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            tau = min(1.0, tau * 1.02)
        else:
            tau = max(0.1, tau * 0.98)
        a = a_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Bussgang iter={iters}, best={best_score:.8f}, "
                  f"tau={tau:.3f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True, n_pairs=500):
    """Coordinate descent with ternary search for optimal transfer."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    iters = 0
    no_improve_count = 0
    max_no_improve = 50

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            base = min(a[i], a[j] * 2.0)
            if base < 1e-14:
                continue

            # Ternary search for optimal transfer amount
            lo = 0.0
            hi = base
            for _ in range(8):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                a1 = a.copy()
                a1[i] -= m1
                a1[j] += m1
                a1 = np.maximum(a1, 0)
                s1 = a1.sum()
                sc1 = compute_score(a1 / s1) if s1 > 1e-15 else 1e10

                a2 = a.copy()
                a2[i] -= m2
                a2[j] += m2
                a2 = np.maximum(a2, 0)
                s2 = a2.sum()
                sc2 = compute_score(a2 / s2) if s2 > 1e-15 else 1e10

                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1

            t_opt = (lo + hi) / 2.0
            a_trial = a.copy()
            a_trial[i] -= t_opt
            a_trial[j] += t_opt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)

            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                no_improve_count = 0

        if not improved:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                noise = rng.standard_normal(n) * (0.001 + 0.002 * no_improve_count / max_no_improve)
                a = best_a.copy() + noise
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                no_improve_count = 0

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def smoothed_objective(z, T, n):
    """Smoothed max objective using log-sum-exp."""
    z_max = z.max()
    a = np.exp(z - z_max)
    s = a.sum()
    if s < 1e-300:
        return 1e10
    a = a / s
    conv = fast_convolve(a)
    shifted = conv / T
    m = shifted.max()
    lse = m + np.log(np.exp(shifted - m).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    """Gradient of smoothed objective w.r.t. z."""
    z_max = z.max()
    a = np.exp(z - z_max)
    s = a.sum()
    if s < 1e-300:
        return np.zeros_like(z)
    a = a / s
    conv = fast_convolve(a)
    shifted = conv / T
    m = shifted.max()
    w = np.exp(shifted - m)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=3000, verbose=False):
    """L-BFGS on smoothed objective in log-space."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-15)
    z0 = np.log(a_safe / a_safe.sum())
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        z_opt = res.x
        z_max = z_opt.max()
        a_out = np.exp(z_opt - z_max)
        a_out = a_out / a_out.sum()
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    if verbose:
        print(f"  L-BFGS done: best={best_score:.8f}, time={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Check a candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Bussgang spectral shaping (quick initial improvement)
    2. LP-FP with momentum (main driver)
    3. LP-FP with tight constraints on hot lags
    4. Coordinate descent with ternary search
    5. L-BFGS polish at multiple temperature levels
    6. Multi-length exploration with LP-FP
    7. Final LP-FP polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: Bussgang spectral shaping
    print(f"\n=== Phase 1: Bussgang (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.08, 80.0)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    a_bg, sc_bg = bussgang_iteration(best_seq, t1, verbose=True)
    best_seq, best_score = check_and_update(a_bg, "bussgang", best_seq, best_score)

    # Phase 2: LP-FP with momentum (main driver, generous budget)
    print(f"\n=== Phase 2: LP-FP Momentum (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    print(f"  Budget: {t2:.0f}s, n={n_parent}")
    a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t2, step=0.45,
                                        use_momentum=True, momentum_beta=0.3,
                                        label="LP-Mom")
    best_seq, best_score = check_and_update(a_lp2, "lp_momentum", best_seq, best_score)

    # Phase 3: LP-FP tight constraints
    print(f"\n=== Phase 3: LP-FP Tight (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if rem > 60:
        print(f"  Budget: {t3:.0f}s")
        a_lp3, sc_lp3 = lp_fictitious_tight(best_seq, t3, verbose=True, label="LP-Tight")
        best_seq, best_score = check_and_update(a_lp3, "lp_tight", best_seq, best_score)

    # Phase 4: Coordinate descent with ternary search
    print(f"\n=== Phase 4: Coord Descent Ternary (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 180.0)
    if rem > 40:
        print(f"  Budget: {t4:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t4, verbose=True, n_pairs=500)
        best_seq, best_score = check_and_update(a_cd, "cd_ternary", best_seq, best_score)

    # Phase 5: L-BFGS polish
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 800, 2000]:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.08, 60.0)
        if rem < 30:
            break
        print(f"  L-BFGS T={T_val}, budget={t5:.0f}s")
        a_lb, sc_lb = lbgfs_optimize(best_seq, t5, T=T_val, maxiter=2000, verbose=True)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    alt_lengths = []
    for delta in [2, 4, 8, -2, -4, 16, 32]:
        alt_n = n_parent + delta
        if 128 <= alt_n <= 4096 and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t6 = min(rem * 0.15, 100.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  LP-FP n={alt_n}, budget={t6:.0f}s")
        a_alt, sc_alt = lp_fictitious_play(init_alt, t6, step=0.3,
                                            use_momentum=True, momentum_beta=0.2,
                                            label=f"LP-n{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"lp_n{alt_n}", best_seq, best_score)

    # Phase 7: Final LP-FP polish on best
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.4, 200.0)
        print(f"  Budget: {t7:.0f}s")
        a_final, sc_final = lp_fictitious_play(best_seq, t7, step=0.2,
                                                use_momentum=True, momentum_beta=0.2,
                                                label="LP-Final")
        best_seq, best_score = check_and_update(a_final, "lp_final", best_seq, best_score)

    # Phase 8: One more coordinate descent
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.5, 100.0)
        print(f"\n=== Phase 8: Final CD (best={best_score:.8f}) ===")
        a_cd2, sc_cd2 = coordinate_descent(best_seq, t8, verbose=True, n_pairs=1000)
        best_seq, best_score = check_and_update(a_cd2, "cd_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        if s > 0.01:
            best_seq = best_seq / s
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")
    return [float(x) for x in best_seq]
