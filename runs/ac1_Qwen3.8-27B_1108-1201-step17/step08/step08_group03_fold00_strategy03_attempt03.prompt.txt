def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_direction(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix A where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, eps=1e-8):
    """Solve: max sum(g) s.t. g>=0, A@g <= M."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        g_val = None
    if g.value is None:
        # Try ECOS for smaller problems
        if n <= 2000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=20000)
            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

def iaa_diversify(f_init, n_iters=200, peak_fraction=0.3):
    """
    IAA-based diversification: iteratively try to flatten the auto-convolution.
    Returns a diverse but valid (non-negative, normalized) sequence.
    """
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]

        # Target: reduce peak while preserving energy
        peak_val = R.max()
        peak_idx = R.argmax()
        # Create target: keep peak, reduce others toward a floor
        floor = peak_fraction * peak_val
        R_target = np.clip(R, floor, peak_val)
        # Preserve total energy (sum of R = sum(f)^2 = 1 for normalized f)
        energy = R.sum()
        if R_target.sum() > 1e-10:
            R_target *= energy / R_target.sum()

        # Update: new |F| from target R
        R_t_fft = rfft(R_target, fft_size)
        # R_target should be close to a valid autocorrelation, so FFT should be ~real non-negative
        real_part = np.maximum(R_t_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        # Preserve phase
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = f_init.copy()
            break

    return f


def run_subgradient(f_init, time_limit, lr_init=0.002, n_top=50, temp_init=0.001):
    """Subgradient method with momentum and adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99)
    momentum = 0.85

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Soft-max weights
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        # Subgradient: grad F(f) = weighted sum of 2*f[k-j] contributions
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * f[k - idx]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 80 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 8:
                    lr *= 0.88
                    temp *= 1.12
                    no_improve = 0
                if no_improve > 40:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                    v = np.zeros(n)
                    no_improve = 0

            if it % 500 == 0:
                print(f"    sg it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 400)

    for sweep in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.75, 0.9, 1.1, 1.25]:
                nv = f[j] * mult
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj
