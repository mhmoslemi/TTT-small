import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    """LP-based fictitious play with adaptive step and restarts."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    stall_threshold = 300

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

def bliss_init(n, iters=200, seed=42):
    """BLISS method: design sequence with flat power spectrum, non-negative."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    # Random initial phases (with conjugate symmetry for real output)
    A = np.fft.fft(np.ones(n) / n, fft_size)
    phases = rng.standard_normal(fft_size // 2 + 1)
    # Ensure conjugate symmetry
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size // 2 + 1] = np.abs(A[:fft_size // 2 + 1]) * np.exp(1j * phases[:fft_size // 2 + 1])
    for k in range(1, fft_size // 2):
        A_full[fft_size - k] = np.conj(A_full[k])

    a = np.zeros(n)
    for i in range(iters):
        A = rfft(a, fft_size) if i > 0 else A_full
        # Target: flat magnitude
        target_mag = np.sqrt(fft_size)
        A_new = np.zeros(fft_size, dtype=complex)
        A_real = np.zeros(fft_size, dtype=complex)
        A_real[:fft_size // 2 + 1] = A[:fft_size // 2 + 1]
        for k in range(1, fft_size // 2):
            A_real[fft_size - k] = np.conj(A_real[k])
        # Blend magnitude toward target
        mag = np.abs(A_real[:fft_size // 2 + 1])
        new_mag = 0.7 * target_mag + 0.3 * mag
        new_mag[mag < 1e-10] = target_mag
        phase = np.angle(A_real[:fft_size // 2 + 1])
        A_real[:fft_size // 2 + 1] = new_mag * np.exp(1j * phase)
        for k in range(1, fft_size // 2):
            A_real[fft_size - k] = np.conj(A_real[k])
        a_full = np.fft.ifft(A_real).real
        a = a_full[:n].copy()
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss
        else:
            a = np.ones(n) / n
    return a


def chirp_init(n, seed=42):
    """Initialize with chirp-like structure shifted to be non-negative."""
    t = np.arange(n)
    # Quadratic chirp
    phase = 2 * np.pi * t * t / (2 * n)
    a = np.cos(phase) + 1.5  # Shift to be positive
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def raised_cosine_init(n):
    """Initialize with raised cosine shape."""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.1
    a = a / a.sum()
    return a


def gauss_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def bussgang_iteration(a0, time_limit, verbose=True, tau=0.5, label="Bussgang"):
    """Bussgang-type spectral flattening."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        # Damping to help convergence
        a = 0.7 * a + 0.3 * a_new
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000, verbose=True, label="L-BFGS"):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    call_count = [0]

    def func(z):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
        if verbose:
            print(f"  {label} T={T}: score={best_score:.8f}, fun_evals={call_count[0]}")
    except Exception as e:
        if verbose:
            print(f"  {label} T={T} failed: {e}")

    return best_a, best_score


def peak_directed_search(a0, time_limit, verbose=True, label="PeakSearch"):
    """
    Direct peak reduction: find the peak of convolution, compute subgradient,
    and take steps to reduce it. Uses line search for step size.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = 0.1  # initial step size

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = np.argmax(conv)
        M = conv[k_star]

        # Subgradient of max(conv) w.r.t. a[j]
        # d(conv[k_star])/d(a[j]) = 2*a[k_star - j] if 0 <= k_star-j <= n-1
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        # Also need to account for sum constraint: project gradient
        # We want to minimize max(conv) subject to sum(a)=1, a>=0
        # Use projected gradient: step in -grad direction, then project
        # But we also want to maintain positivity, so use multiplicative update

        # Multiplicative update: a[j] *= exp(-eta * grad[j] / M)
        # This reduces a[j] more where grad is large (where a[k_star-j] is large)
        log_update = -eta * grad / (M + 1e-30)
        log_update = np.clip(log_update, -5, 5)  # prevent extreme updates
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Slightly increase step if improving
            eta = min(0.5, eta * 1.02)
        else:
            # Decrease step
            eta = max(0.001, eta * 0.8)
            # Keep a as is, or blend
            a = 0.9 * a + 0.1 * a_new
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        # Periodic perturbation to escape local minima
        if iters % 500 == 0:
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            eta = 0.1

        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.5f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=200, verbose=True, label="CD"):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            best_local = best_score
            best_t = 0.0
            # Try moving mass from larger to smaller
            if a[i] < a[j]:
                i, j = j, i

            for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                t = frac * a[i]
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 100:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Diverse initializations + Bussgang ===
    print(f"\n=== Phase 1: Diversified Init + Bussgang (best={best_score:.8f}) ===")
    initializations = {
        "parent": parent.copy(),
    }

    # BLISS init at parent length
    try:
        initializations["bliss"] = bliss_init(n_parent, iters=300, seed=seed)
        initializations["bliss2"] = bliss_init(n_parent, iters=300, seed=seed + 1)
    except Exception as e:
        print(f"  BLISS failed: {e}")

    # Chirp init
    initializations["chirp"] = chirp_init(n_parent, seed=seed)
    # Raised cosine
    initializations["raised_cos"] = raised_cosine_init(n_parent)
    # Gaussian
    initializations["gauss"] = gauss_init(n_parent, sigma_frac=0.4)

    for name, init in initializations.items():
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_bg = min(rem * 0.15, 80.0)
        print(f"  Bussgang [{name}], budget={t_bg:.0f}s")
        a_bg, sc_bg = bussgang_iteration(init, t_bg, verbose=False, tau=0.5, label=f"BG-{name}")
        best_seq, best_score = check_and_update(a_bg, f"bussgang_{name}", best_seq, best_score)

    # === Phase 2: LP Fictitious Play on best ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 540.0)
    print(f"  Budget: {t2:.0f}s, n={len(best_seq)}")
    a_lp1, sc_lp1 = lp_fictitious_play(best_seq, t2, step=0.5, label="LP-s0.5")
    best_seq, best_score = check_and_update(a_lp1, "lp_s0.5", best_seq, best_score)

    # === Phase 3: LP with decreasing step ===
    print(f"\n=== Phase 3: LP Decreasing Step (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if rem > 60:
        print(f"  Budget: {t3:.0f}s")
        a_lp2, sc_lp2 = lp_decreasing_step(best_seq, t3, verbose=True, label="LP-dec")
        best_seq, best_score = check_and_update(a_lp2, "lp_dec", best_seq, best_score)

    # === Phase 4: Peak-directed search ===
    print(f"\n=== Phase 4: Peak-Directed Search (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 150.0)
    if rem > 30:
        print(f"  Budget: {t4:.0f}s")
        a_pk, sc_pk = peak_directed_search(best_seq, t4, verbose=True, label="PeakDir")
        best_seq, best_score = check_and_update(a_pk, "peak_dir", best_seq, best_score)

    # === Phase 5: L-BFGS polish ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 500, 2000, 5000]:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.08, 60.0)
        if rem < 20:
            break
        print(f"  L-BFGS T={T_val}, budget={t5:.0f}s")
        a_lb, sc_lb = lbgfs_optimize(best_seq, t5, T=T_val, maxiter=3000, verbose=True,
                                     label=f"LBFGS")
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    alt_lengths = [n_parent + 2, n_parent - 2, n_parent + 4, n_parent - 4,
                   n_parent + 8, n_parent - 8, n_parent + 16, n_parent - 16]
    # Also try some different scales
    alt_lengths += [int(n_parent * 1.25), int(n_parent * 0.8), int(n_parent * 1.5), int(n_parent * 0.7)]
    alt_lengths = [n for n in alt_lengths if 128 <= n <= 4096 and n != n_parent]
    # Remove duplicates, keep order
    seen = set()
    alt_lengths_unique = []
    for n in alt_lengths:
        if n not in seen:
            seen.add(n)
            alt_lengths_unique.append(n)
    alt_lengths = alt_lengths_unique

    for alt_n in alt_lengths[:6]:  # Limit to 6 alternatives
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alt = min(rem * 0.10, 80.0)
        if t_alt < 15:
            continue
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
        # Quick LP + L-BFGS
        a_alt, sc_alt = lp_fictitious_play(init_alt, t_alt * 0.5, step=0.4,
                                           verbose=False, label=f"LP-n{alt_n}")
        a_alt, sc_alt = lbgfs_optimize(a_alt, t_alt * 0.4, T=500, maxiter=2000,
                                       verbose=False, label=f"LB-n{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.20, 120.0)
    if rem > 20:
        print(f"  Budget: {t7:.0f}s")
        a_final, sc_final = lp_fictitious_play(best_seq, t7, step=0.15,
                                               verbose=True, label="LP-final")
        best_seq, best_score = check_and_update(a_final, "lp_final", best_seq, best_score)

    # === Phase 8: Coordinate descent ===
    print(f"\n=== Phase 8: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.15, 100.0)
    if rem > 20:
        print(f"  Budget: {t8:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t8, n_pairs=300, verbose=True, label="CD")
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # === Phase 9: Final micro-polish ===
    print(f"\n=== Phase 9: Final Micro-Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.5, 60.0)
        # Very small step LP
        a_micro, sc_micro = lp_fictitious_play(best_seq, t9, step=0.05,
                                               verbose=False, label="LP-micro")
        best_seq, best_score = check_and_update(a_micro, "micro", best_seq, best_score)
        # One more L-BFGS
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_last, sc_last = lbgfs_optimize(best_seq, rem2 * 0.8, T=1000,
                                             maxiter=2000, verbose=True, label="FINAL-LB")
            best_seq, best_score = check_and_update(a_last, "final_lb", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
