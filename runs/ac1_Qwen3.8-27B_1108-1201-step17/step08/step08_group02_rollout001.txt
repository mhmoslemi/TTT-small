Actually, I realize there's another important consideration: the **exact structure** of the LP.

The constraint (f*g)[k] <= M means: sum_i f[i] * g[k-i] <= M, where the sum is over valid i.

This is a linear constraint in g. The coefficient matrix A has rows corresponding to k and columns to g's indices. A[k, j] = f[j] if j is a valid index for the k-th convolution (i.e., k-j is in [0, n-1]), else 0.

More specifically, for the full convolution of f (length n) and g (length n):
(f*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * g[k-i]

So the constraint is: sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * g[k-i] <= M

Let j = k-i, then: sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j] <= M

So the row k of A is: A[k, j] = f[k-j] for j in [max(0, k-n+1), min(n-1, k)], and 0 otherwise.

This is a banded matrix (each row has at most n nonzeros, and the band width varies).

For n=1152 and up to 2303 constraints, the LP has about 1152 * 1152 ≈ 1.3M nonzeros. This is manageable for SCS.

"""
LP-based cutting plane optimization with adaptive constraint selection,
proper line search, perturbation restarts, and coordinate descent polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * target_sum / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, peak_indices, n):
    """Build the constraint matrix A where A[k, j] gives coefficient of g[j] in (f*g)[k]."""
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            A[row_i, j] = f[i]
    return A


def solve_lp_direction(f, n, peak_indices, M, solver_prefs=None, max_iters=300000, eps=1e-9):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns g vector or None.
    """
    K = len(peak_indices)
    if K == 0:
        return np.ones(n) / n

    A = build_constraint_matrix(f, peak_indices, n)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)

    if solver_prefs is None:
        solver_prefs = [cp.ECOS, cp.SCS, cp.CLARABEL]

    for solver in solver_prefs:
        try:
            prob.solve(solver=solver, max_iters=max_iters, eps=eps)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)):
                    return g_val
        except Exception:
            continue

    # Last resort: SCS with relaxed settings
    try:
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.SCS, max_iters=1000000, eps=1e-9)
        if g.value is not None:
            g_val = np.maximum(g.value, 0.0)
            if np.all(np.isfinite(g_val)):
                return g_val
    except Exception:
        pass

    return None


def line_search_1d(f, h, M_current, max_evals=25):
    """
    Golden-section + parabolic line search along f -> h direction.
    Minimize max(conv((1-t)*f + t*h)) for t in (0, 1].
    Returns (best_t, best_obj).
    """
    def obj_at(t):
        ft = (1.0 - t) * f + t * h
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-30:
            return np.inf
        ft = ft / s
        return fast_convolve(ft).max()

    best_t = 0.0
    best_obj = M_current
    eval_count = 0

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0

    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = obj_at(c)
    fd = obj_at(d)
    eval_count += 2

    if fc < best_obj:
        best_obj = fc
        best_t = c
    if fd < best_obj:
        best_obj = fd
        best_t = d

    for _ in range(12):
        if eval_count >= max_evals:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = obj_at(c)
            eval_count += 1
            if fc < best_obj:
                best_obj = fc
                best_t = c
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = obj_at(d)
            eval_count += 1
            if fd < best_obj:
                best_obj = fd
                best_t = d

    # Refine around best_t
    for dt in [best_t * 0.9, best_t * 1.1, best_t * 0.8, best_t * 1.2]:
        if dt <= 0 or dt > 0.99:
            continue
        eval_count += 1
        if eval_count > max_evals:
            break
        o = obj_at(dt)
        if o < best_obj - 1e-15:
            best_obj = o
            best_t = dt

    # Try some additional discrete points for robustness
    for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        if abs(t_try - best_t) < 0.01:
            continue
        o = obj_at(t_try)
        if o < best_obj - 1e-15:
            best_obj = o
            best_t = t_try

    return best_t, best_obj


def lp_cutting_plane(f_init, time_limit, label="lp", initial_frac=0.95,
                     final_frac=0.88, max_active=500, patience=30,
                     lp_eps=1e-9, solver_prefs=None):
    """
    LP-based cutting plane optimization.
    Adaptive active set expansion with proper line search.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    frac = initial_frac
    rng = np.random.default_rng(42)

    # Maintain a growing active set (all constraints seen so far)
    all_active = np.array([], dtype=int)
    b_conv = fast_convolve(f)
    M = b_conv.max()
    init_peaks = np.where(b_conv >= frac * M)[0]
    all_active = init_peaks.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        b_conv = fast_convolve(f)
        M = b_conv.max()

        # Update active set: include peaks above current threshold
        threshold = frac * M
        current_peaks = np.where(b_conv >= threshold)[0]
        # Also keep previously active constraints (cutting plane)
        all_active = np.unique(np.concatenate([all_active, current_peaks]))

        # Cap the number of active constraints
        if len(all_active) > max_active:
            # Prioritize the highest current peaks
            peak_vals = b_conv[all_active]
            order = np.argsort(peak_vals)[::-1]
            all_active = all_active[order[:max_active]]

        # Solve LP
        g = solve_lp_direction(f, n, all_active, M,
                               solver_prefs=solver_prefs, eps=lp_eps)

        if g is None:
            stuck += 1
            if stuck > patience:
                # Perturb and reset
                f = f + rng.standard_normal(n) * 0.001 * (f + 1e-8)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                all_active = np.where(fast_convolve(f) >= 0.95 * fast_convolve(f).max())[0]
                stuck = 0
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            # LP says no improvement possible with current constraints
            stuck += 1
            if stuck <= patience // 2:
                # Try expanding the active set
                frac = max(final_frac, frac - 0.02)
                threshold = frac * M
                more_peaks = np.where(b_conv >= threshold)[0]
                all_active = np.unique(np.concatenate([all_active, more_peaks]))
                if len(all_active) > max_active:
                    peak_vals = b_conv[all_active]
                    order = np.argsort(peak_vals)[::-1]
                    all_active = all_active[order[:max_active]]
                continue
            else:
                # Perturb
                f = f + rng.standard_normal(n) * (0.002 + 0.001 * stuck / patience) * (f + 1e-8)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                all_active = np.where(fast_convolve(f) >= 0.98 * fast_convolve(f).max())[0]
                stuck = 0
                frac = initial_frac
                continue

        # Line search
        h = g / S_g
        best_t, best_line_obj = line_search_1d(f, h, M, max_evals=20)

        if best_t <= 1e-6:
            stuck += 1
            if stuck > patience:
                f = f + rng.standard_normal(n) * 0.002 * (f + 1e-8)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                all_active = np.where(fast_convolve(f) >= 0.95 * fast_convolve(f).max())[0]
                stuck = 0
            continue

        # Update f
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            break

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        # Gradually lower the fraction threshold
        if it % 15 == 0:
            frac = max(final_frac, frac - 0.005)

        if it % 20 == 0:
            score = 2.0 * n * best_obj
            print(f"  {label} it={it} obj={best_obj:.12f} score={score:.6f} "
                  f"active={len(all_active)} S_g={S_g:.4f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def multiplicative_weights_optimize(f_init, time_limit, lr_init=0.02, label="mw"):
    """
    Multiplicative weights / mirror descent with exact subgradient.
    Uses uniform weights on ALL peak positions (not softmax).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Find all peaks within 1% of max
        peaks = np.where(b >= 0.99 * M)[0]

        # Exact subgradient: for each peak position k, the gradient contribution
        # of a[i] is a[k-i] (summed over all peak k's)
        grad = np.zeros(n)
        for k in peaks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                grad[j] += f[i]

        # Normalize gradient
        gmax = grad.max()
        if gmax > 0:
            grad /= gmax

        # Multiplicative update
        f_new = f * np.exp(-lr * grad)
        s = f_new.sum()
        if s > 1e-30:
            f_new /= s
            f = f_new
        else:
            break

        if it % 10 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.08)
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.9
                    no_improve = 0

            if it % 50 == 0:
                score = 2.0 * n * best_obj
                print(f"  {label} it={it} obj={best_obj:.12f} score={score:.6f} "
                      f"lr={lr:.5f} peaks={len(peaks)} t={time.monotonic()-t0:.0f}s")

            if no_improve > 50:
                f = best_f + rng.standard_normal(n) * 0.001 * (best_f + 1e-8)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                lr = lr_init
                no_improve = 0

    return best_f, best_obj


def coordinate_descent(f_init, time_limit, label="cd", n_rounds_max=50):
    """
    Coordinate descent: iterate over elements, try scaling each by various factors.
    Focus on elements that contribute most to peak convolution values.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0])

    round_num = 0
    while time.monotonic() - t0 < time_limit and round_num < n_rounds_max:
        round_num += 1
        improved = False
        b = fast_convolve(f)
        M = b.max()
        peaks = np.where(b >= 0.95 * M)[0]

        # Order elements by their contribution to peaks
        contrib = np.zeros(n)
        for k in peaks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                contrib[j] += f[i]
        order = np.argsort(contrib)[::-1]

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(idx)
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_val = new_val
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    n_parent = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.50, 800.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP cutting plane (n={n_parent}, budget={t1:.0f}s) ===")
        f1, obj1 = lp_cutting_plane(parent, time_limit=t1, label="lp1",
                                     initial_frac=0.95, final_frac=0.88,
                                     max_active=400, patience=40, lp_eps=1e-10)
        vs1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(vs1) and vs1 < best_score:
            best_score = vs1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp1]: {vs1:.6f}")

    # === Phase 2: Deeper LP with more constraints (15% time) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 40:
        n_b = len(best_seq)
        print(f"\n=== Phase 2: LP deep (n={n_b}, budget={t2:.0f}s) ===")
        f2, obj2 = lp_cutting_plane(best_seq, time_limit=t2, label="lp2",
                                     initial_frac=0.92, final_frac=0.85,
                                     max_active=600, patience=35, lp_eps=1e-10)
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp2]: {vs2:.6f}")

    # === Phase 3: Multiplicative weights polish (8% time) ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.08, 100.0)
    if t3 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: Multiplicative weights (n={n_b}, budget={t3:.0f}s) ===")
        f3, obj3 = multiplicative_weights_optimize(best_seq, time_limit=t3,
                                                     lr_init=0.015, label="mw")
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [mw]: {vs3:.6f}")

    # === Phase 4: Perturbation + LP restart (12% time) ===
    rem = deadline - time.monotonic()
    if rem > 80:
        n_b = len(best_seq)
        rng = np.random.default_rng(123)
        for pert_iter in range(3):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t4 = min(rem * 0.08, 80.0)
            if t4 < 25:
                break
            print(f"\n=== Phase 4.{pert_iter+1}: Perturbation + LP (budget={t4:.0f}s) ===")
            noise_scale = 0.003 + 0.002 * pert_iter
            f_pert = best_seq + rng.standard_normal(n_b) * noise_scale * (best_seq + 1e-10)
            f_pert = np.maximum(f_pert, 0.0)
            f_pert /= f_pert.sum()
            f4, obj4 = lp_cutting_plane(f_pert, time_limit=t4, label=f"pert{pert_iter+1}",
                                         initial_frac=0.95, final_frac=0.90,
                                         max_active=350, patience=25, lp_eps=1e-10)
            vs4 = float(evaluate_sequence(f4.tolist()))
            if np.isfinite(vs4) and vs4 < best_score:
                best_score = vs4
                best_seq = f4.copy()
                print(f"  NEW BEST [pert{pert_iter+1}]: {vs4:.6f}")

    # === Phase 5: Multi-length exploration (10% time) ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for delta in [-100, -50, 50, 100, 150]:
        tl = n_base + delta
        if 512 <= tl <= 2000 and tl != n_base:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t5 = min(rem * 0.08, 80.0)
        if t5 < 20:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\n=== Phase 5: LP n={target_n} (budget={t5:.0f}s) ===")
        f5, obj5 = lp_cutting_plane(f_interp, time_limit=t5, label=f"n{target_n}",
                                     initial_frac=0.95, final_frac=0.88,
                                     max_active=350, patience=20, lp_eps=1e-10)
        vs5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(vs5) and vs5 < best_score:
            best_score = vs5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {vs5:.6f}")

    # === Phase 6: Coordinate descent polish (5% time) ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.05, 60.0)
    if t6 > 10:
        n_b = len(best_seq)
        print(f"\n=== Phase 6: Coordinate descent (n={n_b}, budget={t6:.0f}s) ===")
        f6, obj6 = coordinate_descent(best_seq, time_limit=t6, label="cd", n_rounds_max=20)
        vs6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(vs6) and vs6 < best_score:
            best_score = vs6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {vs6:.6f}")

    # === Phase 7: Final LP burst with all constraints (5% time) ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.10, 100.0)
    if t7 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 7: Final LP all-constraints (n={n_b}, budget={t7:.0f}s) ===")
        f7, obj7 = lp_cutting_plane(best_seq, time_limit=t7, label="lp_final",
                                     initial_frac=0.90, final_frac=0.85,
                                     max_active=800, patience=30, lp_eps=1e-10)
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp_final]: {vs7:.6f}")

    # === Phase 8: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.10, 50.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final CD (budget={t8:.0f}s) ===")
        f8, obj8 = coordinate_descent(best_seq, time_limit=t8, label="cd_final", n_rounds_max=15)
        vs8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(vs8) and vs8 < best_score:
            best_score = vs8
            best_seq = f8.copy()
            print(f"  NEW BEST [cd_final]: {vs8:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
