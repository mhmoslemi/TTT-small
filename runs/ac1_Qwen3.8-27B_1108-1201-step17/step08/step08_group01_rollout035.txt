def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    _lp_solver = None
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name

    for k in peaks:
        row = np.zeros(n)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for j in range(i_start, i_end + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows)

def solve_lp_best_response(a, solver=None, top_k=None):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Optionally, only enforce the top_k constraints (for speed).
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    
    # Compute full conv to find which constraints are likely binding
    ab_prev = fast_convolve(a)  # Use a*a as proxy for which k are peaks
    if top_k is not None and top_k < 2 * n - 1:
        peak_indices = np.argpartition(ab_prev, -top_k)[-top_k:]
    else:
        peak_indices = np.arange(2 * n - 1)
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    # Build constraints for selected peaks
    constraints = [cp.sum(b_var) == 1]
    for k in peak_indices:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # (a * b)[k] = sum_{j=i_start}^{i_end} a[k-j] * b[j]
        coeffs = a[k - np.arange(i_start, i_end + 1)]
        constraints.append(coeffs @ b_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(t_var.value)

def solve_lp_full(a, solver=None):
    """Solve LP with all constraints (slower but more accurate)."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [cp.sum(b_var) == 1]
    # Build full convolution matrix
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        coeffs = a[k - np.arange(i_start, i_end + 1)]
        constraints.append(coeffs @ b_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(t_var.value)

def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

def lp_guided_optimize(a0, time_limit, solver=None, verbose=True, label=""):
    """
    LP-guided majorization-minimization with line search.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        # Solve LP
        b_star, lp_val = solve_lp_full(a, solver)
        if b_star is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search: minimize f((1-t)*a + t*b_star) for t in [0, 1]
        # Try several values of t
        best_t = 0.0
        best_f = best_score
        
        # Adaptive step sizes: start large, shrink if not improving
        t_candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.4, 0.35, 0.25, 0.15, 0.7, 0.8]
        # Sort by preference (larger first for faster convergence)
        t_candidates = sorted(t_candidates, reverse=True)
        
        found_improvement = False
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * b_star
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            f_trial = compute_objective(a_trial)
            if f_trial < best_f:
                best_f = f_trial
                best_t = t
                found_improvement = True
        
        if found_improvement:
            a = (1.0 - best_t) * a + best_t * b_star
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller steps
            if no_improve > 20:
                # Perturb and restart
                rng_local = np.random.default_rng(it)
                a_pert = a + rng_local.standard_normal(n) * 0.01
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                if compute_objective(a_pert) < compute_objective(a):
                    a = a_pert
                    no_improve = 0
                else:
                    no_improve = 0  # Reset counter to keep trying
                    if no_improve > 50:
                        break
            if no_improve > 100:
                break
        
        cur_score = compute_objective(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-opt {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} cur={cur_score:.6f} lp_val={lp_val:.6f}")
    
    return best_a, best_score, it

def mirror_descent_momentum(a0, time_limit, lr_init=0.005, top_k=50, verbose=False, label=""):
    """Mirror descent with momentum for the convex relaxation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    momentum = 0.9
    grad_avg = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break
        
        # Compute subgradient with top-k peaks and softmax weighting
        k_act = min(top_k, 2 * n - 1)
        # Use softmax over all peaks (not just top-k) for smoother gradient
        temp = 0.005 * max(M, 1e-10)
        w = np.exp((conv - M) / temp)
        w = w / w.sum()
        
        # Subgradient
        grad = np.zeros(n)
        # For efficiency, only use peaks with significant weight
        significant = w > 1e-6 / k_act
        for k in np.where(significant)[0]:
            if w[k] < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[k] * a[k - j]
        
        # Momentum
        grad_avg = momentum * grad_avg + (1 - momentum) * grad
        
        # Mirror descent update
        log_a = np.log(a + 1e-30) - lr * grad_avg
        a_new = np.exp(log_a - log_a.max())
        a_new = a_new / a_new.sum()
        a = a_new
        
        lr *= 0.99995
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-mom {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.8f}")
    
    return best_a, best_score

def coordinated_peak_reduction(a0, time_limit, verbose=False, label=""):
    """
    Targeted peak reduction: identify peaks, compute coordinated updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Find peaks
        peak_threshold = M * 0.95
        peaks = np.where(conv >= peak_threshold)[0]
        
        if len(peaks) == 0:
            break
        
        # For each peak, compute the "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in peaks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for j in range(i_start, i_end + 1):
                pressure[j] += a[k - j] * (conv[k] / M)
        
        # Reduce coordinates with high pressure, increase those with low
        # Use a soft update
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        
        # Scale factor
        scale = 0.02 / p_max
        a_new = a * (1.0 - scale * pressure)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        new_score = compute_objective(a_new)
        if new_score < score:
            a = a_new
        else:
            # Try smaller step
            a_new = a * (1.0 - 0.5 * scale * pressure)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            new_score = compute_objective(a_new)
            if new_score < score:
                a = a_new
            else:
                # No improvement, break
                break
    
    return best_a, best_score

def fast_coord_descent(a0, time_limit, n_trials=8, verbose=False, label=""):
    """Fast coordinate descent with sensitivity analysis."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]  # a[k-i] for k = i..i+n-1 is a[n-1..0] reversed? No.
            # For position k in [i, i+n-1], the coefficient of a[i] in conv[k]
            # is a[k-i]. So the affected values change by 2*delta*a[k-i].
            # a[k-i] for k = i, i+1, ..., i+n-1 gives a[0], a[1], ..., a[n-1].
            # So a_shifted = a (in order).
            
            best_delta = 0.0
            best_new_max = M
            
            for trial in range(n_trials):
                if n_trials == 8:
                    deltas = [-0.4, -0.2, -0.1, -0.05, 0.05, 0.1, 0.2, 0.4]
                elif n_trials == 12:
                    deltas = [-0.5, -0.3, -0.2, -0.1, -0.05, -0.02, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
                elif n_trials == 16:
                    deltas = [-0.5, -0.4, -0.3, -0.2, -0.15, -0.1, -0.05, -0.02,
                              0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
                else:
                    deltas = np.linspace(-0.4, 0.4, n_trials)
                
                delta = a[i] * deltas[trial]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * a_shifted
                # Handle the diagonal case
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                
                changes = 2.0 * best_delta * a_shifted.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]
            
            best_delta = 0.0
            best_new_max = M
            
            for factor in [0.97, 0.94, 0.91, 0.88, 1.03, 1.06, 1.09, 1.12]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = (affected + 2.0 * delta * a_shifted).copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] = max(a[i] + best_delta, 0.0)
                
                changes = (2.0 * best_delta * a_shifted).copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved and sweeps > 10:
            break
    
    return best_a, best_score, sweeps

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():

    7. Final polish

            parent = p / p.sum()

            if np.isfinite(p_score):

                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

        parent = np.ones(1024) / 1024

        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform init: n=1024, score={best_score:.6f}")

    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-guided majorization-minimization with line search (main engine)
2. Mirror descent with momentum for smooth exploration
3. Coordinated peak reduction via sensitivity
4. Fast and fine coordinate descent for local polishing
5. Multi-length exploration with adaptive time allocation

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import coo_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * float(c.max())


def compute_objective_raw(a):
    """Compute the full evaluation: 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * float(c.max()) / (s * s)


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():

    _lp_solver = None

        try:

            prob.solve(solver=solver_name, verbose=False)

                _lp_solver = solver_name

        except Exception:

def solve_lp_best_response(a, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]
    subject to sum(b)=1, b>=0. Returns (b, lp_value).
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a)

    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()

    constraints = [cp.sum(b_var) == 1]

    # Build constraints: for each k, (a*b)[k] <= t
    # (a*b)[k] = sum_{j} a[k-j] * b[j] for valid j
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        row = np.zeros(n)
        row[i_start:i_end + 1] = a[k - np.arange(i_start, i_end + 1)]
        constraints.append(row @ b_var <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue

    status = prob.status
    if status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(t_var.value)

    no_improve = 0
    step_size_decay = 1.0

        # Scale by decay factor

def mirror_descent_momentum(a0, time_limit, lr_init=0.003, verbose=False, label=""):
    """Mirror descent (multiplicative weights) with momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    momentum = 0.85
    grad_avg = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-9:
                break

        # Softmax weighting over peaks
        temp = 0.003 * max(M, 1e-10)
        w = np.exp((conv - M) / temp)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Subgradient computation
        grad = np.zeros(n)
        significant = w > 1e-8
        peak_indices = np.where(significant)[0]
        for k in peak_indices:
            if w[k] < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[k] * a[k - j]

        # Momentum
        grad_avg = momentum * grad_avg + (1.0 - momentum) * grad

        # Mirror descent update (multiplicative)
        log_a = np.log(a + 1e-30)
        a_new = np.exp(log_a - lr * grad_avg)
        a_new = a_new / a_new.sum()
        a = a_new

        lr *= 0.9999

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-mom {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.8f}")

    return best_a, best_score


def coordinated_peak_reduction(a0, time_limit, verbose=False, label=""):
    """Targeted peak reduction using pressure-based updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3.0:
            break

        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Find peaks (top few)
        peak_threshold = M * 0.9
        peaks = np.where(conv >= peak_threshold)[0]
        if len(peaks) == 0:
            break

        # Compute pressure: how much each coordinate contributes to peaks
        pressure = np.zeros(n)
        for k in peaks:
            weight = (conv[k] - M * 0.9) / (M * 0.1 + 1e-30)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for j in range(i_start, i_end + 1):
                pressure[j] += weight * a[k - j]

        p_max = pressure.max()
        if p_max < 1e-20:
            break

        # Try several scales
        improved = False
        for scale in [0.03, 0.02, 0.01, 0.005]:
            a_new = a * (1.0 - scale * pressure / p_max)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            new_score = compute_objective(a_new)
            if new_score < score - 1e-14:
                a = a_new
                improved = True
                break

        if not improved:
            break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

    n = len(a0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            a_shifted = a  # a[k-i] for k=i..i+n-1 is a[0]..a[n-1]

            best_delta = 0.0
            best_new_max = M

            # Generate trial deltas
            if n_trials <= 8:
                deltas = np.array([-0.4, -0.2, -0.1, -0.05, 0.05, 0.1, 0.2, 0.4])
            elif n_trials <= 12:
                deltas = np.array([-0.5, -0.3, -0.2, -0.1, -0.05, -0.02,
                                   0.02, 0.05, 0.1, 0.2, 0.3, 0.5])
            else:
                deltas = np.linspace(-0.4, 0.4, n_trials)

            for delta_frac in deltas:
                delta = a[i] * delta_frac
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)

                changes = 2.0 * best_delta * a_shifted.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0
f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            a_shifted = a

            best_delta = 0.0
            best_new_max = M

            for factor in [0.97, 0.94, 0.91, 0.88, 1.03, 1.06, 1.09, 1.12]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = (affected + 2.0 * delta * a_shifted).copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] = max(a[i] + best_delta, 0.0)

                changes = (2.0 * best_delta * a_shifted).copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break

    return best_a, best_score, sweeps


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate via the actual evaluation function and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def subgradient_polyak(a0, time_limit, t_lower=1.0, verbose=False, label=""):
    """Subgradient method with Polyak step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Polyak step: (f(a) - f*) / ||g||^2
        f_star_est = t_lower  # Lower bound on optimal value of max(conv) with sum=1
        gap = M - f_star_est / (2.0 * n)
        if gap <= 0:
            break

        # Subgradient from top peaks
        k_peak = int(np.argmax(conv))
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        grad = np.zeros(n)
        grad[i_start:i_end + 1] = 2.0 * a[k_peak - np.arange(i_start, i_end + 1)]

        grad_norm_sq = float(grad @ grad)
        if grad_norm_sq < 1e-30:
            break

        step = gap / grad_norm_sq
        step = min(step, 0.1)  # Cap step size

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        new_score = compute_objective(a_new)
        if new_score < score:
            a = a_new
        else:
            # Halve the step
            a_new = a - 0.5 * step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            if compute_objective(a_new) < score:
                a = a_new
            else:
                break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining LP-guided MM, mirror descent,
    peak reduction, and coordinate descent for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform init: n=1024, score={best_score:.6f}")

    n = len(parent)

    # === Phase 1: Quick coordinate descent for baseline ===
    print("\n=== Phase 1: Quick CD ===")
    t1_budget = 30.0
    a_p1, s_p1, _ = fast_coord_descent(parent, t1_budget, n_trials=8, label="p1")
    best_seq, best_score = check_and_update(a_p1, best_seq, best_score, "p1_cd")
    print(f"  After Phase 1: {best_score:.6f}")

    # === Phase 2: LP-guided MM (main engine) ===
    print(f"\n=== Phase 2: LP-guided MM (best={best_score:.6f}) ===")
    t2_total = min(700.0, (deadline - time.monotonic()) * 0.45)
    t2_start = time.monotonic()
    n_restarts = 5
    current = best_seq.copy()

    for restart in range(n_restarts):
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        t_alloc = min(t2_total / max(1, n_restarts - restart), rem * 0.5)
        print(f"  LP-opt restart {restart + 1}/{n_restarts}, budget={t_alloc:.0f}s")
        a_lp, s_lp, it = lp_guided_optimize(current, t_alloc, solver=solver,
                                            verbose=True, label=f"r{restart}")
        best_seq, best_score = check_and_update(a_lp, best_seq, best_score, f"lp_r{restart}")

        # Perturb for next restart if needed
        if best_score > 1.503:
            current = best_seq + rng.standard_normal(len(best_seq)) * 0.003
            current = np.maximum(current, 0)
            current = current / current.sum()
        else:
            current = best_seq.copy()

    elapsed_p2 = time.monotonic() - t2_start
    print(f"  After Phase 2 ({elapsed_p2:.0f}s): {best_score:.6f}")

    # === Phase 3: Mirror descent with momentum ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 30:
        t3 = min(rem * 0.5, 150.0)
        print(f"  MD budget={t3:.0f}s")
        a_md, s_md = mirror_descent_momentum(best_seq, t3, lr_init=0.003, verbose=True, label="md")
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "md")
        print(f"  After Phase 3: {best_score:.6f}")

    # === Phase 4: Coordinated peak reduction ===
    print(f"\n=== Phase 4: Peak Reduction (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t4 = min(rem * 0.5, 100.0)
        a_pr, s_pr = coordinated_peak_reduction(best_seq, t4, verbose=True, label="peak")
        best_seq, best_score = check_and_update(a_pr, best_seq, best_score, "peak_red")
        print(f"  After Phase 4: {best_score:.6f}")

    # === Phase 5: Another LP round (fresh start from current best) ===
    print(f"\n=== Phase 5: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t5 = min(rem * 0.6, 200.0)
        a_lp2, s_lp2, it = lp_guided_optimize(best_seq, t5, solver=solver,
                                              verbose=True, label="lp2")
        best_seq, best_score = check_and_update(a_lp2, best_seq, best_score, "lp_r2")
        print(f"  After Phase 5: {best_score:.6f}")

    # === Phase 6: Fine coordinate descent ===
    print(f"\n=== Phase 6: Fine CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t6 = min(rem * 0.6, 150.0)
        a_fine, s_fine, _ = fine_coord_descent(best_seq, t6, label="fine")
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")
        print(f"  After Phase 6: {best_score:.6f}")

    # === Phase 7: Try alternative length ===
    print(f"\n=== Phase 7: Alt Length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 40:
        n_best = len(best_seq)
        alt_lengths = [n_best + 64, n_best - 64, n_best + 128, n_best - 128]
        for alt_n in alt_lengths:
            if alt_n < 256 or alt_n > 2048:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_alt = min(rem * 0.35, 50.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = fast_coord_descent(init_alt, t_alt, n_trials=10, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")
        print(f"  After Phase 7: {best_score:.6f}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 40
    if rem > 30:
        t8 = min(rem * 0.6, 80.0)
        a_lp3, s_lp3, it = lp_guided_optimize(best_seq, t8, solver=solver,
                                              verbose=True, label="final_lp")
        best_seq, best_score = check_and_update(a_lp3, best_seq, best_score, "final_lp")

    # === Phase 9: Final fine polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 15
    if rem > 15:
        t9 = min(rem * 0.7, 40.0)
        a_fin, s_fin, _ = fine_coord_descent(best_seq, t9, label="final")
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification with the actual evaluation function
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
