"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines: mirror descent (multiplicative weight), LP fictitious play,
active-set equalization, L-BFGS smoothed optimization, and length search.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0. Returns (g, dual_vars)."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return (prev_g if prev_g is not None else f.copy()), None
    if g.value is None:
        return (prev_g if prev_g is not None else f.copy()), None
    g_val = np.maximum(g.value, 0)
    # Try to get dual variables
    dual = None
    try:
        dual = constraints[0].dual_value
        if dual is not None:
            dual = np.maximum(dual, 0)
    except Exception:
        pass
    return g_val, dual

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    solver = cp.CLARABEL
    try:
        prob.solve(solver=solver, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return (prev_g if prev_g is not None else f.copy())
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

Combines: LP fictitious play with line search, mirror descent (multiplicative
weight updates), L-BFGS on smoothed log-sum-exp objective, and multi-length
exploration. Designed to squeeze out the last 0.1% improvement.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    """LP fictitious play with adaptive line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum

        # Line search: try several step sizes
        best_local_score = best_score
        best_local_t = step
        candidates = [step * 0.5, step * 0.75, step, step * 1.25, step * 1.5, step * 2.0, 0.8, 0.9, 0.95]
        for t_c in candidates:
            if t_c > 0.98 or t_c < 0.01:
                continue
            a_trial = (1.0 - t_c) * a + t_c * g_norm
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_local_score - 1e-14:
                best_local_score = sc
                best_local_t = t_c

        step = best_local_t
        a_new = (1.0 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 150:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters + 7777)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True, label="MD"):
    """Mirror descent (multiplicative weight) on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = n * 0.2
    stale_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = conv.max()

        # Active set: within 0.3% of max
        threshold = max_val * 0.997
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient (average over active set)
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if hi >= lo:
                indices = np.arange(lo, hi + 1)
                a_idx = k - indices
                grad[indices] += 2.0 * a[a_idx]
        grad /= len(active)

        grad_tan = grad - grad.mean()

        # Multiplicative weight update
        a_new = a * np.exp(-eta * grad_tan)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        a = a_new
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stale_count = 0
        else:
            stale_count += 1

        # Adaptive step size
        if iters % 200 == 0:
            if stale_count > 100:
                eta *= 0.85
                if stale_count > 300:
                    rng = np.random.default_rng(iters + 4242)
                    a = best_a + rng.standard_normal(n) * 0.003
                    a = np.maximum(a, 1e-30)
                    a = a / a.sum()
                    stale_count = 0
            elif score < best_score:
                eta = min(eta * 1.02, n * 1.0)

            if verbose and iters % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.1f}, stale={stale_count}, time={elapsed:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def chunk_swap_perturb(a0, n_swaps=3, verbose=False):
    """Apply random chunk swaps to perturb a sequence."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    rng = np.random.default_rng(12345)
    for _ in range(n_swaps):
        chunk_size = max(1, n // 20)
        i = rng.integers(0, n - chunk_size)
        j = rng.integers(0, n - chunk_size)
        if j < i:
            i, j = j, i
        if j - i < chunk_size:
            continue
        temp = a[i:i+chunk_size].copy()
        a[i:i+chunk_size] = a[j:j+chunk_size]
        a[j:j+chunk_size] = temp
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    if verbose:
        print(f"  Chunk swap: score went from {compute_score(a0):.8f} to {compute_score(a):.8f}")
    return a


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
