"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP fictitious play with symmetrization, subgradient descent,
and L-BFGS polishing across multiple sequence lengths.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.linalg import toeplitz
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def symmetrize(a):
    return 0.5 * (a + a[::-1].copy())


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def evaluate_smoothed(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted)[np.isfinite(shifted)].sum())
    return (2.0 * n / T) * np.exp(T * lse)


def gradient_smoothed(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lp_fictitious_play(a0, time_limit, step=0.4, symmetrize_flag=True, verbose=True, label="LP-FP"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        if symmetrize_flag:
            g_norm = symmetrize(g_norm)
            ss = g_norm.sum()
            if ss < 1e-15:
                g_norm = a.copy()
            else:
                g_norm = g_norm / ss
        
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        if symmetrize_flag:
            a_new = symmetrize(a_new)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

            step = max(0.05, step * 0.8)
            rng = np.random.default_rng(iters * 7919)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            if symmetrize_flag:
                a = symmetrize(a)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
            step = min(0.5, step * 1.5)
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")
    
    return best_a, best_score


def subgradient_descent(a0, time_limit, verbose=True):
    """Subgradient method minimizing max_k (a*a)[k] on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = 0.01
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = np.argmax(conv)
        M = conv[k_star]
        
        # Subgradient of (a*a)[k_star] w.r.t. a:
        # grad[j] = 2 * a[k_star - j] if 0 <= k_star - j < n, else 0
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        # Projected subgradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        a_new = symmetrize(a_new)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            eta = min(eta * 1.05, 0.1)
        else:
            eta = max(eta * 0.95, 0.001)
        
        a = a_new
        
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG iter={iters}, best={best_score:.8f}, eta={eta:.5f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")
    
    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T_init=500, T_final=50, maxiter=5000):
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    T = T_init
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return evaluate_smoothed(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return gradient_smoothed(z, T, n)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        a_out = symmetrize(a_out)
        ss = a_out.sum()
        if ss > 1e-15:
            a_out = a_out / ss
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    f_interp = symmetrize(f_interp)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def raised_cosine(n, power=2):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** power
    a = np.maximum(a, 0)
    a = symmetrize(a)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP fictitious play (with and without symmetrization), subgradient descent,
L-BFGS polishing, and multi-length exploration. Key improvements over previous:
- Palindromic symmetrization to reduce effective search space
- Adaptive step sizing with tighter stall detection
- Multiple sequence lengths explored
- Subgradient method as escape mechanism from local minima
- Better time allocation across phases
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.linalg import toeplitz
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def symmetrize(a):
    return 0.5 * (a + a[::-1].copy())


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def evaluate_smoothed(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def gradient_smoothed(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lp_fictitious_play(a0, time_limit, step=0.4, use_sym=True, verbose=True, label="LP"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    current_step = step

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        if use_sym:
            g_norm = symmetrize(g_norm)
            gs = g_norm.sum()
            if gs > 1e-15:
                g_norm = g_norm / gs

        a_new = (1 - current_step) * a + current_step * g_norm
        a_new = np.maximum(a_new, 0)
        if use_sym:
            a_new = symmetrize(a_new)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 80:
            current_step = max(0.05, current_step * 0.7)
            rng = np.random.default_rng(iters * 6151 + 7)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            if use_sym:
                a = symmetrize(a)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
            current_step = min(0.5, current_step * 2.0)

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={current_step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def subgradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = 0.02

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = np.argmax(conv)

        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        a_new = symmetrize(a_new)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            eta = min(eta * 1.02, 0.1)
        else:
            eta = max(eta * 0.97, 0.002)

        a = a_new

        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG iter={iters}, best={best_score:.8f}, eta={eta:.5f}, time={elapsed:.0f}s")

    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000, use_sym=True):
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return evaluate_smoothed(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return gradient_smoothed(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        if use_sym:
            a_out = symmetrize(a_out)
        ss = a_out.sum()
        if ss > 1e-15:
            a_out = a_out / ss
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    f_interp = symmetrize(f_interp)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def raised_cosine(n, power=2):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** power
    a = np.maximum(a, 0)
    a = symmetrize(a)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a
