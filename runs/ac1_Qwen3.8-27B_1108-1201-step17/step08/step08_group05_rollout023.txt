Multi-phase optimization for minimum peak auto-convolution problem.
Uses accelerated subgradient method with LP polishing and multi-length search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_at_peaks(a, peak_indices):
    """Compute subgradient of max_k (a*a)[k] at given peak indices.
    grad[i] = sum_k a[k-i] for valid k-i"""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For each i in [lo, hi], grad[i] += a[k-i]
        idx = np.arange(lo, hi + 1)
        grad[idx] += a[k - idx]
    return grad


def accelerated_subgrad(a_init, time_limit, n_top=20, momentum=0.9, label="asg"):
    """
    Nesterov-accelerated subgradient method with backtracking line search.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()  # Nesterov interpolation point
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 1.0 / np.sqrt(n)
    no_improve = 0
    it = 0
    beta = 1.0  # Nesterov parameter
    best_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov interpolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a + (beta / beta_new - 1.0) * (a - y + v)  # hmm this gets complicated
        # Actually let me use a simpler accelerated method
        
        # Compute convolution and find peaks
        c = fast_convolve(y)
        
        # Select top peaks
        if n_top < len(c):
            peak_idx = np.argpartition(c, -n_top)[-n_top:]
        else:
            peak_idx = np.arange(len(c))
        
        # Compute subgradient
        grad = compute_gradient_at_peaks(y, peak_idx)
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad /= g_norm
        
        # Backtracking line search
        step = best_lr
        obj_y = c.max()
        found = False
        for _ in range(10):
            a_new = project_simplex(a - step * grad)
            obj_new = fast_convolve(a_new).max()
            if obj_new <= obj_y - 0.1 * step * g_norm * (obj_y - best_obj + 1e-10):
                found = True
                break
            step *= 0.5
        
        if not found:
            # Just take a small step
            a_new = project_simplex(a - 0.001 * grad)
            obj_new = fast_convolve(a_new).max()
        
        # Update momentum
        v = momentum * v + (a_new - a)
        a = a_new + (0.0)  # without Nesterov it's just gradient descent with momentum
        
        # Actually let me use a cleaner momentum approach
        # a_new = project_simplex(a - lr * grad)
        # v = momentum * v + (a_new - a)  
        # a = a_new + v
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            best_lr = step
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
                no_improve = 0
        
        # Decay step size
        lr = lr / (1.0 + 0.001 * it)
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"lr={lr:.6f} t={elapsed:.0f}s")
        
        # Periodic restart to escape local minima
        if it % 500 == 0 and no_improve > 100:
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj


def lp_polish(a_init, time_limit, max_constraints=200, label="lp"):
    """LP-based polishing with constraint generation."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Build initial constraint set
    c = fast_convolve(a)
    M = c.max()
    top_k = min(50, len(c))
    active = np.argpartition(c, -top_k)[-top_k:].tolist()
    active_set = set(active)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()
        
        # Build constraint matrix
        K = len(active)
        if K > max_constraints:
            # Reduce to most violated
            c_active = c[active]
            keep = np.argsort(c_active)[-max_constraints:]
            active = [active[i] for i in keep]
            active_set = set(active)
            K = len(active)
        
        A_mat = np.zeros((K, n))
        for row_i in range(K):
            k = active[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[row_i, lo:hi+1] = a[lo:hi+1]
        
        # Solve LP: maximize sum(g) s.t. g >= 0, A_mat @ g <= M
        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A_mat @ g_var <= M]
        )
        
        solved = False
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iters=500000)
                solved = True
                break
            except Exception:
                continue
        
        if not solved or g_var.value is None:
            if no_improve > 10:
                break
            no_improve += 1
            continue
        
        g = np.maximum(g_var.value, 0.0)
        g_sum = g.sum()
        if g_sum < 1.0 + 1e-5:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_norm = g / g_sum
        
        # Line search
        best_t = 0.1
        best_trial = best_obj
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            trial = (1.0 - t) * a + t * g_norm
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial /= s
            obj_trial = fast_convolve(trial).max()
            if obj_trial < best_trial:
                best_trial = obj_trial
                best_t = t
        
        a_new = (1.0 - best_t) * a + best_t * g_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        a = a_new
        
        obj_new = fast_convolve(a).max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Constraint generation: find new violations
        if it % 3 == 0:
            fg = fast_cross_convolve(a, g_norm)
            mask = np.zeros(len(c), dtype=bool)
            mask[active] = True
            violations = np.where((fg > M + 1e-8) & (~mask))[0]
            if len(violations) > 0:
                worst = np.argsort(fg[violations])[-min(5, len(violations)):]
                for v_idx in worst:
                    vi = int(v_idx)
                    if vi not in active_set:
                        active_set.add(vi)
                        active.append(vi)
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"K={K} t={elapsed:.0f}s")
    
    return best_a, best_obj


def coordinate_descent(a_init, time_limit, label="coord"):
    """Simple coordinate descent for fine polishing."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try small perturbations at random positions
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            # Try moving mass from i to a random other position
            j = rng.integers(0, n)
            if i == j:
                continue
            eps = a[i] * 0.01
            if eps < 1e-15:
                continue
            a[i] -= eps
            a[j] += eps
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                improved = True
            else:
                a[i] += eps
                a[j] -= eps
        
        if not improved:
            break
    
    return best_a, best_obj


def mirrored_descent(a_init, time_limit, label="mirror"):
    """Mirror descent with negative entropy (KL) Bregman divergence."""
    t0 = time.monotonic()
    n = len(a_init)

    # Or work directly with log-parameterization

    best_a = a_init.copy()
    best_obj = fast_convolve(a_init).max()
    lr = 0.1
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        a = np.exp(theta)
        a /= a.sum()
        
        c = fast_convolve(a)
        n_top = 10
        if n_top < len(c):
            peak_idx = np.argpartition(c, -n_top)[-n_top:]
        else:
            peak_idx = np.arange(len(c))
        
        grad = compute_gradient_at_peaks(a, peak_idx)
        
        # Mirror descent update in log-space
        theta -= lr * grad / (a + 1e-15)  # This is the mirror descent update
        # Actually, mirror descent with entropy: θ_new = θ - lr * grad
        # Then a_new = exp(θ_new) / sum(exp(θ_new))
        
        a_new = np.exp(theta)
        a_sum = a_new.sum()
        if a_sum < 1e-30 or not np.isfinite(a_sum):
            theta = theta - lr * grad * 0.1
            a_new = np.exp(theta)
            a_sum = a_new.sum()
            if a_sum < 1e-30:
                break

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0

            break
    
    return best_a, best_obj

"""
Multi-phase optimization for minimum peak auto-convolution.
Uses LP-based constraint generation with aggressive constraint management,
careful line search, multi-length exploration, and subgradient polishing.
Target: score < 1.5027 (lower is better).
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_at_peaks(a, peak_indices):
    """Compute subgradient of max_k (a*a)[k] at given peak indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += a[k - idx]
    return grad


def precise_line_search(f, direction, t_prev=0.1):
    """Very precise line search for the blend parameter t.
    Minimize max_k ((1-t)*f + t*direction) * ((1-t)*f + t*direction) [k]
    """
    n = len(f)
    
    def obj_at(t):
        trial = (1.0 - t) * f + t * direction
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-10:
            return 1e10
        trial = trial / s
        return fast_convolve(trial).max()
    
    # Initial wide scan
    t_candidates = [0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    # Add t_prev neighborhood
    for dt in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
        t_try = t_prev + dt
        if 0.005 < t_try < 0.8:
            t_candidates.append(t_try)
    t_candidates = sorted(set(t_candidates))
    
    best_t = t_candidates[0]
    best_obj = obj_at(best_t)
    for t in t_candidates[1:]:
        o = obj_at(t)
        if o < best_obj:
            best_obj = o
            best_t = t
    
    # Bisection refinement
    idx = t_candidates.index(best_t)
    lo = t_candidates[max(0, idx - 1)]
    hi = t_candidates[min(len(t_candidates) - 1, idx + 1)]
    if hi - lo < 1e-4:
        return best_t
    
    obj_lo = obj_at(lo)
    obj_hi = obj_at(hi)
    obj_mid = best_obj
    
    # Find bracket around best_t
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        obj_mid = obj_at(mid)
        if obj_lo <= obj_mid:
            lo = mid
            obj_lo = obj_mid
        else:
            hi = mid
            obj_hi = obj_mid
        if hi - lo < 1e-7:
            break
    
    return 0.5 * (lo + hi)


def lp_optimize(f_init, time_limit, max_cons=300, top_k_start=50, label="lp"):
    """
    LP-based optimization with aggressive constraint generation.
    The key improvement: maintain a large pool of constraints and
    use very precise line search.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    # Initial constraint set: top peaks
    c = fast_convolve(f)
    M = c.max()
    top_k = min(top_k_start, len(c))
    active = sorted(np.argpartition(c, -top_k)[-top_k:].tolist())
    active_set = set(active)
    
    it = 0
    no_improve = 0
    t_prev = 0.1
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 10 == 0:
            c = fast_convolve(f)
            M = c.max()
        
        # Manage constraint pool size
        if len(active) > max_cons:
            c_active = c[active]
            # Keep top constraints by violation
            keep = np.argsort(c_active)[-max_cons:]
            active = [active[i] for i in np.sort(keep)]
            active_set = set(active)
        
        K = len(active)
        
        # Build constraint matrix
        A_mat = np.zeros((K, n))
        for row_i in range(K):
            k = active[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[row_i, lo:hi + 1] = f[lo:hi + 1]
        
        # Solve LP
        g_var = cp.Variable(n)
        constraints = [g_var >= 0, A_mat @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        
        solved = False
        g_val = None
        for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                if solver == cp.PDLP:
                    prob.solve(solver=solver, max_iters=500000)
                elif solver == cp.SCS:
                    prob.solve(solver=solver, max_iters=500000, eps=1e-7)
                else:
                    prob.solve(solver=solver, max_iters=200000)
                if g_var.value is not None:
                    g_val = np.maximum(g_var.value, 0.0)
                    if np.all(np.isfinite(g_val)):
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved or g_val is None:
            no_improve += 1
            if no_improve > 15:
                break
            # Increase constraint set
            top_k = min(top_k + 20, len(c))
            new_active = sorted(np.argpartition(c, -top_k)[-top_k:].tolist())
            for a_i in new_active:
                if a_i not in active_set:
                    active_set.add(a_i)
                    active.append(a_i)
            continue
        
        g_sum = g_val.sum()
        if g_sum < 1.0 + 1e-6:
            no_improve += 1
            if no_improve > 25:
                break
            continue
        
        g_norm = g_val / g_sum
        no_improve = 0
        
        # Precise line search
        t_opt = precise_line_search(f, g_norm, t_prev=t_prev)
        t_prev = t_opt
        
        # Update
        f = (1.0 - t_opt) * f + t_opt * g_norm
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        obj_new = fast_convolve(f).max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
        
        # Constraint generation (every 2 iterations)
        if it % 2 == 0:
            fg = fast_cross_convolve(f, g_norm)
            mask = np.zeros(len(c), dtype=bool)
            mask[active] = True
            violations = np.where((fg > M * 1.001 + 1e-10) & (~mask))[0]
            if len(violations) > 0:
                # Add worst violations
                worst = np.argsort(fg[violations])[-min(10, len(violations)):]
                for v_idx in worst:
                    vi = int(v_idx)
                    if vi not in active_set:
                        active_set.add(vi)
                        active.append(vi)
            
            # Also periodically refresh with top peaks
            if it % 10 == 0:
                top_k_cur = min(len(active) + 20, len(c))
                new_top = np.argpartition(c, -top_k_cur)[-top_k_cur:].tolist()
                for a_i in new_top:
                    if a_i not in active_set:
                        active_set.add(a_i)
                        active.append(a_i)
        
        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.12f} score={score:.8f} "
                  f"K={len(active)} t_opt={t_opt:.5f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def subgradient_polish(f_init, time_limit, n_top=15, label="sg"):
    """Subgradient method with adaptive step size and momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)
    lr = 0.005
    momentum = 0.8
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        
        if n_top < len(c):
            peak_idx = np.argpartition(c, -n_top)[-n_top:]
        else:
            peak_idx = np.arange(len(c))
        
        grad = compute_gradient_at_peaks(f, peak_idx)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        
        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
            if no_improve > 200:
                # Restart
                rng = np.random.default_rng(it * 12345)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
                lr = 0.003
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.12f} score={score:.8f} "
                  f"lr={lr:.6f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def try_multiple_starts(f_base, n_starts, time_per_start, label="multi"):
    """Try multiple random perturbations of the base and optimize each."""
    n = len(f_base)
    best_f = f_base.copy()
    best_obj = fast_convolve(f_base).max()
    rng = np.random.default_rng(999)
    
    for s in range(n_starts):
        t_remaining = time_per_start * (n_starts - s) - (time.monotonic() - (time.monotonic() - time_per_start * n_starts))
        # Just use equal time for each
        t_s = time_per_start / n_starts
        if t_s < 5:
            break
        
        # Perturb
        noise = rng.standard_normal(n) * (0.01 + 0.01 * s / max(1, n_starts))
        f_try = project_simplex(f_base + noise)
        
        # Quick LP
        f_opt, obj_opt = lp_optimize(f_try, time_limit=t_s, max_cons=200,
                                      top_k_start=40, label=f"{label}_{s}")
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_f = f_opt.copy()
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization pipeline:
    1. Load parent, try quick improvements
    2. LP optimization with aggressive constraint generation
    3. Subgradient polishing
    4. Multi-length exploration
    5. Final LP burst
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.8f}")
    
    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    
    # === Phase 1: Main LP optimization (45% time) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 750.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP main (n={n_parent}, budget={t1:.0f}s) ===")
        f1, obj1 = lp_optimize(best_seq, time_limit=t1, max_cons=400,
                                top_k_start=60, label="lp1")
        score1 = 2.0 * len(f1) * obj1
        eval1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(eval1) and eval1 < best_score:
            best_score = eval1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp1]: {eval1:.8f}")
        else:
            print(f"  lp1: obj={obj1:.12f} score={score1:.8f} (best remains {best_score:.8f})")
    
    # === Phase 2: Subgradient polish (10% time) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 150.0)
    if t2 > 15:
        print(f"\n=== Phase 2: Subgradient polish (budget={t2:.0f}s) ===")
        f2, obj2 = subgradient_polish(best_seq, time_limit=t2, n_top=15, label="sg1")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [sg1]: {score2:.8f}")
        else:
            print(f"  sg1: {score2:.8f} (no improvement)")
    
    # === Phase 3: LP deep with more constraints (20% time) ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP deep (budget={t3:.0f}s) ===")
        f3, obj3 = lp_optimize(best_seq, time_limit=t3, max_cons=500,
                                top_k_start=100, label="lp3")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp3]: {score3:.8f}")
        else:
            print(f"  lp3: {score3:.8f} (no improvement)")
    
    # === Phase 4: Multi-length exploration (15% time) ===
    rem = deadline - time.monotonic()
    t4_total = min(rem * 0.15, 200.0)
    if t4_total > 30:
        print(f"\n=== Phase 4: Multi-length (budget={t4_total:.0f}s) ===")
        n_base = len(best_seq)
        # Try nearby lengths
        candidates = sorted(set([
            n_base - 48, n_base - 24, n_base + 24, n_base + 48,
            n_base - 96, n_base + 96, n_base + 144, n_base - 144
        ]))
        candidates = [c for c in candidates if 600 <= c <= 2500]
        
        # Sort by proximity to base
        candidates.sort(key=lambda x: abs(x - n_base))
        
        for target_n in candidates[:4]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t4 = min(rem * 0.35, 60.0)
            if t4 < 10:
                break
            
            # Interpolate best_seq to target_n
            old_x = np.linspace(0.0, 1.0, n_base)
            new_x = np.linspace(0.0, 1.0, target_n)
            f_interp = np.interp(new_x, old_x, best_seq)
            f_interp = np.maximum(f_interp, 0.0)
            s = f_interp.sum()
            if s > 1e-10:
                f_interp /= s
            
            f4, obj4 = lp_optimize(f_interp, time_limit=t4, max_cons=200,
                                    top_k_start=50, label=f"n{target_n}")
            score4 = float(evaluate_sequence(f4.tolist()))
            if np.isfinite(score4) and score4 < best_score:
                best_score = score4
                best_seq = f4.copy()
                n_base = target_n

            else:
                print(f"  n{target_n}: {score4:.8f} (no improvement over {best_score:.8f})")
    
    # === Phase 5: Final concentrated LP (remaining time) ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t5 = min(rem * 0.6, 120.0)
        if t5 > 10:

            f5, obj5 = lp_optimize(best_seq, time_limit=t5, max_cons=500,

            score5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(score5) and score5 < best_score:
                best_score = score5
                best_seq = f5.copy()
                print(f"  NEW BEST [lp_final]: {score5:.8f}")
    
    # === Phase 6: Final subgradient burst ===

    if rem > 10:
        t6 = min(rem * 0.4, 60.0)
        if t6 > 5:
            f6, obj6 = subgradient_polish(best_seq, time_limit=t6, n_top=10,
                                           label="sg_final")
            score6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(score6) and score6 < best_score:
                best_score = score6
                best_seq = f6.copy()
                print(f"  NEW BEST [sg_final]: {score6:.8f}")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Gap: {best_score - 1.5027:.8f}")
    print(f"{'=' * 60}")
    
    return [float(x) for x in best_seq]
