def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Score for normalized a (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C(f):
    """Constraint matrix: C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. C*f_conj(g) <= M, g >= 0."""
    import cvxpy as cp
    n = len(f)
    C = build_C(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in ['CLARABEL', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return None


def solve_lp_scipy(f, M):
    """Fallback LP via scipy HiGHS."""
    from scipy.optimize import linprog
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    res = linprog(c, A_ub=C, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def geo_mean(a, g, t):
    """Geometric mean interpolation in log-space."""
    a_safe = np.maximum(a, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1.0 - t) * np.log(a_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    result = np.exp(log_new)
    s = result.sum()
    if s < 1e-30:
        return a.copy()
    return result / s


def linear_interp(a, g, t):
    """Linear interpolation, projected to simplex."""
    result = (1.0 - t) * a + t * g
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-30:
        return a.copy()
    return result / s


def subgradient_step(a, eta, top_k=3):
    """Fast O(n) subgradient step: reduce elements contributing to top peaks."""
    n = len(a)
    conv = fast_convolve(a)
    top_indices = np.argsort(conv)[::-1][:top_k]
    log_a = np.log(np.maximum(a, 1e-30))
    for k in top_indices:
        ki = int(k)
        for j in range(n):
            partner = ki - j
            if 0 <= partner < n:
                log_a[j] -= eta * a[partner]
    log_a -= log_a.max()
    result = np.exp(log_a)
    s = result.sum()
    if s < 1e-30:
        return a.copy()
    return result / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def cosine_init(n, alpha=1.0):
    """Cosine-windowed initialization."""
    t = np.linspace(0, np.pi, n)
    a = np.cos(t)**(2 * alpha) + 0.01
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def structured_random_init(n, seed=0):
    """Structured random initialization with multiple frequency components."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    for freq in [1, 2, 3, 5]:
        a += 0.08 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a

                                           use_adaptive_M=True, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

        # Quick CD on this length
        rem2 = deadline - time.monotonic() - 60
        if rem2 > 10 and a_alt is not None:
            a_alt_cd, sc_alt_cd = coord_descent_v2(a_alt, min(rem2 * 0.3, 30.0), verbose=False)
            best_seq, best_score = check_and_update(a_alt_cd, f"n{alt_n}_cd",
                                                     best_seq, best_score)

    # === Phase 5: Final LP polish ===
    print(f"\n=== Phase 5: Final LP Polish ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_alloc = min(rem * 0.5, 150.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin, sc_fin = lp_fictitious_play(best_seq, t_alloc,
                                           use_adaptive_M=True, verbose=True)
        best_seq, best_score = check_and_update(a_fin, "final_lp", best_seq, best_score)

    # === Phase 6: Final CD polish ===
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(rem * 0.8, 60.0)
        print(f"\n=== Phase 6: Final CD === (budget={t_alloc:.0f}s)")
        a_fin_cd, sc_fin_cd = coord_descent_v2(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_fin_cd, "final_cd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = compute_score(best_seq)

    # Final verification using the actual evaluate_sequence
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
