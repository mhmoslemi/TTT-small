import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_soft_max_gradient(a, p):
    """
    Compute gradient of soft_max(conv) w.r.t. a.
    p: soft-max probabilities over conv values (length 2n-1)
    Returns: grad_a of length n
    """
    n = len(a)
    # g[i] = sum_k p[k] * a[k-i] for valid k,i
    # Use convolution: g = conv(p, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    # Use FFT for speed
    total_len = len(p) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    P = rfft(p, fft_size)
    B = rfft(a_rev, fft_size)
    conv_pb = irfft(P * B, fft_size)[:total_len]
    g = conv_pb[n - 1:2 * n - 1]
    return 2.0 * g


def soft_max_tau_objective_and_grad(theta, tau, n, conv_cache=None):
    """
    Compute soft-max objective and gradient w.r.t. theta.
    a = softmax(theta)
    Returns (objective_value, gradient_vector)
    """
    # Compute a from theta with numerical stability
    theta_shifted = theta - np.max(theta)
    exp_theta = np.exp(theta_shifted)
    a = exp_theta / np.sum(exp_theta)

    # Compute convolution
    conv = fast_convolve(a)

    # Soft-max
    max_conv = np.max(conv)
    scaled = (conv - max_conv) / tau
    exp_scaled = np.exp(scaled)
    sum_exp = np.sum(exp_scaled)
    p = exp_scaled / sum_exp
    soft_max_val = max_conv + tau * np.log(sum_exp)

    # Objective: 2n * soft_max_val
    obj = 2.0 * n * soft_max_val

    # Gradient w.r.t. a: grad_a[i] = 2n * 2 * sum_k p[k] * a[k-i] = 4n * g[i]
    g = compute_soft_max_gradient(a, p)
    grad_a = 4.0 * n * g

    # Gradient w.r.t. theta: grad_theta[j] = a[j] * (grad_a[j] - a @ grad_a)
    mean_grad = np.dot(a, grad_a)
    grad_theta = a * (grad_a - mean_grad)

    return obj, grad_theta

    Run L-BFGS with soft-max objective and tau annealing.
    a_init: initial sequence (will be normalized to sum=1)
    Returns: (best_a, best_score, n_iterations)
    """

    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initial conv and tau
    conv0 = fast_convolve(a)
    M0 = conv0.max()
    if tau_init is None:
        tau_init = M0 / 5.0
    tau = tau_init

    # Start from theta = log(a) (with epsilon for zeros)
    eps = 1e-15
    theta = np.log(a + eps)
    # Remove the mean to fix the gauge (sum theta = 0)
    theta = theta - np.mean(theta)

    best_a = a.copy()
    best_score = 2.0 * n * M0
    t0 = time.monotonic()
    total_iters = 0
    tau_decrease = 0.95
    iters_per_tau = 30

    while time.monotonic() - t0 < time_budget:
        # Run L-BFGS for iters_per_tau iterations at current tau
        results = []
        for _ in range(iters_per_tau):
            if time.monotonic() - t0 > time_budget:
                break

            def fun_and_grad(th):
                obj, grad = soft_max_tau_objective_and_grad(th, tau, n)
                return obj, grad

                               options={'maxiter': 1, 'gtol': 1e-6, 'maxls': 20})
                theta = res.x

            except Exception:

            a_cur = a_cur / a_cur.sum()
            conv_cur = fast_convolve(a_cur)
            score_cur = 2.0 * n * conv_cur.max()

            if score_cur < best_score:
                best_score = score_cur
                best_a = a_cur.copy()

        # Decrease tau
        tau *= tau_decrease
        if tau < M0 * 1e-6:

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SM-LBFGS {label}: tau={tau:.6f}, best={best_score:.6f}, iters={total_iters}, t={elapsed:.1f}s")

    return best_a, best_score, total_iters


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent with exact step optimization via bisection.
    For each coordinate, finds the delta that minimizes the score.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    def score_for_delta(i, delta):
        """Compute the score if we add delta to a[i]."""
        k_start = max(i, 0)
        k_end = min(i + n - 1, 2 * n - 2)
        if k_start > 2 * n - 2 or k_end < 0:
            return best_score, M

        # Unaffected max
        unaff_max = -1e30
        if k_start > 0:
            unaff_max = max(unaff_max, conv[:k_start].max())
        if k_end < 2 * n - 2:
            unaff_max = max(unaff_max, conv[k_end + 1:].max())

        # Affected values with delta applied
        # new_conv[k] = conv[k] + 2*delta*a[k-i] for k in [k_start, k_end]
        # except at k=2i: new_conv[2i] += delta^2
        ks = np.arange(k_start, k_end + 1)
        a_shifted = a[ks - i]  # a[k-i] for k in affected range
        new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted

        # Handle k = 2i
        if 2 * i >= k_start and 2 * i <= k_end:
            idx = 2 * i - k_start
            new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

        new_max = max(unaff_max * 1.0, float(np.max(new_affected)))
        new_sum = sum_a + delta
        if new_sum < 0.01:
            return 1e30, new_max
        return 2.0 * n * new_max / (new_sum * new_sum), new_max

    def apply_delta(i, delta):
        """Apply delta to a[i] and update conv, M, sum_a."""
        nonlocal conv, M, sum_a
        k_start = max(i, 0)
        k_end = min(i + n - 1, 2 * n - 2)
        ks = np.arange(k_start, k_end + 1)
        a_shifted = a[ks - i]
        changes = 2.0 * delta * a_shifted
        if 2 * i >= k_start and 2 * i <= k_end:
            idx = 2 * i - k_start
            changes[idx] = 2.0 * delta * (a[i] - delta) + delta * delta
        new_vals = conv[k_start:k_end + 1] + changes

        unaff_max = -1e30
        if k_start > 0:
            unaff_max = max(unaff_max, conv[:k_start].max())
        if k_end < 2 * n - 2:
            unaff_max = max(unaff_max, conv[k_end + 1:].max())

        conv[k_start:k_end + 1] = new_vals
        M = max(unaff_max, float(np.max(new_vals)))
        a[i] = max(a[i] + delta, 0.0)
        sum_a = max(sum_a + delta, 0.0)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            # Quick check: try a few candidate deltas
            cur_score = 2.0 * n * M / (sum_a * sum_a)
            best_delta = 0.0
            best_new_score = cur_score

            # Candidate deltas: fractions of a[i]
            for frac in [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5, -0.7, 0.7]:
                delta = frac * a[i]
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue
                sc, _ = score_for_delta(i, delta)
                if sc < best_new_score:
                    best_new_score = sc
                    best_delta = delta

            # If we found improvement, refine with bisection
            if best_delta != 0.0:
                # Determine search range
                if best_delta > 0:
                    lo, hi = 0.0, min(2.0 * best_delta, a[i] * 5)
                else:
                    lo, hi = max(2.0 * best_delta, -a[i] * 0.99), 0.0

                # Bisection for the minimum
                for _ in range(12):
                    mid = (lo + hi) / 2.0
                    sc_mid, _ = score_for_delta(i, mid)
                    sc_lo, _ = score_for_delta(i, lo)
                    sc_hi, _ = score_for_delta(i, hi)
                    # Find the minimum among lo, mid, hi
                    if sc_lo <= sc_mid and sc_lo <= sc_hi:
                        hi = mid
                    elif sc_hi <= sc_lo and sc_hi <= sc_mid:
                        lo = mid
                    else:
                        lo = max(lo, mid)
                        hi = min(hi, mid)
                        if hi - lo < 1e-12 * max(1.0, abs(a[i])):
                            break

                best_delta = (lo + hi) / 2.0
                final_score, _ = score_for_delta(i, best_delta)
                if final_score < best_new_score:
                    best_new_score = final_score

            if best_delta != 0.0 and best_new_score < cur_score - 1e-14:
                apply_delta(i, best_delta)
                if best_new_score < best_score:
                    best_score = best_new_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = max(i, 0)
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())

            a_shifted = a[k_start - i:k_end - i + 1]
            affected = conv[k_start:k_end + 1]
            cur_score = 2.0 * n * M / (sum_a * sum_a)
            best_delta = 0.0
            best_new_score = cur_score

            for factor in [0.995, 0.99, 0.985, 0.9975, 1.0025, 1.005, 1.01, 1.015]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue
                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_max = max(unaff_max, float(np.max(new_affected)))
                new_sum = sum_a + delta
                if new_sum < 0.01:
                    continue
                sc = 2.0 * n * new_max / (new_sum * new_sum)
                if sc < best_new_score:
                    best_new_score = sc
                    best_delta = delta

            if best_delta != 0.0 and best_new_score < cur_score - 1e-15:
                new_affected = affected + 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * best_delta * a[i] + best_delta * best_delta
                conv[k_start:k_end + 1] = new_affected
                M = max(unaff_max, float(np.max(new_affected)))
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta
                sum_a = max(sum_a, 0.01)
                if best_new_score < best_score:
                    best_score = best_new_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_alternation(a_init, time_limit, solver=None, verbose=False, label=""):
    """Alternating LP best responses with averaging."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    running_avg = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            a = b_new
        a_new, _ = solve_lp_best_response(a, solver)
        if a_new is not None:
            a = a_new

        running_avg = (it * running_avg + a) / (it + 1)

        for cand in [a, running_avg, 0.5 * (a + running_avg)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            score = 2.0 * n * fast_convolve(cand_n).max()
            if score < best_score:
                best_score = score
                best_a = cand_n.copy()

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")

        # Stop if converged
        if it > 10 and abs(best_score - 2.0 * n * fast_convolve(a).max()) < 1e-10:
            if it > 50:
                break

    return best_a, best_score, it


def init_fourier(n, rng, n_harmonics=None):
    """Initialize using random Fourier superposition."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(5, 25))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_flat_spectrum(n, rng):
    """
    Initialize by trying to make |FFT(a)|^2 as flat as possible.
    This is motivated by the fact that flat power spectrum → flat circular autocorrelation.
    """
    # Start with random phases (Givens' construction idea)
    num_freqs = n // 2 + 1
    target_power = 1.0 / num_freqs

    # Random magnitude (near-flat) and random phases
    mag = target_power * (1.0 + 0.1 * rng.standard_normal(num_freqs))
    mag = np.maximum(mag, 0.01)
    phases = rng.uniform(0, 2 * np.pi, num_freqs)

    # Build the FFT
    A = np.zeros(n, dtype=complex)
    A[:num_freqs] = np.sqrt(mag) * np.exp(1j * phases)
    # Ensure Hermitian symmetry for real signal
    if n % 2 == 0:
        A[2:] = np.conj(A[1:-1][::-1])
        A[1] = np.abs(A[1])  # Nyquist is real
        A[0] = np.abs(A[0])  # DC is real
    else:
        A[1:] = np.conj(A[-1:0:-1])
        A[0] = np.abs(A[0])

    # IFFT to get the time-domain signal
    a = np.real(np.fft.ifft(A)) * n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_score):
    """Evaluate and return updated best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return None, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        return a_norm.copy(), score
    return None, best_score


def perturb_escape(a, rng, magnitude=0.05):
    """Perturb the sequence to escape local optima."""
    n = len(a)
    a_pert = a.copy()
    n_perturb = max(5, n // 10)
    idx = rng.choice(n, size=n_perturb, replace=False)
    a_pert[idx] *= (1.0 + rng.standard_normal(n_perturb) * magnitude)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def propose_candidate():
    """
    Multi-phase optimization:
    1. Soft-max L-BFGS with tau annealing from diverse initializations
    2. Exact coordinate descent for local optimization
    3. LP alternation for global moves
    4. Fine polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Soft-max L-BFGS from diverse initializations ===
    print("\n=== Phase 1: Soft-max L-BFGS ===")
    t1_start = time.monotonic()

    # Generate initializations
    inits = []
    if parent is not None:
        inits.append(('parent', parent))
        for n_try in [len(parent), 1536, 2048]:
            if n_try != len(parent) and 512 <= n_try <= 2048:
                inits.append((f'parent_interp_{n_try}', interpolate_seq(parent, n_try)))

    for n_try in [768, 1024, 1536]:
        if not parent or n_try != len(parent):
            a = init_fourier(n_try, rng)
            inits.append((f'fourier_{n_try}', a))
            a2 = init_flat_spectrum(n_try, rng)
            inits.append((f'flat_spec_{n_try}', a2))

    # Run soft-max L-BFGS on each initialization
    for label, init_a in inits:
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        t_alloc = min(120.0, rem * 0.15)
        print(f"  SM-LBFGS: {label}, n={len(init_a)}, budget={t_alloc:.0f}s")
        a_opt, s_opt, _ = soft_max_lbfgs(init_a, t_alloc, verbose=True, label=label)
        new_best, new_score = check_and_update(a_opt, best_score)
        if new_best is not None:
            best_seq = new_best
            best_score = new_score
            print(f"    -> NEW BEST: {best_score:.6f}")

    print(f"  Phase 1 done ({time.monotonic()-t1_start:.0f}s), best={best_score:.6f}")

    # === Phase 2: Exact coordinate descent ===
    print(f"\n=== Phase 2: Exact Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        t2_start = time.monotonic()
        current = best_seq.copy()

        for rnd in range(4):
            rem = deadline - time.monotonic() - 600
            if rem < 20:
                break
            t_alloc = min(150.0, rem * 0.25)
            print(f"  ECD round {rnd+1}, n={len(current)}, budget={t_alloc:.0f}s")
            a_opt, s_opt, _ = exact_coord_descent(current, t_alloc, verbose=True, label=f"ecd_r{rnd}")
            new_best, new_score = check_and_update(a_opt, best_score)
            if new_best is not None:
                best_seq = new_best
                best_score = new_score
                current = best_seq.copy()
                print(f"    -> NEW BEST: {best_score:.6f}")
            else:
                # Perturb and try again
                current = perturb_escape(best_seq, rng, magnitude=0.03)

    # === Phase 3: LP alternation ===
    print(f"\n=== Phase 3: LP Alternation (best={best_score:.6f}) ===")
    if best_seq is not None:
        for n_try in [len(best_seq), 1024, 1536]:
            rem = deadline - time.monotonic() - 300
            if rem < 60:
                break
            t_alloc = min(120.0, rem * 0.3)
            init_fp = interpolate_seq(best_seq, n_try) if len(best_seq) != n_try else best_seq.copy()
            print(f"  LP n={n_try}, budget={t_alloc:.0f}s")
            a_fp, s_fp, _ = lp_alternation(init_fp, t_alloc, solver=solver, verbose=True, label=f"lp_{n_try}")
            new_best, new_score = check_and_update(a_fp, best_score)
            if new_best is not None:
                best_seq = new_best
                best_score = new_score
                print(f"    -> NEW BEST: {best_score:.6f}")

    # === Phase 4: Soft-max L-BFGS on improved solution ===
    print(f"\n=== Phase 4: SM-LBFGS on improved (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(100.0, rem * 0.4)
            print(f"  SM-LBFGS n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_opt, s_opt, _ = soft_max_lbfgs(best_seq, t_alloc, tau_init=None,
                                              verbose=True, label="sm_phase4")
            new_best, new_score = check_and_update(a_opt, best_score)
            if new_best is not None:
                best_seq = new_best
                best_score = new_score
                print(f"    -> NEW BEST: {best_score:.6f}")

    # === Phase 5: Fine polishing ===
    print(f"\n=== Phase 5: Fine Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(100.0, rem * 0.5)
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_alloc, verbose=True, label="fine")
            new_best, new_score = check_and_update(a_fine, best_score)
            if new_best is not None:
                best_seq = new_best
                best_score = new_score
                print(f"    -> NEW BEST: {best_score:.6f}")

        # Another ECD pass
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(60.0, rem * 0.6)
            a_e2, s_e2, _ = exact_coord_descent(best_seq, t_alloc, verbose=True, label="ecd_final")
            new_best, new_score = check_and_update(a_e2, best_score)
            if new_best is not None:
                best_seq = new_best
                best_score = new_score

    # === Phase 6: Final micro-polish ===
    print(f"\n=== Phase 6: Final (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            t_alloc = min(rem * 0.8, 30.0)
            a_fin, s_fin, _ = fine_coord_descent(best_seq, t_alloc, label="last")
            new_best, new_score = check_and_update(a_fin, best_score)
            if new_best is not None:
                best_seq = new_best
                best_score = new_score

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
