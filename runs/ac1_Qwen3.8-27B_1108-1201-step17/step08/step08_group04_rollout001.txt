"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses Nesterov-accelerated subgradient, LP-based refinement, Fourier basis
parameterization for global shape search, and coordinate descent polishing.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time
import sys

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_at(f, b, weights):
    """Compute subgradient using given weights on convolution entries."""
    n = len(f)
    full_len = 2 * n - 1
    g = np.zeros(n)
    active = np.where(weights > 1e-14)[0]
    for k in active:
        w_k = weights[k]
        if w_k < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * w_k * f[lo:hi+1][::-1]
    return g

def nesterov_sg(f_init, time_limit, n_top=60, temp_init=0.001, 
                 temp_max=0.3, label="ns"):
    """Nesterov-accelerated subgradient method with adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    full_len = 2 * n - 1
    
    # Nesterov variables
    z = f_init.copy().astype(np.float64)
    z = proj_simplex(z)
    y = z.copy()
    k = 0
    
    best_f = z.copy()
    best_obj = fast_convolve(z).max()
    temp = temp_init
    
    while time.monotonic() - t0 < time_limit:
        k += 1
        t_k = 2.0 / (k + 2.0)
        
        # Evaluate at y
        b = fast_convolve(y)
        M = b.max()
        
        # Compute weights using softmax
        if full_len <= n_top:
            w = np.ones(full_len) / full_len
        else:
            scaled = (b - M) / temp
            scaled = np.clip(scaled, -100, 0)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w = np.zeros(full_len)
                w[top_idx] = 1.0 / n_top
            else:
                w = w / w_sum
        
        # Subgradient
        sg = subgradient_at(y, b, w)
        
        # Nesterov update
        z_new = proj_simplex(y - t_k * sg)
        
        # Update y for next iteration
        if k == 1:
            y_new = z_new.copy()
        else:
            mu = (t_k - 1.0) / t_k  # This is (t_{k-1} - 1) / t_k in Nesterov
            # Actually the standard Nesterov update is:
            # y_{k+1} = z_{k+1} + (t_k - 1) / t_{k+1} * (z_{k+1} - z_k)
            # But we use the "heavy ball" style:
            # y_{k+1} = z_{k+1} + (t_k / t_{k+1} - 1) * (z_{k+1} - z_k)
            # Simplified: y = z + (1 - t_k) * (z - z_prev)
            y_new = z_new + (1.0 - t_k) * (z_new - z)
            y_new = proj_simplex(y_new)
        
        z = z_new
        y = y_new
        
        # Periodic evaluation
        if k % 50 == 0:
            obj = fast_convolve(z).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = z.copy()
            else:
                # Anneal temperature and restart
                temp = min(temp * 1.1, temp_max)
                if k % 500 == 0:
                    # Restart from best
                    z = best_f + np.random.default_rng(k).standard_normal(n) * 0.002
                    z = proj_simplex(z)
                    y = z.copy()
                    
            if k % 200 == 0:
                print(f"  {label} k={k} obj={best_obj:.12f} score={2*n*best_obj:.6f} "
                      f"temp={temp:.4f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, 2.0 * n * best_obj

def lp_direction(f, b, M, top_k, solver='ECOS'):
    """Solve LP to find descent direction."""
    n = len(f)
    full_len = 2 * n - 1
    K = min(top_k, full_len)
    
    if K < full_len:
        idx = np.argpartition(b, -K)[-K:]
        # Add boundary and center constraints
        extra = list(range(0, min(5, full_len)))
        extra += list(range(max(0, full_len-5), full_len))
        idx = np.unique(np.concatenate([idx, extra]))
        idx = idx[:K]
    else:
        idx = np.arange(full_len)
    
    idx = np.sort(idx)
    
    # Build constraint matrix
    A = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    
    g_var = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g_var))
    constraints = [g_var >= 0, A @ g_var <= M]
    
    prob = cp.Problem(obj, constraints)
    
    try:
        if n <= 600:
            prob.solve(solver=cp.ECOS, max_iters=30000)
        else:
            prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None
    
    if g_var.value is None:
        return None
    
    g_val = np.maximum(g_var.value, 0.0)
    S = g_val.sum()
    if S < 1e-10:
        return None
    return g_val / S

def lp_optimize(f_init, time_limit, top_k_init=50, top_k_max=500, label="lp"):
    """LP-based optimization with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    f = proj_simplex(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    momentum = np.zeros(n)
    beta = 0.3
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)
        
        b = fast_convolve(f)
        M = b.max()
        
        g = lp_direction(f, b, M, top_k)
        
        if g is None or g.sum() <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 40:
                break
            top_k = min(top_k * 2, min(top_k_max, 2*n-1))
            continue
        
        h = g
        momentum = beta * momentum + (1.0 - beta) * h
        mom_sum = momentum.sum()
        if mom_sum > 1e-10:
            h_mom = momentum / mom_sum
        else:
            h_mom = h
        
        best_t = 0.0
        best_obj_trial = M
        best_f_trial = f.copy()
        
        for direction in [h, h_mom]:
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
                f_trial = (1.0 - t) * f + t * direction
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-15:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj_trial:
                    best_obj_trial = obj
                    best_t = t
                    best_f_trial = f_trial.copy()
        
        if best_t == 0.0:
            stuck += 1
            if stuck > 50:
                break
            continue
        
        f = best_f_trial
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break
        
        if stuck > 25 and it % 10 < 3:
            rng = np.random.default_rng(it * 7)
            f = proj_simplex(best_f + rng.standard_normal(n) * 0.003)
            momentum = np.zeros(n)
            stuck = 0
        
        if it % 50 == 0:
            print(f"  {label} it={it} obj={best_obj:.12f} score={2*n*best_obj:.6f} "

    return best_f, 2.0 * n * best_obj

def fourier_shape_search(f_init, time_limit, K=10):
    """Search for good global shape using Fourier parameterization."""
    t0 = time.monotonic()
    n = len(f_init)
    
    # Define the parametrization: f_j = max(0, c)_j / sum(max(0, c)_j)
    # where c_j = 1 + sum_{m=1}^{K} (a_m * cos(2*pi*m*j/N) + b_m * sin(2*pi*m*j/N))
    # This ensures f is a smooth, non-negative (after clipping) function
    
    N = n
    angles = 2.0 * np.pi * np.arange(N) / N
    cos_basis = np.column_stack([np.cos(m * angles) for m in range(1, K+1)])
    sin_basis = np.column_stack([np.sin(m * angles) for m in range(1, K+1)])
    
    def objective(params):
        a0 = params[0]
        a_m = params[1:1+K]
        b_m = params[1+K:1+2*K]
        c = 1.0 + cos_basis @ a_m + sin_basis @ b_m
        f = np.maximum(c, 0.0)
        s = f.sum()
        if s < 1e-10:
            return 10.0
        f = f / s
        return fast_convolve(f).max()
    
    # Start from current f
    # Try to fit a Fourier approximation
    from scipy.fft import rfft as rfft2
    F = rfft2(f_init)
    # Extract coefficients
    a0_init = 0.0
    a_m_init = np.zeros(K)
    b_m_init = np.zeros(K)
    for m in range(1, K+1):
        a_m_init[m-1] = 2.0 * np.real(F[m]) / N
        b_m_init[m-1] = -2.0 * np.imag(F[m]) / N
    
    x0 = np.concatenate([[a0_init], a_m_init, b_m_init])
    
    best_x = x0.copy()
    best_obj = objective(x0)
    
    # Use Nelder-Mead (works for medium dimension)
    # K=10 gives 21 parameters, which is manageable
    if 2*K + 1 <= 60:
        result = minimize(objective, x0, method='Nelder-Mead',
                         options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12,
                                  'adaptive': True})
        if result.fun < best_obj:
            best_obj = result.fun
            best_x = result.x
    
    # Construct final f
    a0 = best_x[0]
    a_m = best_x[1:1+K]
    b_m = best_x[1+K:1+2*K]
    c = 1.0 + cos_basis @ a_m + sin_basis @ b_m
    f = np.maximum(c, 0.0)
    s = f.sum()
    if s > 1e-10:
        f = f / s
    else:
        f = np.ones(N) / N
    
    elapsed = time.monotonic() - t0
    print(f"  fourier K={K} obj={best_obj:.12f} score={2*N*best_obj:.6f} t={elapsed:.0f}s")
    
    return f, 2.0 * N * best_obj

def coordinate_descent(f_init, time_limit):
    """Coordinate descent with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    
    no_pass = 0
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        # Random order
        order = np.random.default_rng(int(time.monotonic()*1000)%100000).permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f[j] = new_val
                s = f.sum()
                if s < 1e-12:
                    f[j] = best_j_val
                    continue
                f = f / s
                obj = fast_convolve(f).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
                    f = f / s
                else:
                    f[j] = best_j_val
                    f = f / s
            
            f[j] = best_j_val
            s = f.sum()
            f = f / s
            
            if best_j_obj < best_obj - 1e-13:
                best_obj = best_j_obj
                best_f = f.copy()
                any_improved = True
        
        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
        
        if (it := int(time.monotonic() - t0)) % 20 == 19 and it < 60:
            print(f"  CD obj={best_obj:.12f} score={2*n*best_obj:.6f} t={it}s")
    
    return best_f, 2.0 * n * best_obj

def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result
