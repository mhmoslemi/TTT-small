def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

# Get LP solver
def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

    n = len(a)

def lbgfs_refine(a0, time_limit, T_start=200.0, T_end=0.3, n_stages=12, maxiter_per_stage=3000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def func_at_T(z, T_val):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp
    
    def jac_at_T(z, T_val):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    def make_func(T_val):
        def f(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return func_at_T(z, T_val)
        return f
    
    def make_jac(T_val):
        def j(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return jac_at_T(z, T_val)
        return j
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        # Scale iterations based on temperature
        if stage_T > 50:
            iters = maxiter_per_stage // 3
        elif stage_T > 5:
            iters = maxiter_per_stage
        else:
            iters = maxiter_per_stage * 2
        
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y
            else:
                y_new = y_new / sy
            
            # Nesterov average
            weight_

old = (it + 1)
            weight_new = alpha * (it + 2)
            a_new = (weight_old * a + weight_new * b_new) / (weight_old + weight_new)
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa < 1e-15:

            a = a_new
        else:
            a = b_new

        if score < best_score - 1e-15:

def mirror_descent_bregman(a0, time_limit, lr_init=0.05, n_steps=50000):
    """Mirror descent with KL divergence (Bregman) in log-domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        # Use top fraction as active set
        threshold = max_b - 0.02 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient: average of gradients of active constraints
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Bregman (KL) step: z_new = z - lr * g
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = compute_score_normalized(a_new)
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            lr *= 0.9
            no_improve = 0
        lr = max(lr_init / (1.0 + step * 0.001), 1e-8)
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
