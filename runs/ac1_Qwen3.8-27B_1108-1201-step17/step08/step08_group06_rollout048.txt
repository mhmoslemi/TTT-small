def project_to_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

    Mirror descent (multiplicative weight) subgradient method for minimizing
    2n * max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    Uses more-averaging trick.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # More-averaging: weighted average
    more_avg = a.copy()
    more_avg_sum_alpha = 0.0
    
    # Step size schedule: eta_t = C / sqrt(t)
    C = 1.0  # Will tune this
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        max_conv = conv.max()
        
        # Subgradient: uniform over argmax (or top-k for robustness)
        top_k = min(10, len(conv))
        sorted_idx = np.argsort(conv)[::-1]
        top_indices = sorted_idx[:top_k]
        top_vals = conv[top_indices]
        # Soft weights: proportional to exp((val - max)/T_sub) with small T_sub
        T_sub = 0.01 * max_conv
        w_sub = np.exp((top_vals - max_conv) / T_sub)
        w_sub = w_sub / w_sub.sum()
        
        # Create full weight vector
        w_full = np.zeros(2 * n - 1)
        w_full[top_indices] = w_sub
        
        # Compute subgradient
        grad = conv_gradient(a, w_full)
        
        # Mirror descent update (multiplicative weight):
        # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        eta = C / np.sqrt(t)
        # Project to simplex: we want to minimize <grad, a> + (1/eta) * sum a[i]*log(a[i]/a_old[i])
        # Solution: a_new[i] = a_old[i] * exp(-eta * grad[i]) / Z
        # But we also need a_new on the simplex (sum = 1, non-negative)
        # The mirror descent update already gives sum = 1 if we normalize.
        
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # More-averaging
        alpha_t = eta ** 2
        more_avg_sum_alpha += alpha_t
        more_avg = (more_avg_sum_alpha - alpha_t) * more_avg / more_avg_sum_alpha + alpha_t * a_new / more_avg_sum_alpha
        # Simplify: more_avg = more_avg * (1 - alpha_t/more_avg_sum_alpha) + a_new * (alpha_t/more_avg_sum_alpha)
        # Actually let me redo this:
        # more_avg_new = (sum_{s<=t-1} alpha_s * a_s + alpha_t * a_t) / (sum_{s<=t} alpha_s)
        # = more_avg_old * (sum_{s<=t-1} alpha_s) / (sum_{s<=t} alpha_s) + a_t * alpha_t / (sum_{s<=t} alpha_s)
        
        # Recompute properly:
        # Let me just track it as: more_avg = (old_weight * more_avg + alpha_t * a_t) / (old_weight + alpha_t)
        
        a = a_new
        
        sc = compute_score(a)
        sc_avg = compute_score(more_avg)
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = more_avg.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t}, score={sc:.8f}, avg={sc_avg:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def lbfgs_smooth(a0, time_limit, T_schedule, verbose=True, seed=42):
    """
    L-BFGS on the smoothed objective with temperature annealing.
    Works in the simplex parameterization (with projection).
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        T_remaining = time_limit - (time.monotonic() - t0)
        
        # L-BFGS in log-space with the smoothed objective
        # z = log(a), a = exp(z) / sum(exp(z))
        # But this constraints sum(exp(z)) = 1 which is awkward.
        # Instead, let's use z = log(a) and work with a = exp(z - max(z)) / sum(exp(z - max(z)))
        # This is the softmax parametrization.
        
        def func_and_grad(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15, np.zeros_like(z)
            zm = z - z.max()
            exp_z = np.exp(zm)
            a_tmp = exp_z / exp_z.sum()
            obj, grad_a = smooth_obj_grad_a(a_tmp, T)
            # Chain rule: d obj / d z[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
            S = np.dot(a_tmp, grad_a)
            grad_z = a_tmp * (grad_a - S)
            return obj, grad_z
        
        z0 = np.log(np.maximum(a, 1e-300))
        
        try:
            res = minimize(
                fun=lambda z: func_and_grad(z)[0],
                x0=z0,
                jac=lambda z: func_and_grad(z)[1],
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': 50000, 'disp': False}
            )
            # Extract result
            zm = res.x - res.x.max()
            exp_z = np.exp(zm)
            a_new = exp_z / exp_z.sum()
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        except Exception as e:
            print(f"      L-BFGS error at T={T}: {e}")
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS T={T}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def coord_descent_refined(a0, time_limit, verbose=True, seed=42):
    """
    Refined pairwise coordinate descent.
    At each step, pick a pair (i,j) and find the optimal transfer.
    Focus on pairs that reduce the max convolution.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_score = compute_score(a)
    gr = (np.sqrt(5) - 1) / 2
    
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False
        
        conv = fast_convolve(a)
        
        # Strategy: pick pairs based on what will reduce peaks
        # For each peak, the contributing elements are a[j] where j and peak-j are both in [0,n-1]
        # We want to reduce mass at peak-contributing positions and increase at non-contributing ones
        
        top_peak = int(np.argmax(conv))
        
        # Generate candidate pairs
        n_candidates = min(200, n)
        pairs = set()
        
        # Peak-reducing pairs: move mass from peak-contributing to non-contributing
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        peak_set = set(range(js, je + 1))
        non_peak_set = list(set(range(n)) - peak_set)
        
        if non_peak_set:
            for _ in range(n_candidates // 2):
                i = rng.integers(js, je + 1)
                j = non_peak_set[rng.integers(0, len(non_peak_set))]
                if i != j:
                    pairs.add((min(i,j), max(i,j)))
        
        # Random pairs
        for _ in range(n_candidates // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue
            
            # Golden section search for optimal transfer t
            # t > 0: move from i to j
            # t < 0: move from j to i
            t_lo = -a[j] + 1e-16
            t_hi = a[i] - 1e-16
            
            if t_hi <= t_lo:
                continue
            
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            f1 = _eval_transfer(a, i, j, x1)
            f2 = _eval_transfer(a, i, j, x2)
            
            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = _eval_transfer(a, i, j, x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = _eval_transfer(a, i, j, x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                # Perturb
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iteration}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def _eval_transfer(a, i, j, t):
    """Evaluate score after transferring t from i to j."""
    a_trial = a.copy()
    a_trial[i] -= t
    a_trial[j] += t
    if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
        return 1e10
    a_trial = np.maximum(a_trial, 0)
    s = a_trial.sum()
    if s < 1e-15:
        return 1e10
    a_trial = a_trial / s
    return compute_score(a_trial)


def lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    This gives a direction that 'fits' under the current ceiling.
    """
    n = len(f)
    # Build the constraint matrix: conv(f, g)[k] = sum_j f[j] * g[k-j] (with bounds)
    # This is a Toeplitz-like matrix
    # C[k, j] = f[abs(k-j)] ... no, it's f[k-j] if 0 <= k-j < n
    # Actually conv(f, g)[k] = sum_j f[j] * g[k-j] where 0 <= j < n and 0 <= k-j < n
    # So C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, let's build it as:
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        # C[k, j] = f[k-j] for k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        C[j:j+n, j] = f
    
    # LP: max -sum(g) s.t. C*g <= M*1, 0 <= g
    c = -np.ones(n)
    b_ub = np.full(size, M)
    
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def lp_polish(a0, time_limit, verbose=True, seed=42):
    """LP-based fictitious play with geometric mean updates."""
    n = len(a0)
    f = np.maximum(a0, 1e-15).astype(np.float64)
    f = f / f.sum()
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_score = compute_score(f)
    
    f_avg = f.copy()
    avg_weight = 1.0
    
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        g = lp_direction(f, M)
        
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Adaptive step size
        t_step = 2.0 / (k + 2)
        t_step = max(0.02, min(0.5, t_step))
        
        # Geometric mean update
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()
        
        # Update average
        w_new = 1.0 / (k + 1)
        f_avg = (avg_weight * f_avg + w_new * a_new) / (avg_weight + w_new)
        avg_weight += w_new
        
        sc_avg = compute_score(f_avg)
        sc_cur = compute_score(a_new)
        
        if sc_avg < best_score:
            best_score = sc_avg
            best_f = f_avg.copy()
        if sc_cur < best_score:
            best_score = sc_cur
            best_f = a_new.copy()
        
        f = a_new
        
        if verbose and k % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={k}, M={M:.6e}, cur={sc_cur:.8f}, avg={sc_avg:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:

    a_norm = a / s

        best_seq = a_norm.copy()
        if verbose:

    return best_seq, best_score
