So $B_k = c_k - 2a_i a_{k-i} - a_i^2 \mathbf{1}[k=2i]$.

This means for each coordinate $i$, I need to evaluate $f(x) = \max_k [c_k - 2a_i a_{k-i} - a_i^2 \mathbf{1}[k=2i] + 2x a_{k-i} + x^2 \mathbf{1}[k=2i]] / (S+x)^2$

For $k = 2i$: $c_{2i} + (x^2 - a_i^2)$

So the "affected" part is simple. The key computation is:
- $\max_{k \neq 2i} [c_k + 2(x-a_i) a_{k-i}]$
- $\max(c_{2i} + x^2 - a_i^2, \text{above})$

And we need to find $x$ minimizing this divided by $(S+x)^2$.

For the 1D search, at each evaluation of $f(x)$:
- Compute the max of $c_k + 2(x-a_i) a_{k-i}$ over $k \neq 2i$ (with the constraint $0 \leq k \leq 2n-2$ and $a_{k-i}$ is defined)
- Compare with $c_{2i} + x^2 - a_i^2$
- Divide by $(S+x)^2$

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Optimal Coordinate Descent with analytical 1D subproblem solving
2. LP-based best response for global improvements  
3. Accelerated mirror descent (Nesterov) for smooth refinement
4. Multi-length and multi-init exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    if '_lp_solver_global' in globals():
        return _lp_solver_global
    _lp_solver_global = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCIP, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_global = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_global = cp.ECOS
    return _lp_solver_global

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    M = fast_convolve(a).max()
    return 2.0 * n * M / (S * S)

def ocd_optimize(a0, time_limit, verbose=False, label=""):
    """Optimal Coordinate Descent with analytical 1D subproblems."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        S = a.sum()
        sweep_improved = False
        
        for i in range(n):
            ai = a[i]
            S_rest = S - ai
            
            # Affected convolution indices: [i, i+n-1]
            # a_{k-i} for k in [i, i+n-1] is a[0], a[1], ..., a[n-1]
            affected = conv[i:i+n].copy()
            
            # Unaffected max
            umax = -np.inf
            if i > 0:
                umax = max(umax, conv[0:i].max())
            if i + n < 2 * n - 1:
                umax = max(umax, conv[i+n:2*n-1].max())
            
            # k=2i is at position i in the affected array
            idx_2i = i
            conv_2i = conv[2 * i]
            
            # Generate candidate x values
            # 1. Current value
            x_cands = [ai]
            # 2. Zero
            x_cands.append(0.0)
            # 3. Analytical candidates from top active constraints
            #    For each k in [i, i+n-1] where the constraint is "tight":
            #    candidate x_k = S_rest - (conv[k] - 2*ai*a[k-i]) / (2*a[k-i])  [for k != 2i]
            #    candidate x_{2i} = (conv[2i] - ai^2) / S_rest  [for k = 2i]
            
            # Find top affected peaks
            aff_max_vals = affected.copy()
            aff_max_vals[idx_2i] = conv_2i  # Make sure 2i is correct
            top_k = min(5, n)
            top_indices = np.argpartition(aff_max_vals, -top_k)[-top_k:]
            
            for ti in top_indices:
                k = i + ti  # actual convolution index
                if k == 2 * i:
                    # Quadratic constraint: B_{2i} + x^2, where B_{2i} = conv[2i] - ai^2
                    B_2i = conv_2i - ai * ai
                    if S_rest > 1e-15:
                        x_cand = B_2i / S_rest
                        if x_cand > 0:
                            x_cands.append(x_cand)
                else:
                    # Linear constraint: B_k + 2*x*a[k-i]
                    # B_k = conv[k] - 2*ai*a[k-i]
                    vk = a[ti]  # a[k-i] = a[ti]
                    if vk > 1e-15:
                        Bk = conv[k] - 2.0 * ai * vk
                        x_cand = S_rest - Bk / (2.0 * vk)
                        if x_cand > 0:
                            x_cands.append(x_cand)
            
            # 4. Some heuristic values
            x_cands.append(ai * 0.5)
            x_cands.append(ai * 1.5)
            x_cands.append(0.5 / (n * 2))
            
            x_cands = [max(0.0, min(x, 10.0/(n))) for x in x_cands]
            x_cands = list(set(x_cands))
            
            # Evaluate at candidates
            best_x = ai
            best_f = 2.0 * n * conv.max() / (S * S) if S > 0 else np.inf
            
            for x in x_cands:
                new_sum = S_rest + x
                if new_sum < 1e-15:
                    continue
                new_aff = affected + 2.0 * (x - ai) * a
                new_aff[idx_2i] = conv_2i + (x * x - ai * ai)
                aff_max = new_aff.max()
                total_max = max(aff_max, umax) if umax > -np.inf else aff_max
                if total_max < 0:
                    total_max = 0.0
                f = 2.0 * n * total_max / (new_sum * new_sum)
                if f < best_f:
                    best_f = f
                    best_x = x
            
            # Golden section refinement around best_x
            if best_x != ai and best_f < best_score - 1e-14:
                # Bracket
                lo = max(0.0, best_x * 0.5) if best_x > 0 else 0.0
                hi = best_x * 2.0 if best_x > 0 else 1.0 / n
                if lo == hi:
                    lo = 0.0
                    hi = 1.0 / n
                
                def eval_f(x):
                    new_sum = S_rest + x
                    if new_sum < 1e-15:
                        return np.inf
                    new_aff = affected + 2.0 * (x - ai) * a
                    new_aff[idx_2i] = conv_2i + (x * x - ai * ai)
                    aff_max = new_aff.max()
                    total_max = max(aff_max, umax) if umax > -np.inf else aff_max
                    if total_max < 0:
                        total_max = 0.0
                    return 2.0 * n * total_max / (new_sum * new_sum)
                
                # Check bracket
                f_lo = eval_f(lo)
                f_hi = eval_f(hi)
                f_mid = eval_f(best_x)
                if f_mid <= f_lo and f_mid <= f_hi:
                    # best_x is already the minimum in bracket
                    pass
                else:
                    # Golden section
                    if f_lo < f_hi:
                        hi = hi  # keep
                    else:
                        lo = lo  # keep
                    # Standard golden section
                    gr = (np.sqrt(5) - 1) / 2
                    c = hi - gr * (hi - lo)
                    d = lo + gr * (hi - lo)
                    fc = eval_f(c)
                    fd = eval_f(d)
                    for _ in range(20):
                        if fc < fd:
                            hi = d
                            d = c
                            fd = fc
                            c = hi - gr * (hi - lo)
                            fc = eval_f(c)
                        else:
                            lo = c
                            c = d
                            fc = fd
                            d = lo + gr * (hi - lo)
                            fd = eval_f(d)
                    best_x = (lo + hi) / 2.0
                    best_f = eval_f(best_x)
                
                if best_f < best_score - 1e-14:
                    # Apply update
                    delta = best_x - ai
                    conv[i:i+n] += 2.0 * delta * a
                    conv[2 * i] += delta * delta  # Fix the 2i term
                    a[i] = best_x
                    S = S_rest + best_x
                    sweep_improved = True
            
            if best_f < best_score - 1e-14:
                score_now = 2.0 * n * conv.max() / (S * S)
                if score_now < best_score:
                    best_score = score_now
                    best_a = a.copy()
        
        if not sweep_improved:
            no_improve += 1
            if no_improve > 10 and sweeps > 5:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            cur_s = a.sum()
            cur_sc = 2.0 * n * conv.max() / (cur_s * cur_s)
            print(f"    OCD {label} sweep={sweeps} ({elapsed:.1f}s): best={best_score:.8f} cur={cur_sc:.8f}")
    
    return best_a, best_score, sweeps

def accelerated_mirror_descent(a0, time_limit, verbose=False, label=""):
    """Nesterov accelerated mirror descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov momentum point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    t = 1
    lr0 = 0.05
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t += 1
        lr = lr0 / t
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M_y = conv_y.max()
        
        # Use all peaks with softmax weighting
        temp = 0.005 * max(M_y, 1e-10)
        w = np.exp((conv_y - M_y) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros_like(conv_y)
            w[conv_y == M_y] = 1.0
            w /= w.sum()
        else:
            w /= ws
        
        # Subgradient: grad_i = sum_k w_k * 2 * y_{k-i}
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi+1] += 2.0 * w[k] * y[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        
        # Mirror descent step: y_new = proj_simplex(y * exp(-lr * grad))
        y = simplex_project(y * np.exp(-lr * grad))
        
        # Update a (Nesterov)
        alpha = (1 + np.sqrt(1 + 4 * t * t)) / 2
        a_new = y + (t / (t + 1)) * (y - a)  # Hmm, this isn't quite right
        # Actually, standard Nesterov:
        # a_{t+1} = (t/(t+1)) * a_t + (1/(t+1)) * y_{t+1}... 
        # Let me use a simpler accelerated scheme:
        # y_{t+1} = proj_simplex(y_t - lr_t * grad(y_t))
        # a_{t+1} = y_{t+1} + (beta_t / (1+beta_{t+1})) * (y_{t+1} - a_t)
        # where beta_t = (1+sqrt(1+4*t^2))/2
        
        # Actually let me just use plain mirror descent with decreasing lr
        a = y
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AMD {label} it={it} ({elapsed:.1f}s): best={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score
