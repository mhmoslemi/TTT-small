def simplex_project(v):

    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_lse_gradient(w, a, fft_size):
    """
    Compute grad[i] = 2 * sum_k w[k] * a[k-i] using FFT.
    w has length 2n-1, a has length n.
    """
    # correlate(w, a)[i] = sum_n w[n+i] * a[n]
    # We want g[i] = sum_k w[k] * a[k-i] for i = 0..n-1
    # Substituting m = k-i: g[i] = sum_m w[m+i] * a[m] = correlate(w, a)[i]
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)
    n = len(a)
    return 2.0 * corr[:n]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with FFT-based gradients and adaptive tau
2. Fast coordinate descent with O(n) sensitivity per coordinate
3. LP-based best response for global improvement
4. Multi-length, multi-initialization search
5. Nesterov momentum for acceleration

import numpy as np
from scipy.fft import rfft, irfft
import time

# Precomputed FFT sizes for different n values
def get_fft_size(n, mode='conv'):
    if mode == 'conv':
        size = 2 * n - 1
    elif mode == 'grad':
        size = 3 * n - 2
    else:
        size = n
    return 1 << (size - 1).bit_length()

def fast_convolve(a, fft_size=None):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if fft_size is None:
        fft_size = get_fft_size(n, 'conv')
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def lse_gradient(a, conv, tau, fft_size_grad):
    """Compute the log-sum-exp gradient efficiently."""
    n = len(a)
    # Compute weights
    conv_shifted = tau * (conv - conv.max())
    w = np.exp(conv_shifted)
    w /= w.sum()
    
    # Compute grad[i] = 2 * sum_k w[k] * a[k-i]
    W = rfft(w, fft_size_grad)
    A = rfft(a, fft_size_grad)
    corr = irfft(np.conj(W) * A, fft_size_grad)
    return 2.0 * corr[:n], w

def lse_gd(a0, time_limit, tau_init=50.0, tau_rate=1.0005, 
           lr_init=0.3, lr_decay=0.99995, momentum=0.9,
           verbose=False, label=""):
    """
    Log-sum-exp gradient descent with Nesterov momentum.
    Minimizes max_k (a*a)[k] s.t. a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    
    fft_size_conv = get_fft_size(n, 'conv')
    fft_size_grad = get_fft_size(n, 'grad')
    
    best_a = a.copy()
    conv = fast_convolve(a, fft_size_conv)
    best_M = float(conv.max())
    
    t0 = time.monotonic()
    it = 0
    tau = tau_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    no_improve = 0
    y = a.copy()  # Nesterov lookahead
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov step
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta = (alpha - 1.0) / alpha if it > 0 else 0.0
        y = a + beta * (a - y) if it > 0 else a.copy()
        
        # Compute convolution at y
        conv = fast_convolve(np.maximum(y, 0), fft_size_conv)
        M = float(conv.max())
        
        # Compute gradient
        grad, w = lse_gradient(np.maximum(y, 0), conv, tau, fft_size_grad)
        
        # Update
        v = momentum * v + (1 - momentum) * grad
        a_new = simplex_project(y - lr * v)
        a_new = np.maximum(a_new, 0.0)
        a = a_new / a_new.sum()
        
        # Track best (evaluate at a, not y)
        conv_a = fast_convolve(a, fft_size_conv)
        Ma = float(conv_a.max())
        if Ma < best_M - 1e-15:
            best_M = Ma
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            lr *= 0.5
            tau *= 1.5
            no_improve = 0
            if lr < 1e-10:
                break
        
        tau = min(tau * tau_rate, 10000.0)
        lr *= lr_decay
        it += 1
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSGD {label} it={it} ({elapsed:.0f}s): M={Ma:.10f} best={best_M:.10f} tau={tau:.1f} lr={lr:.6f}")
    
    return best_a, best_M

def build_conv_matrix(a):
    """Build T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] s.t. b >= 0, sum(b)=1."""

        solver = _find_lp_solver()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:

                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

            a = a_new
        
        fft_size = get_fft_size(n, 'conv')
        score = 2.0 * n * fast_convolve(a, fft_size).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def power_mean_gd(a0, time_limit, p=3.0, lr_init=0.1, verbose=False, label=""):
    """
    Power mean smoothed gradient descent.
    Minimizes (sum_k ((a*a)[k])^p)^(1/p) s.t. a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    
    fft_size_conv = get_fft_size(n, 'conv')
    fft_size_grad = get_fft_size(n, 'grad')
    
    best_a = a.copy()
    conv = fast_convolve(a, fft_size_conv)
    best_M = float(conv.max())
    
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a, fft_size_conv)
        M = float(conv.max())
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-10:
                break
        
        # Power mean gradient
        # f(a) = (sum_k conv_k^p)^(1/p)
        # grad_i = (1/p) * (sum_k conv_k^p)^(1/p-1) * p * sum_k conv_k^(p-1) * 2*a[k-i]
        #        = (sum_k conv_k^p)^(1/p-1) * 2 * sum_k conv_k^(p-1) * a[k-i]
        
        conv_p = np.power(np.maximum(conv, 0), p)
        sum_conv_p = conv_p.sum()
        if sum_conv_p < 1e-30:
            break
        
        # Normalized weights: w_k = conv_k^(p-1) / sum_j conv_j^(p-1)
        conv_pm1 = np.power(np.maximum(conv, 0), p - 1)
        w = conv_pm1 / conv_pm1.sum()
        
        # Compute gradient via FFT
        W = rfft(w, fft_size_grad)
        A = rfft(a, fft_size_grad)
        corr = irfft(np.conj(W) * A, fft_size_grad)
        grad = 2.0 * (sum_conv_p ** (1.0 / p - 1.0)) * corr[:n]
        
        # Scale: the gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            a = simplex_project(a - lr * grad / grad_norm)
        
        lr *= 0.9999
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PMGD {label} it={it} ({elapsed:.0f}s): M={M:.10f} best={best_M:.10f} lr={lr:.6f}")
    
    return best_a, best_M


def init_fourier(n, rng, n_harmonics=None):
    """Initialize using random Fourier superposition."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s if s > 1e-10 else np.ones(n) / n
    return a


def init_zadoff_chu_like(n, rng):
    """Initialize using a Zadoff-Chu-like sequence (low PAPR)."""
    # Zadoff-Chu: x(k) = exp(-j * pi * u * k^2 / N) for N odd
    # We use the magnitude (which is 1) modulated by a random envelope
    u = int(rng.integers(1, n // 2))
    if n % 2 == 0:
        n_eff = n - 1
    else:
        n_eff = n
    k = np.arange(n_eff)
    phase = np.exp(-1j * np.pi * u * k * k / n_eff)
    a = np.abs(phase)  # This is all 1s, so let's add variation
    # Instead, use a random binary phase sequence with smooth envelope
    a = np.ones(n) * 0.5
    envelope_width = n * rng.uniform(0.3, 0.8)
    center = n / 2
    envelope = np.exp(-0.5 * ((np.arange(n) - center) / envelope_width) ** 2)
    a = 0.3 + 0.7 * envelope
    # Add Fourier components
    for j in range(rng.integers(2, 8)):
        freq = int(rng.integers(1, n // 4))
        amp = rng.uniform(0.05, 0.15)
        ph = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + ph)
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s if s > 1e-10 else np.ones(n) / n
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_score, label=""):
    """Evaluate candidate and return score."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_score
