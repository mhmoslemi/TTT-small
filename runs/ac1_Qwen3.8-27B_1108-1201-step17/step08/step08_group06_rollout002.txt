    lo = -a[j]
    hi = a[i]

            if val > new_max:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def pair_eval_max(conv, d, quad_positions, quad_values, delta):
    """
    Compute max of conv + 2*delta*d + delta^2*quad (sparse).
    d is precomputed = a[k-i] - a[k-j]
    quad is sparse: at most 3 non-zero positions.
    """
    vals = conv + 2.0 * delta * d
    for pos, val in zip(quad_positions, quad_values):
        if 0 <= pos < len(vals):
            vals[pos] += delta * delta * val
    return np.max(vals)


def pair_eval_conv(conv, d, quad_positions, quad_values, delta):
    """Compute full new convolution (for updating)."""
    new_conv = conv + 2.0 * delta * d
    for pos, val in zip(quad_positions, quad_values):
        if 0 <= pos < len(new_conv):
            new_conv[pos] += delta * delta * val
    return new_conv


def golden_section_pair(a, conv, i, j, max_iter=30):
    """
    Find optimal delta for transfer from i to j using golden section search.
    Returns (delta, new_max).
    """
    n = len(a)
    size = 2 * n - 1
    k = np.arange(size)
    ki = k - i
    kj = k - j
    # Precompute d = a[k-i] - a[k-j]
    ai = np.zeros(size)
    valid_i = (ki >= 0) & (ki < n)
    ai[valid_i] = a[ki[valid_i]]
    aj = np.zeros(size)
    valid_j = (kj >= 0) & (kj < n)
    aj[valid_j] = a[kj[valid_j]]
    d = ai - aj

    # Quadratic correction (sparse)
    quad_positions = []
    quad_values = []
    if 0 <= 2 * i < size:
        quad_positions.append(2 * i)
        quad_values.append(1.0)
    if 0 <= i + j < size:
        quad_positions.append(i + j)
        quad_values.append(-2.0)
    if 0 <= 2 * j < size:
        quad_positions.append(2 * j)
        quad_values.append(1.0)

    # Bounds: delta in [-a[j], a[i]]
    lo = -a[j]
    hi = a[i]
    if hi - lo < 1e-15:
        return 0.0

    gr = (np.sqrt(5) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = pair_eval_max(conv, d, quad_positions, quad_values, x1)
    f2 = pair_eval_max(conv, d, quad_positions, quad_values, x2)

    for _ in range(max_iter):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = pair_eval_max(conv, d, quad_positions, quad_values, x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = pair_eval_max(conv, d, quad_positions, quad_values, x2)

    delta_opt = (lo + hi) / 2.0
    return delta_opt

def build_C_for_lp(f):
    """Build constraint matrix for LP: C[k,j] = f[k-j] if valid."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def lp_direction(f, M, verbose=False):
    """
    Solve LP: max sum(g) s.t. C*g <= M, g >= 0.
    Returns g normalized to sum 1.
    """
    n = len(f)
    if n > 2048:
        # Too large for dense LP; use greedy
        return greedy_direction(f, M)
    C = build_C_for_lp(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if res.status != 0:
        return greedy_direction(f, M)
    g = np.maximum(res.x, 0)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def greedy_direction(f, M):
    """Greedy LP approximation."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def lp_fictitious_play(a_init, time_limit, verbose=True, use_greedy=False):
    """LP fictitious play with geometric mean updates."""
    f = a_init.copy()
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        if use_greedy:
            g = greedy_direction(f, M)
        else:
            g = lp_direction(f, M, verbose=verbose)

        if g is None:
            g = greedy_direction(f, M)
        if g is None:
            break

        # Geometric mean update in log space
        t_step = 2.0 / (k + 2)
        t_step = max(0.01, min(1.0, t_step))
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        a_new = np.exp(log_new - log_new.max())
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = a_new.copy()

        f = a_new

        if verbose and k % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={k}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_f, best_score


def lbgfs_smooth(a0, time_limit, T, maxiter=2000):
    """L-BFGS with smooth max (log-sum-exp) in log space."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    z0 = np.log(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func_z(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_tmp = exp_z / exp_z.sum()
        conv = fast_convolve(a_tmp)
        # Smooth max
        shifted = conv / T
        sm = T * (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max()))))
        return 2.0 * n * sm

    def jac_z(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_tmp = exp_z / exp_z.sum()
        conv = fast_convolve(a_tmp)
        shifted = conv / T
        w = np.exp(shifted - shifted.max())
        w = w / w.sum()
        # grad of smooth_max w.r.t. a: 2 * (a * w) (cross correlation)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_tmp, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a_tmp)
        grad_z = a_tmp * (grad_a - S)
        return 2.0 * n * grad_z

    try:
        res = minimize(
            fun=func_z, x0=z0, jac=jac_z,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False}
        )
        z_max = res.x.max()
        a_out = np.exp(res.x - z_max)
        a_out = a_out / a_out.sum()
        sc = compute_score(a_out)
        if sc < best_score:
            best_a = a_out
            best_score = sc
    except Exception:
        pass

    return best_a, best_score


def make_initial_shape(n, shape_type, rng):
    """Create different initial shapes."""
    if shape_type == 0:  # Uniform
        return np.ones(n) / n
    elif shape_type == 1:  # Cosine modulation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif shape_type == 2:  # Multi-frequency
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif shape_type == 3:  # Edge-peaked
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (1 - 2 * x) ** 2
        a = np.maximum(a, 0)
        return a / a.sum()
    elif shape_type == 4:  # Random
        a = rng.random(n)
        return a / a.sum()
    elif shape_type == 5:  # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate against the official evaluator."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining:
    - Fast coordinate descent with exact pair formula
    - LP fictitious play
    - L-BFGS with smooth max
    - Multi-start initialization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Multi-start initialization (30s) ===
    print(f"\n=== Phase 1: Multi-Start (best={best_score:.8f}) ===")
    t1_start = time.monotonic()
    t1_budget = 30.0

    # Try different shapes at the parent length
    for shape_type in range(6):
        if time.monotonic() - t1_start > t1_budget:
            break
        a_init = make_initial_shape(n_parent, shape_type, rng)
        # Quick coordinate descent
        a_test, sc_test = fast_cd_v2(a_init, 3.0, verbose=False)
        best_seq, best_score = check_and_update(a_test, f"shape{shape_type}", best_seq, best_score)

    # Also try starts based on parent with perturbations
    for pert_scale in [0.1, 0.3]:
        if time.monotonic() - t1_start > t1_budget:
            break
        a_pert = parent + rng.standard_normal(n_parent) * pert_scale
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_test, sc_test = fast_cd_v2(a_pert, 4.0, verbose=False)
        best_seq, best_score = check_and_update(a_test, f"pert{pert_scale}", best_seq, best_score)

    print(f"  After Phase 1: best={best_score:.8f}")

    # === Phase 2: Main fast coordinate descent (800s) ===
    print(f"\n=== Phase 2: Fast CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2_budget = min(800.0, rem * 0.5)
    a_cd, sc_cd = fast_cd_v2(best_seq, t2_budget, verbose=True)
    best_seq, best_score = check_and_update(a_cd, "fast_cd", best_seq, best_score)
    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: LP Fictitious Play (300s) ===
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3_budget = min(300.0, rem * 0.25)
    if rem > t3_budget + 200:
        a_lp, sc_lp = lp_fictitious_play(best_seq, t3_budget, verbose=True, use_greedy=(n_parent > 1024))
        best_seq, best_score = check_and_update(a_lp, "lp_fp", best_seq, best_score)
    else:
        a_lp, sc_lp = lp_fictitious_play(best_seq, max(30, rem * 0.2), verbose=False, use_greedy=True)
        best_seq, best_score = check_and_update(a_lp, "lp_fp", best_seq, best_score)
    print(f"  After Phase 3: best={best_score:.8f}")

    # === Phase 4: Second round of fast CD (300s) ===
    print(f"\n=== Phase 4: Fast CD Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4_budget = min(300.0, rem * 0.4)
    a_cd2, sc_cd2 = fast_cd_v2(best_seq, t4_budget, verbose=True)
    best_seq, best_score = check_and_update(a_cd2, "fast_cd2", best_seq, best_score)
    print(f"  After Phase 4: best={best_score:.8f}")

    # === Phase 5: L-BFGS polish (150s) ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for T_val in [500, 2000]:
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(60.0, (deadline - time.monotonic()) * 0.3)
        if t_alloc < 15:
            break
        a_lb, sc_lb = lbgfs_smooth(best_seq, t_alloc, T=T_val)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)
    print(f"  After Phase 5: best={best_score:.8f}")

    # === Phase 6: Final CD polish (remaining time - 60s) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60.0
    if rem > 20:
        a_fin, sc_fin = fast_cd_v2(best_seq, rem, verbose=False)
        best_seq, best_score = check_and_update(a_fin, "final_cd", best_seq, best_score)

    # === Phase 7: Try length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30.0
    if rem > 60:
        for alt_n in [n_parent + 1, n_parent - 1, n_parent + 2, n_parent + 4]:
            if alt_n < 64 or alt_n > 3000 or time.monotonic() > deadline - 40:
                continue
            # Interpolate
            old_x = np.linspace(0, 1, len(best_seq))
            new_x = np.linspace(0, 1, alt_n)
            a_alt = np.interp(new_x, old_x, best_seq)
            a_alt = np.maximum(a_alt, 0)
            a_alt = a_alt / a_alt.sum()
            t_alloc = min(30.0, (deadline - time.monotonic()) * 0.3)
            if t_alloc < 10:
                break
            a_alt2, _ = fast_cd_v2(a_alt, t_alloc, verbose=False)
            best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = parent
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
