import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_logsumexp_obj(z, T_val):
    """Compute the LogSumExp smoothed objective and its gradient."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    objective = (2.0 * len(a) / T_val) * log_sum_exp
    return objective, a, exp_Tb / exp_Tb.sum()


def compute_lse_gradient(z, T_val):
    """Compute gradient of LogSumExp smoothed objective."""
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=8, maxiter_per_stage=3000):

    t0 = time.monotonic()

                return 1e15
            obj, _, _ = compute_logsumexp_obj(z_local, T_val)

        return func

        def jac(z_local):

                return np.zeros_like(z_local)
            return compute_lse_gradient(z_local, T_val)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter_per_stage)

    for stage_idx, stage_T in enumerate(temps):

            break

        maxfun = max(10, min(maxfun, iters_per_stage * 20))

                fun=make_func(stage_T),

            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def subgradient_averaging(a0, time_limit, n_steps=20000, lr_decay=0.0001):
    """Subgradient method with Polyak averaging for the smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient (using the active set)
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.01 * max_b
        active = np.where(b >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Mirror descent step
        lr = 1.0 / np.sqrt(step + 1) * 0.1
        z = z - lr * g
        a_new = softmax_from_z(z)
        
        # Polyak averaging
        weight = 1.0 / (step + 1)
        avg_a = (avg_a * step + a_new * 1.0) / (step + 1)
        avg_a = np.maximum(avg_a, 0)
        avg_s = avg_a.sum()
        if avg_s > 1e-15:
            avg_a = avg_a / avg_s
        
        a = a_new
        
        # Check both current and averaged
        score_cur = 2.0 * n * fast_convolve(a).max()
        score_avg = 2.0 * n * fast_convolve(avg_a).max()
        
        if score_cur < best_score:
            best_score = score_cur
            best_a = a.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradAvg {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-length exploration with
LP-based Nesterov alternating, L-BFGS LogSumExp annealing, subgradient averaging,
and peak shaving. Key improvements: broader length search, per-length state
tracking for warm-starting, and more aggressive optimization budgets.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_lse_gradient(z, T_val):
    """Compute gradient of LogSumExp smoothed objective w.r.t. z."""
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def compute_lse_objective(z, T_val):
    """Compute the LogSumExp smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * len(a) / T_val) * log_sum_exp


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=8, maxiter_per_stage=2000):

    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_local):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return compute_lse_objective(z_local, T_val)
        return func

    def make_jac(T_val):
        def jac(z_local):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_local)
            return compute_lse_gradient(z_local, T_val)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 50,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0
    rng_local = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 300 == 0:
            if best_score < prev_score - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                noise = rng_local.random(n) * 0.005
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_averaging(a0, time_limit, n_steps=30000):
    """Mirror descent with subgradient and Polyak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    lr_base = 0.1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.01 * max_b
        active = np.where(b >= threshold)[0]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        lr = lr_base / np.sqrt(step + 1)
        z = z - lr * g
        a_new = softmax_from_z(z)

        weight = 1.0 / (step + 1)
        avg_a = (avg_a * step + a_new) / (step + 1)
        avg_a = np.maximum(avg_a, 0)
        avg_s = avg_a.sum()
        if avg_s > 1e-15:
            avg_a = avg_a / avg_s

        a = a_new

        score_cur = 2.0 * n * fast_convolve(a).max()
        score_avg = 2.0 * n * fast_convolve(avg_a).max()

        if score_cur < best_score:
            best_score = score_cur
            best_a = a.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradAvg {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=15000):
    """Multi-peak shaving with adaptive coordinate adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        near_peak_threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= near_peak_threshold)[0]

        contrib_count = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:5]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:10]
        improved = False

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88, 0.83, 0.78, 0.72]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:4]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            a_trial = a.copy()
            for i in top_coords[:3]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.97
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, mode=None):
    """Create structured initial sequence with various modes."""
    if mode is None:
        mode = int(rng.integers(0, 8))
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10.0 * x * (1 - x))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (2 * x - 1) ** 2
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 * np.sin(np.pi * x) ** 2 + 0.5 * np.exp(-4 * x * (1 - x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))  # noqa: F821
    except Exception:
        return float('inf')
