Great, so the gradient of the max is sparse! For the active lag k*, only positions i where k*-i is in [0, n) have nonzero gradient.

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None, top_k=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(f)
    if top_k is not None and top_k < 2*n - 1:
        # Tiered constraints: top_k lags get M, rest get M * 2
        # We need to know which lags are top - pass them in
        # For simplicity, just use all constraints but with two levels
        constraints = [C @ g <= M]
    else:
        constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step0=0.5, decay=0.01, verbose=True, label="LP-FP"):

    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        step = step0 / (1.0 + decay * iters)
        step = min(step, 1.0)
        
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-13:

            best_a = a_new.copy()
            stall_count = 0
        else:

def peak_shaving(a0, time_limit, verbose=True):
    """Targeted peak-shaving: reduce the top peaks in the convolution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        # Find the top few lags
        top_lags = np.argsort(conv)[-5:][::-1]  # top 5 lags
        max_val = conv[top_lags[0]]
        
        # For the top lag, find contributing positions
        k = top_lags[0]
        # conv[k] = sum_j a[j]*a[k-j]
        # Gradient wrt a[i] is 2*a[k-i] if 0<=k-i<n
        contrib = np.zeros(n)
        valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
        contrib[valid] = a[np.arange(n)[valid]]  # a[k-i] for valid i
        # Actual contribution of position i to conv[k] is a[i]*a[k-i]
        
        # Find positions to decrease (high contribution to top peaks)
        # and positions to increase (low contribution)
        
        # Try transferring from a high-contribution position to a low-contribution one
        # Score contribution of position i to all top lags
        total_contrib = np.zeros(n)
        for lag in top_lags:
            valid_i = np.arange(n)
            partner = lag - valid_i
            valid_mask = (partner >= 0) & (partner < n)
            # contribution of a[i] to conv[lag] is a[partner[i]] for valid i
            # but we want the "sensitivity": d(conv[lag])/da[i] = 2*a[lag-i]
            total_contrib[valid_mask] += a[partner[valid_mask]]
        
        # We want to decrease a[i] where total_contrib[i] is high
        # and increase a[i] where total_contrib[i] is low
        # But also need to not increase conv at other lags too much
        
        # Simple heuristic: pick a random high-contrib source and low-contrib target
        high_positions = np.where(total_contrib > np.median(total_contrib) + 0.5*np.std(total_contrib))[0]
        low_positions = np.where(total_contrib < np.median(total_contrib) - 0.5*np.std(total_contrib))[0]
        
        if len(high_positions) == 0 or len(low_positions) == 0:
            # Add noise and continue
            a = best_a + rng.standard_normal(n) * 0.0005
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue
        
        i_src = high_positions[rng.integers(0, len(high_positions))]
        i_dst = low_positions[rng.integers(0, len(low_positions))]
        
        if i_src == i_dst or a[i_src] < 1e-10:
            continue
        
        # Try different transfer amounts
        best_local = compute_score(a)
        best_t = 0.0
        max_transfer = a[i_src] * 0.1
        
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
            t = frac * max_transfer
            a_trial = a.copy()
            a_trial[i_src] -= t
            a_trial[i_dst] += t
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = compute_score(a_trial)
            if sc < best_local - 1e-14:
                best_local = sc
                best_t = t
        
        if best_t > 0:
            a[i_src] -= best_t
            a[i_dst] += best_t
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_local < best_score - 1e-13:
                best_score = best_local
                best_a = a.copy()
        else:
            # No improvement, add small noise
            if iters % 50 == 0:
                a = best_a + rng.standard_normal(n) * 0.0003
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PeakShave iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z
