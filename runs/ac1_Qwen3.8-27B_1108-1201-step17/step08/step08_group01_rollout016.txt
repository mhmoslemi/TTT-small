import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_conv(a, conv):
    """Compute score from normalized sequence and its convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]) + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    solvers = []
    for name in ['GUROBI', 'MOSEK', 'GLOP', 'GLPK', 'ECOS', 'PDLP', 'SCS', 'CBC', 'SCIP', 'XPRESS']:
        try:
            solver = getattr(cp, name, None)
            if solver is None:
                continue
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solvers.append(solver)
        except Exception:
            continue
    if solvers:
        return solvers[0]
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution constraint matrix efficiently
    # conv(a_fixed, b)_k = sum_j a_fixed[j] * b[k-j]
    # This is a Toeplitz-like matrix
    # For efficiency, use the fact that it's a convolution
    
    # Actually for moderate n, we can build the matrix directly
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try ECOS as fallback
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return None, None
    except Exception:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

def subgradient_method(a0, time_limit, verbose=True):

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):

    n = len(a)

    A = rfft(a, fft_size)

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1] + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

    t0 = time.monotonic()
    sweeps = 0
    stall = 0

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected_base = conv[k_start:k_end + 1]
            a_shift = a[0:n]
            has_2i = (2 * i >= k_start) and (2 * i <= k_end)
            idx_2i = 2 * i - k_start if has_2i else -1

            # Define objective as function of x = new value of a[i]
            def obj_at_x(x):
                delta = x - a[i]
                aff = affected_base + 2.0 * delta * a_shift
                if has_2i:
                    aff = aff.copy()
                    aff[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                val = max(unaff_max, float(aff.max()))
                return val

            # Quick probe to find a good search interval
            cur_m = obj_at_x(a[i])
            best_probe_x = a[i]
            best_probe_m = cur_m

            for frac in [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.1, 1.3, 0.15]:
                x = a[i] * frac
                if x < 0:
                    x = 0.0
                m = obj_at_x(x)
                if m < best_probe_m:
                    best_probe_m = m
                    best_probe_x = x

            if best_probe_x == a[i]:
                continue

            # Golden section in [max(0, 0.5*best_probe), 2*best_probe]
            lo = max(0.0, best_probe_x * 0.5)
            hi = max(best_probe_x * 2.0, a[i] * 1.5)
            if hi - lo < 1e-15:
                hi = lo + 1e-10

            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = obj_at_x(c1)
            f2 = obj_at_x(c2)

            for _ in range(n_golden):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj_at_x(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj_at_x(c2)

            opt_x = (lo + hi) / 2.0
            opt_m = obj_at_x(opt_x)

            if opt_m < M - 1e-15:
                delta = opt_x - a[i]
                a[i] = opt_x
                s += delta

                changes = 2.0 * delta * a_shift
                if has_2i:
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * delta * (a[i] - delta) + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD-Golden {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, verbose=True):
    """Target the maximum convolution entry directly."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        M = float(conv[k_max])

        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        j = np.arange(i_start, i_end + 1)
        contributions = a[j] * a[k_max - j]

        n_top = min(8, len(j))
        top_idx = np.argsort(contributions)[::-1][:n_top]

        found_improvement = False
        for ti in top_idx:
            i = int(j[ti])
            for frac in [0.7, 0.5, 0.3, 0.8, 0.6, 0.4, 0.2]:
                orig = a[i]
                a[i] = orig * frac
                conv_check = fast_convolve(a)
                M_check = float(conv_check.max())
                if M_check < M - 1e-12:
                    new_score = 2.0 * n * M_check
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                        found_improvement = True
                        break
                else:
                    a[i] = orig
                    break
            if a[i] != orig:
                continue

        if not found_improvement:
            # Try increasing non-contributing coordinates
            non_contrib = np.setdiff1d(np.arange(n), j)
            if len(non_contrib) > 0:
                random_idx = non_contrib[np.random.randint(len(non_contrib))]
                orig = a[random_idx]
                for frac in [1.3, 1.5, 2.0, 1.2]:
                    a[random_idx] = orig * frac
                    s = a.sum()
                    a = a / s
                    conv_check = fast_convolve(a)
                    M_check = float(conv_check.max())
                    if M_check < M - 1e-12:
                        new_score = 2.0 * n * M_check
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                            found_improvement = True
                            break
                    else:
                        a = a / a.sum()  # revert approximately
                        break

        if not found_improvement and it > 200:
            break

        a = best_a.copy() if np.random.rand() < 0.3 else a.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave it={it} ({elapsed:.0f}s): best={best_score:.6f} k_max={k_max}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
