    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""

def lp_alternate_step(f, M, active_indices, time_limit=10.0):

import numpy as np

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_fast_len(size)
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def fast_cross_convolve(f, g):
    """Full linear cross-convolution via FFT."""
    size = len(f) + len(g) - 1
    N = next_fast_len(size)
    F = rfft(f, N)
    G = rfft(g, N)
    return irfft(F * G, N)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = (u * rho_vals) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    result = np.maximum(v - theta, 0.0)
    return result


def compute_smooth_weights(b, beta):
    """Compute softmax weights for the smooth approximation."""
    max_b = b.max()
    scaled = beta * (b - max_b)
    scaled = np.clip(scaled, -500, 0)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:

        w[int(b.argmax())] = 1.0
    else:
        w /= w_sum
    return w

    Wf = rfft(w[::-1], N)

    return grad

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(1152) / 1152.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Uniform n=1152, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # === Phase 1: Smooth PG continuation on parent ===
    print(f"\n=== Phase 1: Smooth PG continuation (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.18, 250.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f_curr = best_seq.copy()
        for beta in [10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            f_curr, obj = smooth_projected_gradient(f_curr, beta, n_iter=400, deadline=deadline)
            score = 2.0 * n_parent * fast_convolve(f_curr).max()
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = f_curr.copy()
                print(f"  beta={beta:.0f}: NEW BEST {score:.8f}")
            else:
                print(f"  beta={beta:.0f}: {score:.8f}")

    # === Phase 2: LP alternating on best ===
    print(f"\n=== Phase 2: LP alternating (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 350.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, s2 = lp_run_with_generation(best_seq, time_limit=t2, top_k_init=50,
                                         max_constraints=400, label="lp1")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp1]: {score2:.8f}")
        else:
            print(f"  lp1: {score2:.8f}")

    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.08, 100.0)
    if t3 > 15:
        f3, s3 = subgradient_polish(best_seq, time_limit=t3, lr_init=0.003, n_top=30)
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [sg1]: {score3:.8f}")
        else:
            print(f"  sg1: {score3:.8f}")

    # === Phase 4: LP deep with more constraints ===
    print(f"\n=== Phase 4: LP deep (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 300.0)
    if t4 > 30:
        f4, s4 = lp_run_with_generation(best_seq, time_limit=t4, top_k_init=80,
                                         max_constraints=600, label="lp_deep")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [lp_deep]: {score4:.8f}")
        else:
            print(f"  lp_deep: {score4:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration (best={best_score:.8f}) ===")
    candidates = [1024, 1088, 1216, 1280, 1408, 1536, 1920]
    candidates = [c for c in candidates if c != len(best_seq) and 512 <= c <= 2500]
    candidates.sort(key=lambda x: abs(x - len(best_seq)))

    for target_n in candidates[:4]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t5 = min(rem * 0.15, 140.0)
        if t5 < 25:
            break

        # Initialize by interpolation
        old_x = np.linspace(0.0, 1.0, len(best_seq))
        new_x = np.linspace(0.0, 1.0, target_n)
        f_init = np.interp(new_x, old_x, best_seq)
        f_init = np.maximum(f_init, 0.0)
        f_init = f_init / f_init.sum()
        
        inits = [f_init, flat_top_window(target_n), cosine_power_window(target_n, 1.3)]
        best_for_len = None
        best_score_len = float('inf')

        for init_i, f_i in enumerate(inits):
            rem_i = deadline - time.monotonic()
            if rem_i < 15:
                break
            t_init = min(rem_i * 0.4, 50.0)
            if t_init < 10:
                break
            # Brief smooth PG then LP
            f_ip, _ = smooth_projected_gradient(f_i, 200.0, n_iter=100, deadline=deadline)
            rem_ip = deadline - time.monotonic()
            if rem_ip < 10:
                continue
            t_lp = min(rem_ip * 0.5, 60.0)
            if t_lp < 10:
                f_final = f_ip
            else:
                f_final, _ = lp_run_with_generation(f_ip, time_limit=t_lp, top_k_init=50,
                                                     max_constraints=300,
                                                     label=f"n{target_n}_i{init_i}")
            sc = float(evaluate_sequence(f_final.tolist()))
            if np.isfinite(sc) and sc < best_score_len:
                best_score_len = sc
                best_for_len = f_final.copy()

        if best_for_len is not None and best_score_len < best_score:
            best_score = best_score_len
            best_seq = best_for_len.copy()
            print(f"  n={target_n}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={target_n}: best={best_score_len:.8f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t6 = min(rem * 0.4, 120.0)
        f6, _ = lp_run_with_generation(best_seq, time_limit=t6, top_k_init=100,
                                        max_constraints=500, label="lp_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [final_lp]: {score6:.8f}")

        rem = deadline - time.monotonic()
        if rem > 10:
            t7 = min(rem * 0.6, 60.0)
            f7, _ = subgradient_polish(best_seq, time_limit=t7, lr_init=0.002, n_top=20)
            score7 = float(evaluate_sequence(f7.tolist()))
            if np.isfinite(score7) and score7 < best_score:
                best_score = score7
                best_seq = f7.copy()
                print(f"  NEW BEST [final_sg]: {score7:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"{'=' * 60}")
    return [float(x) for x in best_seq]
