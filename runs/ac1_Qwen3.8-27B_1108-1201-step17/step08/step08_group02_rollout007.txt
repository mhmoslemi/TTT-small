import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_conv_direction(f, n, peak_indices, M, solver_list=None):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns normalized g vector or None.
    """
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    if solver_list is None:
        solver_list = [cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, max_iters=1000000, eps=1e-10)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)) and g_val.sum() > 1e-10:
                    return g_val
        except Exception:
            continue
    return None


def frank_wolfe_step(f, n, M, peak_indices):
    """
    Find the best basis direction: e_i that minimizes <subgrad, e_i - f>.
    Returns the best index i.
    """
    # Subgradient of max_k (f*f)[k] at f, for active peaks
    # d/d f[j] of (f*f)[k] = f[k-j] (for valid k-j)
    subgrad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        subgrad[idx] += f[k - idx]
    
    # FW: minimize <subgrad, e_i> = subgrad[i]
    # Best atom is argmin subgrad
    best_idx = np.argmin(subgrad)
    return best_idx, subgrad

def coordinate_refine(f_init, time_limit, label="cd"):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    # Fix small negative values from FFT numerical error
    result = np.maximum(result, 0.0)
    return result


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, n, peak_indices):
    """Build the constraint matrix A where A[g] = conv(f,g) at peak indices."""

        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

                best_t = t

    return best_t, best_obj


def benders_lp_optimize(f_init, time_limit, label="benders", max_constraints=800):
    """
    Benders-style optimization: maintain a growing set of active constraints,
    solve LP to find direction, take step, add newly violated constraints.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    rng = np.random.default_rng(42)

    # Initial constraint set: top peaks
    b = fast_convolve(f)
    M = b.max()
    n_init_constraints = min(50, len(b))
    active_set = np.argpartition(b, -n_init_constraints)[-n_init_constraints:].copy()
    
    it = 0
    stuck = 0
    use_qp = True
    mu_schedule = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2]
    mu_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        M = b.max()

        # Update active set: add new peaks that are significantly high
        threshold = 0.85 * M
        all_peaks = np.where(b >= threshold)[0]
        # Add new peaks not in active_set
        new_peaks = np.setdiff1d(all_peaks, active_set)
        if len(new_peaks) > 0 and len(active_set) < max_constraints:
            # Add the most significant new peaks
            n_add = min(len(new_peaks), max_constraints - len(active_set), 20)
            new_peaks_sorted = new_peaks[np.argsort(b[new_peaks])[::-1]][:n_add]
            active_set = np.concatenate([active_set, new_peaks_sorted])

        # If active set is too large, prune to top ones
        if len(active_set) > max_constraints:
            b_active = b[active_set]
            top_k = np.argpartition(b_active, -max_constraints)[-max_constraints:]
            active_set = active_set[top_k]

        # Solve for direction
        g = None
        if use_qp:
            mu = mu_schedule[min(mu_idx, len(mu_schedule) - 1)]
            g = solve_qp_direction(f, n, active_set, M, mu=mu)
            if g is None:
                use_qp = False

        if g is None:
            g = solve_lp_direction(f, n, active_set, M)
            if g is None:
                stuck += 1
                if stuck > 20:
                    f = f + rng.standard_normal(n) * 0.001 * f
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    stuck = 0
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                f = f + rng.standard_normal(n) * 0.002 * f
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                stuck = 0
                # Reset active set
                b = fast_convolve(f)
                n_init = min(50, len(b))
                active_set = np.argpartition(b, -n_init)[-n_init:].copy()
            continue

        # Normalize direction to sum=1
        h = g / S_g

        # Line search
        best_t, best_t_obj = line_search_convolve(f, h, M)

        if best_t <= 0:
            stuck += 1
            if stuck > 15:
                f = f + rng.standard_normal(n) * 0.001 * f
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                stuck = 0
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            break
        f = f / s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
            # Progress: slightly increase mu_idx for more aggressive QP
            if mu_idx < len(mu_schedule) - 1 and it % 30 == 0:
                mu_idx += 1
        else:
            stuck += 1
            if stuck > 20:
                # Perturbation
                f = f + rng.standard_normal(n) * 0.0015 * f
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                stuck = 0
                # Reset mu
                mu_idx = max(0, mu_idx - 1)

        if it % 25 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} score={score:.6f} M={best_b:.10f} "
                  f"constraints={len(active_set)} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def subgradient_polish(f_init, time_limit, label="sg"):
    """
    Subgradient descent with adaptive temperature and peak weighting.
    Good for final polish after LP convergence.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    rng = np.random.default_rng(888)

    lr = 0.005
    temp = 0.0005
    n_top = 150
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights on top peaks
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        b_top = b[top_idx]
        scaled = b_top / (temp * M)
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w[i] * f[k - idx]

        # Update with projection
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        f = f_new

        obj = fast_convolve(f).max()
        if obj < best_b - 1e-14:
            best_b = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.02)
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                lr *= 0.96

        if it % 300 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} score={score:.6f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

        if no_improve > 50:
            f = best_f + rng.standard_normal(n) * 0.0008
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            lr = 0.005
            temp = min(temp * 1.5, 0.01)
            no_improve = 0

    return best_f, best_b


def coordinate_refine(f_init, time_limit, label="cd"):
    """
    Coordinate descent with randomized order and multiple multiplier scales.
    Tries to adjust individual elements to reduce the peak.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    rng = np.random.default_rng(555)

    multipliers = np.array([0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5, 2.0, 0.3])
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        order = rng.permutation(n)
        any_improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue

            orig = f[j]
            best_val = orig
            best_obj = best_b

            for mult in multipliers:
                new_val = orig * mult
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:
                    f[j] = orig
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = new_val
            f[j] = orig

            if best_val != orig:
                f[j] = best_val
                s = f.sum()
                f = f / s
                any_improved = True
                if best_obj < best_b:
                    best_b = best_obj
                    best_f = f.copy()

        if not any_improved:
            if passes > 5:
                break
            # Random restart within the neighborhood
            f = best_f + rng.standard_normal(n) * 0.0003
            f = np.maximum(f, 0.0)
            f = f / f.sum()

    return best_f, best_b


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    constraint management, multi-length exploration, subgradient polish,

        p = np.clip(p, 0.0, 1000.0)

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")

        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Main Benders-LP on parent ===
    rem = deadline - time.monotonic()

    # === Phase 3: Second round of Benders-LP (deeper) ===
