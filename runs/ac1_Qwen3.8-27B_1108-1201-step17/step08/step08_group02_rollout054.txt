Uses softmax-approximated max with Adam optimizer, LP polishing, and coordinate descent.

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)

    # For linear cross-correlation (no wraparound), we need N >= len(p) + len(f) - 1
    # (p ⊛ f)[i] = sum_j p[j] * conj(f[j-i]) for j where f[j-i] is in range
    #
    # We want: grad[i] = sum_{m=0}^{n-1} p[i+m] * f[m]

    # The cross-correlation (p ⊛ f)[i] = sum_j p[j] * f[j-i] (for real f, no conj needed)

    # In the FFT domain: (p ⊛ f) = ifft(FFT(p) * conj(FFT(f)))

    grad = corr[:n]
    # Make sure it's non-negative (it should be since p >= 0 and f >= 0)

    return grad, M

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient (softmax approximation) with Adam, LP polishing,
and coordinate descent. Explores multiple lengths.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import math
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_direct(f, p, n_top=200):
    """Compute gradient of smooth max using direct computation on top peaks."""
    n = len(f)
    len_c = len(p)
    c = None  # We need c to find top peaks
    # Actually, let's pass c separately
    return None


def compute_gradient(f, c, temperature, n_top=200):
    """
    Compute subgradient of softmax-approximated max(c).
    grad[i] = sum_{k} p[k] * f[k-i] for valid k-i.
    """
    n = len(f)
    len_c = len(c)
    
    # Softmax weights
    scaled = c * temperature
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return np.zeros(n)
    p /= p_sum
    
    # Select top peaks
    if n_top is not None and n_top < len_c:
        top_idx = np.argpartition(c, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len_c)
    
    # Only keep peaks with significant weight
    top_idx = top_idx[p[top_idx] > 1e-12]
    
    grad = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # grad[lo:hi+1] += pk * f[k-hi : k-lo+1][::-1]
        grad[lo:hi+1] += pk * f[k-hi:k-lo+1][::-1]
    
    return grad


def adam_smooth_optimize(f_init, time_limit, lr=0.01, temp_init=10.0, temp_min=1.0,
                          temp_decay=0.9995, n_top=200, label="adam"):
    """Adam optimizer with softmax-approximated max."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_convolve(f)
        M = c.max()
        
        # Compute gradient
        grad = compute_gradient(f, c, temp, n_top=n_top)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new, 1.0)
        f = f_new
        
        # Update temperature (increase for more focus on peaks)
        temp = max(temp_min, temp * temp_decay)
        
        # Check improvement
        if t % 10 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    # Reduce learning rate
                    lr *= 0.9
                    no_improve = 0
                    if lr < 1e-6:
                        # Perturb
                        f = best_f + rng.standard_normal(n) * 0.001
                        f = np.maximum(f, 0)
                        f /= f.sum()
                        m = np.zeros(n)
                        v = np.zeros(n)
                        lr = 0.01
                        temp = temp_init
                        no_improve = 0
        
        if t % 500 == 0:
            score = 2.0 * n * best_obj
            print(f"    {label} iter={t} obj={best_obj:.10f} score={score:.6f} "
                  f"lr={lr:.6f} temp={temp:.2f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def solve_lp_direction(f, n, peak_indices, M, eps=1e-9):
    """Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices."""
    K = len(peak_indices)
    if K == 0:
        return None
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[hi-lo+1-1:hi-lo+1][::-1]  # f[k-hi : k-lo+1] reversed
        # Wait: A[row_i, i] = f[k-i] for i in [lo, hi]
        # So A[row_i, lo:hi+1] = [f[k-lo], f[k-lo-1], ..., f[k-hi]]
        # = f[k-hi:k-lo+1][::-1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solved = False
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps=eps)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solved = True
                break
        except Exception:
            continue
    
    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def lp_optimize(f_init, time_limit, label="lp", max_peaks=300, lp_eps=1e-9):
    """LP-based optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Select peaks
        threshold = 0.9 * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(b, -min(20, len(b)))[-min(20, len(b)):]
        
        g = solve_lp_direction(f, n, peaks, M, eps=lp_eps)
        
        if g is None or g.sum() <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            continue
        
        h = g / g.sum()
        
        # Line search
        best_t = 0.0
        best_obj = M
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t
        
        if best_t <= 0:
            stuck += 1
            if stuck > 15:
                break
            continue
        
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0)
        f /= f.sum()
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if it % 5 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.10f} score={score:.6f} "
                  f"peaks={len(peaks)} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                f /= f.sum()
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break
    
    return f, best_b


def interpolate_to_length(f, new_n):
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

So minimizing the score is equivalent to minimizing max(conv(f, f)) for a normalized f.

"""
Smooth gradient + LP hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam with softmax-approximated max (multiple temperature schedules), LP polishing,
coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import math
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, c, temperature, n_top=None):
    """Compute gradient of softmax-approximated max(c) with respect to f."""
    n = len(f)
    len_c = len(c)
    scaled = c * temperature
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return np.zeros(n)
    p /= p_sum
    if n_top is not None and n_top < len_c:
        top_idx = np.argpartition(c, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len_c)
    top_idx = top_idx[p[top_idx] > 1e-14]
    grad = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        grad[lo:hi + 1] += pk * f[k - hi:k - lo + 1][::-1]
    return grad


def adam_optimize(f_init, time_limit, lr=0.05, temp_init=30.0, temp_min=3.0,

    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:

    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(seed)
    lr_sched = lr
    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_convolve(f)
        M = c.max()
        grad = compute_gradient(f, c, temp, n_top=n_top)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = project_simplex(f_new, 1.0)

        temp = max(temp_min, temp * temp_decay)

def solve_lp_direction(f, n, peak_indices, M, eps=1e-9):
    """LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in peaks."""
    K = len(peak_indices)
    if K == 0:
        return None
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    solved = False
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps=eps)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solved = True
                break
        except Exception:
            continue
    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def lp_optimize(f_init, time_limit, label="lp", max_peaks=300, lp_eps=1e-9):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = 0.93
    rng = np.random.default_rng(777)
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(b, -min(20, len(b)))[-min(20, len(b)):]
        g = solve_lp_direction(f, n, peaks, M, eps=lp_eps)
        if g is None or g.sum() <= 1.0 + 1e-7:
            stuck += 1
            if stuck > 25:
                break
            continue
        h = g / g.sum()
        best_t = 0.0
        best_obj = M
        for t in [0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.4]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0)
            s2 = f_trial.sum()
            if s2 < 1e-10:
                continue
            f_trial /= s2
            obj = fast_convolve(f_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t
        if best_t <= 0:
            stuck += 1
            if stuck > 20:
                break
            continue
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0)
        f /= f.sum()
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-15:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        if it % 10 == 0:
            peak_frac = max(0.82, peak_frac - 0.003)
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.10f} score={score:.6f} "
                  f"peaks={len(peaks)} frac={peak_frac:.3f} t={time.monotonic()-t0:.0f}s")
        if stuck > 15 and stuck % 8 == 0:
            f = best_f + rng.standard_normal(n) * 0.001 * f
            f = np.maximum(f, 1e-15)
            f /= f.sum()
            stuck = 5
    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.0]
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s2 = f_trial.sum()
                if s2 < 1e-10:
                    continue
                f_trial /= s2
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                f /= f.sum()
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break
        if passes % 5 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} pass={passes} obj={best_b:.10f} score={score:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")
    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimizer: Adam smooth gradient (multiple schedules) + LP polishing
    + coordinate descent + multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = 2.0
        print(f"No parent, using uniform n=1024, score=2.0")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Adam with medium temperature (bulk work) ===
    n_parent = len(parent)
    f_norm = parent / parent.sum()
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 400.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Adam T=30 (n={n_parent}, budget={t1:.0f}s) ===")
        f1, obj1 = adam_optimize(f_norm, time_limit=t1, lr=0.05,
                                  temp_init=30.0, temp_min=3.0,
                                  temp_decay=0.9999, n_top=250, label="adam_T30", seed=42)
        vs1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(vs1) and vs1 < best_score:
            best_score = vs1
            best_seq = f1.copy()
            print(f"  NEW BEST [adam_T30]: {vs1:.6f}")

    # === Phase 2: Adam with high temperature (focus on peaks) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 250.0)
    if t2 > 25:
        n_b = len(best_seq)
        f_b = best_seq / best_seq.sum()
        print(f"\n=== Phase 2: Adam T=200 (n={n_b}, budget={t2:.0f}s) ===")
        f2, obj2 = adam_optimize(f_b, time_limit=t2, lr=0.02,
                                  temp_init=200.0, temp_min=20.0,
                                  temp_decay=0.99995, n_top=80, label="adam_T200", seed=123)
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [adam_T200]: {vs2:.6f}")

    # === Phase 3: LP polishing ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 300.0)
    if t3 > 30:
        n_b = len(best_seq)
        f_b = best_seq / best_seq.sum()
        print(f"\n=== Phase 3: LP (n={n_b}, budget={t3:.0f}s) ===")
        f3, obj3 = lp_optimize(f_b, time_limit=t3, label="lp1",
                                max_peaks=400, lp_eps=1e-9)
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp1]: {vs3:.6f}")

    # === Phase 4: Second Adam run (different params) ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 150.0)
    if t4 > 20:
        n_b = len(best_seq)
        f_b = best_seq / best_seq.sum()
        print(f"\n=== Phase 4: Adam T=80 (n={n_b}, budget={t4:.0f}s) ===")
        f4, obj4 = adam_optimize(f_b, time_limit=t4, lr=0.03,
                                  temp_init=80.0, temp_min=8.0,
                                  temp_decay=0.9999, n_top=150, label="adam_T80", seed=555)
        vs4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  NEW BEST [adam_T80]: {vs4:.6f}")

    # === Phase 5: Length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for delta in [-80, -40, 40, 80, 120, -120]:
        tl = n_base + delta
        if 512 <= tl <= 2500 and tl != n_base:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t5 = min(rem * 0.12, 130.0)
        if t5 < 15:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\n=== Phase 5: Explore n={target_n} (budget={t5:.0f}s) ===")
        f5, obj5 = adam_optimize(f_interp, time_limit=t5 * 0.65, lr=0.04,
                                  temp_init=50.0, temp_min=5.0,
                                  temp_decay=0.9999, n_top=200,
                                  label=f"adam_n{target_n}", seed=777)
        vs5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(vs5) and vs5 < best_score:
            best_score = vs5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {vs5:.6f}")
            # Quick LP on the new best
            rem2 = deadline - time.monotonic()
            if rem2 > 30:
                t5b = min(rem2 * 0.3, 60.0)
                f5b, _ = lp_optimize(f5 / f5.sum(), time_limit=t5b, label=f"lp_n{target_n}",
                                      max_peaks=300, lp_eps=1e-9)
                vs5b = float(evaluate_sequence(f5b.tolist()))
                if np.isfinite(vs5b) and vs5b < best_score:
                    best_score = vs5b
                    best_seq = f5b.copy()
                    print(f"  NEW BEST [lp_n{target_n}]: {vs5b:.6f}")

    # === Phase 6: Final LP ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.35, 150.0)
    if t6 > 15:
        n_b = len(best_seq)
        f_b = best_seq / best_seq.sum()
        print(f"\n=== Phase 6: Final LP (n={n_b}, budget={t6:.0f}s) ===")
        f6, obj6 = lp_optimize(f_b, time_limit=t6, label="lp_final",
                                max_peaks=500, lp_eps=1e-10)
        vs6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(vs6) and vs6 < best_score:
            best_score = vs6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {vs6:.6f}")

    # === Phase 7: Coordinate descent ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.40, 100.0)
    if t7 > 10:
        print(f"\n=== Phase 7: CD (budget={t7:.0f}s) ===")
        f7, obj7 = coordinate_descent(best_seq, time_limit=t7, label="cd")
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {vs7:.6f}")

    # === Phase 8: Final Adam polish ===
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.50, 60.0)
    if t8 > 10:
        n_b = len(best_seq)
        f_b = best_seq
        print(f"\n=== Phase 8: Final Adam (budget={t8:.0f}s) ===")
        f8, obj8 = adam_optimize(f_b / f_b.sum(), time_limit=t8, lr=0.01,
                                  temp_init=500.0, temp_min=50.0,
                                  temp_decay=0.99999, n_top=50, label="adam_final", seed=999)
        vs8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(vs8) and vs8 < best_score:
            best_score = vs8
            best_seq = f8.copy()
            print(f"  NEW BEST [adam_final]: {vs8:.6f}")

    # Final coordinate descent with remaining time
    rem = deadline - time.monotonic()
    if rem > 8:
        t9 = min(rem * 0.7, 30.0)
        if t9 > 5:
            f9, _ = coordinate_descent(best_seq, time_limit=t9, label="cd_final")
            vs9 = float(evaluate_sequence(f9.tolist()))
            if np.isfinite(vs9) and vs9 < best_score:
                best_score = vs9
                best_seq = f9.copy()
                print(f"  NEW BEST [cd_final]: {vs9:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - t_start:.0f}s")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
