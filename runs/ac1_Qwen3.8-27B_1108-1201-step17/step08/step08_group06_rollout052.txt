    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

def build_conv_matrix(f, n):
    """Build C matrix: C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C

    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

                      bounds=[(0, None)] * n, method='highs',

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP fictitious play with adaptive geometric mean updates (main algorithm)
2. Efficient coordinate descent with FFT-based convolution and smart pair selection
3. Length exploration with interpolation
4. Simulated annealing on peaks for escape from local minima

Key improvements: better time management, more aggressive CD, peak-aware search,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(f, n):
    """Build C matrix: C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    C = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_fictitious_play(a_init, time_limit, verbose=True):
    """LP fictitious play with geometric mean updates and adaptive step."""
    a = normalize(a_init.copy())
    n = len(a)
    f_avg = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    last_improve = 0
    
    for k in range(500000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_lp_direction(a, M)
        if g is None or g.sum() < 1e-15:
            if verbose:
                print(f"    LP failed at iter {k}")
            break
        g = g / g.sum()
        
        # Adaptive step size: larger when improving, smaller when stagnating
        gap = k - last_improve
        if gap < 50:
            alpha = min(1.0, 2.5 / (k + 2))
        elif gap < 200:
            alpha = min(1.0, 1.5 / (k + 2))
        else:
            alpha = min(1.0, 1.0 / (k + 2))
        
        # Geometric mean update
        a_safe = np.maximum(a, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - alpha) * np.log(a_safe) + alpha * np.log(g_safe)
        log_new = log_new - log_new.max()
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()
        
        # Update running average
        f_avg = (k * f_avg + a_new) / (k + 1)
        
        sc_avg = compute_score(f_avg)
        sc_cur = compute_score(a_new)
        
        improved = False
        if sc_avg < best_score - 1e-13:
            best_score = sc_avg
            best_a = f_avg.copy()
            improved = True
        if sc_cur < best_score - 1e-13:
            best_score = sc_cur
            best_a = a_new.copy()
            improved = True
        
        if improved:
            last_improve = k
        
        a = a_new
        
        if verbose and k % 100 == 0:
            print(f"    LP iter={k}, M={M:.6e}, score={sc_avg:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score

def coordinate_descent(a_init, time_limit, verbose=True, seed=42):

        center = n // 2
        for _ in range(n_pairs // 6):

def peak_annealing(a_init, time_limit, verbose=True, seed=55):
    """Simulated annealing focused on reducing top peaks."""
    a = normalize(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T = 0.01  # Initial temperature
    T_min = 0.0001
    cooling = 0.995
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        peak_val = conv.max()
        
        # Find top peak
        pk = int(np.argmax(conv))
        
        # Contributing indices for this peak
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        idx_range = np.arange(js, je + 1)
        
        # Pick a random contributing index to reduce
        i = int(idx_range[rng.integers(0, len(idx_range))])
        
        # Pick a non-contributing index to receive mass
        noncontrib = np.setdiff1d(np.arange(n), idx_range)
        if len(noncontrib) == 0:
            continue
        j = int(noncontrib[rng.integers(0, len(noncontrib))])
        
        if a[i] < 1e-14:
            continue
        
        # Try different transfer amounts
        fracs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]
        fracs = [f for f in fracs if f * a[i] < a[i] * 0.99]
        if not fracs:
            continue
        frac = fracs[rng.integers(0, len(fracs))]
        delta = a[i] * frac
        
        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        a_trial = normalize(a_trial)
        sc_trial = compute_score(a_trial)
        
        # Metropolis acceptance
        delta_sc = sc_trial - best_score
        if delta_sc < 0 or rng.random() < np.exp(-delta_sc / max(T, T_min)):
            a = a_trial
            if sc_trial < best_score - 1e-13:
                best_score = sc_trial
                best_a = a.copy()
        
        T = max(T_min, T * cooling)
        
        # Reset temperature occasionally
        if iters % 200 == 0:
            T = 0.005
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={iters}, T={T:.5f}, score={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    remaining = deadline - time.monotonic()
    t1 = min(550.0, remaining * 0.35)
    print(f"\n=== Phase 1: LP Fictitious Play (budget={t1:.0f}s) ===")
    a1, s1 = lp_fictitious_play(best_seq, t1, verbose=True)
    best_seq, best_score = evaluate_and_update(a1, "lp1", best_seq, best_score)
    print(f"  After P1: best={best_score:.8f}, elapsed={time.monotonic() - (deadline - remaining):.0f}s")
    
    # === Phase 2: Coordinate Descent (~25%) ===
    remaining = deadline - time.monotonic()
    t2 = min(450.0, remaining * 0.25)
    print(f"\n=== Phase 2: Coordinate Descent (budget={t2:.0f}s) ===")
    a2, s2 = coordinate_descent(best_seq, t2, verbose=True, seed=123)
    best_seq, best_score = evaluate_and_update(a2, "cd1", best_seq, best_score)
    
    # === Phase 3: LP from CD result (~15%) ===
    remaining = deadline - time.monotonic()
    t3 = min(250.0, remaining * 0.15)
    if t3 > 30:
        print(f"\n=== Phase 3: LP Polish (budget={t3:.0f}s) ===")
        a3, s3 = lp_fictitious_play(best_seq, t3, verbose=True)
        best_seq, best_score = evaluate_and_update(a3, "lp2", best_seq, best_score)
    
    # === Phase 4: Peak Annealing (~10%) ===
    remaining = deadline - time.monotonic()
    t4 = min(150.0, remaining * 0.10)
    if t4 > 20:
        print(f"\n=== Phase 4: Peak Annealing (budget={t4:.0f}s) ===")
        a4, s4 = peak_annealing(best_seq, t4, verbose=True, seed=55)
        best_seq, best_score = evaluate_and_update(a4, "anneal", best_seq, best_score)
    
    # === Phase 5: Length Exploration (~10%) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Length Exploration ===")
    alt_lengths = [n_parent - 1, n_parent + 1, n_parent - 2, n_parent + 2,
                   n_parent - 3, n_parent + 3, n_parent + 4, n_parent - 4,
                   n_parent + 6, n_parent - 6, n_parent + 8, n_parent - 8]
    # Filter valid and unique
    seen = set()
    unique_lengths = []
    for L in alt_lengths:
        if 256 <= L <= 2048 and L not in seen:
            seen.add(L)
            unique_lengths.append(L)
    
    for alt_n in unique_lengths[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        t_alloc = min(remaining * 0.12, 70.0)
        if t_alloc < 20:
            break
        print(f"  Trying n={alt_n} (budget={t_alloc:.0f}s)")
        init_alt = interpolate_sequence(best_seq, alt_n)
        
        # Quick LP
        t_lp = t_alloc * 0.5
        a_alt, s_alt = lp_fictitious_play(init_alt, t_lp, verbose=False)
        
        # Quick CD
        t_cd = t_alloc * 0.4
        a_alt2, s_alt2 = coordinate_descent(a_alt, t_cd, verbose=False, seed=99 + alt_n)
        
        best_seq, best_score = evaluate_and_update(a_alt2, f"n{alt_n}", best_seq, best_score)
        print(f"    n={alt_n} result: best so far={best_score:.8f}")
    
    # === Phase 6: Final CD Polish (~5%) ===
    remaining = deadline - time.monotonic()
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Final CD Polish (budget={t6:.0f}s) ===")
        a6, s6 = coordinate_descent(best_seq, t6, verbose=True, seed=777)
        best_seq, best_score = evaluate_and_update(a6, "cd_final", best_seq, best_score)
    
    # === Final LP on best ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t7 = min(remaining * 0.5, 60.0)
        print(f"\n=== Final LP (budget={t7:.0f}s) ===")
        a7, s7 = lp_fictitious_play(best_seq, t7, verbose=False)
        best_seq, best_score = evaluate_and_update(a7, "lp_final", best_seq, best_score)
    
    # === Final verification ===
    if best_seq is None:
        best_seq = parent
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = min(best_score, fs)
    except Exception:
        pass
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
