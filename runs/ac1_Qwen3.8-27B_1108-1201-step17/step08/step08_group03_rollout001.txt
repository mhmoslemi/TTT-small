Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max subgradient with Nesterov acceleration and backtracking
2. Cutting-plane LP with adaptive constraint management
3. Multiplicative weights (exponentiated gradient) method
4. Multi-length exploration with upsample/downsample
Key improvements: better temperature schedules, Nesterov momentum,
periodic LP direction injection, and smarter restart strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_direction(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, b, top_k, temp):
    """Compute soft-max weighted subgradient of max(conv(f,f))."""
    n = len(f)
    # Soft-max weights over the peaks
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # Focus on top peaks
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.ones(len(top_idx)) / len(top_idx)
        w = np.zeros(len(b))
        w[top_idx] = w_sel

    # Subgradient: for each i, grad_i = sum_k w_k * 2*f[k-i] (where valid)
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * 2.0 * f[k - idx]
    return g


def subgradient_nesterov(f_init, time_limit, lr_init=0.003, top_k=60,
                         temp_init=0.0005, temp_min=0.00005, momentum=0.9):
    """Subgradient with Nesterov acceleration and adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov step: evaluate at y = f + momentum * v
        y = f + momentum * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-14:
            y = y / s_y
        else:
            y = f.copy()

        b = fast_convolve(y)

        # Compute subgradient
        g = compute_subgradient(y, b, top_k, temp)

        # Update velocity and position
        v = momentum * v - lr * g
        f_new = f + v
        f_new = project_simplex(f_new)

        # Evaluate and possibly backtrack
        obj_new = fast_convolve(f_new).max()
        if obj_new > best_obj * 1.01 and it > 20:
            # Try smaller step
            f_new2 = f - lr * 0.5 * g
            f_new2 = project_simplex(f_new2)
            obj_new2 = fast_convolve(f_new2).max()
            if obj_new2 < obj_new:
                f_new = f_new2
                obj_new = obj_new2
            else:
                f_new = f  # Don't move
                obj_new = best_obj

        f = f_new

        # Update best
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.95
                temp = max(temp * 0.98, temp_min)

        # Periodic restart if stuck
        if no_improve > 500:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.8

        # Periodic LP direction injection
        if it % 300 == 0 and time.monotonic() - t0 < time_limit - 10:
            b_cur = fast_convolve(f)
            M_cur = b_cur.max()
            active = np.argpartition(b_cur, -30)[-30:]
            active = np.sort(active)
            g_lp = _solve_small_lp(f, n, M_cur, active)
            if g_lp is not None:
                S = g_lp.sum()
                if S > 1e-10:
                    h = g_lp / S
                    # Blend: take a small step in LP direction
                    f_trial = 0.9 * f + 0.1 * h
                    f_trial = project_simplex(f_trial)
                    obj_trial = fast_convolve(f_trial).max()
                    if obj_trial < best_obj:
                        f = f_trial
                        best_obj = obj_trial
                        best_f = f.copy()
                        no_improve = 0
                        v = np.zeros(n)

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    sg_nest it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"t={elapsed:.0f}/{time_limit:.0f}s noimp={no_improve}")

    return best_f, best_obj


def _solve_small_lp(f, n, M, active_idx):
    """Solve a small LP: max sum(g) s.t. g>=0, A@g <= M."""
    K = len(active_idx)
    if K == 0 or n > 5000:
        return None
    A = np.zeros((K, n))
    for i, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() <= 1.0 + 1e-6:
        return None
    return g_val


def cutting_plane_lp(f_init, time_limit, initial_k=40, max_k=400, label="cp"):
    """
    LP direction finding with cutting planes.
    Iteratively adds violated constraints to the active set.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        # Solve LP
        g = _solve_lp(f, n, M, active_idx)

        if g is None:
            stuck += 1
            if stuck > 10:
                break
            if len(active_idx) > initial_k:
                b_cur = fast_convolve(f)
                active_idx = np.sort(np.argpartition(b_cur, -initial_k)[-initial_k:])
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                n_add = min(max_k - len(active_idx), 30)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                not_active = new_top[~np.isin(new_top, active_idx)]
                if len(not_active) > 0:
                    active_idx = np.sort(np.concatenate([active_idx, not_active[:n_add]]))
            continue

        # Cutting plane: check violated constraints
        fg = convolve_direction(f, g)
        violated = fg > M * (1 + 1e-7)
        v_count = violated.sum()

        if v_count > 0 and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(25, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.sort(np.unique(np.concatenate([active_idx, to_add])))
                g = _solve_lp(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()
                if S_g <= 1.0 + 1e-6:
                    stuck += 1
                    if stuck > 25:
                        break
                    continue

        stuck = 0
        h = g / S_g

        # Line search
        best_t = 0.05
        best_t_obj = M
        scan_steps = [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]
        for t in scan_steps:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden-section refinement
        if best_t > 0.02:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(7):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = (1 - m1) * f + m1 * h
                fa1 = np.maximum(fa1, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = (1 - m2) * f + m2 * h
                fa2 = np.maximum(fa2, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = (1 - t_mid) * f + t_mid * h
            f_mid = np.maximum(f_mid, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        # Refresh active set periodically
        if it % 10 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k:
                keep_scores = b[active_idx]
                keep = active_idx[np.argsort(keep_scores)[-initial_k:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.sort(np.unique(np.concatenate([keep, new_top])))
                active_idx = active_idx[np.argsort(b[active_idx])[-min(max_k, len(active_idx)):]]

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} t={elapsed:.0f}/{time_limit:.0f}s")

    return best_f, best_obj


def _solve_lp(f, n, M, active_idx, eps=1e-8):
    """Solve the LP relaxation."""
    K = len(active_idx)
    if K == 0:
        return None
    A = np.zeros((K, n))
    for i, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    except Exception:
        pass
    if g.value is None:
        try:
            prob.solve(solver=cp.ECOS, max_iters=30000)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def multiplicative_weights(f_init, time_limit, lr_init=0.01, top_k=50, temp_init=0.001):
    """Multiplicative weights (exponentiated gradient) method."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(float)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Soft-max weights
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        # Subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * f[k - idx]

        # Multiplicative update: a_i <- a_i * exp(-lr * g_i)
        log_f = np.log(f + 1e-30)
        log_f -= lr * g
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = f.copy()

        obj_new = fast_convolve(f_new).max()
        f = f_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr *= 0.93
                temp *= 0.97
                if no_improve > 2000:
                    # Perturb and restart
                    f = project_simplex(best_f + np.random.default_rng(it).standard_normal(n) * 0.005)
                    best_obj = fast_convolve(f).max()
                    best_f = f.copy()
                    no_improve = 0
                    lr = lr_init * 0.7

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    mw it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"t={elapsed:.0f}/{time_limit:.0f}s noimp={no_improve}")

    return best_f, best_obj


def iaa_diversify(f_init, n_iters=150, peak_fraction=0.3):
    """IAA-based diversification to flatten the auto-convolution."""
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]

        peak_val = R.max()
        floor = peak_fraction * peak_val
        R_target = np.clip(R, floor, peak_val)
        energy = R.sum()
        if R_target.sum() > 1e-10:
            R_target *= energy / R_target.sum()

        R_t_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_t_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = f_init.copy()
            break

    return f


def coordinate_polish(f_init, time_limit, n_sweeps=30):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 500)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                f[j] = best_val * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = best_val * mult
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function: multi-phase optimization combining subgradient,
    cutting-plane LP, multiplicative weights, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # === Phase 1: Subgradient Nesterov on parent ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.18, 280.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Subgradient Nesterov (n={n_parent}) budget={t1:.0f}s ===")
        f1, obj1 = subgradient_nesterov(parent, time_limit=t1, lr_init=0.003,
                                        top_k=60, temp_init=0.0008, temp_min=0.00005)
        s1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_seq = f1.copy()
            print(f"  NEW BEST [sg1]: {s1:.8f}")

    # === Phase 2: Cutting-plane LP (main workhorse) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"\n=== Phase 2: CP-LP (n={n_b}, score={best_score:.8f}) budget={t2:.0f}s ===")
        f2, obj2 = cutting_plane_lp(best_seq, time_limit=t2, initial_k=50, max_k=450, label="cp_main")

            best_score = s2
            best_seq = f2.copy()
            print(f"  NEW BEST [cp_main]: {s2:.8f}")

    # === Phase 3: Multiplicative weights ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 150.0)
    if t3 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: Multiplicative Weights (n={n_b}) budget={t3:.0f}s ===")
        f3, obj3 = multiplicative_weights(best_seq, time_limit=t3, lr_init=0.008,
                                          top_k=50, temp_init=0.001)
        s3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(s3) and s3 < best_score:
            best_score = s3
            best_seq = f3.copy()
            print(f"  NEW BEST [mw]: {s3:.8f}")

    # === Phase 4: Continued LP with more constraints ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 4: CP-LP deep (n={n_b}) budget={t4:.0f}s ===")
        f4, obj4 = cutting_plane_lp(best_seq, time_limit=t4, initial_k=80, max_k=600, label="cp_deep")
        s4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [cp_deep]: {s4:.8f}")

    # === Phase 5: Multi-length exploration ===
    rem = deadline - time.monotonic()
    if rem > 100:
        print(f"\n=== Phase 5: Multi-length exploration (best={best_score:.8f}) ===")
        n_base = len(best_seq)
        target_ns = []
        for factor in [1.2, 1.4, 0.7]:
            tn = int(n_base * factor)
            tn = max(512, min(4000, tn))
            if tn != n_base and tn not in target_ns:
                target_ns.append(tn)

        for target_n in target_ns[:2]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t5 = min(rem * 0.35, 150.0)
            if t5 < 30:
                break

            # Upsample/downsample
            old_x = np.linspace(0.0, 1.0, n_base)
            new_x = np.linspace(0.0, 1.0, target_n)
            f_interp = np.interp(new_x, old_x, best_seq)
            f_interp = np.maximum(f_interp, 0.0)
            s_int = f_interp.sum()
            if s_int > 1e-10:
                f_interp /= s_int
            else:
                f_interp = np.ones(target_n) / target_n

            print(f"  CP-LP n={target_n}, budget={t5:.0f}s")
            f5, _ = cutting_plane_lp(f_interp, time_limit=t5, initial_k=40, max_k=300,
                                     label=f"n{target_n}")
            s5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(s5) and s5 < best_score:
                best_score = s5
                best_seq = f5.copy()
                print(f"  NEW BEST [n{target_n}]: {s5:.8f}")

    # === Phase 6: IAA diversification + LP ===
    rem = deadline - time.monotonic()
    if rem > 80:
        print(f"\n=== Phase 6: IAA + LP (best={best_score:.8f}) ===")
        t6_total = min(rem * 0.25, 160.0)
        if t6_total > 40:
            n_b = len(best_seq)
            t6a = t6_total * 0.5
            f_i = iaa_diversify(best_seq, n_iters=120, peak_fraction=0.25)
            f6a, _ = cutting_plane_lp(f_i, time_limit=t6a, initial_k=40, max_k=250, label="iaa")
            s6a = float(evaluate_sequence(f6a.tolist()))
            if np.isfinite(s6a) and s6a < best_score:
                best_score = s6a
                best_seq = f6a.copy()
                print(f"  NEW BEST [iaa]: {s6a:.8f}")

            rem = deadline - time.monotonic()
            if rem > 30:
                t6b = min(rem * 0.5, 80.0)
                f_i2 = iaa_diversify(best_seq, n_iters=150, peak_fraction=0.5)
                f6b, _ = cutting_plane_lp(f_i2, time_limit=t6b, initial_k=40, max_k=250, label="iaa2")
                s6b = float(evaluate_sequence(f6b.tolist()))
                if np.isfinite(s6b) and s6b < best_score:
                    best_score = s6b
                    best_seq = f6b.copy()
                    print(f"  NEW BEST [iaa2]: {s6b:.8f}")

    # === Phase 7: Final subgradient polish ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 7: Final SG polish (best={best_score:.8f}) ===")
        t7 = min(rem * 0.3, 100.0)
        n_b = len(best_seq)
        f7, _ = subgradient_nesterov(best_seq, time_limit=t7, lr_init=0.001,
                                     top_k=80, temp_init=0.0003, temp_min=0.00002)
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [sg7]: {s7:.8f}")

    # === Phase 8: Final LP burst ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 8: Final LP burst (best={best_score:.8f}) ===")
        t8 = min(rem * 0.7, 120.0)
        n_b = len(best_seq)
        f8, _ = cutting_plane_lp(best_seq, time_limit=t8, initial_k=100, max_k=500, label="final")
        s8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(s8) and s8 < best_score:
            best_score = s8
            best_seq = f8.copy()
            print(f"  NEW BEST [final]: {s8:.8f}")

    # === Final coordinate polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t9 = min(rem * 0.8, 30.0)
        f9, _ = coordinate_polish(best_seq, time_limit=t9, n_sweeps=15)
        s9 = float(evaluate_sequence(f9.tolist()))
        if np.isfinite(s9) and s9 < best_score:
            best_score = s9
            best_seq = f9.copy()
            print(f"  NEW BEST [coord]: {s9:.8f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
