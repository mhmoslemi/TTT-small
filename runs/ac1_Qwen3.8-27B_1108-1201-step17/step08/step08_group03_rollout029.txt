from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_subgradient(f, w):
    """Compute subgradient: g[i] = 2 * sum_{k=0}^{n-1} w[i+k] * f[k]"""
    n = len(f)
    f_rev = f[::-1]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    F = rfft(f_rev, fft_size)
    result = irfft(W * F, fft_size)[:size]
    return 2.0 * result[n-1 : 2*n-1]

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_max_weights(b, T):
    """Compute softmax weights for the convolution peaks."""
    scaled = b / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    return w

def lp_direction(f, n, M, active_idx):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    # A[i, j] = f[j] if j is in the range for constraint i

    return np.maximum(res.x, 0)

def spectral_shaping(f_init, time_limit, label="iaa", n_freq_iter=200):
    """
    Spectral shaping: iteratively flatten the power spectrum while maintaining
    non-negativity in the time domain.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for it in range(n_freq_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute spectrum
        F = rfft(f, fft_size)
        power = np.abs(F) ** 2
        
        # Target: reduce peaks in power spectrum
        # Use a soft thresholding approach
        target = power.copy()
        target[target > np.median(target)] = np.median(target)
        
        # Blend: new amplitude = sqrt(blend of old and target)
        alpha = 0.5
        new_amp = np.sqrt(alpha * power + (1 - alpha) * target)
        
        # Preserve phase
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_amp * phase
        
        # Transform back
        f = irfft(F_new, fft_size)[:n]
        
        # Enforce non-negativity
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = best_f.copy()
            break
        
        # Check improvement
        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
    
    return best_f, best_obj

        K = len(active_idx)
        A = np.zeros((K, n))
        for i, k in enumerate(active_idx):

            hi = min(n - 1, k)
            A[i, lo:hi+1] = f[lo:hi+1]
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0

        try:

            best_f = f.copy()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_subgradient(f, w, fft_size):
    """Compute g[i] = 2 * sum_{k=0}^{n-1} w[i+k] * f[k] via FFT."""
    n = len(f)
    f_rev = f[::-1]
    W = rfft(w, fft_size)
    F = rfft(f_rev, fft_size)
    result = irfft(W * F, fft_size)[:2 * n - 1]
    return 2.0 * result[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    step = max(1, n // 250)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for sweep in range(8):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue

            best_val = f[j]
            best_obj_j = best_obj

            for mult in [0.7, 0.85, 1.15, 1.3]:
                f[j] = f[j] * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                F = rfft(f, fft_size)
                b = irfft(F * F, fft_size)[:size]
                obj = b.max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = f[j]

            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s

            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                improved = True

        if not improved:
            break

    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining fast subgradient, spectral shaping, LP refinement,
    multi-length exploration, and coordinate polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        n_init = 1152
        parent = np.ones(n_init) / n_init
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n={n_init}, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(best_seq)

    # === Phase 1: Primary subgradient on parent length ===
    print(f"\n=== Phase 1: Subgradient primary (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.28, 480.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, _ = subgradient_heavy_ball(best_seq, time_limit=t1, label="sg1", seed=42)
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [sg1]: {score1:.6f}")

    # === Phase 2: Continued subgradient with different params ===
    print(f"\n=== Phase 2: Subgradient continued (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 340.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        f2, _ = subgradient_heavy_ball(best_seq, time_limit=t2, label="sg2", seed=99,
                                       T_init=0.001, lr_init=0.03, momentum_init=0.92)
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [sg2]: {score2:.6f}")

    # === Phase 3: Spectral shaping + subgradient ===
    print(f"\n=== Phase 3: Spectral shaping + SG (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 200.0)
    if t3 > 20:
        n_b = len(best_seq)
        f_iaa = spectral_shaping(best_seq, time_limit=min(t3 * 0.4, 80.0), label="iaa")
        iaa_score = float(evaluate_sequence(f_iaa.tolist()))
        f_sg_from_iaa, _ = subgradient_heavy_ball(f_iaa, time_limit=min(t3 * 0.6, 120.0),
                                                   label="sg_iaa", seed=7, T_init=0.001)
        sg_iaa_score = float(evaluate_sequence(f_sg_from_iaa.tolist()))
        if np.isfinite(iaa_score) and iaa_score < best_score:
            best_score = iaa_score
            best_seq = f_iaa.copy()
            print(f"  NEW BEST [iaa]: {iaa_score:.6f}")
        if np.isfinite(sg_iaa_score) and sg_iaa_score < best_score:
            best_score = sg_iaa_score
            best_seq = f_sg_from_iaa.copy()
            print(f"  NEW BEST [sg_iaa]: {sg_iaa_score:.6f}")

    # === Phase 4: LP refinement ===
    print(f"\n=== Phase 4: LP refinement (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 340.0)
    if t4 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n_b}")
        f4, _ = lp_refine(best_seq, time_limit=t4, initial_k=40, max_k=200, label="lp1")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [lp1]: {score4:.6f}")

        # Second LP pass with more constraints
        rem4 = deadline - time.monotonic()
        t4b = min(rem4 * 0.3, 120.0)
        if t4b > 10:
            f4b, _ = lp_refine(best_seq, time_limit=t4b, initial_k=60, max_k=300, label="lp2")
            score4b = float(evaluate_sequence(f4b.tolist()))
            if np.isfinite(score4b) and score4b < best_score:
                best_score = score4b
                best_seq = f4b.copy()
                print(f"  NEW BEST [lp2]: {score4b:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.6f}) ===")
    n_base = len(best_seq)
    candidates = set()
    for factor in [0.85, 0.9, 1.1, 1.15, 1.25]:
        tl = int(n_base * factor)
        tl = max(640, min(2500, tl))
        if tl != n_base:
            candidates.add(tl)
    candidates = sorted(candidates, key=lambda x: abs(x - n_base))

    for target_n in list(candidates)[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t5 = min(rem * 0.10, 100.0)
        if t5 < 20:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_int = np.interp(new_x, old_x, best_seq)
        f_int = np.maximum(f_int, 0.0)
        s = f_int.sum()
        if s > 1e-10:
            f_int /= s
        else:
            f_int = np.ones(target_n) / target_n

        print(f"  Subgradient n={target_n}, budget={t5:.0f}s")
        f5, _ = subgradient_heavy_ball(f_int, time_limit=t5, label=f"n{target_n}",
                                        seed=target_n % 1000, T_init=0.0015)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {score5:.6f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        # Brief LP
        t6a = min(rem * 0.4, 100.0)
        if t6a > 10:
            f6a, _ = lp_refine(best_seq, time_limit=t6a, initial_k=50, max_k=250, label="final_lp")
            s6a = float(evaluate_sequence(f6a.tolist()))
            if np.isfinite(s6a) and s6a < best_score:
                best_score = s6a
                best_seq = f6a.copy()
                print(f"  NEW BEST [final_lp]: {s6a:.6f}")

        # Coordinate descent
        rem6 = deadline - time.monotonic()
        t6b = min(rem6 * 0.4, 60.0)
        if t6b > 5:
            f6b, _ = coordinate_polish(best_seq, time_limit=t6b, label="cd")
            s6b = float(evaluate_sequence(f6b.tolist()))
            if np.isfinite(s6b) and s6b < best_score:
                best_score = s6b
                best_seq = f6b.copy()
                print(f"  NEW BEST [cd]: {s6b:.6f}")

        # Very brief final subgradient
        rem6b = deadline - time.monotonic()
        t6c = min(rem6b * 0.5, 40.0)
        if t6c > 5:
            f6c, _ = subgradient_heavy_ball(best_seq, time_limit=t6c, label="final_sg",
                                             seed=777, T_init=0.0005, lr_init=0.01,
                                             momentum_init=0.93)
            s6c = float(evaluate_sequence(f6c.tolist()))
            if np.isfinite(s6c) and s6c < best_score:
                best_score = s6c
                best_seq = f6c.copy()
                print(f"  NEW BEST [final_sg]: {s6c:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*55}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*55}")

    return [float(x) for x in best_seq]
