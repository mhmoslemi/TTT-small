def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

- max_affected(x) = max of new_b_k for k in [j, j+n-1]

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations:
1. Nesterov accelerated gradient on soft-max objective with adaptive temperature
2. FFT-based gradient computation for O(n log n) per iteration

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_max_gradient(y, tau):
    """
    Compute soft max of auto-convolution and its gradient w.r.t. y.
    Returns (soft_max_value, gradient, b_max, softmax_weights).
    """
    n = len(y)
    b = fast_convolve(y)
    full_len = 2 * n - 1

    # Numerically stable softmax
    b_max = b.max()
    b_shifted = tau * (b - b_max)
    exp_b = np.exp(b_shifted)
    S = exp_b.sum()
    if S < 1e-300:
        w = np.zeros(full_len)
        w[np.argmax(b)] = 1.0
        return b_max, 2.0 * fast_convolve(y / n)[:n], b_max, w
    w = exp_b / S
    soft_max_val = b_max + np.log(S) / tau

    # Gradient: grad_j = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]
    # = 2 * cross_correlation(w, f)[j]
    # Using FFT: ifft(conj(FFT(w)) * FFT(f))[:n]
    L = 1 << (full_len - 1).bit_length()
    if L < 2 * n - 1:
        L = 1 << (2 * n - 1).bit_length()
    W = np.fft.fft(w, L)
    F = np.fft.fft(y, L)
    grad = 2.0 * np.fft.ifft(np.conj(W) * F, L).real[:n]

    return soft_max_val, grad, b_max, w


def nesterov_smooth_max(f_init, time_limit, tau_init=100.0, tau_final=50000.0,
                         label="nesterov"):
    """
    Nesterov accelerated gradient on soft max of auto-convolution.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    y = f.copy()
    it = 0
    no_improve = 0
    tau = tau_init

    # Estimate number of iterations for temperature schedule
    # Assume ~200 iterations per second (rough)
    est_iters = int(time_limit * 150)
    tau_decay = (tau_final / tau_init) ** (1.0 / max(10, est_iters)) if tau_final > tau_init else 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        tau = min(tau * tau_decay, tau_final)

        sm_val, grad, b_max, w = compute_soft_max_gradient(y, tau)

        # Adaptive step size
        base_step = 1.0 / (np.sqrt(it + 1.0) * 0.5 + np.max(np.abs(grad)) * 0.1 + 1e-8)

        # Try several step sizes
        best_y = y.copy()
        best_sm = sm_val
        for mult in [2.0, 1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = base_step * mult
            y_trial = y - step * grad
            y_trial = project_simplex(y_trial)
            sm_trial = fast_convolve(y_trial).max()  # Use hard max for evaluation
            if sm_trial < best_sm:
                best_sm = sm_trial
                best_y = y_trial.copy()

        f_new = best_y

        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta = min(0.95, 1.0 - 2.0 / (beta + 1.0))
        y = f_new + beta * (f_new - f)
        y = project_simplex(y)
        f = f_new

        # Track best
        if it % 3 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            cur_obj = fast_convolve(f).max()
            print(f"    {label} it={it} tau={tau:.0f} obj={cur_obj:.12f} "
                  f"best_score={best_score:.6f} t={elapsed:.0f}s")
            if no_improve > 60:
                f = best_f.copy()
                y = f.copy()
                no_improve = 0

        if no_improve > 150:
            break

    return best_f, best_score


def subgradient_optimize(f_init, time_limit, lr_init=0.005, n_top=100,
                         momentum=0.85, temp_init=0.001, label="sg"):
    """Subgradient optimization with adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)

    full_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weighting
        scaled = (b - M) / temp
        np.clip(scaled, -50, 50, out=scaled)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(full_len)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Restrict to top n_top
        if n_top < full_len:
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w_new = np.zeros(full_len)
            w_new[top_idx] = w_sel
            w = w_new

        # Compute subgradient via FFT
        L = 1 << (full_len - 1).bit_length()
        W = np.fft.fft(w, L)
        F = np.fft.fft(f, L)
        grad = 2.0 * np.fft.ifft(np.conj(W) * F, L).real[:n]

        # Update with momentum
        v = momentum * v - lr * grad
        f = project_simplex(f + v)

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.03)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    no_improve = 0

            if it % 100 == 0:
                temp = min(temp * 1.15, 0.5)
                print(f"    {label} it={it} obj={best_obj:.12f} "
                      f"score={2.0*n*best_obj:.6f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                no_improve = 0

    return best_f, 2.0 * n * best_obj


def incremental_coord_descent(f_init, time_limit, label="cd"):
    """
    Coordinate descent with incremental convolution updates.
    For each coordinate j, find optimal x by evaluating the 1D objective.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    b = fast_convolve(f)
    full_len = 2 * n - 1
    best_f = f.copy()
    best_obj = b.max()
    best_score = 2.0 * n * best_obj

    no_sweep_improve = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_improved = False
        S = f.sum()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            f_j = f[j]
            if f_j < 1e-15:
                continue

            # Affected indices: k from j to min(j+n-1, full_len-1)
            k_start = j
            k_end = min(j + n - 1, full_len - 1)
            affected_ks = np.arange(k_start, k_end + 1)

            # Unaffected max
            mask = np.ones(full_len, dtype=bool)
            mask[k_start:k_end + 1] = False
            M_unaff = b[mask].max() if mask.any() else 0.0

            # For each affected k, compute the coefficient of x in new_b_k
            # new_b_k = b_k + 2*(x - f_j)*f[k-j] for k != 2j
            # new_b_k = b_k - f_j^2 + x^2 for k == 2j
            coeffs = np.zeros(len(affected_ks))
            consts = np.zeros(len(affected_ks))
            quad_mask = np.zeros(len(affected_ks), dtype=bool)

            for idx, k in enumerate(affected_ks):
                if k == 2 * j:
                    quad_mask[idx] = True
                    consts[idx] = b[k] - f_j * f_j
                    coeffs[idx] = 0.0
                else:
                    # new_b_k = b_k + 2*(x - f_j)*f[k-j]
                    # = 2*f[k-j]*x + (b_k - 2*f_j*f[k-j])
                    c = f[k - j]
                    coeffs[idx] = 2.0 * c
                    consts[idx] = b[k] - 2.0 * f_j * c

            # Try several multipliers for x
            S_rest = S - f_j
            best_x = f_j
            best_obj_j = best_obj

            multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0]
            for mult in multipliers:
                x = f_j * mult
                if x < 0:
                    continue
                S_new = S_rest + x
                if S_new < 1e-15:
                    continue

                # Compute new affected b values
                new_b_affected = coeffs * x + consts
                quad_indices = np.where(quad_mask)[0]
                for qi in quad_indices:
                    new_b_affected[qi] = x * x + consts[qi]

                max_affected = new_b_affected.max()
                max_total = max(max_affected, M_unaff)
                obj = 2.0 * n * max_total / (S_new * S_new)

                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_x = x

            if best_x != f_j:
                # Update f and b
                old_b_affected = b[k_start:k_end + 1].copy()
                f[j] = best_x
                S = S_rest + best_x

                # Update b
                for idx, k in enumerate(affected_ks):
                    if k == 2 * j:
                        b[k] = best_x * best_x + consts[idx]
                    else:
                        b[k] = coeffs[idx] * best_x + consts[idx]

                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_score = best_obj_j
                    best_f = f.copy()
                    sweep_improved = True

        if not sweep_improved:
            no_sweep_improve += 1
            if no_sweep_improve >= 3:
                break
        else:
            no_sweep_improve = 0

        if sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} sweep={sweep} obj={best_obj:.12f} "
                  f"score={best_score:.6f} t={elapsed:.0f}s")

    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def build_constraint_matrix(f, indices, n):
    """Build K x n constraint matrix for given peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, indices, M):
    """
    Solve: max sum(g) s.t. g >= 0, A@g <= M.
    """
    n = len(f)
    K = len(indices)
    A = build_constraint_matrix(f, indices, n)

    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(objective, constraints)

    try:
        if K <= 500 and n <= 1200:
            prob.solve(solver=cp.ECOS, max_iters=20000)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def lp_polish(f_init, time_limit, top_k_init=50, top_k_max=400, label="lp"):
    """LP-based polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    it = 0
    stuck = 0
    top_k = top_k_init
    momentum_dir = np.zeros(n)
    full_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

        b = fast_convolve(f)
        M = b.max()

        if top_k < full_len:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
            indices = np.sort(top_idx)
        else:
            indices = np.arange(full_len)

        g = solve_lp_direction(f, indices, M)

        if g is None:
            if len(indices) > 15:
                top_15 = np.sort(np.argpartition(b, -15)[-15:])
                g = solve_lp_direction(f, top_15, M)
            if g is None:
                stuck += 2
                if stuck > 50:
                    break
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 30:
                break
            continue

        h = g / S_g
        momentum_dir = 0.3 * momentum_dir + 0.7 * h
        mom_sum = momentum_dir.sum()
        if mom_sum > 1e-10:
            h_mom = momentum_dir / mom_sum
        else:
            h_mom = h

        # Line search
        best_t_obj = M
        best_f_trial = f.copy()
        best_t = 0.0

        for direction in [h, h_mom]:
            for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                f_trial = (1.0 - t) * f + t * direction
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial = f_trial / st
                obj = fast_convolve(f_trial).max()
                if obj < best_t_obj:
                    best_t_obj = obj
                    best_t = t
                    best_f_trial = f_trial.copy()

        if best_t == 0.0:
            stuck += 1
            if stuck > 50:
                break
            continue

        f = best_f_trial
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break

        if stuck > 25:
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            stuck = 0
            momentum_dir = np.zeros(n)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} "
                  f"score={best_score:.6f} t={elapsed:.0f}s")

    return best_f, best_score

    t1 = min(rem * 0.30, 480.0)

    if t1 > 60:
        # Try parent n and a few nearby values
        n_list = sorted(set([
            n_parent,
            int(n_parent * 0.9),
            int(n_parent * 1.1),
            int(n_parent * 0.75),
            int(n_parent * 1.25),
        ]))
        n_list = [n for n in n_list if 256 <= n <= 3000]

        per_n = t1 / len(n_list)
        for n_trial in n_list:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            alloc = min(per_n * 1.2, rem * 0.85)
            if alloc < 20:
                break

            f_init = interp_sequence(parent, n_trial)
            print(f"  Nesterov n={n_trial}, budget={alloc:.0f}s")
            f_res, score = nesterov_smooth_max(f_init, time_limit=alloc,
                                                tau_init=100.0, tau_final=50000.0,
                                                label=f"ng{n_trial}")

            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [nesterov n={n_trial}]: {real_score:.6f}")

    # ============================================================
    # Phase 2: Subgradient with low temperature
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Subgradient polish (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)

    if t2 > 30:
        n_b = len(best_seq)
        print(f"  SG n={n_b}, budget={t2:.0f}s")
        f_res, _ = subgradient_optimize(
            best_seq, time_limit=t2, lr_init=0.003,
            n_top=max(50, n_b // 8),
            momentum=0.88, temp_init=0.0005, label="sg1")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [sg]: {real_score:.6f}")

    # ============================================================
    # Phase 3: Incremental coordinate descent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Incremental CD (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 400.0)

    if t3 > 30:
        n_b = len(best_seq)
        print(f"  CD n={n_b}, budget={t3:.0f}s")
        f_res, _ = incremental_coord_descent(best_seq, time_limit=t3, label="cd1")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [cd]: {real_score:.6f}")

    # ============================================================
    # Phase 4: LP polishing
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: LP polish (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)

    if t4 > 30:
        n_b = len(best_seq)
        print(f"  LP n={n_b}, budget={t4:.0f}s")
        f_res, _ = lp_polish(best_seq, time_limit=t4,
                             top_k_init=50, top_k_max=min(500, 2 * n_b - 1),
                             label="lp")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [lp]: {real_score:.6f}")

    # ============================================================
    # Phase 5: Final CD + SG
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t5 = min(rem * 0.35, 400.0)

    if t5 > 20:
        n_b = len(best_seq)
        t5a = t5 * 0.6
        t5b = t5 * 0.4

        print(f"  Final CD n={n_b}, budget={t5a:.0f}s")
        f_res, _ = incremental_coord_descent(best_seq, time_limit=t5a, label="cd2")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [cd2]: {real_score:.6f}")

        rem = deadline - time.monotonic()
        if rem > 15:
            t5b = min(t5b, rem * 0.8)
            print(f"  Final SG n={len(best_seq)}, budget={t5b:.0f}s")
            f_res, _ = subgradient_optimize(
                best_seq, time_limit=t5b, lr_init=0.002,
                n_top=max(40, len(best_seq) // 10),
                momentum=0.9, temp_init=0.0002, label="sg2")
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [sg2]: {real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy() if parent is not None else np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
