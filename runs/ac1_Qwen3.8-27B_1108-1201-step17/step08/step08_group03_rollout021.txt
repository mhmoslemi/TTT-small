"""
Optimization of 2*n*max(conv(a,a))/(sum(a))^2 using cutting-plane LP with
adaptive constraint growth, spectral diversification, and multi-start strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import sys

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_pair(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1)
    return np.maximum(v - theta, 0.0)

def build_A(f, indices, n):
    """Build constraint matrix for (f*g)[k] = A[k,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A

def solve_lp(f, n, M, indices):
    """Solve max sum(g) s.t. g>=0, (f*g)[indices] <= M."""
    K = len(indices)
    A = build_A(f, indices, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.MOSEK, max_iters=500000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() <= 0:
        return None
    return g_val

def lp_optimize(f_init, time_limit, init_k=200, max_k=1500, verbose=True):

    s = f.sum()
    if s > 0:
        f /= s
    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_obj = M
    
    # Initial active set: top init_k peaks
    if init_k < len(b):
        active = np.argpartition(b, -init_k)[-init_k:]
    else:
        active = np.arange(len(b))
    active = np.sort(active)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        # Solve LP
        g = solve_lp(f, n, M, active)
        if g is None or g.sum() <= 1.0 + 1e-8:

            if len(active) > init_k:
                active = active[np.argsort(b[active])[:init_k]]  # Keep lowest b values? No, keep top
                active = active[np.argsort(b[active])[-init_k:]]
                continue

        fg = convolve_pair(f, g)

                    g = solve_lp(f, n, M, active)

                    continue

            f_mid /= s_mid

                best_t_obj = o_mid
                best_t = t_mid

        if s > 1e-10:
            f /= s
        else:
            f = best_f.copy()
        
        cur = fast_convolve(f).max()
        M = cur
        if cur < best_obj - 1e-14:
            best_obj = cur
            best_f = f.copy()

def spectral_diversify(f_init, n_iters=100, seed=42):
    """
    Spectral method to diversify and flatten the auto-convolution.
    Works in frequency domain to reduce peak time-domain auto-correlation.
    """
    rng = np.random.default_rng(seed)
    n = len(f_init)
    f = f_init.copy()
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        
        # Identify peaks and try to reduce them
        R_peaks = R.max()
        # Create a "soft penalty" on peaks
        penalty = 0.5
        R_target = R.copy()
        R_target[R_target > penalty * R_peaks] = (R_target[R_target > penalty * R_peaks] + penalty * R_peaks) / 2
        
        # Preserve total sum
        if R_target.sum() > 1e-10:
            R_target *= R.sum() / R_target.sum()
        
        # Update amplitude
        R_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_fft.real, 0)
        new_amp = np.sqrt(real_part)
        
        # Alternative: use iterative amplitude adjustment (like Gerchberg-Saxton)
        # Keep phase, adjust amplitude to reduce peaks
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_amp * phase
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            continue
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
        
        # Accept with some probability (simulated annealing-like)
        if obj <= best_obj or rng.random() < np.exp(-(obj - best_obj) / (best_obj * 0.01)):
            f = f_new
    
    return best_f, best_obj

def subgradient_polish(f_init, time_limit, lr=0.001, n_top=100):

    t0 = time.monotonic()

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    momentum = 0.9

    no_improve = 0
    rng = np.random.default_rng(123)
    it = 0

        it += 1
        b = fast_convolve(f)
        
        # Soft-max weights on top peaks
        if n_top < len(b):

        else:
            top = np.arange(len(b))
        
        w = b[top].copy()

        v = momentum * v - lr * g

        if it % 100 == 0:

"""
Optimization of 2*n*max(conv(a,a))/(sum(a))^2 using cutting-plane LP with
adaptive constraint growth, spectral diversification, multi-start, and
multi-length exploration. Key improvements: larger active constraint sets,
tighter LP solving, parabolic line search, and diverse initializations.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_pair(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    """Build A where A[i,:] @ g = (f*g)[indices[i]]."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, indices, use_mosek=True):
    """Solve max sum(g) s.t. g>=0, (f*g)[k] <= M for k in indices."""
    K = len(indices)
    if K == 0:
        return None
    A = build_constraint_matrix(f, indices, n)
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    solved = False
    if use_mosek:
        try:
            prob.solve(solver=cp.MOSEK, max_iters=1000000)
            solved = True
        except Exception:
            pass
    if not solved:
        try:
            prob.solve(solver=cp.PDLP, max_iter=300000)
            solved = True
        except Exception:
            pass
    if not solved:
        try:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
            solved = True
        except Exception:
            pass
    if g_var.value is None:
        return None
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() <= 1e-10:
        return None
    return g_val


def lp_optimize(f_init, time_limit, init_k=300, max_k=2000, verbose=False):
    """
    Cutting-plane LP optimization: iteratively minimize max(auto-conv)
    by solving LPs with growing constraint sets.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    else:
        f = np.ones(n) / n
    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_obj = M

    if init_k < len(b):
        active = np.argpartition(b, -init_k)[-init_k:]
    else:
        active = np.arange(len(b))
    active = np.sort(active)

    it = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        g = solve_lp_direction(f, n, M, active)

        if g is None or g.sum() <= 1.0 + 1e-7:
            stuck_count += 1
            if stuck_count > 15:
                break
            if len(active) > init_k:
                active = active[np.argsort(b[active])[-init_k:]]
            M = M * (1 + 1e-6)
            continue

        fg = convolve_pair(f, g)
        violated = fg > M * (1 + 1e-5)
        v_idx = np.where(violated)[0]
        new_violated = v_idx[~np.isin(v_idx, active)]

        if len(new_violated) > 0 and len(active) < max_k:
            n_add = min(100, len(new_violated))
            to_add = new_violated[np.argsort(fg[new_violated])[-n_add:]]
            active = np.unique(np.concatenate([active, to_add]))
            if len(active) > max_k:
                active = active[np.argsort(fg[active])[-max_k:]]
            g = solve_lp_direction(f, n, M, active)
            if g is None or g.sum() <= 1.0 + 1e-7:
                if len(active) > init_k:
                    active = active[np.argsort(b[active])[-init_k:]]
                    g = solve_lp_direction(f, n, M, active)
                if g is None or g.sum() <= 1.0 + 1e-7:
                    stuck_count += 1
                    if stuck_count > 15:
                        break
                    continue

        stuck_count = 0
        h = g / g.sum()

        # Line search with parabolic refinement
        best_t = 0.1
        best_t_obj = M
        trial_ts = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8]
        objs = []
        for t in trial_ts:
            ft = np.maximum((1.0 - t) * f + t * h, 0.0)
            st = ft.sum()
            if st < 1e-10:
                objs.append(np.inf)
                continue
            ft /= st
            obj = fast_convolve(ft).max()
            objs.append(obj)
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Parabolic refinement around best_t
        if 0 < best_t < 1.0:
            lo = max(0.0, best_t - 0.15)
            hi = min(1.0, best_t + 0.15)
            for _ in range(10):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.maximum((1.0 - m1) * f + m1 * h, 0.0)
                s1 = f1.sum()
                o1 = fast_convolve(f1 / s1).max() if s1 > 1e-10 else np.inf
                f2 = np.maximum((1.0 - m2) * f + m2 * h, 0.0)
                s2 = f2.sum()
                o2 = fast_convolve(f2 / s2).max() if s2 > 1e-10 else np.inf
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1.0 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        sf = f.sum()
        if sf > 1e-10:
            f /= sf
        else:
            f = best_f.copy()
            M = best_obj
            continue

        cur_obj = fast_convolve(f).max()
        M = cur_obj
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        # Periodically update active set
        if it % 15 == 0:
            b = fast_convolve(f)
            new_top = np.argpartition(b, -min(max_k, len(b)))[-min(max_k, len(b)):]
            active = np.unique(np.concatenate([active, new_top]))
            if len(active) > max_k:
                active = active[np.argsort(b[active])[-max_k:]]

        if verbose and it % 20 == 0:
            score = 2.0 * n * best_obj
            print(f"    LP it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"active={len(active)} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def spectral_diversify(f_init, n_iters=150, seed=42):
    """Spectral method to diversify and flatten auto-convolution."""
    rng = np.random.default_rng(seed)
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        R_peak = R.max()

        # Soft-clip peaks
        threshold = 0.4 * R_peak
        R_target = R.copy()
        mask = R_target > threshold
        R_target[mask] = (R_target[mask] + threshold) / 2.0

        # Preserve total
        tsum = R_target.sum()
        if tsum > 1e-10:
            R_target *= R.sum() / tsum

        # Update amplitude
        R_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn > 1e-10:
            f_new /= sn
        else:
            continue

        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()

        # Accept (greedy with occasional perturbation)
        if obj <= best_obj + 1e-12:
            f = f_new
        elif rng.random() < 0.1:
            f = f_new

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.002, n_top=150):
    """Subgradient method with momentum for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    momentum = 0.85
    no_improve = 0
    rng = np.random.default_rng(77)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        if n_top < len(b):
            top = np.argpartition(b, -n_top)[-n_top:]
        else:
            top = np.arange(len(b))

        w = b[top].copy()
        w_max = w.max()
        if w_max > 0:
            w = w / w_max
        w = w ** 3
        wsum = w.sum()
        if wsum > 1e-10:
            w /= wsum

        g = np.zeros(n)
        for j in range(len(top)):
            if w[j] < 1e-12:
                continue
            k = top[j]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[j] * 2.0 * f[k - idx]

        v = momentum * v - lr * g
        f_new = project_simplex(f + v)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                f_new = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr *= 0.9
                no_improve = 0

        f = f_new
        if lr < 1e-7:
            break

        if it % 200 == 0:
            print(f"    SG it={it} obj={best_obj:.10f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def generate_block_candidate(n, n_blocks=4, seed=0):
    """Generate a multi-block candidate with good initial structure."""
    rng = np.random.default_rng(seed)
    f = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        width = end - start
        # Each block has a smooth bump
        t = np.linspace(0, 1, width)
        bump = np.sin(np.pi * t) ** 2
        f[start:end] = bump
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    return f
