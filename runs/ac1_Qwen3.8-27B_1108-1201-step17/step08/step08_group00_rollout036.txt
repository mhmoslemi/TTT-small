```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp
smoothing, Nesterov-accelerated subgradient descent, and multi-length
exploration. Focuses on efficient gradient computation via FFT and
aggressive temperature annealing.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad_obj(T, a):
    """Compute gradient of (2n/T)*log(sum exp(T*c_k)) w.r.t. a.
    Returns (value, grad_a) where grad_a is gradient w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / exp_Tc.sum()
    log_sum_exp = Tc_max + np.log(exp_Tc.sum())
    val = (2.0 * n / T) * log_sum_exp
    
    # grad_a[i] = 4n * sum_k w_k * a_{k-i} = 4n * correlate(w, a, 'valid')[i]
    # np.correlate(w, a, 'valid') gives sum_j w[j+i]*a[j] for i=0..len(w)-len(a)
    # len(w) = 2n-1, len(a) = n, so we get n values: sum_j w[j+i]*a[j]
    # = sum_k w[k]*a[k-i] where k=j+i
    # This is exactly what we want!
    grad_a = 4.0 * n * np.correlate(w, a, mode='valid')
    
    return val, grad_a

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def lbgfs_refine(a0, time_limit, T_start=500.0, T_end=0.5, n_stages=15, seed=42):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0.copy()).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        # Scale iterations based on remaining time and temperature
        # At high T, fewer iterations needed; at low T, more iterations
        iters = min(5000, max(50, int(remaining * 200 / (n_stages - si))))
        
        def func(z_inner):
            a_inner = softmax_from_z(z_inner)
            val, _ = compute_grad_obj(T_val, a_inner)
            return val
        
        def jac(z_inner):
            a_inner = softmax_from_z(z_inner)
            val, grad_a = compute_grad_obj(T_val, a_inner)
            # Transform: grad_z = a * (grad_a - sum(a * grad_a))
            S = np.dot(a_inner, grad_a)
            return a_inner * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at T={T_val:.1f}: {e}")
        
        a_new = softmax_from_z(z)
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS T={T_val:.1f} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def nesterov_subgrad(a0, time_limit, lr_decay=0.999, seed=42):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    k = 0
    
    while time.monotonic() - t0 < time_limit:
        k += 1
        # Compute subgradient (use smoothing at T=1 for gradient)
        T = 1.0
        _, grad_a = compute_grad_obj(T, y)
        
        # Nesterov step
        alpha = 2.0 / (k + 2)
        
        # Mirror descent step (exponential weights)
        z_y = z_from_a(y)
        z_new = z_y - 0.1 / np.sqrt(k) * grad_a
        y_new = softmax_from_z(z_new)
        
        # Nesterov combination
        a_new = ((k * a + alpha * (k + 2) * y_new) / (k + 1 + alpha * (k + 2)))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        y = y_new
        
        if k % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad k={k} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def peak_redistribute(a, n_iter=10000, time_limit=60.0):

    n = len(a)
    a = a.copy()
    best_a = a.copy()

    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Find top peaks
        top_k = np.argsort(c)[::-1][:3]
        
        improved = False
        for k in top_k:
            if k < 0 or k >= 2*n-1:
                continue
            # Find contributing indices
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            
            # Reduce the top contributor
            top_i = idx[np.argmax(contribs)]
            if a[top_i] < 1e-10:
                continue
            
            # Try reducing and redistributing
            for factor in [0.97, 0.95, 0.93]:
                a_trial = a.copy()
                diff = a_trial[top_i] * (1 - factor)
                a_trial[top_i] -= diff
                # Distribute to random positions
                mask = np.ones(n, dtype=bool)
                mask[top_i] = False
                others = a_trial[mask]
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try coordinated reduction of multiple peaks
            a_trial = a.copy()
            for k in top_k[:2]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[k - idx]
                top_i = idx[np.argmax(contribs)]
                if a_trial[top_i] > 1e-10:
                    a_trial[top_i] *= 0.98
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:

                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp
smoothing at multiple temperatures, multi-length exploration, and
coordinate-based peak redistribution. Uses efficient FFT-based
convolution and correlate for gradient computation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_smoothed_obj(z, T):
    """Compute value and gradient of (2n/T)*log(sum exp(T*c_k)) w.r.t. z."""
    n = len(z)
    a = softmax_from_z(z)
    c = fast_convolve(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    exp_sum = exp_Tc.sum()
    w = exp_Tc / exp_sum
    log_sum_exp = Tc_max + np.log(exp_sum)
    val = (2.0 * n / T) * log_sum_exp

    # grad_a[i] = 4n * sum_k w_k * a_{k-i} = 4n * correlate(w, a, 'valid')[i]
    grad_a = 4.0 * n * np.correlate(w, a, mode='valid')
    # Transform to z-space
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    return val, grad_z


def lbgfs_func(z, T):
    val, _ = compute_smoothed_obj(z, T)
    return val


def lbgfs_jac(z, T):
    _, grad_z = compute_smoothed_obj(z, T)
    return grad_z


def lbgfs_refine(a0, time_limit, T_start=200.0, T_end=0.3, n_stages=20, verbose=True):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0.copy()).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Allocate iterations: more at low T
        frac = (n_stages - si) / n_stages
        iters = max(30, min(8000, int(remaining * 150 / frac)))

        try:
            res = minimize(
                fun=lbgfs_func, x0=z, args=(T_val,),
                jac=lbgfs_jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error T={T_val:.2f}: {e}")

        a_new = softmax_from_z(z)
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        if verbose and (si % 4 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS T={T_val:.2f} ({elapsed:.0f}s/{time_limit:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_multi_restart(a0, time_limit, T_start=200.0, T_end=0.3, n_stages=15,
                        n_restarts=3, verbose=True):
    """L-BFGS with multiple restarts to escape local minima."""
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    current = a0.copy()

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break

        if restart > 0:
            # Perturb current best
            rng = np.random.default_rng(restart * 1000 + 42)
            noise_scale = 0.02 / restart
            perturbed = current + noise_scale * rng.standard_normal(n)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            current = perturbed

        t_alloc = remaining * 0.7 / (n_restarts - restart)
        a_new, _ = lbgfs_refine(current, t_alloc, T_start=T_start, T_end=T_end,
                                n_stages=n_stages, verbose=verbose)
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            current = a_new.copy()

    return best_a, best_score


def structured_init(n, variant=0, seed=0):
    """Create structured initial sequence with variety."""
    x = np.linspace(0, 1, n)
    rng = np.random.default_rng(seed)

    if variant == 0:
        a = np.sin(np.pi * x) ** 2
    elif variant == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 2:
        a = (1 + np.cos(2 * np.pi * x)) / 2
        a = a ** 0.7
    elif variant == 3:
        sigma = rng.uniform(0.15, 0.35)
        a = np.exp(-(x - 0.5) ** 2 / (2 * sigma ** 2))
    elif variant == 4:
        p = rng.uniform(0.5, 2.5)
        a = np.sin(np.pi * x) ** p
    elif variant == 5:
        # Two-peak structure
        a = np.sin(2 * np.pi * x) ** 2 + 0.5 * np.sin(np.pi * x) ** 2
    elif variant == 6:
        # Flat top with cosine taper
        t = 0.3
        a = np.ones(n)
        mask_low = x < t
        mask_high = x > 1 - t
        a[mask_low] = 0.5 * (1 + np.cos(np.pi * x[mask_low] / t))
        a[mask_high] = 0.5 * (1 + np.cos(np.pi * (1 - x[mask_high]) / t))
    elif variant == 7:
        # Power-law taper
        beta = rng.uniform(0.3, 1.5)
        a = 1.0 - (2 * x - 1) ** (2 * beta)
        a = np.maximum(a, 0)
    elif variant == 8:
        # Bimodal
        a = np.exp(-((x - 0.3) ** 2) / 0.04) + np.exp(-((x - 0.7) ** 2) / 0.04)
    else:
        a = np.sin(np.pi * x) ** 2

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_redistribute(a, time_limit=60.0, verbose=True):
    """Reduce peaks by carefully redistributing mass."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_convolve(a)
        c_max = c.max()

        # Find top 5 peaks
        top_peaks = np.argsort(c)[::-1][:5]

        improved = False
        for k in top_peaks:
            if k < 0 or k >= 2 * n - 1:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]

            # Try reducing top 3 contributors
            top_contribs = np.argsort(contribs)[::-1][:3]
            for ci in top_contribs:
                i = idx[ci]
                for factor in [0.96, 0.94, 0.92, 0.89]:
                    diff = a[i] * (1 - factor)
                    if diff < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= diff
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()

                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1

            # Try pair reduction
            k = top_peaks[0]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top2 = np.argsort(contribs)[::-1][:2]
            if len(top2) >= 2:
                i1 = idx[top2[0]]
                i2 = idx[top2[1]]
                if i1 != i2:
                    a_trial = a.copy()
                    a_trial[i1] *= 0.95
                    a_trial[i2] *= 0.95
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s > 1e-15:
                        a_trial = a_trial / s
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            no_improve = 0

        if not improved and no_improve > 50:
            # Try random perturbation
            rng = np.random.default_rng(hash((n, no_improve)) & 0xFFFFFFFF)
            a_trial = a + 0.01 * rng.standard_normal(n)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-10:
                a = a_trial
                best_score = score
                best_a = a.copy()

    if verbose:
        print(f"    PeakRedist: score={best_score:.6f}")
    return best_a, best_score


def coordinate_descent(a, time_limit=60.0, n_top=5, verbose=True):
    """Reorder-based coordinate descent: try swapping values between positions."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0
    while time.monotonic() - t0 < time_limit and no_improve < 500:
        c = fast_convolve(a)
        c_max = c.max()
        top_peaks = np.argsort(c)[::-1][:n_top]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_i = idx[np.argmax(contribs)]

            # Try moving mass from top_i to lowest-contribution position
            if a[top_i] < 1e-10:
                continue
            # Find position with lowest value (to redistribute)
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                continue
            target = candidates[np.argmin(a[candidates])]

            for frac in [0.1, 0.2, 0.3]:
                diff = a[top_i] * frac
                a_trial = a.copy()
                a_trial[top_i] -= diff
                a_trial[target] += diff
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-15:
                    a_trial = a_trial / s
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
            if improved:
                break

        if not improved:
            no_improve += 1

    if verbose:
        print(f"    CoordDesc: score={best_score:.6f}")
    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence safely using evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
```
