Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Smooth gradient descent (log-sum-exp) with Nesterov momentum

combined with fast LP direction finding for global mixing.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized (sum=1) non-negative sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_C_matrix(f):
    """Build the convolution constraint matrix C where (C@g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_fast(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M*1, g >= 0 using scipy HiGHS."""
    n = len(f)
    C = build_C_matrix(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def solve_lp_fast_dual(f, M):
    """Solve dual LP via scipy: min M*sum(y) s.t. C^T y >= 1, y >= 0."""
    n = len(f)
    C = build_C_matrix(f)
    c = M * np.ones(2 * n - 1)
    A_ub = -C.T
    b_ub = -np.ones(n)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * (2 * n - 1), method='highs')
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s

    if s < 1e-15:

    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    f_avg = f.copy()
    t0 = time.monotonic()
    n_iter = 0

    for k in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        n_iter += 1

        conv = fast_convolve(f)
        M = conv.max()

        g = solve_lp_fast(f, M)
        if g is None or g.sum() < 1e-15:
            break

        g = g / g.sum()

        # Adaptive step size
        t_step = 2.0 / (k + 2)
        if k > 200:
            t_step = max(0.02, 1.5 / (k + 2))

        if use_geometric:
            a_new = geometric_mean_update(f, g, t_step)
        else:
            a_new = (1 - t_step) * f + t_step * g
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = f.copy()
            else:
                a_new = a_new / ss

        f_avg = ((k * f_avg) + a_new) / (k + 1)

        score_avg = compute_score(f_avg)
        score_cur = compute_score(a_new)

        if score_avg < best_score - 1e-12:
            best_score = score_avg
            best_f = f_avg.copy()
        if score_cur < best_score - 1e-12:
            best_score = score_cur
            best_f = a_new.copy()

        f = a_new

        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={k}, M={M:.6e}, avg={score_avg:.8f}, "
                  f"best={best_score:.8f}, t={elapsed:.0f}s")

    return best_f, best_score, n_iter


def smooth_grad_descent(a0, time_limit, T_start=5000, T_end=50, n_anneal_steps=200,
                        eta_start=0.1, mu=0.9, verbose=False):
    """Smooth gradient descent with log-sum-exp approximation and Nesterov momentum."""
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    step = 0

    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        progress = min(1.0, step / n_anneal_steps)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start / (1 + 0.01 * step)

        # Compute smooth gradient
        conv = fast_convolve(a)
        shifted = conv / T - conv.max() / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = w / w_sum

        # Gradient: 2 * cross_correlation(w, a)
        fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]

        # Nesterov momentum
        v = mu * v + grad
        a_new = a - eta * v
        a_new = proj_simplex(a_new)

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        step += 1

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD step={step}, T={T:.1f}, score={compute_score(a):.8f}, "
                  f"best={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score


def subgrad_descent(a0, time_limit, eta_start=0.05, n_top=3, verbose=False):
    """Exact subgradient descent using shifted-a trick with top-K peak averaging."""
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    a_avg = a.copy()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Get top-K peaks
        k_top = n_top
        peak_indices = np.argsort(conv)[::-1][:k_top]
        peak_vals = conv[peak_indices]
        # Weight by normalized exp of peaks
        pw = peak_vals - peak_vals[0]
        pw = np.exp(pw / max(conv.max(), 1e-10))
        pw = pw / pw.sum()

        # Gradient: weighted sum of shifted a
        grad = np.zeros(n)
        for pi, w in zip(peak_indices, pw):
            pi = int(pi)
            for j in range(n):
                idx = pi - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * w * a[idx]

        eta = eta_start / (1 + 0.005 * step)
        a_new = a - eta * grad
        a_new = proj_simplex(a_new)

        a_avg = (step * a_avg + a_new) / (step + 1)

        score = compute_score(a_new)
        score_avg = compute_score(a_avg)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        if score_avg < best_score - 1e-12:
            best_score = score_avg
            best_a = a_avg.copy()

        a = a_new
        step += 1

        if verbose and step % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step}, score={score:.8f}, avg={score_avg:.8f}, "
                  f"best={best_score:.8f}, t={elapsed:.0fs}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs_per_round=200, verbose=False):
    """Coordinate descent with golden-section search on pairs."""
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)
        peak_idx = int(conv.argmax())

        for _ in range(n_pairs_per_round):
            if time.monotonic() - t0 > time_limit:
                break

            # Mix: random, peak-focused, edge
            r = rng.random()
            if r < 0.4:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
            elif r < 0.8:
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                i = int(rng.integers(js, je + 1))
                j = int(rng.integers(js, je + 1))
            else:
                quarter = max(1, n // 8)
                i = int(rng.integers(0, quarter))
                j = int(rng.integers(n - quarter, n))

            if i == j:
                continue
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            # Golden section search
            t_lo = -a[j] + 1e-14
            t_hi = a[i] - 1e-14
            if t_hi <= t_lo:
                continue

            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            def eval_t(val):
                a_tmp = a.copy()
                a_tmp[i] -= val
                a_tmp[j] += val
                if a_tmp[i] < -1e-12 or a_tmp[j] < -1e-12:
                    return 1e10
                a_tmp = np.maximum(a_tmp, 0)
                ss = a_tmp.sum()
                if ss < 1e-15:
                    return 1e10
                a_tmp = a_tmp / ss
                return compute_score(a_tmp)

            f1 = eval_t(x1)
            f2 = eval_t(x2)
            for _ in range(15):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose:
            print(f"    CD score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_update(a, label, best_seq, best_score):
    """Evaluate and potentially update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def fast_lp_iterations(f, n_iters, verbose=False):
    """Run a fixed number of LP iterations and return the best found."""
    a = f.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    a_avg = a.copy()

    for k in range(n_iters):
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_lp_fast(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        t = 2.0 / (k + 2)
        a_new = geometric_mean_update(a, g, t)
        a_avg = ((k * a_avg) + a_new) / (k + 1)
        sc = compute_score(a_new)
        sc_avg = compute_score(a_avg)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()
        a = a_new
        if verbose and k % 50 == 0:
            print(f"      LP-iter {k}: score={sc:.8f}, avg={sc_avg:.8f}, best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining LP fictitious play, smooth gradient
    descent, subgradient methods, and coordinate descent.

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t_arr = np.linspace(0, 2 * np.pi * 2, n_init)
        parent = parent + 0.03 * np.cos(t_arr)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Fast LP Fictitious Play (main workhorse) ===
    print(f"\n=== Phase 1: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400.0, rem * 0.25)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a1, sc1, n_it = lp_fictitious_play(best_seq, t1, verbose=True, use_geometric=True)
        best_seq, best_score = check_update(a1, "lp1_geo", best_seq, best_score)

        t1b = min(150.0, rem * 0.1)
        if t1b > 20:
            a1b, sc1b, _ = lp_fictitious_play(best_seq, t1b, verbose=False, use_geometric=False)
            best_seq, best_score = check_update(a1b, "lp1_lin", best_seq, best_score)

    # === Phase 2: Smooth Gradient Descent with Annealing ===
    print(f"\n=== Phase 2: Smooth Grad Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    if t2 > 20:
        print(f"  Budget: {t2:.0f}s, T: 5000->50")
        a2, sc2 = smooth_grad_descent(best_seq, t2, T_start=5000, T_end=50,
                                      eta_start=0.08, mu=0.85, verbose=True)
        best_seq, best_score = check_update(a2, "sgd_anneal", best_seq, best_score)

    # Second SGD run with different params

    t2b = min(200.0, rem * 0.15)
    if t2b > 15:
        print(f"  SGD2 budget: {t2b:.0f}s, T: 2000->30")
        a2b, sc2b = smooth_grad_descent(best_seq, t2b, T_start=2000, T_end=30,
                                        eta_start=0.05, mu=0.9, verbose=False)
        best_seq, best_score = check_update(a2b, "sgd2", best_seq, best_score)

    # === Phase 3: Subgradient Descent ===
    print(f"\n=== Phase 3: Subgradient Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(200.0, rem * 0.15)
    if t3 > 15:
        print(f"  Budget: {t3:.0f}s")
        a3, sc3 = subgrad_descent(best_seq, t3, eta_start=0.03, n_top=5, verbose=True)
        best_seq, best_score = check_update(a3, "subgrad", best_seq, best_score)

    # === Phase 4: Coordinate Descent Polish ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")

        print(f"  Budget: {t4:.0f}s")

                                                  eta_start=0.05, mu=0.9, verbose=False)
            best_seq, best_score = check_update(a_alt2, f"n{alt_n}_sgd", best_seq, best_score)

        # CD polish
        rem3 = deadline - time.monotonic() - 40
        t_cd = min(rem3 * 0.3, 40.0)
        if t_cd > 10:
            a_alt3, sc_alt3 = coordinate_descent(best_seq, t_cd, n_pairs_per_round=200, verbose=False)
            best_seq, best_score = check_update(a_alt3, f"n{alt_n}_cd", best_seq, best_score)

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Quick SGD
        t_end = min(rem * 0.4, 60.0)
        if t_end > 10:
            a_end, sc_end = smooth_grad_descent(best_seq, t_end, T_start=1000, T_end=20,
                                                eta_start=0.03, mu=0.95, verbose=False)

        # Quick CD
        rem2 = deadline - time.monotonic()
        t_end2 = min(rem2 * 0.5, 60.0)
        if t_end2 > 10:
            a_end2, sc_end2 = coordinate_descent(best_seq, t_end2, n_pairs_per_round=400, verbose=False)
            best_seq, best_score = check_update(a_end2, "final_cd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
