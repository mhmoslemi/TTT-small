def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = (u * rho_vals) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = (u * rho_vals) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, b, eps):
    """Compute weighted subgradient using soft-max over peaks."""
    n = len(a)
    # Soft-max weights
    b_s = b / eps
    b_s = b_s - b_s.max()
    w = np.exp(b_s)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback: uniform over top 10
        top_idx = np.argpartition(b, -10)[-10:]
        w = np.zeros(len(b))
        w[top_idx] = 1.0 / 10
    else:
        w /= w_sum
    
    # g[m] = 2 * sum_j w[m+j] * a[j] for j=0..n-1
    # Use matrix-vector product with stride tricks
    # M[m,j] = w[m+j], shape (n, n)
    M = np.lib.stride_tricks.as_strided(w, shape=(n, n), strides=(8, 8))
    g = 2.0 * (M @ a)
    return g


def fista_optimize(f_init, time_limit, eps_init=0.001, eps_end=0.01, L=1.0):
    """FISTA with structured subgradients and adaptive epsilon."""
    t0 = time.monotonic()
    n = len(f_init)
    x = f_init.copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = fast_convolve(x).max()
    it = 0
    no_improve = 0
    
    # Precompute FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Adaptive epsilon: anneal from eps_init to eps_end
    # Use a schedule based on iteration count
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Current epsilon (annealed)
        progress = min(1.0, it / max(1, int(time_limit * 100)))  # rough
        eps = eps_init * (eps_end / eps_init) ** progress
        
        # Compute subgradient at y
        b_y = fast_convolve(y)
        g = compute_subgradient(y, b_y, eps)
        
        # FISTA step
        x_new = project_simplex(y - g / L)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)  # Keep y non-negative (helps with subgradient validity)
        # Renormalize y to simplex (optional, for numerical stability)
        s_y = y.sum()
        if s_y < 1e-10:
            y = x_new.copy()
        else:
            y /= s_y
        
        x = x_new
        t = t_new
        
        # Evaluate and track best
        if it % 100 == 0 or it <= 10:
            obj = fast_convolve(x).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_x = x.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 200:
                # Reset momentum
                y = x.copy()
                x_prev = x.copy()
                t = 1.0
                no_improve = 0
                # Warm restart with perturbation
                if it > 1000:
                    rng = np.random.default_rng(it)
                    x = project_simplex(x + rng.standard_normal(n) * 0.001)
                    y = x.copy()
                    t = 1.0
        
        if it % 5000 == 0:
            cur_obj = fast_convolve(x).max()
            score = 2.0 * n * cur_obj
            print(f"  FISTA it={it} obj={best_obj:.10f} cur={cur_obj:.10f} score={score:.6f} eps={eps:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_x, best_obj


def build_constraint_matrix(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx):
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    g_val = None
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-8)
        if g.value is not None:
            g_val = np.maximum(g.value, 0.0)
            if np.any(np.isnan(g_val)):
                g_val = None
    except Exception:
        pass
    if g_val is None and n <= 2000:
        try:
            prob.solve(solver=cp.ECOS, max_iters=20000)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.any(np.isnan(g_val)):
                    g_val = None
        except Exception:
            g_val = None
    return g_val


def lp_refine(f_init, time_limit, initial_k=50, max_k=300):
    """Cutting-plane LP refinement (from previous code, simplified)."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_obj = M
    
    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)
    
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        g = solve_lp_direction(f, n, M, active_idx)
        if g is None:
            stuck += 1
            if stuck > 5:
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                break
            continue
        
        stuck = 0
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = M
        for t_val in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            f_trial = np.maximum((1 - t_val) * f + t_val * h, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t_val
        
        f = np.maximum((1 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        f /= s
        cur = fast_convolve(f).max()
        M = cur
        
        if cur < best_obj - 1e-13:
            best_obj = cur
            best_f = f.copy()
        
        # Update active set
        if it % 10 == 0 and len(active_idx) < max_k:
            b_new = fast_convolve(f)
            new_top = np.argpartition(b_new, -min(max_k, len(b_new)))[-min(max_k, len(b_new)):]
            active_idx = np.unique(np.concatenate([active_idx, new_top]))
            active_idx = active_idx[np.argsort(b_new[active_idx])[-max_k:]]
        
        if it % 50 == 0:
            print(f"  LP it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def multi_restart_fista(f_init, time_limit, n_restarts=5):
    """Run FISTA from multiple perturbed starting points."""
    t0 = time.monotonic()
    n = len(f_init)
    best_x = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    time_per = time_limit / n_restarts
    rng = np.random.default_rng(12345)
    
    for r in range(n_restarts):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        
        # Create perturbation
        scale = 0.01 * (1 + 0.5 * r)
        perturbed = f_init + rng.standard_normal(n) * scale
        perturbed = np.maximum(perturbed, 0.0)
        s = perturbed.sum()
        if s > 1e-10:
            perturbed /= s
        else:
            perturbed = f_init.copy()
        
        t_rem = min(time_per, rem - 5)
        if t_rem < 5:
            break
        
        f_r, obj_r = fista_optimize(perturbed, time_limit=t_rem, 
                                      eps_init=0.0005 + 0.001 * r,
                                      eps_end=0.005 + 0.005 * r)
        if obj_r < best_obj:
            best_obj = obj_r
            best_x = f_r.copy()
            print(f"  Restart {r}: NEW BEST obj={obj_r:.10f} score={2.0*n*obj_r:.6f}")
    
    return best_x, best_obj


def spectral_flatten(f_init, time_limit, n_iters=5000):
    """Frequency domain spectral flattening."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        
        # Try to flatten R: reduce peaks, boost valleys
        peak = R.max()
        target = peak * 0.85  # Reduce peak by 15%
        R_mod = np.where(R > target, target, R)
        
        # Preserve total energy
        if R_mod.sum() > 0:
            R_mod *= R.sum() / R_mod.sum()
        
        # Get new F: F_new = sqrt(R_mod) * phase(F)
        R_mod_F = rfft(R_mod, fft_size)
        # R_mod should be PSD-like, so its FFT should be real non-negative (approximately)
        real_part = np.maximum(R_mod_F.real, 0.0)
        new_amp = np.sqrt(real_part)
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_amp * phase
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            break
        f = f_new
        
        if it % 1000 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
    
    return best_f, best_obj
