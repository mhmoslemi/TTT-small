"""
Spectral flattening + LP-FP + L-BFGS for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses error reduction (Fienup) algorithm to flatten the power spectrum,

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def spectral_flattening(a0, time_limit, tau=0.5, verbose=True):
    """
    Error reduction algorithm to flatten the power spectrum.
    The goal: make |FFT(a)| as constant as possible while keeping a >= 0.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Forward: go to frequency domain
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.where(mag > 1e-30, mag, 1.0)
        
        # Target magnitude: geometric mean or adjusted mean
        target_mag = np.sqrt(np.mean(mag**2))
        
        # Relaxed update: don't fully flatten, use tau
        new_mag = (mag ** (1-tau)) * (target_mag ** tau)
        
        # Back to time domain
        A_new = phase * new_mag
        a_new = irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={iters}, score={best_score:.10f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):

    a = np.maximum(a0.astype(np.float64), 0)

    best_a = a.copy()

    iters = 0
    prev_g = None
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        if stall_count > 500:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.0005
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
        
        if verbose and iters % 200 == 0:

            print(f"    {label} iter={iters}, best={best_score:.10f}, step={step:.4f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):
    n = len(a0)
    
    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s
    
    def z_from_a(a):
        return np.log(np.maximum(a, 1e-30))
    
    def smoothed_objective(z, T, n):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def smoothed_gradient(z, T, n):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z
    
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def exponentiated_gradient(a0, time_limit, T=100, eta0=1.0, verbose=True):
    """
    Mirror descent (exponentiated gradient) for the log-sum-exp smoothed objective.
    Variable: z (unconstrained), a = softmax(z)
    Update: z ← z - eta * grad_z, then project (no projection needed for softmax)
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = eta0
    iters = 0
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        # Log-sum-exp weights
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        
        # Gradient of LSE objective w.r.t. a
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        
        # Convert to z-gradient: grad_z = a * (grad_a - sum(a*grad_a))
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        
        # Adaptive step size
        # Scale eta so the update has reasonable magnitude
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break
        eta = eta0 / (1.0 + 0.01 * iters)
        
        # Update in z-space
        # z_new = z - eta * grad_z, but we work with a directly
        # a_new ∝ a * exp(-eta * grad_z)
        log_update = -eta * grad_z / grad_norm
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        ss = a_new.sum()
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        else:
            # Try smaller step
            a_trial = a * np.exp(log_update * 0.1)
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial = a_trial / a_trial.sum()
            score_trial = compute_score(a_trial)
            if score_trial < best_score - 1e-14:
                best_score = score_trial
                best_a = a_trial.copy()
                a = a_trial
            else:
                a = a_new
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EG iter={iters}, best={best_score:.10f}, eta={eta:.6f}, time={elapsed:.0f}s")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines spectral flattening (Fienup error reduction), FFT-accelerated subgradient
descent, LP fictitious play, and L-BFGS log-sum-exp polishing.
Explores multiple sequence lengths to avoid local minima.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def spectral_flattening(a0, time_limit, verbose=True, label="Spectral"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0
    taus = [0.3, 0.5, 0.7, 0.4, 0.6, 0.8]
    tau_idx = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        tau = taus[tau_idx % len(taus)]

        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.where(mag > 1e-30, mag, 1.0)

        log_mag = np.log(np.maximum(mag, 1e-30))
        target_log = np.mean(log_mag)

        new_log_mag = (1 - tau) * log_mag + tau * target_log
        new_mag = np.exp(new_log_mag)

        A_new = phase * new_mag
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            stuck += 1
            if stuck > 50:
                break
            continue

        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 100:
                tau_idx += 1
                rng = np.random.default_rng(iters + 7)
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                a_new = a
                score = compute_score(a_new)
                stuck = 0

        a = a_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters}, tau={tau:.1f}, best={best_score:.10f}, time={elapsed:.0f}s")

    return best_a, best_score


def fft_subgradient_descent(a0, time_limit, verbose=True, label="SubGrad"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta0 = 0.3
    iters = 0
    stuck = 0
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        tol = max(1e-12, M * 1e-8)
        peak_mask = conv >= M - tol
        peaks = np.where(peak_mask)[0]

        subgrad = np.zeros(n)
        for k in peaks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            subgrad[idx] += a[k - idx]
        subgrad /= len(peaks)

        eta = eta0 / np.sqrt(1.0 + 0.005 * iters)
        a_new = a - eta * subgrad
        a_new = project_simplex(a_new)

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:
                eta0 = max(0.01, eta0 * 0.5)
                rng = np.random.default_rng(iters)
                a = best_a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                a_new = a
                stuck = 0
                continue

        a = a_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters}, eta0={eta0:.4f}, best={best_score:.10f}, time={elapsed:.0f}s")

    return best_a, best_score


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 300:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.0005
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters}, best={best_score:.10f}, step={step:.4f}, time={elapsed:.0f}s")

    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):
    n = len(a0)

    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    def z_from_a(a):
        return np.log(np.maximum(a, 1e-30))

    def smoothed_objective(z, T, n):
        a_local = softmax_from_z(z)
        conv = fast_convolve(a_local)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def smoothed_gradient(z, T, n):
        a_local = softmax_from_z(z)
        conv = fast_convolve(a_local)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_local, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a_local))
        grad_z = a_local * (grad_a - S)
        return (2.0 * n / T) * grad_z

    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def exponentiated_gradient(a0, time_limit, T=50, verbose=True, label="ExpGrad"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    stuck = 0
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()

        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        eta = 1.0 / (1.0 + 0.01 * iters)
        log_update = -eta * grad_z / grad_norm
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        ss = a_new.sum()
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 300:
                rng = np.random.default_rng(iters)
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                a_new = a
                score = compute_score(a_new)
                stuck = 0

        a = a_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters}, best={best_score:.10f}, time={elapsed:.0f}s")
