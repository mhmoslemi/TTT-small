"""
Optimized sequence search using softmax gradient descent with FFT-based
convolution, LP polishing, and coordinate descent.
Key innovations: efficient gradient computation, temperature annealing,
Adam optimizer, and multi-phase optimization strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.sparse import csr_matrix
import time
from copy import deepcopy


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_gradient(a, temp, n_top=300):
    """Compute gradient of soft-max of convolution using peak-based approach."""
    n = len(a)
    b = fast_convolve(a)
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), b, np.zeros(len(b))
    w /= w_sum

    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        w = np.zeros(len(b))
        w[top_idx] = w_sel

    grad = np.zeros(n)
    nonzero_k = np.where(w > 1e-14)[0]
    for k in nonzero_k:
        wk = w[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        a_slice = a[lo:hi + 1][::-1]
        grad[lo:hi + 1] += wk * a_slice
    return grad, b, w


def softmax_gradient_descent(f_init, time_limit, temp_start=None, temp_end=None,
                             n_top=300, lr_init=0.01, beta_adam=0.9,
                             label="sgd", seed=42):
    """
    Softmax gradient descent with Adam optimizer and temperature annealing.
    """
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return f_init.copy(), fast_convolve(f_init).max()
    a /= s

    b = fast_convolve(a)
    b_max_init = b.max()

    if temp_start is None:
        temp_start = b_max_init * 0.5
    if temp_end is None:
        temp_end = b_max_init * 0.01

    best_a = a.copy()
    best_b = b_max_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    total_iters_estimate = 8000
    no_improve = 0
    best_no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        t += 1
        frac = min(1.0, t / total_iters_estimate)
        frac = frac ** 0.7
        temp = temp_start * (temp_end / temp_start) ** frac

        grad, b, w = compute_softmax_gradient(a, temp, n_top)

        # Skip zero gradient (shouldn't happen but safety)
        if np.all(grad == 0):
            break

        # Adam update
        m = beta_adam * m + (1 - beta_adam) * grad
        v = beta_adam * v + (1 - beta_adam) * grad ** 2
        m_hat = m / (1 - beta_adam ** t)
        v_hat = v / (1 - beta_adam ** t)
        a_new = a - lr_init * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new, 1.0)

        b_new = fast_convolve(a_new)
        b_max = b_new.max()

        if b_max < best_b - 1e-15:
            best_b = b_max
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 50:
            # Perturbation and restart
            a = best_a + rng.standard_normal(n) * 0.002 * best_a
            a = np.maximum(a, 0.0)
            a /= a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            lr_init *= 0.8
            no_improve = 0

        if t % 2000 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={t} temp={temp:.2e} best_b={best_b:.10f} "
                  f"score={score:.6f} lr={lr_init:.6f} elapsed={time.monotonic()-t0:.0f}s")

    return best_a, best_b


def solve_lp_direction_pdlp(f, n, peak_indices, M, eps=1e-8):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)(k) <= M for k in peak_indices.
    Uses sparse matrix and PDLP solver.
    """
    K = len(peak_indices)
    if K == 0:
        return None

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            val = f[j]
            if val > 1e-15:
                rows.append(row_i)
                cols.append(j)
                vals.append(val)

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=200000, eps=eps)
            break
        except Exception:
            continue

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def lp_optimize(f_init, time_limit, label="lp", max_peaks=400, peak_frac=0.95,
                patience=20, eps=1e-8):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return f_init.copy(), fast_convolve(f_init).max()
    a /= s

    best_a = a.copy()
    best_b = fast_convolve(a).max()
    it = 0
    stuck = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()

        # Peak selection
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(b, -min(len(b), 20))[-min(len(b), 20):]

        g = solve_lp_direction_pdlp(a, n, peaks, M, eps=eps)

        if g is None:
            stuck += 1
            if stuck > patience:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > patience:
                break
            # Try with more peaks
            n_peaks_try = min(len(b), max_peaks * 2)
            more_peaks = np.argpartition(b, -n_peaks_try)[-n_peaks_try:]
            g2 = solve_lp_direction_pdlp(a, n, more_peaks, M, eps=eps)
            if g2 is not None and g2.sum() > 1.0 + 1e-5:
                g = g2
                S_g = g2.sum()
            else:
                stuck += 1
                if stuck > patience:
                    break
                continue

        h = g / S_g

        # Line search
        best_t = 0.0
        best_t_obj = M
        for t_trial in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            a_trial = (1.0 - t_trial) * a + t_trial * h
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial /= s_trial
            obj = fast_convolve(a_trial).max()
            if obj < best_t_obj - 1e-14:
                best_t_obj = obj
                best_t = t_trial

        if best_t <= 0:
            stuck += 1
            if stuck > patience:
                break
            continue

        a = (1.0 - best_t) * a + best_t * h
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s

        cur_obj = fast_convolve(a).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Gradually broaden peak selection
        if it % 20 == 0:
            peak_frac = max(0.80, peak_frac - 0.005)

        # Perturbation if stuck
        if stuck > patience // 2 and stuck % 3 == 0:
            a = a + rng.standard_normal(n) * 0.0003 * a
            a = np.maximum(a, 0.0)
            a /= a.sum()

        if it % 10 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} best_b={best_b:.10f} score={score:.6f} "
                  f"peaks={len(peaks)} elapsed={time.monotonic()-t0:.0f}s")

    return best_a, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-10:
        a /= s
    best_b = fast_convolve(a).max()
    best_a = a.copy()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0]

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-15:
                continue
            best_j_val = a[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = a[j] * mult
                if new_val > 1000.0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial /= s_trial
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != a[j]:
                a[j] = best_j_val
                s = a.sum()
                if s > 1e-10:
                    a /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    if time.monotonic() - t0 < time_limit:
        print(f"    {label} converged in {passes} passes, "
              f"final_b={best_b:.10f}, score={2*n*best_b:.6f}")
    return best_a, best_b

def structured_init(n, seed=0):
    """Create a structured initialization with anti-concentrated properties."""
    rng = np.random.default_rng(seed)
    # Start with a "bowl" shape (mass at edges, less in middle)
    x = np.linspace(0, 1, n)
    base = 1.0 + 2.0 * ((x - 0.5) ** 2) / (0.25)  # bowl
    # Add random perturbation
    base += rng.standard_normal(n) * 0.3 * base.mean()
    base = np.maximum(base, 0.0)
    base /= base.sum()
    return base


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s_new = f_new.sum()
    if s_new > 1e-10:
        f_new /= s_new
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Individual values: min={best_seq.min():.8f}, max={best_seq.max():.8f}")
    print(f"Sum={best_seq.sum():.6f}")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
