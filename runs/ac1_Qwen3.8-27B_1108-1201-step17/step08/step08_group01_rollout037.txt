    lr = 1.0

        k_star = int(np.argmax(conv))
        # Also consider top-k peaks for smoothing
        top_k = min(10, 2*n-1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        # Weight by softmax of peak values
        temp = 0.001 * M
        w = np.exp((conv[peaks] - M) / temp)
        w = w / w.sum()
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            # sg[i] += w[idx] * 2 * a[k-i] if 0 <= k-i <= n-1

        # Mirror descent step
        lr = 1.0 / np.sqrt(t + 1)  # or 1/(t+1)
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        conv = conv_new
        M = M_new
    
    return best_a, best_M

**Key improvement: Use FFT-based convolution in the FW steps**

Instead of computing the full convolution at each FW step (O(n log n)), we can use the fact that the FW step only needs the gradient at the current point, which can be computed in O(n) using a shifted version of the fixed sequence.

Let me rewrite the FW step more efficiently:

For the problem min_b max_k (a * b)[k] s.t. sum(b)=1, b>=0:
- At each step, we need k* = argmax (a*b)[k]
- Computing (a*b) is O(n log n) via FFT
- The FW oracle is j* = argmin_i a[k*-i] (over valid i), which is O(n)
- The update b = (1-t)*b + t*e_{j*} is O(n)

So each FW step is O(n log n) for the FFT + O(n) for the rest. With n=1024 and n_steps=1000 per round, that's about 1000 * 1024 * 10 ≈ 10 million operations per round. In 600 seconds, we can do about 60 rounds. That's manageable.

But we can do better: instead of recomputing the full convolution at each FW step, we can update it incrementally. When b changes by t*(e_j - e_i) (moving mass from i to j), the convolution changes by t*(a shifted by j - a shifted by i), which is O(n) to update.

So the FW step becomes O(n) total! This is much faster.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def mirror_descent_multi_peak(a0, time_limit, lr_init=0.05, top_k=20, verbose=True):
    """
    Mirror descent (multiplicative weight update) using multi-peak subgradient.
    Key improvement: uses top-k peaks with softmax weighting to break symmetry.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break

        # Compute multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        # Get top-k peak indices
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient: sg[i] = sum_k w[k] * 2 * a[k-i] (if valid)
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # sg[i] += w[idx] * 2 * a[k-i] for i in [i_start, i_end]
            # a[k-i] for i in [i_start, i_end] = a[k-i_start : k-i_end+1][::-1]
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        # Also track running average (explained by the convergence theory)
        avg_a = (it * avg_a + a_new) / (it + 1)

        a = a_new
        conv = fast_convolve(a)

        # Periodically check the running average
        if it % 100 == 0:
            avg_score = 2.0 * n * fast_convolve(avg_a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score

    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute prefix/suffix maxima for fast "unaffected max" queries
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # The "affected" range: conv[k] changes when a[i] changes
            # conv[k] += 2*delta*a[k-i] (approximately, for k != 2i)
            a_shifted = a[k_start - i:k_end - i + 1]  # a[k-i] for k in [k_start, k_end]

            best_delta = 0.0
            best_new_M = M

            # Adaptive step sizes based on current value
            base_step = a[i]
            factors = [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5]

            for factor in factors:
                delta = base_step * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                # Compute new affected values
                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                # Fix the quadratic term at k = 2i if in range
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                # Apply update
                a[i] = max(a[i] + best_delta, 0.0)

                # Update convolution incrementally
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score


def fine_cd_polish(a0, time_limit, verbose=True):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_M = M

            # Very fine steps
            for factor in [0.99, 0.98, 0.97, 0.985, 1.015, 1.03, 1.02, 1.01]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1].copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

    return best_a, best_score


def lp_refine(a0, time_limit, solver=None):
    """LP-based refinement: solve the LP relaxation for best response."""
    import cvxpy as cp

    if solver is None:
        solver = _find_lp_solver()

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    # Build the convolution matrix T such that T @ b = conv(a, b)
    # T has shape (2n-1, n)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b

        # Evaluate
        score_b = 2.0 * n * fast_convolve(b_val).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_val.copy()

        # Damped update
        alpha = 3.0 / (it + 1.0)
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

        # Rebuild T for next iteration
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a

        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def _find_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _cached_solver
    if '_cached_solver' in globals() and _cached_solver is not None:
        return _cached_solver
    _cached_solver = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = solver_name
                return solver_name
        except Exception:
            continue
    _cached_solver = cp.ECOS
    return cp.ECOS


def periodic_init(n, rng):
    """Initialize with a periodic pattern (alternating high/low)."""
    period = int(rng.integers(2, 8))
    a = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def cosine_window_init(n, rng):
    """Initialize with a cosine raised-to-power window."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(0.5, 3.0)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_hump_init(n, rng):
    """Initialize with two peaks (non-palindromic)."""
    t = np.arange(n, dtype=np.float64)
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width = n * rng.uniform(0.1, 0.25)
    a = np.exp(-0.5 * ((t - center1) / width) ** 2) + np.exp(-0.5 * ((t - center2) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. FW alternating minimization (fast global optimization, O(n) per step)
    2. Multi-peak mirror descent (smooth refinement)
    3. Coordinate descent (local polishing)
    4. LP refinement (final tightening)
    5. Fine coordinate descent (last-mile improvements)
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))

    n_best = len(best_seq)

    # === Phase 1: FW Global Optimization ===
    print(f"\n=== Phase 1: FW Global Optimization (best={best_score:.6f}) ===")
    t1_start = time.monotonic()

    # Try the best sequence and a few diverse initializations
    initializations = [best_seq.copy()]

    # Add diverse inits at the same length
    initializations.append(cosine_window_init(n_best, rng))
    initializations.append(two_hump_init(n_best, rng))
    initializations.append(periodic_init(n_best, rng))

    # Add a noisy version of parent
    noisy = best_seq + rng.standard_normal(n_best) * 0.05
    noisy = np.maximum(noisy, 0)
    noisy = noisy / noisy.sum()
    initializations.append(noisy)

    best_fw = best_seq.copy()
    best_fw_score = best_score

    for idx, init in enumerate(initializations):
        remaining = deadline - time.monotonic() - 400
        if remaining < 60:
            break
        t_alloc = min(remaining * 0.25, 200.0)
        if idx == 0:
            t_alloc = min(remaining * 0.35, 300.0)  # Give more time to best init

        print(f"  FW init {idx}, budget={t_alloc:.0f}s")
        a_fw, s_fw = fw_global_optimize(init, t_alloc, n_fw_per_round=150, verbose=True)
        if s_fw < best_fw_score:
            best_fw_score = s_fw
            best_fw = a_fw.copy()
        best_seq, best_score = check_and_update(a_fw, best_seq, best_score, f"fw_{idx}")

    # === Phase 2: Mirror Descent Refinement ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 300
    if remaining > 60:
        t_md = min(remaining * 0.35, 300.0)
        print(f"  MD budget={t_md:.0f}s")

        # Run MD on best sequence
        a_md, s_md = mirror_descent_multi_peak(best_seq, t_md, lr_init=0.03, top_k=30, verbose=True)
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "mirror_descent")

        # Also run MD on the FW result if different
        if best_fw_score < best_score + 0.001:
            a_md2, s_md2 = mirror_descent_multi_peak(best_fw, min(t_md * 0.5, 100.0),
                                                      lr_init=0.03, top_k=30, verbose=True)
            best_seq, best_score = check_and_update(a_md2, best_seq, best_score, "md_fw")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 200
    if remaining > 60:
        t_cd = min(remaining * 0.4, 300.0)
        print(f"  CD budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent_fast(best_seq, t_cd, n_trials=10, verbose=True, label="main")
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "coord_descent")

        # Run CD on MD result too
        remaining = deadline - time.monotonic() - 150
        if remaining > 60 and a_md is not None:
            t_cd2 = min(remaining * 0.4, 150.0)
            a_cd2, s_cd2 = coord_descent_fast(a_md, t_cd2, n_trials=10, verbose=True, label="md_cd")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "cd_md_result")

    # === Phase 4: LP Refinement ===
    print(f"\n=== Phase 4: LP Refinement (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 120
    if remaining > 60:
        try:
            solver = _find_lp_solver()
            print(f"  LP solver: {solver}")
            t_lp = min(remaining * 0.5, 120.0)
            a_lp, s_lp = lp_refine(best_seq, t_lp, solver=solver)
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, "lp_refine")
        except Exception as e:
            print(f"  LP failed: {e}")

    # === Phase 5: Fine Polishing ===
    print(f"\n=== Phase 5: Fine Polishing (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 40
    if remaining > 30:
        t_fine = min(remaining * 0.5, 120.0)
        print(f"  FineCD budget={t_fine:.0f}s")
        a_fine, s_fine = fine_cd_polish(best_seq, t_fine, verbose=True)
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

    # === Phase 6: Final CD pass ===
    print(f"\n=== Phase 6: Final CD (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 10
    if remaining > 20:
        t_final = min(remaining * 0.7, 60.0)
        a_fin, s_fin = coord_descent_fast(best_seq, t_final, n_trials=8, verbose=False, label="final")
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
