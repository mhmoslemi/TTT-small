def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_direction(f, g):
    """Compute cross-convolution f*g via FFT."""
    size = 2 * len(f) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    """Build matrix A where A[i,:] @ g = (f*g)[indices[i]]."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A


def solve_minimax_lp(f, n, active_idx, solver="SCS", verbose=False):
    """
    Solve: min M' s.t. (f*g)[k] <= M' for k in active_idx, g >= 0, sum(g) = 1.
    Returns g and M'.
    """
    K = len(active_idx)
    if K == 0:
        return None, None
    
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    M_prime = cp.Variable()
    
    constraints = [g >= 0, cp.sum(g) == 1, A @ g <= M_prime]
    prob = cp.Problem(cp.Minimize(M_prime), constraints)
    
    g_val = None
    try:
        if solver == "GUROBI":
            prob.solve(solver=cp.GUROBI)
        elif solver == "MOSEK":
            prob.solve(solver=cp.MOSEK)
        elif solver == "CLARABEL":
            prob.solve(solver=cp.CLARABEL)
        else:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            return None, None
    
    if g.value is None:
        return None, None
    
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, None
    
    s = g_val.sum()
    if s < 1e-10:
        return None, None
    g_val /= s
    
    return g_val, M_prime.value


def solve_maxsum_lp(f, n, M, active_idx, solver="SCS"):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_idx.
    Returns g.
    """
    K = len(active_idx)
    if K == 0:
        return None
    
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
    except Exception:

def spectral_shaping(f, n_iters=100, alpha=0.1):

        F = rfft(f_work, fft_size)

"""
Optimization using minimax LP, spectral shaping, FFT-based subgradient,
and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    size = 2 * len(f) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A


def solve_lp_minimax(f, n, active_idx):
    """min M' s.t. (f*g)[k] <= M' for k in active_idx, g >= 0, sum(g) = 1"""
    K = len(active_idx)
    if K == 0:
        return None, None
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    Mp = cp.Variable()
    constraints = [g >= 0, cp.sum(g) == 1, A @ g <= Mp]
    prob = cp.Problem(cp.Minimize(Mp), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=1000000, eps=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000)
        except Exception:
            return None, None
    if g.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, None
    s = g_val.sum()
    if s < 1e-10:
        return None, None
    g_val /= s
    return g_val, float(Mp.value) if Mp.value is not None else None


def solve_lp_maxsum(f, n, M, active_idx):
    """max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_idx"""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=1000000, eps=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    s = g_val.sum()
    if s < 1e-10:
        return None
    return g_val

    for t in t_scan:
        ft = (1.0 - t) * f + t * h
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-12:
            objs.append(np.inf)
            continue
        ft /= s
        o = fast_convolve(ft).max()
        objs.append(o)
        if o < best_obj:
            best_obj = o
            best_t = t
    
    # Parabolic refinement
    best_idx = int(np.argmin(objs))
    if 0 < best_idx < len(t_scan) - 1:
        t0 = t_scan[best_idx]
        f0 = objs[best_idx]
        if np.isfinite(f0):
            # Check neighbors
            t_m = t_scan[best_idx - 1]
            t_p = t_scan[best_idx + 1]
            fm = objs[best_idx - 1]
            fp = objs[best_idx + 1]
            if np.isfinite(fm) and np.isfinite(fp):
                denom = fm - 2*f0 + fp
                if abs(denom) > 1e-20:
                    tp_ = t0 + 0.5*(t_m - t_p)*(fm - fp)/denom
                    if 0.001 < tp_ < 1.2:
                        ft = np.maximum((1-tp_)*f + tp_*h, 0.0)
                        s = ft.sum()
                        if s > 1e-12:
                            ft /= s
                            o = fast_convolve(ft).max()
                            if o < best_obj:
                                best_obj = o
                                best_t = tp_
    
    # Golden section around best_t
    if best_t > 0.005:
        lo = max(0.0, best_t - 0.04)
        hi = min(1.3, best_t + 0.06)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(12):
            m1 = hi - gr*(hi - lo)
            m2 = lo + gr*(hi - lo)
            f1 = np.maximum((1-m1)*f + m1*h, 0.0)
            s1 = f1.sum()
            o1 = fast_convolve(f1/s1).max() if s1 > 1e-12 else np.inf
            f2 = np.maximum((1-m2)*f + m2*h, 0.0)
            s2 = f2.sum()
            o2 = fast_convolve(f2/s2).max() if s2 > 1e-12 else np.inf
            if o1 < o2:
                hi = m2
            else:
                lo = m1
        tm = (lo + hi) / 2.0
        fm = np.maximum((1-tm)*f + tm*h, 0.0)
        sm = fm.sum()
        if sm > 1e-12:
            fm /= sm
            om = fast_convolve(fm).max()
            if om < best_obj:
                best_obj = om
                best_t = tm
    
    return best_t, best_obj


def fft_subgradient(f, temp=0.01, n_top=None):
    """Compute subgradient of max_k (f*f)_k using FFT."""
    n = len(f)
    b = fast_convolve(f)
    
    # Softmax weights
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.ones(len(top_idx)) / len(top_idx)
        w = np.zeros(len(b))
        w[top_idx] = w_sel
    
    # Subgradient: grad_j = 2 * sum_k w_k * a_{k-j}
    # This is 2 * cross_correlate(w, a) = 2 * convolve(w_flipped, a)
    w_flip = w[::-1]  # Flip for cross-correlation
    # Actually, cross_correlate(w, a)_j = sum_k w_k * a_{k-j}
    # = convolve(w_flipped, a)_j where w_flipped[k] = w[-k]
    # But our w is indexed 0..2n-2, and a is indexed 0..n-1
    # Let me just compute it directly using the convolution structure
    
    # grad_j = 2 * sum_k w_k * f[k-j] (if 0 <= k-j < n)
    # = 2 * sum_i w_{j+i} * f[i] (substituting k=j+i)
    # This is 2 * (w_shifted * f) where w_shifted[j+i] = w[j+i]
    
    # More efficiently: use FFT
    # grad = 2 * cross_correlate(w, f)
    # cross_correlate(w, f) = ifft(conj(fft(w)) * fft(f)) padded appropriately
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    F = rfft(f, fft_size)
    # Cross-correlation: ifft(conj(W) * F)
    cc = irfft(np.conj(W) * F, fft_size)[:n]
    
    return 2.0 * cc


def subgradient_optimize(f_init, time_limit, n_top=50):
    """Subgradient descent with Nesterov acceleration."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    y = f.copy()  # Nesterov momentum
    v = np.zeros(n)
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    temp = 0.01
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = 2.0 / (it + 1.0)  # Nesterov step
        
        # Gradient at y
        g = fft_subgradient(y, temp=temp, n_top=n_top)
        
        # Nesterov update
        t_k = 2.0 / (it + 3.0)
        y_new = (1.0 - t_k) * f + t_k * y  # This is getting complicated
        # Let me use a simpler approach: momentum
        v = 0.9 * v - 0.002 * g
        f = project_simplex(f + v)
        
        if it % 50 == 0:
            M = fast_convolve(f).max()
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    temp *= 1.2
                    no_improve = 0
                if no_improve > 50:
                    f = project_simplex(best_f + np.random.standard_normal(n) * 0.005)
                    v = np.zeros(n)
                    no_improve = 0
        
        if it % 500 == 0:
            print(f"    sg it={it} best_M={best_M:.10f} score={2.0*n*best_M:.6f}")
    
    return best_f, best_M
