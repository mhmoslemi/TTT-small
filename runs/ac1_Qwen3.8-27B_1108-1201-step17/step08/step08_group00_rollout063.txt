import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def get_lp_solver():
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def lbgfs_refine(a0, time_limit, T_values, maxiter_per_stage):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    for T_val in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        
        def jac(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)

            S = np.dot(grad_a, a)

        except Exception:
            pass
    
    return softmax_from_z(z)

    a = a / s

        y = np.maximum(y, 0)
        sy = y.sum()

        y = y / sy

def lbgfs_refine_fast(a0, time_limit, T_values, maxiter_per_stage):
    """L-BFGS with FFT-accelerated gradient."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    fft_size = 1 << (2*n - 2).bit_length()
    
    for T_val in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            a = softmax_from_z(z)
            A = fft(a, fft_size)
            c = np.real(ifft(A * A, fft_size))[:2*n-1]
            max_c = c.max()
            log_sum = max_c + np.log(np.exp(T_val * (c - max_c)).sum())
            return (2.0 * n / T_val) * log_sum
        
        def jac(z):
            a = softmax_from_z(z)
            A = fft(a, fft_size)
            c = np.real(ifft(A * A, fft_size))[:2*n-1]
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            
            # Cross-correlation: grad_a = 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
            W = fft(w, fft_size)
            A_full = fft(a, fft_size)
            grad_a = 2.0 * np.real(ifft(W * np.conj(A_full), fft_size))[:n]
            
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter_per_stage * 200}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_solver_cache = None


def get_lp_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = solver_name
                return solver_name
        except Exception:
            continue
    _solver_cache = cp.ECOS
    return _solver_cache


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-neg, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.PDLP, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

            continue

        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)

        if sy < 1e-15:

        y = y / sy

        if score_y < best_score:

            elapsed = time.monotonic() - t0

            noise = 0.005 * np.random.default_rng(it).random(n)
            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()

    return best_a, best_score


def lbgfs_refine_fft(a0, time_limit, T_stages, iters_per_stage):
    """L-BFGS with FFT-accelerated cross-correlation gradient (O(n log n))."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a_init = a_init / a_init.sum()
    z = np.log(a_init)
    t0 = time.monotonic()

    # FFT size for cross-correlation: need >= (2n-1) + n - 1 = 3n-2
    fft_size_corr = 1 << (3 * n - 2 - 1).bit_length()
    # FFT size for convolution: need >= 2n-1
    fft_size_conv = 1 << (2 * n - 2 - 1).bit_length()

    for T_val in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            a = softmax_from_z(z_val)
            A = rfft(a, fft_size_conv)
            c = irfft(A * A, fft_size_conv)[:2 * n - 1]
            max_c = c.max()
            exp_Tc = np.exp(T_val * (c - max_c))
            log_sum = max_c + np.log(exp_Tc.sum())
            return (2.0 * n / T_val) * log_sum

        def jac(z_val):
            a = softmax_from_z(z_val)
            A = rfft(a, fft_size_conv)
            c = irfft(A * A, fft_size_conv)[:2 * n - 1]
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()

            # Cross-correlation: grad_a[m] = 2 * sum_l w[m+l] * a[l]
            # = 2 * IFFT(FFT(w) * conj(FFT(a)))[:n]
            W = rfft(w, fft_size_corr)
            A_full = rfft(a, fft_size_corr)
            grad_a = 2.0 * irfft(W * np.conj(A_full), fft_size_corr)[:n]

            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300,
                         'maxls': 50}
            )
            z = res.x
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        a_cur = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_cur).max()
        print(f"    L-BFGS T={T_val:.3f} ({elapsed:.0f}s): score={sc:.6f}")

    return softmax_from_z(z)


def peak_shave_local(a0, time_limit, n_steps=20000):
    """Targeted peak shaving on top convolution contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find all near-max peaks
        threshold = max_b * 0.95
        peaks = np.where(b >= threshold)[0]
        # Exclude boundary lags (0 and 2n-2) as they are less impactful
        interior_peaks = peaks[(peaks > 0) & (peaks < 2 * n - 2)]
        if len(interior_peaks) == 0:
            interior_peaks = peaks

        # Find top contributing coordinates across all peaks
        contrib = np.zeros(n)
        for k in interior_peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top = np.argsort(contribs)[::-1][:8]
            contrib[idx[top]] += 1

        top_coords = np.argsort(contrib)[::-1][:15]

        improved = False
        for i in top_coords:
            if a[i] < 1e-15:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.78]:
                a_trial = a.copy()
                diff = a[i] * (1 - factor)
                a_trial[i] -= diff
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] += diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0

    return best_a, best_score

        a = np.exp(-x ** 2 * 3)

        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    else:
        a = 1.0 / (1.0 + np.exp(-5 * (np.linspace(0, 1, n) - 0.5)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def symmetrize(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
