"""
Optimization of non-negative step functions to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with PDLPS solver, spectral equalization,
simulated annealing, and coordinate descent for final polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction_pdlp(f, n, peak_indices, M, eps=1e-9):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Uses PDLPS solver for speed.
    """
    K = len(peak_indices)
    # Build constraint matrix A where A[row, :] gives the coefficients of (f*g)[row_peak]
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solved = False
    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            if solver == cp.PDLP:
                prob.solve(solver=solver, eps_abs=eps, eps_rel=eps, max_iter=200000)
            else:
                prob.solve(solver=solver, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue
    
    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def line_search_best(f, h, M_current, n_trials=20):
    """
    Find best t in (0, 1) minimizing max(conv((1-t)f + t*h)).
    Uses parabolic interpolation on a coarse grid.
    """
    best_t = 0.0
    best_obj = M_current
    
    # Coarse grid
    t_grid = np.linspace(0.02, 0.98, n_trials)
    objs = np.zeros(n_trials)
    for i, t in enumerate(t_grid):
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            objs[i] = np.inf
            continue
        f_trial = f_trial / s
        objs[i] = fast_convolve(f_trial).max()
    
    # Find best on grid
    min_idx = np.argmin(objs)
    if objs[min_idx] >= best_obj:
        return 0.0, best_obj
    
    best_t = t_grid[min_idx]
    best_obj = objs[min_idx]
    
    # Parabolic refinement
    if 0 < min_idx < n_trials - 1:
        t0, t1, t2 = t_grid[min_idx-1], t_grid[min_idx], t_grid[min_idx+1]
        f0, f1, f2 = objs[min_idx-1], objs[min_idx], objs[min_idx+1]
        denom = 2.0 * (f0 - 2*f1 + f2)
        if abs(denom) > 1e-15:
            t_star = t1 - 0.5 * (f0 - f2) * (t2 - t1) / (f0 - 2*f1 + f2)
            # Wait, parabolic interpolation formula
            # t* = t1 - 0.5 * ((t1-t0)^2*(f1-f2) - (t1-t2)^2*(f1-f0)) / ((t1-t0)*(f1-f2) - (t1-t2)*(f1-f0))
            # Simplified: t* = 0.5 * (t0*t1*(f1-f2) + t1*t2*(f0-f1) + t2*t0*(f2-f0)) / ((t0-t1)*(f1-f2) + (t1-t2)*(f0-f1) - (t2-t0)*(f1-f0))
            # Let me use the standard formula
            num = (t2-t0)**2 * (f1-f2) - (t2-t1)**2 * (f0-f1)
            den = (t2-t0) * (f1-f2) - (t2-t1) * (f0-f1)
            if abs(den) > 1e-15:
                t_star = t1 + num / den
                if 0 < t_star < 1:
                    f_trial = (1.0 - t_star) * f + t_star * h
                    f_trial = np.maximum(f_trial, 0.0)
                    s = f_trial.sum()
                    if s > 1e-10:
                        f_trial = f_trial / s
                        obj = fast_convolve(f_trial).max()
                        if obj < best_obj:
                            best_obj = obj
                            best_t = t_star
    
    return best_t, best_obj


def run_lp_optimize(f_init, time_limit, label="lp", initial_peak_frac=0.95,
                    final_peak_frac=0.75, max_peaks=2000, restart_patience=30,
                    lp_eps=1e-9, n_top_line=25):
    """
    LP-based optimization with adaptive peak selection and parabolic line search.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = initial_peak_frac
    rng = np.random.default_rng(20240)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Peak selection: adaptive threshold
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        
        # Cap the number of peaks for LP feasibility
        if len(peaks) > max_peaks:
            # Keep the top max_peaks
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        
        # Ensure we have at least some peaks
        if len(peaks) < 10:
            n_keep = max(50, len(b) // 10)
            peaks = np.argpartition(b, -n_keep)[-n_keep:]
        
        # Solve LP
        g = solve_lp_direction_pdlp(f, n, peaks, M, eps=lp_eps)
        
        if g is None:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        # Direction
        h = g / S_g
        
        # Line search
        best_t, best_t_obj = line_search_best(f, h, M, n_trials=n_top_line)
        
        if best_t <= 0:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s
        else:
            break
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if it % 20 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                  f"peaks={len(peaks)} S_g={S_g:.6f} t={time.monotonic()-t0:.0f}s")
        
        # Gradually lower peak fraction
        if it % 40 == 0:
            peak_frac = max(final_peak_frac, peak_frac - 0.01)
        
        # Periodic perturbation if stuck
        if stuck > restart_patience // 2 and stuck % 10 == 0:
            f = f + rng.standard_normal(n) * 0.001 * (f + 1e-6)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            peak_frac = initial_peak_frac
    
    return best_f, best_b


def spectral_equalize(f_init, time_limit, label="spec"):
    """
    Spectral equalization: iteratively adjust f to flatten its power spectrum.
    This reduces peaks in the autocorrelation.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    lr = 0.1
    no_improve = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Get the "excess" at each peak position
        # For each lag k, the contribution to (f*f)[k] from coefficient i is f[i]*f[k-i]
        # We want to reduce the peaks, so compute a "pressure" vector
        excess = np.maximum(b - 0.5 * M, 0.0)  # only consider significant peaks

        top_k = 200

            top_idx = np.arange(len(b))

            if excess[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # pressure[i] += excess[k] * f[k-i] for i in [lo, hi]
            idx = np.arange(lo, hi + 1)
            pressure[idx] += excess[k] * f[k - idx]

        # Use exponential decay (mirror descent style)
        norm_pressure = pressure / (pressure.max() + 1e-15)
        f_new = f * np.exp(-lr * norm_pressure)
        s = f_new.sum()
        if s < 1e-10:

        f_new = f_new / s
        f = f_new

                      f"lr={lr:.4f} t={time.monotonic()-t0:.0f}s")

def simulated_annealing(f_init, time_limit, label="sa"):
    """
    Simulated annealing for local optimization.
    Move: pick 2 random indices, transfer mass between them.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    
    T = 0.01  # initial temperature
    cooling = 0.995
    it = 0
    rng = np.random.default_rng(999)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and T > 1e-8:
        it += 1
        cur_obj = fast_convolve(f).max()
        
        # Generate move: pick 2 indices, transfer delta mass
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(0, 0.02) * f[i]
        if delta > f[i] * 0.5:
            delta = f[i] * 0.3
        
        f_trial = f.copy()
        f_trial[i] -= delta
        f_trial[j] += delta
        
        if f_trial[i] < 0:
            continue
        
        new_obj = fast_convolve(f_trial).max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / max(T, 1e-10)):
            f = f_trial
            if new_obj < best_b - 1e-14:
                best_b = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        T *= cooling
        if no_improve > 200:
            T = max(T * 2, 0.001)
            no_improve = 0
        
        if it % 1000 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                  f"T={T:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.2, 1.5, 2.0])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break
    
    return f, best_b


def mirror_descent_optimize(f_init, time_limit, lr_init=0.05, n_top=100, label="md"):
    """Mirror descent with negative entropy for polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(555)
    temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Softmax weights on peaks
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Focus on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w[k] * f[k - idx]
        
        # Mirror descent update
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new = f_new / s
        f = f_new
        
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr = lr * 0.93
                    no_improve = 0
        
        if it % 200 == 0:
            temp = min(temp * 1.05, 0.03)
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                  f"lr={lr:.4f} t={time.monotonic()-t0:.0f}s")
        
        if no_improve > 40:
            f = np.maximum(f + rng.standard_normal(n) * 0.001 * (f + 1e-6), 0.0)
            f = f / f.sum()
            lr = lr_init
            no_improve = 0
    
    return best_f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Optimization of non-negative step functions to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with corrected constraint matrix, PDLPS solver,
spectral equalization, simulated annealing, mirror descent, and coordinate descent.
Key improvement over previous: fixed the LP constraint matrix to correctly compute
the cross-correlation (f*g)[k] = sum_j f[k-j]*g[j].

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, peak_indices, n):
    """
    Build constraint matrix A where A[row, j] = f[k - j] for the peak at index k.
    This correctly implements (f*g)[k] = sum_j f[k-j]*g[j].
    """
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi < j_lo:
            continue
        js = np.arange(j_lo, j_hi + 1)
        A[row_i, js] = f[k - js]
    return A


def solve_lp_direction(f, peak_indices, n, M, eps=1e-9):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Uses corrected constraint matrix.
    """
    K = len(peak_indices)
    A = build_lp_matrix(f, peak_indices, n)
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solved = False
    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            if solver == cp.PDLP:
                prob.solve(solver=solver, eps_abs=eps, eps_rel=eps, max_iter=300000)
            elif solver == cp.ECOS:
                prob.solve(solver=solver, max_iters=500000, eps=eps, verbose=False)
            else:
                prob.solve(solver=solver, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue
    
    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def line_search_best(f, h, M_current, n_trials=25):
    """
    Find best t in (0, 1) minimizing max(conv((1-t)f + t*h)).
    Uses grid + parabolic refinement.
    """
    best_t = 0.0
    best_obj = M_current
    
    t_grid = np.linspace(0.02, 0.98, n_trials)
    objs = np.full(n_trials, np.inf)
    for i, t in enumerate(t_grid):
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        objs[i] = fast_convolve(f_trial).max()
    
    min_idx = np.argmin(objs)
    if objs[min_idx] >= best_obj:
        return 0.0, best_obj
    
    best_t = t_grid[min_idx]
    best_obj = objs[min_idx]
    
    # Parabolic refinement
    if 0 < min_idx < n_trials - 1:
        t0, t1, t2 = t_grid[min_idx - 1], t_grid[min_idx], t_grid[min_idx + 1]
        f0, f1, f2 = objs[min_idx - 1], objs[min_idx], objs[min_idx + 1]
        if np.isfinite(f0) and np.isfinite(f2):
            den = (t0 - t1) * (f1 - f2) - (t1 - t2) * (f0 - f1)
            if abs(den) > 1e-15:
                num = 0.5 * ((t0 - t2) * (f1 - f2) - (t1 - t2) * (f0 - f2))
                t_star = t1 + num / (2.0 * (f1 - 0.5 * (f0 + f2)) + 1e-30)
                # Use standard parabolic formula
                a_par = (f0 - f1) / (t0 - t1)
                b_par = (f2 - f1) / (t2 - t1)
                if abs(b_par - a_par) > 1e-15:
                    t_star = t1 - (f1 - f0) / (b_par - a_par + 1e-30) * 0.5
                if 0.01 < t_star < 0.99:
                    f_trial = (1.0 - t_star) * f + t_star * h
                    f_trial = np.maximum(f_trial, 0.0)
                    s = f_trial.sum()
                    if s > 1e-10:
                        f_trial = f_trial / s
                        obj = fast_convolve(f_trial).max()
                        if obj < best_obj:
                            best_obj = obj
                            best_t = t_star
    
    return best_t, best_obj


def run_lp_optimize(f_init, time_limit, label="lp", initial_peak_frac=0.95,
                    final_peak_frac=0.70, max_peaks=2500, restart_patience=30,
                    lp_eps=1e-9, n_top_line=25):
    """
    LP-based optimization with corrected constraint matrix and adaptive peaks.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = initial_peak_frac
    rng = np.random.default_rng(20240)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Peak selection
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 10:
            n_keep = max(50, len(b) // 8)
            peaks = np.argpartition(b, -n_keep)[-n_keep:]
        
        # Solve LP
        g = solve_lp_direction(f, peaks, n, M, eps=lp_eps)
        
        if g is None:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        h = g / S_g
        best_t, best_t_obj = line_search_best(f, h, M, n_trials=n_top_line)
        
        if best_t <= 0:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s
        else:
            break
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if it % 15 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                  f"peaks={len(peaks)} S_g={S_g:.4f} t={time.monotonic()-t0:.0f}s")
        
        if it % 30 == 0:
            peak_frac = max(final_peak_frac, peak_frac - 0.012)
        
        if stuck > restart_patience // 2 and stuck % 8 == 0:
            f = f + rng.standard_normal(n) * 0.001 * (f + 1e-6)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            peak_frac = initial_peak_frac
    
    return best_f, best_b

    lr = 0.15
    no_improve = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Pressure: focus on peaks above 40% of max
        excess = np.maximum(b - 0.4 * M, 0.0)
        
        # Get top peaks for efficiency
        if len(b) > n_top_press:
            top_idx = np.argpartition(excess, -n_top_press)[-n_top_press:]
        else:
            top_idx = np.arange(len(b))
        
        # Compute pressure on each coefficient
        pressure = np.zeros(n)
        for k in top_idx:
            if excess[k] < 1e-15:
                continue
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            # pressure[j] += excess[k] * f[k-j]
            pressure[js] += excess[k] * f[k - js]
        
        # Multiplicative weight update
        max_p = pressure.max()
        if max_p < 1e-15:
            break
        norm_p = pressure / max_p
        f = f * np.exp(-lr * norm_p)
        s = f.sum()
        if s < 1e-10:
            break
        f = f / s
        
        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.4)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr = max(lr * 0.85, 0.02)
                    no_improve = 0
            
            if it % 100 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                      f"lr={lr:.4f} t={time.monotonic()-t0:.0f}s")
            
            if no_improve > 40:
                f = np.maximum(f + rng.standard_normal(n) * 0.003 * (f + 1e-6), 0.0)
                f = f / f.sum()
                lr = 0.15
                no_improve = 0
    
    return best_f, best_b


def mirror_descent_optimize(f_init, time_limit, lr_init=0.05, n_top=150, label="md"):
    """Mirror descent with negative entropy for polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(555)
    temp = 0.002
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel
        
        # Gradient: grad[j] = sum_k w[k] * f[k-j]
        grad = np.zeros(n)
        active = np.where(w > 1e-12)[0]
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            grad[js] += w[k] * f[k - js]
        
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f = f_new / s
        
        if it % 40 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr = lr * 0.92
                    no_improve = 0
        
        if it % 150 == 0:
            temp = min(temp * 1.06, 0.04)
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                  f"lr={lr:.4f} t={time.monotonic()-t0:.0f}s")
        
        if no_improve > 40:
            f = np.maximum(f + rng.standard_normal(n) * 0.002 * (f + 1e-6), 0.0)
            f = f / f.sum()
            lr = lr_init
            no_improve = 0
    
    return best_f, best_b


def simulated_annealing(f_init, time_limit, label="sa"):
    """Simulated annealing: transfer mass between pairs of indices."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    
    T_init = 0.005
    T = T_init
    cooling = 0.998
    it = 0
    rng = np.random.default_rng(999)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and T > 1e-9:
        it += 1
        cur_obj = fast_convolve(f).max()
        
        # Pick 2 indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(0.001, 0.03) * min(f[i], f[j]) + rng.uniform(0, 0.01) * f.max()
        if delta > 0.4 * f[i]:
            delta = 0.3 * f[i]
        if delta < 1e-8:
            continue
        
        f_trial = f.copy()
        f_trial[i] -= delta
        f_trial[j] += delta
        if f_trial[i] < 0:
            continue
        
        new_obj = fast_convolve(f_trial).max()
        delta_obj = new_obj - cur_obj
        
        if delta_obj < 0 or (delta_obj / max(T, 1e-10)) < -math.log(rng.random()):
            f = f_trial
            if new_obj < best_b - 1e-14:
                best_b = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        T *= cooling
        if no_improve > 500:
            T = max(T * 1.5, T_init * 0.3)
            no_improve = 0
        
        if it % 2000 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                  f"T={T:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_f.max() * 0 + best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent targeting peak-reducing moves."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.6])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Prioritize indices that contribute to peaks
        b = fast_convolve(f)
        # Compute "importance" of each index
        importance = np.zeros(n)
        top_k = 100
        if len(b) > top_k:
            peak_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            peak_idx = np.arange(len(b))
        for k in peak_idx:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            importance[js] += b[k] * f[k - js]
        
        # Process in order of importance
        order = np.argsort(-importance)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break
    
    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
