def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.

    # Verify that the input is a list

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    sequence = [max(0, x) for x in sequence]

    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

        return np.inf

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_cvxpy(f, M, solver='CLARABEL'):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0 using cvxpy."""
    import cvxpy as cp
    n = len(f)
    C = build_C(f)
    g_var = cp.Variable(n, nonneg=True)
    constraints = [C @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None
    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0)


def solve_lp_scipy(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0 using scipy linprog."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def greedy_lp(f, M):
    """Greedy approximation to the LP: build g one coordinate at a time."""
    n = len(f)
    g = np.zeros(n)
    conv_f = f.copy()  # (f * g)(k) accumulated
    conv_vals = np.zeros(2 * n - 1)
    
    order = np.arange(n)
    for j in order:
        # Maximum g[j] such that for all k: conv_vals[k] + f[k-j]*g[j] <= M
        # k ranges from j to j+n-1
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:n]  # f[k-j] for k in [j, j+n-1] is f[0], f[1], ..., f[n-1]
        # Actually f[k-j] where k goes from j to j+n-1, so k-j goes from 0 to n-1
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    
    return g


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space: a = f^(1-t) * g^t, normalized."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s

def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective_and_grad(z, T, n):
    """Compute smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    
    # Objective: (2n/T) * sum(w_k) * exp(T * (conv@w - max_shifted))... 
    # Actually: soft_max of conv = T * log(sum(exp(conv/T)))
    # = T * (max_shifted + log(sum(exp(shifted - max_shifted))))
    log_sum_exp = max_shifted + np.log(w.sum())
    obj = (2.0 * n) * T * log_sum_exp
    
    # Gradient w.r.t. a: 2n * (w * a) cross-correlation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    
    # Project to log-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, (2.0 * n) * grad_z


def lbgfs_logspace(a0, time_limit, T, maxiter=5000):

    best_score = compute_score(a0)

            return 1e15
        a_tmp = softmax_from_z(z)
        sc = compute_score(a_tmp)
        if sc < best_score:
            nonlocal_best = sc
        obj, _ = smoothed_objective_and_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = smoothed_objective_and_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',

        pass

            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            # Golden section search for optimal transfer t
            # t > 0: move from i to j; t < 0: move from j to i
            # Effective range: t in [-a[j], a[i]]

            t_hi = a[i] - 1e-15
            
            if t_hi <= t_lo:

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618

                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t

                if ss < 1e-15:
                    return 1e10

                    f2 = f1

                else:

                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)

def peak_focused_search(a0, time_limit, verbose=True):
    """Search that specifically targets the top peaks for reduction."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)

    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        
        # Find top 5 peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Identify contributing pairs for this peak
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[int(pk) - idx]
            
            # Try reducing the top contributing element
            top_contrib_idx = np.argsort(contribs)[::-1][:3]
            
            for ci in top_contrib_idx:
                i_idx = idx[ci]
                if a[i_idx] < 1e-14:
                    continue
                
                # Try different amounts of reduction
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[i_idx] * frac
                    a_trial = a.copy()
                    a_trial[i_idx] -= delta
                    # Redistribute to neighbors
                    mask = np.ones(n, dtype=bool)
                    mask[i_idx] = False
                    others = a_trial[mask]
                    os_sum = others.sum()
                    if os_sum < 1e-15:
                        continue
                    a_trial[mask] = others + delta * others / os_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved:
                # Try adding to a position that doesn't contribute to this peak
                # (i.e., a position far from the peak's contributing range)
                noncontrib = np.setdiff1d(np.arange(n), idx)
                if len(noncontrib) > 0:
                    j_idx = noncontrib[rng.integers(0, len(noncontrib))]
                    for frac in [0.01, 0.02, 0.05]:
                        delta = a[j_idx] * frac
                        a_trial = a.copy()
                        a_trial[j_idx] -= delta
                        mask = np.ones(n, dtype=bool)
                        mask[j_idx] = False
                        others = a_trial[mask]
                        os_sum = others.sum()
                        if os_sum < 1e-15:
                            continue
                        a_trial[mask] = others + delta * others / os_sum
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
        
        if not improved and iters % 50 == 0:
            # Random restart perturbation
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
