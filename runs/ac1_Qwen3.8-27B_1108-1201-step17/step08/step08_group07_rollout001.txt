import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def logsumexp(x, axis=None):
    x_max = np.max(x, axis=axis, keepdims=True)
    return (x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))).squeeze(axis)

def softmax(z):
    z_max = z.max()
    e = np.exp(z - z_max)
    return e / e.sum()

def compute_conv_and_weights(a, T):
    """Compute convolution and softmax weights."""
    n = len(a)
    conv = fast_convolve(a)
    Tconv = T * conv
    Tconv_max = Tconv.max()
    w = np.exp(Tconv - Tconv_max)
    w = w / w.sum()
    return conv, w

def compute_gradient_z(z, T):
    """Compute gradient of smoothed objective w.r.t. z (log-space)."""
    n = len(z)
    a = softmax(z)
    conv, w = compute_conv_and_weights(a, T)
    
    # h[i] = sum_k w[k] * a[k-i] (cross-correlation of w and a)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    h = irfft(W * np.conj(A), fft_size)[:n]
    
    S = np.dot(h, a)
    grad_z = (2.0 * n / T) * a * (h - S)
    return grad_z, a, conv.max()

class AdamOptimizer:
    def __init__(self, dim, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.dim = dim
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
    
    def step(self, x, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def adam_optimize(z0, time_limit, T_start=10000, T_end=2000, lr=0.01, verbose=True, label="Adam"):

"""
Hybrid Adam + LP-Fictitious-Play optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive-temperature log-sum-exp smoothing in log-space with Adam, interleaved
with LP-based fictitious play steps and peak-targeted coordinate descent.
Multiple restarts with perturbation help escape local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_f(z):
    """Numerically stable softmax."""
    z_max = z.max()
    e = np.exp(z - z_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_grad_z(z, T):
    """
    Compute gradient of (2n/T)*logsumexp(T*conv(softmax(z), softmax(z))) w.r.t. z.
    Returns (grad_z, a, max_conv).
    """
    n = len(z)
    a = softmax_f(z)
    conv = fast_convolve(a)
    Tconv = T * conv
    Tconv_max = Tconv.max()
    w = np.exp(Tconv - Tconv_max)
    w = w / w.sum()

    # h[i] = sum_k w[k] * a[k-i] (cross-correlation)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    h = irfft(W * np.conj(A), fft_size)[:n]

    S = np.dot(h, a)
    grad_z = (2.0 * n / T) * a * (h - S)
    return grad_z, a, conv.max()


def adam_step(m, v, t, x, grad, lr, beta1, beta2, eps):
    """Single Adam step, returns updated (m, v, x_new)."""
    m_new = beta1 * m + (1.0 - beta1) * grad
    v_new = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m_new / (1.0 - beta1 ** t)
    v_hat = v_new / (1.0 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return m_new, v_new, x_new


def adam_optimize(z0, time_limit, T_start, T_end, lr, label, max_stall=4000,
                  verbose=True, total_est=60000):
    """
    Adam optimization in log-space with adaptive temperature and restarts.
    Returns (best_a, best_score).
    """
    n = len(z0)
    z = z0.copy()
    a = softmax_f(z)
    best_z = z.copy()
    best_score = compute_score(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        progress = min(iters / total_est, 1.0)
        T = T_start * (T_end / T_start) ** progress

        grad_z, a_curr, _ = compute_grad_z(z, T)
        t += 1
        m, v, z = adam_step(m, v, t, z, grad_z, lr, beta1, beta2, eps)

        # Re-center z to prevent drift
        z = z - z.mean()

        a_new = softmax_f(z)
        score = compute_score(a_new)

        if score < best_score - 1e-13:
            best_score = score
            best_z = z.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            rng = np.random.default_rng(hash(iters) % 100000)
            z = best_z + rng.standard_normal(n) * 0.02
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stall = 0

        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters}, T={T:.0f}, best={best_score:.8f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return softmax_f(best_z), best_score


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    from scipy.linalg import toeplitz
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response_simple(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0 using cvxpy."""
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fp_short(a0, time_limit, step=0.4, label="LP-FP", verbose=True):
    """Short LP fictitious play."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response_simple(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def peak_targeted_cd(a0, time_limit, n_pairs=500, verbose=True):
    """
    Coordinate descent that specifically targets reducing the peak convolution value.
    Moves mass from positions contributing most to the peak to positions contributing least.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        peak_idx = conv.argmax()

        # Find which pairs (i, peak_idx - i) contribute most to the peak
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i]
        contributors = []
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contributors.append((i, j, a[i] * a[j]))

        contributors.sort(key=lambda x: -x[2])
        hot_positions = [c[0] for c in contributors[:max(1, n // 10)]]
        cold_positions = [c[0] for c in contributors[-max(1, n // 10):]]

        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            # Pick a hot source and cold destination
            i = rng.choice(hot_positions) if hot_positions else rng.integers(0, n)
            j = rng.choice(cold_positions) if cold_positions else rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            max_t = a[i] * 0.8

            for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                t = frac * max_t
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 10:
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def random_perturb_redo(a_best, n_restart, time_per_restart, T_start, T_end, lr, label, verbose=True):
    """

    z_best = z_best - z_best.mean()
    best_a = a_best.copy()
    best_score = compute_score(a_best)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for r in range(n_restart):
        if time.monotonic() - t0 > time_per_restart * (r + 1) + 60:
            break
        if time.monotonic() - t0 + time_per_restart > time_per_restart * n_restart + 120:
            break

        perturbation_scale = 0.03 + 0.02 * r
        z_perturbed = z_best + rng.standard_normal(n) * perturbation_scale
        z_perturbed = z_perturbed - z_perturbed.mean()

        t_start = time.monotonic()
        remaining = time_per_restart - (time.monotonic() - t0 - (time_per_restart * r))
        if remaining < 10:
            continue

        a_r, score_r = adam_optimize(
            z_perturbed, min(remaining, time_per_restart - 30),
            T_start=T_start, T_end=T_end, lr=lr,
            label=f"{label}_r{r}", max_stall=3000,
            verbose=verbose, total_est=30000
        )

        if score_r < best_score:
            best_a = a_r
            best_score = score_r
            if verbose:
                print(f"    {label} restart {r}: NEW BEST = {score_r:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
