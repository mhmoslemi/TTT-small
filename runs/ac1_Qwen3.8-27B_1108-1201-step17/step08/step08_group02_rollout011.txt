Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines smooth gradient (mirror descent with log-sum-exp) and LP-based
direction finding. Uses temperature annealing, multi-length exploration,
and coordinate descent for final polishing.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def soft_max_value(b, t):
    """Compute soft max of b with temperature t."""
    shifted = b - b.max()
    return b.max() + t * np.log(np.sum(np.exp(shifted / t)))


def compute_smooth_grad(f, t):
    """
    Compute gradient of soft_max(conv(f,f), t) with respect to f.
    Returns (grad, weights, soft_max_value).
    """
    n = len(f)
    b = fast_convolve(f)
    shifted = b - b.max()
    w = np.exp(shifted / t)
    w /= w.sum()
    smv = b.max() + t * np.log(np.sum(np.exp(shifted / t)))
    
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]
    # Using FFT: this is correlate(w, f, 'valid')
    # = convolve(f[::-1], w, 'full')[n-1 : 2*n-1]
    conv_len = n + (2*n - 1) - 1  # = 3n - 2

    F_rev = rfft(f[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv_full = irfft(F_rev * W, fft_size)
    grad = 2.0 * conv_full[n-1 : 2*n-1]
    
    return grad, w, smv

    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def mirror_descent_phase(f_init, time_limit, t_init=0.1, t_min=0.001,
                         lr_init=0.05, label="md", n_top=None,
                         lp_every=50, lp_peak_frac=0.95):
    """
    Mirror descent with log-sum-exp smoothing and periodic LP steps.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_true_obj = fast_convolve(f).max()
    t = t_init
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, w, smv = compute_smooth_grad(f, t)
        
        # Mirror descent update
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        # Numerical stability
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new /= s
        f = f_new
        
        # Periodically check true objective
        if it % 20 == 0:
            true_obj = fast_convolve(f).max()
            if true_obj < best_true_obj - 1e-12:
                best_true_obj = true_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive learning rate
            if no_improve > 10:
                lr *= 0.9
                no_improve = 0
            elif no_improve < 5 and it > 100:
                lr = min(lr * 1.05, lr_init * 2)
            
            # Anneal temperature
            t = max(t_min, t * 0.995)
            
            if it % 100 == 0:
                score = 2.0 * n * best_true_obj
                print(f"    {label} iter={it} obj={best_true_obj:.10f} "
                      f"score={score:.6f} t={t:.5f} lr={lr:.4f} "
                      f"elapsed={time.monotonic()-t0:.0f}s")
            
            # Perturbation if stuck
            if no_improve > 40:
                f = f + rng.standard_normal(n) * 0.002 * np.maximum(f, 0.01)
                f = np.maximum(f, 1e-10)
                f /= f.sum()
                lr = lr_init
                t = max(t_min, t_init * 0.5)
                no_improve = 0
        
        # Periodic LP step
        if lp_every > 0 and it % lp_every == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 5:
                break
            b = fast_convolve(f)
            M = b.max()
            threshold = lp_peak_frac * M
            peaks = np.where(b >= threshold)[0]
            if len(peaks) > 200:
                peaks = np.argpartition(b, -200)[-200:]
            if len(peaks) < 3:
                peaks = np.argpartition(b, -10)[-10:]
            
            g = solve_lp_direction(f, n, peaks, M, eps=1e-7)
            if g is not None and g.sum() > 1.0 + 1e-4:
                h = g / g.sum()
                # Try a few step sizes
                best_t_step = 0.0
                best_obj = M
                for step in [0.1, 0.2, 0.3, 0.5]:
                    f_trial = (1.0 - step) * f + step * h
                    f_trial = np.maximum(f_trial, 0.0)
                    f_trial /= f_trial.sum()
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_t_step = step

                    f /= f.sum()
                    if best_obj < best_true_obj:
                        best_true_obj = best_obj
                        best_f = f.copy()
    
    return best_f, best_true_obj


def subgradient_phase(f_init, time_limit, lr_init=0.003, n_top=100, label="sg"):
    """Subgradient descent with momentum for polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_true_obj = fast_convolve(f).max()
    lr = lr_init
    temp = 0.01
    no_improve = 0
    it = 0
    rng = np.random.default_rng(444)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Softmax weights focused on top peaks
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel
        
        # Subgradient
        conv_len = 3 * n - 2
        fft_size = 1 << (conv_len - 1).bit_length()
        F_rev = rfft(f[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv_full = irfft(F_rev * W, fft_size)
        grad = 2.0 * conv_full[n-1 : 2*n-1]
        
        # Momentum update
        v = 0.9 * v - lr * grad
        f = f + v
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            continue
        f /= s
        
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_true_obj - 1e-12:
                best_true_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 8:
                    lr *= 0.93
                    no_improve = 0
            
            if it % 200 == 0:
                temp = min(temp * 1.1, 0.05)
                score = 2.0 * n * best_true_obj
                print(f"    {label} iter={it} obj={best_true_obj:.10f} "
                      f"score={score:.6f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
            
            if no_improve > 30:
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 1e-10)
                f /= f.sum()
                v = np.zeros(n)
                no_improve = 0
                lr = lr_init
    
    return best_f, best_true_obj


def coordinate_descent(f_init, time_limit, label="cd", multipliers=None):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_true_obj = fast_convolve(f).max()
    if multipliers is None:
        multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_true_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_true_obj - 1e-14:
                    best_true_obj = best_j_obj
                    improved = True
        if not improved:
            break
    
    return f, best_true_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def run_lp_optimize(f_init, time_limit, label="lp", initial_peak_frac=0.95,
                    final_peak_frac=0.85, max_peaks=300, restart_patience=25,
                    lp_eps=1e-8):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_true_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = initial_peak_frac
    rng = np.random.default_rng(2024)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(b, -max(10, len(b) // 10))[-max(10, len(b) // 10):]
        
        g = solve_lp_direction(f, n, peaks, M, eps=lp_eps)
        
        if g is None or g.sum() <= 1.0 + 1e-5:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        h = g / g.sum()
        # Line search
        best_t = 0.0
        best_t_obj = M
        for t_step in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t_step) * f + t_step * h
            f_trial = np.maximum(f_trial, 0.0)
            f_trial /= f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t_step
        
        if best_t <= 0:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f /= f.sum()
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_true_obj - 1e-14:
            best_true_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if it % 10 == 0:
            score = 2.0 * n * best_true_obj
            print(f"    {label} iter={it} obj={best_true_obj:.10f} "
                  f"score={score:.6f} t={time.monotonic()-t0:.0f}s")
        
        if it % 30 == 0:
            peak_frac = max(final_peak_frac, peak_frac - 0.005)
        
        if stuck > restart_patience // 2 and stuck % 5 == 0:
            f = f + rng.standard_normal(n) * 0.001 * f
            f = np.maximum(f, 0.0)
            f /= f.sum()
    
    return best_f, best_true_obj

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines smooth gradient (mirror descent with log-sum-exp), LP-based
direction finding, subgradient descent, coordinate descent, and
multi-length exploration with temperature annealing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_grad(f, t):
    """
    Compute gradient of soft_max(conv(f,f), t) with respect to f.
    Returns (grad, weights, soft_max_value).
    """
    n = len(f)
    b = fast_convolve(f)
    shifted = b - b.max()
    w = np.exp(shifted / t)
    w /= w.sum()
    smv = b.max() + t * np.log(np.sum(np.exp(shifted / t)))

    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]
    # = 2 * correlate(w, f, 'valid')
    # = 2 * convolve(f[::-1], w, 'full')[n-1 : 2*n-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F_rev = rfft(f[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv_full = irfft(F_rev * W, fft_size)
    grad = 2.0 * conv_full[n - 1:2 * n - 1]

    return grad, w, smv


def solve_lp_direction(f, n, peak_indices, M, eps=1e-8):
    """Solve LP for direction finding."""
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver_name, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def mirror_descent_phase(f_init, time_limit, t_init=0.05, t_min=0.001,
                         lr_init=0.08, label="md", lp_every=40,
                         lp_peak_frac=0.93):
    """Mirror descent with log-sum-exp smoothing and periodic LP steps."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_true_obj = fast_convolve(f).max()
    t = t_init
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, w, smv = compute_smooth_grad(f, t)

        # Mirror descent update
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new /= s
        f = f_new

        # Periodically check true objective
        if it % 20 == 0:
            true_obj = fast_convolve(f).max()
            if true_obj < best_true_obj - 1e-12:
                best_true_obj = true_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 10:
                lr *= 0.9
                no_improve = 0
            elif no_improve < 3 and it > 100:
                lr = min(lr * 1.05, lr_init * 3)

            t = max(t_min, t * 0.995)

            if it % 100 == 0:
                score = 2.0 * n * best_true_obj
                print(f"    {label} iter={it} obj={best_true_obj:.10f} "
                      f"score={score:.6f} t={t:.5f} lr={lr:.4f} "
                      f"elapsed={time.monotonic()-t0:.0f}s")

            if no_improve > 50:
                f = f + rng.standard_normal(n) * 0.003 * np.maximum(f, 0.005)
                f = np.maximum(f, 1e-10)
                f /= f.sum()
                lr = lr_init
                t = max(t_min, t_init * 0.5)
                no_improve = 0

        # Periodic LP step
        if lp_every > 0 and it % lp_every == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 5:
                break
            b = fast_convolve(f)
            M = b.max()
            threshold = lp_peak_frac * M
            peaks = np.where(b >= threshold)[0]
            if len(peaks) > 200:
                peaks = np.argpartition(b, -200)[-200:]
            if len(peaks) < 3:
                peaks = np.argpartition(b, -10)[-10:]

            g = solve_lp_direction(f, n, peaks, M, eps=1e-7)
            if g is not None and g.sum() > 1.0 + 1e-4:
                h = g / g.sum()
                best_t_step = 0.0
                best_obj = M
                for step in [0.1, 0.2, 0.3, 0.5]:
                    f_trial = (1.0 - step) * f + step * h
                    f_trial = np.maximum(f_trial, 0.0)
                    f_trial /= f_trial.sum()
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_t_step = step
                if best_t_step > 0:
                    f = (1.0 - best_t_step) * f + best_t_step * h
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                    if best_obj < best_true_obj:
                        best_true_obj = best_obj
                        best_f = f.copy()

    return best_f, best_true_obj


def subgradient_phase(f_init, time_limit, lr_init=0.003, n_top=120, label="sg"):
    """Subgradient descent with momentum for polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_true_obj = fast_convolve(f).max()
    lr = lr_init
    temp = 0.01
    no_improve = 0
    it = 0
    rng = np.random.default_rng(444)

    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        F_rev = rfft(f[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv_full = irfft(F_rev * W, fft_size)
        grad = 2.0 * conv_full[n - 1:2 * n - 1]

        v = 0.9 * v - lr * grad
        f = f + v
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            continue
        f /= s

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_true_obj - 1e-12:
                best_true_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 8:
                    lr *= 0.93
                    no_improve = 0

            if it % 200 == 0:
                temp = min(temp * 1.1, 0.05)
                score = 2.0 * n * best_true_obj
                print(f"    {label} iter={it} obj={best_true_obj:.10f} "
                      f"score={score:.6f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 30:
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 1e-10)
                f /= f.sum()
                v = np.zeros(n)
                no_improve = 0
                lr = lr_init

    return best_f, best_true_obj


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_true_obj = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_true_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_true_obj - 1e-14:
                    best_true_obj = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_true_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def run_lp_optimize(f_init, time_limit, label="lp", initial_peak_frac=0.94,
                    final_peak_frac=0.85, max_peaks=350, restart_patience=25,
                    lp_eps=1e-8):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_true_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = initial_peak_frac
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(b, -max(10, len(b) // 10))[-max(10, len(b) // 10):]

        g = solve_lp_direction(f, n, peaks, M, eps=lp_eps)

        if g is None or g.sum() <= 1.0 + 1e-5:
            stuck += 1
            if stuck > restart_patience:
                break
            continue

        h = g / g.sum()
        best_t = 0.0
        best_t_obj = M
        for t_step in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t_step) * f + t_step * h
            f_trial = np.maximum(f_trial, 0.0)
            f_trial /= f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t_step

        if best_t <= 0:
            stuck += 1
            if stuck > restart_patience:
                break
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f /= f.sum()

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_true_obj - 1e-14:
            best_true_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 10 == 0:
            score = 2.0 * n * best_true_obj
            print(f"    {label} iter={it} obj={best_true_obj:.10f} "
                  f"score={score:.6f} t={time.monotonic()-t0:.0f}s")

        if it % 30 == 0:
            peak_frac = max(final_peak_frac, peak_frac - 0.005)

        if stuck > restart_patience // 2 and stuck % 5 == 0:
            f = f + rng.standard_normal(n) * 0.001 * f
            f = np.maximum(f, 0.0)
            f /= f.sum()

    return best_f, best_true_obj


def generate_random_init(n, seed):
    """Generate a random initial sequence based on random phase in frequency domain."""
    rng = np.random.default_rng(seed)
    # Random phases in frequency domain
    phases = np.exp(1j * rng.uniform(0, 2 * np.pi, n))
    # Take inverse FFT and take magnitude
    f = np.abs(irfft(phases, n))
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    else:
        f = np.ones(n) / n
    return f

                print(f"  NEW BEST [rr{attempt+1}]: {vs_r:.6f}")

    # === Phase 7: Coordinate descent ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.15, 120.0)
    if t7 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 7: CD (n={n_b}, budget={t7:.0f}s) ===")
        f7, obj7 = coordinate_descent(best_seq, time_limit=t7, label="cd")
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {vs7:.6f}")

    # === Phase 8: Final LP burst ===
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.25, 150.0)
    if t8 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 8: Final LP (n={n_b}, budget={t8:.0f}s) ===")
        f8, obj8 = run_lp_optimize(best_seq, time_limit=t8, label="lp_final",
                                    initial_peak_frac=0.92, final_peak_frac=0.82,
                                    max_peaks=400, restart_patience=20, lp_eps=1e-9)
        vs8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(vs8) and vs8 < best_score:
            best_score = vs8
            best_seq = f8.copy()
            print(f"  NEW BEST [lp_final]: {vs8:.6f}")

    # === Phase 9: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t9 = min(rem * 0.30, 80.0)
    if t9 > 10:
        print(f"\n=== Phase 9: Final CD (budget={t9:.0f}s) ===")
        f9, obj9 = coordinate_descent(best_seq, time_limit=t9, label="cd_final")
        vs9 = float(evaluate_sequence(f9.tolist()))
        if np.isfinite(vs9) and vs9 < best_score:
            best_score = vs9
            best_seq = f9.copy()
            print(f"  NEW BEST [cd_final]: {vs9:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
