def fast_convolve(a):

    size = 1 << (2*n - 2).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2*n-1]

def compute_score(a):

    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:

    A = rfft(a, size)
    return irfft(P * np.conj(A), size)[:n]

def smoothed_gradient_z(z, T):
    """Gradient of smoothed objective w.r.t. z (log-space)."""
    n = len(z)
    a = softmax(z)
    conv = fast_convolve(a)
    # Softmax weights
    m = conv.max()
    p = np.exp(T * (conv - m))
    p = p / p.sum()
    # h[j] = sum_k p[k] * a[k-j]
    h = cross_corr(p, a)
    # ∇_z f = 4n * a * (h - dot(a,h))
    shift = np.dot(a, h)
    return 4.0 * n * a * (h - shift)

# === Optimization methods ===

def adam_logspace(a0, time_limit, T_start=10, T_end=500, lr=0.01):
    """Adam optimizer in log-space on the smoothed objective."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Temperature schedule
        frac = min(1.0, it / max(1, int(time_limit * 10)))  # rough estimate
        T = T_start + (T_end - T_start) * frac
        
        grad = smoothed_gradient_z(z, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a = softmax(z)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
