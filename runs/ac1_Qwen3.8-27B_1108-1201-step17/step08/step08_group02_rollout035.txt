def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)

    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0) * target_sum

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth-max projected gradient descent with momentum (main optimizer)
2. LP-based polishing (using cvxpy)
3. Coordinate descent for final adjustments
4. Multi-start and length exploration

import numpy as np
from scipy.fft import rfft, irfft
import time
import math

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fft_convolve(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        val = max(float(v[0]), 0.0)
        return np.array([val])
    # Scale to project onto sum=1 simplex first, then scale to target
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    proj = np.maximum(v - theta, 0.0)
    # proj now sums to 1, scale to target_sum
    return proj * target_sum

def compute_smooth_gradient(f, b, temp):
    """Compute gradient of log-sum-exp approximation to max(b).
    
    Uses FFT-based cross-correlation for efficiency.
    Returns gradient of shape (n,).
    """
    n = len(f)
    # Softmax weights
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform weights
        w = np.ones(len(b)) / len(b)
    else:
        w /= w_sum
    
    # grad[j] = 2 * sum_{m=0}^{n-1} f[m] * w[j+m]
    # = 2 * conv(flip(f), w)[j + n - 1]
    conv_result = fft_convolve(np.flip(f), w)
    # We need indices n-1 to 2n-2 (inclusive)
    grad = 2.0 * conv_result[n-1:2*n-1]
    
    return grad, w
