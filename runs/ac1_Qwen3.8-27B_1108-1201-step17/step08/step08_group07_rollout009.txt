"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses ADAM with logsumexp-smoothed objective in softmax parameterization,
combined with LP fictitious play and Bussgang spectral flattening.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.linalg import toeplitz

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def softmax_fn(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def logsumexp_smoothed_obj_grad(z, T):
    """Compute smoothed objective and gradient in softmax parameterization.
    L(z) = 2n * T * logsumexp(conv(softmax(z), softmax(z)) / T)
    """
    n = len(z)
    a = softmax_fn(z)
    C = fast_convolve(a)  # length 2n-1
    CT = C / T
    CT_shift = CT - CT.max()
    exp_CT = np.exp(CT_shift)
    sum_exp = exp_CT.sum()
    L = 2.0 * n * T * (CT.max() + np.log(sum_exp))
    W = exp_CT / sum_exp  # length 2n-1
    
    # Compute g = 4n * (a cross-correlate W)[:n]
    # Cross-correlation: (a ⋆ W)[i] = sum_j a[j] * W[j+i]
    # Use FFT: IFFT(conj(FFT(a_pad)) * FFT(W_pad))
    L_fft = 1
    while L_fft < 3 * n - 2:
        L_fft *= 2
    a_pad = np.zeros(L_fft)
    a_pad[:n] = a
    W_pad = np.zeros(L_fft)
    W_pad[:2*n-1] = W
    A_f = rfft(a_pad)
    W_f = rfft(W_pad)
    V = irfft(np.conj(A_f) * W_f, L_fft)[:n]
    g = 4.0 * n * V
    
    dot_ga = float(np.dot(g, a))
    grad_z = a * (g - dot_ga)
    return L, grad_z

def adam_optimize_z(z0, T, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, 
                    time_limit=60.0, max_iter=100000, verbose=False):
    """ADAM optimization of smoothed objective in z-space."""
    n = len(z0)
    z = z0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_z = z.copy()
    best_L = np.inf
    
    L, grad = logsumexp_smoothed_obj_grad(z, T)
    if L < best_L:
        best_L = L
        best_z = z.copy()
    
    while t < max_iter and (time.monotonic() - t0) < time_limit:
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        L, grad = logsumexp_smoothed_obj_grad(z, T)
        if L < best_L:
            best_L = L
            best_z = z.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ADAM T={T:.0e} iter={t} L={L:.8f} best={best_L:.8f} time={elapsed:.0f}s")
    
    return best_z, best_L

def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)

def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

def lp_fictitious_play(a0, time_limit, step=0.3, verbose=True, label="LP"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    
    while (time.monotonic() - t0) < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        
        if stall_count > 200:
            step = max(0.05, step * 0.5)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters} best={best_score:.8f} step={step:.3f} t={elapsed:.0f}s")
    
    return best_a, best_score

def bussgang_step(a, tau=0.3):
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    power = np.abs(A) ** 2
    total = power.sum()
    target = total / fft_size
    ratio = np.where(power > 1e-30, target / power, 1.0)
    ratio = np.clip(ratio, 0.01, 100.0)
    A_new = A * (ratio ** tau)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss < 1e-15:
        return a
    return a_new / ss

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
