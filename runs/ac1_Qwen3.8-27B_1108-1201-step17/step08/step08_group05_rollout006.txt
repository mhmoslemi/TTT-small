def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    return f_new

    if parent is None:
        parent = np.ones(1024) / 1024.0
    
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(parent.tolist()))
    n_parent = len(parent)
    print(f"Initial: n={n_parent}, score={best_score:.8f}")
    
    # === Phase 1: LP optimization on parent ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP optimization (n={n_parent}, budget={t1:.0f}s) ===")
        f1, obj1 = run_lp_optimization(parent, time_limit=t1, label="lp_main")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp_main]: {score1:.8f}")
        else:
            print(f"  lp_main: {score1:.8f} (no improvement)")
    
    # === Phase 2: Mirror descent polish ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 150.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Mirror descent (n={len(best_seq)}, budget={t2:.0f}s) ===")
        f2, obj2 = run_mirror_descent(best_seq, time_limit=t2, label="md_1")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [md_1]: {score2:.8f}")
        else:
            print(f"  md_1: {score2:.8f} (no improvement)")
    
    # === Phase 3: Perturbation + re-optimization ===
    rem = deadline - time.monotonic()
    if rem > 60:
        n_restarts = 3
        t3_total = min(rem * 0.25, 250.0)
        t3_per = t3_total / n_restarts
        print(f"\n=== Phase 3: Perturbation restarts (budget={t3_total:.0f}s) ===")
        
        for restart_i in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_r = min(t3_per, rem - 5)
            if t_r < 15:
                break
            
            # Different perturbation strengths
            strengths = [0.005, 0.01, 0.02]
            f_pert = perturbation_restart(best_seq, rng, strength=strengths[restart_i % len(strengths)])
            
            # Quick LP + mirror descent
            t_lp = t_r * 0.6
            t_md = t_r * 0.4
            
            f_r, obj_r = run_lp_optimization(f_pert, time_limit=t_lp, label=f"restart_{restart_i}_lp")
            score_r = float(evaluate_sequence(f_r.tolist()))
            
            if np.isfinite(score_r) and score_r < best_score:
                best_score = score_r
                best_seq = f_r.copy()
                print(f"  NEW BEST [restart_{restart_i}]: {score_r:.8f}")
            else:
                print(f"  restart_{restart_i}: {score_r:.8f} (no improvement)")
            
            # Mirror descent polish on best so far
            if t_md > 10:
                f_md, obj_md = run_mirror_descent(best_seq, time_limit=t_md, label=f"restart_{restart_i}_md")
                score_md = float(evaluate_sequence(f_md.tolist()))
                if np.isfinite(score_md) and score_md < best_score:
                    best_score = score_md
                    best_seq = f_md.copy()
                    print(f"  NEW BEST [restart_{restart_i}_md]: {score_md:.8f}")
    
    # === Phase 4: Multi-length exploration ===
    rem = deadline - time.monotonic()
    if rem > 80:
        n_base = len(best_seq)
        # Try lengths near the current best
        candidate_lengths = []
        for delta in [-32, -16, -8, 8, 16, 32, 64]:
            cn = n_base + delta
            if 512 <= cn <= 3000:
                candidate_lengths.append(cn)
        
        t4_total = min(rem * 0.3, 400.0)
        t4_per = t4_total / max(1, len(candidate_lengths))
        print(f"\n=== Phase 4: Multi-length exploration (budget={t4_total:.0f}s) ===")
        
        for cn in candidate_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_cn = min(t4_per, rem - 5)
            if t_cn < 15:
                break
            
            # Interpolate best to new length
            f_interp = interpolate_sequence(best_seq, cn)
            
            # Also try a window
            windows = generate_windows(cn)
            
            best_for_len = None
            best_score_len = float('inf')
            
            inits = [f_interp] + windows[:2]  # interpolated + 2 windows
            t_per_init = t_cn / len(inits)
            
            for init_i, f_init in enumerate(inits):
                rem_init = deadline - time.monotonic()
                t_use = min(t_per_init, rem_init - 3)
                if t_use < 10:
                    break
                
                f_opt, obj_opt = run_lp_optimization(f_init, time_limit=t_use * 0.7,
                                                      label=f"n{cn}_init{init_i}")
                score_opt = float(evaluate_sequence(f_opt.tolist()))
                
                if np.isfinite(score_opt) and score_opt < best_score_len:
                    best_score_len = score_opt
                    best_for_len = f_opt.copy()
                
                # Quick mirror descent
                if t_use * 0.3 > 5:
                    f_md, _ = run_mirror_descent(best_for_len if best_for_len is not None else f_init,
                                                  time_limit=t_use * 0.3, label=f"n{cn}_init{init_i}_md")
                    score_md = float(evaluate_sequence(f_md.tolist()))
                    if np.isfinite(score_md) and score_md < best_score_len:
                        best_score_len = score_md
                        best_for_len = f_md.copy()
            
            if best_for_len is not None and best_score_len < best_score:
                best_score = best_score_len
                best_seq = best_for_len.copy()
                print(f"  NEW BEST [n={cn}]: {best_score:.8f}")
            else:
                print(f"  n={cn}: best={best_score_len:.8f} (no improvement over {best_score:.8f})")
    
    # === Phase 5: Final LP deep optimization ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.4, 200.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Final LP deep (n={len(best_seq)}, budget={t5:.0f}s) ===")
        f5, obj5 = run_lp_optimization(best_seq, time_limit=t5, label="lp_final")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [lp_final]: {score5:.8f}")
        else:
            print(f"  lp_final: {score5:.8f} (no improvement)")
    
    # === Phase 6: Final mirror descent polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t6 = min(rem * 0.8, 60.0)
        if t6 > 10:
            print(f"\n=== Phase 6: Final md polish (budget={t6:.0f}s) ===")
            f6, obj6 = run_mirror_descent(best_seq, time_limit=t6, label="md_final")
            score6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(score6) and score6 < best_score:
                best_score = score6
                best_seq = f6.copy()
                print(f"  NEW BEST [md_final]: {score6:.8f}")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Gap: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")
    
    return [float(x) for x in best_seq]
