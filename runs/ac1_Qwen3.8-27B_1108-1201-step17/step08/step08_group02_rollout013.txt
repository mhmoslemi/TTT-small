def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            return np.inf
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    fc = eval_t(c_gs)
    fd = eval_t(d_gs)
    for _ in range(n_evals - 4):
        if fc < fd:
            b_gs = d_gs
            d_gs = c_gs
            fd = fc
            c_gs = b_gs - gr * (b_gs - a_gs)
            fc = eval_t(c_gs)
        else:
            a_gs = c_gs
            c_gs = d_gs
            fc = fd
            d_gs = a_gs + gr * (b_gs - a_gs)
            fd = eval_t(d_gs)

    t_opt = min(c_gs, d_gs) if fc < fd else max(c_gs, d_gs)
    obj_opt = min(fc, fd)
    if obj_opt < best_obj - 1e-15:
        best_obj = obj_opt
        best_t = t_opt

    return best_t, best_obj


def solve_lp(f, M, peak_indices=None, solver_priority=None):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    If peak_indices is None, use all indices.
    Returns (g, sum_g) or (None, 1.0).
    """
    n = len(f)
    total = 2 * n - 1

    if peak_indices is None:
        peak_indices = np.arange(total)

    K = len(peak_indices)

    # Build constraint matrix
    A = np.zeros((K, n))
    indices_i = np.arange(n)
    for j in range(n):
        # f[j] contributes to row k at column (k-j), i.e., A[k, k-j] += f[j]
        # For peak at index pk: A[row, j] where j = pk - i for valid i
        # Actually: (f*g)[pk] = sum_j f[j]*g[pk-j], so A[row, pk-j] = f[j]
        pass

    # More efficient: for each peak index pk, the constraint is
    # sum_j f[j] * g[pk-j] <= M
    # The column index in g is (pk-j), valid when 0 <= pk-j < n, i.e., pk-n+1 <= j <= pk
    for row in range(K):
        pk = int(peak_indices[row])
        j_lo = max(0, pk - n + 1)
        j_hi = min(pk, n - 1)
        col_indices = pk - np.arange(j_lo, j_hi + 1)
        A[row, col_indices] = f[j_lo:j_hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    if solver_priority is None:
        solver_priority = [cp.SCS, cp.ECOS, cp.PDLP, cp.GLPK]

    for solver in solver_priority:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)):
                    return g_val, float(g_val.sum())
        except Exception:
            continue

    return None, 1.0


def run_lp_optimize(f_init, time_limit, label="lp", use_all_peaks=True,
                    peak_frac_start=0.90, peak_frac_end=0.70,
                    max_peaks_partial=500, patience=30):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    total = 2 * n - 1
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = peak_frac_start
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Peak selection
        if use_all_peaks:
            peaks = np.arange(total)
        else:
            threshold = peak_frac * M
            peaks = np.where(b >= threshold)[0]
            if len(peaks) > max_peaks_partial:
                peaks = np.argpartition(b, -max_peaks_partial)[-max_peaks_partial:]
            if len(peaks) < 10:
                peaks = np.argpartition(b, -max(20, total // 10))[-max(20, total // 10):]

        g, S_g = solve_lp(f, M, peak_indices=peaks)

        if g is None or S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > patience:
                # Perturb and try again
                f = f + rng.standard_normal(n) * 0.002 * f
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                stuck = 0
                if it % 20 == 0:
                    # Reset to best
                    f = best_f.copy()
            continue

        # Line search
        d = g / S_g
        best_t, best_t_obj = line_search(f, d, M)

        if best_t <= 0:
            stuck += 1
            if stuck > patience:
                f = f + rng.standard_normal(n) * 0.003 * f
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                stuck = 0
            continue

        # Update
        f_new = (1.0 - best_t) * f + best_t * d
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new

        cur_M = fast_convolve(f_new).max()
        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_f = f_new.copy()
            stuck = 0
            f = f_new
        else:
            # Try smaller step
            for small_t in [best_t * 0.5, best_t * 0.25, best_t * 0.1]:
                f_try = (1.0 - small_t) * f + small_t * d
                f_try = np.maximum(f_try, 0.0)
                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try = f_try / s_try
                m_try = fast_convolve(f_try).max()
                if m_try < best_M - 1e-14:
                    best_M = m_try
                    best_f = f_try.copy()
                    f = f_try
                    stuck = 0
                    break
            else:
                stuck += 1
                if stuck > patience:
                    f = f + rng.standard_normal(n) * 0.003 * f
                    f = np.maximum(f, 1e-15)
                    f = f / f.sum()
                    stuck = 0

        if it % 20 == 0:
            score = 2.0 * n * best_M
            print(f"    {label} iter={it} M={best_M:.10f} score={score:.6f} "
                  f"S_g={S_g:.4f} t={time.monotonic()-t0:.0f}s", flush=True)

        # Gradually lower peak fraction
        if it % 25 == 0:
            peak_frac = max(peak_frac_end, peak_frac - 0.005)

        # Periodic reset to best
        if it % 100 == 0:
            f = best_f.copy()

    return best_f, best_M


def mirror_descent_optimize(f_init, time_limit, lr_init=0.05, n_top=150, label="md"):
    """Mirror descent with negative entropy for polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    total = 2 * n - 1
    f = f_init.copy()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    temp = 0.003

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights on top peaks
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        if n_top < total:
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w = np.zeros(total)
            w[top_idx] = w_sel

        # Gradient: grad[j] = sum_k w[k] * f[k-j] (where valid)
        grad = np.zeros(n)
        for k in range(total):
            if w[k] < 1e-14:
                continue
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            grad[j_lo:j_hi + 1] += w[k] * f[k - np.arange(j_lo, j_hi + 1)]

        # Mirror descent update
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new = f_new / s
        f = f_new

        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_M - 1e-14:
                best_M = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.04, 0.2)
            else:
                no_improve += 1
                if no_improve > 8:
                    lr *= 0.9
                    no_improve = 0

            if it % 150 == 0:
                temp = min(temp * 1.1, 0.03)
                score = 2.0 * n * best_M
                print(f"    {label} iter={it} M={best_M:.10f} score={score:.6f} "
                      f"lr={lr:.4f} t={time.monotonic()-t0:.0f}s", flush=True)

            if no_improve > 40:
                f = best_f + rng.standard_normal(n) * 0.002 * best_f
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                lr = lr_init
                no_improve = 0

    return best_f, best_M


def subgradient_optimize(f_init, time_limit, lr_init=0.003, n_top=100, label="sg"):
    """Projected subgradient descent with momentum and restarts."""
    t0 = time.monotonic()
    n = len(f_init)
    total = 2 * n - 1
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    lr = lr_init
    temp = 0.004
    no_improve = 0
    it = 0
    rng = np.random.default_rng(444)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        if n_top < total:
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w_new = np.zeros(total)
            w_new[top_idx] = w_sel
            w = w_new

        grad = np.zeros(n)
        for k in range(total):
            if w[k] < 1e-14:
                continue
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            grad[j_lo:j_hi + 1] += w[k] * f[k - np.arange(j_lo, j_hi + 1)]

        v = 0.85 * v - lr * grad
        f = project_simplex(f + v)

        if it % 80 == 0:
            obj = fast_convolve(f).max()
            if obj < best_M - 1e-14:
                best_M = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.02)
            else:
                no_improve += 1
                if no_improve > 6:
                    lr *= 0.93
                    no_improve = 0

            if it % 400 == 0:
                temp = min(temp * 1.06, 0.04)
                score = 2.0 * n * best_M
                print(f"    {label} iter={it} M={best_M:.10f} score={score:.6f} "
                      f"t={time.monotonic()-t0:.0f}s", flush=True)

            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_M


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements with multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_M = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.92, 0.96, 1.04, 1.08, 1.15, 1.3, 1.5, 2.0])

    full_pass = 0
    while time.monotonic() - t0 < time_limit:
        improved_this_pass = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = fast_convolve(f).max()
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                new_M = fast_convolve(f).max()
                if new_M < best_M - 1e-14:
                    best_M = new_M
                    improved_this_pass = True
            if time.monotonic() - t0 > time_limit:
                break

        full_pass += 1
        if not improved_this_pass:
            break

    return f, best_M


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def aggressive_peak_chase(f_init, time_limit, label="apc"):
    """
    Aggressively target specific peaks: iteratively solve LP with
    decreasing M threshold to force peak reduction.
    """
    t0 = time.monotonic()
    n = len(f_init)
    total = 2 * n - 1
    f = f_init.copy()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Focus on the single worst peak
        k_max = int(np.argmax(b))
        # Also include nearby peaks
        half_width = max(1, n // 20)
        lo = max(0, k_max - half_width)
        hi = min(total, k_max + half_width + 1)
        peaks = np.arange(lo, hi)

        # Try progressive M reduction
        improved = False
        for frac in [0.999, 0.998, 0.995, 0.99, 0.98, 0.95]:
            M_try = M * frac
            g, S_g = solve_lp(f, M_try, peak_indices=peaks)
            if g is not None and S_g > 1.0 + 1e-4:
                d = g / S_g
                best_t, best_t_obj = line_search(f, d, M, n_evals=15)
                if best_t > 0:
                    f_new = (1.0 - best_t) * f + best_t * d
                    f_new = np.maximum(f_new, 0.0)
                    s_new = f_new.sum()
                    if s_new > 1e-10:
                        f_new = f_new / s_new
                        m_new = fast_convolve(f_new).max()
                        if m_new < best_M - 1e-14:
                            best_M = m_new
                            best_f = f_new.copy()
                            f = f_new
                            improved = True
                            break

        if not improved:
            # Perturb
            f = best_f + rng.standard_normal(n) * 0.001 * best_f
            f = np.maximum(f, 1e-15)
            f = f / f.sum()

        if it % 30 == 0:
            score = 2.0 * n * best_M
            print(f"    {label} iter={it} M={best_M:.10f} score={score:.6f} "
                  f"t={time.monotonic()-t0:.0f}s", flush=True)

        if it % 50 == 0:
            f = best_f.copy()

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-phase LP-based search with all-peak constraints,
    golden-section line search, multi-start diversity, aggressive peak chasing,
    mirror descent, subgradient, and coordinate descent polishing.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = 2.0
        print(f"No parent, using uniform n=1024, score={p_score:.6f}", flush=True)

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Main LP with all peaks (longest phase) ===
    n_parent = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.38, 650.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP all-peaks (n={n_parent}, budget={t1:.0f}s) ===", flush=True)
        f1, obj1 = run_lp_optimize(parent, time_limit=t1, label="lp1",
                                    use_all_peaks=True, patience=35)
        score1 = 2.0 * n_parent * obj1
        vs1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(vs1) and vs1 < best_score:
            best_score = vs1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp1]: {vs1:.6f}", flush=True)

    # === Phase 2: LP with partial peaks (more aggressive) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 300.0)
    if t2 > 40:
        n_b = len(best_seq)
        print(f"\n=== Phase 2: LP partial peaks (n={n_b}, budget={t2:.0f}s) ===", flush=True)
        f2, obj2 = run_lp_optimize(best_seq, time_limit=t2, label="lp2",
                                    use_all_peaks=False,
                                    peak_frac_start=0.85, peak_frac_end=0.65,
                                    max_peaks_partial=400, patience=25)
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp2]: {vs2:.6f}", flush=True)

    # === Phase 3: Aggressive peak chase ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 180.0)
    if t3 > 30:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: Aggressive peak chase (n={n_b}, budget={t3:.0f}s) ===", flush=True)
        f3, obj3 = aggressive_peak_chase(best_seq, time_limit=t3, label="apc")
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [apc]: {vs3:.6f}", flush=True)

    # === Phase 4: Mirror descent polish ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 25:
        n_b = len(best_seq)
        print(f"\n=== Phase 4: Mirror descent (n={n_b}, budget={t4:.0f}s) ===", flush=True)
        f4, obj4 = mirror_descent_optimize(best_seq, time_limit=t4, lr_init=0.04,
                                            n_top=120, label="md")
        vs4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  NEW BEST [md]: {vs4:.6f}", flush=True)

    # === Phase 5: Multi-start LP ===
    rem = deadline - time.monotonic()
    n_b = len(best_seq)
    rng = np.random.default_rng(2024)
    n_starts = 3
    t_each = rem * 0.10 / n_starts
    t_each = min(t_each, 80.0)
    if t_each > 20:
        print(f"\n=== Phase 5: Multi-start ({n_starts} starts, {t_each:.0f}s each) ===", flush=True)
        for si in range(n_starts):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t_s = min(rem * 0.2, t_each)
            if t_s < 20:
                break

            # Create perturbation
            if si == 0:
                perturbed = best_seq + rng.standard_normal(n_b) * 0.01 * best_seq
            elif si == 1:
                # Perturb valleys more
                threshold = np.percentile(best_seq, 30)
                mask = best_seq < threshold
                perturbed = best_seq.copy()
                perturbed[mask] += rng.standard_normal(mask.sum()) * 0.03 * best_seq[mask].mean() + 0.005
            else:
                # Interpolate to different length and back
                new_n = n_b + rng.integers(-100, 100)
                new_n = max(64, min(2048, new_n))
                perturbed = interpolate_to_length(best_seq, new_n)
                perturbed = interpolate_to_length(perturbed, n_b)

            perturbed = np.maximum(perturbed, 1e-15)
            perturbed = perturbed / perturbed.sum()

            f5, obj5 = run_lp_optimize(perturbed, time_limit=t_s,
                                        label=f"ms{si}", use_all_peaks=True,
                                        peak_frac_start=0.90, patience=20)
            vs5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(vs5) and vs5 < best_score:
                best_score = vs5
                best_seq = f5.copy()
                print(f"  NEW BEST [ms{si}]: {vs5:.6f}", flush=True)

    # === Phase 6: Subgradient polish ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 100.0)
    if t6 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 6: Subgradient (n={n_b}, budget={t6:.0f}s) ===", flush=True)
        f6, obj6 = subgradient_optimize(best_seq, time_limit=t6, lr_init=0.003,
                                         n_top=80, label="sg")
        vs6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(vs6) and vs6 < best_score:
            best_score = vs6
            best_seq = f6.copy()
            print(f"  NEW BEST [sg]: {vs6:.6f}", flush=True)

    # === Phase 7: Coordinate descent ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.10, 80.0)
    if t7 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 7: Coordinate descent (budget={t7:.0f}s) ===", flush=True)
        f7, obj7 = coordinate_descent(best_seq, time_limit=t7, label="cd")
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {vs7:.6f}", flush=True)

    # === Phase 8: Final LP burst ===
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.25, 150.0)
    if t8 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 8: Final LP (n={n_b}, budget={t8:.0f}s) ===", flush=True)
        f8, obj8 = run_lp_optimize(best_seq, time_limit=t8, label="lp_final",
                                    use_all_peaks=True,
                                    peak_frac_start=0.88, peak_frac_end=0.75,
                                    patience=30)
        vs8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(vs8) and vs8 < best_score:
            best_score = vs8
            best_seq = f8.copy()
            print(f"  NEW BEST [lp_final]: {vs8:.6f}", flush=True)

    # === Phase 9: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t9 = min(rem * 0.15, 60.0)
    if t9 > 10:
        print(f"\n=== Phase 9: Final CD (budget={t9:.0f}s) ===", flush=True)
        f9, obj9 = coordinate_descent(best_seq, time_limit=t9, label="cd_final")
        vs9 = float(evaluate_sequence(f9.tolist()))
        if np.isfinite(vs9) and vs9 < best_score:
            best_score = vs9
            best_seq = f9.copy()
            print(f"  NEW BEST [cd_final]: {vs9:.6f}", flush=True)

    # === Phase 10: Final aggressive peak chase if time remains ===
    rem = deadline - time.monotonic()
    t10 = min(rem * 0.30, 80.0)
    if t10 > 15:
        print(f"\n=== Phase 10: Final APC (budget={t10:.0f}s) ===", flush=True)
        f10, obj10 = aggressive_peak_chase(best_seq, time_limit=t10, label="apc_final")
        vs10 = float(evaluate_sequence(f10.tolist()))
        if np.isfinite(vs10) and vs10 < best_score:
            best_score = vs10
            best_seq = f10.copy()
            print(f"  NEW BEST [apc_final]: {vs10:.6f}", flush=True)

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in best_seq]
