"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (log-sum-exp) with FFT-based O(n log n) gradients
2. Fast coordinate descent with sensitivity analysis
3. LP-based fictitious play for global improvements
4. Multi-length, multi-initialization exploration
5. Adaptive time allocation and peak-shaving heuristics

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_weights(conv, epsilon):
    """Compute softmax weights for the smoothed max."""
    c_max = conv.max()
    w = np.exp((conv - c_max) / epsilon)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / s
    return w


def compute_smoothed_gradient(a, epsilon):
    """Compute gradient of smoothed max of auto-convolution using FFT."""
    n = len(a)
    conv = fast_convolve(a)
    w = softmax_weights(conv, epsilon)
    
    # grad[i] = 4*n * sum_j w[i+j] * a[j]
    # Compute via cross-correlation using FFT
    a_rev = a[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    corr = irfft(W * Ar, fft_size)[:size]
    grad = 4.0 * n * corr[n-1 : 2*n-1]
    
    return grad, conv

    if s < 1e-15:

        a = a / s

        epsilon_init = 0.05 / n

    v = np.zeros(n)

                a[i] = max(a[i] + best_delta, 0.0)
                
                changes = 2.0 * best_delta * a_seg
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD [{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small steps for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())
            
            affected = conv[k_start:k_end + 1]
            a_seg = a[0:n]
            
            best_delta = 0.0
            best_new_M = M
            
            for factor in [0.97, 0.94, 0.91, 0.88, 0.985, 1.015, 1.03, 1.06, 1.09, 1.12]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * a_seg
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_affected = new_affected.copy()
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-17:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                
                changes = 2.0 * best_delta * a_seg
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    
    return best_a, best_score


def peak_shave(a0, top_k=10, magnitude=0.02, rng=None):
    """Reduce elements that contribute most to the top peaks."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    
    # Find top peaks
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    
    # For each peak, reduce the contributing elements
    for k in peak_indices:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # The contribution of a[i] to conv[k] is a[i]*a[k-i]
        # We want to reduce a[i] where a[i]*a[k-i] is large
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k - idx]
        # Reduce the top contributors
        n_reduce = max(1, len(idx) // 5)
        top_contrib = np.argsort(contrib)[-n_reduce:]
        for j in top_contrib:
            a[idx[j]] *= (1.0 - magnitude * (1.0 + 0.5 * rng.random()))
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def perturb(a0, rng, sigma=0.05):
    """Add random multiplicative noise."""
    n = len(a0)
    a = a0 * np.exp(rng.standard_normal(n) * sigma)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize with random Fourier superposition."""
    n_harmonics = int(rng.integers(4, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.permutation(np.arange(1, n_harmonics + 1))
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_multimodal(n, rng):
    """Initialize with multiple Gaussian bumps."""
    n_bumps = int(rng.integers(2, 6))
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = rng.integers(0, n)
        width = n * rng.uniform(0.1, 0.3)
        amp = rng.uniform(0.5, 1.0)
        t = np.arange(n, dtype=np.float64)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_periodic_jitter(n, rng):
    """Initialize with a periodic pattern plus jitter."""
    period = int(rng.integers(3, max(4, n // 10)))
    t = np.arange(n)
    # Create a periodic pattern
    base = np.ones(n) * 0.5
    for j in range(min(period, n)):
        base[j::period] = 0.5 + rng.uniform(-0.3, 0.3)
    # Add jitter
    a = base * np.exp(rng.standard_normal(n) * 0.2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP, cp.GLPK, cp.MOSEK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver=None):
    """Find b minimizing max_k conv(a_fixed, b)[k] subject to b>=0, sum(b)=1."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution constraint matrix
    # conv(a, b)[k] = sum_j a[j] * b[k-j]
    # For each k, this is a dot product of a "row" with b
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

Actually, I realize I should think about this more carefully. Let me consider what happens when we have a "typical" sequence (roughly uniform with some structure).

For a roughly uniform sequence of length n:
- `a[i] ≈ 1/n`
- `(a*a)[k]` is triangular with peak `1/n` at the center
- The max is `1/n`
- The second max is approximately `(n-1)/n^2 ≈ 1/n - 1/n^2`
- The gap between max and second max is approximately `1/n^2`

So for the softmax to distinguish the max from the rest, we need `epsilon << 1/n^2`. For n=1024, that's `epsilon << 1e-6`.

But if epsilon is that small, the softmax is essentially a one-hot vector at the argmax, and the gradient is just the subgradient at the single peak. This makes the optimization equivalent to subgradient descent on the (nonsmooth) max function, which converges slowly.

On the other hand, if epsilon is larger (say `epsilon = 1/n`), the softmax spreads over many peaks, giving a smoother gradient. But the smoothed objective is then a poor approximation of the true max.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (log-sum-exp) with FFT-based O(n log n) gradients + Adam
2. Fast coordinate descent with O(n) sensitivity analysis per coordinate
3. LP-based fictitious play for global improvements
4. Multi-length, multi-initialization exploration with peak shaving
5. Adaptive temperature scheduling and simulated-annealing restarts

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_weights(conv, epsilon):
    """Compute softmax weights for the smoothed max."""
    c_max = conv.max()
    w = np.exp((conv - c_max) / epsilon)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / s
    return w


def compute_smoothed_gradient(a, epsilon):
    """Compute gradient of smoothed max of auto-convolution using FFT cross-correlation."""
    n = len(a)
    conv = fast_convolve(a)
    w = softmax_weights(conv, epsilon)
    # grad[i] = 4*n * sum_j w[i+j] * a[j]  (cross-correlation)
    a_rev = a[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    corr = irfft(W * Ar, fft_size)[:size]
    grad = 4.0 * n * corr[n - 1:2 * n - 1]
    return grad, conv


def smooth_gd(a0, time_limit, epsilon_init=None, lr_init=0.08, verbose=False, label=""):
    """Smoothed gradient descent with Adam for minimizing max auto-conv."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if epsilon_init is None:
        epsilon_init = 1.0 / n

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()

    t0 = time.monotonic()
    it = 0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    lr = lr_init / n
    epsilon = epsilon_init
    no_improve = 0
    rng_local = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, conv = compute_smoothed_gradient(a, epsilon)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-10)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        # Adapt parameters
        if it % 40 == 0:
            epsilon *= 0.97
        if no_improve > 300:
            lr *= 0.7
            epsilon = max(epsilon * 0.7, 0.005 / n)
            no_improve = 0
            if lr < 0.005 / n:
                # Restart with noise
                a = best_a + 0.02 * rng_local.standard_normal(n)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init / n
                epsilon = epsilon_init
                no_improve = 0
                if best_score < 1.503:
                    break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGD [{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} eps={epsilon:.8f}")

    return best_a, best_score


def fast_coord_descent(a0, time_limit, n_trials=8, verbose=False, label=""):
    """Fast coordinate descent with O(n) sensitivity per coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            a_seg = a[0:n]

            best_delta = 0.0
            best_new_M = M

            for t in range(n_trials):
                if t == 0:
                    delta = -0.5 * a[i]
                elif t == 1:
                    delta = -0.3 * a[i]
                elif t == 2:
                    delta = -0.15 * a[i]
                elif t == 3:
                    delta = -0.05 * a[i]
                elif t == 4:
                    delta = 0.05 * a[i]
                elif t == 5:
                    delta = 0.15 * a[i]
                elif t == 6:
                    delta = 0.3 * a[i]
                else:
                    delta = 0.5 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_seg
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)

                changes = 2.0 * best_delta * a_seg
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 40:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD [{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def peak_shave(a0, top_k=10, magnitude=0.02, rng=None):
    """Reduce elements contributing most to top peaks."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]

    for k in peak_indices:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        if len(idx) == 0:
            continue
        contrib = a[idx] * a[k - idx]
        n_reduce = max(1, len(idx) // 5)
        top_contrib = np.argsort(contrib)[-n_reduce:]
        for j in top_contrib:
            a[idx[j]] *= (1.0 - magnitude * (1.0 + 0.5 * rng.random()))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def perturb(a0, rng, sigma=0.05):
    """Add random multiplicative noise."""
    n = len(a0)
    a = a0 * np.exp(rng.standard_normal(n) * sigma)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize with random Fourier superposition."""
    n_harmonics = int(rng.integers(4, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.permutation(np.arange(1, n_harmonics + 1))
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_multimodal(n, rng):
