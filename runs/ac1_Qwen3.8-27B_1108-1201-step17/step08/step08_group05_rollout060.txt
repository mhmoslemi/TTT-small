Improved optimization for the auto-convolution peak problem.
Uses LP-based descent with constraint generation, smoothed gradient descent
with adaptive temperature, and coordinate descent for fine-tuning.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, n, M, active_set, max_constraints=200):
    """Solve the LP to get the descent direction."""
    K = len(active_set)
    if K == 0:
        return None
    
    # Keep only top constraints if too many
    if K > max_constraints:
        # Prioritize constraints with highest violation potential
        active_set = list(active_set[:max_constraints])
        K = max_constraints
    
    A = np.zeros((K, n))
    for row_i, k in enumerate(active_set):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )
    
    g_val = None
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=200000)
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                g_val = np.maximum(g_var.value, 0.0)
                break
        except Exception:
            continue
    
    return g_val

            best_f = f.copy()
            no_improve = 0

            if no_improve > 50:

                no_improve = 0
            if no_improve > 200:
                break
        
        # Compute softmax weights
        scaled = tau * b - tau * obj  # shift for numerical stability
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        
        # Find significant weight positions (top few)
        n_significant = max(5, min(50, int(50 * (1 - 0.9999 / max(w.max(), 1e-10)))))
        top_idx = np.argsort(w)[-n_significant:]
        w_top = w[top_idx].copy()
        w_top = w_top / w_top.sum()
        
        # Compute gradient: grad[m] = sum_k w[k] * (a[k-m] + a[m-k]) with boundary handling
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):

            k = int(k)

            lo2 = max(0, k)

            grad[lo1:hi1+1] += wk * f[k - np.arange(lo1, hi1+1)]  # a[k-m]
            grad[lo2:hi2+1] += wk * f[np.arange(lo2, hi2+1) - k]  # a[m-k]

        f_new = project_simplex(f + v)

        # Adaptive learning rate

        # Cool down temperature

def subgradient_optimize(f_init, time_limit, tau_start=20.0, lr_start=0.003):
    """Subgradient method with adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_start
    tau = tau_start
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        cur_peak = b.max()
        
        if cur_peak < best_peak - 1e-13:
            best_peak = cur_peak
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
        
        # Softmax weights (temperature-controlled)
        scaled = tau * (b - cur_peak)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[int(b.argmax())] = 1.0
        else:
            w /= w_sum
        
        # Only use top-weight positions for efficiency
        n_top = max(5, min(30, len(b)))
        top_idx = np.argsort(w)[-n_top:]
        w_sub = w[top_idx].copy()
        w_sub_sum = w_sub.sum()
        if w_sub_sum > 1e-30:
            w_sub /= w_sub_sum
        else:
            w_sub = np.ones(n_top) / n_top
        
        # Compute gradient
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            wk = w_sub[i]
            if wk < 1e-12:
                continue
            k = int(k)
            # Contribution from a[k-m]
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            m_range = np.arange(lo1, hi1 + 1)
            grad[m_range] += wk * f[k - m_range]
            # Contribution from a[m-k]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            m_range2 = np.arange(lo2, hi2 + 1)
            grad[m_range2] += wk * f[m_range2 - k]
        
        # Nesterov-style update
        v = 0.9 * v - lr * grad
        f_new = project_simplex(f + v)
        f = f_new
        
        # Adapt parameters
        if no_improve == 0:
            lr = min(lr * 1.01, 0.05)
        else:
            lr = max(lr * 0.95, 0.0001)
            tau = min(tau * 1.05, 100.0)
    
    return best_f, best_peak


def coordinate_refine(f_init, time_limit, n_trials_per_round=50, max_rounds=200):
    """Randomized coordinate descent for fine-tuning."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    rng = np.random.default_rng(42)
    
    round_num = 0
    while time.monotonic() - t0 < time_limit and round_num < max_rounds:
        round_num += 1
        improved = False
        
        for _ in range(n_trials_per_round):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-12:
                continue
            
            # Try transferring
            for eps in [5e-5, 2e-5, 1e-5, 5e-6]:
                if f[i] < eps:
                    continue
                f_trial = f.copy()
                f_trial[i] -= eps
                f_trial[j] += eps
                peak = fast_convolve(f_trial).max()
                if peak < best_peak - 1e-15:
                    f = f_trial
                    best_peak = peak
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved and round_num > 10:
            # Try a different strategy: perturb and project
            f_trial = f + rng.standard_normal(n) * 1e-4
            f_trial = project_simplex(f_trial)
            peak = fast_convolve(f_trial).max()
            if peak < best_peak - 1e-15:
                f = f_trial
                best_peak = peak
                best_f = f.copy()
            else:
                # Restart from best with slightly larger perturbation
                if round_num > 30:
                    break
    
    return best_f, best_peak

"""
Precision optimization for auto-convolution peak minimization using LP-based descent
with dual variable extraction, adaptive constraint management, weighted subgradient
refinement, and coordinate polish. Targets score < 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, n, active_indices):
    """Build constraint matrix for LP."""
    K = len(active_indices)
    A = np.zeros((K, n))
    for row_i, k in enumerate(active_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_indices):
    """Solve LP: max sum(g) s.t. g>=0, A@g <= M. Returns (g, dual_vars)."""
    K = len(active_indices)
    if K == 0:
        return None, None
    A = build_A(f, n, active_indices)
    g_var = cp.Variable(n)
    y_var = cp.Variable(K)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )
    # Track dual via constraint
    con_g = g_var >= 0
    con_A = A @ g_var <= M
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [con_g, con_A])

    g_val = None
    y_val = None
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None:
                g_candidate = np.maximum(g_var.value, 0.0)
                if not np.any(np.isnan(g_candidate)) and not np.any(np.isinf(g_candidate)):
                    g_val = g_candidate
                    try:
                        y_val = con_A.dual_value
                        if y_val is not None:
                            y_val = np.maximum(np.asarray(y_val, dtype=float), 0.0)
                        else:
                            y_val = None
                    except Exception:
                        y_val = None
                    break
        except Exception:
            continue
    return g_val, y_val


def line_search(f, direction, max_evals=35):
    """Golden section line search on the peak objective."""
    n = len(f)

    def obj(t):
        ft = (1.0 - t) * f + t * direction
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-10:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()

    cands = [0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5]
    best_t = 0.05
    best_v = obj(0.05)
    for t in cands:
        v = obj(t)
        if v < best_v:
            best_v = v
            best_t = t

    idx = cands.index(best_t)
    lo = cands[max(0, idx - 1)]
    hi = cands[min(len(cands) - 1, idx + 1)]
    if hi <= lo:
        return best_t

    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = obj(c), obj(d)
    for _ in range(25):
        if hi - lo < 5e-7:
            break
        if fc <= fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = obj(d)

def dual_guided_descent(f_init, time_limit, n_dual=5):
    """Use LP dual variables to construct a weighted descent direction."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()

        else:
            active = np.arange(len(b))

            direction[m1] -= w * f[k - m1]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            m2 = np.arange(lo2, hi2 + 1)
            direction[m2] -= w * f[m2 - k]

        # Try multiple step sizes
        best_t = 0.0
        best_val = best_peak
        for step in [0.001, 0.003, 0.005, 0.01, 0.02]:
            f_trial = project_simplex(f + step * direction)
            pk = fast_convolve(f_trial).max()
            if pk < best_val:
                best_val = pk
                best_t = step

        if best_t > 0:
            f_new = project_simplex(f + best_t * direction)
            f = f_new
            if best_val < best_peak:
                best_peak = best_val
                best_f = f_new.copy()
        else:
            if it > 20:
                break

        if it % 50 == 0:
            print(f"    Dual iter={it} peak={best_peak:.12f} score={2*n*best_peak:.8f} t={elapsed:.0f}s")

    return best_f, best_peak


def subgradient_phase(f_init, time_limit, tau_init=20.0, lr_init=0.002):
    """Subgradient with softmax weighting and momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    tau = tau_init
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    no_imp = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        cur_peak = b.max()

        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 80:
                break

        # Softmax weights
        scaled = tau * (b - cur_peak)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(b))
            w[int(b.argmax())] = 1.0
        else:
            w /= ws

        # Top positions for efficiency
        n_top = max(8, min(40, len(b)))
        top_idx = np.argsort(w)[-n_top:]
        w_sub = w[top_idx].copy()
        wss = w_sub.sum()
        if wss > 1e-30:
            w_sub /= wss
        else:
            w_sub = np.ones(n_top) / n_top

        # Gradient
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            wk = w_sub[i]
            if wk < 1e-12:
                continue
            k = int(k)
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            m1 = np.arange(lo1, hi1 + 1)
            grad[m1] += wk * f[k - m1]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            m2 = np.arange(lo2, hi2 + 1)
            grad[m2] += wk * f[m2 - k]

        v = 0.88 * v - lr * grad
        f = project_simplex(f + v)

        if no_imp == 0:
            lr = min(lr * 1.015, 0.03)
        else:
            lr = max(lr * 0.96, 0.0002)
            tau = min(tau * 1.03, 80.0)

    return best_f, best_peak


def coord_refine(f_init, time_limit, rng_seed=99):
    """Randomized coordinate descent for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    rng = np.random.default_rng(rng_seed)
    trials = 0

    while time.monotonic() - t0 < time_limit:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or f[i] < 1e-10:
            continue

        found = False
        for eps in [3e-5, 1e-5, 5e-6, 2e-6, 1e-6]:
            if f[i] < eps:
                break
            ft = f.copy()
            ft[i] -= eps
            ft[j] += eps
            pk = fast_convolve(ft).max()
            if pk < best_peak - 1e-16:
                f = ft
                best_peak = pk
                best_f = f.copy()
                found = True
                trials += 1
                break

        if not found and trials % 200 == 0 and trials > 0:
            # Random perturbation from best
            ft = best_f + rng.standard_normal(n) * 5e-5
            ft = project_simplex(ft)
            pk = fast_convolve(ft).max()
            if pk < best_peak - 1e-16:
                f = ft
                best_peak = pk
                best_f = f.copy()
        elif not found and trials > 500:
            # Try larger perturbation
            ft = best_f + rng.standard_normal(n) * 2e-4
            ft = project_simplex(ft)
            pk = fast_convolve(ft).max()
            if pk < best_peak - 1e-16:
                f = ft
                best_peak = pk
                best_f = f.copy()
            else:
                break

    return best_f, best_peak
