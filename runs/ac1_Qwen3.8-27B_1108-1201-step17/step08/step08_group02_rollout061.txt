```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2*n-1]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_conv(a, tau):
    """Compute F_tau(a) = (1/tau) * log(sum exp(tau * conv(a,a))) and its gradient."""
    n = len(a)
    c = fast_convolve(a)
    # Numerical stability
    c_max = c.max()
    exp_tau_c = np.exp(tau * (c - c_max))
    log_sum = np.log(exp_tau_c.sum()) + tau * c_max
    value = log_sum / tau  # This is (1/tau) * log(sum exp(tau*c))
    
    # Weights
    w = exp_tau_c / exp_tau_c.sum()
    
    # Gradient: grad[i] = 2 * X[i] - w[2i] * a[i]
    # where X[i] = sum_k w[i+k] * a[k]
    size = 1 << (3 * n).bit_length()
    W = rfft(w, size)
    A = rfft(a, size)
    X = irfft(W * np.conj(A), size)[:n]
    
    grad = 2.0 * X - w[0:2*n-1:2] * a  # w[0::2] has length n (since 2n-1 elements, indices 0,2,...,2n-2)
    
    return value, grad
```

"""
Optimized non-convex sequence search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth FISTA + LP-based cutting plane (PDLP solver) + coordinate descent + multi-length exploration.
Key improvements: PDLP solver for speed, per-peak LP bounds, adaptive peak selection,
smooth gradient warm-starting, and systematic n exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2 * n - 1]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0) if v[0] > 0 else target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    return result


def smooth_objective_and_grad(a, tau):
    """
    Compute F_tau(a) = (1/tau)*log(sum exp(tau*conv(a,a))) and its gradient.
    grad[i] = 2*X[i] - w[2i]*a[i] where X = cross_corr(w, a).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_vals = np.exp(tau * (c - c_max))
    s = exp_vals.sum()
    value = (np.log(s) + tau * c_max) / tau
    w = exp_vals / s
    # X[i] = sum_k w[i+k] * a[k] = cross_correlation(w, a)[i]
    size = 1 << (3 * n).bit_length()
    W = rfft(w, size)
    A = rfft(a, size)
    X = irfft(W * np.conj(A), size)[:n]
    grad = 2.0 * X - w[0:2 * n - 1:2] * a
    return value, grad


def fista_smooth(a_init, tau, n_iter, lr_init=None, label=""):
    """FISTA on smooth approximation with simplex projection."""
    n = len(a_init)
    a = a_init.copy()
    a_kink = a.copy()
    t_k = 1.0
    best_val = smooth_objective_and_grad(a, tau)[0]
    best_a = a.copy()

    if lr_init is None:
        lr_init = 1.0 / (tau * 2.0 * max(a.max() * n, 1e-6))
        lr_init = min(lr_init, 0.5)

    for k in range(n_iter):
        val, grad = smooth_objective_and_grad(a, tau)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a.copy()

        y = a_kink + (t_k - 1) / (t_k + 1) * (a - a_kink)
        a_kink = a
        a_new = project_simplex(y - lr_init * grad)
        a = a_new
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        t_k = t_new

    return best_a


def solve_lp_direction(f, n, peak_indices, bounds, solver_order=None):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= bounds[k] for k in peak_indices.
    bounds: array of per-peak bounds (length = len(peak_indices)).
    """
    if solver_order is None:
        solver_order = [cp.PDLP, cp.SCS, cp.ECOS, cp.CBC]

    K = len(peak_indices)
    if K == 0:
        return None

    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= bounds]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in solver_order:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)):
                    return g_val
        except Exception:
            continue
    return None


def line_search(f, h, current_max, num_trials=20):
    """
    Find best t in (0,1] such that max(conv((1-t)*f + t*h_normalized)) is minimized.
    h_normalized = h / sum(h).
    """
    h_norm = h / h.sum() if h.sum() > 1e-15 else None
    if h_norm is None:
        return 0.0, current_max

    best_t = 0.0
    best_obj = current_max

    # Try a range of t values
    t_vals = np.concatenate([
        np.linspace(0.01, 0.3, num_trials),
        np.array([0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 0.95])
    ])
    for t in t_vals:
        f_trial = (1.0 - t) * f + t * h_norm
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t

    # Refine around best
    if best_t > 0:
        for dt in [best_t * 0.7, best_t * 0.9, best_t * 1.1, best_t * 1.3]:
            if dt <= 0 or dt > 0.99:
                continue
            f_trial = (1.0 - dt) * f + dt * h_norm
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_t = dt

    return best_t, best_obj


def lp_optimize(f_init, time_budget, label="lp", peak_frac_start=0.95,
                peak_frac_end=0.80, max_peaks=400, patience=30,
                use_per_peak_bounds=True):
    """
    LP-based optimization with per-peak bounds and adaptive peak selection.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s

    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = peak_frac_start
    rng = np.random.default_rng(12345)
    solver_order = [cp.PDLP, cp.SCS, cp.ECOS]

    while time.monotonic() - t0 < time_budget:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Peak selection: peaks above threshold
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            # Keep the largest peaks
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 3:
            peaks = np.argpartition(b, -max(10, len(b) // 20))[-max(10, len(b) // 20):]

        # Per-peak bounds
        if use_per_peak_bounds:
            bounds = b[peaks]
        else:
            bounds = np.full(len(peaks), M)

        g = solve_lp_direction(f, n, peaks, bounds, solver_order=solver_order)

        if g is None:
            stuck += 1
            if stuck > patience:
                # Try with fewer peaks
                if len(peaks) > 20:
                    peaks_sub = np.argpartition(b, -20)[-20:]
                    bounds_sub = b[peaks_sub]
                    g = solve_lp_direction(f, n, peaks_sub, bounds_sub, solver_order=solver_order)
                    if g is not None:
                        peaks = peaks_sub
                        stuck = 0
                    else:
                        continue
                else:
                    continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > patience:
                # Perturb and restart
                f = f + rng.standard_normal(n) * 0.001 * f
                f = np.maximum(f, 0.0)
                f /= f.sum()
                stuck = 0
            continue

        h = g / S_g
        best_t, best_t_obj = line_search(f, h, M)

        if best_t <= 0:
            stuck += 1
            if stuck > patience:
                f = f + rng.standard_normal(n) * 0.002 * f
                f = np.maximum(f, 0.0)
                f /= f.sum()
                stuck = 0
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            break

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_b
            print(f"    {label} it={it} obj={best_b:.10f} score={score:.6f} "
                  f"peaks={len(peaks)} t={elapsed:.0f}s")

        # Gradually lower peak fraction
        if it % 25 == 0:
            peak_frac = max(peak_frac_end, peak_frac - 0.008)

    return best_f, best_b

def coordinate_descent(f_init, time_budget, label="cd"):
    """
    Coordinate descent: try scaling each element by various multipliers.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_budget:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = fast_convolve(f).max()
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if abs(best_j_val - f[j]) > 1e-15:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                cur = fast_convolve(f).max()
                if cur < best_b - 1e-13:
                    best_b = cur
                    improved_global = True

    return f, best_b


def interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    # === Phase 2: LP-based refinement (main workhorse) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.40, 600.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"\n{'='*50}")
        print(f"Phase 2: LP optimization (n={n_b}, budget={t2:.0f}s)")
        print(f"{'='*50}")
        f2, obj2 = lp_optimize(best_seq, time_budget=t2, label="lp_main",
                                peak_frac_start=0.94, peak_frac_end=0.78,
                                max_peaks=500, patience=25, use_per_peak_bounds=True)
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp_main]: {vs2:.6f}")

    # === Phase 3: FISTA polish on LP result ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 150.0)
    if t3 > 20:
        n_b = len(best_seq)
        print(f"\n{'='*50}")
        print(f"Phase 3: FISTA polish (n={n_b}, budget={t3:.0f}s)")
        print(f"{'='*50}")
        f3, obj3 = smooth_fista_phase(best_seq, time_budget=t3, label="fista2")
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [fista2]: {vs3:.6f}")

    # === Phase 4: Second LP pass with different parameters ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 300.0)
    if t4 > 30:
        n_b = len(best_seq)
        print(f"\n{'='*50}")
        print(f"Phase 4: LP deep (n={n_b}, budget={t4:.0f}s)")
        print(f"{'='*50}")
        f4, obj4 = lp_optimize(best_seq, time_budget=t4, label="lp_deep",
                                peak_frac_start=0.90, peak_frac_end=0.70,
                                max_peaks=600, patience=30, use_per_peak_bounds=True)
        vs4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  NEW BEST [lp_deep]: {vs4:.6f}")

    # === Phase 5: Multi-length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for delta in [-64, -32, -16, 16, 32, 64, 128, 256]:
        tl = n_base + delta
        if 512 <= tl <= 3000:
            target_lengths.append(tl)
    # Sort by distance from n_base
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t5 = min(rem * 0.15, 150.0)
        if t5 < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        print(f"\n{'='*50}")
        print(f"Phase 5: Multi-n LP (n={target_n}, budget={t5:.0f}s)")
        print(f"{'='*50}")
        f5, obj5 = lp_optimize(f_interp, time_budget=t5, label=f"lp_n{target_n}",
                                peak_frac_start=0.93, peak_frac_end=0.80,
                                max_peaks=400, patience=20, use_per_peak_bounds=True)
        vs5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(vs5) and vs5 < best_score:
            best_score = vs5
            best_seq = f5.copy()
            print(f"  NEW BEST [n={target_n}]: {vs5:.6f}")
        else:
            print(f"  n={target_n}: score={vs5:.6f} (no improvement)")

    # === Phase 6: Coordinate descent ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 120.0)
    if t6 > 10:
        n_b = len(best_seq)
        print(f"\n{'='*50}")
        print(f"Phase 6: Coordinate descent (n={n_b}, budget={t6:.0f}s)")
        print(f"{'='*50}")
        f6, obj6 = coordinate_descent(best_seq, time_budget=t6, label="cd")
        vs6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(vs6) and vs6 < best_score:
            best_score = vs6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {vs6:.6f}")

    # === Phase 7: Final short LP burst ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.30, 180.0)
    if t7 > 10:
        n_b = len(best_seq)
        print(f"\n{'='*50}")
        print(f"Phase 7: Final LP (n={n_b}, budget={t7:.0f}s)")
        print(f"{'='*50}")
        f7, obj7 = lp_optimize(best_seq, time_budget=t7, label="lp_final",
                                peak_frac_start=0.88, peak_frac_end=0.65,
                                max_peaks=700, patience=20, use_per_peak_bounds=True)
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp_final]: {vs7:.6f}")

    # === Phase 8: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.20, 60.0)
    if t8 > 5:
        n_b = len(best_seq)
        print(f"\n{'='*50}")
        print(f"Phase 8: Final CD (n={n_b}, budget={t8:.0f}s)")
        print(f"{'='*50}")
        f8, obj8 = coordinate_descent(best_seq, time_budget=t8, label="cd_final")
        vs8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(vs8) and vs8 < best_score:
            best_score = vs8
            best_seq = f8.copy()
            print(f"  NEW BEST [cd_final]: {vs8:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
