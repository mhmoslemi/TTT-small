```python
"""
Enhanced LP fictitious play with adaptive steps, diverse multi-start,
and gradient polishing for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    # C @ g = convolve(f, g, 'full') has shape 2n-1
    # C is (2n-1) x n Toeplitz
    # First column: f[0], f[1], ..., f[n-1], 0, ..., 0
    # First row: f[0], 0, 0, ..., 0
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    # Try PDLP first (fast for large sparse LPs), then CLARABEL, then ECOS
    for solver in [cp.PDLP, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver, warm_start=(prev_g is not None))
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return prev_g if prev_g is not None else f.copy()
```

```python
Enhanced optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP fictitious play with adaptive steps, diverse multi-start,
and gradient-based polishing.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_circ(a, n):
    """Circular auto-convolution via FFT."""
    fft_size = 1 << max(n - 1, 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:n]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n Toeplitz matrix C such that C @ g = convolve(f, g, 'full')."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None, solver_list=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    if solver_list is None:
        solver_list = [cp.PDLP, cp.CLARABEL, cp.SCS]
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = np.maximum(prev_g, 0)
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    for solver in solver_list:
        try:
            kwargs = {}
            if prev_g is not None and solver != cp.SCS:
                kwargs['warm_start'] = True
            prob.solve(solver=solver, **kwargs)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    
    # Fallback: return proportional to f
    return np.maximum(f, 1e-10)

def lse_objective_and_grad(a, T, n):
    """
    Compute the log-sum-exp smoothed objective and gradient.
    Objective: 2n * T * log(sum(exp(conv_i / T)))
    where conv = circular_conv(a, a)
    """
    conv_circ = fast_convolve_circ(a, n)
    shifted = conv_circ / T
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    lse = max_shifted + np.log(exp_shifted.sum())
    obj_val = 2.0 * n * T * lse
    
    # Gradient
    p = exp_shifted / exp_shifted.sum()  # softmax weights
    fft_size = n
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    grad_a = 4.0 * n * irfft(P * np.conj(A), fft_size)
    
    # Project to tangent space of {a >= 0, sum(a) = 1}
    S = float(np.dot(grad_a, a))
    grad_a = grad_a - S
    
    return obj_val, grad_a


def lbgfs_polish(a0, time_limit, T=100.0, maxiter=5000, verbose=True):
    """L-BFGS optimization on the smoothed log-sum-exp objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Work in log-space to maintain positivity
    z0 = np.log(np.maximum(a, 1e-30))
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros_like(z)
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        obj_val, grad_a = lse_objective_and_grad(a_tmp, T, n)
        # Chain rule: dz -> da
        # a_i = exp(z_i - max(z)) / sum
        # For simplicity, use the gradient in a-space projected
        S = float(np.dot(grad_a, a_tmp))
        grad_z = a_tmp * (grad_a - S)
        return obj_val, grad_z
    
    def func_only(z):
        val, _ = func(z)
        return val
    
    def jac_only(z):
        _, g = func(z)
        return g
    
    try:
        res = minimize(
            fun=func_only, x0=z0, jac=jac_only,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        if verbose:
            print(f"  L-BFGS error: {e}")
    
    return best_a, best_score
```

```python
"""
Enhanced LP fictitious play with adaptive steps, diverse multi-start trajectories,
spectral initialization, gradient polishing, and multi-length exploration
for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_circ(a, n):
    """Circular auto-convolution via FFT."""
    fft_size = 1 << max(n - 1, 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:n]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n Toeplitz matrix C such that C @ g = convolve(f, g, 'full')."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = np.maximum(prev_g, 0)
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)

    solvers = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            kwargs = {}
            if prev_g is not None:
                kwargs['warm_start'] = True
            prob.solve(solver=solver, **kwargs)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return np.maximum(f, 1e-10)


def lp_fictitious_play(a0, time_limit, step_init=0.5, step_decay=0.9995,
                       step_min=0.005, verbose=True, label="LP-FP",
                       restart_interval=400, seed=42):
    """LP-based fictitious play with adaptive step and periodic restarts."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    step = step_init
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        step = max(step_min, step * step_decay)

        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if iters % restart_interval == 0 and iters > restart_interval:
            noise_scale = 0.0005 * (iters / restart_interval)
            noise = rng.standard_normal(n) * min(noise_scale, 0.005)
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            step = max(step_init * 0.3, step)

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.6f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score

def lse_obj_grad(a, T, n):
    """
    Compute LSE smoothed objective and gradient in a-space.
    Uses circular convolution for efficiency.
    """
    conv_circ = fast_convolve_circ(a, n)
    shifted = conv_circ / T
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    obj_val = 2.0 * n * lse  # Note: T * log(sum(exp(x/T))) = log(sum(exp(x/T))) * T... 
    # Actually T * (max/T + log(sum(exp((x-max)/T)))) = max + T*log(sum(exp((x-max)/T)))
    obj_val = 2.0 * n * (max_shifted + T * np.log(exp_shifted.sum()))

    p = exp_shifted / exp_shifted.sum()

    fft_size = n
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    grad_a = 4.0 * n * irfft(P * np.conj(A), fft_size)

    # Project to tangent space
    S = float(np.dot(grad_a, a))
    grad_a = grad_a - S

    return obj_val, grad_a


def lbgfs_polish(a0, time_limit, T=50.0, maxiter=3000, verbose=True, label="L-BFGS"):
    """L-BFGS on LSE smoothed objective in log-space."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    z0 = np.log(a) - np.log(a).mean()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_shift = z - z.max()
        a_tmp = np.exp(z_shift)
        a_tmp = a_tmp / a_tmp.sum()
        obj_val, grad_a = lse_obj_grad(a_tmp, T, n)
        return obj_val

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        z_shift = z - z.max()
        a_tmp = np.exp(z_shift)
        a_tmp = a_tmp / a_tmp.sum()
        obj_val, grad_a = lse_obj_grad(a_tmp, T, n)
        S = float(np.dot(grad_a, a_tmp))
        grad_z = a_tmp * (grad_a - S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        z_out = res.x - res.x.max()
        a_out = np.exp(z_out)
        a_out = a_out / a_out.sum()
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        if verbose:
            print(f"  {label} error: {e}")

    return best_a, best_score


def bussgang_flatten(a0, time_limit, tau=0.3, verbose=True, label="Bussgang"):
    """Spectral flattening via iterative amplitude adjustment."""
    n = len(a0)

    a = a / s

        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power.sum() / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** tau)

        a_new = np.maximum(a_new, 0)

def coordinate_descent(a0, time_limit, n_pairs=200, verbose=True, seed=123):
    """Coordinate descent with pairwise mass transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], max(a[j], a[i] * 0.5))

            for frac in [0.03, 0.08, 0.15, 0.3, 0.5, 0.75]:
                t = frac * base
                if t >= a[i] * 0.9:
                    t = a[i] * 0.3
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
                stall = 0
            else:
                stall += 1

        if not improved and stall > n_pairs * 5:
            noise = rng.standard_normal(n) * 0.002
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def make_initializations(n, seed=0):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Cosine concentration
    x = np.linspace(0, np.pi, n)
    c = np.cos(x) ** 2 + 0.1
    inits.append(('cosine', c / c.sum()))

    # Two bumps
    centers = [n * 0.25, n * 0.75]
    sigmas = [n * 0.1] * 2
    b = np.zeros(n)
    for c, s in zip(centers, sigmas):
        b += np.exp(-0.5 * ((np.arange(n) - c) / s) ** 2)
    inits.append(('twobump', b / b.sum()))

    # Random smoothed
    r = rng.standard_normal(n)
    # Smooth with a small kernel
    kernel = np.exp(-np.arange(-5, 6) ** 2 / 10.0)
    kernel /= kernel.sum()
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0)
    if r_smooth.sum() > 0:
        inits.append(('rand_smooth', r_smooth / r_smooth.sum()))

    # Periodic modulation
    freq = 7
    p = 1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n)
    inits.append(('periodic', p / p.sum()))

    # Power law
    pl = np.linspace(0.1, 1.0, n)
    pl = pl ** 0.5
    inits.append(('powerlaw', pl / pl.sum()))

    return inits

                                        step_min=0.003, verbose=True, label="LP-final", seed=99)
        best_seq, best_score = check_and_update(a_r, "lp_final", best_seq, best_score)

    # === Phase 10: Final coordinate descent polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t10 = min(rem * 0.3, 50.0)
        print(f"\n=== Phase 10: Final CD Polish (budget={t10:.0f}s) ===")
        a_r, sc_r = coordinate_descent(best_seq, t10, n_pairs=400, verbose=True, seed=777)
        best_seq, best_score = check_and_update(a_r, "cd_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        try:
            fs = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(fs) and fs < best_score:
                best_score = fs
        except Exception:
            pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
