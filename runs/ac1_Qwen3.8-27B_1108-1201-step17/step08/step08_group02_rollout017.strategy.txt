                }
                new_generation = generate_initial_population(**parameters)
                current_score = min([evaluate_sequence(seq) for seq in new_generation])
                if current_score < 1.503:
                    best_seq = sorted(new_generation, key=lambda x: evaluate_sequence(x))[:3]
            return best_seq.current_score
        else:
            # Use Lamarckian refinement
            t = tuple(np.log(best_seq))
            parent_sequence = crossover(best_parents, t, max_peaks)
            return parent_sequence.tolist()
