import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_n = a / s
    conv = fast_convolve(a_n)
    return float(2.0 * len(a_n) * conv.max())


def gradient_convolution(a, weights):
    """
    Compute gradient of sum_k weights[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_j weights[i+j] * a[j]
    Using FFT cross-correlation.
    """
    n = len(a)
    fft_size = 1 << (3 * n + 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * A.conj(), fft_size)
    # cross[i] = sum_j weights[j] * a[j-i] (circular)
    # We want sum_j weights[i+j] * a[j] = sum_j weights[j] * a[j-i]
    # which is cross-correlation at lag i = cross[i] for the circular case
    # With sufficient padding, circular = linear for the range we need
    return 2.0 * cross[:n]


def soft_max_gradient_descent(a0, time_limit, T_init=0.01, T_min=0.001,
                               lr_init=0.05, momentum=0.9, verbose=False):
    """
    Soft-max smoothed gradient descent with momentum and adaptive temperature.
    Optimizes 2*n*max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)  # velocity for momentum
    t0 = time.monotonic()
    it = 0
    T = T_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Current soft-max objective and gradient
        conv = fast_convolve(a)
        m = conv.max()
        w = np.exp((conv - m) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient of soft-max conv
        grad = np.zeros(n)
        # Direct computation (vectorized)
        # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
        # Build the Toeplitz-like structure
        for i in range(n):
            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                grad[i] = 2.0 * np.dot(w[i:i + j_max], a[:j_max])

        # Project gradient to tangent space of simplex
        grad_tan = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - lr * grad_tan
        a_new = a + v

        # Project to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        # Evaluate
        score = compute_score(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate and temperature
        if no_improve > 50:
            lr *= 0.95
            T = max(T_min, T * 0.9)
            v *= 0.5
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.6f} T={T:.5f} lr={lr:.6f}")

        if lr < 1e-8:
            break

    return best_a, best_score


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact 1D coordinate descent using scipy.optimize.minimize_scalar.
    For each coordinate, finds the exact optimal value.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                # Try setting to small positive value
                x_min = 0.0
                x_max = max(2.0 * a.max(), 0.1 / n)
            else:
                v = a[i]
                x_min = 0.0
                x_max = max(3.0 * v, 0.1)

            S = 1.0 - a[i]  # sum of all other coordinates (since sum(a)=1)
            v_old = a[i]

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Precompute unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = 0.0
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = 0.0
            unaff_max = max(unaff_left, unaff_right)

            # Define objective for 1D minimization
            def obj_1d(x):
                if x < -1e-15:
                    return np.inf
                x = max(x, 0.0)
                new_sum = S + x
                if new_sum < 0.01:
                    return np.inf
                delta = x - v_old
                # Affected convolution values
                affected = conv[k_start:k_end + 1] + 2.0 * delta * a[0:n]
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    affected = affected.copy()
                    affected[idx_2i] = conv[2 * i] + x * x - v_old * v_old
                new_max = max(unaff_max, float(affected.max()))
                return 2.0 * n * new_max / (new_sum * new_sum)

            try:
                res = minimize_scalar(obj_1d, bounds=(x_min, x_max),
                                      method='bounded',
                                      options={'xatol': 1e-10})
                x_new = res.x
                obj_new = res.fun
            except Exception:
                continue

            if obj_new < best_score - 1e-14:
                # Apply update
                delta = x_new - v_old
                a[i] = x_new
                # Update convolution
                affected_delta = 2.0 * delta * a[0:n]
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    affected_delta = affected_delta.copy()
                    # For k=2i: change is x_new^2 - v_old^2 (not 2*delta*a[i])
                    affected_delta[idx_2i] = x_new * x_new - v_old * v_old
                conv[k_start:k_end + 1] += affected_delta
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                best_score = 2.0 * n * M
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def fast_sweep_cd(a0, time_limit, n_trials=12, verbose=False, label=""):

    Fast coordinate descent with multiple trial step sizes.

    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute trial factors

                        -0.7, 0.7, -0.75, 0.75])

    while time.monotonic() - t0 < time_limit:

            else:
                unaff_right = 0.0

                improved = True

        if not improved:

        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0

def init_from_eigenvector(n, mode=0):

    elif mode == 2:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * t / n) + 0.2 * np.cos(6 * np.pi * t / n)
    else:
        a = 1.0 + 0.15 * np.cos(2 * np.pi * t / n) + 0.15 * np.cos(2 * np.pi * 3 * t / n) + 0.1 * np.cos(2 * np.pi * 5 * t / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

    return a / s


def init_optimal_structure(n, rng):
    """
    Initialize based on the idea that optimal sequences have
    a "two-scale" structure: slow variation + fast oscillation.
    """
    t = np.arange(n, dtype=np.float64)
    # Slow envelope
    env_freq = rng.uniform(0.5, 2.0)
    env_amp = rng.uniform(0.1, 0.4)
    env = env_amp * np.cos(2 * np.pi * env_freq * t / n + rng.uniform(0, 2 * np.pi))
    # Fast oscillation (to reduce autocorrelation peak)
    fast_freq = rng.uniform(3.0, 10.0)
    fast_amp = rng.uniform(0.05, 0.2)
    fast = fast_amp * np.cos(2 * np.pi * fast_freq * t / n + rng.uniform(0, 2 * np.pi))
    a = 1.0 + env + fast
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Linearly interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.05):
    """Perturb a sequence to escape local optima."""
    n = len(a)
    a_new = a.copy()
    # Multiplicative perturbation
    noise = rng.standard_normal(n) * strength
    a_new = a_new * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def sparsify_and_readd(a, rng, sparsity=0.3):
    """Zero out some entries and redistribute."""
    n = len(a)
    a_new = a.copy()
    n_zero = int(n * sparsity)
    zeros = rng.choice(n, size=n_zero, replace=False)
    mass_removed = a_new[zeros].sum()
    a_new[zeros] = 0
    # Redistribute to remaining
    remaining = np.array([i for i in range(n) if i not in set(zeros.tolist())])
    a_new[remaining] += mass_removed * a_new[remaining] / max(a_new[remaining].sum(), 1e-15)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization at multiple n values + quick optimization
    Phase 2: Intensive exact coordinate descent on best candidates
    Phase 3: Soft-max gradient descent for escaping local optima
    Phase 4: Perturbation + re-optimization cycles
    Phase 5: Final polishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")

    # === Phase 1: Multi-n diverse initialization ===
    print("\n=== Phase 1: Diverse Init at Multiple n ===")
    t1_start = time.monotonic()

    n_values = [1024]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
    # Add other candidates
    for alt_n in [768, 1280, 1536, 640]:
        if alt_n not in n_values:
            n_values.append(alt_n)

    candidates = []

    for n in n_values[:6]:
        elapsed = time.monotonic() - t1_start
        if elapsed > 120:
            break

        # Generate diverse initializations
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_from_eigenvector(n, 0))
        inits.append(init_from_eigenvector(n, 1))
        inits.append(init_from_eigenvector(n, 2))
        inits.append(init_from_eigenvector(n, 3))
        inits.append(init_optimal_structure(n, rng))
        inits.append(init_optimal_structure(n, rng))
        inits.append(np.ones(n) / n)

        for idx, init_a in enumerate(inits):
            # Quick fast coordinate descent
            t_left = 120.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(2.0, min(6.0, t_left / 10.0))
            a_opt, s_opt, _ = fast_sweep_cd(init_a, t_alloc, n_trials=12, label=f"p1_{n}_{idx}")
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best: {candidates[0][0]:.6f}")

    for s_c, a_c, n_c in candidates[:4]:
        update_best(a_c, f"p1_n{n_c}")

    # === Phase 2: Intensive exact coordinate descent ===
    print(f"\n=== Phase 2: Exact Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()
        for rnd in range(4):
            rem = deadline - time.monotonic() - 500
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 180.0)
            print(f"  ECD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt = exact_coord_descent(current, t_alloc, verbose=True, label=f"ecd_r{rnd}")
            update_best(a_opt, f"ecd_r{rnd}")

            # Perturb between rounds
            if best_score > 1.5035:
                current = perturb_sequence(best_seq, rng, strength=0.03 + 0.02 * rnd)
            else:
                current = best_seq.copy()

    # === Phase 3: Soft-max gradient descent ===
    print(f"\n=== Phase 3: Soft-max Gradient Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_md = min(rem * 0.4, 150.0)
            print(f"  SGD n={len(best_seq)}, budget={t_md:.0f}s")
            a_md, s_md = soft_max_gradient_descent(best_seq, t_md, T_init=0.005, T_min=0.0005,
                                                    lr_init=0.03, momentum=0.85, verbose=True)
            update_best(a_md, "sgd")

            # Try from perturbed start
            if rem - t_md > 30:
                pert = perturb_sequence(best_seq, rng, strength=0.1)
                t_md2 = min((rem - t_md) * 0.5, 60.0)
                a_md2, s_md2 = soft_max_gradient_descent(pert, t_md2, T_init=0.02, T_min=0.005,
                                                          lr_init=0.05, momentum=0.9, verbose=False)
                update_best(a_md2, "sgd_pert")

    # === Phase 4: Perturbation + re-optimization cycles ===
    print(f"\n=== Phase 4: Perturb + Refine (best={best_score:.6f}) ===")
    n_best = len(best_seq) if best_seq is not None else 1024

    for cycle in range(6):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break

        # Different perturbation strategies
        if cycle % 3 == 0:
            start = perturb_sequence(best_seq, rng, strength=0.05 + 0.03 * cycle)
        elif cycle % 3 == 1:
            start = sparsify_and_readd(best_seq, rng, sparsity=0.1 + 0.05 * cycle)
        else:
            # Mix with a fresh initialization
            fresh = init_optimal_structure(n_best, rng)
            start = 0.5 * best_seq + 0.5 * fresh
            s = start.sum()
            start = start / s

        # Optimize: fast CD then exact CD
        t_cd1 = min(rem * 0.3, 60.0)
        a_c1, s_c1, _ = fast_sweep_cd(start, t_cd1, n_trials=12, label=f"c{cycle}_fast")
        update_best(a_c1, f"cycle{cycle}_fast")

        rem2 = deadline - time.monotonic() - 150
        if rem2 > 20:
            t_cd2 = min(rem2 * 0.5, 80.0)
            a_c2, s_c2 = exact_coord_descent(a_c1, t_cd2, label=f"c{cycle}_exact")
            update_best(a_c2, f"cycle{cycle}_exact")

    # === Phase 5: Try different n ===
    print(f"\n=== Phase 5: Alternative n (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for alt_n in [768, 1536, 2048]:
            if alt_n == n_best:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alt = min(rem * 0.3, 80.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = fast_sweep_cd(init_alt, t_alt * 0.5, n_trials=12, label=f"alt{alt_n}")
            update_best(a_alt, f"alt_{alt_n}_fast")

            rem2 = deadline - time.monotonic() - 80
            if rem2 > 20:
                a_alt2, s_alt2 = exact_coord_descent(a_alt, min(rem2 * 0.5, 40.0), label=f"alt{alt_n}_e")
                update_best(a_alt2, f"alt_{alt_n}_exact")

    # === Phase 6: Final intensive polish on best ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Round 1: Exact CD
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_f1 = min(rem * 0.5, 100.0)
            a_f1, s_f1 = exact_coord_descent(best_seq, t_f1, label="final1")
            update_best(a_f1, "final_exact")

        # Round 2: Fast CD with many trials
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_f2 = min(rem * 0.6, 50.0)
            a_f2, s_f2, _ = fast_sweep_cd(best_seq, t_f2, n_trials=16, label="final2")
            update_best(a_f2, "final_fast")

        # Round 3: Very fast polish
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_f3, s_f3, _ = fast_sweep_cd(best_seq, min(rem * 0.8, 20.0), n_trials=20, label="last")
            update_best(a_f3, "final_last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
