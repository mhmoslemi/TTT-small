"""

1. Parametric U-shape initialization with optimized parameters

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    _lp_solver = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:

    if prob.status not in ('optimal', 'optimal_inaccurate'):

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def init_u_shape(n, alpha):
    """U-shaped initialization: a[j] = 1 + alpha * (2j/(n-1) - 1)^2"""
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / (n - 1) - 1.0
    a = 1.0 + alpha * x ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s

def init_cosine_u(n, alpha):
    """Cosine-based U-shape"""
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / (n - 1) - 1.0
    a = 1.0 + alpha * np.cos(np.pi * x / 2.0) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s

    - Finer line search
    - Incremental convolution update

    else:
        a = a / s

    best_a = a.copy()

        sweeps += 1
        improved = False

        # Peak-directed ordering: process coordinates that contribute to peaks first

        # Collect all coordinates that contribute to peaks
        coord_contrib = np.zeros(n)
        for k in peak_positions:

            i_end = min(n - 1, k)
            coord_contrib[i_start:i_end+1] += 1

                    delta = 0.05 * a[i]

                elif trial == 7:
                    delta = 0.5 * a[i]
                else:

                    delta = best_delta * (1.0 + 0.1 * (trial - 7))

                new_val = a[i] + delta

                if new_sum < 0.01:
                    continue

                # New affected values

                if 2 * i >= k_start and 2 * i <= k_end:

                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2*i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                new_score = 2.0 * n * new_max / (new_sum * new_sum)

                    best_delta = delta

def fine_coord_descent_v2(a0, time_limit, verbose=False, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 1:
                unaff_max = max(unaff_max, float(conv[k_end+1:].max()))

            affected = conv[k_start:k_end+1]
            a_shifted = a

            best_delta = 0.0
            best_new_score = best_score

            # Very fine steps
            for factor in [0.995, 0.99, 0.985, 0.9975, 1.0025, 1.005, 1.01, 1.015, 0.98, 1.02]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue
                new_sum = sum_a + delta
                if new_sum < 0.01:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2*i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                new_score = 2.0 * n * new_max / (new_sum * new_sum)
                if new_score < best_new_score:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0 and best_new_score < best_score - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end+1] += changes

                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                best_score = 2.0 * n * M / (sum_a * sum_a)
                best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break
        no_improve = 0 if improved else no_improve + 1

    return best_a, best_score, sweeps

def lp_alternating(a0, time_limit, solver=None, verbose=False):
    """LP-based alternating minimization."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    b = a.copy()
    a_avg = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Solve LP for b given a
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        # Solve LP for a given b
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
            a_avg = (it * a_avg + a) / (it + 1)

        # Check candidates
        for cand in [a, b, 0.5*(a+b), a_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            score = 2.0 * n * fast_convolve(cand_n).max()
            if score < best_score:
                best_score = score
                best_a = cand_n.copy()

        if verbose and it % 5 == 0:
            print(f"    LP alt n={n} it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score, it

def subgradient_step(a, conv, lr, n_active=10):
    """One subgradient step for minimizing max(conv(a,a))."""
    n = len(a)
    M = conv.max()
    # Find active constraints
    active = np.where(conv >= M * 0.99)[0]
    if len(active) > n_active:
        # Take top n_active
        idx = np.argsort(conv[active])[::-1][:n_active]
        active = active[idx]

    # Equal weights for now (could use softmax)
    w = np.ones(len(active)) / len(active)

    # Subgradient
    grad = np.zeros(n)
    for idx, k in enumerate(active):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    # Mirror descent step
    a_new = a * np.exp(-lr * grad)
    a_new = simplex_project(a_new)
    return a_new

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):

    s = a.sum()

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

def simulated_annealing_step(a, conv, T, n_perturb=50):
    """Perturb a random subset of coordinates (simulated annealing style)."""
    n = len(a)
    a_new = a.copy()
    indices = np.random.choice(n, size=min(n_perturb, n), replace=False)
    for i in indices:
        a_new[i] *= (1.0 + np.random.standard_normal() * 0.05)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals():

                return _lp_solver
        except Exception:
            pass
    _lp_solver = None

        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GUROBI, cp.MOSEK]:

                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def init_u_shape(n, alpha):
    """U-shaped: a[j] = 1 + alpha*(2j/(n-1)-1)^2, normalized."""
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / max(n - 1, 1) - 1.0
    a = 1.0 + alpha * x ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_multi_harmonic(n, alpha, harmonics, rng):

    s = a.sum()
    if s < 1e-15:

    return a / s


def fast_coord_descent(a0, time_limit, n_trials=12, verbose=False, label=""):
    """
    Fast coordinate descent with:
    - Peak-directed coordinate ordering
    - Adaptive line search (multiple trial steps)
    - Incremental convolution update
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute step multipliers
    step_mults = np.array([-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5,
                           0.02, -0.02, 0.2, -0.2, 0.4, -0.4, 0.1, -0.1])
    if n_trials > len(step_mults):
        step_mults = step_mults[:n_trials]
    elif n_trials < len(step_mults):
        step_mults = step_mults[:n_trials]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Determine coordinate ordering
        if sweeps % 10 == 1:
            # Peak-directed: prioritize coordinates contributing to peaks
            peak_thresh = M * 0.9
            peak_mask = conv >= peak_thresh
            if peak_mask.any():
                coord_weight = np.zeros(n)
                for k in np.where(peak_mask)[0]:
                    i_start = max(0, k - n + 1)
                    i_end = min(n - 1, k)
                    coord_weight[i_start:i_end + 1] += 1
                order = np.argsort(-coord_weight, kind='stable')
            else:
                order = np.arange(n)
        else:
            order = np.arange(n)

        for i in order:
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                ul = float(conv[:k_start].max())
                if ul > unaff_max:
                    unaff_max = ul
            if k_end < 2 * n - 1:
                ur = float(conv[k_end + 1:].max())
                if ur > unaff_max:
                    unaff_max = ur

            affected = conv[k_start:k_end + 1]
            a_shifted = a  # a[k-i] for k=i..i+n-1 gives a[0..n-1]

            best_delta = 0.0
            best_new_score = best_score

            for trial in range(n_trials):
                delta = step_mults[trial] * a[i]
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue

                new_sum = sum_a + delta
                if new_sum < 0.01:
                    continue

                # Compute new affected values (linear approx + quadratic correction)
                new_affected = affected + 2.0 * delta * a_shifted
                # Correct the k=2i term if in range
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                new_score = 2.0 * n * new_max / (new_sum * new_sum)

                if new_score < best_new_score:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0 and best_new_score < best_score - 1e-15:
                # Apply update
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                # Incremental convolution update
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                best_score = 2.0 * n * M / (sum_a * sum_a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 80:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            print(f"    FCD {label} sweep={sweeps} ({time.monotonic()-t0:.0f}s): score={best_score:.6f} M={M:.8f}")

    return best_a, best_score, sweeps

def mirror_descent(a0, time_limit, lr_init=0.015, n_active=30, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()

    t0 = time.monotonic()

    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-7:
                break

        # Compute subgradient using top peaks with softmax weighting
        k_act = min(n_active, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = max(0.005 * (pv.max() - pv.min()), 1e-12)
        w = np.exp((pv - pv.max()) / temp)
        w_sum = w.sum()
        w = w / w_sum

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        # Mirror descent step
        scaled = a * np.exp(-lr * grad)
        a = simplex_project(scaled)
        lr *= 0.9997

        if verbose and it % 200 == 0:
            print(f"    MD it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Multi-init + short LP warm-start ===
    print("\n=== Phase 1: Multi-Init + LP Warm ===")
    t1_start = time.monotonic()
    candidates = []

    n_values = [768, 1024, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
        n_values = [max(256, x) for x in n_values if 256 <= x <= 2048]

    all_n = sorted(set(n_values))

    for n in all_n:
        if time.monotonic() - t1_start > 400:
            break
        # Try different U-shape params and harmonic variants
        inits = []
        for alpha in [0.8, 1.2, 1.8, 2.5]:
            inits.append(init_u_shape(n, alpha))
        inits.append(init_multi_harmonic(n, 1.5, [3, 7], rng))
        inits.append(init_multi_harmonic(n, 2.0, [2, 5, 9], rng))
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
            inits.append(shape_perturb(interpolate_seq(parent, n), rng, 2, 0.05))

        for init_idx, a_init in enumerate(inits):
            if time.monotonic() - t1_start > 400:
                break
            # Short LP warm-start
            t_alloc = min(12.0, max(5.0, (400 - (time.monotonic() - t1_start)) / max(1, len(inits) - init_idx)))
            a_lp, s_lp, _ = lp_alternating(a_init, t_alloc, solver=solver)
            candidates.append((s_lp, a_lp, n, f"i{init_idx}"))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates. Top 3: {[f'{c[0]:.6f}(n={c[2]})' for c in candidates[:3]]}")

    for s_c, a_c, n_c, lab in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{lab}")

    # === Phase 2: Intensive coordinate descent ===
    print(f"\n=== Phase 2: Intensive FCD (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t2_start = time.monotonic()

        current = best_seq.copy()
        n_rounds = 6
        for rnd in range(n_rounds):
            rem = deadline - time.monotonic() - 600
            if rem < 30:
                break
            t_alloc = min(rem * 0.25, 300.0 / (n_rounds - rnd))
            print(f"  FCD r{rnd+1}/{n_rounds}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = fast_coord_descent(current, t_alloc, n_trials=14,
                                                      verbose=True, label=f"r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"fcd_r{rnd}")

            # Perturb for next round
            if s_opt > best_score * 0.9999:
                # Stuck - perturb
                current = shape_perturb(best_seq, rng, 3, 0.04)
                print(f"    Perturbed (shape), continuing from perturbed init")
            elif s_opt > best_score - 1e-10:
                current = local_noise_perturb(best_seq, rng, 0.015, 0.08)
            else:
                current = a_opt

    # === Phase 3: LP alternating (longer budget for global improvement) ===
    print(f"\n=== Phase 3: LP Alternating (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        lp_ns = [n_best]
        for alt in [n_best + 256, n_best - 256, 1024, 1280]:
            if alt != n_best and 256 <= alt <= 2048:
                lp_ns.append(alt)
        lp_ns = lp_ns[:3]

        for n_lp in lp_ns:
            rem = deadline - time.monotonic() - 400
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 200.0)
            print(f"  LP n={n_lp}, budget={t_alloc:.0f}s")
            init_a = interpolate_seq(best_seq, n_lp) if best_seq is not None else init_u_shape(n_lp, 1.5)
            a_lp, s_lp, it = lp_alternating(init_a, t_alloc, solver=solver, verbose=True)
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, f"lp_n{n_lp}")

            # Follow up with coordinate descent on LP result
            rem = deadline - time.monotonic() - 250
            if rem > 30:
                t_cd = min(rem * 0.3, 80.0)
                a_cd, s_cd, _ = fast_coord_descent(a_lp, t_cd, n_trials=12, label=f"lp_cd_{n_lp}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"lp_cd_{n_lp}")

    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_md = min(rem * 0.4, 100.0)
            print(f"  MirrorDescent n={len(best_seq)}, budget={t_md:.0f}s")
            a_md, s_md = mirror_descent(best_seq, t_md, lr_init=0.012, n_active=40, verbose=True)
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, "mirror")

            # CD after mirror descent
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                t_cd3 = min(rem * 0.4, 60.0)
