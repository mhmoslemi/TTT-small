"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast FFT-based subgradient with Nesterov momentum, adaptive peak weighting,
LP polishing for escaping local minima, multi-length exploration, and coordinate descent.
Key insight: asymmetry is essential (symmetric sequences have score >= 2).

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, w_peaks, fft_size):
    """
    Compute gradient of weighted sum of convolution peaks.
    w_peaks: weights on convolution indices (length 2n-1)
    Returns gradient w.r.t. a (length n).
    """
    n = len(a)
    W = np.zeros(fft_size)
    W[:len(w_peaks)] = w_peaks
    Wf = rfft(W, fft_size)
    Af = rfft(a, fft_size)
    cross = irfft(Wf * np.conj(Af), fft_size)
    return 2.0 * cross[:n]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(b, top_k, temperature):
    """Compute softmax weights concentrated on top-k peaks."""
    n = len(b)
    if top_k >= n:
        idx = np.arange(n)
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
    vals = b[idx].copy()
    vals = vals / (temperature + 1e-30)
    vals -= vals.max()
    w = np.exp(vals)
    w /= w.sum()
    full_w = np.zeros(n)
    full_w[idx] = w
    return full_w

def solver_lp_direction(f, n, peak_indices, M, eps=1e-8):
    """Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peaks."""
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=200000, eps=eps)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

                  f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Targeted coordinate descent on the most impactful coordinates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7]
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        pass_count += 1
        # Process coordinates in order of impact (larger first)
        order = np.argsort(-f)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                f /= f.sum()
                if best_j_obj < best_b - 1e-15:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_b


def mirror_descent_polish(f_init, time_limit, label="md", n_top=150, lr_init=0.03):
    """Mirror descent (exponential weighting) for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = 0.001
    no_improve = 0
    it = 0
    rng = np.random.default_rng(444)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Focus on top peaks
        idx = np.argpartition(b, -n_top)[-n_top:]
        vals = b[idx] / (temp * M)
        vals -= vals.max()
        w = np.exp(vals)
        w /= w.sum()
        w_full = np.zeros(size)
        w_full[idx] = w

        # Gradient
        grad = compute_grad_fft(f, w_full, fft_size)

        # Mirror descent: f_i *= exp(-lr * grad_i)
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        f_new = np.exp(log_f_new - log_f_new.max())
        s = f_new.sum()
        if s < 1e-30:
            break
        f = f_new / s

        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.93
                    no_improve = 0
                if no_improve > 40:
                    f = best_f + rng.standard_normal(n) * 0.001 * best_f
                    f = np.maximum(f, 1e-15)
                    f /= f.sum()
                    lr = lr_init
                    no_improve = 0

    return best_f, best_b


def interpolate_length(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: hybrid Nesterov subgradient + LP polish + multi-length + CD.
    """
    deadline = time.monotonic() + max(10.0, float(budget_s) - 35.0)
    print(f"Total budget: {budget_s:.0f}s, deadline in {float(budget_s)-35:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    def try_update(candidate_f, source_name):
        nonlocal best_seq, best_score
        vs = float(evaluate_sequence(candidate_f.tolist()))
        if np.isfinite(vs) and vs < best_score:
            best_score = vs
            best_seq = candidate_f.copy()
            print(f"  *** NEW BEST [{source_name}]: {vs:.6f} (n={len(candidate_f)})")
            return True
        return False

    # === Phase 1: Fast Nesterov subgradient on parent (~35% time) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 60:
        n_p = len(parent)
        print(f"\n=== Phase 1: Nesterov SG (n={n_p}, budget={t1:.0f}s) ===")
        f1, obj1 = nesterov_subgradient(parent, time_limit=t1, label="ns1",
                                         top_k=120, temp_init=0.003, temp_final=0.0003,
                                         lr_init=0.015, momentum_init=0.85)
        try_update(f1, "ns1")

    # === Phase 2: LP polish on best (~20% time) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 40:
        n_b = len(best_seq)
        print(f"\n=== Phase 2: LP polish (n={n_b}, budget={t2:.0f}s) ===")
        f2, obj2 = lp_polish(best_seq, time_limit=t2, label="lp2",
                              max_peaks=300, lp_eps=1e-9)
        try_update(f2, "lp2")

    # === Phase 3: More Nesterov on updated best (~15% time) ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 30:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: Nesterov SG round 2 (n={n_b}, budget={t3:.0f}s) ===")
        f3, obj3 = nesterov_subgradient(best_seq, time_limit=t3, label="ns2",
                                         top_k=80, temp_init=0.001, temp_final=0.0002,
                                         lr_init=0.008, momentum_init=0.90)
        try_update(f3, "ns2")

    # === Phase 4: Multi-length exploration (~15% time) ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.9, 1.05, 1.15, 1.3]:
        tl = int(n_base * factor)
        tl = max(512, min(2500, tl))
        if abs(tl - n_base) > 20 and tl not in target_lengths:
            target_lengths.append(tl)
    # Also try powers of 2 nearby
    for p2 in [768, 1024, 1536, 2048]:
        if abs(p2 - n_base) > 50 and p2 not in target_lengths:
            target_lengths.append(p2)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t4 = min(rem * 0.07, 90.0)
        if t4 < 20:
            break
        f_interp = interpolate_length(best_seq, target_n)
        print(f"\n=== Phase 4: Nesterov n={target_n} (budget={t4:.0f}s) ===")
        f4, obj4 = nesterov_subgradient(f_interp, time_limit=t4, label=f"ns_n{target_n}",
                                         top_k=100, temp_init=0.002, temp_final=0.0004,
                                         lr_init=0.01, momentum_init=0.87)
        try_update(f4, f"ns_n{target_n}")

    # === Phase 5: Mirror descent polish (~10% time) ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 120.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 5: Mirror descent (n={n_b}, budget={t5:.0f}s) ===")
        f5, obj5 = mirror_descent_polish(best_seq, time_limit=t5, label="md",
                                          n_top=100, lr_init=0.04)
        try_update(f5, "md")

    # === Phase 6: Coordinate descent (~10% time) ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 100.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 6: Coord descent (n={n_b}, budget={t6:.0f}s) ===")
        f6, obj6 = coordinate_descent(best_seq, time_limit=t6, label="cd")
        try_update(f6, "cd")

    # === Phase 7: Final LP burst on best ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.25, 100.0)
    if t7 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 7: Final LP (n={n_b}, budget={t7:.0f}s) ===")
        f7, obj7 = lp_polish(best_seq, time_limit=t7, label="lp_final",
                              max_peaks=400, lp_eps=1e-9)
        try_update(f7, "lp_final")

    # === Phase 8: Final CD polish ===
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.15, 60.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final CD (budget={t8:.0f}s) ===")
        f8, obj8 = coordinate_descent(best_seq, time_limit=t8, label="cd_final")
        try_update(f8, "cd_final")

    # === Phase 9: One more Nesterov pass if time remains ===
    rem = deadline - time.monotonic()
    t9 = min(rem * 0.30, 60.0)
    if t9 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 9: Final Nesterov (n={n_b}, budget={t9:.0f}s) ===")
        f9, obj9 = nesterov_subgradient(best_seq, time_limit=t9, label="ns_final",
                                         top_k=60, temp_init=0.0005, temp_final=0.0001,
                                         lr_init=0.004, momentum_init=0.92)
        try_update(f9, "ns_final")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
