Spectral-eigen optimized step function minimization.
Combines LP-based direction finding with spectral initialization,
Newton correction, and adaptive peak targeting to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import sys

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

"""
Hybrid LP + spectral-eigen optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses PDLP for fast LP solving, spectral flattening directions, adaptive peak targeting,
and multi-scale exploration with Newton correction.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def build_lp_matrix(f, indices, n):
    """Build constraint matrix for LP: row i is the slice of f that convolves to index i."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, indices, M, n):
    """Solve LP: max 1^T g s.t. g >= 0, (f*g)_k <= M for k in indices."""
    K = len(indices)
    if K == 0:
        return None
    A = build_lp_matrix(f, indices, n)
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        if n <= 2000 and K <= 500:
            prob.solve(solver=cp.PDLP, max_iter=100000, tolerance=1e-8, time_limit_hook=None)
        else:
            prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-8, eps_rel=1e-8, accel_lookback=10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-7, eps_rel=1e-7)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iters=20000, abs_acceleration=1e-8)
            except Exception:
                return None
    if g.value is None:
        return None
    g_val = np.asarray(g.value, dtype=np.float64).ravel()
    g_val = np.maximum(g_val, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    if g_val.sum() <= 0:
        return None
    return g_val


def spectral_direction(f, M, n_peaks=20):
    """
    Compute a spectral-flattening direction:
    FFT the sequence, damp the peaks in frequency domain, IFFT back, project to simplex.
    """
    n = len(f)
    L = 2 * n - 1
    fft_size = 1 << (L - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    power = np.abs(A) ** 2
    # Identify the largest non-DC power bins
    if len(power) > 2:
        non_dc = power[1:]
        if len(non_dc) > n_peaks:
            top_idx = np.argpartition(non_dc, -n_peaks)[-n_peaks:] + 1
        else:
            top_idx = np.arange(1, len(power))
        # Damp these peaks
        target = np.median(power[1:]) if len(power) > 1 else power[0]
        A_new = A.copy()
        for k in top_idx:
            if power[k] > target:
                scale = np.sqrt(target / power[k])
                A_new[k] *= scale
        f_spec = np.fft.irfft(A_new, fft_size)[:n]
        f_spec = np.maximum(f_spec, 0.0)
        s = f_spec.sum()
        if s > 1e-15:
            return f_spec / s
    return None


def subgradient_direction(f, b, n_top=30, temperature=0.001):
    """Compute subgradient direction targeting top peaks with softmax weighting."""
    n = len(f)
    M = b.max()
    # Softmax weights on peaks
    scaled = (b - M) / max(temperature, 1e-10)
    w = np.exp(scaled)
    # Focus on top peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_other = w.copy()
        w_other[np.setdiff1d(np.arange(len(b)), top_idx)] = 0.0
        w = w_other
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    # Compute subgradient: for peak at lag k, gradient is a[i+k] + a[i-k]
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[k]
        for i in range(n):
            idx_p = i + int(k)
            idx_m = int(k) - i
            val = 0.0
            if 0 <= idx_p < n:
                val += f[idx_p]
            if 0 <= idx_m < n:
                val += f[idx_m]
            g[i] += wk * val
    return g


def line_search(f, direction, current_M, max_tries=20):
    """Adaptive line search: try several step sizes and pick the best."""
    best_obj = current_M
    best_t = 0.0
    # Try logarithmically spaced step sizes
    for t in [0.01, 0.02, 0.04, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.65, 0.8, 0.95]:
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-15:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def newton_correction(f, active_indices, target_M, n_iters=5):
    """
    Newton-like correction: try to equalize active peaks to target_M.
    Uses the Hessian of each active quadratic to compute a correction step.
    """
    n = len(f)
    f_curr = f.copy()
    for _ in range(n_iters):
        b = fast_convolve(f_curr)
        M = b.max()
        # Compute gradient at the worst peak
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        for i in range(n):
            ip = i + k_star
            im = k_star - i
            val = 0.0
            if 0 <= ip < n:
                val += f_curr[ip]
            if 0 <= im < n:
                val += f_curr[im]
            grad[i] = val
        # Simple gradient step with small step size
        lr = 1e-4
        f_new = f_curr - lr * grad
        f_new = project_simplex(f_new)
        b_new = fast_convolve(f_new).max()
        if b_new < M - 1e-15:
            f_curr = f_new
        else:
            break
    return f_curr


def run_lp_optimize(f_init, time_limit, top_k_init=80, top_k_max=500,
                    use_spectral=True, label="lp", max_stuck=60):
    """Main LP-based optimization loop with spectral directions."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    rng = np.random.default_rng(12345)
    full_len = 2 * n - 1
    momentum_dir = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        # Adaptive top_k
        if it % 10 == 0 and top_k < top_k_max:
            top_k = min(top_k + min(20, top_k_max - top_k), top_k_max)

        b = fast_convolve(f)
        M = b.max()

        # Peak selection: top-K by value, weighted by excess
        if top_k < full_len:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(full_len)
        indices = np.sort(top_idx)

        # Solve LP
        g = solve_lp(f, indices, M, n)

        direction = None
        if g is not None:
            S_g = g.sum()
            if S_g > 1.0 + 1e-4:
                h = g / S_g
                # Update momentum
                momentum_dir = 0.4 * momentum_dir + 0.6 * h
                mom_sum = momentum_dir.sum()
                if mom_sum > 1e-10:
                    h_mom = momentum_dir / mom_sum
                else:
                    h_mom = h
                # Try both LP direction and momentum-smoothed
                candidates = [h, h_mom]
            else:
                candidates = []
        else:
            candidates = []

        # Add spectral direction
        if use_spectral:
            f_spec = spectral_direction(f, M, n_peaks=max(10, n // 20))
            if f_spec is not None:
                candidates.append(f_spec)

        # Add subgradient direction
        if len(candidates) < 3:
            sg = subgradient_direction(f, b, n_top=max(10, n // 15), temperature=0.002)
            if np.any(sg > 0):
                # Convert subgradient (we want to decrease) to a direction
                f_sg = project_simplex(f - 0.01 * sg)
                if f_sg.sum() > 1e-10:
                    candidates.append(f_sg)

        # Line search over candidates
        best_trial = None
        best_trial_obj = M
        for direction in candidates:
            t_found, obj_found = line_search(f, direction, M)
            if obj_found < best_trial_obj - 1e-14:
                best_trial_obj = obj_found
                best_trial = (1.0 - t_found) * f + t_found * direction
                best_trial = np.maximum(best_trial, 0.0)
                best_trial = best_trial / best_trial.sum()

        if best_trial is None:
            # No improvement from directions; try small random perturbation
            stuck += 2
            if stuck > max_stuck:
                break
            f = project_simplex(f + rng.standard_normal(n) * 0.002)
            continue

        f = best_trial
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > max_stuck:
                break

        # Periodic Newton correction on best
        if it % 15 == 0 and time.monotonic() - t0 < time_limit - 5:
            f_newton = newton_correction(best_f, np.array([int(np.argmax(fast_convolve(best_f)))]), best_obj, n_iters=3)
            bn = fast_convolve(f_newton).max()
            if bn < best_obj - 1e-14:
                best_obj = bn
                best_f = f_newton
                f = best_f.copy()
                stuck = 0

        # Restart with perturbation if stuck
        if stuck > 15 and it % 20 < 3:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            stuck = 0
            momentum_dir = np.zeros(n)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.003, n_top=50, momentum=0.8):
    """Subgradient optimization with adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    v = np.zeros(n)
    lr = lr_init
    temp = 0.001
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weighting
        scaled = (b - M) / max(temp, 1e-10)
        w = np.exp(scaled)
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = np.zeros(len(b))
            w_sel[top_idx] = w[top_idx]
            ws = w_sel.sum()
            if ws > 1e-30:
                w = w_sel / ws
            else:
                w = np.zeros(len(b))
                w[np.argmax(b)] = 1.0
        else:
            ws = w.sum()
            if ws > 1e-30:
                w = w / ws

        # Subgradient
        g = np.zeros(n)
        active = np.where(w > 1e-12)[0]
        for k in active:
            wk = w[k]
            ki = int(k)
            for i in range(n):
                ip = i + ki
                im = ki - i
                val = 0.0
                if 0 <= ip < n:
                    val += f[ip]
                if 0 <= im < n:
                    val += f[im]
                g[i] += wk * val

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.015)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.9
                    no_improve = 0
            if it % 150 == 0:
                temp = min(temp * 1.15, 0.3)
                print(f"    sg it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                      f"lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
            if no_improve > 20:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def coordinate_descent_fast(f_init, time_limit, max_passes=10):
    """Fast coordinate descent using incremental updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    for _pass in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        any_improved = False
        b = fast_convolve(f)
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            # Try a few multipliers
            best_j_obj = b.max()
            best_j_val = f[j]
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                f[j] = f[j] * mult
                s = f sum if len(f) > 0 else 0
                # Actually just recompute
                s_val = f.sum()
                if s_val < 1e-12:
                    f[j] = best_j_val
                    break
                f_norm = f / s_val
                obj_j = fast_convolve(f_norm).max()
                if obj_j < best_j_obj - 1e-13:
                    best_j_obj = obj_j
                    best_j_val = f[j]
                    f[:] = f_norm
            f[j] = best_j_val
            s = f.sum()
            if s > 1e-12:
                f = f / s
            b_new = fast_convolve(f).max()
            if b_new < best_obj - 1e-13:
                best_obj = b_new
                best_f = f.copy()
                any_improved = True
            b = b_new

        if not any_improved:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: dense LP for small n, spectral-guided LP for medium/large n,
    subgradient polish, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)
    t_start = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print("No parent found, using uniform n=512")

    best_seq = parent.copy()
    best_score = p_score

    # ============================================================
    # Phase 1: Dense LP for small n
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Dense LP small n (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.18, 300.0)
    if t1 > 30:
        small_n_list = [128, 192, 256, 384]
        per_n = t1 / len(small_n_list)
        for n_trial in small_n_list:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            alloc = min(per_n, rem * 0.85)
            if alloc < 8:
                break
            f_init = interp_sequence(parent, n_trial)
            print(f"  Dense LP n={n_trial}, budget={alloc:.0f}s")
            f_res, obj_res = run_lp_optimize(
                f_init, time_limit=alloc, top_k_init=2*n_trial-1,
                top_k_max=2*n_trial-1, use_spectral=False,
                label=f"d{n_trial}", max_stuck=40)
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [dense n={n_trial}]: {real_score:.6f}")
            print(f"  Done n={n_trial}: score={real_score:.6f}")

    # ============================================================
    # Phase 2: LP for medium n with spectral directions
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP medium n (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 30:
        n_parent = len(parent)
        med_n_list = []
        for factor in [0.35, 0.5, 0.7, 0.85, 1.0]:
            n_m = int(n_parent * factor)
            n_m = max(256, min(1200, n_m))
            if n_m not in med_n_list:
                med_n_list.append(n_m)
        med_n_list.sort()

        per_n = t2 / len(med_n_list)
        for n_trial in med_n_list:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            alloc = min(per_n, rem * 0.85)
            if alloc < 10:
                break
            f_init = interp_sequence(best_seq, n_trial)
            print(f"  LP n={n_trial}, budget={alloc:.0f}s")
            f_res, obj_res = run_lp_optimize(
                f_init, time_limit=alloc, top_k_init=60,
                top_k_max=min(400, 2*n_trial-1), use_spectral=True,
                label=f"m{n_trial}", max_stuck=50)
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [med n={n_trial}]: {real_score:.6f}")
            print(f"  Done n={n_trial}: score={real_score:.6f}")

    # ============================================================
    # Phase 3: LP for large n (around parent length)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: LP large n (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.30, 500.0)
    if t3 > 30:
        n_base = len(best_seq)
        large_n_list = []
        for factor in [0.85, 1.0, 1.1, 1.3, 1.5]:
            n_l = int(n_base * factor)
            n_l = max(512, min(2500, n_l))
            if n_l not in large_n_list:
                large_n_list.append(n_l)
        large_n_list.sort()

        per_n = t3 / len(large_n_list)
        for n_trial in large_n_list:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            if abs(n_trial - n_base) < n_base * 0.15:
                alloc = min(per_n * 1.5, rem * 0.85)
            else:
                alloc = min(per_n, rem * 0.85)
            if alloc < 12:
                break
            f_init = interp_sequence(best_seq, n_trial)
            print(f"  LP n={n_trial}, budget={alloc:.0f}s")
            f_res, obj_res = run_lp_optimize(
                f_init, time_limit=alloc, top_k_init=80,
                top_k_max=min(600, 2*n_trial-1), use_spectral=True,
                label=f"L{n_trial}", max_stuck=60)
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [large n={n_trial}]: {real_score:.6f}")
            print(f"  Done n={n_trial}: score={real_score:.6f}")

    # ============================================================
    # Phase 4: Deep LP on current best (highest top_k)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep LP (best: {best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 15:
        n_b = len(best_seq)
        print(f"  Deep LP n={n_b}, budget={t4:.0f}s")
        f_res, obj_res = run_lp_optimize(
            best_seq, time_limit=t4, top_k_init=120,
            top_k_max=min(800, 2*n_b-1), use_spectral=True,
            label="deep", max_stuck=80)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [deep]: {real_score:.6f}")

    # ============================================================
    # Phase 5: Subgradient polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: SG polish (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    if t5 > 10:
        n_b = len(best_seq)
        print(f"  SG n={n_b}, budget={t5:.0f}s")
        f_res, obj_res = subgradient_polish(
            best_seq, time_limit=t5, lr_init=0.002,
            n_top=max(30, n_b // 10))
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [sg]: {real_score:.6f}")

    # ============================================================
    # Phase 6: Coordinate descent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: CD (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t6 = min(rem * 0.04, 50.0)
    if t6 > 5:
        n_b = len(best_seq)
        if n_b <= 800:
            print(f"  CD n={n_b}, budget={t6:.0f}s")
            f_res, obj_res = coordinate_descent_fast(best_seq, time_limit=t6, max_passes=3)
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [cd]: {real_score:.6f}")
        else:
            print(f"  Skipping CD (n={n_b} too large)")

    # ============================================================
    # Phase 7: Final burst - LP with max peaks
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 7: Final LP burst (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t7 = rem
    if t7 > 10:
        n_b = len(best_seq)
        print(f"  Final LP n={n_b}, budget={t7:.0f}s")
        f_res, obj_res = run_lp_optimize(
            best_seq, time_limit=t7, top_k_init=150,
            top_k_max=min(1000, 2*n_b-1), use_spectral=True,
            label="final", max_stuck=100)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [final]: {real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    total_time = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={total_time:.0f}s)")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
