import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_subgradient(a):
    """Compute subgradient of max_k conv(a,a)_k.
    The subgradient at the peak k* is: grad[j] = 2*a[k*-j] if 0<=k*-j<n else 0
    """
    n = len(a)
    conv = fast_convolve(a)
    k_star = int(conv.argmax())
    grad = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    indices = np.arange(j_min, j_max + 1)
    grad[indices] = 2.0 * a[k_star - indices]
    return grad, conv.max()


def momentum_subgradient(a0, time_limit, lr=0.1, beta=0.9, verbose=True):
    """Momentum subgradient method with softmax parameterization."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    # Normalize
    z -= z.max()
    a = np.exp(z)
    a /= a.sum()
    
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    lr_current = lr
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        grad, _ = compute_subgradient(a)
        v = beta * v + (1 - beta) * grad
        # Decay learning rate
        lr_current = lr / (1 + 0.01 * iters)
        z -= lr_current * v
        z -= z.max()  # numerical stability
        a_new = np.exp(z)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new /= s
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        if verbose and iters % 500 == 0:
            print(f"    MomSub iter={iters}, score={best_score:.8f}, lr={lr_current:.6f}", end="")
            print(f", time={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def adam_subgradient(a0, time_limit, lr=0.05, beta1=0.9, beta2=0.999, verbose=True):

    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    z -= z.max()
    a = np.exp(z)
    a /= a.sum()

    v = np.zeros(n)

def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = conv(f, g)."""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                C[k, j] = f[i]
    return C


def solve_best_response_simple(f, M, n_inner_iters=10):
    """Simple projected gradient for best response LP.
    max sum(g) s.t. C(f)g <= M, g >= 0
    Using projected subgradient (dual ascent on the constraints).
    """
    n = len(f)
    C = build_conv_matrix(f)
    # Dual: min sum_k M * y_k s.t. C^T y >= 1, y >= 0
    # This is equivalent. Let's use the primal with projected gradient.
    # g = argmax sum(g) s.t. Cg <= M, g >= 0
    # Use multiplicative weights on the constraints
    
    # Actually, let's just use a simple approach:
    # Initialize g uniformly
    g = np.ones(n)
    g /= n
    
    for _ in range(n_inner_iters):
        Cg = C @ g
        # Find violated constraints
        violated = Cg > M
        if not violated.any():
            break
        # Project: reduce g where constraints are violated
        # Simple: g = g * exp(-eta * (Cg - M)^+)
        eta = 0.1
        penalty = np.maximum(Cg - M, 0)
        g = g * np.exp(-eta * (C.T @ penalty))
        g = np.maximum(g, 1e-30)
        g /= g.sum()
    
    return g

def spectral_flatten(a0, time_limit, alpha=0.3, verbose=True):
    """Iterative spectral flattening with non-negativity projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()

def lbgfs_polish(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on smoothed log-sum-exp objective."""
    n = len(a0)
    
    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s
    
    def z_from_a(a):
        return np.log(np.maximum(a, 1e-30))
    
    def smoothed_objective(z):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def smoothed_gradient(z):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z
    
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.arange(1, n + 1)
    cond = sv - (cssv - 1) / rho > 0
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1) / rho_hat
    return np.maximum(v - theta, 0)

        a = a_new

        if verbose and iters % 2000 == 0:
            print(f"    MomSub iter={iters}, score={best_score:.8f}, lr={lr_current:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def adam_subgradient(a0, time_limit, lr=0.1, beta1=0.9, beta2=0.999, verbose=True):
    """Adam on subgradient with simplex projection."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        grad, _ = compute_subgradient(a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** iters)
        v_hat = v / (1 - beta2 ** iters)
        a_trial = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_trial)
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 2000 == 0:
            print(f"    Adam iter={iters}, score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def fista_subgradient(a0, time_limit, verbose=True):
    """FISTA-style accelerated subgradient (non-convex extension)."""
    n = len(a0)
    x = project_simplex(a0.astype(np.float64).copy())
    y = x.copy()
    best_a = x.copy()
    best_score = compute_score(x)
    t0 = time.monotonic()
    t_k = 1.0
    # Estimate Lipschitz constant via powers of 2
    L = 8.0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        grad, _ = compute_subgradient(y)
        step = 1.0 / L
        x_new = y - step * grad
        x_new = project_simplex(x_new)

        # Check if step was too large (non-convex, so just use heuristic)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_simplex(y)
        x = x_new
        t_k = t_new

        # Occasionally increase L to be more conservative
        if iters % 5000 == 0:
            L *= 1.1

        score = compute_score(x)
        if score < best_score:
            best_score = score
            best_a = x.copy()

        if verbose and iters % 2000 == 0:
            print(f"    FISTA iter={iters}, score={best_score:.8f}, L={L:.1f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def spectral_flatten(a0, time_limit, alpha=0.2, verbose=True):
    """Iterative spectral flattening: equalize |FFT(a)|^2 across bins, project to non-negativity."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Target uniform power for non-DC bins
        non_dc = power[1:]
        target = non_dc.mean()

        # Adjust magnitudes toward target
        mags = np.sqrt(power)
        new_mags = mags.copy()
        new_mags[1:] = alpha * mags[1:] + (1 - alpha) * np.sqrt(target)
        # Preserve phases
        phases = np.angle(A)
        A_new = new_mags * np.exp(1j * phases)
        A_new[0] = A[0]  # Keep DC fixed

        a_new = irfft(A_new, fft_size)[:n]
        a_new = project_simplex(a_new)

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 200 == 0:
            print(f"    Spectral iter={iters}, score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def lp_fictitious_play_v2(a0, time_limit, step=0.3, verbose=True):
    """LP fictitious play with toeplitz matrix and iterative best response."""
    from scipy.linalg import toeplitz
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    def solve_br(f, M):
        nn = len(f)
        first_col = np.zeros(2 * nn - 1)
        first_col[:nn] = f
        C = toeplitz(first_col, f[:1])
        g = np.ones(nn) / nn
        for _ in range(30):
            Cg = C @ g
            if (Cg <= M + 1e-10).all():
                break
            violations = np.maximum(Cg - M, 0)
            penalty = C.T @ violations
            g = g * np.exp(-0.3 * penalty)
            g = np.maximum(g, 1e-30)
            g /= g.sum()
        return g

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_br(a, M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = project_simplex((1 - step) * a + step * g_norm)
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 100 == 0:
            print(f"    LP2 iter={iters}, score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)

    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    def z_from_a(a):
        return np.log(np.maximum(a, 1e-30))

    def smoothed_objective(z):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def smoothed_gradient(z):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        a_out = project_simplex(a_out)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=True):
    """Pairwise coordinate descent with multiple trial transfers."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]
            best_local = best_score
            best_t = 0.0
            for frac in [0.05, 0.1, 0.2, 0.4, 0.6, 0.9]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = project_simplex(a_trial)
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = project_simplex(a)
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 30:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)

        if verbose and iters % 100 == 0:
            print(f"    CD iter={iters}, score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

    5. Spectral flattening (frequency domain)
    6. L-BFGS smoothing polish
    7. Coordinate descent (local refinement)

    best_seq = None

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)

    n_parent = len(parent)
    print(f"\n=== Starting multi-phase optimization (best={best_score:.8f}) ===")

    # Phase 1: Momentum subgradient - fast iterations, good for large improvements
    print(f"\n--- Phase 1: Momentum Subgradient ---")
    rem = deadline - time.monotonic()

        best_seq, best_score = check_and_update(a_fista, "fista", best_seq, best_score)

    # Phase 5: Spectral flattening (new approach)
    print(f"\n--- Phase 5: Spectral Flattening ---")

        t6 = min(rem * 0.10, 80.0)

    rem = deadline - time.monotonic()
    if rem > 30:

        a_final, sc_final = adam_subgradient(best_seq, t10, lr=0.05, verbose=False)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
