def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Select k peaks from b using a diversity criterion.
    Balances peak magnitude with spread across the convolution.
    """

    # b[i] * (distance to nearest selected)
    candidates = np.arange(n)

def solve_lp_direction(f, indices, M):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)(k) <= M for k in indices.
    Returns g or None on failure.
    """
    import cvxpy as cp
    n = len(f)
    K = len(indices)
    
    # Build constraint matrix: for each k, (f*g)(k) = sum_j f[j] * g[k-j]
    # = dot product of f reversed/shifted with g
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        # (f*g)(k) = sum_j f[j] * g[k-j] where 0<=j<n and 0<=k-j<n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # g index = k - j, so as j goes from lo to hi, g index goes from k-lo to k-hi
        g_lo = k - hi
        g_hi = k - lo
        A[i, g_lo:g_hi + 1] = f[lo:hi + 1][::-1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        if K * n <= 2000000:
            prob.solve(solver=cp.ECOS, max_iters=50000, abs_acceleration=1e-10)
        else:

    except Exception:

            return None

def golden_section_line_search(f, direction, current_obj, n_eval=30):
    """
    Golden section search for step size t in [0, 1].
    Minimize obj((1-t)*f + t*direction) projected to simplex.
    """
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    def obj_at(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-15:
            return np.inf
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()
    
    a, b = 0.0, 1.0
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(n_eval - 2):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = obj_at(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = obj_at(d)
    
    t_best = (a + b) / 2
    obj_best = obj_at(t_best)
    return t_best, obj_best

            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sel
            w = w_new

        g = np.zeros(n)
        active = np.where(w > 1e-14)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * f[k - idx]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 40 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.015)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.9
                    no_improve = 0

            if it % 150 == 0:
                temp = min(temp * 1.2, 0.4)

            if no_improve > 20:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

    return best_f, 2.0 * n * best_obj


def pairwise_swap_local_search(f_init, time_limit, n_trials=5000):
    """Try random pairwise swaps to improve the objective."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    rng = np.random.default_rng(42)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        f[i], f[j] = f[j], f[i]
        s = f.sum()
        if s < 1e-15:
            f[i], f[j] = f[j], f[i]
            continue
        f_norm = f / s
        obj = fast_convolve(f_norm).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_score = 2.0 * n * obj
        else:
            f[i], f[j] = f[j], f[i]

    return f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src = np.asarray(src, dtype=np.float64)
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def cosine_modulated_start(n, freq=1.0, amplitude=0.3):
    """Create a cosine-modulated uniform sequence as a diverse start."""
    j = np.arange(n, dtype=np.float64)
    base = np.ones(n) / n
    mod = 1.0 + amplitude * np.cos(2.0 * np.pi * freq * j / n)
    seq = base * mod
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s > 1e-15:
        seq = seq / s
    else:
        seq = np.ones(n) / n
    return seq


def structured_random_start(n, seed=0, smoothness=0.8):
    """Create a smooth random sequence as a diverse start."""
    rng = np.random.default_rng(seed)
    coarse = rng.random(max(4, n // 10))
    full = np.interp(np.linspace(0, len(coarse) - 1, n), np.arange(len(coarse)), coarse)
    # Smooth
    kernel_size = max(3, n // 20)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.ones(kernel_size) / kernel_size
    full = np.convolve(full, kernel, mode='same')
    full = np.maximum(full, 0.0)
    s = full.sum()
    if s > 1e-15:
        full = full / s
    else:
        full = np.ones(n) / n
    return full


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: diverse starts, LP with diversity peaks,
    n exploration, subgradient polish, and pairwise swap local search.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print("No parent, using uniform n=512")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    def try_update(f_res):
        nonlocal best_seq, best_score
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST: score={real_score:.6f} n={len(f_res)}")
        return real_score

    # ============================================================
    # Phase 1: Diverse starts with short LP optimization
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Diverse starts (target: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)

    if t1 > 30:
        # Generate diverse starts at the parent's length
        starts = []
        starts.append(("parent_interp", parent.copy()))
        starts.append(("cosine1", cosine_modulated_start(n_parent, freq=1.0, amplitude=0.3)))
        starts.append(("cosine2", cosine_modulated_start(n_parent, freq=2.0, amplitude=0.2)))
        starts.append(("smooth_rand1", structured_random_start(n_parent, seed=1, smoothness=0.8)))
        starts.append(("smooth_rand2", structured_random_start(n_parent, seed=2, smoothness=0.6)))

        per_start = t1 / len(starts)
        for name, f_init in starts:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            alloc = min(per_start, rem * 0.9)
            if alloc < 8:
                break
            print(f"  Start '{name}', budget={alloc:.0f}s")
            f_res, _ = run_lp_optimization(
                f_init, time_limit=alloc, top_k_init=40,
                top_k_max=min(300, 2 * n_parent - 1),
                use_diversity=True, label=name)
            try_update(f_res)

    # ============================================================
    # Phase 2: LP for various n values
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: n exploration (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 450.0)

    if t2 > 30:
        n_base = len(best_seq)
        n_candidates = []
        for factor in [0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0, 2.5]:
            n_c = int(n_base * factor)
            n_c = max(128, min(4000, n_c))
            if n_c not in n_candidates:
                n_candidates.append(n_c)
        n_candidates.sort()

        # Weight allocation: give more time to n values closer to current best
        weights = []
        for n_c in n_candidates:
            w = 1.0 / (1.0 + abs(n_c - len(best_seq)) / len(best_seq))
            weights.append(w)
        total_w = sum(weights)
        weights = [w / total_w for w in weights]

        for i, n_c in enumerate(n_candidates):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            alloc = max(10.0, weights[i] * t2 * 0.9)
            alloc = min(alloc, rem * 0.85)
            if alloc < 8:
                break

            f_init = interp_sequence(best_seq, n_c)
            tk_max = min(500, 2 * n_c - 1)
            print(f"  LP n={n_c}, budget={alloc:.0f}s")
            f_res, _ = run_lp_optimization(
                f_init, time_limit=alloc, top_k_init=50,
                top_k_max=tk_max, use_diversity=True,
                label=f"n{n_c}")
            try_update(f_res)

    # ============================================================
    # Phase 3: Deep LP on current best
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Deep LP (best: {best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)

    if t3 > 20:
        n_b = len(best_seq)
        print(f"  Deep LP n={n_b}, budget={t3:.0f}s")
        f_res, _ = run_lp_optimization(
            best_seq, time_limit=t3, top_k_init=80,
            top_k_max=min(700, 2 * n_b - 1),
            use_diversity=True, label="deep",
            momentum_beta=0.4, max_stuck=80)
        try_update(f_res)

    # ============================================================
    # Phase 4: Subgradient polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: SG polish (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 150.0)

    if t4 > 10:
        n_b = len(best_seq)
        print(f"  SG n={n_b}, budget={t4:.0f}s")
        f_res, _ = subgradient_polish(
            best_seq, time_limit=t4, lr_init=0.003,
            n_top=max(50, n_b // 8))
        try_update(f_res)

    # ============================================================
    # Phase 5: Pairwise swap local search
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Pairwise swap (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)

    if t5 > 5:
        print(f"  Pairwise swap n={len(best_seq)}, budget={t5:.0f}s")
        f_res, _ = pairwise_swap_local_search(best_seq, time_limit=t5)
        try_update(f_res)

    # ============================================================
    # Phase 6: Final LP burst with high peak count
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final LP (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t6 = max(0.0, rem * 0.95)

    if t6 > 10:
        n_b = len(best_seq)
        print(f"  Final LP n={n_b}, budget={t6:.0f}s")
        f_res, _ = run_lp_optimization(
            best_seq, time_limit=t6, top_k_init=120,
            top_k_max=min(900, 2 * n_b - 1),
            use_diversity=True, label="final",
            momentum_beta=0.35, max_stuck=100)
        try_update(f_res)

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy() if parent is not None else np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
