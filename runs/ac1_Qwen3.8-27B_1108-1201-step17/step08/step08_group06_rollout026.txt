Actually, I want to add one more novel idea: **spectral shaping**.

The insight: in the frequency domain, the auto-convolution corresponds to |A(ω)|^2 where A is the DFT of a. To minimize the peak of the time-domain auto-convolution, we want |A(ω)|^2 to be as flat as possible (by the uncertainty principle / Wiener-Khinchin theorem).

Algorithm:
1. Start with sequence a
2. Compute A = DFT(a) (zero-padded to size 2n-1)
3. Compress the spectrum: |A_new| = |A|^α for α < 1 (e.g., α = 0.7)
4. Compute a_new = IDFT(A_new * exp(i*phase(A)))
5. Clip: a_new = max(a_new, 0)
6. Normalize
7. Iterate

This is similar to the "iterative amplitude modification" (IAM) algorithm used in crystallography and holography.

The parameter α controls the aggressiveness:
- α = 1: no change
- α → 0: completely flat spectrum (uniform auto-conv)
- α < 1: partial compression

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Score for normalized sequence (sum=1): 2*n*max_conv."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def softmax(z):
    z_max = z.max()
    e = np.exp(z - z_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_smoothed_objective_and_grad_z(z, T):
    """
    Compute the smoothed objective and its gradient w.r.t. z (log-space).
    F = 2n * T * log(sum_k exp(c_k / T)) where c = a*a, a = softmax(z).
    Returns (objective, grad_z).
    """
    n = len(z)
    a = softmax(z)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Auto-convolution
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)[:size]
    
    # Softmax weights
    c_max = c.max()
    w = np.exp((c - c_max) / T)
    w_sum = w.sum()
    w = w / w_sum
    
    # Objective: 2n * T * log_sum_exp = 2n * T * (c_max + log(w_sum))
    obj = 2.0 * n * T * (c_max + np.log(w_sum))
    
    # h[j] = sum_k w[k] * (a[k-j] + a[j-k])
    # Part 1: sum_k w[k] * a[k-j] = conv(a[::-1], w)[n-1+j]
    # Part 2: sum_k w[k] * a[j-k] = conv(w, a)[j]
    a_flipped = a[::-1]
    AF = rfft(a_flipped, fft_size)
    W = rfft(w, fft_size)
    Afull = rfft(a, fft_size)
    
    conv1 = irfft(AF * W, fft_size)[:size]
    conv2 = irfft(W * Afull, fft_size)[:size]
    
    h = conv1[n-1:n-1+n] + conv2[:n]
    
    # grad_z[j] = 2n*T * a[j] * (h[j] - <a,h>)
    ah = np.dot(a, h)
    grad_z = 2.0 * n * T * a * (h - ah)
    
    return obj, grad_z

def nesterov_optimize(z0, time_limit, T_schedule, lr0=0.1, mu=0.9, verbose=True):
    """
    Nesterov accelerated gradient in log-space.
    T_schedule: callable(t_frac) -> T, or a list of (t_frac, T) pairs for interpolation.
    """
    z = z0.copy().astype(np.float64)
    z_prev = z.copy()
    velocity = np.zeros_like(z)
    n = len(z)
    
    t0 = time.monotonic()
    best_z = z.copy()
    best_obj = compute_score(softmax(z))
    iters = 0
    
    # Normalize z (remove mean for numerical stability)
    z = z - z.mean()
    z_prev = z_prev - z_prev.mean()
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        t_frac = (time.monotonic() - t0) / time_limit
        T = T_schedule(t_frac)
        
        # Nesterov lookahead
        y = z + mu * (z - z_prev)
        
        # Compute gradient at lookahead point
        obj_y, grad_y = compute_smoothed_objective_and_grad_z(y, T)
        
        # Update
        velocity = mu * velocity - lr0 * grad_y
        z_new = z + velocity
        
        # Numerical stability: normalize z
        z_new = z_new - z_new.mean()
        
        # Check improvement
        z_prev = z
        z = z_new
        
        if iters % 100 == 0:
            a = softmax(z)
            sc = compute_score(a)
            if sc < best_obj:
                best_obj = sc
                best_z = z.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NG iter={iters}, T={T:.2f}, obj={obj_y:.8f}, best={best_obj:.8f}, elapsed={elapsed:.0f}s")
        
        # If the objective is getting very large, reset velocity
        if not np.isfinite(obj_y):
            velocity = np.zeros_like(z)
            z = best_z.copy()
            z_prev = z.copy()
    
    return softmax(best_z), best_obj

def spectral_shaping(a0, n_iters=50, alpha=0.7):
    """Iterative amplitude modification for flat spectrum."""
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for _ in range(n_iters):
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude
        mag_new = mag ** alpha
        A_new = mag_new * np.exp(1j * phase)
        
        a = irfft(A_new, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    
    return a

    f = f.copy()

    t0 = time.monotonic()
    
    f_avg = f.copy()
    
    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fast_convolve(f)
        M = conv_f.max()
        
        # Build constraint matrix (sparse pattern)
        # C[k,j] = f[k-j] if 0<=k-j<n else 0
        c_vec = -np.ones(n)
        A_ub = np.zeros((2*n-1, n))

        b_ub = np.full(2*n-1, M)
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if res.status != 0 or res.x is None:
            break
        
        g = np.maximum(res.x, 0)
        if g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        # Geometric mean update
        t_step = 2.0 / (k + 2)
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()
        
        f_avg = ((k * f_avg) + a_new) / (k + 1)

        sc_avg = compute_score(f_avg)

            best_score = sc_cur
            best_f = a_new.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_f = f_avg.copy()
        
        f = a_new
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={k}, M={M:.6e}, best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_score
