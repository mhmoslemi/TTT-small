Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines FISTA-accelerated subgradient, iterative ammunition algorithm (IAA) 
variants, cutting-plane LP, and multi-coordinate descent for final polish.
Novel elements: adaptive spectral water-filling, multi-scale peak shaving,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, b, n_peaks=50, beta=3.0):
    """Compute subgradient of max(conv(f,f)) with softmax weighting."""
    n = len(f)
    if n_peaks < len(b):
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    else:
        top_idx = np.arange(len(b))
    
    b_top = b[top_idx].copy()
    b_max = b_top.max()
    # Sharper weighting on the actual peak
    w = np.exp(beta * (b_top - b_max) / max(b_max, 1e-15))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)

        hi = min(n - 1, k)
        g[lo:hi + 1] += w[i] * 2.0 * f[k - lo: k - lo + (hi - lo + 1)]
    
    return g

def iaa_variant(f_init, n_iters=200, shave_fraction=0.05, floor_frac=0.1, 
                sharpen=1.0):
    """
    IAA variant with adjustable peak shaving.
    shave_fraction: fraction of autocorrelation values to shave
    floor_frac: floor as fraction of peak
    """
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        
        peak = R.max()
        if peak < 1e-15:
            break
        
        # Shave: reduce values above a threshold
        threshold = peak * (1.0 - shave_fraction)
        R_modified = R.copy()
        mask = R_modified > threshold
        if mask.any():
            # Smoothly reduce: R_mod = threshold + (R - threshold) * floor_frac
            R_modified[mask] = threshold + (R_modified[mask] - threshold) * floor_frac
        
        # Ensure R_modified has the right total energy
        # sum(R) = (sum(f))^2 = 1 for normalized f
        total = R.sum()
        mod_total = R_modified.sum()
        if mod_total > 1e-15:
            R_modified *= total / mod_total
        
        # Get new amplitude spectrum
        R_mod_fft = rfft(R_modified, fft_size)
        new_power = np.maximum(R_mod_fft.real, 0.0)
        new_amp = np.sqrt(new_power)
        
        # Update phase
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            break
    
    return f


def spectral_water_filling(f_init, n_iters=100, n_bands=64, alpha=0.1):
    """
    Water-filling in frequency domain: distribute spectral power more evenly
    to flatten the autocorrelation.
    """
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    n_freq = fft_size // 2 + 1
    
    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        
        # Compute spectral power in bands
        power = np.abs(F) ** 2
        band_size = max(1, n_freq // n_bands)
        
        # For each band, compute its contribution to the peak autocorrelation
        # and redistribute
        for band in range(n_bands):
            lo = band * band_size
            hi = min((band + 1) * band_size, n_freq)
            if lo >= hi:
                continue
            band_power = power[lo:hi].sum()
            mean_power = power.sum() / n_bands
            
            # If this band has too much power, reduce it
            if band_power > mean_power * (1 + alpha):
                scale = (mean_power / band_power) ** 0.5
                F[lo:hi] *= scale
            elif band_power < mean_power * (1 - alpha):
                scale = (mean_power / band_power) ** 0.3  # gentler increase
                F[lo:hi] *= scale
        
        f = irfft(F, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
    
    return f


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix for LP."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)[k] <= M for k in active_idx."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
    except Exception:
        g = None
    if g.value is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=50000)
            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def lp_cutting_planes(f_init, time_limit, initial_k=50, max_k=400, label="lp"):
    """Cutting-plane LP with adaptive constraint management."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s
    best_f = f.copy()
    
    b = fast_convolve(f)
    M = b.max()
    best_obj = M
    
    active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    active_idx = np.sort(active_idx)
    
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        g = solve_lp_direction(f, n, M, active_idx)
        
        if g is None:
            stuck += 1
            if stuck > 10:
                break
            if len(active_idx) > initial_k:
                b_cur = fast_convolve(f)
                active_idx = np.argpartition(b_cur, -initial_k)[-initial_k:]
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue
        stuck = 0
        
        # Cutting plane check
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        F_f = rfft(f, fft_size)
        F_g = rfft(g, fft_size)
        fg = irfft(F_f * F_g, fft_size)[:size]
        
        violated = fg > M * (1 + 1e-7)
        if violated.any() and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(20, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp_direction(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()
        
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                break
            continue
        
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = M
        for t in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            f_trial = np.maximum((1 - t) * f + t * h, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        # Golden section
        lo = max(0.0, best_t - 0.05)
        hi = min(1.0, best_t + 0.1)
        for _ in range(8):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
            sa1 = fa1.sum()
            fa1 = fa1 / sa1 if sa1 > 1e-10 else f
            fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
            sa2 = fa2.sum()
            fa2 = fa2 / sa2 if sa2 > 1e-10 else f
            o1 = fast_convolve(fa1).max()
            o2 = fast_convolve(fa2).max()
            if o1 < o2:
                hi = m2
            else:
                lo = m1
        t_mid = (lo + hi) / 2.0
        f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
        s_mid = f_mid.sum()
        if s_mid > 1e-10:
            f_mid /= s_mid
            o_mid = fast_convolve(f_mid).max()
            if o_mid < best_t_obj:
                best_t_obj = o_mid
                best_t = t_mid
        
        f = np.maximum((1 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        f = f / s if s > 1e-10 else np.ones(n) / n
        
        cur_obj = fast_convolve(f).max()
        M = cur_obj
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
        
        # Refresh active set periodically
        if it % 20 == 0:
            b = fast_convolve(f)
            active_idx = np.argpartition(b, -max_k)[-max_k:]
            active_idx = np.sort(active_idx)
        
        if it % 30 == 0:
            print(f"    {label} it={it} score={2*n*best_obj:.6f} active={len(active_idx)} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def coordinate_descent(f_init, time_limit, n_sweeps=50):
    """
    Multi-step coordinate descent with golden section per coordinate.
    Scales with problem size.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    
    # For large n, step over coordinates
    step = max(1, n // 500)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            
            orig_val = f[j]
            if orig_val < 1e-15:
                continue
            
            best_val = orig_val
            best_local = best_obj
            
            # Try multiplicative updates
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                f[j] = orig_val * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = orig_val
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_local - 1e-14:
                    best_local = obj
                    best_val = orig_val * mult
            
            f[j] = best_val
            s = f.sum()
            f = f / s if s > 1e-10 else f
            if best_local < best_obj - 1e-14:
                best_obj = best_local
                improved = True
        
        if not improved:
            break
    
    return f, best_obj


def fine_coordinate_descent(f_init, time_limit):
    """
    Very fine coordinate descent: try to adjust each coordinate with 
    golden section search.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s
    best_obj = fast_convolve(f).max()
    
    for j in range(n):
        if time.monotonic() - t0 > time_limit:
            break
        if f[j] < 1e-15:
            continue
        
        # Golden section search for optimal value of f[j]
        # Keep other coordinates fixed (in normalized space)
        # f[j] ranges from 0 to f[j]*3 (or so)
        # But we need to renormalize after each change
        
        lo = max(0.0, f[j] * 0.3)
        hi = f[j] * 3.0
        best_val = f[j]
        best_local = best_obj
        
        # First, scan a few points
        for trial in np.linspace(0, f[j] * 4, 10):
            f[j] = trial
            s = f.sum()
            if s < 1e-10:
                continue
            f_norm = f / s
            obj = fast_convolve(f_norm).max()
            if obj < best_local - 1e-14:
                best_local = obj
                best_val = trial
        
        f[j] = best_val
        s = f.sum()
        f = f / s if s > 1e-10 else f
        best_obj = best_local
    
    return f, best_obj
