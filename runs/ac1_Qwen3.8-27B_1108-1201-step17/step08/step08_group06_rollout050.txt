"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping (constant modulus algorithm variant) for global optimization
2. Adam gradient descent with adaptive soft-max smoothing for local refinement  
3. Targeted peak reduction via coordinate moves with golden-section search
4. Multi-initialization with cosine modulation
5. Length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for sequence (assumed non-negative)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def softmax_vec(x, T):
    """Stable softmax of x/T."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted / T)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def grad_smoothed(a, T):
    """
    Compute value and gradient of f_T(a) = 2n * soft_max(conv(a,a), T).
    soft_max(x, T) = T * log(sum exp(x_k / T))
    Gradient: 4n * sum_j a[j] * w[i+j] where w = softmax(conv, T)
    """
    n = len(a)
    conv = fast_convolve(a)
    w = softmax_vec(conv, T)
    
    # grad[i] = 4n * dot(a, w[i:i+n])
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 4.0 * n * np.dot(a, w[i:i+n])
    
    # Objective value
    conv_shifted = conv - conv.max()
    log_sum_exp = conv.max() + T * np.log(np.sum(np.exp(conv_shifted / T)))
    f_val = 2.0 * n * log_sum_exp
    
    return f_val, grad


def adam_step(params, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Standard Adam update. Returns new params, m, v."""
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad**2
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    params_new = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params_new, m, v


def spectral_shaping(a, alpha=0.5, time_limit=60.0, verbose=True):
    """
    Constant modulus algorithm variant: iteratively flatten the power spectrum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        A = rfft(a, fft_size)
        mag = np.abs(A)
        # Target: flat spectrum
        target = np.sqrt(np.sum(mag**2) / len(mag))
        
        # Geometric interpolation toward flat
        mag_new = mag * (target / np.maximum(mag, 1e-30))**alpha
        
        # Preserve the phase
        phase = A / np.maximum(mag, 1e-30)
        A_new = mag_new * phase
        
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        
        # Also try the original phase with new magnitude
        a = a_new
        it += 1
        
        if verbose and it % 50 == 0:
            print(f"    Spectral iter={it}, score={best_score:.8f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a


def adam_optimize(a0, time_limit, T0=1.0, T_min=0.001, lr=1e-3, 
                  T_decay=0.995, verbose=True):
    """
    Adam gradient descent with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    T = T0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        f_val, grad = grad_smoothed(a, T)
        
        a_new, m, v = adam_step(a, grad, m, v, t, lr)
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Keep old a but still update Adam state
            a_new = a  # revert
            # Actually, let's always use the new a but track best separately
            a = a_new  # use new a for next iteration regardless
        
        # Decay temperature
        T = max(T_min, T * T_decay)
        
        if verbose and t % 100 == 0:
            print(f"    Adam iter={t}, T={T:.6f}, score={best_score:.8f}, "
                  f"t={time.monotonic()-t0:.1f}s")
    
    return best_a


def targeted_peak_reduction(a0, time_limit, verbose=True):
    """
    Identify top peaks and reduce them via coordinate moves.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Find top peaks
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Contributing pairs: a[j] * a[pk-j] for j = max(0, pk-n+1) ... min(n-1, pk)
            j_start = max(0, int(pk) - n + 1)
            j_end = min(n - 1, int(pk))
            
            if j_start > j_end:
                continue
            
            contribs = a[j_start:j_end+1] * a[int(pk)-j_end:int(pk)-j_start+1][::-1]
            
            # Find top contributing index
            top_local = np.argmax(contribs)
            i_top = j_start + top_local
            
            if a[i_top] < 1e-14:
                continue
            
            # Try reducing a[i_top] by various amounts
            best_t = 0
            best_f = best_score
            
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                delta = a[i_top] * frac
                a_trial = a.copy()
                a_trial[i_top] -= delta
                if a_trial[i_top] < 0:
                    continue
                # Redistribute to positions that DON'T contribute to this peak
                safe_positions = np.setdiff1d(np.arange(n), 
                    range(j_start, j_end + 1))
                if len(safe_positions) == 0:
                    continue
                # Redistribute proportionally to current values at safe positions
                safe_vals = a_trial[safe_positions]
                safe_sum = safe_vals.sum()
                if safe_sum < 1e-15:
                    # Distribute uniformly
                    a_trial[safe_positions] += delta / len(safe_positions)
                else:
                    a_trial[safe_positions] += delta * safe_vals / safe_sum
                
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_f:
                    best_f = sc
                    best_t = delta
            
            if best_f < best_score - 1e-13:
                delta = best_t
                a_trial = a.copy()
                a_trial[i_top] -= delta
                safe_positions = np.setdiff1d(np.arange(n),
                    range(j_start, j_end + 1))
                safe_vals = a_trial[safe_positions]
                safe_sum = safe_vals.sum()
                if safe_sum > 1e-15:
                    a_trial[safe_positions] += delta * safe_vals / safe_sum
                else:
                    a_trial[safe_positions] += delta / len(safe_positions)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                a = a_trial
                best_score = best_f
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Try random pairwise moves
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-14:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True

                    break
        
        if verbose:

            if int(elapsed) % 10 == 0 and int(elapsed) != getattr(targeted_peak_reduction, '_last_print', -1):

def coord_descent_polish(a0, time_limit, n_pairs=200, verbose=False):
    """
    Pairwise coordinate descent with golden-section search.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        
        # Generate pairs
        pairs = set()
        # Random pairs
        for _ in range(n_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Peak-focused pairs
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        for _ in range(n_pairs // 2):
            i = rng.integers(js, je + 1)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            # Golden section search for transfer t from i to j
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_t(t_val):
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                if a_trial[i] < 0 or a_trial[j] < 0:
                    return 1e10
                a_trial = a_trial / a_trial.sum()
                return compute_score(a_trial)
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)

def cosine_modulated_init(n, freq=1.0, alpha=0.3, seed=0):
    """Create a cosine-modulated initialization."""
    x = np.arange(n)
    base = 1.0 + alpha * np.cos(2 * np.pi * freq * x / n)
    base = np.maximum(base, 0.1)
    base = base / base.sum()
    return base


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
