"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses HiGHS solver for cutting-plane LP, multi-M direction finding,
spectral equalization diversification, adaptive line search, and
coordinate descent polishing. Each phase builds on the best sequence

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    """Build constraint matrix: row i gives (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_idx):
    """
    Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in active_idx.
    Returns optimal g or None.
    """
    K = len(active_idx)
    if K == 0:
        return None
    A = build_A(f, active_idx, n)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if np.all(np.isfinite(g)) and g.sum() > 0:
                return g
    except Exception:
        pass
    return None

        # Normalize direction

            f_trial = np.maximum(f_trial, 0.0)

        # Golden section refinement

            for _ in range(8):

                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)

        else:

def spectral_equalize(f_init, n_iters=300, floor_frac=0.35, label="spec"):
    """
    Spectral equalization: flatten auto-convolution by adjusting |F|.
    Preserves phase, adjusts amplitude to reduce peak.
    """
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    for it in range(n_iters):
        F = rfft(f, fft_size)
        ac = irfft(F * F, fft_size)[:size]
        peak = ac.max()
        if peak < 1e-15:
            break

        # Target: raise the floor of ac
        floor = floor_frac * peak
        ac_target = np.maximum(ac, floor)

        # Preserve total energy
        e_old = ac.sum()
        e_new = ac_target.sum()
        if e_new > 1e-15:
            ac_target *= e_old / e_new

        # Enforce symmetry (auto-convolution is symmetric)
        ac_target = (ac_target + ac_target[::-1].copy()) / 2.0

        # Compute new |F|
        ac_fft = rfft(ac_target, fft_size)
        real_part = np.maximum(ac_fft.real, 0.0)
        new_amp = np.sqrt(real_part)

        # Preserve phase
        abs_F = np.abs(F)
        phase = np.ones_like(F, dtype=complex)
        mask = abs_F > 1e-15
        phase[mask] = F[mask] / abs_F[mask]
        F_new = new_amp * phase
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            break

        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
        f = f_new

        if it % 50 == 0:
            print(f"    [{label}] it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f}")

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, n_top=80):
    """
    Subgradient method focused on top peaks of the auto-convolution.
    Uses momentum and adaptive learning rate.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = 0.001
    momentum = 0.8
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    rng = np.random.default_rng(2024)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Focus on top peaks
        k = min(n_top, len(b))
        top_idx = np.argpartition(b, -k)[-k:]

        # Subgradient: for max_k (f*f)_k, grad wrt f is 2*f reversed at offset k
        # Use weighted combination of top-k peaks
        weights = b[top_idx]
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights /= wsum

        g = np.zeros(n)
        for j, k_idx in enumerate(top_idx):
            if weights[j] < 1e-10:
                continue
            lo = max(0, k_idx - n + 1)
            hi = min(n - 1, k_idx)
            # (f*f)[k_idx] = sum_j f[j] * f[k_idx - j]
            # d/d f[m] = f[k_idx - m] + f[m - k_idx] (if valid)
            # Actually: d/d f[m] of sum_j f[j]*f[k-j] = f[k-m] + f[m-k] (when valid)
            # But since we're computing subgradient of the max, we just need grad at the max point
            # grad of (f*f)[k] wrt f[m] = f[k-m] if 0<=k-m<n, plus f[m-k] if 0<=m-k<n
            # But for auto-conv (not autocorrelation), it's just f[k-m] (one term)
            m_lo = max(0, k_idx - n + 1)
            m_hi = min(n - 1, k_idx)
            ms = np.arange(m_lo, m_hi + 1)
            g[ms] += weights[j] * f[k_idx - ms]

        v = momentum * v - lr * g
        f_new = project_simplex(f + v)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.85
            if no_improve > 100:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0
                lr = 0.001

        if it % 100 == 0:
            print(f"    [sg] it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates on selected coordinates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)
    no_improve_sweeps = 0

    for sweep in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 1.15, 1.3]:
                f[j] = best_val * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f_normed = f / s
                obj = fast_convolve(f_normed).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = best_val * mult
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 3:
                break
        else:
            no_improve_sweeps = 0

    return f, best_obj

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # === Phase 1: Main CP-LP with HiGHS (40% budget) ===
    print(f"\n=== Phase 1: CP-LP HiGHS (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 700.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, _ = cp_lp_optimize(parent, time_limit=t1, initial_k=60, max_k=400, label="cp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [cp1]: {score1:.6f}")

    # === Phase 2: Multi-M LP (20% budget) ===
    print(f"\n=== Phase 2: Multi-M LP (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 350.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, _ = multi_m_lp(best_seq, time_limit=t2, label="mmlp")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [mmlp]: {score2:.6f}")

    # === Phase 3: Spectral equalization + LP polish (15% budget) ===
    print(f"\n=== Phase 3: Spectral + LP (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3_total = min(rem * 0.15, 200.0)
    if t3_total > 30:
        # Spectral equalization
        t3a = t3_total * 0.4
        f_spec = spectral_equalize(best_seq, n_iters=min(300, int(t3a * 50)),
                                   floor_frac=0.35, label="spec")
        # Short LP on spectrally-equalized result
        t3b = t3_total * 0.6
        f3, _ = cp_lp_optimize(f_spec, time_limit=t3b, initial_k=40, max_k=250, label="spec+lp")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [spec+lp]: {score3:.6f}")

    # === Phase 4: Try larger sequence length (15% budget) ===
    print(f"\n=== Phase 4: Larger n (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 30:
        target_n = max(2048, int(n_parent * 1.8))
        # Interpolate/extend
        old_x = np.linspace(0.0, 1.0, n_parent)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  Trying n={target_n}, budget={t4:.0f}s")
        f4, _ = cp_lp_optimize(f_interp, time_limit=t4, initial_k=40, max_k=250, label="big")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [n={target_n}]: {score4:.6f}")

    # === Phase 5: Subgradient polish (10% budget) ===
    print(f"\n=== Phase 5: Subgradient (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 120.0)
    if t5 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}")
        f5, _ = subgradient_polish(best_seq, time_limit=t5, n_top=max(50, n_b // 15))
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg]: {score5:.6f}")

    # === Phase 6: Final CP-LP burst (10% budget) ===
    print(f"\n=== Phase 6: Final LP burst (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 150.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, _ = cp_lp_optimize(best_seq, time_limit=t6, initial_k=80, max_k=500, label="final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {score6:.6f}")

    # === Phase 7: Coordinate descent polish (remaining time) ===
    rem = deadline - time.monotonic()
    if rem > 10:
        print(f"\n=== Phase 7: Coord descent (score={best_score:.6f}) ===")
        t7 = min(rem * 0.5, 60.0)
        f7, _ = coordinate_polish(best_seq, time_limit=t7)
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {score7:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
