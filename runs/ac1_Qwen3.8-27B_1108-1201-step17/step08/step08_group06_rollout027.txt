def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

def fast_convolve(a):

    return irfft(A * A, fft_size)[:size]

    if s < 1e-300:

    grad_half = grad_full[:m].copy()
    if m > 1:
        grad_half[1:] += grad_full[n_full - 1:m - 1:-1]

    return obj, grad_half


def adam_optimize(z0, T_schedule, lr, beta1, beta2, eps, time_limit, verbose=True):
    """
    ADAM optimization in log-space with temperature annealing.
    T_schedule: list of (start_iter_frac, T_value) pairs or a single T.
    Returns (best_z, best_score, avg_a).
    """
    n = len(z0)
    if isinstance(T_schedule, (int, float)):
        T_schedule = [(0.0, T_schedule)]
    else:
        T_schedule = sorted(T_schedule, key=lambda x: x[0])

    z = z0.copy().astype(np.float64)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t_step = 0

    max_iters = 200000
    t0 = time.monotonic()

    # Polyak averaging
    a_avg = softmax_from_z(z).copy()
    n_avg = 0
    warmup = max(100, max_iters // 10)

    best_score = compute_score_normalized(softmax_from_z(z))
    best_z = z.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        t_step += 1

        # Get current T
        T_cur = T_schedule[-1][1]
        for frac, T_val in T_schedule:
            if it / max_iters <= frac:
                T_cur = T_val
                break
        # Actually let's interpolate: T decreases over time
        # Use the schedule entries as breakpoints
        T_cur = T_schedule[0][1]
        for i in range(len(T_schedule) - 1):
            if it / max_iters >= T_schedule[i][0]:
                T_cur = T_schedule[i + 1][1]

        obj, grad_z = smooth_obj_and_grad_z(z, T_cur)

        # ADAM update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)

        # Adaptive learning rate (cosine decay)
        progress = min(1.0, t_step / max_iters)
        lr_cur = lr * (1.0 + np.cos(np.pi * progress)) / 2.0

        z = z - lr_cur * m_hat / (np.sqrt(v_hat) + eps)

        # Get current a
        a_cur = softmax_from_z(z)

        # Update average (Polyak: average over iterates after warmup)
        if t_step > warmup:
            n_avg += 1
            a_avg = (a_avg * (n_avg - 1) + a_cur) / n_avg

        # Track best
        if t_step % 50 == 0:
            sc_cur = compute_score_normalized(a_cur)
            sc_avg = compute_score_normalized(a_avg)
            if sc_cur < best_score:
                best_score = sc_cur
                best_z = z.copy()
            if sc_avg < best_score:
                best_score = sc_avg

        if verbose and t_step % 5000 == 0:
            sc_cur = compute_score_normalized(a_cur)
            print(f"    ADAM iter={t_step}, T={T_cur:.0f}, score_cur={sc_cur:.8f}, "
                  f"best={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")

    # Final check
    a_final = softmax_from_z(z)
    sc_final = compute_score_normalized(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_z = z.copy()

    a_avg_final = compute_score_normalized(a_avg)
    if a_avg_final < best_score:
        best_score = a_avg_final

    # Return the better of best_z and a_avg
    a_best = softmax_from_z(best_z)
    if compute_score_normalized(a_avg) < compute_score_normalized(a_best):
        a_best = a_avg

    return a_best, best_score


def adam_optimize_symmetric(z_half0, T_schedule, lr, beta1, beta2, eps, time_limit, verbose=True):
    """ADAM optimization with palindromic symmetry constraint."""
    m = len(z_half0)
    n_full = 2 * m - 1

    z = z_half0.copy().astype(np.float64)
    mom = np.zeros_like(z)
    vel = np.zeros_like(z)
    t_step = 0

    max_iters = 200000
    t0 = time.monotonic()

    a_half_avg = softmax_from_z(z).copy()
    n_avg = 0
    warmup = max(100, max_iters // 10)

    def get_full_a(zh):
        zh_m = len(zh)
        af = np.zeros(2 * zh_m - 1)
        af[:zh_m] = softmax_from_z(zh)
        if zh_m > 1:
            af[zh_m:] = af[zh_m - 1:0:-1]
        return af / af.sum()

    def get_score(zh):
        return compute_score_normalized(get_full_a(zh))

    best_score = get_score(z)
    best_z = z.copy()

    if isinstance(T_schedule, (int, float)):
        T_schedule = [(0.0, T_schedule)]
    T_schedule = sorted(T_schedule, key=lambda x: x[0])

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        t_step += 1

        T_cur = T_schedule[0][1]
        for i in range(len(T_schedule) - 1):
            if it / max_iters >= T_schedule[i][0]:
                T_cur = T_schedule[i + 1][1]

        obj, grad_half = smooth_obj_and_grad_z_symmetric(z, T_cur)

        mom = beta1 * mom + (1 - beta1) * grad_half
        vel = beta2 * vel + (1 - beta2) * grad_half ** 2
        mom_hat = mom / (1 - beta1 ** t_step)
        vel_hat = vel / (1 - beta2 ** t_step)

        progress = min(1.0, t_step / max_iters)
        lr_cur = lr * (1.0 + np.cos(np.pi * progress)) / 2.0

        z = z - lr_cur * mom_hat / (np.sqrt(vel_hat) + eps)

        a_half_cur = softmax_from_z(z)

        if t_step > warmup:
            n_avg += 1
            a_half_avg = (a_half_avg * (n_avg - 1) + a_half_cur) / n_avg

        if t_step % 50 == 0:
            sc_cur = get_score(z)
            sc_avg = get_score(a_half_avg)
            if sc_cur < best_score:
                best_score = sc_cur
                best_z = z.copy()
            if sc_avg < best_score:
                best_score = sc_avg

        if verbose and t_step % 5000 == 0:
            sc_cur = get_score(z)
            print(f"    ADAM-sym iter={t_step}, T={T_cur:.0f}, score={sc_cur:.8f}, "
                  f"best={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")

    a_best = get_full_a(best_z)
    a_avg_full = get_full_a(a_half_avg)
    if compute_score_normalized(a_avg_full) < compute_score_normalized(a_best):
        a_best = a_avg_full

    return a_best, best_score


def build_C_matrix(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. C*f_cross_g <= M, g >= 0 using HiGHS."""
    n = len(f)
    C = build_C_matrix(f)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            return None
        return np.maximum(res.x, 0)
    except Exception:
        return None


def lp_fictitious_play(f_init, time_limit, use_greedy=True, verbose=True):
    """LP-based fictitious play with geometric mean updates."""
    f = f_init.copy()
    s = f.sum()
    if s < 1e-15:
        return f, compute_score_normalized(f)
    f = f / s
    n = len(f)

    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t0 = time.monotonic()

    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_f = fast_convolve(f)
        M = conv_f.max()

        if use_greedy:
            g = greedy_lp_solve(f, M)
        else:
            g = solve_lp_for_g(f, M)
            if g is None:
                g = greedy_lp_solve(f, M)

        if g is None or g.sum() < 1e-15:
            break

        g = g / g.sum()

        # Geometric mean update in log-space
        t_step = 2.0 / (k + 2)
        if k > 200:
            t_step = max(0.005, 2.0 / (k + 2))

        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()

        f_avg = ((k * f_avg) + a_new) / (k + 1)

        score_cur = compute_score_normalized(a_new)
        score_avg = compute_score_normalized(f_avg)

        if score_cur < best_score:
            best_score = score_cur
            best_f = a_new.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_f = f_avg.copy()

        f = a_new

        if verbose and k % 100 == 0:
            print(f"    LP iter={k}, M={M:.6e}, score={best_score:.8f}, "
                  f"elapsed={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def greedy_lp_solve(f, M):
    """Greedy approximation to the LP."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)

    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:k_end - k_start + 1]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.min(slack[valid] / f_slice[valid])
        max_gj = max(0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj

    return g


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with golden-section search for optimal 2-point transfer."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    no_improve = 0
    outer_iter = 0

    while time.monotonic() - t0 < time_limit:
        outer_iter += 1
        improved = False

        # Generate candidate pairs
        n_pairs = min(n, 1000)
        pairs = set()

        # Random pairs
        for _ in range(n_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        # Peak-focused pairs
        conv = fast_convolve(a)
        top_peaks = np.argsort(conv)[::-1][:5]
        for pk in top_peaks:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for _ in range(20):
                i = rng.integers(js, je + 1)
                j = rng.integers(js, je + 1)
                if i != j:
                    pairs.add((min(i, j), max(i, j)))

        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] + a[j] < 1e-15:
                continue

            # Golden section search for transfer t
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            a_trial = a.copy()

            def eval_transfer(t):
                a_trial[i] = a[i] - t
                a_trial[j] = a[j] + t
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    return 1e10
                a_trial[i] = max(0, a_trial[i])
                a_trial[j] = max(0, a_trial[j])
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial = a_trial / ss
                return compute_score_normalized(a_trial)

            f1 = eval_transfer(x1)
            f2 = eval_transfer(x2)

            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(x2)

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-12:
                a[i] = max(0, a[i] - t_opt)
                a[j] = max(0, a[j] + t_opt)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                    best_score = compute_score_normalized(a)
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                no_improve = 0
        else:
            no_improve = 0

        if verbose and outer_iter % 50 == 0:
            print(f"    CD iter={outer_iter}, score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_initialization(n, seed, style='random'):
    """Create initial sequences of various styles."""
    rng = np.random.default_rng(seed)
    if style == 'random':
        a = rng.random(n)
    elif style == 'bump':
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.5) ** 2) + 0.01
    elif style == 'sinusoidal':
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.01)
    elif style == 'bilinear':
        x = np.linspace(0, 1, n)
        a = np.minimum(x, 1 - x) * 4 + 0.01
    elif style == 'uniform_plus_noise':
        a = np.ones(n) + rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def run_adam_attempt(a_init, n, time_limit, seed, verbose=True):
    """Run one ADAM optimization attempt from a given initialization."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z0 = np.log(a)

    # Temperature schedule: start high, anneal down
    # T controls the "smoothness" of the max approximation
    n_iters_est = time_limit * 50  # rough estimate
    T_schedule = [
        (0.0, 5.0 * n),
        (0.3, 2.0 * n),
        (0.6, 0.5 * n),
        (0.8, 0.1 * n),
        (0.9, 0.02 * n),
        (1.0, 0.005 * n),
    ]

    # Multiple lr values to try
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    a_best, score = adam_optimize(z0, T_schedule, lr, beta1, beta2, eps,
                                   time_limit, verbose=verbose)
    return a_best, score


def run_adam_symmetric_attempt(a_init, time_limit, seed, verbose=True):
    """Run ADAM optimization with symmetry constraint."""
    n = len(a_init)
    # Make palindromic
    m = (n + 1) // 2
    a_half = (a_init[:m] + a_init[n - 1:n - m - 1:-1] if m > 1 else a_init[:1]) / 2.0
    a_half = np.maximum(a_half, 1e-15)
    a_half = a_half / a_half.sum()

    z0 = np.log(a_half)

    n_full = 2 * m - 1
    T_schedule = [
        (0.0, 5.0 * n_full),
        (0.3, 2.0 * n_full),
        (0.6, 0.5 * n_full),
        (0.8, 0.1 * n_full),
        (0.9, 0.02 * n_full),
        (1.0, 0.005 * n_full),
    ]

    a_best, score = adam_optimize_symmetric(z0, T_schedule, 0.01, 0.9, 0.999,
                                             1e-8, time_limit, verbose=verbose)
    return a_best, score


def subgradient_mirror_descent(a_init, time_limit, verbose=True):
    """
    Subgradient mirror descent (multiplicative weights) with Polyak averaging.
    Directly minimizes max(conv) using subgradients.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score_normalized(a)

    # Polyak averaging
    a_avg = a.copy()
    n_avg = 0
    warmup = 200

    # Step size schedule: eta_t = c / sqrt(t)
    c = 0.5
    max_iters = 500000

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        max_val = conv.max()
        peak_idx = np.argmax(conv)

        # Subgradient: g[j] = 2 * a[peak_idx - j] if 0 <= peak_idx - j < n
        g = np.zeros(n)
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        for j in range(js, je + 1):
            g[j] = 2.0 * a[peak_idx - j]

        # Update in log-space (multiplicative weights)
        eta = c / np.sqrt(t)
        z = np.log(a) - eta * g
        z -= z.max()
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()

        # Polyak average
        if t > warmup:
            n_avg += 1
            a_avg = (a_avg * (n_avg - 1) + a_new) / n_avg

        a = a_new

        if t % 1000 == 0:
            sc = compute_score_normalized(a)
            sc_avg = compute_score_normalized(a_avg)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
            if verbose and t % 5000 == 0:
                print(f"    SGD iter={t}, score={sc:.8f}, avg={sc_avg:.8f}, "
                      f"best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining:
    - ADAM with temperature annealing (main workhorse)
    - Symmetry-constrained ADAM
    - Subgradient mirror descent
    - LP fictitious play polish
    - Coordinate descent polish
    - Length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = make_initialization(n_init, seed, 'bump')
        best_seq = parent.copy()
        best_score = compute_score_normalized(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: ADAM with temperature annealing (multiple attempts) ===
    print(f"\n=== Phase 1: ADAM Temperature Annealing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_attempts = 4
    per_attempt_time = rem * 0.35 / n_attempts

    for attempt in range(n_attempts):
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(per_attempt_time, rem * 0.3)
        if t_alloc < 20:
            break

        # Different initializations
        styles = ['random', 'bump', 'sinusoidal', 'uniform_plus_noise']
        style = styles[attempt % len(styles)]
        a_init = make_initialization(n_parent, seed + attempt * 100 + 7, style)

        # Also try starting from parent
        if attempt == 0:
            a_init = parent.copy()

        print(f"  Attempt {attempt}: style={style}, budget={t_alloc:.0f}s")
        a_res, sc = run_adam_attempt(a_init, n_parent, t_alloc, seed + attempt,
                                      verbose=(attempt == 0))
        if sc < best_score:
            best_score = sc
            best_seq = a_res.copy()
            print(f"  NEW BEST: {best_score:.8f}")
        # Verify with evaluate_sequence
        actual_sc = float(evaluate_sequence(a_res.tolist()))
        if np.isfinite(actual_sc) and actual_sc < best_score:
            best_score = actual_sc
            best_seq = a_res.copy()
            print(f"  VERIFIED BEST: {best_score:.8f}")

    # === Phase 2: Symmetry-constrained ADAM ===
    print(f"\n=== Phase 2: Symmetric ADAM (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t_alloc = min(rem * 0.2, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_sym, sc_sym = run_adam_symmetric_attempt(best_seq, t_alloc, seed + 999)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.copy()
            print(f"  NEW BEST (sym): {best_score:.8f}")
        actual_sc = float(evaluate_sequence(a_sym.tolist()))
        if np.isfinite(actual_sc) and actual_sc < best_score:
            best_score = actual_sc
            best_seq = a_sym.copy()

    # === Phase 3: Subgradient Mirror Descent ===
    print(f"\n=== Phase 3: Subgradient Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 60:
        t_alloc = min(rem * 0.25, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, sc_sg = subgradient_mirror_descent(best_seq, t_alloc, verbose=True)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  NEW BEST (SGD): {best_score:.8f}")

    # === Phase 4: LP Fictitious Play Polish ===
    print(f"\n=== Phase 4: LP Fictitious Play Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 60:
        t_alloc = min(rem * 0.3, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t_alloc, use_greedy=True, verbose=True)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  NEW BEST (LP): {best_score:.8f}")
        actual_sc = float(evaluate_sequence(a_lp.tolist()))
        if np.isfinite(actual_sc) and actual_sc < best_score:
            best_score = actual_sc
            best_seq = a_lp.copy()

    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: Coordinate Descent Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem * 0.4, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, verbose=True)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  NEW BEST (CD): {best_score:.8f}")
        actual_sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(actual_sc) and actual_sc < best_score:
            best_score = actual_sc
            best_seq = a_cd.copy()

    # === Phase 6: Second ADAM pass on best found ===
    print(f"\n=== Phase 6: Second ADAM Pass (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 60:
        t_alloc = min(rem * 0.5, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_init2 = best_seq + rng.standard_normal(n_parent) * 0.005
        a_init2 = np.maximum(a_init2, 0)
        a_init2 = a_init2 / a_init2.sum()
        a_res2, sc2 = run_adam_attempt(a_init2, n_parent, t_alloc, seed + 5000,
                                        verbose=False)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a_res2.copy()
            print(f"  NEW BEST (ADAM2): {best_score:.8f}")

    # === Phase 7: Length Exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = [n_parent - 2, n_parent + 2, n_parent - 4, n_parent + 4,
                 n_parent - 8, n_parent + 8]
    for alt_n in nearby_ns:
        if alt_n < 128 or alt_n > 3000:
            continue
        rem = deadline - time.monotonic() - 60
        if rem < 40:
            break
        t_alloc = min(rem * 0.15, 80.0)
        if t_alloc < 20:
            break
        print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
        a_alt = interpolate_sequence(best_seq, alt_n)
        a_res_alt, sc_alt = run_adam_attempt(a_alt, alt_n, t_alloc * 0.7,
                                              seed + alt_n, verbose=False)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_res_alt.copy()
            print(f"  NEW BEST (n={alt_n}): {best_score:.8f}")

    # === Phase 8: Final Coordinate Descent ===
    print(f"\n=== Phase 8: Final CD Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t_alloc = min(rem * 0.6, 100.0)
        a_final, sc_final = coordinate_descent(best_seq, t_alloc, verbose=False)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  FINAL BEST: {best_score:.8f}")

    # === Phase 9: Final LP Polish ===
    rem = deadline - time.monotonic() - 15
    if rem > 15:
        t_alloc = min(rem * 0.7, 50.0)
        a_lp_final, sc_lp_final = lp_fictitious_play(best_seq, t_alloc,
                                                      use_greedy=True, verbose=False)
        if sc_lp_final < best_score:
            best_score = sc_lp_final
            best_seq = a_lp_final.copy()

    # Final verification
    if best_seq is None:
        best_seq = parent
    else:
        best_seq = np.maximum(best_seq, 0)
        ss = best_seq.sum()
        best_seq = best_seq / ss

    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
