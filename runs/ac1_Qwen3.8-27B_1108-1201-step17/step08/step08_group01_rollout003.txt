import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = fast_convolve(a).max()
    return 2.0 * n * M / (s * s)

def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.GLPK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.MOSEK, cp.GUROBI]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val

def line_search(a, b, n, max_iter=40):
    """Find t in [0,1] minimizing score of (1-t)*a + t*b."""
    def obj(t):
        x = (1 - t) * a + t * b
        return score_of(x)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    f_lo = obj(0.0)
    f_hi = obj(1.0)
    
    if f_hi < f_lo:

        lo = 0.5
    # Actually, do full golden section
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1 - gr) * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)

            break
    
    t_best = (lo + hi) / 2.0

            f = f_hi

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based alternating minimization with exact quadratic line search
2. Fast coordinate descent with O(n) sensitivity analysis for polishing
3. Adaptive perturbation to escape local optima

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear cross-convolution of a and b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = float(fast_convolve(a).max())
    return 2.0 * n * M / (s * s)


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.GLPK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.MOSEK, cp.GUROBI]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def line_search_conv(a, b):
    """
    Find t in [0,1] minimizing max_k conv((1-t)*a + t*b, (1-t)*a + t*b).
    Uses precomputed convolutions for O(n) per evaluation.
    Returns (best_t, best_max_conv).
    """
    conv_aa = fast_convolve(a)
    conv_ab = fast_convolve_ab(a, b)
    conv_bb = fast_convolve(b)

    def max_conv(t):
        val = (1.0 - t) * (1.0 - t) * conv_aa + 2.0 * t * (1.0 - t) * conv_ab + t * t * conv_bb
        return float(val.max())

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = lo + (1.0 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = max_conv(c)
    fd = max_conv(d)

    for _ in range(50):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1.0 - gr) * (hi - lo)
            fc = max_conv(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = max_conv(d)
        if hi - lo < 1e-10:
            break

    t_best = (lo + hi) / 2.0
    f_best = max_conv(t_best)

    # Check endpoints
    f0 = max_conv(0.0)
    f1 = max_conv(1.0)
    if f0 < f_best:
        t_best, f_best = 0.0, f0
    if f1 < f_best:
        t_best, f_best = 1.0, f1

    return t_best, f_best


def coord_descent_polish(a0, time_limit, n_trials=12, verbose=False):
    """Fast coordinate descent with sensitivity analysis for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a  # a[k-i] for k = i, ..., i+n-1 is just a[0], ..., a[n-1]

            best_delta = 0.0
            best_new_max = M

            # Try various multiplicative changes
            factors = []
            for j in range(1, n_trials + 1):
                factors.append(1.0 - j / (n_trials + 1.0))
            for j in range(1, n_trials // 2 + 1):
                factors.append(1.0 + j * 0.3 / (n_trials // 2))

            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0:
                old_ai = a[i]
                a[i] = max(a[i] + best_delta, 0.0)

                # Update conv incrementally
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M  # sum is ~1
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def perturb_escape(a, rng, intensity=0.02):
    """Perturb the sequence to escape a local optimum."""
    n = len(a)
    a_new = a.copy()
    # Perturb a random subset of coordinates
    n_perturb = max(10, n // 20)
    idx = rng.choice(n, size=n_perturb, replace=False)
    a_new[idx] *= (1.0 + rng.standard_normal(n_perturb) * intensity)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

    rng = np.random.default_rng(777)

    # Get parent sequence
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
        if parent_raw.ndim != 1 or parent_raw.size == 0 or not np.isfinite(parent_raw).all():
            raise ValueError("Invalid parent")
        if parent_raw.sum() < 0.01:
            raise ValueError("Parent sum too small")
        parent = parent_raw / parent_raw.sum()
    except Exception:
        parent = np.ones(1024) / 1024
        print("Using default parent (uniform 1024)")

    n = len(parent)
    best_seq = parent.copy()
    best_score = float(score_of(parent))
    print(f"Parent: n={n}, score={best_score:.8f}")

    # Verify with evaluate_sequence
    ev_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(ev_score):
        best_score = min(best_score, ev_score)
        if ev_score < best_score + 1e-6:
            best_score = ev_score

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # === Phase 1: LP alternating with line search ===
    print(f"\n=== Phase 1: LP Alternating + Line Search ===")
    a = parent.copy()
    t0 = time.monotonic()
    lp_iters = 0
    no_improve_streak = 0
    improvement_threshold = 1e-12

    while time.monotonic() < deadline - 300:
        lp_iters += 1

        # Solve LP best response
        b = solve_lp_best_response(a, solver)
        if b is None:
            no_improve_streak += 1
            if no_improve_streak > 20:
                # Perturb and try again
                a = perturb_escape(best_seq, rng, intensity=0.03)
            continue

        # Line search
        t_opt, f_opt = line_search_conv(a, b)
        new_a = (1.0 - t_opt) * a + t_opt * b
        new_score = 2.0 * n * f_opt  # sum is 1 (both on simplex)

        if new_score < best_score - 1e-10:
            best_score = new_score
            best_seq = new_a.copy()
            no_improve_streak = 0
            a = new_a
            if lp_iters % 10 == 1:
                elapsed = time.monotonic() - t0
                print(f"  LP iter={lp_iters} ({elapsed:.0f}s): score={best_score:.8f} t={t_opt:.4f}")
        elif new_score < best_score:
            a = new_a
            no_improve_streak += 1
        else:
            # LP didn't help from this point; try from best_seq
            no_improve_streak += 1
            if no_improve_streak >= 15:
                # Perturb best_seq and restart LP from there
                a = perturb_escape(best_seq, rng, intensity=0.02 + 0.01 * (no_improve_streak // 15))
                no_improve_streak = 0
                print(f"  LP iter={lp_iters}: perturbing to escape, intensity={0.02 + 0.01*(no_improve_streak//15):.4f}")
            else:
                a = new_a

        # Periodic coordinate descent interlude
        if lp_iters % 30 == 0 and time.monotonic() < deadline - 350:
            a_cd, s_cd = coord_descent_polish(best_seq, time_limit=15, n_trials=8)
            if s_cd < best_score - 1e-10:
                best_score = s_cd
                best_seq = a_cd.copy()
                a = a_cd.copy()
                print(f"  CD interlude: improved to {best_score:.8f}")

    print(f"  Phase 1 done: {lp_iters} LP iters, best={best_score:.8f}")

    # === Phase 2: Intensive coordinate descent polishing ===
    print(f"\n=== Phase 2: CD Polishing ===")
    if time.monotonic() < deadline - 150:
        t_cd = min(150.0, (deadline - time.monotonic() - 100))
        a_poly, s_poly = coord_descent_polish(best_seq, time_limit=t_cd, n_trials=10)
        if s_poly < best_score:
            best_score = s_poly
            best_seq = a_poly.copy()
        print(f"  After CD: {best_score:.8f}")

    # === Phase 3: More LP rounds from polished point ===
    print(f"\n=== Phase 3: LP from Polished Point ===")
    a = best_seq.copy()
    no_improve_streak = 0
    lp2_iters = 0

    while time.monotonic() < deadline - 100:
        lp2_iters += 1
        b = solve_lp_best_response(a, solver)
        if b is None:
            no_improve_streak += 1
            if no_improve_streak > 20:
                a = perturb_escape(best_seq, rng, intensity=0.01)
                no_improve_streak = 0
            continue

        t_opt, f_opt = line_search_conv(a, b)
        new_a = (1.0 - t_opt) * a + t_opt * b
        new_score = 2.0 * n * f_opt

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = new_a.copy()
            a = new_a
            no_improve_streak = 0
            if lp2_iters % 5 == 0:
                print(f"  LP2 iter={lp2_iters}: score={best_score:.8f}")
        else:
            a = new_a
            no_improve_streak += 1
            if no_improve_streak > 20:
                a = perturb_escape(best_seq, rng, intensity=0.01)
                no_improve_streak = 0

    print(f"  Phase 3 done: {lp2_iters} iters, best={best_score:.8f}")

    # === Phase 4: Fine coordinate descent ===
    print(f"\n=== Phase 4: Fine CD ===")
    if time.monotonic() < deadline - 50:
        t_fine = min(80.0, deadline - time.monotonic() - 30)
        a_fine, s_fine = fine_coord_descent(best_seq, time_limit=t_fine)
        if s_fine < best_score:
            best_score = s_fine
            best_seq = a_fine.copy()
        print(f"  After FineCD: {best_score:.8f}")

    # === Phase 5: Final LP + fine polish ===
    if time.monotonic() < deadline - 20:
        a = best_seq.copy()
        b = solve_lp_best_response(a, solver)
        if b is not None:
            t_opt, f_opt = line_search_conv(a, b)
            new_a = (1.0 - t_opt) * a + t_opt * b
            new_score = 2.0 * n * f_opt
            if new_score < best_score:
                best_score = new_score
                best_seq = new_a.copy()

        if time.monotonic() < deadline - 10:
            a_final, s_final = fine_coord_descent(best_seq, time_limit=15)
            if s_final < best_score:
                best_score = s_final
                best_seq = a_final.copy()

    # Final verification with evaluate_sequence
    final_list = [float(max(0.0, x)) for x in best_seq]
    fs = float(evaluate_sequence(final_list))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
        if fs < best_score + 1e-6:
            best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return final_list
