- There exist weights w_k >= 0, sum w_k = 1, such that:

- sum_k w_k = 1, w_k >= 0
- sum_j a_j = 1, a_j >= 0

This is a system of nonlinear equations that could potentially be solved with Newton's method. But the active set is unknown and changes.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def lbgfs_refine_v2(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """Improved L-BFGS with log-sum-exp and more stages."""
    n = len(a0)
    z = z_from_a(np.asarray(a0, dtype=np.float64)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = np.geomspace(500.0, 0.1, 12)
    
    best_a = softmax_from_z(z)
    best_score = compute_score(best_a)
    
    for T_val in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            a = softmax_from_z(z)
            c = fast_convolve(a)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            log_sum_exp = max_Tc + np.log(exp_Tc.sum())
            return (2.0 * n / T_val) * log_sum_exp
        
        def jac(z):
            a = softmax_from_z(z)
            c = fast_convolve(a)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            # grad_a = 2 * sum_k w_k * (conv of a with delta at k)
            # conv(a, delta_k)[j] = a[j-k] (shifted a)
            # Actually, d/dc_k (a_j * a_{k-j}) gives us the gradient
            # grad_{a_i} conv(a,a)[k] = a[k-i] (for valid i)
            grad_a = np.zeros(n)
            for k in range(len(c)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            # a is parameterized via softmax, so grad_z = a * (grad_a - <grad_a, a>)
            S = np.dot(grad_a, a)
            return 2.0 * a * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"  L-BFGS stage T={T_val:.1f} failed: {e}")
        
        a = softmax_from_z(z)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

from scipy.optimize import minimize

        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(np.asarray(a0, dtype=np.float64)).copy()
    t0 = time.monotonic()
    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_score = compute_score(best_a)

    for stage_idx, T_val in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_inner):
            a = softmax_from_z(z_inner)
            c = fast_convolve(a)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            lse = max_Tc + np.log(exp_Tc.sum())
            return (2.0 * n / T_val) * lse

        def jac(z_inner):
            a = softmax_from_z(z_inner)
            c = fast_convolve(a)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            # Sparse gradient: only process significant weights
            significant = w > 1e-12
            grad_a = np.zeros(n)
            for k in np.where(significant)[0]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return 2.0 * a * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a = softmax_from_z(z)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    Stage {stage_idx+1}/{len(T_stages)} T={T_val:.2f}: score={score:.6f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_rounds=200):
    """Exact coordinate descent on the nonsmooth max objective."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        # Process coordinates in random order
        order = np.random.default_rng(rnd).permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-18:
                continue

            # Compute current convolution
            c = fast_convolve(a)
            current_max = c.max()

            # For coordinate i, find optimal value
            # conv[k] as function of a_i:
            # If k-i is valid: conv[k] += 2*a_i*a[k-i] (or a_i^2 if k=2i)
            # We need to find a_i_new >= 0 that minimizes max_k conv[k](a_i_new)
            # subject to maintaining sum = 1 (redistribute to others)

            # Actually, let's do a simpler approach:
            # Reduce a[i] by a small amount and redistribute
            # Try several reduction factors
            best_trial = None
            best_trial_score = best_score

            for factor in [0.999, 0.995, 0.99, 0.98, 0.96, 0.93, 0.90]:
                diff = a[i] * (1 - factor)
                if diff < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor
                # Redistribute to other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-18:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()

                score = compute_score(a_trial)
                if score < best_trial_score:
                    best_trial_score = score
                    best_trial = a_trial

            if best_trial is not None:
                a = best_trial
                best_a = a.copy()
                best_score = best_trial_score
                improved = True

        if not improved:
            # Try pair operations
            c = fast_convolve(a)
            peak_k = int(np.argmax(c))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top = np.argsort(contribs)[::-1][:6]

            pair_improved = False
            for ci in top:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-15 or a[j] < 1e-15:
                    continue

                for f1 in [0.99, 0.97, 0.94]:
                    for f2 in [0.99, 0.97, 0.94]:
                        a_trial = a.copy()
                        diff = a[i] * (1 - f1) + a[j] * (1 - f2)
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        others_sum = a_trial[mask].sum()
                        if others_sum < 1e-18:
                            continue
                        a_trial[mask] += diff * a_trial[mask] / others_sum
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_a = a.copy()
                            best_score = score
                            pair_improved = True
                            break
                    if pair_improved:
                        break
                if pair_improved:
                    break

            if not pair_improved:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif choice == 2:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x / 2) ** 2
    elif choice == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    else:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 0.5 * x ** 0.5

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced L-BFGS LogSumExp
optimization with ultra-fine temperature annealing (20+ stages down to T=0.005),
multi-restart with structured perturbations, and exact nonsmooth coordinate
descent polishing. Focuses computational budget on the parent sequence length
with aggressive low-temperature refinement to squeeze out the last 0.2%.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_stage(a_in, T_val, maxiter):
    """Single L-BFGS stage at temperature T_val. Returns updated a."""
    n = len(a_in)
    z = z_from_a(np.asarray(a_in, dtype=np.float64)).copy()

    def func(z_inner):
        a = softmax_from_z(z_inner)
        c = fast_convolve(a)
        Tc = T_val * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        lse = max_Tc + np.log(exp_Tc.sum())
        return (2.0 * n / T_val) * lse

    def jac(z_inner):
        a = softmax_from_z(z_inner)
        c = fast_convolve(a)
        Tc = T_val * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()
        grad_a = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return 2.0 * a * (grad_a - S)

    try:
        res = minimize(
            fun=func, x0=z, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20,
                     'gtol': 1e-15, 'maxfun': maxiter * 300,
                     'disp': False}
        )
        z = res.x
    except Exception:
        pass

    return softmax_from_z(z)


def lbgfs_anneal(a0, time_limit, T_start, T_end, n_stages=20, maxiter_per_stage=6000):
    """L-BFGS with geometric temperature annealing over n_stages."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_idx, T_val in enumerate(temps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 8:
            break

        # Scale maxiter based on remaining time
        effective_iter = maxiter_per_stage
        if remaining < 60:
            effective_iter = max(500, maxiter_per_stage // 3)
        elif remaining < 200:
            effective_iter = max(1000, maxiter_per_stage // 2)

        a = lbgfs_stage(a, T_val, effective_iter)
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if (stage_idx + 1) % 5 == 0 or stage_idx == 0:
            print(f"    T={T_val:.4f} ({stage_idx+1}/{n_stages}) [{time.monotonic()-t0:.0f}s]: score={score:.8f}")

    return best_a, best_score


def peak_shave_exact(a0, time_limit, n_rounds=300):
    """Exact coordinate descent on the nonsmooth max(conv(a,a)) objective.
    Tries to reduce peak by Adjusting contributing coordinates and
    redistributing mass to low-convolution positions."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        current_peak = c.max()
        peak_thresh = current_peak - 0.02 * current_peak
        active_peaks = np.where(c >= peak_thresh)[0]

        # Find top contributing coordinates across all active peaks
        contrib = np.zeros(n)
        for k in active_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        # Find "safe" positions (low convolution contribution)
        pos_weight = np.zeros(n)
        for k in active_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            pos_weight[js:je+1] += 1
        safe_positions = np.argsort(pos_weight)[:n // 3]

        top_contributing = np.argsort(contrib)[::-1][:15]
        improved = False

        for i in top_contributing:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-16:
                continue

            # Try reducing a[i] and redistributing to safe positions
            for factor in [0.999, 0.997, 0.995, 0.99, 0.98, 0.96]:
                diff = a[i] * (1.0 - factor)
                if diff < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor

                # Redistribute to safe positions (top 10 safest)
                safe_sorted = safe_positions[:20]
                safe_weights = a_trial[safe_sorted].copy()
                sw_sum = safe_weights.sum()
                if sw_sum < 1e-18:
                    # Redistribute proportionally to all others
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    os = others.sum()
                    if os < 1e-18:
                        continue
                    a_trial[mask] = others + diff * others / os
                else:
                    a_trial[safe_sorted] = safe_weights + diff * safe_weights / sw_sum

                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_a = a.copy()
                    best_score = score
                    improved = True
                    break

        if not improved:
            # Try pair reduction on the main peak
            main_peak = int(np.argmax(c))
            js = max(0, main_peak - n + 1)
            je = min(n - 1, main_peak)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[main_peak - idx]
            top_pairs = np.argsort(contribs)[::-1][:8]

            pair_improved = False
            for ci in top_pairs:
                i = idx[ci]
                j = main_peak - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-16 or a[j] < 1e-16:
                    continue
                for f1 in [0.995, 0.99, 0.98]:
                    for f2 in [0.995, 0.99, 0.98]:
                        a_trial = a.copy()
                        diff = a[i] * (1 - f1) + a[j] * (1 - f2)
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        others_sum = a_trial[mask].sum()
                        if others_sum < 1e-18:
                            continue
                        a_trial[mask] += diff * a_trial[mask] / others_sum
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_a = a.copy()
                            best_score = score
                            pair_improved = True
                            break
                    if pair_improved:
                        break
                if pair_improved:
                    break

            if not pair_improved:
                # Try small random perturbations
                n_pert = 5
                pert_improved = False
                for _ in range(n_pert):
                    pert_idx = rng.integers(0, n, size=3)
                    a_trial = a.copy()
                    for pi in pert_idx:
                        a_trial[pi] *= (1.0 + 0.02 * rng.standard_normal())
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_a = a.copy()
                        best_score = score
                        pert_improved = True
                        break
                if not pert_improved:
                    break
        else:
            pass  # Continue if improved

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif choice == 2:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x / 2) ** 2
    elif choice == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    else:
        x = np.linspace(0, 1, n)
        a = np.sqrt(x * (1 - x))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization focused on ultra-fine L-BFGS temperature annealing
    and exact coordinate descent polishing. Allocates most time to low-temperature
    refinement of the parent sequence with multiple perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Main L-BFGS - coarse to medium temperature ===
    print(f"\n=== Phase 1: L-BFGS Main n={n_best} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 50:
        print(f"  Budget: {t_alloc:.0f}s, T: 100 -> 1")
        a1, _ = lbgfs_anneal(best_seq, t_alloc, T_start=100.0, T_end=1.0,
                             n_stages=12, maxiter_per_stage=8000)
        try_update(a1, "lbgfs_main")

    # === Phase 2: L-BFGS - medium to low temperature ===
    print(f"\n=== Phase 2: L-BFGS Low-T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s, T: 2 -> 0.05")
        a2, _ = lbgfs_anneal(best_seq, t_alloc, T_start=2.0, T_end=0.05,
                             n_stages=15, maxiter_per_stage=10000)
        try_update(a2, "lbgfs_low_t")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: Peak Shave (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a3, _ = peak_shave_exact(best_seq, t_alloc, n_rounds=500)
        try_update(a3, "peak_shave_1")

    # === Phase 4: Perturbation restarts ===
    print(f"\n=== Phase 4: Perturbation Restarts (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for r in range(4):
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_restart = min(rem * 0.22, 250.0)
        if t_restart < 40:
            break

        noise_scale = 0.02 * (r + 1)
        if r == 3:
            # Larger perturbation for diversity
            noise_scale = 0.10
        perturbed = best_seq + noise_scale * rng.standard_normal(len(best_seq))
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()

        print(f"  Restart {r+1} (noise={noise_scale:.3f}), budget={t_restart:.0f}s")
        T_lo = 0.5 if r < 3 else 5.0
        a_r, _ = lbgfs_anneal(perturbed, t_restart, T_start=T_lo, T_end=0.01,
                             n_stages=10, maxiter_per_stage=7000)
        try_update(a_r, f"restart_{r+1}")

        # Quick polish after restart
        rem = deadline - time.monotonic()
        if rem > 20 and best_score < 1.503:
            t_polish = min(rem * 0.3, 40.0)
            a_rp, _ = peak_shave_exact(best_seq, t_polish, n_rounds=100)
            try_update(a_rp, f"restart_{r+1}_polish")

    # === Phase 5: Final ultra-low-T L-BFGS ===
    print(f"\n=== Phase 5: Ultra-Low-T L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.30, 120.0)
        print(f"  Budget: {t_fin:.0f}s, T: 1 -> 0.005")
        a5, _ = lbgfs_anneal(best_seq, t_fin, T_start=1.0, T_end=0.005,
                             n_stages=12, maxiter_per_stage=12000)
        try_update(a5, "ultra_low_t")

    # === Phase 6: Final coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.40, 50.0)
        a_cd2, _ = peak_shave_exact(best_seq, t_cd, n_rounds=300)
        try_update(a_cd2, "final_cd")

    # === Phase 7: One more L-BFGS pass if time remains ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_last = min(rem * 0.50, 60.0)
        a_last, _ = lbgfs_anneal(best_seq, t_last, T_start=0.5, T_end=0.002,
                                 n_stages=8, maxiter_per_stage=15000)
        try_update(a_last, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
