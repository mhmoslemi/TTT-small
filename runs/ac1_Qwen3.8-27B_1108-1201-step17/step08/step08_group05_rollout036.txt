"""
Hybrid optimization for non-negative auto-convolution peak minimization.
Combines LP with constraint generation, smoothed-max gradient descent,
and multi-length exploration. Targets score < 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_weights(b, t):
    """Compute smoothed max value and softmax weights."""
    b_max = b.max()
    shifted = t * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    value = b_max + log_sum / t
    weights = exp_shifted / exp_shifted.sum()
    return value, weights


def smooth_max_gradient(a, t):

    Compute gradient of smooth_max(a*a, t) w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    """
    n = len(a)
    b = fast_convolve(a)
    _, w = smooth_max_and_weights(b, t)

    # T2_j = sum_i w_{j-i} * a_i (convolution)
    # grad = T1 + T2

    size = 3 * n
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    W_rev = rfft(w[::-1], fft_size)

    # T2 = convolve(w, a)[:n]
    T2 = irfft(W * A, fft_size)[:n]

    # T1 = convolve(w_rev, a)[n-1:2n-1][::-1]
    T1_full = irfft(W_rev * A, fft_size)
    T1 = T1_full[n - 1:2 * n - 1][::-1]

    return T1 + T2


def run_lp_phase(f_init, time_limit, top_k_init=50, top_k_max=250, label="lp"):
    """LP-based optimization with constraint generation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptively increase top_k
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        g, M = _solve_lp(f, n, top_k, max_constraints=350,
                         time_budget=max(3, (time_limit - (time.monotonic() - t0)) * 0.08))

        if g is None:
            # Fallback: use steepest descent direction on top peaks
            b = fast_convolve(f)
            n_peaks = min(8, len(b))
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:] if n_peaks < len(b) else np.arange(len(b))
            g = np.zeros(n)
            for pk in top_idx:
                pk = int(pk)
                lo = max(0, pk - n + 1)
                hi = min(n - 1, pk)
                g[lo:hi + 1] += f[pk - hi: pk - lo + 1]
            M = b[top_idx].max()

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 25:
                # Perturb and try again
                rng = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
                f_trial = f + rng.standard_normal(n) * 0.002
                f = project_simplex(f_trial)
                top_k = min(top_k * 2, top_k_max)
                stuck = 0
                continue
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g

        # Adaptive line search
        best_t = _adaptive_line_search(f, h)

        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s

        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            f = f_new  # Still move, but don't update best
            stuck += 1
            if stuck > 60:
                break

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0 * n * best_obj:.6f} tk={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def _solve_lp(f, n, top_k, max_constraints, time_budget):
    """Solve LP with constraint generation."""
    t_start = time.monotonic()
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    # Initialize active set with top-k peaks
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    active_list = [int(x) for x in top_idx]
    active_set = set(active_list)

    g = None

    for cg_iter in range(12):
        if time.monotonic() - t_start > time_budget * 1.5:
            break

        K = len(active_list)
        if K > max_constraints:
            b_active = np.array([b[k] for k in active_list])
            keep = np.argsort(b_active)[-max_constraints:]
            active_list = [active_list[i] for i in keep]
            active_set = set(active_list)
            K = len(active_list)

        # Build constraint matrix
        A = np.zeros((K, n))
        for row_i in range(K):
            k = active_list[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )

        solved = False
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iters=500000)
                solved = True
                break
            except Exception:
                continue

        if not solved or g_var.value is None:
            if g is not None:
                break
            return None, M

        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)) or np.any(np.isinf(g)):
            if cg_iter > 0:
                break
            return None, M

        # Check for violated constraints
        if time.monotonic() - t_start > time_budget:
            break

        fg = fast_cross_convolve(f, g)
        mask = np.zeros(len(b), dtype=bool)
        for k in active_list:
            mask[k] = True
        violations = np.where((fg > M + 1e-9) & (~mask))[0]

        if len(violations) == 0:
            break

        worst = np.argsort(fg[violations])[-min(5, len(violations)):]
        for v in worst:
            vi = int(v)
            if vi not in active_set:
                active_set.add(vi)
                active_list.append(vi)

    return g, M


def _adaptive_line_search(f, direction, max_evals=15):
    """Adaptive line search using backtracking."""
    n = len(f)
    best_t = 0.05
    best_obj = _peak_at_combo(f, direction, 0.05)

    candidates = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    for t in candidates:
        obj = _peak_at_combo(f, direction, t)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t

    # Refine around best
    t_lo = max(0.001, best_t * 0.5)
    t_hi = min(0.99, best_t * 2.0)
    if t_hi - t_lo < 0.01:
        return best_t

    for _ in range(8):
        t_mid = (t_lo + t_hi) / 2
        obj = _peak_at_combo(f, direction, t_mid)
        if obj < best_obj:
            best_obj = obj
            best_t = t_mid
            t_lo = t_mid
        else:
            t_hi = t_mid
        if t_hi - t_lo < 0.005:
            break

    return best_t


def _peak_at_combo(f, direction, t):
    """Compute peak of (1-t)*f + t*direction, normalized."""
    trial = (1.0 - t) * f + t * direction
    trial = np.maximum(trial, 0.0)
    s = trial.sum()
    if s < 1e-10:
        return 1e10
    trial = trial / s
    return fast_convolve(trial).max()
