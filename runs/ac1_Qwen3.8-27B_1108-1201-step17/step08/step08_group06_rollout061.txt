"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent (multiplicative updates) with soft-max gradient
2. Adaptive temperature annealing
3. Periodic coordinate descent refinement with golden-section search
4. Multiple initializations and length exploration
5. Peak-aware pairwise transfers
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    return irfft(A * A, n=fft_size)[:size]


def compute_score(a):
    """Compute the score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def cross_correlate(a, w):
    """Compute g[j] = sum_k w[k] * a[k-j] via FFT.
    w has length 2n-1, a has length n.
    Result has length n.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # g[j] = sum_k w[k] * a[k-j]
    # This is the cross-correlation of a with w
    # Using FFT: ifft(conj(fft(w)) * fft(a)) gives correlation
    # But we need to be careful with the definition
    
    # Alternative: g = ifft(fft(w) * conj(fft(a)))[:n] ... no
    # Let me think: correlation(a, w)[j] = sum_k a[k] * w[k+j] (standard definition)
    # We want: g[j] = sum_k w[k] * a[k-j]
    # Let's define b[m] = a[-m] (reversed a, zero-padded)
    # Then g[j] = sum_k w[k] * b[j-k] = (w * b)[j] (convolution!)
    
    # Actually, let's just use the direct definition with FFT:
    # g[j] = sum_k w[k] * a[k-j] for j = 0, ..., n-1
    # where a[m] = 0 if m not in [0, n-1]
    
    # This is: g = ifft(fft(w) * fft(a_reversed_shifted))... 
    
    # Simpler approach: use the fact that g[j] = sum_k w[k] * a[k-j]
    # = sum_m w[m+j] * a[m] (substituting m = k-j)
    # = sum_m a[m] * w[m+j]
    # This is the cross-correlation of a and w, evaluated at lag j.
    
    # Using FFT: correlation(a, w) = ifft(conj(fft(a)) * fft(w))
    # But this gives correlation[m] = sum_k a[k] * conj(w[k-m])
    # For real signals: = sum_k a[k] * w[k-m]
    
    # So correlation(a, w)[j] = sum_k a[k] * w[k-j]
    # We want g[j] = sum_m a[m] * w[m+j]
    # = correlation(a, w)[-j] (negative lag)
    # = correlation(w, a)[j] (by symmetry of real correlation)
    
    # So: g = correlation(w, a)[0:n] = ifft(conj(fft(w)) * fft(a))[0:n]
    
    Wa = rfft(w, n=fft_size)
    Aa = rfft(a, n=fft_size)
    g_full = irfft(np.conj(Wa) * Aa, n=fft_size)
    # g_full[m] = sum_k w[k] * a[k-m]... let me verify
    # Actually: ifft(conj(fft(w)) * fft(a))[m] = sum_k w[k-m] * a[k]
    # (this is the cross-correlation (w ⋆ a)[m] = sum_k w[k-m] a[k])
    # We want g[j] = sum_k w[k] a[k-j] = sum_m w[m+j] a[m] (let m=k-j)
    # = (w ⋆ a)[-j] ... hmm this is getting confusing.
    
    # Let me just do it directly for correctness:
    # g[j] = sum_{k=0}^{2n-2} w[k] * a[k-j] where a is zero-padded
    # For j in [0, n-1]: k-j ranges from -j to 2n-2-j
    # a[k-j] is non-zero only when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
    
    # So g[j] = sum_{k=j}^{j+n-1} w[k] * a[k-j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is the dot product of a with the slice w[j:j+n]
    
    # This is a "sliding dot product" which can be computed as a cross-correlation.
    # g = ifft(fft(w) * conj(fft(a_pad))) ... let me just verify with a small example.
    
    # Actually, the cleanest way: g is the cross-correlation of w and a.
    # numpy.correlate(w, a, mode='full') gives c[k] = sum_n w[n+k] * a[n] (approximately)
    # Let's just use scipy or compute directly.
    
    # For efficiency with large n, use FFT:
    # Define: g[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    # This is the "valid" cross-correlation of w and a.
    # Using FFT: create W = fft(w, fft_size), A = fft(a, fft_size)
    # Then ifft(W * conj(A)) gives the full cross-correlation
    # ifft(fft(w) * conj(fft(a)))[k] = sum_n w[n] * a[n-k] (for real)
    # = sum_n w[n+k] * a[n] (substituting n -> n+k)
    # = g[k] for k >= 0!
    
    # So: g[k] = ifft(fft(w) * conj(fft(a)))[k] for k = 0, ..., n-1
    
    W = rfft(w, n=fft_size)
    A = rfft(a, n=fft_size)
    g_full = irfft(W * np.conj(A), n=fft_size)
    return g_full[:n]


def smoothed_mirror_descent(a0, time_limit, T_init=1.0, T_min=0.01, eta_init=0.1, 
                             anneal_rate=0.999, verbose=True):
    """Smoothed mirror descent with adaptive annealing."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    T = T_init
    eta = eta_init
    iterations = 0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        b = fast_convolve(a)
        T = max(T_min, T * anneal_rate)
        
        # Soft-max weights
        b_max = b.max()
        exp_b = np.exp((b - b_max) / T)
        p = exp_b / exp_b.sum()
        
        # Gradient: g[j] = sum_k p[k] * a[k-j]
        g = cross_correlate(a, p)
        
        # Ensure gradient is non-negative (it should be since a >= 0, p >= 0)
        g = np.maximum(g, 0)
        
        # Mirror descent update
        log_factor = -eta * g
        # Clip to prevent underflow
        log_factor = np.clip(log_factor, -50, 50)
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score_new = compute_score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        
        # Adaptive step size: reduce if not improving for a while
        if patience > 50:
            eta *= 0.5
            patience = 0
        
        if verbose and iterations % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SMD iter={iterations}, T={T:.4f}, eta={eta:.4f}, "
                  f"score={score_new:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_random_pairs=200, n_peak_pairs=100, verbose=False):
    """Coordinate descent with golden-section search for optimal pairwise transfer."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)
        
        # Generate candidate pairs
        pairs = set()
        
        # Random pairs
        for _ in range(n_random_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Peak-focused pairs
        top_idx = np.argmax(conv)
        js = max(0, top_idx - n + 1)
        je = min(n - 1, top_idx)
        for _ in range(n_peak_pairs):
            i = rng.integers(js, je + 1)
            j = rng.integers(js, je + 1)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        gr = (np.sqrt(5) - 1) / 2
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            # Golden section search
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_t(t_val):
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    return 1e10
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial = a_trial / ss
                return compute_score(a_trial)
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-12:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                # Perturb
                noise = rng.standard_normal(n) * 0.001
                a = best_a + noise
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  CD score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def gradient_descent_smoothed(a0, time_limit, T_init=10.0, T_decay=0.9995, lr_init=0.01, verbose=True):
    """Gradient descent on the smoothed objective in the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        p = np.exp((b - b_max) / T)
        p = p / p.sum()
        
        # Gradient of smoothed objective: 2 * cross_correlate(a, p)
        g = 2.0 * cross_correlate(a, p)
        
        # Projected gradient step (project onto simplex)
        a_new = a - lr * g
        # Project onto simplex: sum=1, a>=0
        a_new = project_simplex(a_new)
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        T = max(0.01, T * T_decay)
        lr = lr * 0.9999
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  GD iter={it}, T={T:.4f}, lr={lr:.6f}, score={score:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def mirror_descent_with_line_search(a0, time_limit, T_init=5.0, T_decay=0.999, verbose=True):
    """Mirror descent with adaptive step size (backtracking line search)."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = T_init
    eta = 0.1
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        p = np.exp((b - b_max) / T)
        p = p / p.sum()
        
        g = cross_correlate(a, p)
        g = np.maximum(g, 1e-15)
        
        # Mirror descent: minimize f(a_new) subject to D_KL(a_new || a) <= eta^2/2
        # Closed form: a_new = a * exp(-eta * g) / sum(a * exp(-eta * g))
        # With line search: try decreasing eta until we improve
        
        improved = False
        for trial in range(20):
            log_factor = -eta * g
            log_factor = np.clip(log_factor, -100, 100)
            a_new = a * np.exp(log_factor)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                eta *= 0.5
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                improved = True
                # Try to increase eta slightly
                eta = min(eta * 1.1, 1.0)
                break
            else:
                eta *= 0.5
        
        if not improved:
            # Keep a the same
            pass
        
        T = max(0.01, T * T_decay)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD-LB iter={it}, T={T:.4f}, eta={eta:.4f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:

def make_initializations(n, rng):

    inits = []

    inits.append(('uniform', np.ones(n) / n))

    x = np.linspace(0, np.pi, n)
    dome = np.sin(x)
    dome = dome / dome.sum()
    inits.append(('dome', dome))
    
    # Inverted dome (U-shape)
    ushape = 1.0 - 0.8 * np.sin(x)

    inits.append(('gaussian', exp_shape))

    pw = np.ones(n) ** 0.5 + 0.5 * np.sin(np.pi * np.arange(n) / n)

    pw = pw / pw.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Smoothed mirror descent (multiplicative updates) with soft-max gradient
2. Adaptive step size with backtracking
3. Periodic coordinate descent with golden-section search

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    return irfft(A * A, n=fft_size)[:size]


def compute_score(a):
    """Compute the score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_gradient(a):
    """Compute gradient of smoothed max-convolution.
    Returns (g, p, b) where g[i] = sum_k p[k]*a[k-i], p is softmax of b, b is convolution.
    """
    n = len(a)
    b = fast_convolve(a)
    return b


def softmax_weights(b, T):
    """Compute softmax weights over convolution values."""
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    s = exp_b.sum()
    if s < 1e-300:
        return np.full(len(b), 1.0 / len(b))
    return exp_b / s


def cross_correlate(a, w):
    """Compute g[j] = sum_k w[k] * a[k-j] for j=0,...,n-1 via FFT.
    w has length 2n-1, a has length n.
    Uses proper FFT size for linear correlation.
    """
    n = len(a)
    # For linear cross-correlation of lengths (2n-1) and n, need FFT size >= 3n-2
    needed = 3 * n - 2
    fft_size = 1 << (needed - 1).bit_length()
    W = rfft(w, n=fft_size)
    A = rfft(a, n=fft_size)
    c = irfft(W * np.conj(A), n=fft_size)

def mirror_descent_step(a, g, eta):
    """Perform one mirror descent step: a_new = a * exp(-eta*g), normalized."""
    log_factor = -eta * g
    log_factor = np.clip(log_factor, -80, 80)
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def smoothed_mirror_descent(a0, time_limit, T_init=2.0, T_min=0.005, 
                             eta_init=0.1, eta_decay=0.9995, 
                             T_decay=0.9999, verbose=True, verbose_every=500):
    """Smoothed mirror descent with adaptive step size."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = T_init
    eta = eta_init
    patience = 0
    max_patience = 200
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        p = softmax_weights(b, T)
        g = cross_correlate(a, p)
        g = np.maximum(g, 0)
        
        # Mirror descent step
        a_new = mirror_descent_step(a, g, eta)
        score = compute_score(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            patience = 0
            a = a_new
            # Slightly increase eta (be more aggressive)
            eta = min(eta * 1.02, 1.0)
        else:
            patience += 1
            if patience > max_patience:
                # Reduce eta and perturb
                eta *= 0.5
                patience = 0
                # Add small noise to escape
                a_new = a_new + 0.001 * np.random.default_rng(it).standard_normal(n)
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
                a = a_new
            
        # Anneal temperature
        T = max(T_min, T * T_decay)
        eta = max(eta * eta_decay, 1e-4)
        
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"  SMD it={it}, T={T:.5f}, η={eta:.5f}, "
                  f"sc={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def gradient_descent_projected(a0, time_limit, T_init=5.0, T_decay=0.9998, 
                                lr_init=0.005, lr_decay=0.9999, verbose=True, verbose_every=500):
    """Projected gradient descent on smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        p = softmax_weights(b, T)
        g = 2.0 * cross_correlate(a, p)
        
        a_new = a - lr * g
        a_new = project_simplex(a_new)
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        T = max(0.01, T * T_decay)
        lr = max(lr * lr_decay, 1e-8)
        
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"  PGD it={it}, T={T:.5f}, lr={lr:.7f}, "
                  f"sc={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rand=200, n_peak=100, 
                       seed=42, verbose=False, verbose_every=100):
    """Coordinate descent with golden-section search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5) - 1) / 2
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it_count += 1
        improved = False
        conv = fast_convolve(a)
        
        # Generate pairs
        pairs = set()
        for _ in range(n_rand):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((int(min(i, j)), int(max(i, j))))
        
        # Peak-focused
        top_k = min(3, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        for pk in peak_indices:
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            for _ in range(n_peak // top_k):
                i = rng.integers(js, je + 1)
                j = rng.integers(js, je + 1)
                if i != j:
                    pairs.add((int(min(i, j)), int(max(i, j))))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            t_lo = -a[j] + 1e-14
            t_hi = a[i] - 1e-14
            if t_hi <= t_lo:
                continue
            
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            f1 = eval_transfer(a, i, j, x1)
            f2 = eval_transfer(a, i, j, x2)
            
            for _ in range(12):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2; x2 = x1; f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(a, i, j, x1)
                else:
                    t_lo = x1; x1 = x2; f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(a, i, j, x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and it_count % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD it={it_count}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def eval_transfer(a, i, j, t):
    """Evaluate score after transferring t from a[i] to a[j]."""
    a_trial = a.copy()
    a_trial[i] -= t
    a_trial[j] += t
    if a_trial[i] < -1e-14 or a_trial[j] < -1e-14:
        return 1e10
    a_trial = np.maximum(a_trial, 0)
    s = a_trial.sum()
    if s < 1e-15:
        return 1e10
    a_trial = a_trial / s
    return compute_score(a_trial)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n)
    
    if parent is not None:
        inits.append(('parent', interpolate_sequence(parent, n)))
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Dome (raised cosine)
    dome = np.sin(np.pi * x / (n - 1)) ** 0.5
    dome = np.maximum(dome, 0.01)
    inits.append(('dome', dome / dome.sum()))
    
    # Inverted dome
    inv = 1.0 - 0.7 * np.sin(np.pi * x / (n - 1))
    inv = np.maximum(inv, 0.05)
    inits.append(('inv_dome', inv / inv.sum()))
    
    # Gaussian
    center = n / 2.0
    gauss = np.exp(-((x - center) ** 2) / (2 * (n / 4) ** 2))
    inits.append(('gauss', gauss / gauss.sum()))
    
    # Multi-frequency
    multi = 1.0 + 0.3 * np.sin(2 * np.pi * x / n * 3) + 0.2 * np.sin(2 * np.pi * x / n * 7)
    multi = np.maximum(multi, 0.05)
    inits.append(('multi', multi / multi.sum()))
    
    # Random
    r = rng.random(n)
    inits.append(('rand', r / r.sum()))
    
    # Bimodal
    bim = np.exp(-((x - n*0.25) ** 2) / (n/20)) + np.exp(-((x - n*0.75) ** 2) / (n/20))
    inits.append(('bimodal', bim / bim.sum()))
    
    return inits


def check_score(a, label, best_seq, best_score):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = compute_score(a_norm)
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
