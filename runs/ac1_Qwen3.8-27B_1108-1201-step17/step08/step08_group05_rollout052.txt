    F = rfft(f, fft_s)

    #   k ranges from j to j+n-1 (so that k-j ranges from 0 to n-1)
    #   and k must be < 2n-1

    # where f_flipped[j] = f[n-1-j]

    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:

    return f_new / s

def mwu_pass(f_init, time_limit, tau_init=0.005, eta_init=0.3, decay=0.999, n_top=None):
    """Run MWU for a given time."""
    t0 = time.monotonic()
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    tau = tau_init
    eta = eta_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f_new = mwu_step(f, tau=tau, eta=eta, n_top=n_top)
        
        if not np.allclose(f_new, f, atol=1e-15):
            obj = fast_convolve(f_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_new.copy()
                no_improve = 0
                f = f_new
            else:
                no_improve += 1
                if no_improve > 50:
                    tau *= 1.5
                    eta *= 0.9
                    no_improve = 0
                    if eta < 0.01:
                        break
        else:
            break
        
        if it % 100 == 0:
            if time.monotonic() - t0 > time_limit:
                break
            # Slowly increase tau (anneal)
            tau *= decay
    
    return best_f, best_obj

def lp_solve_fast(f, n, active_indices, M, warm_start=None):
    """Solve the LP with given active constraints, using warm start if available."""
    K = len(active_indices)
    if K == 0:
        return warm_start if warm_start is not None else np.ones(n) / n
    
    # Build constraint matrix (sparse-friendly)
    # Row k: A[k, lo:hi+1] = f[lo:hi+1]
    A_rows = []
    A_cols = []
    A_vals = []
    row_offset = 0
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A_rows.extend([row_offset] * len(idx))
        A_cols.extend(idx.tolist())
        A_vals.extend(f[lo:hi + 1].tolist())
        row_offset += 1
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(K, n))
    
    g_var = cp.Variable(n)
    constraints = [g_var >= 0]
    if warm_start is not None and np.all(warm_start >= -1e-9) and np.allclose(A @ warm_start, A @ warm_start, atol=1e-6):
        # Try to use warm start
        pass
    
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
    
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                return np.maximum(g_var.value, 0.0)
        except Exception:
            continue
    
    return warm_start if warm_start is not None else None

def run_lp_optimization(f_init, time_limit, top_k_init=50, top_k_max=300, label="lp"):
    """LP-based optimization with constraint generation and warm-starting."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    warm_g = None
    it = 0
    stuck = 0
    top_k = top_k_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Gradually increase top_k
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)
        
        b = fast_convolve(f)
        M = b.max()
        
        # Select active constraints
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        
        # Solve LP
        g = lp_solve_fast(f, n, top_idx.tolist(), M, warm_start=warm_g)
        
        if g is None:
            if stuck > 30:
                break
            stuck += 1
            continue
        
        warm_g = g
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 40:
                break
            continue
        stuck = 0
        
        # Normalize direction
        h = g / S_g
        
        # Check for violated constraints and add them
        fg = fast_cross_convolve(f, g)
        mask = np.zeros(len(b), dtype=bool)
        mask[top_idx] = True
        violations = np.where((fg > M + 1e-9) & (~mask))[0]
        if len(violations) > 0:
            # Add worst violations to top_idx for next iteration
            worst = np.argsort(fg[violations])[-min(10, len(violations)):]
            top_idx = np.sort(np.concatenate([top_idx, worst]))
            if len(top_idx) > top_k_max:
                top_k = min(top_k + 10, top_k_max)
                best_top = np.argpartition(b, -top_k)[-top_k:]
                top_idx = np.sort(np.union1d(top_idx, best_top))
                top_idx = np.sort(top_idx)[-top_k_max:]
        
        # Line search (backtracking)
        best_t = 0.05
        best_trial_obj = best_obj
        t = 0.5
        for _ in range(8):
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            f_trial /= s
            trial_obj = fast_convolve(f_trial).max()
            if trial_obj < best_trial_obj - 1e-14:
                best_trial_obj = trial_obj
                best_t = t
                if t > 0.3:
                    break
            t *= 0.5
        
        if best_t < 1e-6:
            # Try larger steps
            for t_try in [0.3, 0.5, 0.7]:
                f_trial = (1.0 - t_try) * f + t_try * h
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                trial_obj = fast_convolve(f_trial).max()
                if trial_obj < best_obj - 1e-14:
                    best_t = t_try
                    best_trial_obj = trial_obj
                    break
        
        if best_t < 1e-6:
            stuck += 1
            if stuck > 50:
                break
            continue
        
        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
        else:
            # Revert
            f = best_f.copy()
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")
    
    return best_f, best_obj

def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):

    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:

    return w / s

def super_gaussian_window(n, p=3.0):
    """Super-Gaussian window: exp(-|x|^p)."""
    x = np.linspace(-1, 1, n)
    w = np.exp(-np.abs(x) ** p)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def interpolate_sequence(f_src, n_dst):
    """Interpolate a sequence to a different length."""
    n_src = len(f_src)
    if n_src == n_dst:
        return f_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    f_dst = np.interp(x_dst, x_src, f_src)
    f_dst = np.maximum(f_dst, 0)
    s = f_dst.sum()
    if s < 1e-10:
        return np.ones(n_dst) / n_dst
    return f_dst / s

p[j] = sum_k w[k] * f[k-j] for valid k and k-j

Using the convolution trick:
- f_flipped[m] = f[n-1-m]
- (w * f_flipped)[m] = sum_k w[k] * f_flipped[m-k] = sum_k w[k] * f[n-1-(m-k)] = sum_k w[k] * f[n-1-m+k]
- We want p[j] = sum_k w[k] * f[k-j]
- So we need: n-1-m+k = k-j => m = n-1+j
- Therefore: p[j] = (w * f_flipped)[n-1+j]

This looks correct. The convolution (w * f_flipped) has length len(w) + n - 1 = (2n-1) + n - 1 = 3n-2.

"""
Optimized auto-convolution peak minimization using MWU pre-pass, warm-started LP
with adaptive constraint generation, perturbation escapes, and multi-length exploration.
Targets score < 1.5027 (lower is better) for the auto-convolution peak problem.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    result[np.abs(result) < 1e-14] = 0.0
    return result


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    result = irfft(F * G, fft_size)[:size]
    result[np.abs(result) < 1e-14] = 0.0
    return result


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pressure(w, f):
    """Compute p[j] = sum_k w[k] * f[k-j] for j=0..n-1 using FFT."""
    n = len(f)
    f_flipped = f[::-1]
    conv_size = len(w) + n - 1
    fft_s = 1 << (conv_size - 1).bit_length()
    W = rfft(w, fft_s)
    Ff = rfft(f_flipped, fft_s)
    conv = irfft(W * Ff, fft_s)
    return conv[n - 1:n - 1 + n]


def mwu_step(f, tau=0.01, eta=0.1, n_top=None):
    """Single multiplicative weight update step."""
    n = len(f)
    b = fast_convolve(f)

    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        b_sel = b[top_idx]
        b_max = b_sel.max()
        w_sel = np.exp((b_sel - b_max) / tau)
        w = np.zeros(len(b))
        w[top_idx] = w_sel
    else:
        b_max = b.max()
        w = np.exp((b - b_max) / tau)

    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

    p = compute_pressure(w, f)
    p_mean = p.mean()
    if p_mean < 1e-30:
        return f.copy()

    f_new = f * (1.0 - eta * p / p_mean)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    return f_new / s


def mwu_pass(f_init, time_limit, tau_init=0.005, eta_init=0.2, n_top=None):
    """Run MWU iterations for a given time budget."""
    t0 = time.monotonic()
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    tau = tau_init
    eta = eta_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        f_new = mwu_step(f, tau=tau, eta=eta, n_top=n_top)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            eta = min(eta * 1.02, 0.5)
        else:
            no_improve += 1
            if no_improve > 80:
                tau *= 1.4
                eta *= 0.85
                no_improve = 0
                if eta < 0.005:
                    break
        if it % 200 == 0:
            tau *= 0.97

    return best_f, best_obj


def solve_lp(f, n, active_idx, M, g_warm=None):
    """Solve max sum(g) s.t. g>=0, A@g<=M using sparse matrix."""
    K = len(active_idx)
    if K == 0:
        return g_warm if g_warm is not None else np.ones(n) / n

    rows = []
    cols = []
    vals = []
    row_i = 0
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rows.extend([row_i] * len(idx))
        cols.extend(idx.tolist())
        vals.extend(f[lo:hi + 1].tolist())
        row_i += 1

    A = csr_matrix((vals, (rows, cols)), shape=(K, n))

    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maxitize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])

    for solver in [cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=300000)
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                return np.maximum(g_var.value, 0.0)
        except Exception:
            continue

    return g_warm if g_warm is not None else None


def run_lp_optimization(f_init, time_limit, top_k_init=50, top_k_max=250, label="lp"):

    t0 = time.monotonic()
    n = len(f_init)

    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 25 == 0 and top_k < top_k_max:

        b = fast_convolve(f)

        top_idx_sorted = np.sort(top_idx).tolist()

        g = solve_lp(f, n, top_idx_sorted, M, g_warm=warm_g)

        if g is None:

            continue

        warm_g = g.copy()
        S_g = g.sum()

                break
            continue
        stuck = 0

        h = g / S_g

        # Check for violated constraints
        fg = fast_cross_convolve(f, h)

        mask[top_idx] = True
        violations = np.where((fg > M + 1e-9) & (~mask))[0]
        if len(violations) > 0:

            top_idx = np.sort(np.concatenate([top_idx, worst]))

            else:

                    break

            stuck += 1

                break
            # Nudge constraint set

def perturbation_escape(f, strength=0.03):
    """Apply a structured random perturbation to escape local minima."""
    n = len(f)
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
    noise = rng.standard_normal(n) * strength
    f_new = f + noise
    f_new = project_simplex(f_new)
    return f_new


def interpolate_sequence(f_src, n_dst):
    """Interpolate a sequence to a different length."""
    n_src = len(f_src)
    if n_src == n_dst:
        return f_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    f_dst = np.interp(x_dst, x_src, f_src)
    f_dst = np.maximum(f_dst, 0)
    s = f_dst.sum()
    if s < 1e-10:
        return np.ones(n_dst) / n_dst
    return f_dst / s


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / n if s < 1e-10 else w / s


def super_gaussian_window(n, p=2.5):
    x = np.linspace(-1, 1, n)
    w = np.exp(-np.abs(x) ** p)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s
