import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent_update(a, grad, lr):
    """Multiplicative weight update (mirror descent step)."""
    # a_new = a * exp(-lr * grad) / sum(a * exp(-lr * grad))
    log_a = np.where(a > 1e-300, np.log(a), -300)
    log_new = log_a - lr * grad
    log_new -= log_new.max()
    new = np.exp(log_new)
    s = new.sum()
    if s < 1e-30:
        return a.copy()
    return new / s

    # = (w * a_rev)[i + (n-1)]  ... this is getting confusing.
    
    # Let me just use the direct convolution approach:
    # grad[i] = 2*tau * sum_{j=0}^{n-1} w[i+j] * a[j]
    # 
    # Define v[i+j] = w[i+j] for i+j in range [0, 2n-2]
    # Then grad[i]/(2*tau) = sum_j v[i+j] * a[j]
    # = sum_j a[j] * v[i+j]
    # This is the (i)-th element of the "anti-convolution" of a and v.
    # 
    # Anti-convolution: (a # v)[i] = sum_j a[j] * v[i+j]
    # = (a * v_rev)[i + (len(v)-1)]  where v_rev is reversed v
    
    # v has length 2n-1. v_rev has length 2n-1.
    # (a * v_rev) has length n + (2n-1) - 1 = 3n-2.
    # (a * v_rev)[i + (2n-2)] = sum_j a[j] * v_rev[(2n-2) - j + i]... 
    # This is getting too complicated. Let me just use a different approach.
    
    # Alternative: use the fact that grad[i] = 2*tau * (w_shift * a)[i]
    # where w_shift is w shifted/appropriate.
    
    # Actually, the simplest way: 
    # c[k] = sum_i a[i] * a[k-i]
    # dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (for valid k-i)
    # grad[i] = sum_k w[k] * dc[k]/da[i] = 2 * sum_k w[k] * a[k-i]
    # 
    # Let's define the "flipped" weight: wf[m] = w[m]
    # Then sum_k w[k] * a[k-i] for k in [i, i+n-1] (valid range)
    # = (w * a_flipped)[...] 
    # 
    # Let a_flipped[m] = a[n-1-m]. Then:
    # sum_k w[k] * a[k-i] = sum_k w[k] * a_flipped[n-1-(k-i)]
    # = sum_k w[k] * a_flipped[n-1-i-k]
    # = (w * a_flipped)[n-1-i]
    
    # So grad[i] = 2 * (w * a_flipped)[n-1-i]
    
    a_flipped = a[::-1]
    # Compute w * a_flipped (linear convolution)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AF = rfft(a_flipped, fft_size)
    conv_w_af = irfft(W * AF, fft_size)[:size]
    
    # grad[i] = 2 * conv_w_af[n-1-i] for i = 0, ..., n-1
    grad = 2.0 * conv_w_af[n-1 :: -1][:n]
    # grad[i] = 2 * conv_w_af[n-1-i]
    # So grad[0] = 2*conv_w_af[n-1], grad[1] = 2*conv_w_af[n-2], etc.
    grad = 2.0 * conv_w_af[n-1::-1][:n]  # This gives grad[0]=2*conv[n-1], grad[1]=2*conv[n-2], ...
    
    return grad, c


def smooth_optimize(a_init, time_limit, tau_init=0.01, tau_final=0.001, 
                    lr_init=0.001, top_k=None, label="smooth"):
    """
    Smooth optimization using logsumexp + mirror descent.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-10:
        a /= s
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Anneal parameters
        progress = (time.monotonic() - t0) / time_limit
        tau = tau_init * (tau_final / tau_init) ** progress
        lr = lr_init * (1.0 - 0.5 * progress)
        
        # Compute smooth gradient
        grad, c = smooth_gradient(a, tau, top_k=top_k)
        
        # Mirror descent update
        a_new = mirror_descent_update(a, grad, lr)
        
        # Occasionally add small noise for exploration
        if it % 500 == 499:
            noise = np.random.default_rng(it).standard_normal(n) * 0.001
            a_new = np.maximum(a_new + noise, 0)
            a_new /= a_new.sum()
        
        a = a_new
        it += 1
        
        # Evaluate every 100 iterations
        if it % 100 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 10:
                # Reset to best and add noise
                a = best_a + np.random.default_rng(it).standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a /= a.sum()
                no_improve = 0
            
            if it % 1000 == 0:
                print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                      f"tau={tau:.5f} t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, eps=1e-7):
    """Solve: max sum(g) s.t. g>=0, A@g <= M. Returns g or None."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    g_val = None
    # Try PDP first (fastest for this type of problem)
    try:
        prob.solve(solver=cp.PLP)
        if g.value is not None:
            g_val = np.maximum(g.value, 0.0)
    except Exception:
        pass
    if g_val is None:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
        except Exception:
            pass
    if g_val is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=20000)
                if g.value is not None:
                    g_val = np.maximum(g.value, 0.0)
            except Exception:
                pass
    if g_val is None or np.any(np.isnan(g_val)):
        return None
    return g_val
