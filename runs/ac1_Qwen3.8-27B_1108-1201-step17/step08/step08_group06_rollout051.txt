"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based subgradient direction finding (novel: linearized QP step)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def lp_subgradient_step(a):
    """
    Solve LP to find optimal perturbation direction.
    min t s.t. conv[k] + 2*(delta*a)[k] <= t, sum(delta)=0, delta >= -a
    
    Returns the optimal delta direction.
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    # Build the constraint matrix for (delta * a)[k]
    # For k in [0, 2n-2], (delta*a)[k] = sum_j delta[j] * a[k-j] where 0<=k-j<n
    # So coefficient of delta[j] in constraint k is a[k-j] if 0<=k-j<n else 0
    
    # The LP:
    # Variables: delta[0..n-1], t
    # min t
    # s.t. 2 * sum_j a[k-j]*delta[j] - t <= conv[k] - t ... 
    
    # Actually let me reformulate:
    # min t
    # s.t. sum_j 2*a[k-j]*delta[j] <= t - conv[k] for k=0..2n-2
    #      sum_j delta[j] = 0
    #      delta[j] >= -a[j] for all j
    
    # Rewrite: -sum_j 2*a[k-j]*delta[j] + t >= conv[k] - 2n+1... 
    # Let me use standard form for linprog (minimizes c^T x s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq)
    
    # Variables x = [delta_0, ..., delta_{n-1}, t]  (n+1 variables)
    # Objective: minimize t, so c = [0, 0, ..., 0, 1]
    # Inequality constraints: 2*sum_j a[k-j]*delta[j] - t <= conv[k] for each k
    #   => sum_j 2*a[k-j]*delta[j] + (-1)*t <= conv[k]
    # Equality constraint: sum_j delta[j] = 0 => sum_j delta[j] + 0*t = 0
    # Bounds: delta[j] >= -a[j], delta[j] <= inf; t >= -inf, t <= inf
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.full(2 * n - 1, 0.0)
    
    for k in range(2 * n - 1):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_ub[k, j] = 2.0 * a[kj]
        A_ub[k, n] = -1.0  # coefficient of t
        b_ub[k] = conv[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(delta) = 0
    b_eq = np.zeros(1)
    
    bounds = [(-a[j], None) for j in range(n)] + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x[:n]


def lp_subgradient_step_fast(a):
    """
    Faster version using sparse structure for the LP.
    For large n, builds the matrix more efficiently.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Build A_ub efficiently
    # A_ub[k, j] = 2*a[k-j] if 0 <= k-j < n else 0
    # This is a banded matrix structure
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ub[j:j+n, j] = 2.0 * a
    A_ub[:, n] = -1.0
    
    b_ub = conv.copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lower = np.zeros(n + 1)
    lower[:n] = -a
    upper = np.zeros(n + 1)
    upper[:] = np.inf
    
    bounds = list(zip(lower, upper))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x[:n]


def backtrack_line_search(a, delta, initial_score, max_iter=30):
    """Backtracking line search along a + alpha * delta."""
    alpha = 1.0
    best_alpha = 0.0
    best_score = initial_score
    
    for _ in range(max_iter):
        a_trial = a + alpha * delta
        if (a_trial < -1e-15).any():
            alpha *= 0.5
            continue
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            break
        a_trial = a_trial / s
        sc = compute_score(a_trial)
        if sc < best_score - 1e-12:
            best_score = sc
            best_alpha = alpha
        alpha *= 0.5
        if alpha < 1e-8:
            break
    
    return best_alpha, best_score


def multiplicative_updates(a, time_limit, T_init=100.0, T_final=10.0, 
                           n_iters=None, verbose=True):
    """
    Mirror descent / multiplicative weight updates.
    Update: a_new[j] proportional to a[j] * exp(-eta * grad_j)
    where grad_j is the subgradient of the max-convolution at a.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    if n_iters is None:
        n_iters = 100000
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Temperature: anneal from T_init to T_final
        progress = min(1.0, it / max(1, n_iters * 0.8))
        T = T_init * (T_final / T_init) ** progress
        
        # Softmax weights on the convolution
        shifted = (conv - M) / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(conv)
            w[M.argmax()] = 1.0
        else:
            w = w / w_sum
        
        # Gradient: grad_j = 2 * sum_k w[k] * a[k-j] (for valid k-j)
        # This is 2 * (w * a) restricted to first n elements... 
        # Actually: grad_j = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
        # = 2 * (full_convolve_w_a)[j] where we need the right slice
        # conv_wa[k] = sum_j w[k-j]*a[j] ... no that's not right either.
        # Let me be careful: (a*a)[k] = sum_j a[j]*a[k-j]
        # d/d a[j] of (a*a)[k] = a[k-j] + a[j] (by symmetry, since k-j and j both appear)
        # Wait no. (a*a)[k] = sum_j a[j]*a[k-j]. 
        # d/dx_j = a[k-j] + (if k-j != j) a[k-j]... 
        # Actually d/dx_j [sum_i a[i]*a[k-i]] = a[k-j] (from the i=j term) + a[k-j] (from the i=k-j term if k-j != j)
        # = 2*a[k-j] if k != 2j, or a[j] if k == 2j
        # For simplicity, let's use 2*a[k-j] (off-by-one at the center doesn't matter much)
        
        # grad_j = 2 * sum_k w[k] * a[k-j] (approximately, ignoring the diagonal correction)
        # This is 2 * (w_conv_a_reversed)[j] where w_conv_a is the full conv of w and a
        
        # Actually: sum_k w[k] * a[k-j] for k in [j, j+n-1] 
        # = sum_{k'=0}^{n-1} w[j+k'] * a[k']
        # This is the cross-correlation of w and a at lag j
        
        # Let me compute it via FFT:
        # cross_corr[j] = sum_k w[k]*a[k-j] = sum_{k'} w[j+k']*a[k']
        # = (w * reverse(a))[2n-2-j] ... hmm this is getting complicated.
        
        # Let me just use the direct computation via convolution:
        # grad[j] = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j]
        # = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
        
        # This is the cross-correlation of w and a.
        # cross_corr[j] = <w[j:j+n], a[0:n]>
        
        # Can compute as: reverse(a) convolved with w, then take appropriate slice
        # (w * rev_a)[j + n - 1] = sum_m w[j+m] * rev_a[n-1-m] = sum_m w[j+m] * a[m]
        
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        rev_a = a[::-1]
        RA = rfft(rev_a, fft_size)
        cross = irfft(W * RA, fft_size)
        grad = 2.0 * cross[n - 1: n - 1 + n]
        
        # Multiplicative update
        eta = 1.0 / (n * 0.1)  # learning rate
        log_update = -eta * grad
        log_update -= log_update.max()  # for numerical stability
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MU iter={it}, T={T:.2f}, score={compute_score(a):.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def peak_aware_coord_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent that specifically targets peak reduction.
    For each peak, identify the top contributors and try to reduce them.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        conv = fast_convolve(a)
        # Find top peaks
        n_peaks = min(10, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Contributing indices for peak pk
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            
            if js >= je:
                continue
            
            # Contribution of each pair (i, pk-i)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[int(pk) - idx]
            total = contribs.sum()
            
            if total < 1e-15:
                continue
            
            # Try moving mass from high-contribution positions to low-contribution ones
            # Sort by contribution (descending)
            order = np.argsort(contribs)[::-1]
            
            # Top contributors to reduce
            top_idx = idx[order[:min(5, len(idx))]]
            # Bottom contributors (or non-contributors) to receive mass
            non_contrib = np.setdiff1d(np.arange(n), idx)
            
            for i_idx in top_idx:
                if a[i_idx] < 1e-14:
                    continue
                # Find a target to move mass to
                targets = list(non_contrib[rng.integers(0, len(non_contrib), size=min(3, len(non_contrib)))] if len(non_contrib) > 0 else np.array([]))
                # Also try other low-contribution positions in the same peak range
                if len(order) > 5:
                    targets += [int(idx[order[5] + rng.integers(0, len(order)-5)])]
                
                for j_idx in targets:
                    if j_idx == i_idx:
                        continue
                    # Try different transfer amounts
                    for frac in [0.01, 0.03, 0.08, 0.15]:
                        delta = a[i_idx] * frac
                        a_trial = a.copy()
                        a_trial[i_idx] -= delta
                        a_trial[j_idx] += delta
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Perturb to escape
                noise_scale = 0.01 / sqrt(max(1, no_improve))  # Hmm, need to be careful
                a = best_a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PC iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss

def lp_iteration(a, time_limit, max_iters=500, verbose=True):
    """
    Main LP-based iteration with backtracking line search.
    At each step: solve LP for direction, line search for step size.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        delta = lp_subgradient_step_fast(a)
        if delta is None:
            break
        
        # Check if delta is meaningful
        delta_norm = np.abs(delta).max()
        if delta_norm < 1e-15:
            break
        
        # Normalize delta for the line search
        # The LP gives us a direction; we scale it
        # The constraint delta >= -a ensures a + alpha*delta >= 0 for alpha in [0,1]
        # So we line search over alpha in [0, 1]
        
        best_alpha, best_sc = backtrack_line_search(a, delta, best_score, max_iter=25)
        
        if best_alpha > 0:
            a = a + best_alpha * delta
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={it}, alpha={best_alpha:.4f}, score={compute_score(a):.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def lp_iteration_batched(a, time_limit, batch_size=5, verbose=True):
    """
    LP iteration that takes multiple LP steps in a batch before line searching.
    This is more efficient when LP solving is the bottleneck.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Accumulate directions
        direction = np.zeros(n)
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            delta = lp_subgradient_step_fast(a + direction)
            if delta is None:
                break
            direction += delta
            it += 1
        
        if np.abs(direction).max() < 1e-15:
            break
        
        best_alpha, best_sc = backtrack_line_search(a, direction, best_score, max_iter=25)
        
        if best_alpha > 0:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LPB it={it}, alpha={best_alpha:.4f}, score={compute_score(a):.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def subgradient_logspace(a0, time_limit, eta_init=0.01, verbose=True):
    """
    Subgradient method in log-space.
    z_{k+1} = z_k - eta * proj(a * grad_subgrad)
    where grad_subgrad is the subgradient of max(conv) at a.
    """
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        k_star = conv.argmax()
        M = conv[k_star]
        
        # Subgradient of max(conv) w.r.t. a:
        # grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        # (approximately, ignoring the diagonal correction)
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[k_star - j]
        
        # Project to log-space: d_z = a * (grad - dot(grad, a))
        S = np.dot(grad, a)
        d_z = a * (grad - S)
        
        # Step size: eta_k = eta_init / (1 + 0.1*k)
        eta = eta_init / (1.0 + 0.01 * it)
        
        z = z - eta * d_z
        a = softmax_from_z(z)
        
        sc = compute_score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG iter={it}, eta={eta:.6f}, score={sc:.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score
