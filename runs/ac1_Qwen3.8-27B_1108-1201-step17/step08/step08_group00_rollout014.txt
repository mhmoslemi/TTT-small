def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def variance_objective(z):
    """Variance of autocorrelation as a function of log-domain z."""
    a = softmax_from_z(z)
    c = fast_convolve(a)
    mc = c.mean()
    return float(np.sum((c - mc) ** 2))


def variance_gradient(z):
    """Gradient of variance objective w.r.t. log-domain z."""
    n = len(z)
    a = softmax_from_z(z)
    c = fast_convolve(a)
    mc = c.mean()
    d = c - mc
    # grad_a[j] = 4 * sum_m a[m] * d[j+m]
    grad_a = np.zeros(n)
    for j in range(n):
        m_max = min(n, len(d) - j)

            grad_a[j] = 4.0 * np.dot(a[:m_max], d[j:j + m_max])
    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def lbgfs_logsumexp_refine(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=8, iters_per_stage=2000):
    """L-BFGS with LogSumExp smoothing and multi-stage temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    def make_func(T_val):
        def func(zv):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            av = softmax_from_z(zv)
            b = fast_convolve(av)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(zv):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zv)
            av = softmax_from_z(zv)
            b = fast_convolve(av)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * av[k - idx]
            S = np.dot(grad_a, av)
            return av * (grad_a - S)
        return jac

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        max_it = min(iters_per_stage, int(remaining * 200))
        try:
            res = minimize(
                fun=make_func(stage_T), x0=z, jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': max_it, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': max_it * 50, 'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Variance minimization (new approach) ===
    print(f"\n=== Phase 1: Variance Minimization n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 180.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_var = variance_minimize(best_seq, t_alloc, maxiter=3000)
        try_update(a_var, "variance")

        # Now do a short LP run from the variance-minimized solution
        rem = deadline - time.monotonic()
        t_lp_short = min(rem * 0.1, 90.0)
        if t_lp_short > 20:
            a_lp_from_var, _ = nesterov_lp_refine(a_var, t_lp_short, solver=solver, n_max_iters=15000)
            try_update(a_lp_from_var, "lp_from_variance")

    # === Phase 2: Extended Nesterov LP ===
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=80000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 3: L-BFGS LogSumExp with aggressive annealing ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 30:
        print(f"  T: 100->0.1, budget={t_alloc:.0f}s")
        a_lb = lbgfs_logsumexp_refine(best_seq, t_alloc, T_start=100.0, T_end=0.1,
                                      n_stages=9, iters_per_stage=1500)
        try_update(a_lb, "lbgfs_main")

    # === Phase 4: Second Nesterov LP round ===
    print(f"\n=== Phase 4: Nesterov LP round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=30000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Multi-perturbation escape ===
    print(f"\n=== Phase 6: Multi-Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_cur = len(best_seq)
        perturbations = [
            ("gauss_03", best_seq + 0.03 * rng.standard_normal(n_cur)),
            ("gauss_06", best_seq + 0.06 * rng.standard_normal(n_cur)),
            ("sinusoidal", best_seq * (1 + 0.05 * np.sin(2 * np.pi * 3 * np.arange(n_cur) / n_cur))),
            ("frequency", _freq_perturb(best_seq, rng)),
            ("block", _block_perturb(best_seq, rng)),
        ]

        for name, perturbed in perturbations:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps

            # Short LP refinement from perturbed start
            t_pert = min(rem * 0.25, 60.0)
            if t_pert > 15:
                a_pert, _ = nesterov_lp_refine(perturbed, t_pert, solver=solver, n_max_iters=8000)
                try_update(a_pert, f"pert_{name}")

            # Very short L-BFGS polish
            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None:
                t_pert_lb = min(rem * 0.15, 30.0)
                if t_pert_lb > 10:
                    a_pert2 = lbgfs_logsumexp_refine(best_seq, t_pert_lb,
                                                     T_start=5.0, T_end=0.2,
                                                     n_stages=5, iters_per_stage=500)
                    try_update(a_pert2, f"pert_lb_{name}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for mult in [1.25, 1.5, 2.0]:
        alt_n = int(n_best * mult)
        if 512 <= alt_n <= 3072 and alt_n != n_best:
            alt_lengths.append(alt_n)
    # Also try slightly shorter
    alt_n_short = int(n_best * 0.75)
    if 384 <= alt_n_short <= 4096 and alt_n_short != n_best:
        alt_lengths.append(alt_n_short)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.30, 180.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Variance + LP
        t_var = t_total * 0.2
        a_alt_var = variance_minimize(init_alt, t_var, maxiter=1000)
        try_update(a_alt_var, f"var_n{alt_n}")

        rem = deadline - time.monotonic()
        t_lp_alt = min(rem * 0.5, 80.0)
        if t_lp_alt > 20:
            a_alt, _ = nesterov_lp_refine(a_alt_var, t_lp_alt, solver=solver, n_max_iters=15000)
            try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 25 and len(best_seq) == alt_n:
            t_lb_alt = min(rem * 0.3, 40.0)
            if t_lb_alt > 10:
                a_alt2 = lbgfs_logsumexp_refine(best_seq, t_lb_alt,
                                                T_start=10.0, T_end=0.3,
                                                n_stages=5, iters_per_stage=500)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 60.0)
        a_fin = lbgfs_logsumexp_refine(best_seq, t_fin, T_start=3.0, T_end=0.05,
                                       n_stages=6, iters_per_stage=1000)
        try_update(a_fin, "final_lbgfs_lowT")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.25, 40.0)
        a_ps2, _ = peak_shave(best_seq, t_ps, n_steps=20000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.4, 30.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def _freq_perturb(a, rng):
    """Perturb in frequency domain."""
    n = len(a)
    X = np.fft.rfft(a.astype(np.float64))
    X_perturbed = X * (1 + 0.05 * rng.standard_normal(len(X)) + 0j * rng.standard_normal(len(X)))
    a_perturbed = np.fft.irfft(X_perturbed, n)
    return a_perturbed


def _block_perturb(a, rng):
    """Perturb by scaling random blocks."""
    n = len(a)
    a_perturbed = a.copy()
    n_blocks = rng.integers(3, 8)
    for _ in range(n_blocks):
        start = rng.integers(0, n - 10)
        length = rng.integers(5, max(6, n // 10))
        scale = rng.uniform(0.8, 1.2)
        a_perturbed[start:start + length] *= scale
    a_perturbed = np.maximum(a_perturbed, 0)
    s = a_perturbed.sum()
    if s > 1e-15:
        a_perturbed = a_perturbed / s
    return a_perturbed
