            # Restart with perturbation
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            no_improve = 0
            tau = max(tau * 2.0, tau_init * 0.5)
            continue

        _, w = softmax_loss_weights(b, tau)
        grad = compute_grad_softmax(f, w)

        # Subtract mean to ensure we stay on simplex
        grad = grad - grad.sum() / n

        t_adam += 1
        bc1 = 1.0 - beta1 ** t_adam
        bc2 = 1.0 - beta2 ** t_adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / bc1
        v_hat = v / bc2

        lr = lr_base / np.sqrt(t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)
        f = f_new

        if it % 200 == 0:
            tau = max(tau * tau_decay ** 200, tau_min)

    return best_f, best_peak


def lp_optimize(f_init, time_limit, top_k_init=50, top_k_max=300):
    """LP-based optimization with constraint generation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    f /= s

    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        b = fast_convolve(f)
        M = b.max()

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        active_list = top_idx.tolist()
        active_set = set(active_list)

        g = None
        for cg_iter in range(20):
            if time.monotonic() - t0 > time_limit:
                break

            K = len(active_list)
            if K > 400:
                b_active = b[active_list]
                keep = np.argsort(b_active)[-400:]
                active_list = [active_list[i] for i in keep]
                active_set = set(active_list)
                K = len(active_list)

            A = np.zeros((K, n))
            for row_i in range(K):
                k = active_list[row_i]
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                A[row_i, lo:hi + 1] = f[lo:hi + 1]

            g_var = cp.Variable(n)
            prob = cp.Problem(
                cp.Maximize(cp.sum(g_var)),
                [g_var >= 0, A @ g_var <= M]
            )

            solved = False
            for solver in [cp.PDLP, cp.SCS]:
                try:
                    prob.solve(solver=solver, max_iters=200000)
                    solved = True
                    break
                except Exception:
                    continue

            if not solved or g_var.value is None:
                if g is not None:
                    break
                break

            g = np.maximum(g_var.value, 0.0)
            if np.any(np.isnan(g)) or np.any(np.isinf(g)):
                if cg_iter > 0:
                    break
                break

            if time.monotonic() - t0 > time_limit * 0.7:
                break

            fg = fast_cross_convolve(f, g)
            mask = np.zeros(len(b), dtype=bool)
            mask[active_list] = True
            violations = np.where((fg > M + 1e-9) & (~mask))[0]

            if len(violations) == 0:
                break

            worst = np.argsort(fg[violations])[-min(8, len(violations)):]
            for vi in worst:
                v_int = int(vi)
                if v_int not in active_set:
                    active_set.add(v_int)
                    active_list.append(v_int)

        if g is None or g.sum() <= 1.0 + 1e-5:
            # Fallback: steepest descent on top peaks
            b = fast_convolve(f)
            n_peaks = min(5, len(b))
            top_peaks = np.argpartition(b, -n_peaks)[-n_peaks:]
            g = np.zeros(n)
            for pk in top_peaks:
                lo = max(0, pk - n + 1)
                hi = min(n - 1, pk)
                for j in range(lo, hi + 1):
                    g[j] += f[pk - j]
            if g.sum() <= 1.0 + 1e-5:
                stuck += 1
                if stuck > 30:
                    break
                top_k = min(top_k * 2, top_k_max)
                continue

        h = g / g.sum()

        # Line search
        best_t = 0.05
        best_trial = best_peak
        for t_candidate in [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t_candidate) * f + t_candidate * h
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-10:
                continue
            f_trial /= ss
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t_candidate

        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-10:
            break
        f_new /= ss
        f = f_new

        cur_peak = fast_convolve(f).max()
        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break

        if it % 20 == 0:
            score = 2.0 * n * best_peak
            print(f"    LP iter={it} peak={best_peak:.10f} score={score:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_peak


def local_search(f_init, time_limit, n_trials=5000):
    """Local coordinate descent: perturb pairs of elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    f /= f.sum()

    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    rng = np.random.default_rng(12345)

    for it in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        if (it + 1) % 500 == 0:
            remaining_fractions = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        else:
            remaining_fractions = [0.01, 0.02, 0.03, 0.05, 0.1, 0.2]

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        delta = rng.choice([-1, 1]) * rng.uniform(0.001, 0.05)
        f_trial = f.copy()
        f_trial[i] += delta
        f_trial[j] -= delta
        f_trial = np.maximum(f_trial, 0.0)
        ss = f_trial.sum()
        if ss < 1e-10 or abs(ss - 1.0) > 0.01:
            continue
        f_trial /= ss

        obj = fast_convolve(f_trial).max()
        if obj < best_peak - 1e-14:
            f = f_trial
            best_peak = obj
            best_f = f.copy()

    return best_f, best_peak


def multi_length_optimize(f_best, score_best, time_limit):
    """Try different lengths by interpolation and new windows."""
    t0 = time.monotonic()
    n_base = len(f_best)

    candidate_lengths = []
    for factor in [0.75, 0.8, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0]:
        target_n = int(round(n_base * factor))
        if 512 <= target_n <= 3500 and target_n != n_base:
            candidate_lengths.append(target_n)

    # Also try some fixed good lengths
    for ln in [768, 1024, 1152, 1280, 1536, 1920, 2048]:
        if ln != n_base and 512 <= ln <= 3500 and ln not in candidate_lengths:
            candidate_lengths.append(ln)

    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_base))[:5]

    best_seq_overall = f_best.copy()
    best_score_overall = score_best

    per_time = time_limit / max(1, len(candidate_lengths))

    for target_n in candidate_lengths:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        t_alloc = min(per_time * 1.2, remaining)

        # Initialize by interpolation
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, f_best)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        # Quick softmax GD
        t1 = t_alloc * 0.5
        f_new, peak_new = softmax_gradient_descent(f_interp, time_limit=t1,
                                                     tau_init=0.002, tau_min=0.0005)
        score_new = 2.0 * target_n * peak_new
        if np.isfinite(score_new) and score_new < best_score_overall:
            best_score_overall = score_new
            best_seq_overall = f_new.copy()
            print(f"  Multi-len n={target_n}: score={score_new:.6f} (NEW BEST)")

        # Quick LP if time remains
        remaining2 = t_alloc - (time.monotonic() - t0 + (time_limit - t_alloc))
        t2 = t_alloc * 0.5
        if t_lock_check(remaining):
            f_lp, peak_lp = lp_optimize(f_new, time_limit=t2, top_k_init=40, top_k_max=150)
            score_lp = 2.0 * target_n * peak_lp
            if np.isfinite(score_lp) and score_lp < best_score_overall:
                best_score_overall = score_lp
                best_seq_overall = f_lp.copy()
                print(f"  Multi-len n={target_n} LP: score={score_lp:.6f} (NEW BEST)")

    return best_seq_overall, best_score_overall


def t_lock_check(remaining):
    return remaining > 30


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    s = w.sum()
    return w / s


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def raised_cosine(n):
    x = np.linspace(0, 1, n)
    w = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    w = np.maximum(w, 0)
    s = w.sum()
    return w / s
