import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, weights):
    """Compute weighted subgradient of max_k (f*f)[k]."""
    n = len(f)
    b = fast_convolve(f)
    grad = np.zeros(n)
    for k in range(len(b)):

        wm = float(weights[k])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * wm * f[k - m]
    return grad

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(f, b, weights):
    """Compute weighted subgradient using FFT.
    
    grad[m] = 2 * sum_k w[k] * f[k-m] for valid k-m in [0, n-1]
    = 2 * (w * f_rev)[n-1+m] where f_rev[i] = f[n-1-i]
    """
    n = len(f)
    # w has length 2n-1
    # We want: grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * f[j]
    # Let f_rev[j] = f[n-1-j], then f[j] = f_rev[n-1-j]
    # grad[m] = 2 * sum_j w[m+j] * f_rev[n-1-j]
    # Let i = n-1-j, j = n-1-i:
    # grad[m] = 2 * sum_i w[m+n-1-i] * f_rev[i]
    # = 2 * (w_conv * f_rev)[m+n-1] where w_conv is w reversed... 
    
    # Actually, let's use the convolution identity:
    # (a * b)[k] = sum_i a[i] * b[k-i]
    # We want sum_j w[m+j] * f[j]
    # Let's define w2[j] = w[j] (length 2n-1), f2[j] = f[j] (length n)
    # sum_j w[m+j] * f[j] = sum_j w2[m+j] * f2[j]
    # This is the cross-correlation of w2 and f2 at lag m
    # = (w2 * f2_rev)[m + n - 1] where f2_rev[j] = f2[n-1-j]
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    W = rfft(weights, fft_size)
    Frev = rfft(f[::-1], fft_size)
    corr = irfft(W * Frev, fft_size)[:n + n - 1]
    # cross-correlation at lag m is corr[m + n - 1]... 
    # Actually (w2 * f2_rev)[k] = sum_i w2[i] * f2_rev[k-i] = sum_i w2[i] * f2[n-1-k+i]
    # We want sum_j w2[m+j] * f2[j]. Let i = m+j, j = i-m:
    # = sum_i w2[i] * f2[i-m] (for i from m to m+n-1)
    # = (w2 * f2_rev)[m + n - 1] since f2_rev[k-i] = f2[n-1-k+i]
    # Need n-1-k+i = i-m, so k = n-1+m
    # (w2 * f2_rev)[n-1+m] = sum_i w2[i] * f2[n-1-(n-1+m)+i] = sum_i w2[i] * f2[i-m]
    # Yes! So grad[m] = 2 * corr[n-1+m]
    
    # Hmm wait, let me re-derive. 
    # (a*b)[k] = sum_i a[i]*b[k-i]
    # Let a = weights (length 2n-1), b = f[::-1] (length n)
    # (a*b)[k] = sum_i weights[i] * f[n-1-(k-i)] = sum_i weights[i] * f[i-k+n-1]
    # We want sum_j weights[m+j] * f[j]
    # Set i = m+j: sum_i weights[i] * f[i-m]
    # So we need f[i-k+n-1] = f[i-m], which means k = n-1+m
    # grad[m] = 2 * (weights * f[::-1])[n-1+m]
    
    return 2.0 * corr[n - 1 : n - 1 + n]


def softmax_weights(b, temperature):
    """Compute softmax weights over the convolution peaks."""
    scaled = b / max(temperature, 1e-10)
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= s
    return w


def run_lp_optimization(f_init, time_limit, label="lp", top_k_init=30, max_constraints=300, m_reduction=0.0):
    """
    LP-based optimization with constraint generation.
    Maximizes sum(g) s.t. g >= 0, (f*g)[k] <= (1-m_reduction)*M for active constraints.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(f)
        M = b.max()

        # Use reduced M to force improvement
        M_bound = M * (1.0 - m_reduction)

        # Get active constraints (top peaks)
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        active_list = np.sort(top_idx).tolist()
        active_set = set(active_list)

        g = None
        for cg_iter in range(10):
            if time.monotonic() - t0 > time_limit:
                break

            K = len(active_list)
            if K > max_constraints:
                b_active = b[active_list]
                keep = np.argsort(b_active)[-max_constraints:]
                active_list = [active_list[i] for i in keep]
                active_set = set(active_list)
                K = len(active_list)

            # Build constraint matrix: A[i, j] = f[j] if j is in [max(0, k-n+1), min(n-1, k)] where k = active_list[i]
            A = np.zeros((K, n))
            for row_i in range(K):
                k = active_list[row_i]
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                A[row_i, lo:hi + 1] = f[lo:hi + 1]

            g_var = cp.Variable(n)
            prob = cp.Problem(
                cp.Maximize(cp.sum(g_var)),
                [g_var >= 0, A @ g_var <= M_bound]
            )

            solved = False
            for solver in [cp.PDLP, cp.SCS]:
                try:
                    prob.solve(solver=solver, max_iters=500000)
                    solved = True
                    break
                except Exception:
                    continue

            if not solved or g_var.value is None:
                if g is not None:
                    break
                return None, M

            g = np.maximum(g_var.value, 0.0)
            if np.any(np.isnan(g)) or np.any(np.isinf(g)):
                if cg_iter > 0:
                    break
                return None, M

            # Check for violated constraints
            if time.monotonic() - t0 > time_limit * 0.8:
                break

            fg = fast_cross_convolve(f, g)
            mask = np.zeros(len(b), dtype=bool)
            mask[active_list] = True
            violations = np.where((fg > M_bound + 1e-9) & (~mask))[0]

            if len(violations) == 0:
                break

            worst = np.argsort(fg[violations])[-min(5, len(violations)):]
            for v in worst:
                vi = int(v)
                if vi not in active_set:
                    active_set.add(vi)
                    active_list.append(vi)

        if g is None:
            break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            # Try with less aggressive M
            m_reduction = max(0.0, m_reduction - 0.005)
            continue
        else:
            stuck = 0

        h = g / S_g

        # Line search
        best_t = 0.05
        best_trial_obj = fast_convolve((1 - 0.05) * f + 0.05 * h).max()
        for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial_obj:
                best_trial_obj = obj
                best_t = t

        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s

        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            # Gradually increase m_reduction
            m_reduction = min(m_reduction + 0.002, 0.02)
        else:
            stuck += 1
            m_reduction = max(0.0, m_reduction - 0.003)

        if it % 20 == 0:
            print(f"    {label} iter={it} obj={best_obj:.12f} score={2*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def nesterov_subgradient(f_init, time_limit, label="nesterov"):
    """Nesterov-accelerated projected subgradient descent."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    y = f.copy()  # Nesterov look-ahead
    v = np.zeros(n)  # momentum
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    lr = 1.0 / (np.sqrt(n) * 10)  # initial learning rate
    temperature = 0.01
    it = 0
    cost_history = []
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        b = fast_convolve(y)
        M = b.max()
        
        # Softmax weights with temperature
        w = softmax_weights(b, temperature)
        
        # Compute subgradient
        grad = compute_subgradient_fft(y, b, w)
        
        # Nesterov update
        v = 0.9 * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Nesterov look-ahead
        beta = 2.0 / (it + 2.0)  # Nesterov step
        y = f + beta * (f_new - f)
        y = project_simplex(y)
        
        f = f_new
        
        # Adaptive learning rate
        if it % 100 == 0:
            obj = fast_convolve(f).max()
            cost_history.append(obj)
            if len(cost_history) > 5:
                recent = np.array(cost_history[-5:])
                if recent[-1] < recent[0] - 1e-8:
                    lr = min(lr * 1.05, 0.5)
                else:
                    lr = max(lr * 0.9, 1e-6)
            
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
            
            # Anneal temperature
            temperature = max(temperature * 0.95, 0.001)
            
            if it % 500 == 0:
                print(f"    {label} iter={it} obj={best_obj:.12f} score={2*n*best_obj:.6f} lr={lr:.6f} t={time.monotonic()-t0:.0f}s")
    
    # Final check
    obj = fast_convolve(f).max()
    if obj < best_obj - 1e-14:
        best_obj = obj
        best_f = f.copy()
    
    return best_f, best_obj


def direct_subgradient(f_init, time_limit, label="sg", n_top=50):
    """Direct projected subgradient with backtracking line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = 1.0 / n
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Top-k weights
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w = np.zeros(len(b))
            w[top_idx] = 1.0 / n_top
        else:
            w = np.ones(len(b)) / len(b)
        
        # Subgradient
        grad = compute_subgradient_fft(f, b, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(10):
            f_trial = project_simplex(f - alpha * grad)
            obj = fast_convolve(f_trial).max()
            if obj < M - 1e-12:
                f = f_trial
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if improved:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.1, 0.5)
            else:
                no_improve += 1
        else:
            no_improve += 1
            lr = max(lr * 0.9, 1e-8)
        
        if no_improve > 50:
            break
    
    return best_f, best_obj


def bimodal_init(n, separation=0.5, width=0.15):
    """Bimodal Gaussian initialization."""
    x = np.linspace(0, 1, n)
    c1 = 0.5 - separation / 2
    c2 = 0.5 + separation / 2
    f = np.exp(-((x - c1) / width) ** 2) + np.exp(-((x - c2) / width) ** 2)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def trimodal_init(n, width=0.12):
    """Trimodal Gaussian initialization."""
    x = np.linspace(0, 1, n)
    f = (np.exp(-((x - 0.2) / width) ** 2) +
         np.exp(-((x - 0.5) / width) ** 2) +
         np.exp(-((x - 0.8) / width) ** 2))
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def smooth_noise_init(n, noise_level=0.3, seed=42):
    """Smoothed random noise initialization."""
    rng = np.random.default_rng(seed)
    f = 1.0 + noise_level * rng.standard_normal(n)
    # Smooth with moving average
    kernel_size = max(1, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    f = np.convolve(f, kernel, mode='same')
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def cosine_window_init(n, alpha=0.5):
    """Tukey-like window with adjustable alpha."""
    x = np.linspace(0, 1, n)
    if alpha >= 1.0:
        w = np.ones(n)
    elif alpha <= 0.0:
        w = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        w = np.ones(n)
        r = int(alpha * n / 2)
        if r > 0:
            for i in range(r):
                w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n) - 1)))
                w[n - 1 - i] = w[i]
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(f, weights):
    """Compute weighted subgradient of max_k (f*f)[k] using FFT.
    
    grad[m] = 2 * sum_j weights[m+j] * f[j] for j=0..n-1
    Computed as 2 * (weights * f_rev)[n-1+m] via convolution.

    conv_len = (2 * n - 1) + n - 1  # = 3n - 2

    W = rfft(weights, fft_size)
    Frev = rfft(f[::-1], fft_size)
    conv = irfft(W * Frev, fft_size)[:conv_len]
    return 2.0 * conv[n - 1: n - 1 + n]


def softmax_weights(b, temperature):
    """Compute softmax weights over convolution values."""
    scaled = b / max(temperature, 1e-15)

    if s < 1e-30:

        w[b.argmax()] = 1.0
    else:

            stuck = 0

        h = g / S_g

        # Line search: evaluate several step sizes
        best_t = 0.05
        best_trial = fast_convolve(np.maximum((1 - 0.05) * f + 0.05 * h, 0)).max()
        for t_val in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4]:
            f_trial = (1.0 - t_val) * f + t_val * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t_val

        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s

        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            f = f_new
            m_reduction = min(m_reduction + 0.002, 0.02)
        else:
            f = f_new  # still accept for exploration
            stuck += 1
            m_reduction = max(0.0, m_reduction - 0.005)

        if it % 20 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} sc={2 * n * best_obj:.6f} "
                  f"mr={m_reduction:.4f} t={time.monotonic() - t0:.0f}s")

    return best_f, best_obj


def nesterov_subgradient(f_init, time_limit, label="nesterov"):
    """Momentum-based projected subgradient with adaptive step size."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    y = f.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    lr = 1.0 / (np.sqrt(n) * 5.0)
    temperature = 0.01
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(y)
        M = b.max()
        w = softmax_weights(b, temperature)
        grad = compute_subgradient_fft(y, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Momentum update
        v = 0.9 * v - lr * grad / max(grad_norm, 1e-15)
        f_new = project_simplex(f + v)

        # Nesterov look-ahead
        beta = 2.0 / (it + 2.0)
        y = project_simplex(f + beta * (f_new - f))
        f = f_new

        # Periodic evaluation
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 1.0)
            else:
                no_improve += 1
                if no_improve > 10:
                    lr = max(lr * 0.9, 1e-6)
                    no_improve = 0
                if no_improve > 40:
                    # Restart with perturbation
                    rng = np.random.default_rng(it)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    y = f.copy()
                    v = np.zeros(n)
                    no_improve = 0

            temperature = max(temperature * 0.97, 0.001)

            if it % 500 == 0:
                print(f"    {label} it={it} obj={best_obj:.12f} sc={2 * n * best_obj:.6f} "
                      f"lr={lr:.6f} t={time.monotonic() - t0:.0f}s")

    obj = fast_convolve(f).max()
    if obj < best_obj - 1e-14:
        best_obj = obj
        best_f = f.copy()

    return best_f, best_obj


def direct_descent(f_init, time_limit, label="descent", n_top=30):
    """Direct projected subgradient with backtracking line search on top peaks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = 1.0 / (n * 0.5)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Top-k uniform weights
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w = np.zeros(len(b))
            w[top_idx] = 1.0 / n_top
        else:
            w = np.ones(len(b)) / len(b)

        grad = compute_subgradient_fft(f, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Backtracking
        alpha = lr
        improved = False
        for _ in range(8):
            f_trial = project_simplex(f - alpha * grad / gn)
            obj = fast_convolve(f_trial).max()
            if obj < M - 1e-12:
                f = f_trial
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break

        if improved:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            lr = min(lr * 1.05, 1.0)
        else:
            no_improve += 1
            lr = max(lr * 0.9, 1e-10)

        if no_improve > 60:
            break

    return best_f, best_obj


def bimodal_init(n, separation=0.4, width=0.18):
    """Bimodal Gaussian initialization."""
    x = np.linspace(0, 1, n)
    c1 = 0.5 - separation / 2
    c2 = 0.5 + separation / 2
    f = np.exp(-((x - c1) / width) ** 2) + np.exp(-((x - c2) / width) ** 2)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def trimodal_init(n, width=0.12):
    """Trimodal Gaussian initialization."""
    x = np.linspace(0, 1, n)
    f = (np.exp(-((x - 0.2) / width) ** 2) +
         np.exp(-((x - 0.5) / width) ** 2) +
         np.exp(-((x - 0.8) / width) ** 2))
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def smooth_noise_init(n, noise_level=0.2, seed=42):
    """Smoothed random noise initialization."""
    rng = np.random.default_rng(seed)
    f = 1.0 + noise_level * rng.standard_normal(n)
    ksize = max(1, n // 40)
    kernel = np.ones(ksize) / ksize
    f = np.convolve(f, kernel, mode='same')
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def tukey_init(n, r=0.3):
    """Tukey window initialization."""
    x = np.linspace(0, 1, n)
    w = np.ones(n)
     ri = int(r * n / 2)
    if ri > 0:
        for i in range(ri):
            val = 0.5 * (1 + np.cos(np.pi * (2 * i / (r * n) - 1)))
            w[i] = val
            w[n - 1 - i] = val
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interp_to_length(f, target_n):
    """Interpolate sequence to target length and normalize."""
    old_n = len(f)
    if old_n == target_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    return f_new / s
