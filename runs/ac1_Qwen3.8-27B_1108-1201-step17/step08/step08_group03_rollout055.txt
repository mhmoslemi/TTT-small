def line_search_quadratic(f, h, b_f, b_fh, b_h):
    """
    Find optimal t in [0,1] minimizing max_k [(1-t)^2 * b_f[k] + 2t(1-t) * b_fh[k] + t^2 * b_h[k]]
    where b_f = conv(f,f), b_fh = conv(f,h), b_h = conv(h,h).
    """
    n = len(f)
    
    def objective(t):
        # b(k)(t) = (1-t)^2 * b_f[k] + 2t(1-t) * b_fh[k] + t^2 * b_h[k]
        val = (1-t)**2 * b_f + 2*t*(1-t) * b_fh + t**2 * b_h
        # Clip to non-negative (should be non-negative for non-negative sequences, but numerical issues)
        return val.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    lo, hi = 0.0, 1.0
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = objective(c)
    fd = objective(d)
    
    for _ in range(60):  # 60 iterations gives precision of ~1e-12
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1 - gr) * (hi - lo)
            fc = objective(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = objective(d)
    
    t_opt = (lo + hi) / 2
    return t_opt

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix A where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, time_per_solve=2.0):
    """Solve: max sum(g) s.t. g>=0, f*g[k] <= M for k in active_idx."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    t0 = time.monotonic()
    # Try PDLP first (fastest for this structure)
    try:
        prob.solve(solver=cp.PDLP, max_iters=50000, time_limit=time_per_solve)
    except Exception:
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):

def quadratic_line_search(b_f, b_fh, b_h, n_max_iter=50):
    """
    Find t in [0,1] minimizing max_k [(1-t)^2*b_f[k] + 2t(1-t)*b_fh[k] + t^2*b_h[k]].
    Uses golden section search.
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    
    def obj(t):
        coeff = (1.0 - t) * (1.0 - t) * b_f + 2.0 * t * (1.0 - t) * b_fh + t * t * b_h
        return coeff.max()
    
    c = lo + (1.0 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(n_max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1.0 - gr) * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)
    
    t_opt = (lo + hi) / 2.0
    return t_opt

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Gradient warmup (fast, to escape uniform if parent is bad) ===
    if best_score > 1.55:
        rem = deadline - time.monotonic()
        t1 = min(rem * 0.1, 100.0)
        if t1 > 30:
            print(f"\n=== Phase 1: Gradient warmup (n={len(parent)}, budget={t1:.0f}s) ===")
            f1, _ = gradient_warmup(parent, time_limit=t1, n_top=50, label="gw1")
            s1 = float(evaluate_sequence(f1.tolist()))
            if np.isfinite(s1) and s1 < best_score:
                best_score = s1
                best_seq = f1.copy()
                print(f"  NEW BEST [gw1]: {s1:.6f}")

    # === Phase 2: Main LP with cutting planes ===
    print(f"\n=== Phase 2: CP-LP main (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 50:
        n2 = len(best_seq)
        print(f"  Budget: {t2:.0f}s, n={n2}, k=50->400")
        f2, obj2 = lp_cutting_plane(best_seq, time_limit=t2, initial_k=50, max_k=400, label="cp_main")
        s2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(s2) and s2 < best_score:
            best_score = s2
            best_seq = f2.copy()
            print(f"  NEW BEST [cp_main]: {s2:.6f}")

    # === Phase 3: Smoothing pass ===
    print(f"\n=== Phase 3: Smoothing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.08, 80.0)
    if t3 > 10:
        f3, _ = smoothing_pass(best_seq, time_limit=t3, label="sm1")
        s3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(s3) and s3 < best_score:
            best_score = s3
            best_seq = f3.copy()
            print(f"  NEW BEST [sm1]: {s3:.6f}")
        # LP polish after smoothing
        rem3 = deadline - time.monotonic()
        if rem3 > 50 and s3 < best_score + 0.01:
            t3b = min(rem3 * 0.2, 150.0)
            f3b, _ = lp_cutting_plane(best_seq, time_limit=t3b, initial_k=60, max_k=350, label="cp_sm")
            s3b = float(evaluate_sequence(f3b.tolist()))
            if np.isfinite(s3b) and s3b < best_score:
                best_score = s3b
                best_seq = f3b.copy()
                print(f"  NEW BEST [cp_sm]: {s3b:.6f}")

    # === Phase 4: Deeper LP with more constraints ===
    print(f"\n=== Phase 4: CP-LP deep (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.2, 300.0)
    if t4 > 50:
        n4 = len(best_seq)
        print(f"  Budget: {t4:.0f}s, n={n4}, k=80->500")
        f4, _ = lp_cutting_plane(best_seq, time_limit=t4, initial_k=80, max_k=500, label="cp_deep")
        s4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [cp_deep]: {s4:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.9, 1.1, 1.2]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    
    for target_n in target_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t5 = min(rem * 0.15, 180.0)
        if t5 < 40:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s_int = f_interp.sum()
        if s_int > 1e-10:
            f_interp = f_interp / s_int
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  CP-LP n={target_n}, budget={t5:.0f}s")
        f5, _ = lp_cutting_plane(f_interp, time_limit=t5, initial_k=40, max_k=300, label=f"n{target_n}")
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {s5:.6f}")

    # === Phase 6: LP on different initializations ===
    rem = deadline - time.monotonic()
    if rem > 100:
        print(f"\n=== Phase 6: Multi-start (best={best_score:.6f}) ===")
        n_b = len(best_seq)
        
        # Try a Gaussian-like initialization at the best length
        x = np.linspace(0, 1, n_b)
        f_gauss = np.exp(-4.0 * (x - 0.5) ** 2)
        f_gauss = np.maximum(f_gauss, 0.0)
        s_g = f_gauss.sum()
        if s_g > 1e-10:
            f_gauss = f_gauss / s_g
        
        t6 = min(rem * 0.12, 150.0)
        if t6 > 40:
            print(f"  Gaussian init, n={n_b}, budget={t6:.0f}s")
            f6, _ = lp_cutting_plane(f_gauss, time_limit=t6, initial_k=40, max_k=300, label="gauss")
            s6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(s6) and s6 < best_score:
                best_score = s6
                best_seq = f6.copy()
                print(f"  NEW BEST [gauss]: {s6:.6f}")

        # Try interpolated mix of best and uniform
        rem6 = deadline - time.monotonic()
        if rem6 > 60:
            f_mix = 0.7 * best_seq + 0.3 * np.ones(n_b) / n_b
            s_m = f_mix.sum()
            f_mix = f_mix / s_m
            t6b = min(rem6 * 0.15, 150.0)
            if t6b > 30:
                print(f"  Mix init, budget={t6b:.0f}s")
                f6b, _ = lp_cutting_plane(f_mix, time_limit=t6b, initial_k=40, max_k=300, label="mix")
                s6b = float(evaluate_sequence(f6b.tolist()))
                if np.isfinite(s6b) and s6b < best_score:
                    best_score = s6b
                    best_seq = f6b.copy()
                    print(f"  NEW BEST [mix]: {s6b:.6f}")

    # === Phase 7: Final LP burst ===
    print(f"\n=== Phase 7: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.2, 200.0)
    if t7 > 30:
        n7 = len(best_seq)
        print(f"  Budget: {t7:.0f}s, n={n7}, k=100->600")
        f7, _ = lp_cutting_plane(best_seq, time_limit=t7, initial_k=100, max_k=600, label="final")
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [final]: {s7:.6f}")

    # === Phase 8: Smoothing + coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 8: Polish (best={best_score:.6f}) ===")
        t8 = min(rem * 0.4, 100.0)
        f8, _ = smoothing_pass(best_seq, time_limit=t8 * 0.4, label="sm2")
        s8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(s8) and s8 < best_score:
            best_score = s8
            best_seq = f8.copy()
            print(f"  NEW BEST [sm2]: {s8:.6f}")
        
        rem8 = deadline - time.monotonic()
        if rem8 > 20:
            t8b = min(rem8 * 0.5, 80.0)
            f8b, _ = coordinate_polish(best_seq, time_limit=t8b, label="cd")
            s8b = float(evaluate_sequence(f8b.tolist()))
            if np.isfinite(s8b) and s8b < best_score:
                best_score = s8b
                best_seq = f8b.copy()
                print(f"  NEW BEST [cd]: {s8b:.6f}")

    # Final evaluation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        print("WARNING: best_seq invalid, reverting to parent!")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    # Final safety: ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    s_final = best_seq.sum()
    if s_final < 0.01:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
