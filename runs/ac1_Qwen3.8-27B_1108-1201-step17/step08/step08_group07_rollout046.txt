import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response_max_sum(f, M, prev_g=None):
    """max sum(g) s.t. C(f)@g <= M, g >= 0"""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is None else prev_g.copy()
    if g.value is None:
        return prev_g if prev_g is not None else np.ones(n) / n
    return np.maximum(g.value, 0)


def solve_best_response_min_max(f, prev_g=None):
    """min t s.t. C(f)@g <= t, sum(g)=1, g >= 0"""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()

def lp_fictitious_play_nesterov(a0, time_limit, step=0.5, verbose=True, label="LP-Nest"):
    """LP fictitious play with Nesterov momentum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    y = a.copy()  # Nesterov lookahead
    beta = 0.9  # momentum parameter
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        beta = (1 + np.sqrt(1 + 4 * beta * beta)) / 2  # Nesterov momentum schedule
        
        conv = fast_convolve(y)
        M = conv.max()

        g = solve_best_response_max_sum(y, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        # Nesterov update for y
        y = a_new + (beta - 1) / (beta + 1) * (a_new - a)
        y = np.maximum(y, 0)
        y = y / y.sum()

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score

For `k = 0`: `sum_n w[n] * a[n]` for `n = 0, ..., len(a)-1 = n-1`. Since `w` has length `2n-1`, `w[n]` is valid for `n = 0, ..., n-1`. So this is `sum_{n=0}^{n-1} w[n] * a[n]`.

For `k = j` (small j): `sum_n w[n+j] * a[n]` for `n = 0, ..., n-1` (where `n+j <= 2n-2`). This is exactly `sum_i a[i] * w[i+j]`!

So `np.correlate(w, a, mode='full')[:n]` gives me `cross_corr[0], ..., cross_corr[n-1]`.

But for large n, this O(n^2) computation might be slow. Let me use FFT instead.

Using FFT: The cross-correlation `c[j] = sum_i a[i] * w[i+j]` can be computed as:
`c = IFFT(FFT(a_reversed) * conj(FFT(w)))` ... 

Actually, let me just use the property:
`sum_i a[i] * w[i+j]` = element `j` of the "valid" cross-correlation.

In FFT terms, if we define:
- `A = FFT(a_padded)` where `a_padded` is `a` zero-padded to length `fft_size`
- `W = FFT(w_padded)` where `w_padded` is `w` zero-padded to length `fft_size`
- `fft_size >= (n-1) + (2n-2) + 1 = 3n-2`

Then the linear cross-correlation is:
`cross = IFFT(conj(A) * W)` (or `IFFT(A * conj(W))` depending on convention)
