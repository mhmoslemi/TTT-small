def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

- Careful golden-section line search
- Multi-length exploration with window initialization

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, top_k=50, temperature=None):
    """Compute (soft) subgradient of max auto-convolution peak.
    
    If temperature is None, use hard subgradient (top-k peaks, equal weight).
    Otherwise, use log-sum-exp weighting with given temperature.
    """
    n = len(f)
    b = fast_convolve(f)
    
    if temperature is not None and temperature > 0:
        # Soft weights via log-sum-exp
        shifted = b / temperature
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(w)
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        # Restrict to top-k for efficiency
        if top_k < len(w):
            top_idx = np.argpartition(w, -top_k)[-top_k:]
            w_new = np.zeros_like(w)
            w_new[top_idx] = w[top_idx]
            w = w_new / w_new.sum()
    else:
        # Hard: top-k equal weight
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        w = np.zeros_like(b)
        w[top_idx] = 1.0 / len(top_idx)
    
    # Gradient: g[i] = sum_k w[k] * 2 * f[k-i] for valid k-i
    g = np.zeros(n)
    active = np.where(w > 1e-15)[0]
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        idx = np.arange(lo, hi + 1)
        g[idx] += 2.0 * w[k] * f[k - idx]
    
    return g, b

            K = len(active_list)

def golden_line_search(f, direction, max_evals=25):
    """Golden section search for optimal step size.
    
    Minimizes peak of auto-convolution along f -> (1-t)*f + t*direction.
    """
    n = len(f)

    def obj_at(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    # Coarse search
    candidates = [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    best_t = 0.05
    best_obj = obj_at(0.05)
    for t in candidates:
        obj = obj_at(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Golden section refinement
    idx = candidates.index(best_t)
    lo = candidates[max(0, idx - 1)]
    hi = candidates[min(len(candidates) - 1, idx + 1)]
    if lo == hi:
        return best_t

    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(15):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
        if hi - lo < 1e-6:
            break

    return (lo + hi) / 2

        return None, M

    return g, M


def golden_line_search(f, direction, max_evals=30):
    """Golden section search for optimal step size minimizing convolution peak."""
    n = len(f)

    def obj_at(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return float(fast_convolve(f_trial).max())

    candidates = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    best_t = 0.05
    best_obj = obj_at(0.05)
    for t in candidates:
        obj = obj_at(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    idx = candidates.index(best_t)
    lo = candidates[max(0, idx - 1)]
    hi = candidates[min(len(candidates) - 1, idx + 1)]
    if hi - lo < 1e-8:
        return best_t

    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(14):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
        if hi - lo < 1e-7:
            break

    return (lo + hi) / 2


def lp_optimize(f_init, time_limit, top_k_init=50, top_k_max=200, label="lp"):
    """Full LP-based optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())
    it = 0
    stuck = 0
    top_k = top_k_init
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 25 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        rem = time_limit - (time.monotonic() - t0)
        t_lp = min(rem * 0.2, 20.0, max(5.0, rem * 0.4))

        g, M = lp_direction(f, n, top_k_init=top_k,
                           max_constraints=min(400, top_k + 150), time_limit=t_lp)

        if g is None:
            # Fallback: soft subgradient direction
            grad, _ = compute_soft_subgradient(f, T=5.0, top_k=min(top_k, 30))
            if grad.max() > 1e-15:
                f_trial = project_simplex(f - 0.01 * grad)
                g = f_trial.copy()
            else:
                g = np.ones(n) / n
            S_g = g.sum()
            if S_g <= 1.0 + 1e-5:
                break
        else:
            S_g = g.sum()
            if S_g <= 1.0 + 1e-5:
                stuck += 1
                if stuck > 25:
                    # Perturb and try again
                    f = project_simplex(f + rng.standard_normal(n) * 0.002)
                    stuck = 0
                    continue
                top_k = min(top_k * 2, top_k_max)
                continue

        h = g / S_g
        best_t = golden_line_search(f, h)
        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s

        cur_obj = float(fast_convolve(f_new).max())
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 35:
                # Reset to best and perturb
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                stuck = 0

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"top_k={top_k} t={elapsed:.0f}s")

    return best_f


def nesterov_optimize(f_init, time_limit, T_init=15.0, T_min=1.0, top_k=40):
    """Nesterov-accelerated subgradient with adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    y = f.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())
    lr = 0.003
    T = T_init
    momentum = 0.65
    no_improve = 0
    it = 0
    t_nesterov = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        T = max(T * 0.995, T_min)

        g, b = compute_soft_subgradient(y, T=T, top_k=top_k)

        v = momentum * v - lr * g
        f_candidate = y + v
        f = project_simplex(f_candidate)

        # Nesterov extrapolation
        t_nesterov = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov))
        y = f + ((t_nesterov - 1.0) / t_nesterov) * (f - f_candidate + lr * g)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-10:
            y /= sy

        if it % 10 == 0:
            obj = float(fast_convolve(f).max())
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.015, 0.015)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    no_improve = 0
                if no_improve > 20:
                    y = f.copy()
                    v = np.zeros(n)
                    no_improve = 0

        if it % 200 == 0:
            print(f"    nest it={it} T={T:.2f} lr={lr:.5f} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f


def direct_descent(f_init, time_limit, n_peaks=8, lr_schedule=None):
    """Direct subgradient descent with adaptive step size on true objective."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        k_eff = min(n_peaks, len(b))
        if k_eff < len(b):
            top_idx = np.argpartition(b, -k_eff)[-k_eff:]
        else:
            top_idx = np.arange(len(b))

        grad = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            grad[lo:hi + 1] += f[lo:hi + 1]

        if grad.max() < 1e-15:
            break

        # Try multiple step sizes
        alphas = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]
        best_alpha = None
        best_trial = best_obj

        for alpha in alphas:
            f_trial = project_simplex(f - alpha * grad)
            obj = float(fast_convolve(f_trial).max())
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            for alpha in [0.0003, 0.0005, 0.0008]:
                f_trial = project_simplex(f - alpha * grad)
                obj = float(fast_convolve(f_trial).max())
                if obj < best_trial - 1e-14:
                    best_trial = obj
                    best_alpha = alpha

        if best_alpha is None:
            stuck += 1
            if stuck > 30:
                break
            continue

        f = project_simplex(f - best_alpha * grad)
        obj = float(fast_convolve(f).max())
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break

        if it % 50 == 0:
            print(f"    direct it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return best_f


def flat_top_window(n):
    """Generate normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2.0 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Generate normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    w = np.maximum(w, 1e-10)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolate_sequence(f, target_n):
    """Interpolate sequence to target length, preserving sum."""
    old_n = len(f)
    if old_n == target_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(target_n) / target_n
    return f_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-phase hybrid (LP + Nesterov + direct descent)
    with multi-length exploration and perturbation restarts.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1152) / 1152.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent found, using uniform n=1152, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    print(f"Initial: n={n_parent}, score={best_score:.8f}")
    print(f"Target: 1.5027, gap={best_score - 1.5027:.6f}")

    def try_update(f_new, label):
        nonlocal best_seq, best_score
        if f_new is None:
            return
        s = f_new.sum()
        if s < 1e-10:
            return
        f_norm = f_new / s
        score = float(evaluate_sequence(f_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f_norm.copy()
            print(f"  *** NEW BEST [{label}]: score={score:.8f} (n={len(f_norm)})")

    # === Phase 1: LP optimization on parent (35% time) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP on parent (n={n_parent}, {t1:.0f}s) ===")
        f1 = lp_optimize(parent, time_limit=t1, top_k_init=50, top_k_max=180, label="lp1")
        try_update(f1, "lp1")

    # === Phase 2: Nesterov subgradient (10% time) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 120.0)
    if t2 > 15:
        print(f"\n=== Phase 2: Nesterov (n={len(best_seq)}, {t2:.0f}s) ===")
        f2 = nesterov_optimize(best_seq, time_limit=t2, T_init=10.0, T_min=1.0, top_k=50)
        try_update(f2, "nest")

    # === Phase 3: LP deep with more constraints (25% time) ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 380.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP deep (n={len(best_seq)}, {t3:.0f}s) ===")
        f3 = lp_optimize(best_seq, time_limit=t3, top_k_init=80, top_k_max=250, label="lp3")
        try_update(f3, "lp3")

    # === Phase 4: Direct descent (8% time) ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 80.0)
    if t4 > 10:
        print(f"\n=== Phase 4: Direct descent (n={len(best_seq)}, {t4:.0f}s) ===")
        f4 = direct_descent(best_seq, time_limit=t4, n_peaks=10)
        try_update(f4, "direct")

    # === Phase 5: Multi-length exploration (15% time) ===
    rem = deadline - time.monotonic()
    t5_total = min(rem * 0.15, 180.0)
    if t5_total > 30:
        print(f"\n=== Phase 5: Multi-length exploration ({t5_total:.0f}s) ===")
        n_base = len(best_seq)
        candidate_lengths = [n_base - 128, n_base + 128, n_base + 256,
                            1024, 1280, 1536]
        candidate_lengths = [c for c in candidate_lengths if 512 <= c <= 3200 and c != n_base]
        candidate_lengths = sorted(set(candidate_lengths))[:4]

        for target_n in candidate_lengths:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t5 = min(rem * 0.3, t5_total / max(1, len(candidate_lengths)), 80.0)
            if t5 < 15:
                break

            # Initialize by interpolation
            f_init = interpolate_sequence(best_seq, target_n)

            # Also try window initialization
            f_window = flat_top_window(target_n)

            # Pick the better initialization
            s1 = float(evaluate_sequence(f_init.tolist()))
            s2 = float(evaluate_sequence(f_window.tolist()))
            if s1 < s2:
                f_start = f_init
                init_name = "interp"
            else:
                f_start = f_window
                init_name = "window"

            print(f"  n={target_n} init={init_name} score={min(s1,s2):.8f}, budget={t5:.0f}s")

            # LP optimize
            f5 = lp_optimize(f_start, time_limit=t5, top_k_init=40, top_k_max=150,
                           label=f"n{target_n}")
            try_update(f5, f"n{target_n}")

    # === Phase 6: Final Nesterov polish (7% time) ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.07, 60.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Final Nesterov polish ({t6:.0f}s) ===")
        f6 = nesterov_optimize(best_seq, time_limit=t6, T_init=5.0, T_min=0.5, top_k=60)
        try_update(f6, "nest_final")

    # === Phase 7: Final LP burst (remaining time, min 10s) ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t7 = rem * 0.7
        print(f"\n=== Phase 7: Final LP ({t7:.0f}s) ===")
        f7 = lp_optimize(best_seq, time_limit=t7, top_k_init=100, top_k_max=300, label="lp_final")
        try_update(f7, "lp_final")

    # === Final verification ===
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027, gap: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
