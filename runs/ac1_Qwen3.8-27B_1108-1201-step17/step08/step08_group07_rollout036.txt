import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def gradient_at_max(a, k):
    """Gradient of (a*a)[k] at point a."""
    n = len(a)
    g = np.zeros(n)
    for m in range(n):
        idx = k - m
        if 0 <= idx < n:
            g[m] = 2.0 * a[idx]
    return g

def cutting_plane(a0, time_limit, max_cuts=500, verbose=True):
    """Cutting plane method for min max_k (a*a)[k] s.t. a in simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Accumulated cuts: list of (gradient, constant) where grad^T a + const <= t
    cuts_grads = []
    cuts_consts = []
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        f_val = conv[k_star]
        
        # Add new cut
        g = gradient_at_max(a, k_star)
        const = f_val - g @ a  # Linearization: (a*a)[k] ≈ f_val + g^T(a-a) = g^T a + (f_val - g^T a)
        cuts_grads.append(g)
        cuts_consts.append(const)
        
        # Limit number of cuts
        if len(cuts_grads) > max_cuts:
            # Remove oldest cuts (or least tight ones)
            cuts_grads = cuts_grads[-max_cuts:]
            cuts_consts = cuts_consts[-max_cuts:]
        
        # Solve LP
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, t_var >= 0]
        for g_c, c_c in zip(cuts_grads, cuts_consts):
            constraints.append(g_c @ a_var + c_c <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CLARABEL, warm_start=True)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break
        
        if a_var.value is None:
            break
        
        a_new = np.maximum(a_var.value, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        t_lp = t_var.value

        conv_new = fast_convolve(a_new)
        f_new = conv_new.max()
        gap = abs(f_new - t_lp)
        
        if f_new < best_score - 1e-14:
            best_score = f_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CP iter={iters}, t_lp={t_lp:.8f}, f={f_new:.8f}, gap={gap:.2e}, "
                  f"best={best_score:.8f}, cuts={len(cuts_grads)}, time={elapsed:.0f}s")
        
        if gap < 1e-8:
            if verbose:
                print(f"  Converged at iter {iters}, score={best_score:.8f}")
            break
    
    return best_a, best_score

Uses the fact that max_k (a*a)[k] is convex in a, enabling cutting plane method.
Combined with spectral initialization (flat-top window) and multi-length search.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def gradient_at_max(a, k):
    """Gradient of (a*a)[k] at point a. g[m] = 2*a[k-m] if valid."""
    n = len(a)
    g = np.zeros(n)
    m = np.arange(n)
    idx = k - m
    valid = (idx >= 0) & (idx < n)
    g[valid] = 2.0 * a[idx[valid]]
    return g


def flat_top_window(n):
    """Generate a flat-top window normalized to sum 1."""
    if n <= 2:
        return np.ones(n) / n
    j = np.arange(n)
    c0 = 0.21557895
    c1 = 0.41663158
    c2 = 0.277263158
    c3 = 0.083578947
    c4 = 0.006947368
    w = (c0 - c1 * np.cos(2 * np.pi * j / (n - 1))
           + c2 * np.cos(4 * np.pi * j / (n - 1))
           - c3 * np.cos(6 * np.pi * j / (n - 1))
           + c4 * np.cos(8 * np.pi * j / (n - 1)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def raised_cosine_window(n):
    """Raised cosine window normalized to sum 1."""
    j = np.arange(n)
    w = np.sin(np.pi * (j + 0.5) / n) ** 2
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def cutting_plane(a0, time_limit, max_cuts=300, top_k=3, verbose=True, label="CP"):
    """
    Cutting plane method for min max_k (a*a)[k] s.t. a in simplex.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    
    best_a = a.copy()
    best_f = fast_convolve(a).max()
    t0 = time.monotonic()
    
    cuts_grads = []
    cuts_consts = []
    
    # Add initial cuts for top k values
    conv = fast_convolve(a)
    top_indices = np.argsort(conv)[-top_k:]
    for k in top_indices:
        g = gradient_at_max(a, int(k))
        f_k = conv[int(k)]
        const = f_k - g @ a
        cuts_grads.append(g)
        cuts_consts.append(const)
    
    iters = 0
    stall = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Solve LP
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, t_var >= 0]
        for g_c, c_c in zip(cuts_grads, cuts_consts):
            constraints.append(g_c @ a_var + c_c <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CLARABEL, warm_start=(iters > 1))
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break
        
        if a_var.value is None:
            break
        
        a_new = np.maximum(a_var.value, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        t_lp = float(t_var.value)
        
        # Evaluate true objective at new point
        conv_new = fast_convolve(a_new)
        f_new = conv_new.max()
        
        if f_new < best_f - 1e-14:
            best_f = f_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Add new cuts for top k values at the new point
        top_indices = np.argsort(conv_new)[-top_k:]
        for k in top_indices:
            g = gradient_at_max(a_new, int(k))
            f_k = conv_new[int(k)]
            const = f_k - g @ a_new
            cuts_grads.append(g)
            cuts_consts.append(const)
        
        # Prune cuts if too many
        if len(cuts_grads) > max_cuts:
            # Keep only the most recent cuts
            idx_keep = np.arange(len(cuts_grads))
            # Remove from the front
            prune_count = len(cuts_grads) - max_cuts
            cuts_grads = cuts_grads[prune_count:]
            cuts_consts = consts[prune_count:]  # BUG: should be cuts_consts
        
        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, t_lp={t_lp:.8f}, f={f_new:.8f}, "
                  f"gap={abs(f_new-t_lp):.2e}, best={best_f:.8f}, "
                  f"cuts={len(cuts_grads)}, time={elapsed:.0f}s/{time_limit:.0f}s")
        
        # Convergence check
        if abs(f_new - t_lp) < 1e-10 * max(1, f_new):
            if verbose:
                print(f"  {label} Converged at iter {iters}")
            break
        
        if stall > 200:
            if verbose:
                print(f"  {label} Stalled, breaking")
            break
    
    return best_a, best_f


def frank_wolfe(a0, time_limit, verbose=True, label="FW"):
    """Frank-Wolfe for min max_k (a*a)[k] s.t. a in simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_f = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        g = gradient_at_max(a, k_star)
        
        # FW step: v = e_{argmin g}
        m_star = int(np.argmin(g))
        v = np.zeros(n)
        v[m_star] = 1.0
        
        # Step size: gamma_t = 2/(t+2) for standard FW
        gamma = 2.0 / (t + 2)
        a = (1 - gamma) * a + gamma * v
        
        f = fast_convolve(a).max()
        if f < best_f - 1e-14:
            best_f = f
            best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={t}, f={f:.8f}, best={best_f:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_f


def mirror_descent(a0, time_limit, eta=0.1, verbose=True, label="MD"):
    """Mirror descent with entropy regularization on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_f = fast_convolve(a).max()
    t0 = time.monotonic()
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        g = gradient_at_max(a, k_star)
        
        # Mirror descent step: a_new[m] ∝ a[m] * exp(-eta * g[m])
        log_a = np.log(np.maximum(a, 1e-30))
        new_weights = log_a - eta * g
        new_weights = new_weights - new_weights.max()
        a_new = np.exp(new_weights)
        a_new = a_new / a_new.sum()
        
        f = fast_convolve(a_new).max()
        if f < best_f - 1e-14:
            best_f = f
            best_a = a_new.copy()
        
        # Decrease eta
        eta *= 0.999
        a = a_new
        
        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, f={f:.8f}, best={best_f:.8f}, eta={eta:.4e}, time={elapsed:.0f}s")
    
    return best_a, best_f

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def grad_of_conv_k(a, k):
    """Gradient of (a*a)[k] at point a. g[m] = 2*a[k-m] if 0<=k-m<n."""
    n = len(a)
    g = np.zeros(n)
    m = np.arange(n)
    idx = k - m
    valid = (idx >= 0) & (idx < n)
    g[valid] = 2.0 * a[idx[valid]]
    return g


def project_simplex(v):
    """Project vector v onto the probability simplex {a >= 0, sum(a)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) - cssv > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=False, label="LP-FP"):

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()

        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

            break

        g_norm = g / g_sum

        # Line search: try several step sizes

        best_local_a = a

            if ss < 1e-15:

            a_trial = a_trial / ss
            sc = compute_score(a_trial)

                best_local_score = sc
                best_local_a = a_trial.copy()

            best_a = best_local_a

            prev_g = None

def bussgang_iteration(a0, time_limit, tau=0.5, verbose=False, label="Bussgang"):
    """Bussgang-type spectral flattening."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.01, 100.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def smoothed_objective(z, T, n):
    """Smoothed max via log-sum-exp approximation."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    """Gradient of smoothed objective w.r.t. z."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):

    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=False):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]

            for frac in [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 30:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def flat_top_window(n):
    """Flat-top window normalized to sum 1."""
    if n <= 2:
        return np.ones(n) / n
    j = np.arange(n)
    c0, c1, c2, c3, c4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
    w = (c0 - c1 * np.cos(2 * np.pi * j / (n - 1))
           + c2 * np.cos(4 * np.pi * j / (n - 1))
           - c3 * np.cos(6 * np.pi * j / (n - 1))
           + c4 * np.cos(8 * np.pi * j / (n - 1)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def raised_cosine_window(n):
    """Raised cosine window normalized to sum 1."""
    j = np.arange(n)
    w = np.sin(np.pi * (j + 0.5) / n) ** 2
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def blackmanharris_window(n):
    """Blackman-Harris window normalized to sum 1."""
    if n <= 2:
        return np.ones(n) / n
    j = np.arange(n)
    a = np.arange(n)
    c0, c1, c2, c3 = 0.35875, 0.48829, 0.14128, 0.01168
    w = c0 - c1 * np.cos(2 * np.pi * a / (n - 1)) + c2 * np.cos(4 * np.pi * a / (n - 1)) \
        - c3 * np.cos(6 * np.pi * a / (n - 1))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def check_and_update(a, label, best_seq, best_score):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: LP-FP on parent (main driver, longest budget)
    print(f"\n=== Phase 1: LP-FP on parent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 480.0)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
