"""
Accelerated optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses FISTA (accelerated projected gradient) with log-sum-exp smoothing as the
main engine, combined with LP-based refinement for final polishing.
Key innovations over previous approach:
1. FFT-based O(n log n) gradient computation enables millions of iterations
2. Adaptive beta schedule (coarse-to-fine smoothing)
3. PDLP/MOSEK solver for fast LP refinement
4. Multi-scale exploration with spectral initialization
5. Peak-targeting restarts to escape local optima
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, beta):
    """
    Compute the log-sum-exp smoothed max of conv(a,a) and its gradient.
    Returns (smoothed_value, gradient, weights).
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        val = b_max
    else:
        w = exp_shifted / sum_exp
        val = b_max + np.log(sum_exp) / beta

    # Gradient: grad[j] = 2 * sum_m a[m] * w[m+j]
    # Compute via FFT: g[j] = conv(a_flipped, w)[n-1+j]
    a_flip = a[::-1]
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    AF = rfft(a_flip, fft_size)
    WF = rfft(w, fft_size)
    conv_res = irfft(AF * WF, fft_size)
    g = conv_res[n - 1:2 * n - 1]
    # Ensure non-negative (numerical)
    g = np.maximum(g, 0.0)

    grad = 2.0 * g
    return val, grad, w


def fista_optimize(f_init, time_limit, beta_start=1.0, beta_end=200.0,
                   beta_ramp_frac=0.3, label="fista", max_stuck=200):
    """
    FISTA (accelerated projected gradient) on the log-sum-exp smoothed
    max auto-convolution, with adaptive beta schedule.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    t = 1.0  # FISTA momentum parameter: t_{k+1} = (1+sqrt(1+4t_k^2))/2
    y = f.copy()  # FISTA extrapolation point
    lr = 1.0 / (4.0 * n * beta_start)  # initial step size

    ramp_iters = int(0)  # we'll track by time

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive beta: increase over time
        frac = (time.monotonic() - t0) / time_limit
        if frac < beta_ramp_frac:
            beta = beta_start * (beta_end / beta_start) ** (frac / beta_ramp_frac)
        else:
            beta = beta_end

        # Update step size
        lr = 1.0 / (4.0 * n * beta)

        # FISTA iteration
        val, grad, w = smooth_grad(y, beta)

        z = project_simplex(y - lr * grad)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = z + ((t - 1.0) / t_new) * (z - f)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        f = z
        t = t_new

        # Check improvement periodically
        if it % 50 == 0:
            cur_obj = fast_convolve(f).max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > max_stuck:
                    # Restart with perturbation
                    rng = np.random.default_rng(it * 31 + 7)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    y = f.copy()
                    t = 1.0
                    no_improve = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            cur = fast_convolve(f).max()
            print(f"    {label} it={it} best={best_obj:.12f} cur={cur:.12f} "
                  f"beta={beta:.1f} t={elapsed:.0f}s")

    return best_f, 2.0 * n * best_obj


def multiplicative_weights(f_init, time_limit, label="mw", eta_init=0.02,
                           beta_smooth=5.0):
    """
    Multiplicative weights / exponential gradient method.
    Naturally stays in the simplex.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    eta = eta_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights on peaks
        temp = 1.0 / beta_smooth
        shifted = (b - M) / temp
        shifted = np.clip(shifted, -50, 50)
        w = np.exp(shifted)
        ws = w.sum()
        if ws > 1e-30:
            w = w / ws
        else:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0

        # Gradient: grad[j] = 2 * sum_m a[m] * w[m+j]
        a_flip = f[::-1]
        conv_size = 2 * n - 1
        fft_size = 1 << (conv_size - 1).bit_length()
        AF = rfft(a_flip, fft_size)
        WF = rfft(w, fft_size)
        conv_res = irfft(AF * WF, fft_size)
        g = np.maximum(conv_res[n - 1:2 * n - 1], 0.0)

        # Multiplicative update
        log_update = -eta * g
        log_update -= log_update.max()  # for numerical stability
        f_new = f * np.exp(log_update)
        s_new = f_new.sum()
        if s_new < 1e-300:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / s_new
        f = f_new

        # Decay eta
        eta *= 0.9995

        if it % 100 == 0:
            cur_obj = fast_convolve(f).max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 50:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    no_improve = 0
                    eta = eta_init * np.exp(-no_improve * 0.01)

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} best={best_obj:.12f} eta={eta:.6f} "
                  f"t={elapsed:.0f}s")

    return best_f, 2.0 * n * best_obj


def solve_lp_refine(f, top_k, solver_list=None, time_per_lp=5.0):
    """
    LP refinement: max sum(g) s.t. g>=0, conv(f,g)[top_k peaks] <= M.
    Returns the optimal g (normalized) or None.
    """
    n = len(f)
    b_full = fast_convolve(f)
    M = b_full.max()
    full_len = 2 * n - 1

    if top_k < full_len:
        top_idx = np.argpartition(b_full, -top_k)[-top_k:]
    else:
        top_idx = np.arange(full_len)
    top_idx = np.sort(top_idx)

    K = len(top_idx)
    # Build constraint matrix
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    g_var = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(objective, constraints)

    if solver_list is None:
        solver_list = ['PDLP', 'MOSEK', 'ECOS', 'SCS']

    g_val = None
    for solver in solver_list:
        try:
            if solver == 'PDLP':
                prob.solve(solver=cp.PDLP, max_iterations=100000,
                           abstol=1e-10, rhotol=1e-10)
            elif solver == 'MOSEK':
                prob.solve(solver=cp.MOSEK, msdk_params={"MSK_DPAR_INTPNT_CO_TOL_PFEAS": 1e-10})
            elif solver == 'ECOS':
                prob.solve(solver=cp.ECOS, max_iters=30000, abs_acceleration=1e-10)
            elif solver == 'SCS':
                prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-10, eps_rel=1e-10)
            else:
                prob.solve(solver=cp.ECOS)

            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0.0)
                if not np.any(np.isnan(g_val)) and not np.any(np.isinf(g_val)):
                    break
                else:
                    g_val = None
        except Exception:
            continue

    if g_val is None or g_val.sum() <= 1.0 + 1e-4:
        return None

    g_val = g_val / g_val.sum()
    return g_val


def lp_iteration(f_init, time_limit, top_k_init=50, top_k_max=500,
                 label="lp", growth_rate=1.15, max_stuck=60):
    """LP-based direction finding with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        g = solve_lp_refine(f, top_k)
        if g is None:
            stuck += 2
            if stuck > max_stuck:
                break
            top_k = min(top_k + 10, 2 * n - 1)
            continue

        # Line search
        best_t = 0.0
        best_obj_try = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s_t = f_trial.sum()
            if s_t < 1e-15:
                continue
            f_trial = f_trial / s_t
            obj_t = fast_convolve(f_trial).max()
            if obj_t < best_obj_try:
                best_obj_try = obj_t
                best_t = t

        if best_t == 0.0:
            stuck += 1
            if stuck > max_stuck:
                break
            top_k = min(int(top_k * growth_rate), 2 * n - 1)
            continue

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > max_stuck:
                break

        # Grow constraints
        if it % 15 == 0 and top_k < top_k_max:
            top_k = min(int(top_k * 1.2), top_k_max)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} k={top_k} "
                  f"t={elapsed:.0f}s")

    return best_f, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def coordinate_descent(f_init, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    multipliers = [0.3, 0.5, 0.7, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5, 2.0]
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j = f[j]
            best_obj_j = best_obj
            for mult in multipliers:
                f[j] *= mult
                s = f.sum()
                if s < 1e-12:
                    f[j] = best_j
                    continue
                f = f / s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_j = f[j]
            f[j] = best_j
            s = f.sum()
            if s > 1e-12:
                f = f / s
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                best_f = f.copy()
                improved = True
        if not improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, 2.0 * n * best_obj

        f_res, _ = lp_iteration(best_seq, time_limit=t6,
                                top_k_init=100, top_k_max=min(800, 2 * n_b - 1),
                                label="lp2", growth_rate=1.3, max_stuck=50)
        try_update(f_res, f"lp2 n={n_b}")

    # ============================================================
    # Phase 7: Coordinate descent polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 7: CD polish (best={best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t7 = min(rem * 0.05, 50.0)
    if t7 > 5:
        print(f"  CD n={len(best_seq)}, budget={t7:.0f}s")
        f_res, _ = coordinate_descent(best_seq, time_limit=t7)
        try_update(f_res, "cd")

    # ============================================================
    # Phase 8: One more FISTA with very high beta (fine-tuning)
    # ============================================================
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.50, 120.0)
    if t8 > 10:
        n_b = len(best_seq)
        print(f"\n  Final FISTA n={n_b}, budget={t8:.0f}s")
        f_res, _ = fista_optimize(best_seq, time_limit=t8,
                                  beta_start=100.0, beta_end=1000.0,
                                  beta_ramp_frac=0.4,
                                  label="fista_final", max_stuck=500)
        try_update(f_res, "fista_final")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
