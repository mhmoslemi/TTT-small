"""
Adam-based LSE optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothing with adaptive temperature, Adam optimizer,
and multiple restarts with different initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_gradient(a, T, n):
    """Compute gradient of LSE-smoothed objective w.r.t. z (log parameters)."""
    conv = fast_convolve(a)
    shifted = conv / T
    max_sh = shifted.max()
    w = np.exp(shifted - max_sh)
    w = w / w.sum()
    
    # w has length 2n-1
    # grad_a[i] = 2n * (S1[i] + S2[i])
    # S1[i] = sum_j a[j] * w[i+j] (for valid indices)
    # S2[i] = sum_j a[j] * w[i-j] (for valid indices)
    
    fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    A = rfft(a, fft_size)
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    
    S2 = irfft(A * W, fft_size)[:n]
    S1 = irfft(A_rev * W, fft_size)[n - 1:n - 1 + n]
    
    grad_a = 2.0 * n * (S1 + S2)
    
    # Convert to z-gradient via softmax parameterization
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    return grad_z


def adam_lse_optimize(a0, time_limit, T_init=1000.0, lr_init=0.01, 
                       beta1=0.9, beta2=0.999, eps=1e-8, 
                       T_min=50.0, T_decay=0.9995, verbose=True, label="Adam-LSE"):
    """Adam optimizer on LSE-smoothed objective with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    T = T_init
    lr = lr_init
    
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        t += 1
        a = softmax(z)
        grad_z = lse_gradient(a, T, n)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Anneal temperature
        T = max(T_min, T * T_decay)
        
        # Compute actual score
        a_new = softmax(z)
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={t}, T={T:.1f}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=200, maxiter=5000):
    """L-BFGS polish on LSE-smoothed objective."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_sh = shifted.max()
        lse = max_sh + np.log(np.exp(shifted - max_sh).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax(z)
        return lse_gradient(a, T, n)
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 30, 'disp': False})
        a_out = softmax(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def random_restart(n, seed):
    """Generate a random valid initial sequence."""
    rng = np.random.default_rng(seed)
    # Start with a semi-structured initialization
    x = np.linspace(0, 1, n)
    # Cosine-shaped
    a = np.cos(np.pi * x) ** 2
    a = np.maximum(a, 0)
    # Add random perturbation
    a += rng.standard_normal(n) * 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

"""
Adam-based LSE optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses correct FFT-based gradient (4n * correlation), Adam with temperature
annealing, multiple restarts, and L-BFGS polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_fn(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_gradient(a, T, n):
    """Gradient of LSE-smoothed objective w.r.t. z (log-softmax params).
    Uses the identity: d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n else 0.
    So grad_a[i] = 4n * sum_j a[j]*w[i+j] = 4n * conv(a_rev, w)[n-1+i].
    """
    conv = fast_convolve(a)
    shifted = conv / T
    max_sh = shifted.max()
    w = np.exp(shifted - max_sh)
    w = w / w.sum()
    # grad_a[i] = 4n * sum_j a[j] * w[i+j]
    # = 4n * conv(a_rev, w)[n-1+i]
    fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    grad_a = 4.0 * n * irfft(A_rev * W, fft_size)[n - 1:n - 1 + n]
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    return grad_z


def adam_lse_optimize(a0, time_limit, T_init=1000.0, lr_init=0.01,
                      beta1=0.9, beta2=0.999, eps=1e-8,
                      T_min=5.0, T_decay=0.9995, verbose=True, label="Adam"):
    """Adam optimizer on LSE-smoothed objective with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    T = T_init
    lr = lr_init
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        a = softmax_fn(z)
        grad_z = lse_gradient(a, T, n)
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        T = max(T_min, T * T_decay)
        if t % 100 == 0:
            lr = lr_init * (T / T_init) ** 0.5
        a_new = softmax_fn(z)
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            rng = np.random.default_rng(t)
            z = np.log(np.maximum(best_a, 1e-30)) + rng.standard_normal(n) * 0.01
            no_improve = 0
            m = np.zeros(n)
            v = np.zeros(n)
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={t}, T={T:.1f}, best={best_score:.8f}, time={elapsed:.0f}s")
    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=200, maxiter=5000):
    """L-BFGS polish on LSE-smoothed objective."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_fn(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_sh = shifted.max()
        lse = max_sh + np.log(np.exp(shifted - max_sh).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_fn(z)
        return lse_gradient(a, T, n)

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 30, 'disp': False})
        a_out = softmax_fn(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def peak_reduction(a0, time_limit, verbose=False):
    """Direct peak reduction: iteratively reduce the max convolution lag."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = np.argmax(conv)
        # Find which a[i] contribute most to conv[k_star]
        # conv[k_star] = sum_j a[j]*a[k_star-j]
        # Contribution of a[i]: a[i]*a[k_star-i] (if valid)
        contrib = np.zeros(n)
        for j in range(n):
            ki = k_star - j
            if 0 <= ki < n:
                contrib[j] = a[j] * a[ki]
        # Reduce top contributors, add to bottom contributors
        top_idx = np.argsort(contrib)[-3:]
        bot_idx = np.argsort(contrib)[:3]
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            i = top_idx[rng.integers(0, len(top_idx))]
            j = bot_idx[rng.integers(0, len(bot_idx))]
            if i == j or a[i] < 1e-6:
                continue
            t_val = a[i] * 0.01
            a_trial = a.copy()
            a_trial[i] -= t_val
            a_trial[j] += t_val
            sc = compute_score(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
        if verbose and iters % 50 == 0:
            print(f"  PR iter={iters}, best={best_score:.8f}")
    return best_a, best_score


def cosine_init(n, peak=2.0, noise=0.0):
    """Cosine-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** peak
    a = np.maximum(a, 0)
    if noise > 0:
        a += np.random.default_rng(42).standard_normal(n) * noise
        a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    return a_interp / s if s > 1e-10 else np.ones(new_n) / new_n


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Adam-LSE with temperature annealing (main driver)
    2. L-BFGS polish at multiple temperatures
    3. Peak reduction heuristic
    4. Random restarts
    5. Different sequence lengths
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = cosine_init(n_init, peak=2.0)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: Adam-LSE hot (exploration)
    print(f"\n=== Phase 1: Adam Hot T (best={best_score:.8f}, n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    a1, _ = adam_lse_optimize(best_seq, t1, T_init=5000.0, lr_init=0.02,
                               T_min=500.0, T_decay=0.999, label="Adam-Hot")
    best_seq, best_score = check_and_update(a1, "adam_hot", best_seq, best_score)

    # Phase 2: Adam-LSE warm (refinement)
    print(f"\n=== Phase 2: Adam Warm T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if rem > 60:
        a2, _ = adam_lse_optimize(best_seq, t2, T_init=500.0, lr_init=0.01,
                                   T_min=50.0, T_decay=0.999, label="Adam-Warm")
        best_seq, best_score = check_and_update(a2, "adam_warm", best_seq, best_score)

    # Phase 3: Adam-LSE cool (fine-tuning)
    print(f"\n=== Phase 3: Adam Cool T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if rem > 60:
        a3, _ = adam_lse_optimize(best_seq, t3, T_init=50.0, lr_init=0.005,
                                   T_min=5.0, T_decay=0.9995, label="Adam-Cool")
        best_seq, best_score = check_and_update(a3, "adam_cool", best_seq, best_score)

    # Phase 4: L-BFGS polish
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [50, 200, 1000]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t4 = min(rem * 0.12, 80.0)
        a4, _ = lbgfs_polish(best_seq, t4, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a4, f"lbfgs_T{T_val}", best_seq, best_score)

    # Phase 5: Peak reduction
    print(f"\n=== Phase 5: Peak Reduction (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t5 = min(rem * 0.08, 60.0)
        a5, _ = peak_reduction(best_seq, t5)
        best_seq, best_score = check_and_update(a5, "peak_reduce", best_seq, best_score)

    # Phase 6: Random restarts
    print(f"\n=== Phase 6: Restarts (best={best_score:.8f}) ===")
    for restart in range(4):
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t6 = min(rem * 0.12, 80.0)
        rng = np.random.default_rng(200 + restart * 13)
        init = cosine_init(n_parent, peak=1.0 + rng.random() * 3.0,
                           noise=0.05 + rng.random() * 0.1)
        a6, _ = adam_lse_optimize(init, t6, T_init=1000.0, lr_init=0.01,
                                   T_min=20.0, T_decay=0.999, label=f"R{restart}")
        best_seq, best_score = check_and_update(a6, f"restart_{restart}", best_seq, best_score)

    # Phase 7: Different lengths
    print(f"\n=== Phase 7: Other Lengths (best={best_score:.8f}) ===")
    for alt_n in [n_parent + 2, n_parent - 2, n_parent + 4, 512, 1024]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        if alt_n < 64 or alt_n > 4096:
            continue
        if alt_n == n_parent:
            continue
        t7 = min(rem * 0.15, 80.0)
        a_interp = interpolate_seq(best_seq, alt_n)
        a7, _ = adam_lse_optimize(a_interp, t7, T_init=200.0, lr_init=0.005,
                                   T_min=10.0, T_decay=0.9995, label=f"n{alt_n}")
        best_seq, best_score = check_and_update(a7, f"len_{alt_n}", best_seq, best_score)

    # Phase 8: Final Adam polish (very low T)
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.3, 200.0)
        a8, _ = adam_lse_optimize(best_seq, t8, T_init=10.0, lr_init=0.002,
                                   T_min=1.0, T_decay=0.9998, label="Final")
        best_seq, best_score = check_and_update(a8, "final_adam", best_seq, best_score)

    # Phase 9: Final L-BFGS
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.5, 50.0)
        a9, _ = lbgfs_polish(best_seq, t9, T=50, maxiter=2000)
        best_seq, best_score = check_and_update(a9, "final_lbfgs", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
