Let me also think about the **multiplicative weight approach** more carefully. The key insight is:

We want to minimize max_k (a*a)(k) subject to a >= 0, sum(a) = 1.

This is equivalent to: find a probability distribution a on {0,...,n-1} that minimizes the maximum pointwise value of its self-convolution.

The self-convolution (a*a)(k) = P(X+Y=k) where X,Y are i.i.d. with distribution a.

So we want a distribution on {0,...,n-1} such that the sum of two i.i.d. copies has a flat distribution (minimizes the maximum probability).

The minimum possible max probability for the sum is at least 1/(2n-1) (since the sum ranges over 0,...,2n-2, which is 2n-1 values). For a uniform distribution, (a*a)(k) = (n-|k-n+1|)/(n^2) for 0 <= k <= 2n-2, which peaks at (n-1+1)/n^2 = n/n^2 = 1/n at k=n-1.

So the uniform gives max = 1/n, and the theoretical lower bound is 1/(2n-1).

The score for uniform: 2*n*(1/n)/1^2 = 2.
The theoretical lower bound on the score: 2*n*(1/(2n-1)) = 2n/(2n-1) ≈ 1 + 1/(2n-1).

For n=1152: theoretical lower bound ≈ 1 + 1/2303 ≈ 1.0004.

But achieving the theoretical lower bound requires the sum X+Y to be perfectly uniform, which is impossible for a distribution on a finite set (it would require a "perfect difference set" or similar combinatorial structure that doesn't exist for most n).

In practice, the best sequences achieve scores around 1.5, which means max(conv) ≈ 1.5 / (2*n) * (sum)^2 = 1.5/(2*1152) ≈ 0.000651 when sum=1.

The theoretical minimum is 1/(2*1152-1) ≈ 0.000434.

So we're at about 1.5x the theoretical minimum. There's still room for improvement.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

def mirror_descent(f_init, time_limit, eta_init=0.5, top_peaks=5, 
                   decay=0.001, perturb_interval=200, label="md"):
    """Fast mirror descent for minimizing max auto-convolution."""
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    
    eta = eta_init
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        b = fast_convolve(a)
        M = b.max()
        
        # Select top peaks
        if top_peaks >= len(b):
            peak_indices = np.arange(len(b))
        else:
            # Use top peaks weighted by excess
            sorted_idx = np.argsort(b)[::-1][:top_peaks]
            peak_indices = sorted_idx
        
        # Compute subgradient (averaged over selected peaks)
        g = np.zeros(n)
        for k in peak_indices:
            # g[i] += 2 * a[k-i] for valid i
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # a[k - i] for i in [lo, hi]
            # When i = lo, k-i = k-lo; when i = hi, k-i = k-hi
            # a[k-lo], a[k-lo-1], ..., a[k-hi]
            # = a[k-lo : k-hi-1 : -1] reversed = a[k-hi : k-lo+1] reversed
            # Actually: for i in range(lo, hi+1), a[k-i]
            # k-i ranges from k-lo down to k-hi
            # So we need a[k-hi : k-lo+1] flipped
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        
        g /= len(peak_indices)
        
        # Exponentiated gradient update
        # a_i *= exp(-eta * g_i)
        # For numerical stability, subtract max
        log_a = np.log(a + 1e-300) - eta * g
        log_a -= log_a.max()
        a = np.exp(log_a)
        s = a.sum()
        a = a / s
        
        # Anneal eta
        eta = eta_init / (1.0 + decay * it)
        
        # Check improvement
        if it % 10 == 0:
            if M < best_obj - 1e-14:
                best_obj = M
                best_score = 2.0 * n * M
                best_a = a.copy()
            
            # Perturbation when stuck
            if it % perturb_interval == 0:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
        
        if it % 500 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"eta={eta:.6f} t={time.monotonic()-t0:.0f}s")
    
    # Final check
    final_obj = fast_convolve(best_a).max()
    best_score = 2.0 * n * final_obj
    return best_a, best_score

def build_constraint_matrix(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A

def solve_lp(f, indices, M):
    n = len(f)
    A = build_constraint_matrix(f, indices, n)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000)
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8)
    except:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8)
        except:
            return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)

def lp_refine(f, time_limit, top_k=100, label="lp"):
    """LP-based refinement, called periodically."""
    t0 = time.monotonic()
    n = len(f)
    a = f.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        top_idx = np.sort(top_idx)
        
        g = solve_lp(a, top_idx, M)
        if g is None:
            break
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            break
        
        h = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_trial = M
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = (1.0 - t) * a + t * h
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            obj_t = fast_convolve(a_trial).max()
            if obj_t < best_obj_trial:
                best_obj_trial = obj_t
                best_t = t
        
        if best_t == 0.0:
            break
        
        a = (1.0 - best_t) * a + best_t * h
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        if best_obj_trial < best_obj - 1e-14:
            best_obj = best_obj_trial
            best_a = a.copy()
            best_score = 2.0 * n * best_obj
    
    return best_a, best_score

def coordinate_descent(f_init, time_limit):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    no_pass_improve = 0
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0]:
                new_val = f[j] * mult
                f[j] = new_val
                s = f.sum()
                if s < 1e-12:
                    f[j] = best_j_val
                    continue
                f_n = f / s
                obj = fast_convolve(f_n).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
                f[j] = best_j_val  # reset
        
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break
        else:
            no_pass_improve = 0
    
    return best_f, best_score

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses fast exponentiated gradient (mirror descent) as the main workhorse with
FFT-based cross-correlation for O(n log n) gradient computation, combined with
periodic LP refinement for precision, multi-length exploration, and coordinate
descent polish. The key advantage over pure LP approaches is that mirror descent
iterations are ~1000x faster, enabling vastly more optimization steps.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def compute_subgradient_fft(a, peak_weights, n, fft_size):
    """
    Compute subgradient g[i] = 2 * sum_k w[k] * a[k-i] using FFT cross-correlation.
    peak_weights: array of length 2n-1 with weights at peak positions.
    """
    A = rfft(a, fft_size)
    W = rfft(peak_weights, fft_size)
    # Cross-correlation: ifft(FFT(W) * conj(FFT(A)))
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def compute_subgradient_direct(a, peak_indices, weights):
    """Compute subgradient by direct summation over selected peaks."""
    n = len(a)
    g = np.zeros(n)
    for wi, k in zip(weights, peak_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * wi * a[k-hi:k-lo+1][::-1]
    return g


def mirror_descent(f_init, time_limit, eta_init=1.0, eta_min=0.01,
                   top_peaks=8, use_fft_grad=True, perturb_interval=500,
                   perturb_scale=0.008, label="md", seed=42):
    """
    Fast exponentiated gradient (mirror descent) for minimizing max auto-convolution.
    Works in log-space for numerical stability.
    """
    t0 = time.monotonic()
    n = len(f_init)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    a = f_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n

    # Work in log-space
    log_a = np.log(a + 1e-300)
    log_a -= log_a.max()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj

    eta = eta_init
    it = 0
    rng = np.random.default_rng(seed)
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Recover a from log-space
        a = np.exp(log_a)
        a = a / a.sum()

        b = fast_convolve(a)
        M = b.max()

        # Select top peaks with soft weights
        if top_peaks >= size:
            peak_indices = np.arange(size)
            peak_weights = np.ones(size) / size
        else:
            # Use top-K peaks with exponential softmax weighting
            top_idx = np.argpartition(b, -top_peaks)[-top_peaks:]
            top_vals = b[top_idx]
            # Soft weight: favor higher peaks
            temp = max((top_vals.max() - top_vals.min()) * 0.1, 1e-10)
            w_raw = np.exp((top_vals - top_vals.max()) / temp)
            peak_weights = w_raw / w_raw.sum()

            if use_fft_grad:
                pw = np.zeros(size)
                pw[top_idx] = peak_weights
                g = compute_subgradient_fft(a, pw, n, fft_size)
            else:
                g = compute_subgradient_direct(a, top_idx, peak_weights)

        # Clip gradient for stability
        g_max = np.abs(g).max()
        if g_max > 10.0:
            g = g * (10.0 / g_max)

        # Exponentiated gradient update in log-space
        log_a = log_a - eta * g
        log_a -= log_a.max()

        # Anneal eta
        progress = min(1.0, it / max(1, int(time_limit * 50)))
        eta = eta_init * (eta_min / eta_init) ** progress

        # Periodic perturbation to escape local minima
        if it % perturb_interval == 0:
            noise = rng.standard_normal(n) * perturb_scale
            log_a = log_a + noise
            log_a -= log_a.max()

        # Periodic check and improvement tracking
        if it % 5 == 0:
            a_check = np.exp(log_a)
            a_check = a_check / a_check.sum()
            obj_check = fast_convolve(a_check).max()
            if obj_check < best_obj - 1e-15:
                best_obj = obj_check
                best_a = a_check.copy()
                best_score = 2.0 * n * best_obj
                no_improve_count = 0
            else:
                no_improve_count += 1

            # Adaptive perturbation scale
            if no_improve_count > 100:
                perturb_scale = min(perturb_scale * 1.05, 0.05)

            if it % 200 == 0:
                print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                      f"eta={eta:.6f} t={time.monotonic()-t0:.0f}s")

    # Final
    a_final = np.exp(log_a)
    a_final = a_final / a_final.sum()
    final_obj = fast_convolve(a_final).max()
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a_final.copy()
        best_score = 2.0 * n * best_obj

    return best_a, best_score


def build_constraint_matrix(f, indices, n):
    """Build K x n constraint matrix for given peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A


def solve_lp_direction(f, indices, M):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    A = build_constraint_matrix(f, indices, n)
    g_var = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(objective, constraints)
    try:
        if len(indices) <= 500 and n <= 1500:
            prob.solve(solver=cp.ECOS, max_iters=15000)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-9)
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8)
        except Exception:
            return None
    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0.0)


def lp_refine(f_init, time_limit, top_k=150, label="lp"):
    """LP-based refinement with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()

        top_idx = np.argpartition(b, -top_k)[-top_k:]
        top_idx = np.sort(top_idx)

        g = solve_lp_direction(a, top_idx, M)
        if g is None:
            stuck += 2
            if stuck > 10:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10:
                break
            continue

        h = g / S_g

        # Backtracking line search
        best_t = 0.0
        best_obj_trial = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_trial = (1.0 - t) * a + t * h
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            obj_t = fast_convolve(a_trial).max()
            if obj_t < best_obj_trial - 1e-14:
                best_obj_trial = obj_t
                best_t = t

        if best_t == 0.0:
            stuck += 1
            if stuck > 10:
                break
            # Try with fewer constraints
            if top_k > 50:
                top_k = max(50, top_k // 2)
            continue

        a = (1.0 - best_t) * a + best_t * h
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        if best_obj_trial < best_obj - 1e-14:
            best_obj = best_obj_trial
            best_a = a.copy()
            best_score = 2.0 * n * best_obj
            stuck = 0
        else:
            stuck += 1

        if it % 5 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]
    no_pass_improve = 0

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue

            best_j_val = f[j]
            best_j_obj = best_obj
            orig_val = f[j]

            for mult in multipliers:
                f[j] = orig_val * mult
                s = f.sum()
                if s < 1e-12:
                    f[j] = orig_val
                    continue
                f_n = f / s
                obj = fast_convolve(f_n).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = orig_val * mult
            f[j] = orig_val

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True

        if not any_improved:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break
        else:
            no_pass_improve = 0

    return best_f, best_score


def alternating_optimization(f_init, time_limit, label="alt"):
    """
    Alternating method: alternate between "raise valleys" and "lower peaks".
    At each step, either:
    - Identify the lowest point in the convolution and increase weights there
    - Identify the highest point and decrease weights there
    This creates a more uniform convolution profile.
    """
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    it = 0
    alpha = 0.05

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        m = b.min()

        # Strategy: reduce weight on elements contributing to peaks,
        # increase weight on elements contributing to valleys
        peak_idx = np.argmax(b)
        valley_idx = np.argmin(b)

        # Gradient from peak (want to reduce)
        peak_grad = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        peak_grad[lo:hi+1] = a[peak_idx-hi:peak_idx-lo+1][::-1]

        # Gradient from valley (want to increase)
        valley_grad = np.zeros(n)
        lo = max(0, valley_idx - n + 1)
        hi = min(n - 1, valley_idx)
        valley_grad[lo:hi+1] = a[valley_idx-hi:valley_idx-lo+1][::-1]

        # Combined update: reduce where it contributes to peaks, increase where valleys
        direction = valley_grad - peak_grad
        direction = direction / (np.abs(direction).sum() + 1e-15)

        a_new = a + alpha * direction
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_score = 2.0 * n * best_obj
            a = a_new
            alpha = min(0.1, alpha * 1.02)
        else:
            alpha *= 0.7

        if it % 100 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"alpha={alpha:.6f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_score
