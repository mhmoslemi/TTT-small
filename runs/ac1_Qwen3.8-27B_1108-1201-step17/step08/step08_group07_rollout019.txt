    n = len(a)
    size = 2 * n - 1

    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')

    e = np.exp(z - z_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_global = 0
    
    t0 = time.monotonic()
    best_score = compute_score(a0)
    best_a = a0.copy()
    
    n_phases = len(T_schedule)
    phase_budget = time_limit / n_phases
    
    for phase_idx, T in enumerate(T_schedule):
        # Reset Adam at each temperature change? Or keep?
        # Keeping helps with continuity, but the gradient scale changes with T.
        # Let's keep it but scale lr.
        phase_t0 = time.monotonic()
        iters = 0
        no_improve = 0
        
        while time.monotonic() - phase_t0 < phase_budget and time.monotonic() - t0 < time_limit - 5:
            t_global += 1
            iters += 1
            
            grad_z = compute_grad_z(z, T, n)
            
            # Gradient clipping
            gnorm = np.linalg.norm(grad_z)
            if gnorm > 5.0:
                grad_z = grad_z * 5.0 / gnorm
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_global)
            v_hat = v / (1 - beta2 ** t_global)
            
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = z - z.max()
            
            # Evaluate
            a = softmax(z)
            score = compute_score(a)
            if score < best_score - 1e-10:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Restart if stuck
            if no_improve > 500:
                rng = np.random.default_rng(t_global)
                z = z + rng.standard_normal(n) * 0.1
                z = z - z.max()
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  {label} T={T}, iters={iters}, best={best_score:.8f}, time={elapsed:.0f}s/{time_limit:.0f}s")
    
    return best_a, best_score

# --- LP Fictitious Play ---

def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)

def solve_best_response(f, M, timeout=10.0):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return f.copy()
    if g.value is None:
        return f.copy()
    return np.maximum(g.value, 0)

def lp_fictitious_play(a0, time_limit, step=0.3, verbose=True, label="LP-FP"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        
        if stall_count > 200:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, time={elapsed:.0f}s")
    
    return best_a, best_score

# --- Coordinate Descent ---

def coordinate_descent(a0, time_limit, n_pairs_per_iter=200, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        iters += 1
        improved = False
        for _ in range(n_pairs_per_iter):
            if time.monotonic() - t0 > time_limit - 5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            best_local = best_score
            best_t = 0.0
            base = a[i]
            
            for frac in [0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            
            if best_t > 0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        
        if not improved and iters > 10:
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score

# --- Helper functions ---

def check_and_update(a, label, best_seq, best_score):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
