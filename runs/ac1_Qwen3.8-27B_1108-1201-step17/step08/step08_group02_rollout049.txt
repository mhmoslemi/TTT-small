import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(f):
    """Compute the objective value for a normalized sequence."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * M / (s * s)


def compute_subgradient(f, peak_positions):
    """Compute 2 * sum_{k in peaks} f[k-j] for each j (subgradient of max conv)."""
    n = len(f)
    grad = np.zeros(n)
    for k in peak_positions:
        k = int(k)
        if k < n:
            grad[0:k+1] += f[k:-1:-1]
        else:
            start = k - n + 1
            grad[start:n] += f[n-1:start-1:-1]
    return 2.0 * grad


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_init_arcsine(n, rng):
    """Arcsine-like initialization: heavy at edges."""
    x = np.arange(n, dtype=np.float64)
    x = (x + 0.5) / n
    f = 1.0 / np.sqrt(np.maximum(x * (1.0 - x), 1e-10))
    f /= f.sum()
    return f


def make_init_beta(n, alpha, rng):
    """Beta distribution initialization with parameter alpha."""
    x = np.arange(n, dtype=np.float64)
    x = (x + 0.5) / n
    x = np.clip(x, 1e-6, 1.0 - 1e-6)
    f = x ** (alpha - 1) * (1.0 - x) ** (alpha - 1)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s
    return f


def make_init_powerlaw(n, alpha, rng):
    """Power law: f[i] ∝ (i+1)^(-alpha) * (n-i)^(-alpha)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    f = i ** (-alpha) * (n - i + 1) ** (-alpha)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:

def mirror_descent_optimize(f_init, time_limit, lr_init=0.05, n_top=80, label="md",
                            temp_init=0.001, temp_decay=1.005, temp_max=0.05,
                            restart_every=50, seed=42):
    """Mirror descent with softmax peak weighting and adaptive learning rate."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights on peaks
        scaled = b / (temp * M + 1e-30)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Focus on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w[top_idx] = w_sel
            else:
                w[top_idx] = 1.0 / len(top_idx)

        # Compute gradient via subgradient at top peaks
        if n_top < len(b):
            grad = compute_subgradient(f, top_idx)
        else:
            grad = compute_subgradient(f, np.where(w > 1e-6)[0])

        # Mirror descent update
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new /= s
        f = f_new

        # Adaptive temperature
        temp = min(temp * temp_decay, temp_max)

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.3)
            else:
                no_improve += 1
                if no_improve > restart_every:
                    # Restart with perturbation
                    f = best_f + rng.standard_normal(n) * 0.002 * best_f
                    f = np.maximum(f, 1e-15)
                    f /= f.sum()
                    lr = lr_init
                    temp = temp_init
                    no_improve = 0
                elif no_improve > 5:
                    lr *= 0.95
                    no_improve = 0

            if it % 200 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                      f"lr={lr:.4f} temp={temp:.5f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def subgradient_with_momentum(f_init, time_limit, lr_init=0.003, n_top=60,
                              momentum=0.9, label="sg", seed=42):
    """Projected subgradient descent with Nesterov momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    y = f.copy()  # momentum variable
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Evaluate at y (Nesterov lookahead)
        b_y = fast_convolve(y)
        M = b_y.max()

        # Top peaks
        if n_top < len(b_y):
            top_idx = np.argpartition(b_y, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b_y))

        grad = compute_subgradient(y, top_idx)

        # Subgradient step + momentum
        v = momentum * v - lr * grad
        f_new = project_simplex(y + v)

        # Update y for next iteration
        y = momentum * f_new + (1.0 - momentum) * f
        # Actually standard Nesterov: y_{k+1} = (v_{k+1} + x_{k+1}) / (1+m) where v is the momentum
        # Simpler: y = f + momentum * v (lookahead)
        y = f_new + momentum * v

        f = f_new

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.05)
            else:
                no_improve += 1
                if no_improve > 30:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    y = f.copy()
                    lr = max(lr_init, lr * 0.9)
                    no_improve = 0
                elif no_improve > 5:
                    lr *= 0.97
                    no_improve = 0

            if it % 200 == 0:
                score = 2.0 * n * best_b

                      f"lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

                best_f = f.copy()
                no_improve = 0
                push_frac_cur = min(push_frac_cur * 1.02, 0.1)
            else:
                no_improve += 1
                if no_improve > 20:
                    push_frac_cur *= 0.9
                    if no_improve > 50:
                        f = best_f + rng.standard_normal(n) * 0.002 * best_f
                        f = np.maximum(f, 1e-15)
                        f /= f.sum()
                        no_improve = 0
                        push_frac_cur = push_frac

            if it % 200 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                      f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def weighted_subgradient(f_init, time_limit, lr_init=0.02, n_top=50,
                         label="ws", temp_init=0.003, seed=42):
    """
    Weighted subgradient: use softmax-weighted mix of peak contributions.
    Fast and effective polishing method.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights
        scaled = b / (temp * M + 1e-30)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Top peaks for gradient
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        # Weighted subgradient
        grad = np.zeros(n)
        for k in top_idx:
            wk = w[k]
            if wk < 1e-10:
                continue
            k_int = int(k)
            if k_int < n:
                grad[0:k_int+1] += wk * f[k_int:-1:-1]
            else:
                start = k_int - n + 1
                grad[start:n] += wk * f[n-1:start-1:-1]
        grad *= 2.0

        # Update: multiplicative (mirror descent style)
        f_new = f * np.exp(-lr * grad)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new /= s
        f = f_new

        # Decay temperature
        temp = min(temp * 1.008, 0.1)

        if it % 10 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.2)
            else:
                no_improve += 1
                if no_improve > 40:
                    f = best_f + rng.standard_normal(n) * 0.003 * best_f
                    f = np.maximum(f, 1e-15)
                    f /= f.sum()
                    lr = lr_init * 0.5
                    temp = temp_init
                    no_improve = 0
                elif no_improve > 5:
                    lr *= 0.96
                    no_improve = 0

            if it % 200 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                      f"lr={lr:.4f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd", n_multipliers=7):
    """Coordinate descent: try scaling each element by various factors."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break
        if it % 5 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} sweep={it} obj={best_b:.12f} score={score:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def smooth_filter(f, sigma=2.0):
    """Apply Gaussian smoothing."""
    n = len(f)
    x = np.arange(n)
    mid = n // 2
    kernel = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
    # Circular convolution for smoothing
    from scipy.signal import fftconvolve
    f_smooth = fftconvolve(f, kernel / kernel.sum(), mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    s = f_smooth.sum()
    if s > 1e-10:
        f_smooth /= s
    return f_smooth


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization combining diverse initializations with fast
    subgradient/mirror descent, peak-pushing, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    best_seq = None
    best_score = np.inf

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    n_parent = len(parent) if parent is not None else 1024

    # Define initializations to try
    inits = []

    if parent is not None:
        inits.append(("parent", parent.copy()))

    # Diverse smooth initializations at parent length
    n_try = n_parent
    inits.append(("arcsine", make_init_arcsine(n_try, rng)))
    for alpha in [0.3, 0.5, 0.7, 1.5, 2.0, 3.0]:
        inits.append((f"beta_{alpha}", make_init_beta(n_try, alpha, rng)))
    for alpha in [0.1, 0.3, 0.5, 1.0]:
        inits.append((f"power_{alpha}", make_init_powerlaw(n_try, alpha, rng)))
    inits.append(("rand_exp", make_init_random(n_try, 'exponential', rng)))
    inits.append(("rand_lap", make_init_random(n_try, 'laplacian', rng)))

    # Also try different lengths
    for factor in [0.75, 0.85, 1.1, 1.25, 1.5]:
        n_alt = int(n_try * factor)
        n_alt = max(256, min(3000, n_alt))
        if n_alt != n_try:
            inits.append((f"arcsine_n{n_alt}", make_init_arcsine(n_alt, rng)))
            inits.append((f"beta05_n{n_alt}", make_init_beta(n_alt, 0.5, rng)))

    print(f"Trying {len(inits)} initializations, budget={budget_s}s")

    # Phase 1: Quick evaluation and ranking of initializations
    print("\n=== Phase 1: Quick initialization ranking ===")
    init_scores = []
    for name, f_init in inits:
        f_n = f_init.copy()
        s = f_n.sum()
        if s > 1e-10:
            f_n /= s
        b = fast_convolve(f_n)
        M = b.max()
        sc = 2.0 * len(f_n) * M
        init_scores.append((sc, name, f_n))
        print(f"  {name}: score={sc:.6f} (n={len(f_n)})")

    init_scores.sort(key=lambda x: x[0])

    # Phase 2: Optimize top initializations
    print("\n=== Phase 2: Optimization of top initializations ===")
    n_budget = min(len(init_scores), 12)  # optimize top 12

    for rank in range(n_budget):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        sc, name, f_init = init_scores[rank]
        n_cur = len(f_init)

        # Allocate time based on rank
        if rank < 3:
            t_alloc = min(rem * 0.15, 150.0)
        elif rank < 6:
            t_alloc = min(rem * 0.08, 80.0)
        else:
            t_alloc = min(rem * 0.04, 40.0)
        if t_alloc < 10:
            break

        print(f"\n--- Optimizing '{name}' (n={n_cur}, initial={sc:.6f}, budget={t_alloc:.0f}s) ---")

        f = f_init.copy()
        cur_best = f.copy()
        cur_best_b = fast_convolve(f).max()

        # Step A: Weighted subgradient (fast, many iterations)
        tA = t_alloc * 0.35
        if tA > 5:
            fA, bA = weighted_subgradient(f, time_limit=tA, lr_init=0.02,
                                           n_top=50, label=f"ws_{name}")
            if bA < cur_best_b:
                cur_best_b = bA
                cur_best = fA.copy()
            f = fA.copy() if bA < cur_best_b else cur_best.copy()

        # Step B: Mirror descent
        rem2 = deadline - time.monotonic()
        tB = min(rem2 * 0.15, t_alloc * 0.30)
        if tB > 5:
            fB, bB = mirror_descent_optimize(f, time_limit=tB, lr_init=0.03,
                                              n_top=60, label=f"md_{name}")
            if bB < cur_best_b:
                cur_best_b = bB
                cur_best = fB.copy()
            f = fB.copy() if bB < cur_best_b else cur_best.copy()

        # Step C: Peak pushing
        rem3 = deadline - time.monotonic()
        tC = min(rem3 * 0.10, t_alloc * 0.20)
        if tC > 5:
            fC, bC = peak_push_optimize(f, time_limit=tC, n_top=30,
                                         push_frac=0.01, label=f"pp_{name}")
            if bC < cur_best_b:
                cur_best_b = bC
                cur_best = fC.copy()

        # Step D: Subgradient with momentum
        rem4 = deadline - time.monotonic()
        tD = min(rem4 * 0.10, t_alloc * 0.15)
        if tD > 5:
            fD, bD = subgradient_with_momentum(cur_best, time_limit=tD,
                                                lr_init=0.002, n_top=50,
                                                label=f"sg_{name}")
            if bD < cur_best_b:
                cur_best_b = bD
                cur_best = fD.copy()

        # Step E: Coordinate descent polish
        rem5 = deadline - time.monotonic()
        tE = min(rem5 * 0.15, t_alloc * 0.20)
        if tE > 10:
            fE, bE = coordinate_descent(cur_best, time_limit=tE, label=f"cd_{name}")
            if bE < cur_best_b:
                cur_best_b = bE
                cur_best = fE.copy()

        # Evaluate
        vs = float(evaluate_sequence(cur_best.tolist()))
        score_cur = 2.0 * n_cur * cur_best_b
        print(f"  Result: verified={vs:.6f} (fft={score_cur:.6f}), n={n_cur}")

        if np.isfinite(vs) and vs < best_score:
            best_score = vs
            best_seq = cur_best.copy()
            print(f"  *** NEW BEST: {vs:.6f} ***")

    # Phase 3: Deep optimization on the best found
    print("\n=== Phase 3: Deep optimization on best ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        n_best = len(best_seq)
        print(f"  Best so far: score={best_score:.6f}, n={n_best}")

        # Extended weighted subgradient
        t_deep1 = min(rem * 0.30, 300.0)
        if t_deep1 > 30:
            print(f"  Deep WS (budget={t_deep1:.0f}s)")
            f_deep1, b_deep1 = weighted_subgradient(best_seq, time_limit=t_deep1,
                                                     lr_init=0.01, n_top=80,
                                                     label="ws_deep")
            vs = float(evaluate_sequence(f_deep1.tolist()))
            if np.isfinite(vs) and vs < best_score:
                best_score = vs
                best_seq = f_deep1.copy()
                print(f"  *** NEW BEST [ws_deep]: {vs:.6f} ***")

        # Extended mirror descent
        rem = deadline - time.monotonic()
        t_deep2 = min(rem * 0.25, 200.0)
        if t_deep2 > 20:
            print(f"  Deep MD (budget={t_deep2:.0f}s)")
            f_deep2, b_deep2 = mirror_descent_optimize(best_seq, time_limit=t_deep2,
                                                        lr_init=0.02, n_top=100,
                                                        label="md_deep")
            vs = float(evaluate_sequence(f_deep2.tolist()))
            if np.isfinite(vs) and vs < best_score:
                best_score = vs
                best_seq = f_deep2.copy()
                print(f"  *** NEW BEST [md_deep]: {vs:.6f} ***")

    # Phase 4: Multi-length exploration from best
    print("\n=== Phase 4: Multi-length exploration ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        n_base = len(best_seq)
        target_lengths = []
        for factor in [0.8, 0.9, 1.1, 1.2, 1.4]:
            tl = int(n_base * factor)
            tl = max(256, min(3000, tl))
            if tl != n_base and tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_base))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_ml = min(rem * 0.15, 100.0)
            if t_ml < 15:
                break
            f_interp = interpolate_to_length(best_seq, target_n)
            print(f"  Interpolating to n={target_n}, budget={t_ml:.0f}s")
            f_ml, b_ml = weighted_subgradient(f_interp, time_limit=t_ml * 0.5,
                                               lr_init=0.02, n_top=50, label=f"ml_{target_n}")
            f_ml2, b_ml2 = mirror_descent_optimize(f_ml, time_limit=t_ml * 0.5,
                                                    lr_init=0.02, n_top=60, label=f"mld_{target_n}")
            f_best_ml = f_ml.copy() if b_ml < b_ml2 else f_ml2.copy()
            b_best_ml = min(b_ml, b_ml2)
            vs = float(evaluate_sequence(f_best_ml.tolist()))
            if np.isfinite(vs) and vs < best_score:
                best_score = vs
                best_seq = f_best_ml.copy()
                print(f"  *** NEW BEST [n={target_n}]: {vs:.6f} ***")

    # Phase 5: Final coordinate descent polish
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        t_cd = min(rem * 0.4, 120.0)
        if t_cd > 10:
            print(f"\n=== Phase 5: Final CD polish (budget={t_cd:.0f}s) ===")
            f_cd, b_cd = coordinate_descent(best_seq, time_limit=t_cd, label="cd_final")
            vs = float(evaluate_sequence(f_cd.tolist()))
            if np.isfinite(vs) and vs < best_score:
                best_score = vs
                best_seq = f_cd.copy()
                print(f"  *** NEW BEST [cd_final]: {vs:.6f} ***")

    # Phase 6: One more weighted subgradient burst with remaining time
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        t_final = rem * 0.7
        print(f"\n=== Phase 6: Final WS burst (budget={t_final:.0f}s) ===")
        f_fin, b_fin = weighted_subgradient(best_seq, time_limit=t_final,
                                             lr_init=0.008, n_top=100,
                                             label="ws_final")
        vs = float(evaluate_sequence(f_fin.tolist()))
        if np.isfinite(vs) and vs < best_score:
            best_score = vs
            best_seq = f_fin.copy()
            print(f"  *** NEW BEST [ws_final]: {vs:.6f} ***")

    # Final verification
    if best_seq is None:
        # Fallback
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Should not happen, but fallback
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
