import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, n_peaks=100, tau=0.05):
    """Compute gradient of 2*n*logsumexp(conv(a,a)/tau)/tau using top-peak approximation."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    
    if n_peaks < len(b):
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    else:
        top_idx = np.arange(len(b))
    
    # Softmax weights on selected peaks
    w_sel = np.exp((b[top_idx] - b_max) / tau)
    w_sum = w_sel.sum()
    if w_sum < 1e-300:
        w_sel = np.zeros(len(top_idx))
        w_sel[np.argmax(b[top_idx])] = 1.0
    else:
        w_sel = w_sel / w_sum
    
    # Compute gradient: grad_m = 2*n * sum_k w_k * 2 * a_{k-m} / tau
    # Actually the gradient of logsumexp(b/tau) is (1/tau) * softmax(b/tau) @ (db/da)
    # and db_k/da has component m = 2*a_{k-m}
    # So grad = 2*n * (1/tau) * sum_k softmax_k * 2 * a_{k-m}
    #         = 4*n/tau * sum_k softmax_k * a_{k-m}
    
    # Wait, let me be more careful.
    # F(a) = 2*n * logsumexp(b/tau)  where logsumexp(b/tau) = tau*log(sum(exp(b/tau)))
    # Actually let's use: F(a) = 2*n * (1/tau) * log(sum(exp(b_k/tau)))
    # dF/da_m = 2*n/tau * sum_k (exp(b_k/tau)/sum(exp(b_j/tau))) * db_k/da_m
    #         = 2*n/tau * sum_k softmax_k * db_k/da_m
    # db_k/da_m = 2*a_{k-m} (for valid indices)
    
    scale = 2.0 * n / tau
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        w = w_sel[i]
        if w < 1e-14:

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = a[lo:hi + 1]
        grad[lo:hi + 1] += w * idx[::-1]
    
    grad *= 2.0 * scale  # 2 (from db/da) * scale
    
    return grad, b, b_max

def lp_direction(f, peak_indices, M, solver='HiGHS'):
    """
    Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns g or None on failure.
    """
    n = len(f)
    K = len(peak_indices)
    
    # Build constraint matrix A (K x n)
    # A[i, j] = f[j] if j is in the valid range for peak peak_indices[i]
    rows = np.zeros((K, n))
    for i, k in enumerate(peak_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        rows[i, lo:hi + 1] = f[lo:hi + 1]
    
    # scipy linprog: minimize c @ x s.t. A_ub @ x <= b_ub, bounds
    c = -np.ones(n)
    A_ub = rows
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            if np.isfinite(g).all() and g.sum() > 1.0:
                return g
    except Exception:
        pass
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            if np.isfinite(g).all() and g.sum() > 1.0:
                return g
    except Exception:
        pass
    
    return None

    """LP-based optimization with scipy solver."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    it = 0
    stuck = 0
    top_k = top_k_init
    momentum_dir = np.zeros(n)
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)
        
        b = fast_convolve(f)
        M = b.max()
        
        # Select peaks
        if top_k < 2 * n - 1:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
            # Also add boundary peaks
            n_bound = min(n, 5)
            boundary = list(range(0, n_bound)) + list(range(2*n-2-n_bound+1, 2*n-1))
            all_cands = list(set(list(top_idx) + boundary))
            all_cands.sort(key=lambda i: b[i], reverse=True)
            indices = np.array(all_cands[:top_k])
        else:
            indices = np.arange(2 * n - 1)
        
        # Solve LP
        g = lp_direction(f, indices, M)
        
        if g is None:
            stuck += 2
            if stuck > 50:
                break
            top_k = max(10, top_k // 2)
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 40:
                break
            top_k = min(top_k * 2, max(top_k_max, 2*n-1))
            continue
        
        h = g / S_g
        
        # Momentum
        momentum_dir = 0.3 * momentum_dir + 0.7 * h
        mom_sum = momentum_dir.sum()
        if mom_sum > 1e-10:
            h_mom = momentum_dir / mom_sum
        else:
            h_mom = h
        
        # Line search
        best_t_obj = M
        best_f_trial = f.copy()
        best_t = 0.0
        
        for direction in [h, h_mom]:
            for t in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                f_trial = (1.0 - t) * f + t * direction
                f_trial = np.maximum(f_trial, 0.0)
                s2 = f_trial.sum()
                if s2 < 1e-15:
                    continue
                f_trial = f_trial / s2
                obj = fast_convolve(f_trial).max()
                if obj < best_t_obj:
                    best_t_obj = obj
                    best_t = t
                    best_f_trial = f_trial.copy()
        
        if best_t == 0.0:
            stuck += 1
            if stuck > 50:
                break
            continue
        
        f = best_f_trial
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break
        
        if stuck > 25 and it % 20 == 0:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            stuck = 0
            momentum_dir = np.zeros(n)
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"k={len(indices)} t={elapsed:.0f}s")
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit):
    """Fast coordinate descent with multi-step updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.1, 1.2, 1.4, 1.6]
    no_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(0, n, 1):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass += 1
            if no_pass >= 3:
                break
        else:
            no_pass = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def make_u_shaped(n, depth=0.5):
    """Create a U-shaped (bowl) non-negative sequence."""
    x = np.linspace(0, 1, n)
    a = 1.0 + depth * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_gaussian(n, sigma=0.3):
    """Create a Gaussian-shaped sequence."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-x**2 / (2 * sigma**2))
    return a / a.sum()


def make_random(n, seed=42):
    """Create a random non-negative sequence."""
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n))
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass
    
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512, score={p_score:.6f}")
    
    best_seq = parent.copy()
    best_score = p_score
    
    def try_update(f_res, tag):
        nonlocal best_seq, best_score
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [{tag}]: score={real_score:.6f} n={len(f_res)}")
            return True
        return False
    
    # ============================================================
    # Phase 1: FISTA on parent (quick and effective)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: FISTA on parent (target: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    
    if t1 > 30:
        n_p = len(parent)
        print(f"  FISTA n={n_p}, budget={t1:.0f}s")
        f_res, _ = fista_optimize(parent, time_limit=t1,
                                  tau_init=0.08, tau_min=0.003,
                                  n_peaks_init=50, n_peaks_max=250,
                                  label="fista_p")

    # Phase 2: LP refinement on current best
    # ============================================================

    # ============================================================
    # Phase 3: FISTA on current best (deeper)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Deep FISTA (best: {best_score:.6f})")
    print(f"{'='*60}")

                                  tau_init=0.03, tau_min=0.002,
