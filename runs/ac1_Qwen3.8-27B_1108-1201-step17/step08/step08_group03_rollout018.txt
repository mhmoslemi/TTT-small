"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a minimax LP to find optimal blending directions, combined with
frequency-domain diversification, multiplicative updates, and multi-length exploration.
Key innovation: Direct minimax LP (no cutting planes needed) finds the direction
h that minimizes max_k (f*h)[k] subject to h>=0, sum(h)=1.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(f, n):
    """Build matrix A where (A @ h)[k] = (f*h)[k] for linear convolution of length-n sequences."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    for l in range(n):
        A[l:l + n, l] = f
    return A


def solve_minimax_lp(f, n, use_solver=None):
    """Solve: min t s.t. (f*h)[k] <= t for all k, h>=0, sum(h)=1.
    Returns (h, t_val) or (None, None) on failure.
    """
    h = cp.Variable(n)
    t = cp.Variable()

    A = build_conv_matrix(f, n)

    constraints = [
        A @ h <= t,
        h >= 0,
        cp.sum(h) == 1
    ]

    prob = cp.Problem(cp.Minimize(t), constraints)

    if use_solver is None:
        use_solver = cp.SCS

    try:
        prob.solve(solver=use_solver, max_iters=200000, eps=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            pass

    if h.value is None:
        return None, None

    h_val = np.maximum(h.value, 0.0)
    if np.any(np.isnan(h_val)):
        return None, None
    s = h_val.sum()
    if s < 1e-10:
        return None, None
    h_val /= s
    return h_val, t.value


def line_search_alpha(f, h, cf, ch, cfh):
    """Find alpha in [0,1] minimizing max_k [(1-a)^2*cf[k] + 2a(1-a)*cfh[k] + a^2*ch[k]]."""
    def obj_at(a):
        vals = (1 - a) ** 2 * cf + 2 * a * (1 - a) * cfh + a ** 2 * ch
        return vals.max()

    best_a = 0.0
    best_obj = obj_at(0.0)

    # Coarse grid
    for i in range(1, 21):
        a = i / 20.0
        o = obj_at(a)
        if o < best_obj:
            best_obj = o
            best_a = a

    # Golden section refinement
    if best_a > 0.02:
        lo = max(0.0, best_a - 0.08)
        hi = min(1.0, best_a + 0.08)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(15):
            if hi - lo < 1e-6:
                break
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)
            o1 = obj_at(a1)
            o2 = obj_at(a2)
            if o1 < o2:
                hi = a2
            else:
                lo = a1
        a_mid = (lo + hi) / 2.0
        o_mid = obj_at(a_mid)
        if o_mid < best_obj:
            best_obj = o_mid
            best_a = a_mid

    return best_a, best_obj


def iaа_diversify(f, n_iters=30, peak_fraction=0.3):
    """IAA-based diversification in frequency domain."""
    n = len(f)
    f_work = f.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        F = rfft(f_work, fft_size)
        R = irfft(F * F, fft_size)[:size]
        peak_val = R.max()
        floor = peak_fraction * peak_val
        R_target = np.clip(R, floor, peak_val)
        energy = R.sum()
        if R_target.sum() > 1e-10:
            R_target *= energy / R_target.sum()
        R_t_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_t_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f_work = irfft(F_new, fft_size)[:n]
        f_work = np.maximum(f_work, 0.0)
        s = f_work.sum()
        if s > 1e-10:
            f_work /= s
        else:
            return f.copy()

    return f_work


def multiplicative_update(f, n_steps=10, eta=0.1, n_top=200):
    """Multiplicative update based on convolution pressure."""
    n = len(f)
    f_work = f.copy()

    for step in range(n_steps):
        b = fast_convolve(f_work)
        threshold = b.mean() + 0.5 * b.std()
        w = np.maximum(b - threshold, 0.0)
        if w.sum() < 1e-30:
            w = np.maximum(b - b.mean(), 0.0)
        if w.sum() < 1e-30:
            break
        w /= w.sum()

        # Pressure: p[i] = sum_k w[k] * f[k-i] (contribution of f[i] to weighted peak)
        p = np.zeros(n)
        # p[i] = sum_k w[k] * A[k,i] where A is the conv matrix
        # A[k,i] = f[k-i] if 0<=k-i<n
        # So p[i] = sum_k w[k] * f[k-i] for valid k-i
        # This is cross_convolve of w_reversed with f... let me just do it directly
        # Actually p[i] = (w * f)_reversed... let me use FFT
        # p = sum_k w[k] * f[k-i] = cross_convolve_rev(w, f)[i]
        # cross_convolve(w, f)[k] = sum_j w[j] * f[k-j]
        # We want sum_k w[k] * f[k-i], which is cross_convolve(w, f)[2*(n-1) - i]... 
        # Actually let's just compute it via the matrix
        # p[i] = sum_k w[k] * A[k,i] = A^T @ w
        # But A is (2n-1) x n, so A^T @ w is n-dimensional
        # A[k,i] = f[k-i] for max(0,k-n+1) <= i <= min(n-1,k)
        # p[i] = sum_k w[k] * f[k-i] for valid k
        # Let's compute via FFT: cross_convolve(f, w) gives c[k] = sum_j f[j]*w[k-j]
        # We want sum_k w[k]*f[k-i]. Let j=k-i, so k=j+i:
        # p[i] = sum_j w[j+i] * f[j]
        # This is (w shifted by i) dot f... it's like a cross-correlation.
        # cross_convolve(w, f)[i] = sum_j w[j] * f[i-j]... not quite.
        # Let me just do the direct computation for the top few indices
        top_k = np.argsort(w)[-n_top:]
        for k in top_k:
            if w[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            p[idx] += w[k] * f[k - idx]

        f_work = f_work * np.exp(-eta * p / (p.max() + 1e-30))
        f_work = np.maximum(f_work, 0.0)
        s = f_work.sum()
        if s > 1e-10:
            f_work /= s
        else:
            return f.copy()

    return f_work

                        else:
                            # Random perturbation
                            f_pert = f + np.random.standard_normal(n) * 0.01
                            f_pert = np.maximum(f_pert, 0)
                            f_pert /= f_pert.sum()
                            f = f_pert
                            cf = fast_convolve(f)
                            no_improve = 0
                else:
                    break
            continue

        # Update
        f_new = (1 - alpha) * f + alpha * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = f.copy()
            continue

        f = f_new
        cf = fast_convolve(f)
        cur_obj = cf.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"alpha={alpha:.4f} no_imp={no_improve} t={elapsed:.0f}s")

    return best_f, best_obj


def symmetric_minimax_lp(f, n):
    """Solve minimax LP with symmetry constraint h[i]=h[n-1-i]."""
    h = cp.Variable(n)
    t = cp.Variable()
    A = build_conv_matrix(f, n)

    constraints = [
        A @ h <= t,
        h >= 0,
        cp.sum(h) == 1
    ]

    # Symmetry constraints
    half = n // 2
    for i in range(half):
        constraints.append(h[i] == h[n - 1 - i])

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            pass

    if h.value is None:
        return None, None

    h_val = np.maximum(h.value, 0.0)
    if np.any(np.isnan(h_val)):
        return None, None
    s = h_val.sum()
    if s < 1e-10:
        return None, None
    h_val /= s
    return h_val, t.value


def minimize_symmetric_lp(f_init, time_limit, label="sym"):
    """Optimization using symmetric minimax LP directions."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    cf = fast_convolve(f)
    best_obj = cf.max()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        h, t_val = symmetric_minimax_lp(f, n)
        if h is None:
            # Fallback to unsymmetric
            h, t_val = solve_minimax_lp(f, n)
            if h is None:
                break

        cfh = cross_convolve(f, h)
        ch = fast_convolve(h)
        alpha, new_obj = line_search_alpha(f, h, cf, ch, cfh)

        if alpha < 1e-4:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        f_new = (1 - alpha) * f + alpha * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        f = f_new
        cf = fast_convolve(f)
        cur_obj = cf.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 50 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj
