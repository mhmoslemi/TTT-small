"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating optimization
with CLARABEL solver, dual-gap monitoring, Adam-LogSumExp refinement,
spectral equalization, and multi-length exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_fast(n):
    """Build the convolution matrix for a given length using structured approach."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = 1.0
    return T


_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = name
                print(f"  Using solver: {name}")
                return name
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Fallback solver: ECOS")
    return _solver


def solve_lp_dual(n, a_fixed, solver=None):
    """Solve the LP: min M s.t. T_a @ b <= M, sum(b)=1, b>=0.
    Returns (b, M) or (None, None) on failure."""
    if solver is None:
        solver = get_solver()
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_lp_dual_gap(n, a_fixed, solver=None):
    """Solve both primal and dual LPs to compute the gap.
    Returns (b, M, dual_value, gap)."""
    b, M = solve_lp_dual(n, a_fixed, solver)
    if b is None:
        return None, None, None, None
    size = 2 * n - 1
    # Dual: max sum(y) subject to T^T y >= lambda*1, y >= 0, sum(y) = 1
    # Actually dual is: max lambda s.t. T^T y >= lambda*1, sum(y)=1, y>=0
    # Which is: max min_j (T^T y)_j s.t. sum(y)=1, y>=0
    y_var = cp.Variable(size, nonneg=True)
    lam_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T.T @ y_var >= lam_var, cp.sum(y_var) == 1]
    prob = cp.Problem(cp.Maximize(lam_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status in ('optimal', 'optimal_inaccurate'):
        dual_val = float(lam_var.value)
        return b, M, dual_val, M - dual_val
    return b, M, None, None


def nesterov_alternating(a0, time_limit, solver=None, n_max_iters=200000):
    """Nesterov dual averaging for the LP-based alternating optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_M
    stall = 0

    for t in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_dual(n, y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (t + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Nesterov extrapolation
        a_new = y  # Simplified: just use y
        # More careful Nesterov:
        # a = ((t+1)*a + alpha*(t+2)*b_new) / (t+1 + alpha*(t+2))
        a_candidate = ((t + 1) * a + alpha * (t + 2) * b_new) / (t + 1 + alpha * (t + 2))
        a_candidate = np.maximum(a_candidate, 0)
        sa = a_candidate.sum()
        if sa > 1e-15:
            a = a_candidate / sa
        else:
            a = y.copy()

        M_a = fast_convolve(a).max()
        M_y = fast_convolve(y).max()

        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()
            stall = 0
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            stall = 0

        if (t + 1) % 500 == 0:
            if best_M < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_M
            if stall >= 10:
                # Perturb to escape
                rng = np.random.default_rng(t)
                y = y + 0.005 * rng.random(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

        if (t + 1) % 2000 == 0:
            print(f"    NesterovLP t={t+1}: M={best_M:.8f} (score={2*best_M:.6f})")

    return best_a, best_M


def adam_logsumexp(a0, time_limit, T_start=100.0, T_end=0.1, n_steps=50000,
                   lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LogSumExp approximation of the max-convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s

    # Work in log-space for the simplex constraint
    z = np.log(np.maximum(a, 1e-300))
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Interpolate temperature
        progress = step / max(1, n_steps)
        T = T_start * (T_end / T_start) ** progress

        # Softmax
        z_shift = z - z.max()
        exp_z = np.exp(z_shift)
        a = exp_z / exp_z.sum()

        # Compute convolution
        c = fast_convolve(a)
        Tc = T * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()

        # Gradient with respect to a
        grad_a = np.zeros(n)
        # Only iterate over positions with significant weight
        thresh = w.max() * 1e-6
        active = np.where(w > thresh)[0]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]

        # Convert to gradient in z-space (simplex gradient)
        S = np.dot(grad_a, a)
        grad_z = a * (2.0 * n * w.sum() * grad_a - S)  # Hmm this isn't right

        # Actually, let me recompute. The objective is f(z) = (2n/T) * log(sum(exp(T*c(a(z))))).
        # df/dz_j = (2n/T) * sum_k w_k * dc[k]/da_i * da_i/dz_j
        # da_i/dz_j = a_i * (delta_ij - a_j)
        # dc[k]/da_i = sum_j a[k-j] * delta (i in the valid range for k)
        # So dc[k]/da_i = a[k-i] if i is in [max(0,k-n+1), min(n-1,k)], else 0
        #
        # df/dz_j = (2n/T) * sum_k w_k * [sum_i dc[k]/da_i * a_i*(delta_ij - a_j)]
        # = (2n/T) * sum_k w_k * [dc[k]/da_j * a_j - a_j * sum_i dc[k]/da_i * a_i]
        # = (2n/T) * a_j * [sum_k w_k * dc[k]/da_j - sum_k w_k * c[k]]
        # = (2n/T) * a_j * [sum_k w_k * c_contrib(k,j) - c_max_weighted]
        #
        # where c_contrib(k,j) = a[k-j] if valid, else 0
        # and c_max_weighted = sum_k w_k * c[k]
        #
        # Actually this is getting complicated. Let me just use the chain rule more carefully.
        #
        # Let g(a) = (2n/T) * LSE(T*c(a)) where LSE(x) = log(sum(exp(x)))
        # dg/da = (2n/T) * T * w * (dc/da)^T = 2n * w @ (dc/da)
        # where (dc/da)[k,i] = a[k-i] if i in valid range for k, else 0
        #
        # So dg/da_i = 2n * sum_k w_k * a[k-i] * 1{i valid for k}
        #
        # Then dg/dz_j = sum_i dg/da_i * da_i/dz_j = sum_i dg/da_i * a_i * (delta_ij - a_j)
        # = a_j * (dg/da_j - sum_i dg/da_i * a_i)
        # = a_j * (dg/da_j - dot(dg/da, a))

        # Recompute gradient properly
        dg_da_j = 2.0 * n * np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            dg_da_j[idx] += w[k] * a[k - idx]

        dot_dg_da_a = np.dot(dg_da_j, a)
        grad_z = a * (dg_da_j - dot_dg_da_a)

        # Adam update
        step_num = step + 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** step_num)
        v_hat = v / (1 - beta2 ** step_num)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if step % 1000 == 0:
            a_eval = exp_z / exp_z.sum()
            score = 2.0 * n * fast_convolve(a_eval).max()
            if score < best_score:
                best_score = score
                best_a = a_eval.copy()

    # Final evaluation
    z_shift = z - z.max()
    a_final = np.exp(z_shift)
    a_final = a_final / a_final.sum()
    score = 2.0 * n * fast_convolve(a_final).max()
    if score < best_score:
        best_score = score
        best_a = a_final.copy()

    return best_a


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=8):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    def make_func(T_val):
        def func(z_in):
            z_shift = z_in - z_in.max()
            a_in = np.exp(z_shift)
            a_in = a_in / a_in.sum()
            c = fast_convolve(a_in)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            lse = max_Tc + np.log(exp_Tc.sum())
            return (2.0 * n / T_val) * lse
        return func

    def make_jac(T_val):
        def jac(z_in):
            z_shift = z_in - z_in.max()
            a_in = np.exp(z_shift)
            a_in = a_in / a_in.sum()
            c = fast_convolve(a_in)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            dg_da = np.zeros(n)
            thresh = w.max() * 1e-8
            active = np.where(w > thresh)[0]
            for k in active:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                dg_da[idx] += w[k] * a_in[k - idx]
            S = np.dot(dg_da, a_in)
            return a_in * (dg_da - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(100, 8000 // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
            # Evaluate
            z_shift = z - z.max()
            a_eval = np.exp(z_shift)
            a_eval = a_eval / a_eval.sum()
            score = 2.0 * n * fast_convolve(a_eval).max()
            if score < best_score:
                best_score = score
                best_a = a_eval.copy()
            print(f"    L-BFGS T={stage_T:.2f}: score={best_score:.6f}")
        except Exception as e:
            print(f"    L-BFGS error: {e}")

    return best_a


def spectral_equalize(a0, time_limit, n_steps=20000):
    """Try to flatten the auto-convolution in the spectral domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 2).bit_length()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT of a (padded)
        A = np.fft.rfft(a, fft_size)
        # |A|^2 is the power spectrum
        power = np.abs(A) ** 2

        # We want |A|^2 to be as flat as possible (in the non-DC part)
        # This means we want to reduce peaks in the power spectrum
        # While maintaining the a >= 0 constraint

        # Compute current convolution
        c = fast_convolve(a)
        c_max = c.max()
        c_mean = c.mean()
        # Target: make c as flat as possible
        # In frequency domain, flat c means flat |A|^2

        # Simple approach: apply a slight equalization to the spectrum
        # Find where |A|^2 has peaks and reduce them
        power_max = power.max()
        if power_max < 1e-15:
            break

        # Compute gradient: which direction in a-space reduces the peak of c?
        # This is equivalent to the mirror descent / subgradient approach
        c_peak_k = int(np.argmax(c))
        # Subgradient of max(c) at peak
        g = np.zeros(n)
        js = max(0, c_peak_k - n + 1)
        je = min(n - 1, c_peak_k)
        idx = np.arange(js, je + 1)
        g[idx] = a[c_peak_k - idx]

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Step in the negative gradient direction (reduce a where it contributes to peak)
        lr = 0.001 * (1.0 + 0.1 * (1 - step / n_steps))
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for f in [0.5, 0.1, 0.01]:
                a_try = a - lr * f * g
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 0.01:
                    continue
                a_try = a_try / st
                sc = 2.0 * n * fast_convolve(a_try).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, reduce learning rate
                lr *= 0.9

    return best_a


def peak_shave(a0, time_limit, n_steps=20000):
    """Coordinate-wise peak shaving."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_work = c.copy()
        if len(c_work) > 2:
            c_work[0] = -np.inf
            c_work[-1] = -np.inf
        peak_k = int(np.argmax(c_work))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88, 0.83, 0.78]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair modification
            for ci in top[:4]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.93, 0.88]:
                    for f2 in [0.97, 0.93, 0.88]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # No improvement found in this step; try a random perturbation
            if step % 500 == 0 and step > 0:
                rng = np.random.default_rng(step)
                a_trial = a.copy()
                a_trial[rng.integers(n)] *= 0.9
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break  # Truly stuck

    return best_a


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x)) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        # Two-peak structure
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    else:
        # Cosine with modulation
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(2 * np.pi * x) ** 2
        a = a ** 1.5

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def eval_obj(a):
    """Evaluate the objective function (score)."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

    """
    Multi-phase optimization combining LP-based alternating method,
    L-BFGS LogSumExp refinement, spectral equalization, peak shaving,

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    print(f"Total budget: {budget_s:.0f}s")

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 1e-10:
            parent = parent / parent.sum()
            p_score = eval_obj(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)

        sc = eval_obj(a)

    rem = deadline - time.monotonic()

        print(f"  Budget: {t_alloc:.0f}s")
