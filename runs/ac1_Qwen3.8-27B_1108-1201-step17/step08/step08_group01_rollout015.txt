Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated proximal gradient with log-sum-exp smoothing
2. Adaptive temperature annealing for peak focus
3. Block coordinate descent with exact 1D/2D subproblems
4. Structured Fourier perturbations for escaping local optima
5. Multi-length exploration with diverse initialization
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def nesterov_proximal(a0, time_limit, T_init=10.0, T_end=100.0, verbose=False):
    """
    Nesterov-accelerated proximal gradient on smoothed objective.
    f_T(a) = (1/T) * log(sum_k exp(T * (a*a)[k]))
    with constraint a >= 0, sum(a) = 1 (simplex).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    y = a.copy()  # momentum variable
    t = 1.0
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    step = 1.0 / (2.0 * n)  # initial step size
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive temperature
        progress = min(1.0, it / (time_limit * 50))  # ~50 iterations per second target
        T = T_init * (T_end / T_init) ** progress
        
        # Compute smoothed objective and gradient
        conv = fast_convolve(y)  # use momentum point for gradient
        M = conv.max()
        # Softmax weights
        shifted = T * (conv - M)
        w = np.exp(shifted)
        w_sum = w.sum()
        w = w / w_sum
        
        # Gradient of f_T at y
        grad = np.zeros(n)
        for k_idx in range(len(conv)):
            if w[k_idx] < 1e-10:
                continue
            k = k_idx
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            j = np.arange(i_start, i_end + 1)
            # d(a*a)[k]/da[j] = 2*a[k-j] (for linear conv, this needs care)
            # Actually for linear conv: (a*a)[k] = sum_i a[i]*a[k-i] where 0<=i<n, 0<=k-i<n
            # d/d(a[j]) = sum_i [i==j]*a[k-i] + a[i]*[k-i==j] = a[k-j] + a[k-j] = 2*a[k-j]
            # (when both indices are valid)
            grad[j] += 2.0 * w[k_idx] * a[k - j]
        
        # Nesterov update
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta = (t_old - 1.0) / t
        
        # Extrapolation
        y_new = y + beta * (y - a)  # momentum
        # Gradient step
        y_new = y_new - step * grad
        # Project to simplex
        a_new = simplex_project(y_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Adaptive step size (decrease if oscillating)
        if it % 10 == 0:
            step *= 0.99
        
        a = a_new
        y = a_new  # reset momentum occasionally to avoid divergence
        if it % 50 == 0:
            y = a.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov it={it} ({elapsed:.1f}s): score={score:.6f}, best={best_score:.6f}, T={T:.1f}, step={step:.6f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov-accelerated proximal gradient) with p-norm continuation
2. FFT-based gradient computation in O(n log n) per iteration
3. Backtracking line search for robust step-size selection
4. Structured Fourier-mode perturbations for escaping local optima
5. Multi-length exploration with diverse initialization
6. Final coordinate descent polishing

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_gradient(a, p, fft_size=None, fft_a_rev=None):
    """
    Compute gradient of g_p(a) = (sum_k (a*a)[k]^p)^(1/p) on the simplex.
    Returns (grad, g_p_value, c) where c = conv(a,a).
    """
    n = len(a)
    c = fast_convolve(a)
    g_p = (np.sum(c ** p)) ** (1.0 / p)
    
    if g_p < 1e-30:
        return np.zeros(n), 0.0, c
    
    # h[k] = c[k]^(p-1)
    h = c ** (p - 1)
    
    # Cross-correlation: corr[i] = sum_k h[k] * a[k-i]
    # = conv(h, a_rev)[i + n - 1]
    a_rev = a[::-1].copy()
    
    need_size = (2 * n - 1) + n - 1  # = 3n - 2
    fft_sz = 1 << (need_size - 1).bit_length()
    
    H = rfft(h, fft_sz)
    if fft_a_rev is None:
        A_rev = rfft(a_rev, fft_sz)
    else:
        A_rev = fft_a_rev[:fft_sz] if len(fft_a_rev) >= fft_sz else rfft(a_rev, fft_sz)
    
    corr_full = irfft(H * A_rev, fft_sz)
    corr = corr_full[n - 1:n - 1 + n]
    
    grad = 2.0 * (g_p ** (1.0 - p)) * corr
    return grad, g_p, c


def fista_optimize(a0, time_limit, p_values=None, initial_step=None,
                   max_iters_per_p=2000, verbose=False, label=""):
    """
    FISTA with p-continuation to minimize max(conv(a,a)) on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if p_values is None:
        p_values = [2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64]
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for pi, p in enumerate(p_values):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Time allocation: more time for larger p
        remaining = time_limit - (time.monotonic() - t0)
        n_remaining_p = len(p_values) - pi
        t_alloc = remaining * (1.0 + 0.5 * pi) / (sum(1.0 + 0.5 * j for j in range(pi, len(p_values))))
        t_alloc = min(t_alloc, time_limit - (time.monotonic() - t0))
        
        if t_alloc < 1.0:
            break
        
        # FISTA inner loop
        t_fista = time.monotonic()
        step = initial_step if initial_step is not None else 1.0 / (n ** 0.6)
        y = a.copy()
        t_mom = 1.0
        a_prev = a.copy()
        it = 0
        stalling = 0
        
        while time.monotonic() - t_fista < t_alloc and it < max_iters_per_p:
            it += 1
            # Compute gradient at y
            grad, g_p, c = compute_gradient(y, p)
            
            # Backtracking line search
            step_local = step
           .Accepted = False
            for bt in range(8):
                a_trial = simplex_project(y - step_local * grad)
                # Check Armijo condition
                g_trial = (np.sum(fast_convolve(a_trial) ** p)) ** (1.0 / p)
                if g_trial <= g_p + 1e-4 * step_local * np.sum(grad * (a_trial - y)):
                    Accepted = True
                    a_new = a_trial
                    break
                step_local *= 0.5
            
            if not Accepted:
                a_new = simplex_project(y - step_local * grad)
            
            # Nesterov momentum update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            beta = (t_mom - 1.0) / t_new
            y = a_new + beta * (a_new - a_prev)
            t_mom = t_new
            a_prev = a_new.copy()
            a = a_new
            
            # Track best
            c_new = fast_convolve(a)
            score = 2.0 * n * c_new.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stalling = 0
            else:
                stalling += 1
            
            # Adaptive step: increase if making progress, decrease if stalling
            if stalling == 50:
                step *= 0.8
                stalling = 0
            elif stalling == 100:
                step *= 0.6
                stalling = 0
            elif it % 100 == 0 and stalling < 20:
                step *= 1.05  # cautiously increase
            
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"      FISTA p={p} it={it} ({elapsed:.1f}s): g_p={g_p:.8f}, max_c={c_new.max():.8f}, score={score:.6f}")
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA p={p} done ({elapsed:.1f}s): iters={it}, best={best_score:.6f}")
    
    return best_a, best_score


def structured_perturb(a0, rng, mode="fourier", strength=0.1):
    """Apply a structured perturbation to escape local optima."""
    n = len(a0)
    a = a0.copy()
    
    if mode == "fourier":
        # Add a random Fourier mode
        freq = rng.integers(2, max(3, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        pert = strength * np.cos(2 * np.pi * freq * t / n + phase)
        a = a + pert * a.sum()
    elif mode == "anti_peak":
        # Reduce mass at coordinates contributing most to the peak
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        a[i_start:i_end + 1] *= (1.0 - strength * rng.random(i_end - i_start + 1))
    elif mode == "spread":
        # Spread mass toward edges
        t = np.linspace(0, 1, n)
        edge_bias = strength * (t ** 2 + (1 - t) ** 2) - 0.5 * strength
        a = a + edge_bias * a.sum()
    elif mode == "random_block":
        # Perturb a random block
        block_size = rng.integers(n // 10, n // 3)
        start = rng.integers(0, n - block_size)
        a[start:start + block_size] += strength * rng.standard_normal(block_size) * a.mean()
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def coord_descent_polish(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent polishing using sensitivity analysis.
    Only considers the top few peaks for efficiency.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        conv = fast_convolve(a)
        M = float(conv.max())
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]
            
            best_delta = 0.0
            best_new_max = M
            
            for factor in [0.85, 0.9, 0.95, 0.97, 0.99, 1.01, 1.03, 1.05, 1.1, 1.15]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def init_fourier(n, rng):
    """Initialize with random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 15))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, max(2, n // 4)), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Initialize with cosine bell."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_tapered(n, rng):
    """Initialize with tapered (raised cosine) shape."""
    t = np.linspace(0, 1, n)
    alpha = rng.uniform(0.5, 2.0)
    a = 0.5 * (1 + np.cos(2 * np.pi * alpha * (t - 0.5)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_two_hump(n, rng):
    """Initialize with two-hump shape."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.35)
    c2 = n * rng.uniform(0.65, 0.8)
    w1 = n * rng.uniform(0.08, 0.15)
    w2 = n * rng.uniform(0.08, 0.15)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
