Optimized auto-convolution peak minimization using improved LP with adaptive
constraint bands, combined direction strategies, warm-started solves,
multi-length exploration with diverse initializations, and annealed perturbations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_vals = np.arange(1, n + 1)
    cond = v_sorted * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(n, f, indices):
    """Build constraint matrix A where A[i,j] = f[j] shifted for lag indices[i]."""
    K = len(indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = indices[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, active_indices, M, time_budget=30.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, A@g <= M.
    Returns (g, M, dual_values) or (None, M, None) on failure.
    """
    t0 = time.monotonic()
    K = len(active_indices)
    if K == 0:
        return None, M, None

    A = build_constraint_matrix(n, f, active_indices)

    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )

    constraints_lp = A @ g_var <= M

    solved = False
    g = None
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        if time.monotonic() - t0 > time_budget * 0.7:
            break
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                if not np.any(np.isnan(g)):
                    solved = True
                    break
        except Exception:
            continue

    if not solved or g is None:
        return None, M, None

    # Get dual values
    duals = None
    try:
        duals = np.array([c.dual_value for c in constraints_lp])
    except Exception:
        pass

    return g, M, duals


def select_constraints_adaptive(b, M, epsilon, max_constraints=500):
    """
    Select constraint indices: all where b[k] >= (1-epsilon)*M,
    plus top-k if fewer than max_constraints.
    """
    threshold = (1.0 - epsilon) * M
    active = np.where(b >= threshold)[0]

    if len(active) > max_constraints:
        # Keep the most violated
        keep = np.argsort(b[active])[-max_constraints:]
        active = active[keep]

    return sorted(active.tolist())


def lp_direction_with_band(f, n, epsilon=0.01, max_constraints=400, time_budget=30.0):
    """
    Compute LP direction with adaptive epsilon band.
    Returns (direction, M, active_count).
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M, 0

    active_indices = select_constraints_adaptive(b, M, epsilon, max_constraints)
    g, M, duals = solve_lp(f, n, active_indices, M, time_budget)

    if g is None:
        return None, M, len(active_indices)

    S_g = g.sum()
    if S_g <= 1.0 + 1e-5:
        return None, M, len(active_indices)

    return g / S_g, M, len(active_indices)


def subgradient_direction(f, n, top_k=20):
    """Compute averaged subgradient of max auto-convolution."""
    b = fast_convolve(f)
    M = b.max()
    threshold = 0.99 * M
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([b.argmax()])
    if len(active) > top_k:
        active = np.argsort(b[active])[-top_k:]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += f[k - j]

    if grad.max() < 1e-15:
        return None
    return grad / grad.sum()

        )

        if dir_lp is not None:
            direction = dir_lp
        else:
            # Fallback to subgradient
            direction = subgradient_direction(f, n, top_k=min(20, n // 10))
            if direction is None:
                break

        # Line search
        t_step = line_search_exact(f, direction)

        if t_step <= 0:
            stuck += 1
            epsilon = min(epsilon * 1.5, epsilon_max)
            continue

        # Update
        f_new = (1.0 - t_step) * f + t_step * direction
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            stuck += 1
            continue
        f_new = f_new / s

        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
            # Slightly tighten epsilon on success
            epsilon = max(epsilon * 0.97, epsilon_init)
            top_k = min(top_k + 2, top_k_max)
        else:
            stuck += 1
            epsilon = min(epsilon * 1.3, epsilon_max)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"eps={epsilon:.4f} t={elapsed:.0f}s/{time_limit:.0f}s")

    return best_f, best_obj


def spectral_init(n, mode="flatten"):
    """
    Generate initial sequence using spectral methods.
    'flatten': try to make |F(a)|^2 flat (which makes a*a flat)
    """
    if mode == "flatten":
        # Start with random non-negative, iterate to flatten spectrum
        rng = np.random.default_rng(42)
        a = rng.random(n)
        for _ in range(50):
            A = rfft(a)
            target = np.ones(len(A))
            # Water-filling: reduce high frequency components
            power = np.abs(A) ** 2
            peak_power = power.max()
            if peak_power < 1e-10:
                break
            # Reduce peaks in frequency domain
            mask = power > 1.5 * power.mean()
            A[mask] *= 0.95
            a = irfft(A, n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                break
            a /= s
        return a
    elif mode == "cosine":
        # Cosine-based: a[j] proportional to cos^2(pi*j/(n-1))
        j = np.arange(n)
        a = np.cos(np.pi * j / (n - 1)) ** 2
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    elif mode == "gaussian":
        j = np.arange(n) - n / 2.0
        sigma = n / 4.0
        a = np.exp(-j ** 2 / (2 * sigma ** 2))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    elif mode == "bilateral":
        # Two bumps
        j = np.arange(n)
        center1 = n * 0.3
        center2 = n * 0.7
        sigma = n * 0.15
        a = np.exp(-(j - center1) ** 2 / (2 * sigma ** 2)) + \
            np.exp(-(j - center2) ** 2 / (2 * sigma ** 2))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    else:
        return np.ones(n) / n


def fine_polish(f, time_limit, label="polish"):
    """
    Very fine-grained optimization: small steps with many constraints.
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()
        # Use all constraints above 99% of max
        threshold = 0.99 * M
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([b.argmax()])
        active = sorted(active.tolist())

        time_per_lp = min(10.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        g, M2, _ = solve_lp(f, n, active, M, time_budget=time_per_lp)

        if g is not None and g.sum() > 1.0 + 1e-5:
            direction = g / g.sum()
            t_step = line_search_exact(f, direction)
            if t_step > 0:
                f_new = (1.0 - t_step) * f + t_step * direction
                f_new = np.maximum(f_new, 0.0)
                s = f_new.sum()
                if s > 1e-10:
                    f_new /= s
                    obj_new = fast_convolve(f_new).max()
                    if obj_new < best_obj - 1e-15:
                        best_obj = obj_new
                        best_f = f_new.copy()
                        f = f_new
        else:
            # Subgradient fallback with very small steps
            grad = np.zeros(n)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    grad[j] += f[k - j]
            if grad.max() > 1e-15:
                for alpha in [0.001, 0.002, 0.005]:
                    f_trial = f - alpha * grad
                    f_trial = project_simplex(f_trial)
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_f = f_trial.copy()
                        f = f_trial
                        break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):

    adaptive constraint bands, and fine polishing.
    """

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Quick LP on parent ===
    print(f"\n=== Phase 1: LP on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 480.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_optimization(parent, time_limit=t1, label="lp_parent",
                                        epsilon_init=0.003, epsilon_max=0.04,

        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1

        else:
            print(f"  lp_parent: {score1:.8f}")

    # === Phase 2: Multi-length exploration with diverse inits ===
    print(f"\n=== Phase 2: Multi-length exploration ===")
    rem = deadline - time.monotonic()
    n_base = len(best_seq)

    # Generate candidate lengths
    candidate_lengths = set()
    for factor in [0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.2, 1.3]:
        candidate_lengths.add(int(round(n_base * factor / 64)) * 64)

        if 512 <= nl <= 3000:
            candidate_lengths.add(nl)
    candidate_lengths = sorted([c for c in candidate_lengths if c != n_base and 512 <= c <= 3000])
    candidate_lengths = candidate_lengths[:5]  # Limit to 5 lengths

    for target_n in candidate_lengths:

            break

        t_explore = min(rem * 0.12, 120.0)
        if t_explore < 20:
            break

        print(f"  Exploring n={target_n}, budget={t_explore:.0f}s")

        # Multiple initializations
        inits = []
        # Interpolate best
        old_x = np.linspace(0.0, 1.0, len(best_seq))
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        f_interp = project_simplex(f_interp)
        inits.append(("interp", f_interp))

        # Spectral flatten

        # Cosine
