import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(w, a):
    """Compute subgradient g[j] = 2 * sum_k w[k] * a[k-j] using FFT."""
    n = len(a)
    N = 1 << ((3 * n - 1) - 1).bit_length()  # >= 3n-2
    W = fft(w, N)
    A = fft(a, N)
    C = ifft(W * np.conj(A))
    return 2.0 * C.real[:n]

def score_normalized(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

Key improvements: fast O(n log n) subgradient via FFT, adaptive annealing,

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(w, a):
    """Compute subgradient g[j] = 2*sum_k w[k]*a[k-j] via FFT. O(n log n)."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    C = ifft(W * np.conj(A))
    return 2.0 * C.real[:n]

    """Build LP constraint matrix for given peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_idx, eps=1e-8, max_iters=200000):
    """Solve: max sum(g) s.t. g>=0, A@g <= M. Returns g or None."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=max_iters, eps=eps)
    except Exception:
        return None
    if g.value is None:
        if n <= 2000:

            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Solve LP
        g = solve_lp(f, n, M, active_idx, eps=1e-8, max_iters=150000)

        if g is None:
            stuck += 1
            if stuck > 6:
                break
            if len(active_idx) > initial_k:
                active_idx = active_idx[np.argsort(b[active_idx])[-initial_k:]]
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                break
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[
                    -min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue

        # Cutting plane check
        fg = convolve_cross(f, g)
        violated = fg > M * (1 + 1e-7)
        v_count = violated.sum()

        if v_count > 0 and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(20, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp(f, n, M, active_idx, eps=1e-8, max_iters=100000)
                if g is None:
                    stuck += 1
                    if stuck > 4:
                        break
                    continue
                S_g = g.sum()

        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                break
            continue

        stuck = 0
        h = g / S_g

        # Line search with golden section refinement
        best_t = 0.1
        best_t_obj = M
        for t in [0.03, 0.06, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            ft = np.maximum((1.0 - t) * f + t * h, 0.0)
            st = ft.sum()
            if st < 1e-10:
                continue
            ft /= st
            ot = fast_convolve(ft).max()
            if ot < best_t_obj:
                best_t_obj = ot
                best_t = t

        # Refine
        if best_t > 0.02:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(5):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        # Update
        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        # Refresh active set periodically
        if it % 10 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k * 2:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]

        if it % 20 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} active={len(active_idx)} "
                  f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def spectral_diversify(f_init, n_iters=100, peak_fraction=0.3, seed=42):
    """
    Diversify by flattening the power spectrum (|FFT|^2),
    then project back to non-negative time domain.
    """
    rng = np.random.default_rng(seed)
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]

        # Target: clip R to be flatter
        peak_val = R.max()
        floor = peak_fraction * peak_val
        R_target = np.clip(R, floor, peak_val)

        # Scale to preserve energy
        energy = R.sum()
        rt_sum = R_target.sum()
        if rt_sum > 1e-10:
            R_target *= energy / rt_sum

        # Convert back: adjust amplitude spectrum
        R_fft = fft(R_target, fft_size)
        real_part = np.maximum(R_fft.real, 0.0)
        new_amp = np.sqrt(real_part)

        # Preserve phase, add small random perturbation
        abs_F = np.abs(F)
        if np.any(abs_F > 1e-15):
            phase = F / (abs_F + 1e-30)
        else:
            phase = np.ones_like(F)
        # Small phase perturbation for diversity
        if it % 10 == 5:
            phase_pert = np.exp(1j * rng.standard_normal(len(F)) * 0.01)
            phase = phase * phase_pert

        F_new = new_amp * phase
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            break
        f = f_new

    return f


def coordinate_polish(f_init, time_limit, n_sweeps=10):
    """Fast coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)
    rng = np.random.default_rng(7)

    improved_any = True
    sweep = 0
    while improved_any and time.monotonic() - t0 < time_limit and sweep < n_sweeps:
        sweep += 1
        improved_any = False
        # Randomize order
        order = np.arange(0, n, step)
        rng.shuffle(order)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 1.15, 1.3]:
                f[j] = best_val * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = f[j]
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                improved_any = True

    return f, best_obj


def reversal_averaging(f):
    """Average with reversed sequence (preserves convolution spectrum structure)."""
    f_rev = f[::-1].copy()
    f_avg = 0.5 * (f + f_rev)
    s = f_avg.sum()
    if s > 1e-10:
        f_avg /= s
    return f_avg


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search algorithm: multi-phase optimization combining
    FFT-based subgradient, cutting-plane LP, spectral diversification,
    and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Fast subgradient for initial improvement ===
    print(f"\n{'='*50}")
    print(f"Phase 1: Subgradient (n={len(parent)})")
    print(f"{'='*50}")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = subgradient_phase(parent, time_limit=t1, lr_init=0.003,
                                     temp_init=0.005, n_top=80, label="sg1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [sg1]: {score1:.6f}")

            print(f"  No improvement: {score1:.6f} vs {best_score:.6f}")

    print(f"\n{'='*50}")

    print(f"{'='*50}")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)

                                          initial_k=60, max_k=400, label="cp1")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [cp1]: {score2:.6f}")
        else:
            print(f"  No improvement: {score2:.6f} vs {best_score:.6f}")

    # === Phase 3: Spectral diversification + LP ===
    print(f"\n{'='*50}")
    print(f"Phase 3: Diversify + LP (score={best_score:.6f})")
    print(f"{'='*50}")
    rem = deadline - time.monotonic()
    t3_total = min(rem * 0.18, 250.0)
    if t3_total > 30:
        # Variant A: peak_fraction=0.2
        f_da = spectral_diversify(best_seq, n_iters=100, peak_fraction=0.2, seed=1)
        f_da = reversal_averaging(f_da)
        t3a = t3_total * 0.4
        f3a, _ = lp_cutting_plane_phase(f_da, time_limit=t3a,
                                        initial_k=40, max_k=250, label="div_a")

            t3b = min(rem3 * 0.5, 100.0)

                                            initial_k=40, max_k=250, label="div_b")

    print(f"Phase 4: Multi-length (score={best_score:.6f})")

    n_base = len(best_seq)

                                   temp_init=0.004, n_top=50, label=f"n{target_n}_sg")
        t4_lp = t4 - t4_sg - 2
        f4, _ = lp_cutting_plane_phase(f4s, time_limit=t4_lp,
                                       initial_k=40, max_k=250, label=f"n{target_n}")
        s4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4) and s4 < best_score:
