Actually, there's an even simpler and potentially better approach: **direct subgradient descent with proper step size**.

The objective is F(f) = max_k (f*f)[k] (with sum(f)=1, f>=0).
The subgradient at f (when the max is attained at lag k*) is:
∇F(f) = 2 * R_{k*} f (where R_{k*} is the reversal-shift operator)

For multiple active lags, we use a convex combination.

The subgradient descent step is:
f_new = project_simplex(f - alpha * grad)

Where alpha is the step size. The key question is: what's a good step size?

For the max function, the subgradient has norm ||2*R_k f||_2 = 2*||f||_2. With sum(f)=1, ||f||_2 <= 1.

A safe step size for subgradient methods is alpha_t = c/sqrt(t) for some constant c. But this converges slowly.

A better approach: use the step size that would make the peak decrease by a certain amount. If the peak is M and the subgradient direction is d, then:
F(f - alpha*d) <= M - alpha * (some lower bound on the decrease)

The decrease is at least alpha * (M - min_k (f*f)[k]) / ||d|| (by convexity of the max of linear functions).

Uses a direct LP formulation (minimize max(f*g) s.t. sum(g)=1, g>=0)

from scipy.fft import rfft, irfft
import cvxpy as cp
import time

    return irfft(A * A, fft_size)[:size]

    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    if not cond.any():

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the matrix B such that B @ g = f * g (full linear convolution)."""
    n = len(f)
    K = 2 * n - 1
    B = np.zeros((K, n))
    # B[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # For column i: B[k, i] = f[k-i] for k in [i, i+n-1]
    for i in range(n):
        B[i:i+n, i] = f
    return B


def solve_lp_direction(f, n, verbose=False):
    """
    Solve: min t s.t. g >= 0, sum(g) = 1, (f*g)[k] <= t for all k.
    Returns (g, t_opt) or (None, None).
    """
    K = 2 * n - 1
    B = build_conv_matrix(f)
    
    g_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        g_var >= 0,
        cp.sum(g_var) == 1,
        (B @ g_var) - t_var <= 0
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for solver in [cp.PDLP, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None and t_var.value is not None:
                g = np.maximum(np.real(g_var.value), 0.0)
                s = g.sum()
                if s > 1e-10 and np.all(np.isfinite(g)):

    return None, None


def line_search(f, g, M):

    n = len(f)
    best_t = 0.0

    candidates = [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]

def subgradient_optimize(f_init, time_limit, label="sg", verbose=True):
    """Temperature-based subgradient optimization with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)
    lr = 0.005
    temp = 0.1
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    momentum = 0.9
    
    while time.monotonic() - t0 < time_limit:
        # Batch of iterations
        batch = min(50, max(10, int((time_limit - (time.monotonic() - t0)) * 3)))
        
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            
            b = fast_convolve(f)
            M = b.max()
            
            # Temperature-based weights
            scaled = b / max(temp, 1e-10)
            scaled = scaled - M / max(temp, 1e-10)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum
            
            # Compute subgradient
            grad = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                for i in range(n):
                    if 0 <= k - i < n:
                        grad[i] += 2.0 * w[k] * f[k - i]
            
            # Update with momentum
            v = momentum * v - lr * grad
            f_new = project_simplex(f + v)
            
            obj_new = fast_convolve(f_new).max()
            f = f_new
        
        # Check progress
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.02)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 5:
                # Anneal temperature
                temp = min(temp * 1.5, 1.0)
                if no_improve > 15:
                    # Restart with perturbation
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                    v = np.zeros(n)
                    no_improve = 0
                    lr = 0.005
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"  {label} iter={it} obj={best_obj:.10f} score={score:.6f} temp={temp:.4f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def peak_shaving(f_init, time_limit, label="peak"):
    """Direct peak shaving: reduce the peak by adjusting contributing elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        peak_idx = int(b.argmax())
        M = b[peak_idx]
        
        # Find which elements of f contribute most to the peak
        # (f*f)[peak_idx] = sum_i f[peak_idx - i] * f[i]
        contributions = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contributions[i] = f[j]
        
        # Reduce the largest contributors
        top_contrib = np.argsort(contributions)[-5:]
        
        # Try reducing each contributor slightly
        best_alpha = 0
        best_trial = best_obj
        for ci in top_contrib:
            for alpha in [0.001, 0.005, 0.01, 0.02]:
                f_trial = f.copy()
                reduction = alpha * f[ci]
                if reduction <= 0:
                    continue
                f_trial[ci] -= reduction
                # Redistribute to the smallest elements
                f_trial += reduction / n
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_trial:
                    best_trial = obj
                    best_alpha = (ci, alpha)
        
        if best_alpha == 0:
            break
        
        ci, alpha = best_alpha
        f_trial = f.copy()
        reduction = alpha * f[ci]
        f_trial[ci] -= reduction
        f_trial += reduction / n
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        f_trial /= s
        
        if best_trial < best_obj:
            best_obj = best_trial
            best_f = f_trial.copy()
            f = f_trial
        else:
            break
    
    return best_f, best_obj

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    n = len(f)
    K = 2 * n - 1
    B = np.zeros((K, n))
    for i in range(n):
        B[i:i + n, i] = f
    return B


def solve_lp_full(f, n):
    B = build_conv_matrix(f)
    g_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        g_var >= 0,
        cp.sum(g_var) == 1,
        (B @ g_var) - t_var <= 0
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    for solver in [cp.PDLP, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None and t_var.value is not None:
                g = np.maximum(np.real(g_var.value), 0.0)
                s = g.sum()
                if s > 1e-10 and np.all(np.isfinite(g)):
                    g /= s
                    return g, float(t_var.value)
        except Exception:
            continue
    return None, None


def solve_lp_active(f, n, M, top_k=100):

    active_set = set(top_idx.tolist())
    K = 2 * n - 1
    B = np.zeros((len(active_set), n))
    for row_i, k in enumerate(sorted(active_set)):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            B[row_i, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    g_var = cp.Variable(n)

def subgradient_optimize(f_init, time_limit, label="sg", verbose=True):

def peak_shaving(f_init, time_limit, label="peak"):

        peak_idx = int(b.argmax())
        M = b[peak_idx]
        contributions = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contributions[i] = f[j]
        top_contrib = np.argsort(contributions)[-8:]
        found = False
        for ci in top_contrib:
            for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                reduction = alpha * f[ci]
                if reduction <= 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[ci] -= reduction
                f_trial += reduction / n
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:

                f_trial /= s
                obj = fast_convolve(f_trial).max()

                    f = f_trial

    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = subgradient_optimize(best_seq, time_limit=t2, label="sg1")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [sg1]: {score2:.8f}")
        else:
            print(f"  sg1: {score2:.8f}")

    # Phase 3: LP with active constraints (faster per-iteration)
    print(f"\n=== Phase 3: LP active (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 320.0)
    if t3 > 60:
        print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
        f3, obj3 = lp_optimize(best_seq, time_limit=t3, label="lp_act", use_full=False, top_k=150)
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp_act]: {score3:.8f}")
        else:
            print(f"  lp_act: {score3:.8f}")

    # Phase 4: Multi-length exploration
    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    candidate_lengths = sorted(set(
        l for l in [n_best - 128, n_best - 64, n_best + 64, n_best + 128,
                    n_best + 256, 1024, 1280, 1536, 1920, 2048]
        if 512 <= l <= 3072 and l != n_best
    ))

    for target_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t4 = min(rem * 0.12, 180.0)
        if t4 < 40:
            break

        old_x = np.linspace(0.0, 1.0, n_best)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s

        print(f"  n={target_n}, budget={t4:.0f}s")
        f4, obj4 = lp_optimize(f_interp, time_limit=t4, label=f"n{target_n}", use_full=True)
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {score4:.8f}")
        else:
            print(f"  n{target_n}: {score4:.8f}")

    # Phase 5: Final subgradient polish
    print(f"\n=== Phase 5: Final subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 100.0)
    if t5 > 20:
        print(f"  Budget: {t5:.0f}s, n={len(best_seq)}")
        f5, obj5 = subgradient_optimize(best_seq, time_limit=t5, label="sg_fin")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg_fin]: {score5:.8f}")
        else:
            print(f"  sg_fin: {score5:.8f}")

    # Phase 6: Peak shaving
    print(f"\n=== Phase 6: Peak shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.12, 50.0)
    if t6 > 15:
        f6, obj6 = peak_shaving(best_seq, time_limit=t6, label="peak")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [peak]: {score6:.8f}")
        else:
            print(f"  peak: {score6:.8f}")

    # Phase 7: Final LP burst on remaining time
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 7: Final LP burst (score={best_score:.8f}, rem={rem:.0f}s) ===")
        t7 = min(rem * 0.6, 80.0)
        f7, obj7 = lp_optimize(best_seq, time_limit=t7, label="lp_fin", use_full=True)
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp_fin]: {score7:.8f}")
        else:
            print(f"  lp_fin: {score7:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
