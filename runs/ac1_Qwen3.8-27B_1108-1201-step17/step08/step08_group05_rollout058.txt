def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad_fft(f, w):
    """
    Compute subgradient of sum_k w[k]*(f*f)[k] w.r.t. f.
    f: length n
    w: length 2n-1
    Returns: gradient of length n

    n = len(f)

def softmax_weights(b, temp):
    """Compute softmax weights from convolution values."""
    if temp < 1e-8:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
        return w
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
        return w
    return w / s

"""
Ensemble subgradient + LP optimization for auto-convolution peak minimization.
Uses FFT-based gradient computation, momentum, full-constraint LP refinement,
temperature scheduling, and multi-length exploration.
Targets score < 1.5027 (lower is better).
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_fft(f, w):
    """
    Compute subgradient of sum_k w[k]*(f*f)[k] w.r.t. f using FFT.
    f: length n
    w: length 2n-1
    Returns: gradient of length n
    """
    n = len(f)
    conv_size = 3 * n - 2
    fft_size = 1 << (conv_size - 1).bit_length()

    W = rfft(w.astype(np.float64), fft_size)
    F = rfft(f, fft_size)
    F_rev = rfft(f[::-1], fft_size)

    # Part B: (f * w) linear convolution, take first n
    fw = irfft(F * W, fft_size)
    part_b = fw[:n]

    # Part A: (f_rev * w) linear convolution, take [n-1 : 2n-1]
    frw = irfft(F_rev * W, fft_size)
    part_a = frw[n - 1:2 * n - 1]

    return part_a + part_b


def softmax_weights(b, temp):
    """Compute softmax weights from convolution peak values."""
    if temp < 1e-8:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
        return w
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
        return w
    return w / s


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    len_b = 2 * n - 1
    A = np.zeros((len_b, n), dtype=np.float64)
    for k in range(len_b):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - hi:k - lo + 1]
    return A


def lp_solve_full(f, M, time_limit=30.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns g (length n) or None.
    """
    t0 = time.monotonic()
    n = len(f)

    if time.monotonic() - t0 > time_limit:
        return None

    A = build_constraint_matrix(f)

    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )

    g = None
    solvers = [cp.PDLP, cp.SCS]
    for solver in solvers:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            prob.solve(solver=solver)
            if g_var.value is not None:
                g_candidate = np.maximum(g_var.value, 0.0)
                if np.all(np.isfinite(g_candidate)):
                    g = g_candidate
                    break
        except Exception:
            continue

    return g

def lp_descent(f_init, time_limit, top_k=100, blend=0.3):
    """
    Pure LP-based descent with constraint selection (top-k peaks).
    Faster than full LP for large n.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.astype(np.float64).copy())
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5.0:
            break

        b = fast_convolve(f)
        M = b.max()

        # Select top-k constraints
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))

        # Build reduced constraint matrix
        A = np.zeros((len(top_idx), n), dtype=np.float64)
        for row_i in range(len(top_idx)):
            k = int(top_idx[row_i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[k - hi:k - lo + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )

        g = None
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None:
                    g_candidate = np.maximum(g_var.value, 0.0)
                    if np.all(np.isfinite(g_candidate)):
                        g = g_candidate
                        break
            except Exception:
                continue

        if g is None:
            stuck += 1
            if stuck > 5:
                break
            continue

        g_sum = g.sum()
        if g_sum <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10:
                # Increase top_k
                top_k = min(top_k * 2, len(b))
                stuck = 0
            continue

        # Blend
        h = g / g_sum
        f_new = project_simplex((1.0 - blend) * f + blend * h)
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 15:
                break

        if it % 5 == 0:
            score = 2.0 * n * best_obj
            print(f"    lp it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"top_k={top_k} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def mixed_optimize(f_init, time_limit,
                   sg_time_frac=0.6,
                   lr_init=0.02, momentum=0.85,
                   temp_init=0.001, temp_decay=1.003,
                   top_k_lp=80):
    """
    Mixed subgradient + LP optimization.
    Alternates between fast subgradient steps and LP refinement.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.astype(np.float64).copy())
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()

    # Phase A: Subgradient
    sg_time = time_limit * sg_time_frac
    f_sg, obj_sg = subgradient_optimize(
        f, time_limit=sg_time,
        lr_init=lr_init, momentum=momentum,
        temp_init=temp_init, temp_decay=temp_decay,
        lp_every=500, lp_blend=0.3,
        restart_every=200,
        verbose_every=5000)
    if obj_sg < best_obj:

        best_f = f_sg.copy()
        f = f_sg.copy()

        if obj_lp < best_obj:
            best_obj = obj_lp
            best_f = f_lp.copy()

    return best_f, best_obj

    """Generate a normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Generate a normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def kaiser_window(n, beta=5.0):
    """Generate a normalized Kaiser window."""
    if n == 1:
        return np.array([1.0])
    from scipy.special import i0
    x = np.linspace(-1, 1, n)
    w = i0(beta * np.sqrt(1 - x**2)) / i0(beta)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolate_sequence(f, n_target):
    """Interpolate a sequence to a different length."""
    n_src = len(f)
    if n_src == n_target:
        return f.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    f_interp = np.interp(x_tgt, x_src, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return f_interp / s
