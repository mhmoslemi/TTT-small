"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses spectral flattening (Bussgang), multiplicative weights,
log-sum-exp L-BFGS, and coordinate descent.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def evaluate_normalized(a):
    """Evaluate using the actual evaluate_sequence function."""
    a = np.maximum(a, 0.0)

    a_norm = a / s
    return float(evaluate_sequence(a_norm.tolist()))


def bussgang_flatten(a, n_iter=1000, tau=0.3, time_limit=None, t0=None):
    """Bussgang spectral flattening iteration."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for it in range(n_iter):
        if t0 is not None and time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        mean_power = power.mean()
        
        # Flatten: A_new = A * (mean_power / power)^tau
        ratio = np.where(power > 1e-30, mean_power / power, 1.0)
        ratio = np.clip(ratio, 1e-3, 1e3)
        A_new = A * (ratio ** tau)
        
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:

        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score


def compute_smoothed_gradient(a, T):
    """Compute gradient of 2n * T * log(sum exp(C/T)) w.r.t. a.
    
    Returns (4n/T) * cross_corr(w, a) where w = softmax(C/T).
    Actually returns 4n * cross_corr(w, a) since T cancels.
    """
    n = len(a)
    C = fast_convolve(a)
    
    # Softmax weights
    C_shifted = C - C.max()
    w = np.exp(C_shifted / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    
    # Cross-correlation: grad[i] = 4n * sum_m w[m] * a[m-i]
    N = 1 << (3 * n - 3).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    cross = irfft(np.conj(W) * A, N)
    
    return 4.0 * n * cross[:n]

    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Stable softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def bussgang_flatten(a, time_limit, t0=None, tau_start=0.4, tau_end=0.05,
                     n_phases=6, verbose=True):
    """Bussgang spectral flattening with decreasing tau."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    fft_size = 1 << (2 * n - 2).bit_length()

    if t0 is None:
        t0 = time.monotonic()

    iters_per_phase = 10000
    total_iters = n_phases * iters_per_phase

    for phase in range(n_phases):
        tau = tau_start + (tau_end - tau_start) * phase / max(n_phases - 1, 1)
        for it in range(iters_per_phase):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            A = rfft(a, fft_size)
            power = np.abs(A) ** 2
            mean_power = power.mean()

            # Safety: avoid division by zero
            ratio = np.ones_like(power)
            mask = power > 1e-30
            ratio[mask] = mean_power / power[mask]
            ratio = np.clip(ratio, 1e-6, 1e6)

            A_new = A * (ratio ** tau)
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                break
            a_new = a_new / ss

            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Bussgang phase={phase}, tau={tau:.3f}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def subgradient_multiplicative(a, time_limit, t0=None, verbose=True):
    """Subgradient multiplicative weights with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    if t0 is None:
        t0 = time.monotonic()

    C = fast_convolve(a)
    T = C.max() * 2.0
    eta = 0.5
    stall = 0
    iters = 0
    fft_size = 1 << (3 * n - 3).bit_length()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        C = fast_convolve(a)
        max_C = C.max()

        # Softmax weights
        C_norm = C / T
        C_shifted = C_norm - C_norm.max()
        w = np.exp(C_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w = w / w_sum

        # Gradient: grad[i] = 4n * sum_m w[m] * a[m-i]
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        cross = irfft(np.conj(W) * A, fft_size)
        grad = 4.0 * n * cross[:n]

        # Center the gradient (preserves product of multiplicative factors)
        grad_centered = grad - grad.mean()

        # Scale for stability
        grad_range = np.abs(grad_centered).max()
        if grad_range < 1e-15:
            break
        effective_eta = eta * T / (grad_range * 0.5 + 1e-10)
        effective_eta = min(effective_eta, 2.0)

        # Multiplicative update
        log_update = np.exp(-effective_eta * grad_centered / (grad_range + 1e-10))
        a = a * log_update
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive tuning
        if stall > 200:
            T = max(C.max() * 0.7, 1e-10)
            eta = max(eta * 0.8, 0.01)
            stall = 0
            # Small perturbation
            rng = np.random.default_rng(iters * 7919)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad-MW iter={iters}, best={best_score:.8f}, T={T:.6f}, eta={eta:.4f}, time={elapsed:.0f}s")

    return best_a, best_score


def lbgfs_smoothed(a0, time_limit, n_phases=4, iters_per_phase=2000, verbose=True):
    """L-BFGS on log-sum-exp smoothed objective with temperature annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    z0 = np.log(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    best_z = z0.copy()

    C = fast_convolve(a)
    T_start = C.max() * 4.0
    T_end = C.max() * 0.02
    fft_size_grad = 1 << (3 * n - 3).bit_length()

    for phase in range(n_phases):
        T = T_start * (T_end / T_start) ** (phase / max(n_phases - 1, 1))
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_local = exp_z / exp_z.sum()
            C_local = fast_convolve(a_local)
            C_norm = C_local / T
            C_shifted = C_norm - C_norm.max()
            lse = C_norm.max() + np.log(np.exp(C_shifted).sum())
            return 2.0 * n * T * lse

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_local = exp_z / exp_z.sum()
            C_local = fast_convolve(a_local)
            C_norm = C_local / T
            C_shifted = C_norm - C_norm.max()
            w = np.exp(C_shifted)
            w = w / w.sum()
            W = rfft(w, fft_size_grad)
            A = rfft(a_local, fft_size_grad)
            cross = irfft(np.conj(W) * A, fft_size_grad)
            grad_a = 4.0 * n * cross[:n]
            S = float(np.dot(a_local, grad_a))
            return a_local * (grad_a - S)

        try:
            maxiter = min(iters_per_phase, max(100, int(remaining * 50)))
            res = minimize(
                fun=func, x0=best_z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter * 40,
                         'disp': False}
            )
            a_test = softmax_from_z(res.x)
            score = compute_score(a_test)
            if score < best_score:
                best_score = score
                best_a = a_test.copy()
                best_z = res.x.copy()
            else:
                best_z = res.x.copy()  # Still use for next phase
        except Exception:
            pass

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS phase={phase}, T={T:.6f}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=500, verbose=True, t0=None):
    """Coordinate descent with pairwise transfers and greedy line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    if t0 is None:
        t0 = time.monotonic()

    rng = np.random.default_rng(42)
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12:
                continue

            best_local = best_score
            best_t = 0.0
            base = a[i] * 0.5

            for frac in [0.1, 0.3, 0.5, 0.8]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_local < best_score:
                    best_score = best_local
                    best_a = a.copy()
                    improved = True
                    stall = 0

        if not improved:
            stall += 1
            if stall > 50:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stall = 0

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Check if a new sequence improves the score."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def raised_cosine_init(n):
    """Initialize with a raised-cosine window for better starting point."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a
