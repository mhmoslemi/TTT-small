def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""

def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP fictitious play (min-peak formulation) with both additive and 
multiplicative updates, L-BFGS log-sum-exp polishing, spectral methods,
and multi-length exploration to beat 1.5027.

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_min_peak_lp(f, a_init=None):
    """Solve: min t s.t. C(f) @ g <= t, sum(g) = 1, g >= 0. Returns (g, t)."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    if a_init is not None:
        g.value = np.maximum(a_init, 0)
        g.value = g.value / max(g.value.sum(), 1e-15)
    C = build_conv_matrix(f)
    constraints = [
        C @ g <= t,
        cp.sum(g) == 1
    ]
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, warm_start=(a_init is not None),
                       verbose=False)
            if g.value is not None and t.value is not None:
                g_val = np.maximum(g.value, 0)
                ss = g_val.sum()
                if ss > 1e-15:
                    g_val = g_val / ss
                return g_val, float(t.value)
        except Exception:
            continue
    # Fallback: return uniform
    return np.ones(n) / n, np.inf


def solve_max_sum_lp(f, M, a_init=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if a_init is not None:
        g.value = np.maximum(a_init, 0)
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, warm_start=(a_init is not None),
                       verbose=False)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return a_init if a_init is not None else np.ones(n) / n


def lp_fictitious_play_additive(a0, time_limit, step_init=0.4, verbose=True, label="LP-add"):
    """LP fictitious play with additive (exponential) averaging update."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    step = step_init
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        g, t_lp = solve_min_peak_lp(a, a_init=prev_g)
        prev_g = g
        g = g / g.sum()

        a_new = (1 - step) * a + step * g
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        # Adaptive step: decrease when stalled
        if stall_count > 150:
            step = max(0.02, step * 0.6)
            stall_count = 0

        # Periodic restart
        if stall_count > 400:
            rng = np.random.default_rng(iters * 7919)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            step = step_init
            stall_count = 0

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, t_lp={t_lp:.6f}, "
                  f"step={step:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def lp_fictitious_play_multiplicative(a0, time_limit, eta_init=0.5, verbose=True, label="LP-mul"):
    """LP fictitious play with multiplicative (mirror descent) update."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    eta = eta_init
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        g, t_lp = solve_min_peak_lp(a, a_init=prev_g)
        prev_g = g
        g = g / g.sum()

        # Multiplicative update: a_new[i] proportional to a[i] * (g[i]/a[i])^eta
        # = a[i]^(1-eta) * g[i]^eta (geometric mean)
        eps = 1e-20
        a_safe = np.maximum(a, eps)
        g_safe = np.maximum(g, eps)
        a_new = a_safe ** (1 - eta) * g_safe ** eta
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 200:
            eta = max(0.05, eta * 0.5)
            stall_count = 0

        if stall_count > 500:
            rng = np.random.default_rng(iters * 104729)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            eta = eta_init
            stall_count = 0

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, t_lp={t_lp:.6f}, "
                  f"eta={eta:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def lp_fictitious_play_uniform_avg(a0, time_limit, verbose=True, label="LP-uniform"):
    """LP fictitious play with uniform averaging of all past iterates."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    # Running average
    avg = a.copy()
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        g, t_lp = solve_min_peak_lp(a, a_init=prev_g)
        prev_g = g
        g = g / g.sum()

        # Uniform average update
        avg = ((iters - 1) / iters) * avg + (1.0 / iters) * g

        # Use the average as current iterate for next LP
        a = avg.copy()

        score = compute_score(a)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 300:
            # Switch to exponential averaging with small step
            step = 0.1
            g2, _ = solve_min_peak_lp(a, a_init=prev_g)
            a = (1 - step) * a + step * g2
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
            prev_g = None

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, t_lp={t_lp:.6f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, verbose=True):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    z0 = np.log(np.maximum(a, 1e-30))
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        ss = exp_z.sum()
        if ss < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / ss

    def objective_and_grad(z, T):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        obj = 2.0 * n * T * np.exp(lse)

        # Gradient of 2n * T * LSE(conv/T) w.r.t. a
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return obj, grad_z

    z_current = z0.copy()
    for T_val in [100, 300, 1000, 3000, 5000]:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = objective_and_grad(z, T_val)
            return obj

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, g = objective_and_grad(z, T_val)
            return g

        try:
            res = minimize(
                fun=func, x0=z_current, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 100000, 'disp': False}
            )
            a_out = softmax_from_z(res.x)
            score = compute_score(a_out)
            if score < best_score:
                best_score = score
                best_a = a_out.copy()
                z_current = res.x.copy()
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        if verbose:
            print(f"  L-BFGS T={T_val}: best={best_score:.8f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with pairwise transfers and structured search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        any_improved = False

        # Try structured pairs
        n_tries = min(300, n)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            best_local = best_score
            best_t = 0.0
            best_dir = 0

            for direction in [1, -1]:
                src = i if direction == 1 else j
                dst = j if direction == 1 else i
                if a[src] < 1e-15:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5]:
                    t = frac * a[src]
                    a_trial = a.copy()
                    a_trial[src] -= t
                    a_trial[dst] += t
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_local:
                        best_local = sc
                        best_t = t
                        best_dir = direction

            if best_t > 0:
                src = i if best_dir == 1 else j
                dst = j if best_dir == 1 else i
                a[src] -= best_t
                a[dst] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                any_improved = True
                no_improve = 0
                break

        if not any_improved:
            no_improve += 1
            if no_improve > 20:
                a = best_a.copy() + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def spectral_method(a0, time_limit, verbose=True):
    """Spectral flattening via iterative projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power.mean()
        # Vary tau
        tau = 0.2 + 0.3 * (iters % 3) / 3.0
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.3, 3.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Spectral iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def active_set_lp_refinement(a0, time_limit, verbose=True):
    """Solve LP restricting to top-K active constraints for exact local optimum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    conv = fast_convolve(a)
    M = conv.max()

    for K in [20, 50, 100, 200, 500]:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        peak_indices = np.argsort(conv)[-K:]

        g = cp.Variable(n, nonneg=True)
        t = cp.Variable()
        C_full = build_conv_matrix(a)
        C_red = C_full[peak_indices, :]
        constraints = [
            C_red @ g <= t,
            cp.sum(g) == 1
        ]
        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints)

        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None:
                    g_val = np.maximum(g.value, 0)
                    ss = g_val.sum()
                    if ss > 1e-15:
                        g_val = g_val / ss
                    score = compute_score(g_val)
                    if score < best_score:
                        best_score = score
                        best_a = g_val.copy()
                        if verbose:
                            print(f"  Active-set K={K}: score={score:.8f}")
                    break
            except Exception:
                continue

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate using the actual evaluation function and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
