def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def greedy_lp(f, M, order=None):
    """Greedy LP: find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    if order is None:
        order = np.arange(n)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    
    for j in order:
        # Max g[j] such that conv_vals[k] + f[k-j]*g[j] <= M for all k in [j, j+n-1]
        k_start = j
        k_end = j + n  # exclusive
        if k_end > 2*n-1:
            k_end = 2*n-1
        slack = M - conv_vals[k_start:k_end]
        f_slice = f[:k_end-k_start]  # f[k-j] for k in [j, j+n-1]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.min(slack[valid] / f_slice[valid])
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end] += f_slice * max_gj
    
    return g

def geometric_mean_update(f, g, t):

    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s

def multi_peak_reduction(a0, time_limit, verbose=True, seed=99):
    """
    Explicitly target the top peaks for reduction.
    For each peak, identify contributing pairs and try to reduce them.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    
    best_a = a.copy()

    t0 = time.monotonic()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        
        # Get top 5 peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            pk = int(pk)
            # Contributing pairs: (j, pk-j) for valid j
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contribs = np.array([a[j] * a[pk - j] for j in range(js, je + 1)])
            
            if contribs.sum() < 1e-15:
                continue
            
            # Try reducing the top contributing elements
            top_contrib = np.argsort(contribs)[::-1][:3]
            
            for ci in top_contrib:
                j_idx = js + ci
                other_idx = pk - j_idx
                
                if a[j_idx] < 1e-14:
                    continue
                
                # Try reducing a[j_idx] and redistributing to a position
                # that doesn't contribute to this peak
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = a[j_idx] * frac
                    a_trial = a.copy()
                    a_trial[j_idx] -= delta
                    
                    # Redistribute to a position far from the peak
                    # Positions that don't contribute to peak pk: j where pk-j < 0 or pk-j >= n
                    # i.e., j > pk or j < pk-n+1
                    noncontrib = np.concatenate([
                        np.arange(0, max(1, pk - n + 1)),  # j < pk-n+1
                        np.arange(min(n, pk + 1), n)  # j > pk
                    ])
                    noncontrib = noncontrib[noncontrib != j_idx]
                    if len(noncontrib) == 0:
                        continue
                    
                    # Choose the position with smallest a value among non-contributing
                    target = noncontrib[np.argmin(a_trial[noncontrib])]
                    a_trial[target] += delta
                    
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            # Perturb
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and iter_count % 100 == 0:
            print(f"  MP iter={iter_count}, score={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
