"""
Optimized step function search using fast LP (PDLP solver), aggressive adaptive
constraint management, spectral diversification, and multi-phase polishing.
Key improvements over previous: faster LP via PDLP, backtracking line search,
more aggressive constraint sets, spectral flattening between LP phases,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    """Build constraint matrix A where (f@g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_fast(f, M, active_idx, n):
    """Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_idx."""
    if len(active_idx) == 0:
        return None
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solved = False
    try:
        prob.solve(solver=cp.PDP, max_iters=8000, verbose=False)
        if g.value is not None:
            solved = True
    except Exception:
        pass
    
    if not solved:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-9, verbose=False)
        except Exception:
            pass
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() <= 1e-15:
        return None
    return g_val


def lp_optimize_aggressive(f_init, time_limit, initial_k=60, max_k=500, label="lp"):

def spectral_flatten(f_init, n_iters=400, time_limit=80.0, clip_frac=0.4):
    """
    Spectral domain iterative flattening of the auto-convolution.
    Alternates between time domain (non-negativity + normalization)
    and frequency domain (amplitude adjustment to flatten correlation).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        F = rfft(f, fft_size)
        b = irfft(F * F, fft_size)[:size]
        b_max = b.max()
        if b_max < 1e-15:
            break

        # Create target: clip peaks, preserve total
        floor = clip_frac * b_max
        b_target = np.maximum(b, floor)
        # Soft clipping: blend between original and clipped
        b_target = 0.5 * b + 0.5 * b_target
        # Preserve total sum (= 1 for normalized f)
        total = b.sum()
        if b_target.sum() > 1e-15:
            b_target *= total / b_target.sum()

        # Frequency domain: new amplitude from target
        B_fft = rfft(b_target, fft_size)
        amp_new = np.sqrt(np.maximum(B_fft.real, 0.0))

        # Phase preservation
        abs_F = np.abs(F)
        phase = np.ones_like(F)
        mask = abs_F > 1e-14
        phase[mask] = F[mask] / abs_F[mask]
        F_new = amp_new * phase

        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f = f_init.copy()
            break

        f = f_new
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Perturb to escape
                f = project_simplex(f + 0.001 * np.random.default_rng(it).standard_normal(n))
                no_improve = 0

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.0008, n_peaks=40, momentum=0.85):
    """
    Subgradient method on the simplex with momentum for final polishing.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        b_max = b.max()

        # Select active peaks (top n_peaks)
        if n_peaks < len(b):
            thresh = b[np.partition(b, -n_peaks)[-n_peaks]]
            active = np.where(b >= thresh * 0.99)[0]
        else:
            active = np.arange(len(b))

        # Compute subgradient of max_k b[k]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * f[hi:lo - 1:-1] if lo < hi else f[lo:hi + 1]
        if len(active) > 0:
            grad /= len(active)

        # Momentum + projection
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 30:
            lr *= 0.92
            no_improve = 0
        if no_improve > 150:
            f = project_simplex(0.7 * best_f + 0.3 * (best_f + 0.003 * rng.standard_normal(n)))
            v = np.zeros(n)
            no_improve = 0

        if it % 200 == 0:
            print(f"    sg it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.8f} "
                  f"lr={lr:.6e} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def try_different_length(best_seq, target_n, time_limit, label="len"):

    n_old = len(best_seq)

    print(f"\n=== Phase 2: Spectral flatten (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.1, 80.0)
    if t2 > 15:
        f2, obj2 = spectral_flatten(best_seq, n_iters=600, time_limit=t2, clip_frac=0.35)
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [spectral]: {score2:.8f}")
        else:
            print(f"  [spectral] score={score2:.8f}")

    # === Phase 3: Deep LP with more constraints ===
    print(f"\n=== Phase 3: Deep LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 350.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s, initial_k=100, max_k=700")
        f3, obj3 = lp_optimize_aggressive(best_seq, time_limit=t3,
                                           initial_k=100, max_k=700, label="lp3")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp3]: {score3:.8f}")
        else:
            print(f"  [lp3] score={score3:.8f}")

    # === Phase 4: Spectral + LP combo ===
    print(f"\n=== Phase 4: Spectral + LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4_total = min(rem * 0.12, 120.0)
    if t4_total > 30:
        # Variant A: aggressive clipping
        f4a = spectral_flatten(best_seq, n_iters=300, time_limit=t4_total * 0.3, clip_frac=0.2)
        rem4 = deadline - time.monotonic()
        if rem4 > 20:
            f4a_opt, _ = lp_optimize_aggressive(f4a, time_limit=min(rem4 * 0.5, 60.0),
                                                  initial_k=50, max_k=400, label="spec_a")
            s4a = float(evaluate_sequence(f4a_opt.tolist()))
            if np.isfinite(s4a) and s4a < best_score:
                best_score = s4a
                best_seq = f4a_opt.copy()
                print(f"  NEW BEST [spec_a]: {s4a:.8f}")
            else:
                print(f"  [spec_a] score={s4a:.8f}")

        # Variant B: gentle clipping
        rem4 = deadline - time.monotonic()
        if rem4 > 20:
            f4b = spectral_flatten(best_seq, n_iters=300, time_limit=min(rem4 * 0.3, 40.0), clip_frac=0.6)
            rem4 = deadline - time.monotonic()
            if rem4 > 15:
                f4b_opt, _ = lp_optimize_aggressive(f4b, time_limit=min(rem4 * 0.6, 50.0),
                                                     initial_k=50, max_k=400, label="spec_b")
                s4b = float(evaluate_sequence(f4b_opt.tolist()))
                if np.isfinite(s4b) and s4b < best_score:
                    best_score = s4b
                    best_seq = f4b_opt.copy()
                    print(f"  NEW BEST [spec_b]: {s4b:.8f}")
                else:
                    print(f"  [spec_b] score={s4b:.8f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 15:
        f5, obj5 = subgradient_polish(best_seq, time_limit=t5, lr_init=0.0005, n_peaks=50)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg]: {score5:.8f}")
        else:
            print(f"  [sg] score={score5:.8f}")

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Length exploration (best={best_score:.8f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.9, 1.1, 1.2]:
        tl = int(n_base * factor)
        tl = max(512, tl)
        target_lengths.append(tl)
    target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t6 = min(rem * 0.1, 90.0)
        if t6 < 20:
            break
        print(f"  Trying n={target_n}, budget={t6:.0f}s")
        f6 = try_different_length(best_seq, target_n, time_limit=t6, label=f"n{target_n}")
        s6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(s6) and s6 < best_score:
            best_score = s6
            best_seq = f6.copy()
            print(f"  NEW BEST [n{target_n}]: {s6:.8f}")
        else:
            print(f"  [n{target_n}] score={s6:.8f}")

    # === Phase 7: Final LP burst ===
    print(f"\n=== Phase 7: Final LP burst (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.25, 250.0)
    if t7 > 20:
        print(f"  Budget: {t7:.0f}s, initial_k=150, max_k=800")
        f7, obj7 = lp_optimize_aggressive(best_seq, time_limit=t7,
                                           initial_k=150, max_k=800, label="final")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [final]: {score7:.8f}")
        else:
            print(f"  [final] score={score7:.8f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
