def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def max_conv_val(f0, direction, t):
    """Compute max of conv for (1-t)*f0 + t*direction, normalized."""
    f_trial = (1.0 - t) * f0 + t * direction
    f_trial = np.maximum(f_trial, 0.0)
    s = f_trial.sum()
    if s < 1e-15:
        return np.inf
    f_trial = f_trial / s
    return fast_convolve(f_trial).max()


def ternary_line_search(f0, direction, t_max=1.0, iterations=40):
    """Ternary search for optimal t in [0, t_max]."""
    a = 0.0
    b = t_max
    for _ in range(iterations):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = max_conv_val(f0, direction, m1)
        f2 = max_conv_val(f0, direction, m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    t_opt = (a + b) / 2.0
    return t_opt, max_conv_val(f0, direction, t_opt)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_subgradient(f, eps, n_top=300):
    """Compute subgradient of log-sum-exp smoothing of max(f*f) with softmax weights."""
    b = fast_convolve(f)
    n = len(f)
    full_len = 2 * n - 1

    b_max = b.max()
    scaled = (b - b_max) / eps
    # Clamp for numerical stability
    scaled = np.clip(scaled, -100, 100)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(full_len)
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum

    # Restrict to top peaks for efficiency
    if n_top < full_len:
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_new = np.zeros(full_len)
        w_new[top_idx] = w[top_idx]
        ws = w_new.sum()
        if ws > 1e-30:
            w_new = w_new / ws
        w = w_new

    # Compute gradient: grad[i] = 2 * sum_k w[k] * f[k-i] for valid k-i
    active = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2*w[k] * f[k-hi : k-lo+1][::-1]
        segment = f[k - hi:k - lo + 1][::-1].copy()
        grad[lo:hi + 1] += 2.0 * w[k] * segment

    return grad, b_max


def smooth_optimize(f_init, time_limit, eps_init=1.0, eps_min=0.02,
                    lr_init=0.005, n_top=200, label="smooth"):
    """
    Smooth optimization using log-sum-exp gradient with Adam-like updates.
    Temperature anneals from eps_init to eps_min.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    eps = eps_init
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Anneal temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        eps = eps_init * (eps_min / eps_init) ** progress
        eps = max(eps, eps_min)

        grad, b_max = lse_subgradient(f, eps, n_top)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)

        if it % 10 == 0:
            obj = fast_convolve(f_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_new.copy()
                no_improve = 0
                f = f_new.copy()
                lr = min(lr * 1.02, 0.05)
            else:
                no_improve += 1
                f = f_new.copy()
                if no_improve > 8:
                    lr *= 0.9
                    no_improve = 0

        # Periodic diversification
        if it % 200 == 0 and no_improve > 15:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002 * best_f.max() / max(best_f.mean(), 1e-10))
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

        if it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={2.0*n*best_obj:.6f} "
                  f"eps={eps:.4f} lr={lr:.5f} t={elapsed:.0f}s")

    return best_f, 2.0 * n * best_obj


def solve_lp(f, indices, M, n):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])

    try:
        if K <= 600 and n <= 1200:
            prob.solve(solver=cp.ECOS, max_iters=30000)
        else:
            prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def single_coordinate_descent(f_init, n_passes=3):
    """Quick single-pass coordinate descent polish."""
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]

    for _ in range(n_passes):
        for j in range(n):
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_val = f[j] * mult
            f[j] = best_val
            s = f.sum()
            if s > 1e-12:
                f = f / s

    return f, 2.0 * n * best_obj

        n_b = len(best_seq)
        print(f"  Smooth n={n_b}, budget={t1:.0f}s")
        f_res, _ = smooth_optimize(
            best_seq, time_limit=t1, eps_init=0.8, eps_min=0.03,
            lr_init=0.004, n_top=200, label="sm1")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [smooth]: {real_score:.6f}")

    # ============================================================
    # Phase 2: LP with convex line search
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP + ternary search (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.28, 400.0)

    if t2 > 30:
        n_b = len(best_seq)
        print(f"  LP n={n_b}, budget={t2:.0f}s")
        f_res, _ = lp_optimize(
            best_seq, time_limit=t2, top_k_init=80, top_k_max=min(600, 2*n_b-1),
            label="lp1")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [lp]: {real_score:.6f}")

    # ============================================================
    # Phase 3: Multi-n exploration
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-n exploration (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 350.0)

    if t3 > 40:
        n_base = len(best_seq)
        n_candidates = []
        for factor in [0.80, 0.90, 1.0, 1.10, 1.25, 1.40]:
            n_c = int(n_base * factor)
            n_c = max(512, min(2000, n_c))
            if n_c not in n_candidates:
                n_candidates.append(n_c)
        n_candidates.sort()

        per_n = t3 / len(n_candidates)
        for n_trial in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            alloc = min(per_n, rem * 0.85)
            if alloc < 15:
                break

            f_init = interp_sequence(best_seq, n_trial)
            # Quick smooth + LP combo
            t_smooth = alloc * 0.4
            t_lp = alloc * 0.6

            print(f"  Multi-n: n={n_trial}, budget={alloc:.0f}s")

            f_res, _ = smooth_optimize(
                f_init, time_limit=t_smooth, eps_init=0.5, eps_min=0.05,
                lr_init=0.003, n_top=150, label=f"sn{n_trial}")

            rem_inner = deadline - time.monotonic()
            if rem_inner > 15:
                t_lp_actual = min(t_lp, rem_inner * 0.85)
                if t_lp_actual > 10:
                    f_res2, _ = lp_optimize(
                        f_res, time_limit=t_lp_actual,
                        top_k_init=50, top_k_max=min(300, 2*n_trial-1),
                        label=f"ln{n_trial}")
                    f_res = f_res2

            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [n={n_trial}]: {real_score:.6f}")

    # ============================================================
    # Phase 4: Deep continued LP on current best
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep LP (best: {best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)

    if t4 > 20:
        n_b = len(best_seq)
        print(f"  Deep LP n={n_b}, budget={t4:.0f}s")
        f_res, _ = lp_optimize(
            best_seq, time_limit=t4, top_k_init=150,
            top_k_max=min(800, 2*n_b-1), label="deep")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [deep]: {real_score:.6f}")

    # ============================================================
    # Phase 5: Final smooth polish + coordinate descent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 150.0)

    if t5 > 15:
        n_b = len(best_seq)
        print(f"  Final smooth n={n_b}, budget={t5:.0f}s")
        f_res, _ = smooth_optimize(
            best_seq, time_limit=t5, eps_init=0.3, eps_min=0.01,
            lr_init=0.002, n_top=300, label="final")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [final smooth]: {real_score:.6f}")

    # Coordinate descent polish
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"  CD polish, budget={rem:.0f}s")
        f_res, _ = single_coordinate_descent(best_seq, n_passes=2)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [CD]: {real_score:.6f}")

    # ============================================================
    # Final verification
    # ============================================================
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy() if parent is not None else np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
