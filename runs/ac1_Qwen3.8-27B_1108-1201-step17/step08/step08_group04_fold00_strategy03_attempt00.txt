                new_individual[k] *= np.random.uniform(0.85, 1.15)
            else:
                new_individual[k] *= np.random.uniform(0.95, 1.05)
        population.append(new_individual)
        num_evaluations += 1
        if num_evaluations > 3:
            break
    seq_set = sorted(population, key=lambda x: x.tolist().index(project_simplex_clip(x)))
    best_seq = None
    current = np.zeros(len(seq_set))
    for candidate in seq_set:
        if not temp_negative_check(candidate) and number_negative(candidate) == 0:
            score_val = float(evaluate_sequence(candidate.tolist()))
        else:
            score_val = float('inf')
        if score_val < 0 and best_seq is None:
            best_seq = candidate
    if best_seq is None:
        best_seq = np.random.uniform(size=512)
    try:
        deadline = time.time() + timeout
        cores = 1 if 'ECOS' not in globals() else 2
        for i in range(cores*10):
            seq = np.random.uniform(size=512)
            conv_result = fast_convolve_fft(seq.tolist())
            max_peak = np.argmax(conv_result)
            total_sum = np.sum(seq)
            score_val = 2*len(seq)*max_peak / (total_sum**2)
            if score_val < 0:
                best_seq = seq
    except KeyboardInterrupt:
        pass
    finally:
        return [float(i) for i in best_seq]

def evaluate_sequence(seq):
    seq_array = np.array(seq)
    if seq_array.shape is None:
        return float('inf')
    convolve_seq = np.convolve(seq_array, seq_array, mode='full')
    max_conv = np.max(convolve_seq)
    total_sequence = np.sum(seq_array)
    sigma_total = total_sequence ** 2
    filter = lambda a: float('inf') if max_conv > 1000 else max_conv/((total_sequence)**2) if sigma_total < (total_sequence)**2 else float('inf')
    return filter(max_conv) if sigma_total > (total_sequence)**2 else float('inf')
