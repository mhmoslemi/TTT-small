def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):

    n = len(a)

    """Cross-convolution f*g via FFT."""

            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"|A|={len(active_idx)} st={stuck} t={elapsed:.0f}s")
    
    return best_f, best_obj


def random_blend_diversify(f_best, n_trials=5, rng_seed=0):
    """Generate diverse candidates by blending with random sequences."""
    rng = np.random.default_rng(rng_seed)
    n = len(f_best)
    candidates = []
    
    for i in range(n_trials):
        # Different blend strategies
        blend_type = i % 3
        if blend_type == 0:
            # Blend with uniform
            w = 0.1 + 0.15 * i
            f = (1 - w) * f_best + w * np.ones(n) / n
        elif blend_type == 1:
            # Add structured perturbation (low frequency)
            freqs = np.fft.rfftfreq(n)
            phase = rng.standard_normal(len(freqs))
            amp = np.zeros(len(freqs))
            amp[1:5] = 0.05 / np.sqrt(np.arange(1, 5))
            pert = irfft(amp * phase, n)
            f = f_best + pert
        else:
            # Random block scaling
            f = f_best.copy()
            n_blocks = 8
            bs = n // n_blocks
            for bi in range(n_blocks):
                f[bi*bs:(bi+1)*bs] *= 0.7 + 0.6 * rng.random()
        
        f = normalize(f)
        candidates.append(f)
    
    return candidates


def multi_length_search(f_best, time_budget, n_base):
    """Try different lengths around n_base."""
    n_values = set()
    for factor in [0.8, 0.9, 1.0, 1.1, 1.2]:
        nv = int(n_base * factor)
        nv = max(256, min(3000, nv))
        n_values.add(nv)
    n_values = sorted(n_values)
    
    results = []
    per_budget = time_budget / len(n_values)
    
    for i, nv in enumerate(n_values):
        rem = time_budget - i * per_budget
        if rem < 20:
            break
        t_alloc = min(per_budget * 1.3, rem - 10)
        if t_alloc < 15:
            break
        
        # Interpolate from best
        old_x = np.linspace(0, 1, len(f_best))
        new_x = np.linspace(0, 1, nv)
        f_init = np.interp(new_x, old_x, f_best)
        f_init = normalize(f_init)
        
        # Add small random perturbation for diversity
        if nv != len(f_best):
            rng = np.random.default_rng(nv)
            f_init = normalize(f_init + rng.standard_normal(nv) * 0.01)
        
        print(f"  Multi-len: n={nv}, budget={t_alloc:.0f}s")
        f_opt, obj = lp_optimize(f_init, time_limit=t_alloc,
                                  initial_k=40, max_k=min(400, 2*nv),
                                  label=f"ml{nv}")
        score = 2.0 * nv * obj
        results.append((score, f_opt, nv))
        print(f"    score={score:.6f}")
    
    results.sort(key=lambda x: x[0])
    return results


def coordinate_descent(f_init, time_limit, n_sweeps=30):
    """Simple coordinate descent: adjust each element up/down multiplicatively."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 500)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-12:
                continue
            orig = f[j]
            for mult in [0.7, 0.85, 1.15, 1.3]:
                f[j] = orig * mult
                f = normalize(f)
                obj = fast_convolve(f).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    improved = True
                    break
            # Keep best
            f[j] = orig
            if improved:
                # Redo with best mult
                for mult in [0.7, 0.85, 1.15, 1.3]:
                    f[j] = orig * mult
                    f = normalize(f)
                    obj = fast_convolve(f).max()
                    if obj <= best_obj + 1e-13:
                        break
    return f, best_obj

    Main search: LP cutting-plane optimization with adaptive constraints,
    multi-length exploration, random-blend diversification, and coordinate polish.
    """

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:

        parent = np.ones(1024) / 1024

    best_seq = parent.copy()

    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
