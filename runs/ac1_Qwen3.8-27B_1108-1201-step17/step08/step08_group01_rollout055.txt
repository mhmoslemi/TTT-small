        grad = compute_grad_from_weights(w, y)

        # Step size: Lipschitz const of grad is ~ 4/(eps) * max|a| ~ 4/(eps*n)
        # So lr ~ eps*n/4 is safe. Use a fraction for stability.
        lr = min(eps * n * 0.25, 2.0)

        a_new = y - lr * grad
        a_new = simplex_project(a_new)

        # Evaluate
        conv_a = fast_convolve(a_new)
        score = 2.0 * n * conv_a.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Anneal epsilon
        eps = max(eps_min, eps * eps_decay)

        if no_improve > 500:
            break

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} eps={eps:.6f}")

    return best_a, best_score

def coord_descent_optimal(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent with bisection-based optimal step size.
    For each coordinate, finds the delta that minimizes the new max convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            a_slice = a[:affected_len]  # a[k-i] for k = i, i+1, ..., i+n-1

            # Find optimal delta using bisection on the subgradient
            # f(delta) = max(unaff_max, max_k (affected[k] + 2*delta*a_slice[k] + [k==2i-k_start]*delta^2))
            # We want to minimize f(delta) subject to a[i] + delta >= 0

            # Check if current value is already optimal (gradient points "inward")
            cur_max = float(affected.max())
            if cur_max <= unaff_max + 1e-15:
                # Already at or below unaffected max; try small positive delta
                # to potentially help other coordinates
                best_delta = 0.0
                best_val = cur_max
                for trial_d in [0.5/n, 1.0/n, 2.0/n, 5.0/n, 10.0/n]:
                    if a[i] + trial_d < 0:
                        break
                    new_aff = affected + 2.0 * trial_d * a_slice
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_aff = new_aff.copy()
                        new_aff[idx] += trial_d * trial_d
                    new_max = max(unaff_max, float(new_aff.max()))
                    if new_max < best_val:
                        best_val = new_max
                        best_delta = trial_d
                if best_delta != 0.0 and best_val < M - 1e-14:
                    a[i] += best_delta
                    sum_a_change = best_delta
                    changes = 2.0 * best_delta * a_slice
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        changes = changes.copy()
                        changes[idx] += best_delta * best_delta
                    conv[k_start:k_end + 1] += changes
                    M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                    cur_score = 2.0 * n * M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True
                continue

            # Current max > unaff_max, need to reduce
            # Bisection: find delta where the max affected entry crosses unaff_max
            # or where subgradient = 0

            # The function f(delta) is convex in delta (for delta > -a[i])
            # Find the interval [lo, hi] containing the minimum
            lo = -a[i]  # delta that makes a[i] = 0
            hi = 0.0

            # Check if f(0) > f(lo) (i.e., decreasing a[i] helps)
            # At delta = 0: f = cur_max
            # At delta = lo: a[i] = 0, affected entries become affected - 2*a[i]*a_slice
            # (plus the quadratic correction)

            # Try bisection
            best_delta = 0.0
            best_f = cur_max

            # Quick scan with a few candidate values
            candidates = [0.0]
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * a[i])
            # Also try values that would equalize the top entries
            top_entries = np.where(affected > cur_max * 0.9)[0]
            for k_idx in top_entries[:5]:
                if a_slice[k_idx] > 1e-15:
                    d = (unaff_max - affected[k_idx]) / (2.0 * a_slice[k_idx])
                    d = max(d, -a[i] * 0.99)
                    d = min(d, 0.0)
                    candidates.append(d)

            for d in candidates:
                if a[i] + d < -1e-15:
                    continue
                new_aff = affected + 2.0 * d * a_slice
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * d * a[i] + d * d
                val = max(unaff_max, float(new_aff.max()))
                if val < best_f:
                    best_f = val
                    best_delta = d

            # Bisection refinement around best_delta
            if best_f < cur_max - 1e-14:
                # Refine with bisection
                mid = best_delta
                search_lo = best_delta - abs(a[i]) * 0.1
                search_hi = best_delta + abs(a[i]) * 0.1
                search_lo = max(search_lo, -a[i] * 0.99)

                for _ in range(8):
                    mid_left = (search_lo + mid) / 2.0
                    mid_right = (mid + search_hi) / 2.0

                    # Evaluate at mid_left
                    new_aff_l = affected + 2.0 * mid_left * a_slice
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_aff_l = new_aff_l.copy()
                        new_aff_l[idx] = conv[2 * i] + 2.0 * mid_left * a[i] + mid_left * mid_left
                    val_l = max(unaff_max, float(new_aff_l.max()))

                    # Evaluate at mid_right
                    new_aff_r = affected + 2.0 * mid_right * a_slice
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_aff_r = new_aff_r.copy()
                        new_aff_r[idx] = conv[2 * i] + 2.0 * mid_right * a[i] + mid_right * mid_right
                    val_r = max(unaff_max, float(new_aff_r.max()))

                    if val_l < val_r:
                        search_hi = mid
                    else:
                        search_lo = mid
                    mid = (search_lo + search_hi) / 2.0

                # Evaluate at refined mid
                if a[i] + mid >= -1e-15:
                    new_aff_m = affected + 2.0 * mid * a_slice
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_aff_m = new_aff_m.copy()
                        new_aff_m[idx] = conv[2 * i] + 2.0 * mid * a[i] + mid * mid
                    val_m = max(unaff_max, float(new_aff_m.max()))
                    if val_m < best_f:
                        best_f = val_m
                        best_delta = mid

                if best_f < M - 1e-14:
                    a[i] += best_delta
                    a[i] = max(a[i], 0.0)
                    changes = 2.0 * best_delta * a_slice
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        changes = changes.copy()
                        changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                    conv[k_start:k_end + 1] += changes
                    M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                    cur_score = 2.0 * n * M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            a_slice = a[:affected_len]

            best_delta = 0.0
            best_new_max = float(affected.max())

            # Try small multiplicative changes
            for factor in [0.995, 0.99, 0.985, 0.98, 0.97, 1.01, 1.02, 1.03, 1.05, 1.07]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_slice
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                changes = 2.0 * best_delta * a_slice
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    _lp_solver = None
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given a (nonneg, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build constraint matrix: (2n-1) x n, T[k,j] = a_fixed[k-j] if valid
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.PDLP, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lp_guided_restart(a0, solver, n_alpha=5):
    """
    Solve LP best response, then interpolate back to a0 at various alphas.
    Returns the best interpolated candidate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    b_star, M_lp = solve_lp_best_response(a, solver)
    if b_star is None:
        return a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for alpha in np.linspace(0.1, 0.9, n_alpha):
        cand = (1 - alpha) * a + alpha * b_star
        cand = np.maximum(cand, 0)
        sc = cand.sum()
        if sc < 1e-10:
            continue
        cand = cand / sc
        score = 2.0 * n * fast_convolve(cand).max()
        if score < best_score:
            best_score = score
            best_a = cand.copy()

    return best_a

def init_u_shape(n, alpha=1.0, beta=0.5):
    """
    U-shaped initialization: a_j ∝ 1 + alpha * cos(pi * j / (n-1))^beta
    Higher values at edges to compensate for convolution boundary effects.
    """
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + alpha * (np.cos(np.pi * t) ** 2) ** beta
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_power_u(n, p=2.0):
    """U-shape with power: a_j ∝ (j/(n-1))^p + ((n-1-j)/(n-1))^p + 0.1"""
    if n == 1:
        return np.array([1.0])
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = t ** p + (1.0 - t) ** p + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_fourier(n, rng):
    """Random Fourier superposition initialization."""
    n_harm = int(rng.integers(3, 15))
    amps = rng.random(n_harm) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_noise(n, base, rng, sigma=0.05):
    """Add noise to a base sequence."""
    a = base.copy() + rng.standard_normal(n) * sigma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse U-shaped initializations + quick coordinate descent
    Phase 2: Intensive coordinate descent on best candidates
    Phase 3: LP-guided restarts for global escape
    Phase 4: FISTA smoothing pass
    Phase 5: Fine polishing
    Phase 6: Alt-length exploration
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + quick CD ===
    print("\n=== Phase 1: Diverse Init + Quick CD ===")
    t1_start = time.monotonic()
    candidates = []

    n_values = [768, 1024, 1280, 1536]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    for n in n_values:
        elapsed = time.monotonic() - t1_start
        if elapsed > 180:
            break

        # Generate multiple initializations
        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('u1', init_u_shape(n, alpha=1.5, beta=0.5)))
        inits.append(('u2', init_u_shape(n, alpha=2.5, beta=0.7)))
        inits.append(('u3', init_u_shape(n, alpha=3.5, beta=1.0)))
        inits.append(('pu1', init_power_u(n, p=1.5)))
        inits.append(('pu2', init_power_u(n, p=2.5)))
        inits.append(('fourier', init_fourier(n, rng)))
        if parent is not None:
            inits.append(('noise', init_noise(n, interpolate_seq(parent, n), rng, 0.05)))

        for itype, icand in inits:
            t_left = 180.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(3.0, min(10.0, t_left / (len(inits) + 1)))
            a_opt, s_opt = coord_descent_optimal(icand, t_alloc, label=f"p1_n{n}_{itype}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, top score: {candidates[0][0]:.6f}")
    for s_c, a_c, n_c, it_c in candidates[:3]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: Intensive CD on top candidates ===
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        current = best_seq.copy()
        for rnd in range(4):
            rem = deadline - time.monotonic() - 500
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  CD round {rnd+1}, n={len(current)}, budget={t_alloc:.0f}s")
            a_opt, s_opt = coord_descent_optimal(current, t_alloc, verbose=True, label=f"p2_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_cd_r{rnd}")
            current = best_seq.copy()

    # === Phase 3: LP-guided restarts ===
    print(f"\n=== Phase 3: LP-Guided Restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        for rnd in range(3):
            rem = deadline - time.monotonic() - 350
            if rem < 60:
                break

            # Solve LP and interpolate
            a_restarted = lp_guided_restart(best_seq, solver, n_alpha=7)
            best_seq, best_score = check_and_update(a_restarted, best_seq, best_score, f"lp_restart_r{rnd}")

            # Run CD from the restarted point
            t_cd = min(rem * 0.4, 120.0)
            a_opt, s_opt = coord_descent_optimal(a_restarted, t_cd, verbose=True, label=f"lp_cd_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"lp_cd_r{rnd}")

            # Also try FISTA from the LP solution
            rem2 = deadline - time.monotonic() - 200
            if rem2 > 30:

                a_fista, s_fista = fista_smoothed(a_restarted, t_fista, verbose=True, label=f"lp_fista_r{rnd}")
                best_seq, best_score = check_and_update(a_fista, best_seq, best_score, f"lp_fista_r{rnd}")

    # === Phase 4: FISTA pass on best ===
    print(f"\n=== Phase 4: FISTA Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_fista = min(rem * 0.6, 100.0)
            a_fista, s_fista = fista_smoothed(best_seq, t_fista, eps_init=0.001, verbose=True, label="fista_main")
            best_seq, best_score = check_and_update(a_fista, best_seq, best_score, "fista_main")

            # CD from FISTA result
            rem2 = deadline - time.monotonic() - 50
            if rem2 > 15:
                t_cd2 = min(rem2 * 0.6, 60.0)
                a_cd2, _ = coord_descent_optimal(a_fista, t_cd2, label="fista_cd")
                best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "fista_cd")

    # === Phase 5: Fine polishing ===
    print(f"\n=== Phase 5: Fine Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_fine = min(rem * 0.6, 80.0)
            a_fine, _ = fine_coord_descent(best_seq, t_fine, verbose=True, label="fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

            # Another intensive CD
            rem2 = deadline - time.monotonic() - 20
            if rem2 > 10:
                a_cd3, _ = coord_descent_optimal(best_seq, min(rem2 * 0.7, 50.0), label="final_cd")
                best_seq, best_score = check_and_update(a_cd3, best_seq, best_score, "final_cd")

    # === Phase 6: Alt-length exploration ===
    print(f"\n=== Phase 6: Alt-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            alt_ns = []
            for delta_n in [-128, -64, 64, 128, 256]:
                alt = n_best + delta_n
                if 512 <= alt <= 2048 and alt != n_best:
                    alt_ns.append(alt)

            for alt_n in alt_ns[:2]:
                rem2 = deadline - time.monotonic() - 15
                if rem2 < 10:
                    break
                init_alt = interpolate_seq(best_seq, alt_n)
                t_alt = min(rem2 * 0.6, 40.0)
                a_alt, _ = coord_descent_optimal(init_alt, t_alt, label=f"alt{alt_n}")
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = init_u_shape(1024, alpha=2.0, beta=0.7)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
