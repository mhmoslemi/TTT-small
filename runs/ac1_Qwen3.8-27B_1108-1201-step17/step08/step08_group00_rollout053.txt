"""

- Direct LP solving with best available solver (GLOP/MOSEK/GUROBI)
- Nesterov-accelerated alternating minimization with symmetry

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def symmetrize(a):
    """Enforce palindromic symmetry: a_j = a_{n-1-j}."""
    return 0.5 * (a + a[::-1])


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find the fastest working LP solver."""
    global _sol
    if '_sol' not in dir():
        _sol = None
    if _sol is not None:
        return _sol
    for solver_name in ['GLOP', 'GUROBI', 'MOSEK', 'APPSP', 'CLARABEL', 'ECOS', 'SCS', 'PDLP']:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x[:5] >= 0.1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _sol = solver_name
                print(f"LP solver: {solver_name} ({time.monotonic()-t0:.3f}s)")
                return solver_name
        except Exception:
            continue
    _sol = 'ECOS'
    return _sol


def solve_lp_response(a_fixed, solver=None):
    """Given fixed a (sum=1), solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build T matrix: (T @ b)_k = sum_j a_fixed[k-j] * b_j
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.ECOS)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=50000, use_symmetry=True):
    """Nesterov-accelerated alternating LP with symmetry."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if use_symmetry:
        a = symmetrize(a)
    
    y = a.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    it = 0
    
    while it < n_max_iters:
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_response(y, solver)
        if b_new is None:
            it += 1
            continue
        
        if use_symmetry:
            b_new = symmetrize(b_new)
        b_new = normalize(b_new)
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        it += 1
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 10:
                rng = np.random.default_rng(42 + it)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                if use_symmetry:
                    y = symmetrize(y)
                y = y / y.sum()
                stall = 0
            print(f"    NL it={it} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.1, maxiter=8000, use_symmetry=True):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if use_symmetry:
        a = symmetrize(a)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            if use_symmetry:
                a = symmetrize(a)
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            if use_symmetry:
                a = symmetrize(a)
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            g = a * (grad_a - S)
            if use_symmetry:
                g = symmetrize(g)
            return g
        return jac
    
    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_final = softmax_from_z(z)
    if use_symmetry:
        a_final = symmetrize(a_final)
    a_final = normalize(a_final)
    return a_final


def frank_wolfe(a0, time_limit, n_steps=50000, use_symmetry=True):
    """Frank-Wolfe (conditional gradient) on the probability simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if use_symmetry:
        a = symmetrize(a)
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Subgradient of max at peak_k
        g = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[peak_k - idx]
        if use_symmetry:
            g = symmetrize(g)
        
        # FW: find vertex that minimizes g^T v (i.e., argmin g_i)
        v_idx = int(np.argmin(g))
        v = np.zeros(n)
        v[v_idx] = 1.0
        
        # Symmetrize the vertex direction
        if use_symmetry:
            v = symmetrize(v)
            v = v / v.sum()
        
        # Step size: gamma = 2/(step+2) for FW
        gamma = 2.0 / (step + 2)
        a_new = (1 - gamma) * a + gamma * v
        a_new = normalize(a_new)
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=100000, use_symmetry=True):
    """Coordinate descent targeting peak reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if use_symmetry:
        a = symmetrize(a)
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        top_idx = np.argsort(contribs)[::-1][:5]
        improved = False
        
        for ci in top_idx:
            i = idx[ci]
            if a[i] < 1e-12:
                continue
            for factor in [0.99, 0.98, 0.97, 0.95, 0.93]:
                a_trial = a.copy()
                diff = a[i] * (1 - factor)
                if diff < 1e-15:
                    continue
                a_trial[i] = a[i] * factor
                # Redistribute to others
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = normalize(a_trial)
                if use_symmetry:
                    a_trial = symmetrize(a_trial)
                    a_trial = normalize(a_trial)
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pair reduction
            if len(top_idx) >= 2:
                i1 = idx[top_idx[0]]
                j1 = peak_k - i1
                if 0 <= j1 < n and i1 != j1 and a[j1] > 1e-12:
                    a_trial = a.copy()
                    a_trial[i1] *= 0.97
                    a_trial[j1] *= 0.97
                    a_trial = normalize(a_trial)
                    if use_symmetry:
                        a_trial = symmetrize(a_trial)
                        a_trial = normalize(a_trial)
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
        
        if not improved:
            # Log-space perturbation
            z = np.log(np.maximum(a, 1e-300))
            rng = np.random.default_rng(step)
            z_noise = z + 0.01 * rng.standard_normal(n)
            a_trial = softmax_from_z(z_noise)
            if use_symmetry:
                a_trial = symmetrize(a_trial)
            a_trial = normalize(a_trial)
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                if step > 100:
                    break
    
    return best_a, best_score


def structured_init(n, rng, method=0):
    """Create structured symmetric initial sequence."""
    if method == 0:  # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x - np.pi))
        a = a ** 1.5
    elif method == 1:  # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 1.5)
    elif method == 2:  # Sine squared
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif method == 3:  # Parzen
        x = np.linspace(-1, 1, n)
        a = np.where(np.abs(x) < 0.5,
                     1 - 6 * x**2 * (1 - np.abs(x)),
                     2 * (1 - np.abs(x))**3)
        a = np.maximum(a, 0)
    elif method == 4:  # Blackman-like
        x = np.linspace(0, 1, n, endpoint=False)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    elif method == 5:  # Tukey
        x = np.linspace(-1, 1, n)
        alpha = 0.5
        a = np.ones(n)
        mask = np.abs(x) > (1 - alpha) / 2
        x_edge = (np.abs(x[mask]) - (1 - alpha) / 2) / (alpha / 2)
        a[mask] = 0.5 * (1 + np.cos(np.pi * x_edge))
    else:
        a = np.ones(n) * (0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n)))
    
    a = np.maximum(a, 0)
    a = symmetrize(a)
    return normalize(a)


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    f_interp = symmetrize(f_interp)
    return normalize(f_interp)

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = symmetrize(p) / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: Extended Nesterov LP (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t_alloc, solver=solver, n_max_iters=100000, use_symmetry=True)
        try_update(a_lp, "nesterov_lp_1")
    
    # === Phase 2: L-BFGS LogSumExp refinement ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  L-BFGS budget={t_alloc:.0f}s")
        a_lb = lbgfs_logsumexp(best_seq, t_alloc, T_start=100.0, T_end=0.1,
                                maxiter=10000, use_symmetry=True)
        try_update(a_lb, "lbgfs_1")
    
    # === Phase 3: Second Nesterov LP (continue convergence) ===
    print(f"\n=== Phase 3: Nesterov LP r2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp(best_seq, t_alloc, solver=solver, n_max_iters=80000, use_symmetry=True)
        try_update(a_lp2, "nesterov_lp_2")
    
    # === Phase 4: Frank-Wolfe ===
    print(f"\n=== Phase 4: Frank-Wolfe (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 80.0)
    if t_alloc > 10:
        a_fw, _ = frank_wolfe(best_seq, t_alloc, n_steps=100000, use_symmetry=True)
        try_update(a_fw, "frank_wolfe")
    
    # === Phase 5: Peak coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 80.0)
    if t_alloc > 10:
        a_cd, _ = peak_coordinate_descent(best_seq, t_alloc, n_steps=200000, use_symmetry=True)
        try_update(a_cd, "coord_desc")
    
    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.8f}) ===")
    alt_lengths = set()
    for mult in [1.25, 1.5, 2.0, 0.8, 0.65]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 5120 and alt_n != n_best:
            alt_lengths.add(alt_n)
    # Also try specific lengths known to work well
    alt_lengths.update([768, 1024, 1280, 1536])
    alt_lengths = sorted(alt_lengths)
    
    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.25, 200.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        
        init_alt = interpolate_seq(best_seq, alt_n)
        
        t_lp = t_total * 0.70
        a_alt, _ = nesterov_lp(init_alt, t_lp, solver=solver, n_max_iters=50000, use_symmetry=True)
        try_update(a_alt, f"lp_n{alt_n}")
        
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lb = min(rem * 0.3, 60.0)
            if t_lb > 10:
                a_alt2 = lbgfs_logsumexp(best_seq, t_lb, T_start=30.0, T_end=0.5,
                                          maxiter=4000, use_symmetry=True)
                try_update(a_alt2, f"lbgfs_n{alt_n}")
    
    # === Phase 7: Multi-init exploration ===
    print(f"\n=== Phase 7: Multi-Init (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for init_method in [1, 3, 4]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.3, 80.0)
            if t_sub < 10:
                break
            init_new = structured_init(n_cur, rng, init_method)
            a_mi, _ = nesterov_lp(init_new, t_sub, solver=solver, n_max_iters=20000, use_symmetry=True)
            try_update(a_mi, f"multi_init_{init_method}")
    
    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 60.0)
        a_fin = lbgfs_logsumexp(best_seq, t_fin, T_start=5.0, T_end=0.05,
                                 maxiter=5000, use_symmetry=True)
        try_update(a_fin, "final_lbgfs")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.4, 40.0)
        a_cd2, _ = peak_coordinate_descent(best_seq, t_cd, n_steps=50000, use_symmetry=True)
        try_update(a_cd2, "final_cd")
    
    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.5, 30.0)
        a_lp_fin, _ = nesterov_lp(best_seq, t_lp_fin, solver=solver, n_max_iters=5000, use_symmetry=True)
        try_update(a_lp_fin, "final_lp")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
