"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating optimization
with Nesterov acceleration, multi-length search, and L-BFGS LogSumExp refinement.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.5, maxiter=5000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for stage_T in np.geomspace(T_start, T_end, 5):
        if time.monotonic() - t0 > time_limit:
            break
            
        def func(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)

def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method for large sequences where LP is too expensive."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find active constraints (peaks above threshold)
        threshold = max_b - 0.05 * max_b
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Update in log-space (mirror descent)
        z = z_from_a(a)
        z = z - lr * g
        a_new = softmax_from_z(z)
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            lr *= 0.9
            no_improve = 0
        lr = max(0.01, 0.05 / (1.0 + step * 0.001))

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating optimization
with Nesterov acceleration, multi-length exploration, L-BFGS LogSumExp refinement,
and subgradient descent for longer sequences. Key improvements: longer sequences
(up to 4096), more aggressive time allocation to LP phase, better length search,

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:
        solver = get_lp_solver()

                    break

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)

    if s < 1e-15:
        return None, None

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.3, maxiter=6000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

            a = softmax_from_z(z)
            b = fast_convolve(a)

            return a * (grad_a - S)

                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 100,

        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        near_thresh = peak_val - 0.04 * peak_val
        near_peaks = np.where(b_work >= near_thresh)[0]

        # Find top contributors
        contrib_count = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:5]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:10]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:4]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            a_trial = a.copy()
            for i in top_coords[:3]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.97
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely using the provided evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def do_try_update(best_state, a, label):
    """Update best state if a gives better score. best_state is [seq, score]."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return best_state
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state[1]:
        best_state[0] = a.copy()
        best_state[1] = sc
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    return best_state


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with multi-length search:
    1. Load parent, run LP alternation at parent length
    2. L-BFGS polish
    3. Explore longer lengths (interpolated + optimized)
    4. Subgradient for very long sequences
    5. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = [None, float('inf')]
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state = [parent.copy(), p_score]
    except Exception:
        pass

    if best_state[0] is None:
        n_init = 1024
        x = np.linspace(0, 1, n_init)
        init_seq = np.sin(np.pi * x) ** 2
        init_seq = init_seq / init_seq.sum()
        init_score = evaluate_safe(init_seq)
        print(f"Default init: n={n_init}, score={init_score:.6f}")
        best_state = [init_seq, init_score]

    n_best = len(best_state[0])
    print(f"\nStarting optimization from n={n_best}, score={best_state[1]:.6f}")

    # === Phase 1: Extended LP at current length ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_state[0], t_alloc, solver=solver)
        best_state = do_try_update(best_state, a_lp, "lp_phase1")

    # === Phase 2: L-BFGS polish ===
    print(f"\n=== Phase 2: L-BFGS polish ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        a_lb = lbgfs_refine(best_state[0], t_alloc, T_start=80.0, T_end=0.3, maxiter=6000)
        best_state = do_try_update(best_state, a_lb, "lbgfs_phase2")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_state[0], t_alloc, n_steps=8000)
        best_state = do_try_update(best_state, a_ps, "peak_shave")

    # === Phase 4: Explore longer lengths ===
    print(f"\n=== Phase 4: Longer Length Search ===")
    # Generate candidate lengths: prioritize near-doubling and powers of 2
    n_cur = len(best_state[0])
    candidate_lengths = []
    for target in [int(n_cur * 1.2), int(n_cur * 1.5), int(n_cur * 1.75),
                   int(n_cur * 2.0), 1024, 1280, 1536, 2048, 2560, 3072, 4096]:
        if target > n_cur and target <= 4096 and target >= 512 and target % 32 == 0:
            candidate_lengths.append(target)
    candidate_lengths = sorted(set(candidate_lengths))

    # Prioritize: 2x first, then 1.5x, then larger
    def length_priority(ln):
        diff_2x = abs(ln - 2 * n_cur)
        diff_15x = abs(ln - 1.5 * n_cur)
        return min(diff_2x, 0.7 * diff_15x)

    candidate_lengths.sort(key=length_priority)

    for alt_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break

        # Allocate time based on proximity to 2x
        prox = 1.0 / (1.0 + abs(alt_n - 2 * n_cur) / max(1, n_cur))
        t_total = min(rem * 0.22 * prox, 200.0)
        if t_total < 30:
            t_total = min(rem * 0.12, 80.0)

        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state[0], alt_n)

        if alt_n <= 2048:
            t_lp = t_total * 0.7
            a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver)
            best_state = do_try_update(best_state, a_alt, f"lp_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 20 and best_state[0] is not None and len(best_state[0]) == alt_n:
                t_lb = min(rem * 0.25, 50.0)
                if t_lb > 15:
                    a_alt2 = lbgfs_refine(best_state[0], t_lb, T_start=30.0, T_end=0.5, maxiter=3000)
                    best_state = do_try_update(best_state, a_alt2, f"lbgfs_n{alt_n}")
        else:
            a_alt, _ = subgradient_refine(init_alt, t_total * 0.6, n_steps=40000)
            best_state = do_try_update(best_state, a_alt, f"subgrad_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 15 and best_state[0] is not None:
                t_lb = min(rem * 0.3, 50.0)
                if t_lb > 10:
                    a_alt2 = lbgfs_refine(best_state[0], t_lb, T_start=20.0, T_end=1.0, maxiter=2000)
                    best_state = do_try_update(best_state, a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 5: Perturbation + re-optimization ===
    print(f"\n=== Phase 5: Perturbation Escape ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pert = min(rem * 0.20, 150.0)
        n_cur = len(best_state[0])

        # Try structured perturbation
        for noise_scale in [0.03, 0.08]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state[0] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(t_pert * 0.5, rem * 0.4)
            if t_sub > 15:
                if len(perturbed) <= 2048:
                    a_pert, _ = nesterov_lp_refine(perturbed, t_sub, solver=solver, n_max_iters=15000)
                else:
                    a_pert, _ = subgradient_refine(perturbed, t_sub)
                best_state = do_try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.35, 100.0)
        a_fin = lbgfs_refine(best_state[0], t_fin, T_start=10.0, T_end=0.1, maxiter=4000)
        best_state = do_try_update(best_state, a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15 and best_state[0] is not None and len(best_state[0]) <= 2048:
        t_lp_fin = min(rem * 0.4, 80.0)
        a_lp_fin, _ = nesterov_lp_refine(best_state[0], t_lp_fin, solver=solver, n_max_iters=10000)
        best_state = do_try_update(best_state, a_lp_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.3, 30.0)
        a_ps2, _ = peak_shave_local(best_state[0], t_ps, n_steps=5000)
        best_state = do_try_update(best_state, a_ps2, "final_ps")

    # Final verification
    if best_state[0] is not None:
        fs = evaluate_safe(best_state[0])
        if np.isfinite(fs):
            best_state[1] = fs

    print(f"\nFINAL SCORE: {best_state[1]:.6f} (n={len(best_state[0])})")
    return [float(x) for x in best_state[0]]
