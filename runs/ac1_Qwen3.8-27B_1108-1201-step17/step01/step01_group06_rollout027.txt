import time
from scipy.fft import fft, ifft

def fft_conv_full(a, b):
    """Full convolution of a and b using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    A = fft(a, size)
    B = fft(b, size)
    return np.real(ifft(A * B))[:n_out]

def lse_gradient(a, beta):
    """Compute gradient of log-sum-exp(beta * conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fft_conv_full(a, a)  # full self-convolution, length 2n-1
    # LSE
    beta_c = beta * c
    max_bc = beta_c.max()
    log_sum = max_bc + np.log(np.sum(np.exp(beta_c - max_bc)))
    w = np.exp(beta_c - max_bc) / np.sum(np.exp(beta_c - max_bc))  # softmax(beta*c)
    # gradient = 2 * correlate(w, a)

    a_rev = a[::-1]
    corr = fft_conv_full(w, a_rev)
    grad = 2.0 * corr[n-1:2*n-1]
    return grad, c.max(), log_sum

Minimize 2*n*max_conv/sum^2 using smooth log-sum-exp (LSE) objective.

Algorithm:
- Fix sum(a) = sqrt(2n) so objective = max(conv(a,a))
- Use LSE(β*conv(a,a)) as smooth surrogate for max
- Gradient computed via FFT-based correlation
- Adam optimizer with Euclidean projection onto non-negative simplex
- Adaptive temperature β (exponential schedule)
- Multiple initializations and lengths for exploration
- Polish phase with fine steps at the end

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve(x, y):
    """Full convolution of x and y via FFT."""
    n_out = len(x) + len(y) - 1
    size = 1 << (n_out - 1).bit_length()
    X = fft(x, size)
    Y = fft(y, size)
    return np.real(ifft(X * Y))[:n_out]


def compute_lse_gradient(a, beta):
    """
    Compute gradient of LSE(beta * conv(a,a)) w.r.t. a.
    Returns (gradient, cmax, lse_value).
    """
    n = len(a)
    # Full self-convolution
    c = fft_convolve(a, a)
    cmax = c.max()
    
    # Softmax weights (numerically stable)
    beta_c = beta * c
    max_bc = beta_c.max()
    exp_bc = np.exp(beta_c - max_bc)
    sum_exp = exp_bc.sum()
    w = exp_bc / sum_exp  # softmax(beta * c)
    
    # LSE value
    lse_val = max_bc + np.log(sum_exp)
    
    # Gradient: grad_m = 2*beta * sum_k w_k * a_{k-m}
    # This is 2*beta * correlate(w, a) at lags 0..n-1
    # = 2*beta * convolve(w, a[::-1])[n-1 : 2*n-1]
    a_rev = a[::-1]
    corr = fft_convolve(w, a_rev)
    grad = 2.0 * beta * corr[n - 1: 2 * n - 1]
    
    return grad, cmax, lse_val


def run_smooth_optimize(a_init, n, S, time_budget, beta_start, beta_end, lr_init, rng):
    """
    Smooth optimization using LSE objective with Adam.
    """
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    s = a.sum()
    if s > 0:
        a = a * (S / s)
    else:
        a = np.full(n, S / n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    
    # Precompute total iterations estimate
    # Each iteration: 2 FFTs of size 2*n ~ 0.5ms for n=8192
    # So we can do about 1000-2000 iterations per second
    
    while time.monotonic() - start < time_budget:
        t += 1
        
        # Current beta (exponential schedule)
        progress = min(1.0, t / max(1, int(time_budget * 1500)))  # rough estimate
        progress = min(1.0, (time.monotonic() - start) / max(1e-6, time_budget))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Learning rate schedule (cosine decay)
        lr = lr_init * 0.5 * (1 + np.cos(np.pi * progress))
        lr = max(lr, lr_init * 0.01)
        
        # Compute gradient
        grad, cmax, lse_val = compute_lse_gradient(a, beta)
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Occasional noise for exploration
        if t % 500 == 0 and rng.random() < 0.1:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
    
    return best_a, best_cmax

"""
Minimize 2*n*max_conv/sum^2 using smooth log-sum-exp (LSE) objective with FFT-based gradients.

Key improvements over previous approach:
- LSE smooth surrogate gives true gradient (not just top-k subgradient)
- Gradient computed via single FFT correlation (efficient)
- Adaptive temperature schedule (smooth to sharp)
- Adam + Mirror Descent dual optimizers
- Multiple structured initializations and lengths
- Polishing phase with fine parameters
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.full(len(a), S / len(a), dtype=np.float64)
    return a * (S / s)


def compute_lse_grad(a, beta):
    """
    Compute gradient of (1/beta)*logsumexp(beta * conv(a,a)) w.r.t. a.
    Returns (grad, cmax).
    Uses FFT-based correlation for efficiency.
    """
    n = len(a)
    # FFT size: need N >= 2n-1 for both conv and correlation
    N = 1 << (2 * n - 1).bit_length()
    if N < 2 * n - 1:
        N <<= 1

    A = fft(a, N)

    # Full self-convolution
    c = np.real(ifft(A * A))[:2 * n - 1]
    cmax = c.max()

    # Softmax weights (numerically stable)
    beta_c = beta * c
    max_bc = beta_c.max()
    exp_bc = np.exp(beta_c - max_bc)
    sum_exp = exp_bc.sum()
    w = exp_bc / sum_exp

    # Gradient: grad[m] = 2 * sum_j a[j] * w[m+j]
    # = 2 * circular_correlation(w, a)[m] for m=0..n-1
    # FFT(corr) = FFT(w) * conj(FFT(a))
    W = fft(w, N)
    R = np.real(ifft(W * np.conj(A)))[:n]
    grad = 2.0 * R

    return grad, cmax


def run_adam_smooth(a_init, n, S, time_budget, beta_start, beta_end, lr_init, rng):
    """Adam optimizer with LSE smooth objective and simplex projection."""
    a = normalize_sum(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))

        # Adaptive temperature
        beta = beta_start * (beta_end / beta_start) ** progress

        # Learning rate (cosine decay)
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr_init * 0.005)

        # Compute gradient
        grad, cmax = compute_lse_grad(a, beta)

        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()

        # Gradient normalization for stability
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad_norm = grad / gnorm
        else:
            break

        # Adam update
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad_norm
        v = b2 * v + (1 - b2) * (grad_norm ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        # Occasional noise injection
        if t % 1000 == 0 and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def run_mirror_smooth(a_init, n, S, time_budget, beta_start, beta_end, eta_init, rng):
    """Mirror descent (multiplicative weights) with LSE smooth objective."""
    a = normalize_sum(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))

        beta = beta_start * (beta_end / beta_start) ** progress
        eta = eta_init * max(0.05, 1.0 / (1.0 + 0.002 * t))

        grad, cmax = compute_lse_grad(a, beta)

        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()

        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        grad_scaled = grad / gmax

        # Multiplicative update
        a_new = a * np.exp(-eta * grad_scaled)
        a_new = np.maximum(a_new, 1e-15)
        a = normalize_sum(a_new, S)

        # Occasional noise
        if t % 2000 == 0 and rng.random() < 0.03:
            a = normalize_sum(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []

    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        given = np.clip(given, 0, 1000)
        if len(given) == n and given.sum() > 0:
            inits.append(('given', given.copy()))
        elif len(given) > n:
            inits.append(('given_trim', given[:n].copy()))
        elif len(given) > 0:
            padded = np.zeros(n)
            padded[:min(len(given), n)] = given[:min(len(given), n)]
            inits.append(('given_pad', padded))
    except Exception:
        pass

    # Raised cosine window
    x = np.linspace(0, 1, n)
    inits.append(('raised_cos', 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.1))

    # Gaussian
    x = np.linspace(-3, 3, n)
    inits.append(('gaussian', np.exp(-0.5 * x ** 2)))

    # Symmetric raised cosine
    half = (n + 1) // 2
    xh = np.linspace(0, 1, half)
    h = 0.5 * (1 + np.cos(2 * np.pi * xh)) + 0.1
    if n % 2 == 0:
        sym = np.concatenate([h, h[-2::-1]])
    else:
        sym = np.concatenate([h, h[-1::-1]])
    inits.append(('sym_cos', sym))

    # Multi-frequency sinusoid (non-negative)
    x = np.linspace(0, 2, n)
    sig = 1.0 + 0.3 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x) + 0.1 * np.sin(3 * np.pi * x)
    inits.append(('multi_sin', np.maximum(sig, 0.01)))

    # Random smooth (low-pass filtered random)
    r = rng.random(n)
    # Simple box filter
    k = max(3, n // 512)
    kernel = np.ones(k) / k
    from scipy.signal import convolve as sig_conv
    smooth_r = sig_conv(r, kernel, mode='same')
    inits.append(('smooth_rand', smooth_r + 0.1))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """Search for the best non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + total_budget - 10.0

    global_best = np.inf
    global_a = None

    def check_deadline():
        return time.monotonic() < deadline

    # Phase 1: Exploration (70% of time)
    explore_time = 0.70 * (deadline - time.monotonic())
    polish_time = 0.25 * (deadline - time.monotonic())
    explore_deadline = time.monotonic() + explore_time

    lengths = [8192, 16384]
    all_results = []

    for n in lengths:
        S = np.sqrt(2.0 * n)
        inits = generate_initializations(n, rng)

        # Time per initialization
        per_init_time = explore_time / (len(lengths) * len(inits))

        for name, arr in inits:
            if time.monotonic() > explore_deadline - 5:
                break
            time_left = explore_deadline - time.monotonic()
            if time_left < 3:
                break

            # Run Adam with two parameter settings
            settings = [
                {'beta_start': 0.5, 'beta_end': 8.0, 'lr_init': 0.8 * S / n},
                {'beta_start': 1.0, 'beta_end': 15.0, 'lr_init': 0.5 * S / n},
            ]

            best_local = arr.copy()
            best_local_cmax = np.inf
            per_setting = time_left / (len(settings) + 1)  # +1 for mirror

            for i, ps in enumerate(settings):
                if time.monotonic() > explore_deadline - 10:
                    break
                seg_time = min(per_setting, explore_deadline - time.monotonic() - 10)
                if seg_time < 2:
                    break

                a_best, cmax = run_adam_smooth(
                    best_local, n, S, seg_time,
                    ps['beta_start'], ps['beta_end'], ps['lr_init'], rng
                )
                if cmax < best_local_cmax:
                    best_local_cmax = cmax
                    best_local = a_best.copy()
                if cmax < global_best:
                    global_best = cmax
                    global_a = a_best.copy()
                    print("[Explore] n=%d init=%s adam%d: cmax=%.8f (global=%.8f) t=%.0fs"
                          % (n, name, i, cmax, global_best, time.monotonic() - t_start))

            # Try mirror descent from best so far
            if time.monotonic() < explore_deadline - 5:
                seg_time = min(per_setting, explore_deadline - time.monotonic() - 5)
                if seg_time > 2:
                    a_mir, cmax_mir = run_mirror_smooth(
                        best_local, n, S, seg_time,
                        1.0, 10.0, 0.3, rng
                    )
                    if cmax_mir < best_local_cmax:
                        best_local_cmax = cmax_mir
                        best_local = a_mir.copy()
                    if cmax_mir < global_best:
                        global_best = cmax_mir
                        global_a = a_mir.copy()
                        print("[Explore] n=%d init=%s mirror: cmax=%.8f (global=%.8f) t=%.0fs"
                              % (n, name, cmax_mir, global_best, time.monotonic() - t_start))

            all_results.append((best_local, best_local_cmax, n))

    # Print exploration summary
    print("\n=== Exploration Complete ===")
    print("Global best cmax: %.8f" % global_best)
    print("Time used: %.1f / %.1f s" % (time.monotonic() - t_start, total_budget))

    # Verify best with evaluate_sequence
    if global_a is not None:
        ver = evaluate_sequence(global_a.tolist())
        print("Verify after exploration: %.8f" % ver)
        if np.isfinite(ver):
            global_best = min(global_best, ver)
        else:
            print("WARNING: evaluate returned inf, trying alternatives")

    # Phase 2: Polish (25% of time)
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        polish_start = time.monotonic()
        polish_end = deadline - 5.0
        polish_left = polish_end - polish_start

        if polish_left > 5:
            # Successive polishing with increasing beta and decreasing lr
            a_current = global_a.copy()
            cmax_current = global_best

            polish_configs = [
                {'beta_start': 5.0, 'beta_end': 20.0, 'lr_init': 0.3 * S / n, 'frac': 0.4},
                {'beta_start': 10.0, 'beta_end': 40.0, 'lr_init': 0.1 * S / n, 'frac': 0.35},
                {'beta_start': 20.0, 'beta_end': 80.0, 'lr_init': 0.03 * S / n, 'frac': 0.25},
            ]

            for cfg in polish_configs:
                seg = cfg['frac'] * polish_left
                if time.monotonic() > polish_end - 3:
                    break
                a_new, cmax_new = run_adam_smooth(
                    a_current, n, S, seg,
                    cfg['beta_start'], cfg['beta_end'], cfg['lr_init'], rng
                )
                if cmax_new < cmax_current:
                    cmax_current = cmax_new
                    a_current = a_new.copy()
                    global_best = cmax_new
                    global_a = a_new.copy()
                    print("[Polish] cmax=%.8f (beta=%.0f-%.0f)" % (cmax_new, cfg['beta_start'], cfg['beta_end']))

            # Try mirror descent in polish too
            if time.monotonic() < polish_end - 5:
                seg = polish_end - time.monotonic() - 5
                if seg > 3:
                    a_m, cmax_m = run_mirror_smooth(
                        a_current, n, S, seg,
                        20.0, 60.0, 0.1, rng
                    )
                    if cmax_m < global_best:
                        global_best = cmax_m
                        global_a = a_m.copy()
                        print("[Polish-mirror] cmax=%.8f" % cmax_m)

    # Phase 3: Final verification and return
    if global_a is not None:
        final_a = np.clip(global_a, 0.0, 1000.0)
        if final_a.sum() < 0.01:
            final_a = np.ones(256, dtype=np.float64)
        ver = evaluate_sequence(final_a.tolist())
        print("\n=== FINAL ===")
        print("Final verify: %.8f" % ver)
        print("Sequence length: %d" % len(final_a))
        print("Sum: %.4f" % final_a.sum())
        print("Max element: %.4f" % final_a.max())
        print("Total time: %.1f s" % (time.monotonic() - t_start))

        if np.isfinite(ver):
            return [float(x) for x in final_a]
        else:
            print("Final verify returned inf! Using fallback.")

    # Fallback
    fallback = np.ones(256, dtype=np.float64)
    print("Using fallback sequence.")
    return [float(x) for x in fallback]
