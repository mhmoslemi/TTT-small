```python
def project_simplex(v):
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (cssv - 1))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def score_sequence(a):
    """Compute the score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Compute full convolution of a with itself."""
    return np.convolve(a, a)

    # Actually, sum_k w_k * a[k-j] for k in [0, 2n-2] and j in [0, n-1]:
    # This is like a cross-correlation between w (length 2n-1) and a (length n).
    # np.correlate(w, a, mode='full') gives sum_k w[k] * a[k-j+n-1] ... 
    # Let me just compute it directly.
    # grad_j = 2 * sum_k w_k * a[k-j] where 0 <= k-j < n, i.e., j <= k < j+n
    # For a given j, k ranges from j to j+n-1 (clipped to [0, 2n-2]).
    # This is equivalent to: for each j, sum the w[k] * a[k-j] for valid k.
    # Let's use the fact that this is a convolution.
    # Define w_padded of length 2n-1, and we want for each j in [0,n-1]:
    #   sum_{k=0}^{2n-2} w[k] * a[k-j] * 1[0 <= k-j < n]
    # = sum_{k=j}^{j+n-1} w[k] * a[k-j] (with k clipped to [0, 2n-2])
    # This is the cross-correlation of w and a at lag j.
    # np.correlate(w, a, 'full')[n-1+j] = sum_k w[k+j] * a[k] ... hmm.
    # Let me just use FFT.
    
    # Method: pad a to length 2n-1, compute FFT-based cross-correlation
    L = 2 * n - 1
    w_ext = np.zeros(L)
    w_ext[:len(w)] = w  # w has length 2n-1 = L
    a_ext = np.zeros(L)
    a_ext[:n] = a
    
    # Cross-correlation: sum_k w[k] * a[k-j] = (w * flip(a))[j] ... 
    # Actually, let's use: correlate(w, a, v) = sum_k w[k] * conj(a[k-v])
    # We want: grad_j/2 = sum_k w[k] * a[k-j] for 0 <= k-j < n
    # This is correlate(w, a, j) restricted to valid a indices.
    # = np.correlate(w, a, mode='full') ... let me check the indexing.
    # np.correlate(w, a, 'full')[k] = sum_i w[i+k] * a[i] (for valid i)
    # Hmm, this doesn't quite work with different lengths.
    
    # Simpler: use FFT
    N_fft = 2 * L  # or next power of 2
    W = np.fft.rfft(w_ext, N_fft)
    A = np.fft.rfft(a_ext, N_fft)
    # correlate(w, a) = ifft(conj(W) * A) ... no.
    # In 1D: correlate(f, g, lag) = sum_k f[k] * g[k-lag]
    # This equals convolve(f, flip(g))[len(g)-1+lag]
    # Or: ifft(conj(FFT(f)) * FFT(g)) gives the cross-correlation
    cross = np.fft.irfft(np.conj(W) * A, N_fft)
    # cross[j] = sum_k w[k] * a[k-j] (circular, with our padding it's linear)
    # We want cross[j] for j = 0, ..., n-1
    grad = 2.0 * cross[:n]
    
    return smooth_max, grad

def lp_refine(a, time_limit=60.0, verbose=True, relax=1.0):
    """
    LP-based refinement. Find direction g that maximizes sum(g)
    subject to (a * g)_k <= M * relax for all k.
    """
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    it = 0
    for it in range(500):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv = np.convolve(a, a)
        M = conv.max() * relax
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 30:
            break
        
        # Build LP: maximize sum(g) s.t. (a*g)_k <= M, g >= 0
        # (a*g)_k = sum_i a[i] * g[k-i]
        # For each k, the constraint is: sum_{i: 0<=k-i<n} a[i] * g[k-i] <= M
        # Let j = k-i, so i = k-j: sum_{j: 0<=k-j<n} a[k-j] * g[j] <= M
        # = sum_j a[k-j] * g[j] <= M (with a[k-j]=0 if k-j out of bounds)
        # This is (A @ g)_k <= M where A is a Toeplitz-like matrix.
        
        # Build sparse matrix A of shape (2n-1, n)
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        row_idx = []
        col_idx = []
        data = []
        for k in range(2 * n - 1):
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    row_idx.append(k)
                    col_idx.append(j)
                    data.append(a[i])
        
        A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        remaining = time_limit - (time.monotonic() - start_time)
        if remaining < 5:
            break
        timeout = max(3.0, min(30.0, remaining * 0.2))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if not res.success or res.x is None:
                break
            g = np.maximum(0, res.x)
        except Exception:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        
        g = g / sum_g
        # Mix
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1 - alpha) * a + alpha * g
        a_new = normalize(a_new)
        
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-10:
            a = a_new
        else:
            # Try smaller step
            alpha2 = alpha * 0.1
            a_new = (1 - alpha2) * a + alpha2 * g
            a_new = normalize(a_new)
            a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f}, iters={it}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score

```python
- Smooth-max gradient descent with projected steps
- LP-based direction finding with relaxation
- Multi-start search with diverse initializations and lengths
- Local search polishing

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_grad(a, T=20.0):
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = conv.max()
    exp_vals = np.exp(T * (conv - max_conv))
    w = exp_vals / exp_vals.sum()
    smooth_max = max_conv + np.log(exp_vals.sum()) / T
    L = 2 * n - 1
    w_ext = np.zeros(L)
    w_ext[:] = w
    a_ext = np.zeros(L)
    a_ext[:n] = a

    W = np.fft.rfft(w_ext, N_fft)
    A = np.fft.rfft(a_ext, N_fft)
    cross = np.fft.irfft(np.conj(W) * A, N_fft)
    grad = 2.0 * cross[:n]
    return smooth_max, grad

    print(f"Total initializations: {len(starts)}")

    # Phase 1: Score all starts
    scored = []
    for name, a in starts:
        s = score_sequence(a)
        scored.append((s, name, a))
        print(f"  {name}: {s:.6f}")
    scored.sort()
    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_seq = scored[0][2].copy()

    # Phase 2: Optimize top candidates
    n_top = min(10, len(scored))
    main_budget = (deadline - total_start) * 0.7
    time_per = main_budget / n_top

    for idx in range(n_top):
        s0, name, a = scored[idx]
        remaining = deadline - total_start
        if remaining < 20:
            break
        t_alloc = min(time_per, remaining * 0.8)
        if t_alloc < 10:
            break
        print(f"\n=== Opt {name} (init={s0:.6f}, n={len(a)}, t={t_alloc:.0f}s) ===")

        # Smooth max gradient
        t1 = t_alloc * 0.35
        a1, s1 = smooth_max_optimize(a, time_limit=t1, T=15.0, eta0=0.05, verbose=True)
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()
            print(f"  *** SM: {best_score:.6f}")

        # LP refinement
        remaining = deadline - total_start
        if remaining > 20:
            t2 = min(t_alloc * 0.35, remaining * 0.5)
            a2, s2 = lp_refine(a1, time_limit=t2, verbose=True, relax=1.0)
            if s2 < best_score:
                best_score = s2
                best_seq = a2.copy()
                print(f"  *** LP: {best_score:.6f}")

        # Second SM with higher T
        remaining = deadline - total_start
        if remaining > 15:
            t3 = min(t_alloc * 0.2, remaining * 0.5)
            a3, s3 = smooth_max_optimize(a2 if 'a2' in dir() else a1,
                                         time_limit=t3, T=40.0, eta0=0.02, verbose=True)
            if s3 < best_score:
                best_score = s3
                best_seq = a3.copy()
                print(f"  *** SM2: {best_score:.6f}")

    # Phase 3: Cross-pollinate - try optimizing best with different methods
    remaining = deadline - total_start
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Polishing best ({best_score:.6f}), remaining={remaining:.0f}s ===")
        # LP on best
        t_lp = remaining * 0.3
        bp1, s_bp1 = lp_refine(best_seq, time_limit=t_lp, verbose=True, relax=1.02)
        if s_bp1 < best_score:
            best_score = s_bp1
            best_seq = bp1.copy()
            print(f"  *** Polish LP: {best_score:.6f}")

        # Local search
        remaining = deadline - total_start
        if remaining > 15:
            t_ls = remaining * 0.3
            bp2, s_bp2 = local_search(best_seq, time_limit=t_ls, rng=rng, verbose=True)
            if s_bp2 < best_score:
                best_score = s_bp2
                best_seq = bp2.copy()
                print(f"  *** Polish LS: {best_score:.6f}")

        # Final SM
        remaining = deadline - total_start
        if remaining > 10:
            t_sm = remaining * 0.4
            bp3, s_bp3 = smooth_max_optimize(best_seq, time_limit=t_sm, T=60.0, eta0=0.01, verbose=True)
            if s_bp3 < best_score:
                best_score = s_bp3
                best_seq = bp3.copy()
                print(f"  *** Final SM: {best_score:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = normalize(initial_raised_cosine(500))
        best_score = score_sequence(best_seq)

    print(f"\nFinal: score={best_score:.6f}, n={len(best_seq)}")
    verify = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified: {verify:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")
    return [float(x) for x in best_seq]
```
