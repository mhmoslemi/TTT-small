"""
Hybrid optimization for minimizing peak self-convolution of non-negative sequences.
Combines LP-based refinement, gradient descent on smooth approximations of the 
L-infinity objective, and stochastic perturbations with adaptive scheduling.

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_seq(a):
    """Compute the score for a sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def solve_lp(a, M, timeout=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(n_rows, M)

    options = {'presolve': True, 'dual_feasibility_tolerance': 1e-9,
               'primal_feasibility_tolerance': 1e-9}
    if timeout is not None:
        options['time_limit'] = timeout
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options=options)
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

        rows_list.append(k_vals)
        cols_list.append(np.full(len(k_vals), j, dtype=np.int32))
        data_list.append(a[k_vals - j].astype(np.float64))
    row_idx = np.concatenate(rows_list)
    col_idx = np.concatenate(cols_list)
    data = np.concatenate(data_list)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp(a, M, timeout=60.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    options = {'presolve': True, 'time_limit': timeout}
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options=options)
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def gradient_of_lip(a, conv, p=4):
    """
    Compute gradient of sum(conv[k]^p) w.r.t. a.
    Uses FFT for efficiency.
    Returns gradient vector of length n.
    """
    n = len(a)
    # grad[j] = 2p * sum_i a[i] * conv[j+i]^{p-1}  (for valid j+i)
    conv_p = np.power(np.maximum(conv, 0.0), p - 1)
    # grad[j] = 2p * sum_{i=0}^{n-1} a[i] * conv_p[j+i]
    # = 2p * conv(a[::-1], conv_p)[n-1 : 2n-1]
    full = np.convolve(a[::-1], conv_p)
    grad = 2.0 * p * full[n-1 : 2*n-1]
    return grad


def gradient_refine(a, time_limit_s=60.0, p=4, verbose=False):
    """
    Gradient descent on sum(conv^p) to approximate L-infinity minimization.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    lr = 1.0 / (n * p)
    it = 0
    no_improve = 0
    
    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        grad = gradient_of_lip(a, conv, p)
        
        # Normalized gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        a_new = a - lr * grad
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        new_score = score_seq(a_new)
        
        if new_score < best_score - 1e-10:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 200 or lr < 1e-8:
                break
        
        it += 1
    
    if verbose:
        print(f"    Grad: score={best_score:.6f}, {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
