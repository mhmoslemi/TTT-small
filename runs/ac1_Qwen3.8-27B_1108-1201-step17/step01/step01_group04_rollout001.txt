"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam optimizer with log-parameterization, soft-max multi-peak gradient,
LP-based refinement with tightening bounds, and diverse multi-start strategy.

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) * target / n
    return a * (target / S)


def _soft_peak_grad(a, tau=5.0):
    """Compute soft-max weighted subgradient of the objective."""
    n = len(a)
    S = float(a.sum())
    c = np.convolve(a, a)
    c_max = c.max()
    
    # Soft-max weights over convolution positions
    shifted = c - c_max
    w = np.exp(tau * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Compute dC/da[j] = sum_m w[m] * 2*a[m-j] (valid indices)
    dC = np.zeros(n)
    for j in range(n):
        # m ranges over valid values: max(0, j) to min(2n-2, j+n-1)
        m_lo = max(0, j)
        m_hi = min(2 * n - 2, j + n - 1)
        # m-j ranges from m_lo-j to m_hi-j, must be in [0, n-1]
        # a[m-j] where m goes from m_lo to m_hi
        k_lo = max(0, m_lo - j)
        k_hi = min(n - 1, m_hi - j)
        if k_hi >= k_lo:
            dC[j] = 2.0 * np.dot(w[m_lo:m_hi + 1], a[k_lo:k_hi + 1])
    
    # dF/da[j] = 2*n * (dC[j]/S^2 - 2*C/S^3) = 2*n/S^3 * (dC[j]*S - 2*c_max)
    grad_F = 2.0 * n / (S ** 3) * (dC * S - 2.0 * c_max)
    return grad_F, c_max, c


def _hard_peak_grad(a):
    """Compute subgradient using only the hardest peak."""
    n = len(a)
    S = float(a.sum())
    c = np.convolve(a, a)
    c_max = c.max()
    m_star = int(c.argmax())
    
    dC = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        dC[j] = 2.0 * a[m_star - j]
    
    grad_F = 2.0 * n / (S ** 3) * (dC * S - 2.0 * c_max)
    return grad_F, c_max, c


def _adam_update(x, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step. Returns updated parameters and moments."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v


def _optimize_adam(a0, deadline, max_iters=5000):
    """Optimize using Adam with soft-max gradient. Returns (best_seq, best_score)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 1e-6, 1000.0)
    a = _normalize(a)
    best = a.copy()
    best_score = _compute_obj(a)
    S_t = np.sqrt(2.0 * n)
    
    # Initialize Adam moments
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Learning rate schedule
    lr_init = 0.05
    lr_min = 0.001
    
    # Soft-max temperature schedule
    tau_init = 3.0
    tau_max = 20.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        t += 1
        frac = t / max_iters
        lr = lr_init * (1.0 + np.cos(np.pi * frac)) * 0.5 + lr_min
        tau = tau_init + (tau_max - tau_init) * frac
        
        # Compute gradient
        grad, c_max, c = _soft_peak_grad(a, tau=tau)
        
        # Project gradient to maintain sum constraint
        grad = grad - grad.mean()
        
        # Adam step
        a, m, v = _adam_update(a, grad, m, v, t, lr=lr)
        
        # Clip and renormalize
        a = np.maximum(a, 1e-8)
        a = _normalize(a, S_t)

def _lp_refine_single(a, M, n):
    """Solve LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0. Returns g or None."""
    if n > 2048:
        return None
    
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = a[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine_sequence(a, deadline, max_iters=100):
    """LP-based refinement: alternate fixing sides, tighten M. Returns (best_seq, best_score)."""
    n = len(a)
    best = a.copy()
    best_score = _compute_obj(a)
    S_t = np.sqrt(2.0 * n)
    
    c = np.convolve(a, a)
    M = float(c.max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Solve LP with current a as the fixed side
        g = _lp_refine_single(a, M, n)
        if g is None:
            break
        
        # Try blending
        improved = False
        for t_blend in [0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            s_new = _compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

            # Tighten M
            c = np.convolve(a, a)

        else:
            # Try with slightly tighter M

    print(f"Generated {len(scored_starts)} starts, best initial score: {scored_starts[0][0]:.6f}")
    
    # Phase 1: Adam optimization on top starts
    max_starts_phase1 = min(12, len(scored_starts))
    per_start_budget = min(120.0, (deadline - time.monotonic()) / max(1, max_starts_phase1))
    
    for idx in range(max_starts_phase1):
        if time.monotonic() > deadline - 60.0:
            break
        
        init_score, a0 = scored_starts[idx]
        start_time = time.monotonic()
        max_time = min(per_start_budget, deadline - time.monotonic() - 5.0)
        
        if max_time < 10.0:
            break
        
        seq, score = _optimize_adam(a0, t_start + start_time + max_time)
        
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Phase1 start {idx}: n={len(seq)}, score={score:.6f} "
                  f"(elapsed={time.monotonic()-t_start:.0f}s)")
        else:
            print(f"  Phase1 start {idx}: n={len(a0)}, score={score:.6f} "
                  f"(not best, elapsed={time.monotonic()-t_start:.0f}s)")
    
    # Phase 2: LP refinement on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        print(f"\nPhase 2: LP refinement (current best={best_score:.6f})")
        n_best = len(best_seq)
        lp_deadline = min(deadline - 10.0, t_start + time.monotonic() - t_start + 400.0)
        lp_deadline = min(lp_deadline, deadline - 10.0)
        
        if n_best <= 2048:
            seq, score = _lp_refine_sequence(best_seq, lp_deadline, max_iters=80)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  After LP: score={score:.6f}")
    
    # Phase 3: Fine gradient refinement
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"\nPhase 3: Gradient refinement (current best={best_score:.6f})")
        grad_deadline = deadline - 5.0
        seq, score = _gradient_refine(best_seq, grad_deadline, max_iters=3000)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  After gradient refine: score={score:.6f}")
    
    # Phase 4: Another LP pass if time remains
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        n_best = len(best_seq)
        if n_best <= 2048:
            print(f"\nPhase 4: Final LP pass (current best={best_score:.6f})")
            seq, score = _lp_refine_sequence(best_seq, deadline - 5.0, max_iters=40)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  After final LP: score={score:.6f}")
    
    # Phase 5: Try a different length if we have time
    if time.monotonic() < deadline - 60.0:
        print(f"\nPhase 5: Try alternative length (current best={best_score:.6f})")
        # Try a few specific lengths that weren't tried yet
        tried_lengths = {len(s) for _, s in scored_starts[:max_starts_phase1]}
        
        for alt_n in [384, 768, 1536]:
            if alt_n in tried_lengths or time.monotonic() > deadline - 30.0:
                continue
            a_test = rng.random(alt_n) * 0.5 + 0.25
            a_test = _normalize(a_test)
            s_test = _compute_obj(a_test)
            
            seq, score = _optimize_adam(a_test, deadline - 15.0, max_iters=2000)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Alt n={alt_n}: score={score:.6f}")
    
    # Final validation
    if best_seq is not None:
        # Ensure all values are in valid range
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        final_score = _compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.6f}")
        
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
