"""
LP-based iterative optimization for minimizing the peak of linear
autocorrelation (convolution) of non-negative sequences.
Uses the structure that for fixed f, finding the best update direction g
is a linear program.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_conv(a):
    """Compute autocorrelation (linear convolution with itself)."""
    n = len(a)
    if n <= 128:
        return np.convolve(a, a)
    else:
        return fft_convolve(a, a)

def score_normalized(f):
    """Score assuming sum(f) = 1. Returns 2n * max(conv(f,f))."""
    n = len(f)
    return 2.0 * n * np.max(compute_conv(f))

def build_A(f):
    """Build constraint matrix A where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    return A

def solve_lp(f, M):
    """Solve the LP to find best direction g."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        g = result.x
        if np.sum(g) > 1e-10:
            return g / np.sum(g)
    return None

def line_search(f, g, n_points=50):
    """Find best t in [0,1] minimizing max(conv((1-t)f + tg, ...))."""
    best_t = 0.0
    best_val = np.max(compute_conv(f))
    
    for t in np.linspace(0.01, 1.0, n_points):
        candidate = (1-t)*f + t*g
        val = np.max(compute_conv(candidate))
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

def optimize_lp(n, f_init, time_limit=30, max_iter=500):
    """Run LP-based optimization for a given sequence length."""
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    s = np.sum(f)
    if s < 1e-10:
        return f_init, np.inf
    f = f / s
    
    best_f = f.copy()
    best_conv_peak = np.max(compute_conv(f))
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iter):

            break
        
        M = np.max(compute_conv(f))
        g = solve_lp(f, M)
        
        if g is None:
            break

            f = (1-t)*f + t*g
            f = np.clip(f, 0, None)
            fs = np.sum(f)
            if fs < 1e-10:
                break
            f = f / fs
            best_f = f.copy()
            best_conv_peak = np.max(compute_conv(f))
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    final_score = 2.0 * n * best_conv_peak
    return best_f, final_score

"""
LP-based iterative optimization for minimizing the peak of linear convolution
of non-negative sequences. Uses the key insight that for a fixed sequence f,
finding the best update direction g (maximizing sum(g) subject to f*g <= M, g>=0)
is a linear program. Combined with line search and multiple restarts, this
achieves near-optimal sequences for various lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def auto_conv(a):
    """Compute linear convolution of a with itself."""
    n = len(a)
    if n <= 128:
        return np.convolve(a, a)
    return fft_convolve(a, a)


def score_from_conv_peak(n, peak):
    """Score = 2n * peak (assuming sum(a) = 1)."""
    return 2.0 * n * peak


def build_constraint_matrix(f):
    """Build A where (A@g)[k] = (f*g)[k] for k=0..2n-2."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_lp_direction(f, M, time_limit_lp=3.0):
    """
    Solve LP: maximize sum(g) s.t. A@g <= M, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_limit_lp})
        if result.success and result.x is not None:
            g = result.x
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def line_search(f, g, n_coarse=30, n_fine=20):
    """Two-stage line search for best t in [0,1]."""
    best_t = 0.0
    best_val = np.max(auto_conv(f))

    # Coarse search
    ts = np.linspace(0.0, 1.0, n_coarse)
    for t in ts[1:]:
        candidate = (1 - t) * f + t * g
        val = np.max(auto_conv(candidate))
        if val < best_val:
            best_val = val
            best_t = t

    if best_t == 0.0:
        return 0.0, best_val

    # Fine search around best
    if best_t < 1.0:
        t_low = max(0.0, best_t - 2.0 / n_coarse)
        t_high = min(1.0, best_t + 2.0 / n_coarse)
    else:
        t_low = best_t - 4.0 / n_coarse
        t_high = 1.0
    ts_fine = np.linspace(t_low, t_high, n_fine)
    for t in ts_fine:
        candidate = (1 - t) * f + t * g
        val = np.max(auto_conv(candidate))
        if val < best_val:
            best_val = val
            best_t = t

    return best_t, best_val


def optimize_lp_sequence(n, f_init, time_limit, max_iter=2000):
    """
    Run LP-based iterative optimization.
    f_init should be non-negative. Returns (best_f_normalized, best_score).
    """
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0.0, None)
    s = np.sum(f)
    if s < 1e-10:
        return f_init, np.inf
    f = f / s

    best_f = f.copy()
    best_peak = np.max(auto_conv(f))
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit and iter_count < max_iter:
        M = np.max(auto_conv(f))
        g = solve_lp_direction(f, M, time_limit_lp=min(2.0, max(0.5, time_limit / 10)))

        if g is None:
            break

        t, val = line_search(f, g)

        if val < best_peak * (1.0 - 1e-9):
            f_new = (1 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, None)
            fs = np.sum(f_new)
            if fs < 1e-10:
                break
            f_new = f_new / fs
            f = f_new
            new_peak = np.max(auto_conv(f))
            if new_peak < best_peak:
                best_f = f.copy()
                best_peak = new_peak
            no_improve = 0
        else:
            # Try a smaller step or different direction
            no_improve += 1
            if no_improve > 40:
                break
            # Occasionally try a random perturbation to escape
            if no_improve % 20 == 0:
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                f = f * (1.0 + 0.02 * rng.standard_normal(n))
                f = np.clip(f, 0.0, None)
                fs = np.sum(f)
                if fs > 1e-10:
                    f = f / fs

        iter_count += 1
        if iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      iter={iter_count}, peak={best_peak:.6e}, score={score_from_conv_peak(n, best_peak):.6f}, t={elapsed:.1f}s")

    final_score = score_from_conv_peak(n, best_peak)
    return best_f, final_score


def make_monotone_init(n, growth=1.0, rng=None):
    """Monotonically increasing initialization."""
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + growth * t / n
    if rng is not None:
        f += 0.01 * rng.standard_normal(n)
    return np.clip(f, 0.01, None)


def make_exponential_init(n, rate=1.0, rng=None):
    """Exponentially increasing initialization."""
    t = np.arange(n, dtype=np.float64)
    f = np.exp(rate * t / n)
    if rng is not None:
        f += 0.01 * rng.standard_normal(n)
    return np.clip(f, 0.01, None)


def make_quadratic_init(n, rng=None):
    """Quadratically varying initialization."""
    t = np.linspace(0, 1, n)
    f = 1.0 + 2.0 * t  # linear ramp
    if rng is not None:
        f += 0.02 * rng.standard_normal(n)
    return np.clip(f, 0.01, None)


def random_init(n, rng):
    """Random positive initialization."""
    return 0.5 + rng.random(n)


def resample_sequence(seq, n_target):
    """Interpolate a sequence to a target length."""
    n_src = len(seq)
    if n_src == n_target:
        return seq.astype(np.float64).copy()
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_target)
    return np.interp(x_new, x_old, seq.astype(np.float64))


def local_polish(f, time_limit, rng, n_steps=5000):
    """
    Simple local search polish: random multiplicative perturbations + LP re-optimization.
    """
    f = f.astype(np.float64).copy()
    f = np.clip(f, 0.0, None)
    s = np.sum(f)
    if s < 1e-10:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_peak = np.max(auto_conv(f))
    n = len(f)
    t0 = time.monotonic()
    temperature = 0.005

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Random perturbation
        candidate = f.copy()
        r = rng.random()
        if r < 0.4:
            # Small global perturbation
            candidate *= np.exp(rng.normal(0, 0.01 * temperature / 0.005, size=n))
        elif r < 0.7:
            # Block perturbation
            width = max(1, n // 32)
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02 * temperature / 0.005, size=width))
        else:
            # Target peak reduction: reduce elements contributing to peak
            conv = auto_conv(f)
            peak_idx = int(np.argmax(conv))
            # Elements around the peak position
            center = peak_idx // 2
            half_w = max(1, n // 16)
            lo = max(0, center - half_w)
            hi = min(n, center + half_w)
            candidate[lo:hi] *= np.exp(rng.normal(-0.01, 0.005, size=hi - lo))

        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            continue
        candidate = candidate / cs
        new_peak = np.max(auto_conv(candidate))

        if new_peak < best_peak:
            best_peak = new_peak
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_peak - best_peak) * 1000 / max(temperature, 1e-10)):
            f = candidate

        temperature *= 0.999

    final_score = score_from_conv_peak(n, best_peak)
    return best_f, final_score
