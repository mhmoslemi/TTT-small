    size = 2 * n - 1

        j_end = min(n-1, k)
        if j_start <= j_end:

        return None
    return res.x

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative optimization.
The key idea: given current sequence a, solve the LP max sum(b) s.t. (a*b)[k] <= M, b >= 0.
The LP solution gives a direction to reduce the peak autocorrelation. Iterate until convergence.
Uses multiple sequence lengths, diverse initializations, and simulated annealing restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_constraint_matrix(a):
    """Build A matrix where A[k,j] = a[k-j] if 0<=k-j<n else 0. Shape: (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # A[k, j] = a[k-j] for j in [j_lo, j_hi]
            # k-j ranges from k-j_hi to k-j_lo
            A[k, j_lo:j_hi+1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A

def solve_lp_for_b(a, M):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0. Returns b or None."""
    n = len(a)
    A = build_constraint_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.status == 0:
        return res.x
    return None

def lp_iterate(a0, max_iter=100, time_limit=None, verbose=False):
    """
    Run LP iteration. Returns (best_a, best_M) where best_M = max(conv(best_a, best_a)).
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M
    
    t_start = time.monotonic()
    prev_M = M
    
    for it in range(max_iter):
        if time_limit is not None and (time.monotonic() - t_start) > time_limit:
            break
            
        b = solve_lp_for_b(a, M)
        if b is None:
            break
        
        S = b.sum()
        if S <= 1.0 + 1e-12:
            break
        
        a_new = b / S
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
        
        # Check for stagnation
        if abs(M_new - M) < 1e-12 * max(1.0, M):
            a = a_new
            break
        
        a = a_new
        M = M_new
        
        if verbose and it % 10 == 0:
            print(f"    iter {it}: M={M:.8f}, sum_b={S:.6f}")
    
    return best_a, best_M

    for step in range(n_steps):
        frac = step / n_steps
        # Decrease step size over time
        si = min(int(frac * len(step_size_schedule)), len(step_size_schedule) - 1)
        step_frac = step_size_schedule[si]
        
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        
        # Try transferring mass from j to i (and vice versa)
        best_local_M = best_M
        best_delta = 0.0
        
        # Transfer from j to i: a[i] += t, a[j] -= t
        for t_frac in [step_frac, step_frac * 0.5, step_frac * 2, step_frac * 0.25]:
            t = t_frac * a[j]
            if t <= 0 or t >= a[j]:
                continue
            a_trial = a.copy()
            a_trial[i] += t
            a_trial[j] -= t
            M_trial = fast_convolve(a_trial).max()
            if M_trial < best_local_M:
                best_local_M = M_trial
                best_delta = t
        
        # Transfer from i to j
        for t_frac in [step_frac, step_frac * 0.5, step_frac * 2, step_frac * 0.25]:
            t = t_frac * a[i]
            if t <= 0 or t >= a[i]:
                continue
            a_trial = a.copy()
            a_trial[j] += t
            a_trial[i] -= t
            M_trial = fast_convolve(a_trial).max()
            if M_trial < best_local_M:
                best_local_M = M_trial
                best_delta = -t  # negative means transfer from i to j
        
        if best_delta > 0:
            a[i] += best_delta
            a[j] -= best_delta
        elif best_delta < 0:
            a[i] -= (-best_delta)
            a[j] += (-best_delta)
        
        if best_local_M < best_M:
            best_M = best_local_M
    
    a = np.maximum(a, 0)
    a /= a.sum()
    return a, best_M

def make_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian (various widths)
    x = np.arange(n, dtype=np.float64)
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Edge-concentrated (bimodal)
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Random with structure
    inits['random2'] = rng.dirichlet(np.ones(n) * 10.0)
    inits['random3'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Low-frequency sinusoid
    inits['sin'] = (1.0 + np.sin(2 * np.pi * x / n)) / (2.0 * n)
    
    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    cw /= cw.sum()
    inits['cosine'] = cw
    
    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    
    # Triangular
    tri = 1.0 - np.abs(x - n * 0.5) / (n * 0.5)
    tri = np.maximum(tri, 0)
    tri /= tri.sum()
    inits['triangular'] = tri
    
    # White noise (uniform random, not dirichlet)
    wn = rng.random(n)
    wn += 1e-6
    wn /= wn.sum()
    inits['white'] = wn
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative optimization.
Core idea: Given sequence a, solve LP max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
The LP solution gives a direction to reduce peak autocorrelation. Iterate until convergence.
Combined with diverse initializations, local refinement via coordinate descent,
simulated annealing restarts, and adaptive time management across sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_constraint_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0. Shape: (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def solve_lp(a, M, top_k=None):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0. Returns b or None."""
    n = len(a)
    A = build_constraint_matrix(a)
    
    if top_k is not None and top_k < 2 * n - 1:
        conv = fast_convolve(a)
        idx = np.argsort(conv)[::-1][:top_k]
        A_sel = A[idx, :]
    else:
        A_sel = A
    
    c = -np.ones(len(A_sel))
    c[:n] = -1.0
    c[n:] = 0.0
    # Actually c should just be -ones(n) for the n variables
    c = -np.ones(n)
    b_ub = np.full(A_sel.shape[0], M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sel, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_iterate(a0, max_iter=200, time_limit=None, top_k=None):
    """Run LP iteration to convergence. Returns (best_a, best_M)."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a), dtype=np.float64) / len(a)
    else:
        a /= s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M
    
    t_start = time.monotonic()
    
    for it in range(max_iter):
        if time_limit is not None and (time.monotonic() - t_start) > time_limit:
            break
        
        b = solve_lp(a, M, top_k=top_k)
        if b is None:
            break
        
        S = b.sum()
        if S <= 1.0 + 1e-12:
            break
        
        a_new = b / S
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
        
        # Check for stagnation
        if abs(M_new - M) < 1e-13 * max(1.0, M):
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
            break
        
        a = a_new
        M = M_new
    
    # One more final check
    conv_final = fast_convolve(a)
    M_final = conv_final.max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a, best_M


def local_refine(a0, n_steps=3000, seed=42):
    """Local refinement using random coordinate pair mass transfers."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 0.0
    a /= s
    n = len(a)
    rng = np.random.default_rng(seed)
    
    conv = fast_convolve(a)
    best_M = conv.max()
    
    # Adaptive step sizes (decreasing)
    for step in range(n_steps):
        frac = step / n_steps
        # Step size decreases from 0.5 to 0.01
        base_step = 0.5 * (0.02 ** frac)
        
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        
        best_local_M = best_M
        best_delta = 0.0
        
        # Try several transfer amounts in both directions
        for t_frac in [base_step, base_step * 0.5, base_step * 2.0, base_step * 0.25, base_step * 4.0]:
            # Transfer from j to i
            t = t_frac * a[j]
            if 0 < t < a[j] * 0.999:
                a_trial = a.copy()
                a_trial[i] += t
                a_trial[j] -= t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_local_M:
                    best_local_M = M_trial
                    best_delta = t
            
            # Transfer from i to j
            t = t_frac * a[i]
            if 0 < t < a[i] * 0.999:
                a_trial = a.copy()
                a_trial[j] += t
                a_trial[i] -= t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_local_M:
                    best_local_M = M_trial
                    best_delta = -t
        
        if best_delta > 0:
            a[i] += best_delta
            a[j] -= best_delta
        elif best_delta < 0:
            a[i] += best_delta
            a[j] -= best_delta
        
        if best_local_M < best_M:
            best_M = best_local_M
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a, best_M


def trimmed_evaluate(a):
    """Evaluate with trimming of leading/trailing zeros. Returns (score, trimmed_a)."""
    # Find first and last non-zero
    thresh = 1e-12
    nonzero = np.where(a > thresh)[0]
    if len(nonzero) == 0:
        return float('inf'), a
    lo = nonzero[0]
    hi = nonzero[-1]
    a_trimmed = a[lo:hi + 1].copy()
    try:
        score = evaluate_sequence(a_trimmed.tolist())
    except Exception:
        score = float('inf')
    return score, a_trimmed


def make_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussians
    for sf in [0.1, 0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random Dirichlet (various concentrations)
    inits['rand_alpha03'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_alpha10'] = rng.dirichlet(np.ones(n) * 10.0)
    inits['rand_alpha01'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Low-freq sinusoid
    inits['sin'] = (1.0 + np.sin(2 * np.pi * x / n)) / (2.0 * n)
    inits['sin2'] = (1.0 + np.sin(4 * np.pi * x / n)) / (2.0 * n)
    
    # Cosine window
    if n > 1:
        cw = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    else:
        cw = np.ones(1)
    cw = np.maximum(cw, 0)
    cw /= cw.sum()
    inits['cosine'] = cw
    
    # Triangular
    tri = 1.0 - np.abs(x - n * 0.5) / (n * 0.5)
    tri = np.maximum(tri, 1e-6)
    tri /= tri.sum()
    inits['triangular'] = tri
    
    # White noise
    wn = rng.random(n) + 1e-6
    wn /= wn.sum()
    inits['white'] = wn
    
    # Parent-based
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses LP-based iterative optimization with:
    - Multiple sequence lengths (64-2048)
    - Diverse initializations
    - LP iteration as main optimizer
    - Local refinement via coordinate descent
    - Simulated annealing restarts
    - Adaptive time management
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    deadline = t_global_start + float(budget_s) - 30.0

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={p.size}, sum={p.sum():.4f}")
    except Exception:
        pass

    best_sequence = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_sequence, best_score
        # Try both trimmed and untrimmed
        score_trimmed, a_trimmed = trimmed_evaluate(a)
        try:
            score_full = float(evaluate_sequence(a.tolist()))
        except Exception:
            score_full = float('inf')
        
        if score_trimmed < score_full:
            score = score_trimmed
            a_use = a_trimmed
        else:
            score = score_full
            a_use = a
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_use.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  NEW BEST [{label}]: score={score:.6f} (n={len(a_use)}, t={elapsed:.1f}s)")

    # Phase 1: LP iteration scan
    print("=== Phase 1: LP Iteration Scan ===")
    lengths = [64, 128, 256, 512, 1024, 2048]
    
    # Build config list
    phase1_configs = []
    for n in lengths:
        inits = make_initializations(n, rng, parent)
        for name, a0 in inits.items():
            phase1_configs.append((n, name, a0))

    # Priority: parent first, then medium lengths
    def config_priority(cfg):
        n, name, a0 = cfg
        pri = 0
        if 'parent' in name:
            pri -= 200
        if n in [128, 256, 512]:
            pri -= 50
        if n == 64:
            pri += 0
        if n >= 1024:
            pri += 30
        if 'uniform' in name:
            pri -= 10
        return pri

    phase1_configs.sort(key=config_priority)

    # Time allocation
    n_configs = len(phase1_configs)
    phase1_budget = (deadline - time.monotonic()) * 0.45
    time_per_config = max(3.0, phase1_budget / n_configs)

    lp_results = []  # (score, n, a_opt, M_opt)
    t_phase1_start = time.monotonic()

    for i, (n, name, a0) in enumerate(phase1_configs):
        if time.monotonic() > deadline - 90:
            break
        
        t_alloc = min(time_per_config, deadline - time.monotonic() - 90)
        if t_alloc < 2.0:
            break
        
        # Adjust top_k for large n to speed up LP
        if n <= 512:
            tk = None
        elif n <= 1024:
            tk = 1024
        else:
            tk = 2048
        
        try:
            a_opt, M_opt = lp_iterate(a0, max_iter=300, time_limit=t_alloc, top_k=tk)
            score_trimmed, a_trimmed = trimmed_evaluate(a_opt)
            try:
                score_full = float(evaluate_sequence(a_opt.tolist()))
            except Exception:
                score_full = float('inf')
            score = min(score_trimmed, score_full)
            
            if np.isfinite(score):
                best_a = a_trimmed if score_trimmed < score_full else a_opt
                lp_results.append((score, n, name, best_a, M_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = best_a.copy()
                    print(f"  P1 n={n:4d} {name:12s}: {score:.6f}")
        except Exception:
            pass

    print(f"Phase 1 done. Best: {best_score:.6f}, configs tried: {i+1}/{n_configs}")

    # Phase 2: Deep optimization on top results
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.6f}) ===")
    
    lp_results.sort(key=lambda x: x[0])
    
    top_configs = []
    used_lengths = set()
    for score, n, name, a_opt, M_opt in lp_results:
        if len(top_configs) < 12:
            top_configs.append((n, a_opt))
            used_lengths.add(n)
        elif n not in used_lengths and len(top_configs) < 16:
            top_configs.append((n, a_opt))
            used_lengths.add(n)
    
    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 90
    time_per_deep = max(8.0, remaining / max(1, len(top_configs)))

    for n, a0 in top_configs:
        if time.monotonic() > deadline - 90:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 90)
        if t_alloc < 4:
            break
        
        tk = None if n <= 512 else (1024 if n <= 1024 else 2048)
        
        # LP with more iterations
        a_opt, M_opt = lp_iterate(a0, max_iter=500, time_limit=t_alloc * 0.5, top_k=tk)
        
        # Local refinement
        t_rem = deadline - time.monotonic() - 90
        if t_rem > 5:
            n_refine = min(5000, max(500, int(t_rem * 300)))
            a_opt, M_opt = local_refine(a_opt, n_steps=n_refine, seed=int(rng.integers(99999)))
        
        try_update(a_opt, f"deep n={n}")

    # Phase 3: Simulated annealing restarts
    print(f"\n=== Phase 3: Annealing Restarts (best={best_score:.6f}) ===")
    
    if best_sequence is not None and time.monotonic() < deadline - 60:
        n_best = len(best_sequence)
        n_restarts = 30
        t_remaining = deadline - time.monotonic() - 60
        t_per_restart = t_remaining / n_restarts
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            
            t_alloc = min(t_per_restart, deadline - time.monotonic() - 60)
            if t_alloc < 3:
                break
            
            # Perturb with decreasing noise
            noise_scale = (0.2 + 0.5 * (r / n_restarts)) / np.sqrt(n_best)
            a_pert = best_sequence.copy()
            a_pert += rng.normal(0, noise_scale, n_best)
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 1e-10:
                continue
            a_pert /= sp
            
            tk = None if n_best <= 512 else (1024 if n_best <= 1024 else 2048)
            a_opt, M_opt = lp_iterate(a_pert, max_iter=150, time_limit=t_alloc * 0.6, top_k=tk)
            
            t_rem = deadline - time.monotonic() - 60
            if t_rem > 3:
                a_opt, M_opt = local_refine(a_opt, n_steps=min(2000, int(t_rem * 200)),
                                            seed=int(rng.integers(99999)))
            
            try_update(a_opt, f"anneal {r}")

    # Phase 4: Final intensive polish on the best
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")
    
    if best_sequence is not None and time.monotonic() < deadline - 15:
        t_rem = deadline - 15 - time.monotonic()
        
        # Extended local refinement
        n_steps = min(50000, max(1000, int(t_rem * 500)))
        a_opt, M_opt = local_refine(best_sequence, n_steps=n_steps, seed=77777)
        try_update(a_opt, "polish1")
        
        # Another round of LP + refine
        t_rem = deadline - 15 - time.monotonic()
        if t_rem > 10:
            n_best = len(a_opt)
            tk = None if n_best <= 512 else (1024 if n_best <= 1024 else 2048)
            a_opt2, M_opt2 = lp_iterate(a_opt, max_iter=200, time_limit=t_rem * 0.5, top_k=tk)
            t_rem2 = deadline - 15 - time.monotonic()
            if t_rem2 > 3:
                a_opt2, M_opt2 = local_refine(a_opt2, n_steps=min(5000, int(t_rem2 * 300)),
                                              seed=88888)
            try_update(a_opt2, "polish2")
        
        # Multi-seed local refinement
        t_rem = deadline - 15 - time.monotonic()
        if t_rem > 10:
            for s in [111, 222, 333]:
                if time.monotonic() > deadline - 15:
                    break
                n_s = min(3000, int((deadline - 15 - time.monotonic()) * 100))
                if n_s < 100:
                    break
                a_opt3, _ = local_refine(best_sequence, n_steps=n_s, seed=s)
                try_update(a_opt3, f"polish_s{s}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        try:
            best_score = float(evaluate_sequence(best_sequence.tolist()))
        except Exception:
            best_score = float('inf')

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    print(f"{'='*50}")
    
    return [float(x) for x in best_sequence]
