"""
Step-function optimization via accelerated mirror descent (multiplicative weights)
with adaptive active-set subgradient on the probability simplex.

The problem is convex: minimize max_k (q*q)[k] subject to q >= 0, sum(q) = 1.
We use the multiplicative weight update (mirror descent with entropy regularizer)
combined with adaptive step sizes and active-set focusing.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len


def _conv_full(q):
    """Full self-convolution of q using FFT."""
    n = len(q)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(2 * n - 1)
    F = fft(q, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def _score_normalized(q):
    """Score for a nonnegative vector (not necessarily normalized)."""
    n = len(q)
    s = q.sum()
    if s < 1e-12:
        return np.inf
    c = _conv_full(q)
    cmax = c.max() if len(c) else 0.0
    return 2.0 * n * cmax / (s * s)


def _subgradient_fft(q, active_lags):
    """
    Compute the subgradient of max_k c[k] at q, averaged over active_lags.
    g[i] = 2 * sum_{k in active} (1/|active|) * q[k-i] (valid indices only).
    Uses FFT for O(n log n) computation.
    """
    n = len(q)
    m = len(active_lags)
    if m == 0:
        return np.zeros(n)
    
    L = next_fast_len(3 * n)  # need >= (2n-1) + n - 1
    w = np.zeros(L)
    w[active_lags] = 1.0 / m
    # g[i] = 2 * (w * qc)[i] where qc = q (cross-correlation)
    # Using: ifft(fft(w) * conj(fft(q))) gives cross-correlation
    g = 2.0 * np.real(ifft(fft(w, L) * np.conj(fft(q, L))))[:n]
    return np.maximum(g, 0.0)


def _mirror_step(q, g, alpha):
    """Multiplicative weight update: q_new = softmax(log(q) - alpha * g)."""
    log_q = np.log(q + 1e-300)
    log_q -= alpha * g
    log_q -= log_q.max()
    q_new = np.exp(log_q)
    s = q_new.sum()
    if s < 1e-300:
        return np.full(len(q), 1.0 / len(q))
    return q_new / s


def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _normalize(q):
    q = np.maximum(q, 0.0)
    s = q.sum()
    if s < 1e-300:
        return np.full(len(q), 1.0 / max(1, len(q)))
    return q / s

    alpha = 1.0 / n  # step size
    beta = 0.9  # momentum
    
    L_conv = next_fast_len(2 * n - 1)
    L_sub = next_fast_len(3 * n)
    
    t = 0
    while time.monotonic() < end:
        t += 1
        beta_t = (1.0 - np.sqrt(1 + 4.0 * t * t)) / (1.0 - 2.0 * t)  # Nesterov beta
        beta_t = (1.0 - np.sqrt(1 + 4.0 / max(t, 1))) / 2.0
        
        y = q + beta_t * (y - q + alpha * np.zeros(n))  # This is getting complicated
        # Let me use the standard Nesterov: y = q + beta_t * (q - q_prev)
        
        # Simplified: use standard momentum
        # y = q + momentum * (q - q_prev)
        # ...
        
        # Compute gradient of smoothed max at y
        Fy = fft(y, L_conv)
        c = np.real(ifft(Fy * Fy))[:2 * n - 1]
        cmax = c.max()
        temp = max(T * cmax, 1e-10)
        
        log_w = (c - cmax) / temp
        log_w -= log_w.max()
        w = np.exp(log_w)
        w /= w.sum()
        
        # Gradient: g = 2 * (w ⊛ y)
        g = 2.0 * np.real(ifft(fft(w, L_sub) * np.conj(fft(y, L_sub))))[:n]
        
        # Projected gradient step
        q_prev = q
        q_new = _project_simplex(y - alpha * g)
        
        score_new = _score_normalized(q_new)
        if score_new < best_score:
            best_score = score_new
            best_q = q_new.copy()
        
        q = q_new
        
        # Decay temperature
        T = max(T * 0.999, 1e-6)
        # Decay step size
        alpha = max(alpha * 0.9999, 1e-6)
    
    return best_q, best_score
