"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines: (1) direct soft-max gradient descent with simplex projection,
(2) LP-based direction finding for global exploration,
(3) targeted local search for final polish.
Uses FFT for fast convolution and HiGHS for LP solving.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * float(np.max(c)) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_soft_max_conv(a, tau):
    """
    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Returns (value, gradient, convolution).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w cross-correlate a at i)
    # = 2 * conv(w, a_rev)[i + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    corr = fft_convolve(w, a_rev)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c


def build_constraint_matrix(a):
    """Build the (2n-1) x n constraint matrix for (a*b)[k] = sum_j A[k,j]*b[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_direction(a, M, upper_bound=10.0):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, 0 <= b <= upper_bound.
    Returns the optimal b (normalized to sum=1) or None if infeasible.
    """
    n = len(a)
    A = build_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            b_opt = result.x
            s = np.sum(b_opt)
            if s > 1e-10:
                return b_opt / s
    except Exception:
        pass
    return None


def optimize_direct_gradient(a_init, time_limit, rng):
    """
    Direct gradient descent with soft-max temperature annealing.
    Optimizes a directly on the simplex (no reparameterization).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score

    tau_start = 5.0 / n
    tau_end = 0.05 / n
    lr = 0.1
    patience = 0
    max_patience = 3000

    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed * 1.3)

        value, g, c = grad_soft_max_conv(a, tau)
        # Project gradient to tangent space of simplex (remove mean)
        g_mean = np.mean(g)
        g_tangent = g - g_mean
        a = a - lr * g_tangent
        a = project_simplex(a)

        step += 1
        score = local_score(a)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive learning rate
        if patience > 1000 and patience % 1000 == 0:
            lr *= 0.7
            # Try a perturbation
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = project_simplex(a)

        if patience > max_patience:
            break

        if step % 2000 == 0:
            print(f"    direct_grad step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, lr={lr:.4f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def optimize_softmax_gradient(n, theta_init, time_limit, rng):
    """
    Gradient descent on softmax parameterization with annealed temperature.
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr_base = 2.5
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = np.inf
    tau_start = 40.0 / n
    tau_end = 0.2 / n
    patience = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed * 1.1)
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = lr_base / np.sqrt(step + 1)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        a_check = softmax_normalize(theta) * S
        score = local_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            patience = 0
        else:
            patience += 1

        if patience > 5000:
            break

        if step % 3000 == 0:
            print(f"    softmax_grad step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

    return best_theta, best_score


def lp_based_optimization(a_init, time_limit, rng):
    """
    LP-based iterative method: find descent directions via LP solving.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)
    M = float(np.max(fft_convolve(a, a)))
    a_norm = np.sum(a)  # = 1
    best_a = a.copy()
    best_score = local_score(a)

    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        # Solve LP for direction
        b_dir = solve_lp_direction(a, M, upper_bound=5.0)

        if b_dir is not None:
            # Try multiple step sizes
            improved = False
            for t in [0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
                a_new = (1.0 - t) * a + t * b_dir
                a_new = np.clip(a_new, 0.0, None)
                s = np.sum(a_new)
                if s < 1e-10:
                    continue
                a_new = a_new / s
                M_new = float(np.max(fft_convolve(a_new, a_new)))
                if M_new < M * 1.001:
                    a = a_new
                    M = M_new
                    improved = True
                    break
            if not improved:
                # Try smaller steps
                for t in [0.05, 0.1, 0.15, 0.2]:
                    a_new = (1.0 - t) * a + t * b_dir
                    a_new = np.clip(a_new, 0.0, None)
                    s = np.sum(a_new)
                    if s < 1e-10:
                        continue
                    a_new = a_new / s
                    M_new = float(np.max(fft_convolve(a_new, a_new)))
                    if M_new < M:
                        a = a_new
                        M = M_new
                        improved = True
                        break

        # Occasionally add perturbation
        if iteration % 20 == 0:
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = project_simplex(a)

        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if iteration % 10 == 0:
            print(f"    LP opt iter={iteration}, M={M:.6f}, score={score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def polish_local_search(a_init, time_limit, rng):
    """
    Simulated annealing local search with targeted peak reduction.
    """
    current = a_init.astype(np.float64).copy()
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        candidate = current.copy()
        r = rng.random()

        conv = fft_convolve(current, current)
        peak_idx = int(np.argmax(conv))

        if r < 0.3:
            # Targeted: reduce peak contributors
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.012, size=n_inc))
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.65:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.8:
            # Symmetrize partially with noise
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.008, size=n))
        elif r < 0.9:
            # Second peak reduction
            sorted_peaks = np.argsort(conv)[::-1]
            second_peak = sorted_peaks[1] if len(sorted_peaks) > 1 else peak_idx
            contribs2 = np.zeros(n)
            for i in range(n):
                j = second_peak - i
                if 0 <= j < n:
                    contribs2[i] = current[i] * current[j]
            top_k2 = max(2, n // 20)
            top_idx2 = np.argsort(contribs2)[-top_k2:]
            candidate[top_idx2] *= np.exp(rng.normal(-0.02, 0.01, size=len(top_idx2)))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        s = np.sum(candidate)
        if s > 1e-10:
            candidate = candidate / s

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling

        if iteration % 2000 == 0:
            print(f"    polish iter={iteration}, score={current_score:.6f}, best={best_score:.6f}, temp={temperature:.2e}, t={time.monotonic()-t0:.1f}s")

    return best, best_score


def make_chirp_init(n, alpha, rng):

    a += 0.015 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    num_tones = rng.integers(3, 8)

    a += 0.015 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
