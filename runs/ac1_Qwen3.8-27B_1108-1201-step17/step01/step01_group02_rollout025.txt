import numpy as np

def score_fn(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)

def subgradient_polish(a, time_limit, verbose=True):
    """Subgradient descent to polish the solution."""
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_fn(a)
    start = time.monotonic()
    eta = 0.02
    patience = 0

    it = 0
    while time.monotonic() - start < time_limit:
        conv = np.convolve(a, a)
        max_val = conv.max()

        # Find positions near peak
        threshold = 0.98 * max_val
        peak_pos = np.where(conv >= threshold)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_pos:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                grad[j] += a[k - j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Try a few step sizes
        best_step = 0
        best_new_score = score_fn(a)
        for e in [eta * 0.5, eta, eta * 1.5, eta * 2.0]:
            a_try = a - e * grad / gnorm
            a_try = normalize(a_try)
            s = score_fn(a_try)
            if s < best_new_score:
                best_new_score = s
                best_step = e

        if best_step == 0:
            eta *= 0.9
            patience += 1
            if patience > 200:
                break
        else:
            a = a - best_step * grad / gnorm
            a = normalize(a)
            patience = 0
            eta = best_step

        if best_new_score < best_score - 1e-12:
            best_score = best_new_score
            best_a = a.copy()

        if verbose and it % 5000 == 0 and it > 0:
            print(f"    SGD iter {it:5d}: score={best_score:.6f}, eta={eta:.6f}, t={time.monotonic()-start:.0f}s")
        it += 1

    if verbose:
        print(f"    SGD done: best={best_score:.6f}, iters={it}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score


def local_search(a, time_limit, rng, verbose=True):
    """Random local search with segment modifications."""
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_fn(a)
    start = time.monotonic()
    accept_count = 0

    while time.monotonic() - start < time_limit:
        # Random perturbation strategy
        strategy = rng.integers(0, 5)

        if strategy == 0:
            # Segment scaling
            width = max(2, int(rng.integers(max(2, n // 300), max(3, n // 6))))
            pos = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[pos:pos+width] *= np.exp(rng.normal(0, 0.15, size=width))
        elif strategy == 1:
            # Global smoothing
            candidate = a.copy()
            kernel_size = int(rng.integers(3, max(5, n // 50)))
            kernel = np.ones(kernel_size) / kernel_size
            candidate = np.convolve(candidate, kernel, mode='same')
        elif strategy == 2:
            # Random element perturbation
            candidate = a.copy()
            n_perturb = int(rng.integers(1, max(2, n // 20)))
            idx = rng.integers(0, n, size=n_perturb)
            candidate[idx] *= np.exp(rng.normal(0, 0.2, size=n_perturb))
        elif strategy == 3:
            # Symmetrize
            candidate = 0.5 * (a + a[::-1])
            candidate *= (1 + rng.normal(0, 0.02))
        else:
            # Frequency-domain perturbation
            candidate = a.copy()
            F = np.fft.rfft(candidate)
            F[1:-1] *= np.exp(rng.normal(0, 0.05, size=len(F[1:-1])))
            candidate = np.fft.irfft(F, n=n)

        candidate = normalize(candidate)
        score = score_fn(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 20 == 0:
                print(f"    LS accept {accept_count:3d}: score={best_score:.6f}, t={time.monotonic()-start:.0f}s")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, accepts={accept_count}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score

    deadline = total_start + budget_s - 30.0  # Leave 30s for final checks

    best_score = float('inf')
    best_seq = None

    # Get starting point from previous best if available
    global height_sequence_1
    start_seqs = []
    try:
        if height_sequence_1 is not None and len(height_sequence_1) > 10:
            hs = np.array(height_sequence_1, dtype=np.float64)
            hs = normalize(hs)
            s = score_fn(hs)
            start_seqs.append((hs, s, len(hs), "prev_best"))
            print(f"Previous best: n={len(hs)}, score={s:.6f}")
    except Exception as e:
        print(f"Could not load previous best: {e}")

    # Define configurations: (n, init_type, time_frac)
    # Weight toward n=500 (previous best length) but explore nearby
    configs = []
    if start_seqs:
        n_prev = len(start_seqs[0][0])
        # Refine previous best
        configs.append((n_prev, 'prev', 0.20))
        # Nearby lengths
        for dn in [-100, -50, 0, 50, 100, 150, 200]:
            n_try = n_prev + dn
            if 100 <= n_try <= 3000:
                configs.append((n_try, 'raised_cosine', 0.06))
    else:
        configs.append((500, 'raised_cosine', 0.15))
        configs.append((800, 'raised_cosine', 0.10))
        configs.append((300, 'raised_cosine', 0.08))

    # Additional diverse initializations
    configs.append((500, 'gaussian', 0.05))
    configs.append((600, 'blackman', 0.05))
    configs.append((500, 'nuttall', 0.05))
    configs.append((700, 'random', 0.05))
    configs.append((450, 'random', 0.04))
    configs.append((1000, 'raised_cosine', 0.05))

    # Sort by time allocation (largest first)
    configs.sort(key=lambda x: -x[2])

    for n, init_type, t_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        time_limit = min(t_frac * budget_s, remaining * 0.7)
        if time_limit < 20.0:
            continue

        print(f"\n=== n={n}, init={init_type}, time={time_limit:.0f}s, remaining={remaining:.0f}s ===")

        if init_type == 'prev':
            a = start_seqs[0][0].copy()
        elif init_type == 'raised_cosine':
            a = init_raised_cosine(n)
        elif init_type == 'gaussian':
            a = init_gaussian(n, sigma_frac=float(rng.uniform(0.15, 0.35)))
        elif init_type == 'blackman':
            a = init_blackman(n)
        elif init_type == 'nuttall':
            a = init_nuttall(n)
        elif init_type == 'random':
            a = init_random_smooth(n, rng)
        else:
            a = init_raised_cosine(n)

        a = normalize(a)
        init_score = score_fn(a)
        print(f"  Init: n={n}, score={init_score:.6f}")

        if init_score < best_score:
            best_score = init_score
            best_seq = a.copy()

        # LP refinement
        sub_budget = time_limit * 0.7
        a_ref, score = lp_refine(a, time_limit=sub_budget, verbose=True, use_relaxed=False)
        if score < best_score:
            best_score = score
            best_seq = a_ref.copy()
            print(f"  *** NEW BEST (LP): score={score:.6f}, n={n}")

        # Subgradient polish if time remains
        remaining_sub = time_limit - (time.monotonic() - (total_start + (budget_s - 30) - remaining))
        # Simpler: check against deadline
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20 and score < best_score + 0.02:
            sg_time = min(30.0, remaining2 * 0.3)
            a_sg, sg_score = subgradient_polish(a_ref, time_limit=sg_time, verbose=False)
            if sg_score < best_score:
                best_score = sg_score
                best_seq = a_sg.copy()
                print(f"  *** NEW BEST (SGD): score={score:.6f}, n={n}")

    # Final polishing phase
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final Polish phase, remaining={remaining:.0f}s ===")
        if remaining > 30:
            # Local search
            ls_time = min(remaining * 0.3, 120.0)
            polished, ls_score = local_search(best_seq, time_limit=ls_time, rng=rng, verbose=True)
            if ls_score < best_score:
                best_score = ls_score
                best_seq = polished.copy()
                print(f"  *** After LS: score={best_score:.6f}")

            # Final LP on polished
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                lp_time = min(remaining2 * 0.7, 180.0)
                final, lp_score = lp_refine(best_seq, time_limit=lp_time, verbose=True, use_relaxed=True)
                if lp_score < best_score:
                    best_score = lp_score
                    best_seq = final.copy()
                    print(f"  *** After final LP: score={best_score:.6f}")

            # One more SGD pass
            remaining3 = deadline - time.monotonic()
            if remaining3 > 20:
                sg_time = min(remaining3 * 0.5, 60.0)
                sg_final, sg_score = subgradient_polish(best_seq, time_limit=sg_time, verbose=False)
                if sg_score < best_score:
                    best_score = sg_score
                    best_seq = sg_final.copy()
                    print(f"  *** After final SGD: score={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = init_raised_cosine(500)
        best_score = score_fn(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Verify
    best_score_verified = score_fn(best_seq)
    print(f"\nFinal: score={best_score_verified:.6f}, n={len(best_seq)}")
    try:
        verify = evaluate_sequence([float(x) for x in best_seq])
        print(f"Verified via evaluate_sequence: {verify:.6f}")
    except Exception as e:
        print(f"Verification error: {e}")

    print(f"Total time: {time.monotonic()-total_start:.1f}s")
    return [float(x) for x in best_seq]
