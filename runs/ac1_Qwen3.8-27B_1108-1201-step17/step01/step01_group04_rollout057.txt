Let me verify: `correlate(w, a, 'full')[j] = sum_m w[j+m]*a[m]` where `m` ranges so that both indices are valid. For `j >= 0`, `m` ranges from `0` to `min(len(a)-1, len(w)-1-j)`.     """Solve LP: max sum(g) s.t. A_a @ g <= M, A_H @ g <= M, g >= 0.

    A = np.concatenate([A_a, A_H], axis=0)

        pass

    tau_start = 20.0
    tau_end = 100.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 15:
            break
        
        progress = it / max_iters
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.003)
        tau = tau_start + (tau_end - tau_start) * progress
        
        a = _projected_subgrad_step(a, lr, tau=tau, target=target)
        
        if (it + 1) % 100 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                if verbose:
                    print(f"    grad iter {it}: score={best_score:.6f}")
    
    return best, best_score


# Global RNG for perturbations (avoid closure issues)
rng_perturb = np.random.default_rng(12345)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple starting sequences using LP-based
    direction finding with adaptive line search and subgradient refinement.
    Returns the best sequence found within the time budget.
    """
    global rng_perturb
    rng_perturb = np.random.default_rng(12345)
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    def _update(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  *** NEW BEST: score={best_score:.6f}, n={len(seq)} ***")

    # --- Generate initial sequences ---
    inits = []

    # Use previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= 32 and np.all(np.isfinite(prev)):
            inits.append(('lp', prev.copy()))
    except NameError:
        pass

    # n=512 initializations
    n = 512
    inits.append(('lp', np.ones(n)))
    
    # Two-level: concentrated mass
    for ratio in [0.3, 0.4, 0.5, 0.6]:
        a = np.ones(n)
        a[:int(n * (1 - ratio))] = 1.8
        a[int(n * (1 - ratio)):] = 0.6
        inits.append(('lp', a.copy()))
    
    # Ramped
    inits.append(('lp', np.linspace(0.5, 1.5, n).copy()))
    inits.append(('lp', np.linspace(1.5, 0.5, n).copy()))
    
    # Sine-modulated
    i = np.arange(n, dtype=np.float64)
    inits.append(('lp', (0.7 + 0.3 * np.sin(2 * np.pi * i * 2.0 / n)).copy()))
    inits.append(('lp', (0.7 + 0.3 * np.sin(2 * np.pi * i * 3.0 / n)).copy()))
    
    # Random
    inits.append(('lp', (rng.random(n) * 0.4 + 0.3).copy()))
    
    # n=256 (faster, might find different basin)
    n2 = 256
    inits.append(('lp', np.ones(n2)))
    a2 = np.ones(n2)
    a2[:128] = 2.0
    inits.append(('lp', a2.copy()))
    
    # n=1024 (more degrees of freedom, gradient method)
    n3 = 1024
    inits.append(('grad', np.ones(n3)))
    a3 = np.ones(n3)
    a3[:512] = 1.5
    a3[512:] = 0.7
    inits.append(('grad', a3.copy()))
    i3 = np.arange(n3, dtype=np.float64)
    inits.append(('grad', (0.6 + 0.4 * np.cos(2 * np.pi * i3 * 5.0 / n3)).copy()))

    print(f"Total initializations: {len(inits)}, budget={budget_s}s")

    # --- Allocate time per initialization ---
    # Give more time to promising ones (previous best, n=512)
    # Sort: previous best first, then n=512 LP, then others
    def _priority(item):
        method, a = item
        if 'height_sequence' in str(type(a)) and False:
            return 0
        if len(a) == 1024:
            return 3
        if len(a) == 256:
            return 4
        return 1  # n=512

    # Keep previous best first
    order = sorted(range(len(inits)), key=lambda idx: (0 if idx == 0 and 'prev' in str(inits[idx]) else _priority(inits[idx])))
    
    # Actually let's just run them in order, with time allocation
    # Allocate: first init gets 40%, rest split evenly
    n_inits = len(inits)
    times_per_init = []
    remaining = deadline - t_start - 30  # 30s reserve for final refinement
    for i in range(n_inits):
        if i == 0:
            times_per_init.append(remaining * 0.35)
        elif i < 4:
            times_per_init.append(remaining * 0.12)
        else:
            times_per_init.append(remaining * 0.06)
    
    # --- Run optimization on each initialization ---
    for i, (method, a0) in enumerate(inits):
        if time.monotonic() > deadline - 20:
            break
        
        alloc_time = min(times_per_init[i], deadline - t_start - 20)
        if alloc_time < 10:
            break
        
        local_deadline = t_start + (t_start + deadline - t_start) * 0 + time.monotonic() + alloc_time
        # Simpler: just use a per-init deadline
        local_deadline = time.monotonic() + alloc_time
        local_deadline = min(local_deadline, deadline - 15)
        
        n_init = len(a0)
        print(f"\n  Init {i+1}/{n_inits}: n={n_init}, method={method}, time={alloc_time:.0f}s")
        t0 = time.monotonic()
        
        if method == 'lp' and n_init <= 1024:
            seq, score = _optimize_lp(a0, local_deadline, max_iters=400)
        else:
            seq, score = _optimize_gradient(a0, local_deadline, max_iters=5000)
        
        _update(seq, score)
        elapsed = time.monotonic() - t0
        print(f"  Done in {elapsed:.1f}s, current best={best_score:.6f}")

    # --- Final refinement of the best sequence ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = len(best_seq)
        print(f"\n  Final refinement: n={n_best}, score={best_score:.6f}")
        
        # Multiple refinement passes
        for pass_i in range(8):
            if time.monotonic() > deadline - 8:
                break
            
            local_dl = deadline - 5
            if n_best <= 1024:
                seq, score = _optimize_lp(best_seq, local_dl, max_iters=200)
            else:
                seq, score = _optimize_gradient(best_seq, local_dl, max_iters=2000)
            
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine pass {pass_i}: score={best_score:.6f}")
            else:
                # Try a different perturbation strategy
                a_try = best_seq + rng_perturb.random(n_best) * 0.03 * best_seq.mean()
                a_try = np.maximum(a_try, 0.0)
                a_try = _normalize(a_try)
                if n_best <= 1024:
                    seq2, score2 = _optimize_lp(a_try, local_dl, max_iters=100)
                else:
                    seq2, score2 = _optimize_gradient(a_try, local_dl, max_iters=1000)
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2.copy()
                    print(f"  Refine pass {pass_i} (perturbed): score={best_score:.6f}")
                else:
                    break

    # --- Final evaluation and return ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\n  Final: n={len(best_seq)}, computed_score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"  evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback: uniform sequence.")
    return [1.0] * 256
