def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    if np.all(v >= 0) and abs(np.sum(v) - target) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _fft_cross_corr(a, w):
    """
    Compute cross-correlation: result[j] = sum_{l=0}^{len(a)-1} a[l]*w[l+j]
    for j = 0, ..., len(a)-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    m = len(w)
    N = 1
    while N < n + m - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    result = np.fft.irfft(np.conj(A) * W, N)[:n]
    return result

    """Compute 2*n*max(conv(a,a)) for a normalized sequence (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _soft_gradient(a, beta):
    """
    Compute the gradient of the softmax-smoothed max of conv(a,a).
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    p = np.exp(beta * (b - b_max))
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        # All weights concentrated on the max
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
        smoothed_val = b_max

        p /= p_sum
        smoothed_val = b_max + np.log(p_sum) / beta
    
    grad = 2.0 * _fft_cross_corr(a, p)
    return grad, smoothed_val

import numpy as np

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fft_cross_corr(a, w):
    """
    Compute cross-correlation: result[j] = sum_{l=0}^{n-1} a[l]*w[l+j]
    for j = 0, ..., n-1 where n = len(a).
    Uses FFT for O((n+m) log(n+m)) computation.
    """
    n = len(a)
    m = len(w)
    N = 1
    while N < n + m - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    result = np.fft.irfft(np.conj(A) * W, N)[:n]
    return result

    """Compute 2*n*max(conv(a,a)) for a normalized sequence (sum=1)."""
    n = len(a)

def _soft_gradient(a, beta):
    """
    Compute the gradient of the softmax-smoothed max of conv(a,a).
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    shifted = beta * (b - b_max)
    # Clip to avoid overflow

    p = np.exp(shifted)

        p[np.argmax(b)] = 1.0
        smoothed_val = b_max
    else:
        p = p / p_sum
        smoothed_val = b_max + np.log(p_sum) / beta

    grad = 2.0 * _fft_cross_corr(a, p)
    return grad, smoothed_val

        print(f"  Improved: {best_score:.6f}")
    else:
        print(f"  No improvement (stays at {best_score:.6f})")

    # Phase 6: Perturbation restarts with short optimization
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_pert = max(1, int(remaining / 40))
        per_pert = remaining / n_pert
        print(f"Phase 6: {n_pert} perturbation restarts, {per_pert:.0f}s each")
        for i in range(n_pert):
            if time.monotonic() >= deadline:
                break
            # Create perturbed version
            a_pert = _perturb_smooth(best_a, rng, scale=0.02 + 0.02 * (i % 3))
            a_p, s_p = _optimize_smooth(a_pert, 50.0, 500.0, per_pert - 2,
                                        lr=0.2, momentum=0.6)
            # Quick subgradient polish
            if per_pert > 10:
                a_p, s_p = _subgradient_polish(a_p, 2.0, top_k=3)
            if s_p < best_score:
                best_score = s_p
                best_a = a_p
                print(f"  Perturb {i+1}: new best {best_score:.6f}")

    # Final verification
    result = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.6f}")
    return result
