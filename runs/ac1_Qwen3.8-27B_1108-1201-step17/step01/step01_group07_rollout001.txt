Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based direction-finding
algorithm. At each step, solves a linear program to find the best direction
g (with g >= 0) that maintains the autocorrelation peak bound M to first
order, then takes a step in that direction. This is analogous to the
"proximity" method in MPSL sequence design. Multiple sequence lengths and
initializations are tried, with a two-phase strategy (scan then deep dive).
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size, s=fft_size)
    return result[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lp_direction(a, R, M):
    """
    Solve the LP to find the best direction g.
    
    maximize sum(g)
    subject to: for all k, sum_j a[k-j]*g[j] <= (M - R[k])/2
                0 <= g[j] <= UB for all j
    
    Returns: g (optimal direction) or None if infeasible.
    """
    n = len(a)
    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:

            A[k, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]

    b = np.maximum(b, 0.0)  # safety

    c = -np.ones(n)

        if res.status == 0:
            return res.x
        elif res.status == 2:  # infeasible
            return None
        else:
            return None
    except Exception:
        return None

    n = len(a)
    a_sum = a.sum()
    a /= a_sum  # normalize to sum=1
    
    R = fast_convolve(a)
    M = R.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    iter_count = 0
    stuck_count = 0

        M = R.max()

            best_obj = current_obj
            best_a = a.copy()

        # Solve LP
        g = lp_direction(a, R, M)

            b = (M_relaxed - R) / 2.0
