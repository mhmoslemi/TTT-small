def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:

def optimize_annealed(n, a_init, time_limit, rng):
    """
    Annealed softmax gradient descent to minimize the score.
    Parameters are in log-space (theta), and a = softmax(theta) * sqrt(2n).
    Returns best (a, score).
    """
    S = np.sqrt(2.0 * n)
    theta = np.log(np.clip(a_init, 1e-10, None) / np.sum(np.clip(a_init, 1e-10, None)))
    
    # Initialize momentum for Adam
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    tau = 1.0 / n  # initial temperature for soft-max of convolution
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Anneal temperature
        progress = (time.monotonic() - t0) / time_limit
        tau = (1.0 / n) * (1.0 + 20.0 * (1.0 - progress))
        
        # Compute a from theta
        theta_shifted = theta - np.max(theta)
        exp_theta = np.exp(theta_shifted)
        a = exp_theta / np.sum(exp_theta) * S
        
        # Compute score and gradient
        c = fft_convolve(a, a)
        # Soft-max of c at temperature tau
        c_max = np.max(c)
        w = np.exp((c - c_max) / tau)
        w /= np.sum(w)
        
        # Gradient of soft_max(c, tau) w.r.t. a
        # d/da_j soft_max = 2 * sum_k w[k+j] * a[k]  (for the appropriate range)
        # Using correlation: g[j] = 2 * (a[::-1] * w)[j + n - 1]
        w_full = np.zeros(2 * n - 1)

        corr = fft_convolve(a[::-1], w_full)
        g = 2.0 * corr[n-1:2*n-1]
        
        # Gradient w.r.t. theta: g_theta = a * (g - sum(a*g)/S)
        # (because da/dtheta_j = a_j * (delta_jk - a_k/S))
        inner = np.dot(a, g) / S
        g_theta = a * (g - inner)
        
        # Adam update
        lr = 1.0 / np.sqrt(step + 1)

        v = beta2 * v + (1 - beta2) * g_theta**2

            best_a = a_check.copy()
    
    return best_a, best_score

    """Fast computation of the score matching evaluate_sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def grad_soft_max_conv(a, tau):
    """
    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Returns (value, gradient_a, convolution).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    # Gradient: g[j] = 2 * sum_k a[k] * w[k+j] (valid range)
    # Computed as cross-correlation
    corr = fft_convolve(a[::-1], w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c


def optimize_annealed(n, a_init, time_limit, rng):
    """
    Annealed softmax Adam gradient descent.
    Parameters in log-space; a = softmax(theta) * sqrt(2n).
    """
    S = np.sqrt(2.0 * n)
    a_init_safe = np.clip(a_init, 1e-10, None)
    theta = np.log(a_init_safe / np.sum(a_init_safe))

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0

    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    tau_start = 5.0 / np.sqrt(n)
    tau_end = 0.1 / np.sqrt(n)

    no_improve = 0
    patience = 5000

    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac * 1.2)

        # Compute a
        a = softmax_normalize(theta) * S

        # Gradient
        _, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)

        # Adam update
        lr = 0.5 / np.sqrt(step + 1)
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate periodically (every step for speed)
        a_check = softmax_normalize(theta) * S
        score = compute_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_check.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience and step > 500:
            # Try a restart with perturbation
            theta += rng.normal(0, 0.1, size=n)
            no_improve = 0

    return best_a, best_score


def polish_local(a_init, time_limit, rng):
    """Simulated annealing local search for final polish."""
    current = a_init.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.3:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(2, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 2:
                inc_idx = rng.choice(others, size=min(4, len(others)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=len(inc_idx)))
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.7:
            # Symmetrize nudge
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.008, size=n))
        elif r < 0.9:
            # Small random subset
            idx = rng.choice(n, size=max(1, n // 20), replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.015, size=len(idx)))
        else:
            # Tiny global
            candidate *= np.exp(rng.normal(0.0, 0.003, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score

        temp *= cooling

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Random initialization around 1."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    1. Establish uniform baseline (theoretically optimal, score=2.0)
    2. Evaluate provided sequence
    3. Gradient optimization from multiple starts
    4. Local search polish
    Returns best sequence found.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 25.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Phase 0: Uniform baseline (score should be exactly 2.0)
    print("=== Phase 0: Uniform baseline ===")
    for n in [128, 256, 512, 1024, 2048]:
        a_uniform = np.ones(n)
        score = compute_score(a_uniform)
        print(f"  Uniform n={n}: score={score:.12f}")
        if score < best_score:
            best_score = score
            best_seq = a_uniform.copy()

    # Phase 1: Check provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = compute_score(start_seq)
        print(f"\n=== Phase 1: Provided sequence ===")
        print(f"  n={len(start_seq)}, score={s:.12f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
            print(f"  *** New best from provided: {best_score:.12f} ***")

    # Phase 2: Gradient optimization
    print(f"\n=== Phase 2: Gradient optimization ===")
    configs = []

    # Optimize from provided sequence
    if start_seq is not None and len(start_seq) >= 64:
        configs.append(('provided', len(start_seq), start_seq))

    # Various chirp and random initializations
    for n in [128, 256, 384, 512, 768, 1024]:
        alpha = 0.3 + 0.2 * rng.random()
        configs.append((f'chirp_{n}', n, make_chirp_init(n, alpha, rng)))
        configs.append((f'rand_{n}', n, make_random_init(n, rng, spread=0.2 + 0.3 * rng.random())))

    # Sort by expected value (prefer shorter sequences first for speed)
    configs.sort(key=lambda x: x[1])

    # Allocate time proportionally
    total_configs = len(configs)
    base_time_per = max(20.0, (deadline - time.monotonic()) / (total_configs + 2))

    for name, n, a0 in configs:
        if time.monotonic() > deadline - 30:
            break
        time_left = deadline - time.monotonic()
        t_alloc = min(base_time_per, time_left * 0.3)
        if t_alloc < 5:
            continue

        print(f"  Optimizing '{name}' (n={n}), time_alloc={t_alloc:.1f}s...")
        t0 = time.monotonic()
        a_opt, score_opt = optimize_annealed(n, a0, t_alloc, rng)
        elapsed = time.monotonic() - t0
        print(f"    score={score_opt:.12f}, time={elapsed:.1f}s")

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.12f} ***")

    # Phase 3: Polish
    print(f"\n=== Phase 3: Local search polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        polish_time = min(120.0, max(5.0, deadline - time.monotonic() - 10))
        print(f"  Starting polish: score={best_score:.12f}, time={polish_time:.1f}s")
        t0 = time.monotonic()
        a_polished, score_polished = polish_local(best_seq, polish_time, rng)
        elapsed = time.monotonic() - t0
        print(f"  After polish: score={score_polished:.12f}, time={elapsed:.1f}s")
        if score_polished < best_score:
            best_score = score_polished
            best_seq = a_polished.copy()
            print(f"  *** New best after polish: {best_score:.12f} ***")

    # Phase 4: Try a few more diverse starts if time permits
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 4: Additional exploration ===")
        remaining = deadline - time.monotonic()
        for trial in range(4):
            if time.monotonic() > deadline - 15:
                break
            n = int(rng.integers([128, 256, 512]))
            a0 = make_random_init(n, rng, spread=0.1 + 0.4 * rng.random())
            t_alloc = min(30.0, remaining / (4 - trial))
            a_opt, score_opt = optimize_annealed(n, a0, t_alloc, rng)
            print(f"  Trial {trial}: n={n}, score={score_opt:.12f}")
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"  *** New best: {best_score:.12f} ***")

    # Final verification with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.12f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    return [1.0] * 256
