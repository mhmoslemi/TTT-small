"""
Hybrid LP + gradient-based optimization to minimize the autocorrelation peak
of a non-negative step function. Score = 2*n*max(f*f)/(sum(f))^2.

Strategy:
  1. Multiple structured initializations per length
  2. LP-based iterative refinement with backtracking line search
  3. Smooth-softmax gradient polish using L-BFGS-B
  4. Multi-length search (512-4096) with global best tracking
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize
import time


def fast_score(f):
    """Compute 2*n*max(conv)/sum^2 without calling evaluate_sequence."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n): A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit_s=8.0):
    """Solve max sum(g) s.t. A*g <= M, g >= 0. Returns normalized g or None."""
    n = int(len(f))
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_step_with_backtrack(f, t_init=0.3, t_min=0.02, t_decay=0.99,
                           lp_time=8.0, max_backtracks=10):
    """One LP step with backtracking to ensure score doesn't increase."""
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return f.copy(), fast_score(f)
    
    g = solve_lp(f, M, lp_time)
    if g is None:
        return f.copy(), fast_score(f)
    
    if np.linalg.norm(g - f) < 1e-10:
        return f.copy(), fast_score(f)
    
    current_score = fast_score(f)
    t = t_init
    for _ in range(max_backtracks):
        f_new = np.maximum((1.0 - t) * f + t * g, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s
        new_score = fast_score(f_new)
        if new_score <= current_score + 1e-12:
            return f_new, new_score
        t = max(t * 0.5, t_min * 0.1)
    
    # If no step worked, return original
    return f.copy(), current_score

    f = start.copy()
    best = f.copy()
    best_sc = fast_score(f)
    t = 0.3
    stall = 0
    rng_local = np.random.default_rng(int(time.time() * 1000) % (2**31))
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        f_new, sc = lp_step_with_backtrack(f, t, lp_time=lp_time)
        
        if sc < best_sc - 1e-10:
            best_sc = sc
            best = f_new.copy()
            stall = 0
        else:
            stall += 1
        
        f = f_new
        t = max(0.02, t * 0.995)
        
        # Periodic perturbation to escape local minima
        if stall > 40:
            w = int(rng_local.integers(1, max(2, len(f) // 8)))
            s0 = int(rng_local.integers(0, max(1, len(f) - w)))
            f[s0:s0+w] *= np.exp(rng_local.normal(0, 0.4, w))
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            stall = 0
            t = 0.3  # reset step size after perturbation
    
    return best, best_sc


def smooth_objective(x, alpha=50.0):
    """Smooth approximation of 2n*max(conv)/sum^2 using log-sum-exp."""
    n = len(x)
    s = x.sum()
    if s < 1e-10:
        return 1e10
    conv = np.convolve(x, x)
    aconv = alpha * conv
    max_val = aconv.max()
    log_sum = max_val + np.log(np.sum(np.exp(aconv - max_val)))
    m_soft = log_sum / alpha
    return 2.0 * n * m_soft / (s * s)


def smooth_gradient(x, alpha=50.0):
    """Gradient of smooth_objective w.r.t. x."""
    n = len(x)
    s = x.sum()
    if s < 1e-10:
        return np.ones(n) * 1e-10
    conv = np.convolve(x, x)
    aconv = alpha * conv
    max_val = aconv.max()
    exp_vals = np.exp(aconv - max_val)
    p = exp_vals / exp_vals.sum()  # softmax weights
    
    # d M_soft / dx_i = sum_k p_k * d conv[k]/dx_i
    # d conv[k]/dx_i = 2*x[k-i] if 0 <= k-i < n
    # So d M_soft/dx_i = 2 * sum_j x[j] * p[j+i]
    # This is 2 * (p flipped convolved with x) 
    # Use FFT for efficiency
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    
    fx = np.fft.rfft(x, fft_size)
    fp = np.fft.rfft(p, fft_size)
    
    # cross-correlation: sum_j x[j]*p[j+i] = ifft(conj(fft(x)) * fft(p))[i]
    # Actually: sum_j x[j]*p[j+i] = (x * reverse(p))[n-1+i] in convolution terms
    # Let's use: R[i] = sum_j x[j] * p[j+i]
    # = ifft(fft(x) * conj(fft(p)))... no
    # R[i] = sum_j x[j] * p[j+i]
    # Let q[j] = p[n-1-j] (reversed p)
    # R[i] = sum_j x[j] * q[n-1-j-i]
    # This is (x*q)[n-1-i] in standard convolution
    # So R = convolve(x, q)[n-1::-1]... 
  
    # Simpler: use the property that sum_j x[j]*p[j+i] is the cross-correlation
    # np.correlate(p, x, 'full')[n-1:] gives sum_j p[j]*x[j+k] for k=0,1,...
    # We want sum_j x[j]*p[j+i] which is the same as sum_j p[j+i]*x[j]
    # = cross_correlation at lag i (p relative to x)
    
    # Let me just compute it directly using the convolution structure
    # d M_soft/dx_i = 2 * sum_{k: 0<=k-i<n} p[k] * x[k-i]
    # = 2 * sum_j p[j+i] * x[j]  (letting j = k-i)
    
    # For efficiency, compute all i at once:
    # grad[i] = 2 * sum_j x[j] * p[j+i] for i = 0,...,n-1
    # This is like a "half cross-correlation"
    
    # Use FFT: ifft(fft(x_padded) * conj(fft(p_padded))) gives cross-correlation
    x_pad = np.zeros(fft_size)
    p_pad = np.zeros(fft_size)
    x_pad[:n] = x
    p_pad[:len(p)] = p
    
    xx = np.fft.rfft(x_pad)
    pp = np.fft.rfft(p_pad)
    
    # cross-correlation: corr[k] = sum_j x[j+k] * conj(p[j])
    # We want sum_j x[j] * p[j+i] = sum_j p[j+i] * x[j]
    # = corr_p_x[i] where corr_p_x[i] = sum_j p[j] * x[j-i]... 
    # Let me use: sum_j x[j] * p[j+i] = sum_j p[j+i] * x[j]
    # Substitute m = j+i: = sum_m p[m] * x[m-i] (for valid m)
    # = (p * reverse(x))[n-1-i] ... this is getting circular.
    
    # Let me just do it with a direct O(n^2) approach for n <= 4096
    # grad[i] = 2 * sum_{j=0}^{n-1} x[j] * p[j+i] where j+i < len(p) = 2n-1
    grad = np.zeros(n)
    # p has length 2n-1
    for i in range(n):
        # sum_j x[j] * p[j+i] for j such that j < n and j+i < 2n-1
        j_max = min(n, 2*n-1-i)
        if j_max > 0:
            grad[i] = 2.0 * np.dot(x[:j_max], p[i:i+j_max])
    
    # Now compute dF/dx where F = 2n * M_soft / S^2
    # dF/dx_i = 2n * (1/S^2) * dM_soft/dx_i - 2n * M_soft * 2/(S^3) * 1
    m_soft = (max_val + np.log(exp_vals.sum())) / alpha
    dF = (2.0 * n / (s * s)) * grad - (4.0 * n * m_soft / (s * s * s))
    
    return dF


def gradient_polish(f, deadline, alpha_start=20.0, alpha_end=100.0):
    """Polish using L-BFGS-B with smooth objective."""
    f = f.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s
    
    best = f.copy()
    best_sc = fast_score(f)
    
    # Anneal alpha from low to high (soft to sharp)
    alphas = np.linspace(alpha_start, alpha_end, 5)
    for alpha in alphas:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            return smooth_objective(x, alpha)
        
        def grad(x):
            return smooth_gradient(x, alpha)
        
        try:
            res = minimize(obj, f, jac=grad, method='L-BFGS-B',
                          bounds=[(0, None)] * len(f),
                          options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxls': 50})
            if res.x is not None:
                f_new = np.maximum(res.x, 0.0)
                sn = f_new.sum()
                if sn > 1e-10:
                    f_new = f_new / sn
                    sc = fast_score(f_new)
                    if sc < best_sc:
                        best_sc = sc
                        best = f_new.copy()
                        f = f_new
        except Exception:
            pass
    
    return best


def block_perturb(f, rng, strength=0.3):
    """Random block perturbation for exploration."""
    n = len(f)
    cand = f.copy()
    n_blocks = int(rng.integers(1, 4))
    for _ in range(n_blocks):
        w = int(rng.integers(1, max(2, n // 10)))
        s0 = int(rng.integers(0, max(1, n - w)))
        cand[s0:s0+w] *= np.exp(rng.normal(0, strength, w))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f.copy()
    return cand / s


def propose_candidate(seed=42, budget_s=1800):
    """Main search: find best non-negative sequence minimizing the score."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    
    # Try to load previous best
    best_seq = None
    best_score = np.inf
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            best_seq = parent.copy()
            best_score = fast_score(parent)
            # Verify with actual evaluator
            try:
                vs = float(evaluate_sequence(parent.tolist()))
                if np.isfinite(vs) and vs < best_score:
                    best_score = vs
            except Exception:
                pass
            print("Loaded parent: n=%d, score=%.6f" % (parent.size, best_score), flush=True)
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_initial(512, 'uniform', rng)
        best_score = fast_score(best_seq)
        print("Starting fresh: score=%.6f" % best_score, flush=True)
    
    def try_update(seq):
        nonlocal best_seq, best_score
        sc = fast_score(seq)
        if np.isfinite(sc) and sc < best_score:
            # Verify with authoritative evaluator
            try:
                sc2 = float(evaluate_sequence(seq.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = np.maximum(seq, 0.0).copy()
                    print("  NEW BEST: score=%.6f (n=%d)" % (best_score, seq.size), flush=True)
            except Exception:
                pass
    
    # Phase 1: LP-based optimization from multiple starts
    print("\n=== Phase 1: LP-based optimization ===", flush=True)
    
    lengths = [512, 1024, 2048]
    if best_seq is not None and best_seq.size not in lengths and best_seq.size <= 4096:
        lengths.insert(0, int(best_seq.size))
    
    methods = ['cosine', 'gaussian', 'two_peaks', 'four_peaks', 'edges_high',
               'raised_cos', 'alternating_damped', 'perturbed_uniform']
    
    # Budget allocation: more time for smaller sequences (faster LP)
    total_combos = len(lengths) * len(methods)
    time_per = max(20.0, (budget_s * 0.55) / total_combos)
    time_per = min(time_per, 120.0)
    
    # Prioritize: start with the best sequence's length first
    if best_seq is not None:
        n_best = int(best_seq.size)
        # Also try the best sequence itself
        if n_best <= 4096:
            print("\n[LP] Polishing parent: n=%d, budget=%.0fs" % (n_best, time_per * 2), flush=True)
            dl = min(deadline, time.monotonic() + time_per * 2)
            lp_best, lp_sc = lp_optimize(best_seq.copy(), dl, lp_time=min(10.0, time_per / 3))
            if np.isfinite(lp_sc):
                try_update(lp_best)
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        for method in methods:
            if time.monotonic() > deadline:
                break
            start = make_initial(n, method, rng)
            dl = min(deadline, time.monotonic() + time_per)
            lp_best, lp_sc = lp_optimize(start, dl, lp_time=min(8.0, time_per / 4))
            if np.isfinite(lp_sc):
                try_update(lp_best)
    
    # Phase 2: Extended LP on the current best
    print("\n=== Phase 2: Extended LP on best ===", flush=True)
    if best_seq is not None and best_seq.size <= 4096:
        ext_dl = min(deadline, time.monotonic() + (budget_s * 0.25))
        lp_best, lp_sc = lp_optimize(best_seq.copy(), ext_dl, lp_time=10.0)
        if np.isfinite(lp_sc):
            try_update(lp_best)
    
    # Phase 3: Gradient-based polish
    print("\n=== Phase 3: Gradient polish ===", flush=True)
    if best_seq is not None and best_seq.size <= 4096:
        grad_dl = min(deadline, time.monotonic() + (budget_s * 0.15))
        polished = gradient_polish(best_seq.copy(), grad_dl)
        try_update(polished)
    
    # Phase 4: Final short LP passes with perturbation
    print("\n=== Phase 4: Final refinement ===", flush=True)
    if best_seq is not None and best_seq.size <= 4096:
        while time.monotonic() < deadline:
            perturbed = block_perturb(best_seq, rng, strength=0.15)
            short_dl = min(deadline, time.monotonic() + 60.0)
            lp_best, lp_sc = lp_optimize(perturbed, short_dl, lp_time=5.0)
            if np.isfinite(lp_sc):
                try_update(lp_best)
            if time.monotonic() > deadline:
                break
    
    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    if best_seq.size == 0 or not np.isfinite(best_seq).all() or best_seq.sum() < 0.01:
        best_seq = np.ones(256, dtype=np.float64)
    
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print("\nFINAL: score=%.6f, n=%d" % (final_score, best_seq.size), flush=True)
    
    return [float(x) for x in best_seq]
