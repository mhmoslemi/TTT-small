```python
def lp_min_max_cross(p):
    """Solve: min M s.t. conv(p,q) <= M, sum(q)=1, q>=0."""
    n = len(p)
    # Variables: q_0, ..., q_{n-1}, M  (n+1 variables)
    # Minimize: M  => c_obj = [0,0,...,0,1]
    # Constraints:
    #   conv(p,q)[k] - M <= 0 for k=0,...,2n-2  (2n-1 inequalities)
    #   sum(q) = 1  (1 equality)
    #   q_i >= 0  (bounds)
    
    # Build the constraint matrix
    # For each k, conv(p,q)[k] = sum_j A[k,j] * q[j]
    # where A[k,j] = p[k-j] if 0 <= k-j < n, else 0
    
    # Inequality constraints: A[k,j]*q[j] - M <= 0
    # As a row: [A[k,0], ..., A[k,n-1], -1] @ [q, M] <= 0
    
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            data.append(p[k-j])
        # Add -1 for M (variable n)
        rows.append(k)
        cols.append(n)
        data.append(-1.0)
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(q) = 1
    A_eq = csr_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n+1))
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]  # q_i >= 0, M free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'disp': False})
    
    if res.status == 0 and res.x is not None:
        q = res.x[:n]
        M_opt = res.x[n]
        return np.maximum(q, 0), M_opt
    return None, np.inf
```

```python
"""
LP-based alternating optimization for minimum peak autocorrelation step functions.

Core approach:
  - Score = 2*n*max(conv(p,p))/sum(p)^2 is scale-invariant.
  - At each step, solve an LP: min M s.t. conv(p,q)[k] <= M, sum(q)=1, q>=0.
  - Blend p toward the LP solution q to reduce peak autocorrelation.
  - Iterate with line search, symmetrization, and length exploration.
  - Final gradient-based polish using power-mean surrogate.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_len(n):
    return int(next_fast_len(max(1, 2 * n - 1)))


def _convolve_full(p):
    """Full self-convolution via FFT."""
    n = len(p)
    if n == 0:
        return np.array([])
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def _score(p):
    """Compute evaluation score (internal, assumes valid input)."""
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s < 0.01:
        return np.inf
    c = _convolve_full(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))


def _safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 1e-15:
        return np.full(len(p), 1.0 / max(1, len(p)))
    return p / s


def _symmetrize(p):
    """Symmetrize: p_new[i] = (p[i] + p[n-1-i]) / 2."""
    n = len(p)
    p_sym = 0.5 * (p + p[::-1])
    return _safe_normalize(p_sym)


def _build_conv_matrix(p):
    """Build sparse matrix A where A[k,j] = p[k-j] if 0<=k-j<n else 0."""
    n = len(p)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            data.append(p[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_min_max_cross(p):
    """
    Solve: min M s.t. conv(p,q)[k] <= M for all k, sum(q)=1, q>=0.
    Variables: [q_0,...,q_{n-1}, M]
    Returns (q, M_opt) or (None, inf).
    """
    n = len(p)
    if n < 2:
        return np.array([1.0]), 1.0

    A_conv = _build_conv_matrix(p)

    # Inequality constraints: conv(p,q)[k] - M <= 0
    # Matrix shape: (2n-1, n+1), last column is -1 for M
    n_ineq = 2 * n - 1
    A_ub_data = []
    A_ub_row = []
    A_ub_col = []
    for k in range(n_ineq):
        # Add conv terms
        start = A_conv.indptr[k]
        end = A_conv.indptr[k + 1]
        for idx in range(start, end):
            A_ub_row.append(k)
            A_ub_col.append(A_conv.indices[idx])
            A_ub_data.append(A_conv.data[idx])
        # Add -1 for M (variable index n)
        A_ub_row.append(k)
        A_ub_col.append(n)
        A_ub_data.append(-1.0)

    A_ub = csr_matrix((A_ub_data, (A_ub_row, A_ub_col)), shape=(n_ineq, n + 1))
    b_ub = np.zeros(n_ineq)

    # Equality: sum(q) = 1
    A_eq = csr_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])

    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'disp': False})
        if res.status == 0 and res.x is not None:
            q = np.maximum(res.x[:n], 0.0)
            M_opt = float(res.x[n])
            s = float(np.sum(q))
            if s > 1e-15:
                q = q / s
            return q, M_opt
    except Exception:
        pass

    return None, np.inf

def _gradient_descent(p, time_limit, label=""):
    """
    Gradient-based refinement using power-mean surrogate.
    Minimize (sum c_i^p)^(1/p) on the simplex for large p.
    """
    p = _safe_normalize(p.copy())
    n = len(p)
    best_p = p.copy()
    best_score = _score(p)
    current = best_score
    start_time = time.time()
    rng = np.random.default_rng(12345)

    power = 12.0
    base_lr = 0.02
    it = 0
    stall = 0

    while time.time() - start_time < time_limit:
        it += 1
        c = _convolve_full(p)
        cmax = float(np.max(c))
        if cmax < 1e-30:
            break

        # Power-mean gradient
        cpow = np.power(np.maximum(c, 0), power - 1)
        cpow_sum = float(np.sum(cpow))
        if cpow_sum < 1e-30:
            break
        w = cpow / cpow_sum  # weights over lags

        # Gradient of power mean w.r.t. p:
        # d/dp_j [sum_k w_k * c_k] where c_k = sum_i p_i * p_{k-i}
        # = sum_k w_k * (p_{k-j} + p_{j-k}) for valid indices
        # Use FFT: grad = 2 * convolve(w_reversed, p) (circular, truncated)
        L = _fft_len(n)
        Fp = fft(p, L)
        Fw = fft(w[::-1] + w[1:][::-1] if n > 1 else w, L)
        # Actually, let's compute directly via the relation:
        # c_k = sum_i p_i * p_{k-i}, so dc_k/dp_j = p_{k-j} + p_{j-k} (with boundary handling)
        # Simpler: use the fact that grad_j = 2 * sum_k w_k * p_{k-j} (for the 'valid' part)
        # Let's use convolution:
        # h_j = sum_k w_k * p_{k-j}
        # This is (w * p)_j = sum_k w_k * p_{k-j}
        # = conv(w, p)[j + (n-1)] ... it's a cross-correlation
        # Actually h = ifft(FFT(p) * conj(FFT(w))) ... let me just do it explicitly for small n
        # For large n, use FFT:
        h = np.zeros(n)
        for j in range(n):
            # h_j = sum_k w_k * p_{k-j} where k-j must be in [0, n-1]
            # k ranges from j to j + n - 1 (capped at 2n-2)
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            p_idx = np.arange(k_lo - j, k_hi - j + 1)  # = 0, 1, ..., k_hi-j
            w_idx = np.arange(k_lo, k_hi + 1)
            h[j] = np.dot(w[w_idx], p[p_idx])
        grad = 2.0 * h

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-30:
            break

        # Projected gradient step (with multiple step sizes)
        accepted = False
        for lr in [base_lr / gnorm, 0.5 * base_lr / gnorm, 0.1 * base_lr / gnorm,
                   0.02 * base_lr / gnorm, 0.005 * base_lr / gnorm]:
            v = p - lr * grad
            v = np.maximum(v, 0)
            vs = float(np.sum(v))
            if vs < 1e-12:
                continue
            v = v / vs
            sc = _score(v)
            if np.isfinite(sc) and sc < current:
                current = sc
                p = v
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    stall = 0
                accepted = True
                break

        if not accepted:
            stall += 1
            if stall > 50:
                # Random perturbation
                r = p + rng.normal(0, 0.05 / n, n)
                r = np.maximum(r, 0)
                r = r / max(1e-15, np.sum(r))
                sc = _score(r)
                if np.isfinite(sc):
                    p = r
                    current = sc
                    if sc < best_score:
                        best_score = sc
                        best_p = p.copy()
                        stall = 0
            if stall > 200:
                break

        if it % 50 == 0:
            elapsed = time.time() - start_time
            print(f"{label} grad iter={it} t={elapsed:.0f}s best={best_score:.10f}", flush=True)

        if best_score < 1.5030 and stall > 100:
            break

    return best_p, float(best_score)


def _resize_sequence(p, new_n):
    """Resize sequence via linear interpolation + resampling."""
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    return _safe_normalize(p_new)


def _make_initials(n, parent=None):
    """Generate diverse initial points."""
    inits = []
    rng = np.random.default_rng(777)

    # Uniform
    inits.append(np.full(n, 1.0 / n))

    # Symmetric cap (Gaussian)
    x = np.linspace(0, 1, n)
    for sigma in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        g = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
        inits.append(_safe_normalize(g))

    # Power shapes
    for a in [0.2, 0.5, 1.0, 2.0, 3.0]:
        p = np.power(np.minimum(x, 1 - x) * 2, a)
        inits.append(_safe_normalize(p))

    # Cosine window
    for a in [0.5, 1.0, 2.0]:
        c = np.cos(np.pi * (x - 0.5)) ** a
        inits.append(_safe_normalize(np.maximum(c, 0)))

    # Beta-like
    for alpha in [0.3, 0.5, 1.0, 2.0, 3.0, 5.0]:
        b = np.power(np.maximum(x, 1e-10), alpha) * np.power(np.maximum(1 - x, 1e-10), alpha)
        inits.append(_safe_normalize(b))

    # Sine window
    for a in [1.0, 2.0, 3.0]:
        s = np.sin(np.pi * x) ** a
        inits.append(_safe_normalize(s))

    # Random smooth
    for _ in range(8):
        f1 = rng.integers(1, max(2, n // 100))
        f2 = rng.integers(1, max(2, n // 50))
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        p = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * x + ph1) + 0.3 * np.sin(2 * np.pi * f2 * x + ph2)
        inits.append(_safe_normalize(np.maximum(p, 0.01)))

    # Parent-based
    if parent is not None and len(parent) > 0:
        p_resized = _resize_sequence(parent, n)
        inits.append(p_resized)
        # Symmetrized parent
        inits.append(_symmetrize(p_resized))
        # Smoothed parent
        from scipy.ndimage import gaussian_filter1d
        p_smooth = gaussian_filter1d(p_resized, sigma=n // 50)
        inits.append(_safe_normalize(p_smooth))

    # Deduplicate and score
    scored = []
    for p in inits:
        sc = _score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    return scored


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for nonnegative step function minimizing 2*n*max(conv(p,p))/sum(p)^2.
    Uses LP-based alternating optimization as the main engine.
    """
    start_time = time.time()
    deadline = start_time + float(budget_s) - 15.0  # safety margin
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0.01:
            parent = _safe_normalize(parent_arr)
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}", flush=True)
    else:
        print("No parent available, starting fresh.", flush=True)
        parent = np.full(700, 1.0 / 700)

    best_p = parent.copy()
    best_score = _score(parent)
    print(f"Initial best: {best_score:.10f}", flush=True)

    def try_update(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = _safe_normalize(p)
        if not np.isfinite(sc):
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"  *** New best: {best_score:.10f} (n={len(p)})", flush=True)

    # Phase 1: LP alternating on parent length and nearby lengths
    n_parent = len(parent)
    lengths_to_try = []
    lengths_to_try.append(n_parent)
    # Try a few nearby lengths
    for dn in [-50, -30, -10, 10, 30, 50, 100]:
        nl = n_parent + dn
        if 100 <= nl <= 3000:
            lengths_to_try.append(nl)
    # Also try some round numbers
    for nl in [400, 500, 600, 800, 1000]:
        if nl not in lengths_to_try and 100 <= nl <= 3000:
            lengths_to_try.append(nl)

    lengths_to_try = sorted(set(lengths_to_try))
    print(f"Phase 1: Trying lengths {lengths_to_try}", flush=True)

    # Allocate time: prefer parent length, then others
    time_per_lp = {}
    total_lp_time = (deadline - start_time) * 0.55  # 55% for LP
    n_lens = len(lengths_to_try)
    # Give parent length 40% of LP time, rest split among others
    time_per_lp[n_parent] = total_lp_time * 0.4
    remaining_lp = total_lp_time * 0.6
    for nl in lengths_to_try:
        if nl != n_parent:
            time_per_lp[nl] = remaining_lp / max(1, n_lens - 1)

    for nl in lengths_to_try:
        if time.time() > deadline - 300:
            break
        if nl not in time_per_lp:
            continue
        t_alloc = time_per_lp[nl]
        if t_alloc < 30:
            t_alloc = 30
        t_alloc = min(t_alloc, deadline - time.time() - 300)
        if t_alloc < 20:
            break

        # Get initial for this length
        p_init = _resize_sequence(parent, nl)
        p_sym = _symmetrize(p_init)
        sc_init = _score(p_init)
        sc_sym = _score(p_sym)
        try_update(p_init, sc_init)
        try_update(p_sym, sc_sym)

        # Run LP alternating
        p_best, sc_best = _lp_alternating(p_init, t_alloc, label=f"LP(n={nl})")
        try_update(p_best, sc_best)

        # Also try from symmetrized start
        if time.time() < deadline - 350:
            t2 = min(t_alloc * 0.5, deadline - time.time() - 350)
            if t2 > 20:
                p_best2, sc_best2 = _lp_alternating(p_sym, t2, label=f"LP-sym(n={nl})")
                try_update(p_best2, sc_best2)

    print(f"\nAfter Phase 1: best={best_score:.10f}", flush=True)

    # Phase 2: Multi-start LP from diverse initializations at best length
    if time.time() < deadline - 200:
        n_best = len(best_p)
        print(f"Phase 2: Multi-start at n={n_best}", flush=True)
        initials = _make_initials(n_best, parent=best_p)

        # Sort by score, take top 8
        initials = initials[:8]
        t_per_init = min(40.0, (deadline - time.time() - 200) / max(1, len(initials)))

        for sc, p_init in initials:
            if time.time() > deadline - 220:
                break
            try_update(p_init, sc)
            t_alloc = min(t_per_init, deadline - time.time() - 220)
            if t_alloc < 15:
                break
            p_best, sc_best = _lp_alternating(p_init, t_alloc, label=f"LP-ms(n={n_best})")
            try_update(p_best, sc_best)

    print(f"\nAfter Phase 2: best={best_score:.10f}", flush=True)

    # Phase 3: Gradient-based polish on the best
    if time.time() < deadline - 60:
        n_best = len(best_p)
        t_grad = min(300.0, deadline - time.time() - 30)
        print(f"Phase 3: Gradient polish at n={n_best}, budget={t_grad:.0f}s", flush=True)
        p_best, sc_best = _gradient_descent(best_p, t_grad, label="GRAD")
        try_update(p_best, sc_best)

    # Phase 4: Final LP polish
    if time.time() < deadline - 30:
        n_best = len(best_p)
        t_final = min(120.0, deadline - time.time() - 15)
        print(f"Phase 4: Final LP at n={n_best}, budget={t_final:.0f}s", flush=True)
        p_best, sc_best = _lp_alternating(best_p, t_final, label="FINAL-LP")
        try_update(p_best, sc_best)

    # Final: try symmetrization
    p_sym_final = _symmetrize(best_p)
    sc_sym_final = _score(p_sym_final)
    try_update(p_sym_final, sc_sym_final)

    # Ensure valid output
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(np.sum(best_p))
    if s < 0.01:
        best_p = np.full(len(best_p), 0.01 / max(1, len(best_p)))
    best_p = np.minimum(best_p, 1000.0)

    final_score = _score(best_p)
    print(f"\nFinal result: n={len(best_p)}, score={final_score:.10f}", flush=True)
    print(f"Total time: {time.time() - start_time:.1f}s", flush=True)

    return [float(x) for x in best_p]
```
