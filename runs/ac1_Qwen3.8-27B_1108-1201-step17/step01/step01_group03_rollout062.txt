def mw_step(a, eta, n_active=10):
    """One multiplicative weights step."""
    n = len(a)
    corr = fast_autocorr(a)
    # Find the top n_active lags
    top_lags = np.argsort(corr)[-n_active:]
    # Compute the subgradient
    g = np.zeros(n)
    for k in top_lags:
        # g_m += 2 * a[k-m] for 0 <= k-m < n
        # This is a shifted version of a
        for m in range(n):

    # Multiplicative update

    a_new = a_new / a_new.sum()

Multi-phase optimization to minimize the autocorrelation peak of a non-negative
step function. Combines parametric initialization, multiplicative weights local
search, and LP-based refinement.

Objective: score = 2 * n * max(a*a) / (sum(a))^2
"""
import numpy as np
import time
import scipy.fft as sfft
import scipy.sparse as sp
from scipy.optimize import linprog, minimize_scalar, minimize
import sys

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT padding."""
    n = len(a)
    size = 1
    target = 2 * n - 1
    while size < target:
        size *= 2
    A = sfft.rfft(a, size)
    corr = sfft.irfft(A * A, size)[:target]
    # Clip tiny negatives from numerical error
    return np.maximum(corr, 0.0)

def compute_score(a):
    """Compute the score (same as evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = len(a)
    corr = fast_autocorr(a)
    return float(2.0 * n * corr.max() / (s * s))

def normalize(a):
    """Normalize to sum 1."""
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_chirp(n, alpha=0.8, beta=1.0, gamma=0.0):
    """Chirp sequence: 1 + alpha * cos(2*pi*beta*m^2/n + gamma)."""
    m = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * beta * m * m / n + gamma)
    return np.maximum(a, 0.01)

def make_harmonic(n, alpha=0.8, beta=1.0, gamma=0.0):
    """Harmonic: 1 + alpha * cos(2*pi*beta*m/n + gamma)."""
    m = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * beta * m / n + gamma)
    return np.maximum(a, 0.01)

def make_gaussian(n, width=0.3, center=0.5):
    """Gaussian bump."""
    m = np.arange(n, dtype=np.float64) / n
    a = np.exp(-0.5 * ((m - center) / width) ** 2)
    return np.maximum(a, 0.01)

def make_polynomial(n, power=1.0):
    """Polynomial: (1 + cos(2*pi*m/n))^power."""
    m = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(2.0 * np.pi * m / n)) ** power
    return np.maximum(a, 0.01)

def make_dual_chirp(n, alpha=0.8, beta=1.0):
    """Dual chirp: 1 + alpha * cos(2*pi*beta*(m-n/2)^2/n)."""
    m = np.arange(n, dtype=np.float64)
    centered = m - n / 2.0
    a = 1.0 + alpha * np.cos(2.0 * np.pi * beta * centered * centered / n)
    return np.maximum(a, 0.01)

def make_random(n, rng):
    a = rng.random(n)
    return np.maximum(a, 0.01)

def compute_subgradient(a, top_k=20):
    """Compute subgradient of max autocorrelation."""
    n = len(a)
    corr = fast_autocorr(a)
    # Get top_k lags
    idx = np.argpartition(corr, -top_k)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        segment = a[lo:k+1][::-1]
        g[lo:hi+1] += 2.0 * segment
    g /= top_k
    return g, corr

def mw_optimize(a, eta0=0.1, n_steps=10000, deadline=None, top_k=20):
    """Multiplicative weights optimization."""
    a = normalize(np.asarray(a, dtype=np.float64).copy())
    best = a.copy()
    best_score = compute_score(a)
    eta = eta0
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g, corr = compute_subgradient(a, top_k)
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        score = compute_score(a_new)
        if score < best_score:
            best = a_new.copy()
            best_score = score
        
        # Keep the update if it improved, otherwise revert and reduce eta
        if score > best_score * 1.01:
            eta *= 0.9
        else:
            a = a_new
            eta = max(eta * 0.999, 0.001)
    
    return best, best_score

def lp_refine(a, n_iterations=50, deadline=None, lp_time=5.0):
    """LP-based refinement (same as previous approach)."""
    a = normalize(np.asarray(a, dtype=np.float64).copy())
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    t = 0.3
    t_decay = 0.98
    t_min = 0.05
    
    for it in range(n_iterations):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        corr = fast_autocorr(a)
        M = float(corr.max())
        if M < 1e-15:
            break
        
        # Build sparse A matrix
        i_idx = np.tile(np.arange(n), n)
        j_idx = np.repeat(np.arange(n), n)
        row_idx = j_idx + i_idx
        data = a[i_idx]
        A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
        
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'time_limit': lp_time, 'presolve': True})
            if res.status != 0:
                continue
            g = np.maximum(res.x, 0.0)
        except Exception:
            continue
        
        gs = g.sum()
        if gs < 1e-10:
            continue
        g = g / gs
        
        if np.linalg.norm(g - a) < 1e-8:
            t *= 0.9
            continue
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        score = compute_score(a_new)
        if score < best_score:
            best = a_new.copy()
            best_score = score
            a = a_new
            t = max(t_min, t * t_decay)
        else:
            t *= 0.9
    
    return best, best_score

def block_perturb(a, rng, max_block=50):
    """Random block multiplicative perturbation."""
    n = len(a)
    cand = a.copy()
    w = int(rng.integers(1, min(max_block, n)))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.3)))
    cand = np.maximum(cand, 0.0)
    return normalize(cand)

"""
Multi-phase optimization to minimize the autocorrelation peak of a non-negative
step function, using:
1. FFT-based soft-max gradient descent with epsilon annealing
2. LP-based refinement for final polish
3. Multi-start with diverse parametric initializations

Objective: score = 2 * n * max(a*a) / (sum(a))^2
"""
import numpy as np
import time
import scipy.fft as sfft
import scipy.sparse as sp
from scipy.optimize import linprog

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    target = 2 * n - 1
    size = 1
    while size < target:
        size <<= 1
    A = sfft.rfft(a, size)
    corr = sfft.irfft(A * A, size)[:target]
    return np.maximum(corr, 0.0)

def compute_score(a):
    """Compute the objective score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = len(a)
    corr = fast_autocorr(a)
    return float(2.0 * n * corr.max() / (s * s))

def normalize(a):
    """Normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_chirp(n, alpha=0.8, beta=1.0, gamma=0.0):
    m = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * beta * m * m / (n * n) + gamma)
    return np.maximum(a, 0.01)

def make_harmonic(n, alpha=0.8, freq=1):
    m = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * m / n)
    return np.maximum(a, 0.01)

def make_mixed(n, alpha=0.7, beta=1.0, freq=3):
    m = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0*np.pi*beta*m*m/(n*n)) * np.cos(2.0*np.pi*freq*m/n)
    return np.maximum(a, 0.01)

def make_random(n, rng):
    return np.maximum(rng.random(n), 0.01)

def soft_max_gradient(a, corr, epsilon):
    """Compute soft-max gradient using FFT cross-correlation."""
    n = len(a)
    M = corr.max()
    weights = np.exp((corr - M) / epsilon)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n), M
    weights /= wsum
    
    # grad[m] = 2 * sum_k weights[k] * a[k-m] for valid k-m
    # = 2 * cross_correlation(weights, a)[m]
    a_rev = a[::-1]
    size = 1
    target = len(weights) + n - 1
    while size < target:
        size <<= 1
    W = sfft.rfft(weights, size)
    Ar = sfft.rfft(a_rev, size)
    conv = sfft.irfft(W * Ar, size)
    grad = 2.0 * conv[n-1 : 2*n-1]
    return grad, M

def soft_max_optimize(a, deadline=None, n_steps=500000, eps0=0.1, eps_min=0.01):
    """Soft-max optimization with epsilon annealing."""
    a = normalize(a.copy())
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    epsilon = eps0
    eta = 0.3
    stall = 0
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        corr = fast_autocorr(a)
        M = corr.max()
        grad, _ = soft_max_gradient(a, corr, epsilon)
        
        # Scale gradient for numerical stability
        gmax = grad.max()
        if gmax < 1e-15:
            break
        grad_scaled = grad / gmax
        
        # Multiplicative weights update
        a_new = a * np.exp(-eta * grad_scaled)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score - 1e-10:
            best = a_new.copy()
            best_score = score
            a = a_new
            stall = 0
            # Slightly increase eta if making good progress
            eta = min(eta * 1.001, 1.0)
        else:
            stall += 1
            eta *= 0.95
            if stall > 500:
                # Reheat: increase epsilon temporarily
                epsilon = min(epsilon * 1.5, eps0)
                # Large perturbation
                rng = np.random.default_rng(step)
                a = normalize(a * np.exp(rng.normal(0, 0.3, n)))
                stall = 0
        
        # Anneal epsilon
        if step % 200 == 0:
            epsilon = max(eps_min, epsilon * 0.995)
    
    return best, best_score

def lp_refine(a, deadline=None, max_iters=200, lp_time=5.0):
    """LP-based refinement."""
    a = normalize(a.copy())
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    t = 0.2
    t_decay = 0.99
    t_min = 0.02
    
    # Precompute indices for sparse matrix
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        corr = fast_autocorr(a)
        M = float(corr.max())
        if M < 1e-15:
            break
        
        data = a[i_idx]
        A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
        
        c_vec = -np.ones(n)
        b_vec = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds,
                         method='highs', options={'time_limit': lp_time, 'presolve': True})
            if res.status != 0:
                continue
            g = np.maximum(res.x, 0.0)
        except Exception:
            continue
        
        gs = g.sum()
        if gs < 1e-10:
            continue
        g = g / gs
        
        if np.linalg.norm(g - a) < 1e-9:
            t *= 0.9
            continue
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        score = compute_score(a_new)
        if score < best_score:
            best = a_new.copy()
            best_score = score
            a = a_new
            t = max(t_min, t * t_decay)
        else:
            t *= 0.9
            if t < t_min:
                break
    
    return best, best_score

def block_perturb(a, rng, max_block=50):
    n = len(a)
    cand = a.copy()
    w = int(rng.integers(1, min(max_block, n)))

    stop = min(n, start + w)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.3)))
    cand = np.maximum(cand, 0.0)
    return normalize(cand)

"""
Multi-phase optimization to minimize the autocorrelation peak of a non-negative
step function. Uses FFT-based soft-max gradient descent with epsilon annealing,
LP refinement for final polish, and multi-start with diverse parametric
initializations (chirps, harmonics, mixed signals).

Objective: score = 2 * n * max(a*a) / (sum(a))^2
"""
import numpy as np
import time
import scipy.fft as sfft
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    target = 2 * n - 1
    size = 1
    while size < target:
        size <<= 1
    A = sfft.rfft(a, size)
    corr = sfft.irfft(A * A, size)[:target]
    return np.maximum(corr, 0.0)


def compute_score(a):
    """Compute the objective score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = len(a)
    corr = fast_autocorr(a)
    return float(2.0 * n * corr.max() / (s * s))


def normalize(a):
    """Normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_chirp(n, alpha=0.8, beta=1.0, gamma=0.0):
    m = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * beta * m * m / (n * n) + gamma)
    return np.maximum(a, 0.01)


def make_harmonic(n, alpha=0.8, freq=1):
    m = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * m / n)
    return np.maximum(a, 0.01)

def soft_max_gradient(a, corr, epsilon):
    """Compute soft-max gradient using FFT cross-correlation. O(n log n)."""
    n = len(a)
    M = corr.max()
    weights = np.exp((corr - M) / epsilon)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n), M
    weights /= wsum

    a_rev = a[::-1]
    size = 1
    target_len = len(weights) + n - 1
    while size < target_len:
        size <<= 1
    W = sfft.rfft(weights, size)
    Ar = sfft.rfft(a_rev, size)
    conv = sfft.irfft(W * Ar, size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, M
