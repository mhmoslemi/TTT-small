Multi-stage optimization for minimum peak autocorrelation of non-negative sequences.
Stage 1: Smoothed gradient descent (softmax-approximated max)
Stage 2: Subgradient method with Nesterov momentum
Stage 3: LP-based active set refinement
Stage 4: Local search polishing

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_simplex(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), 1.0 / len(a))
    return a / s


def score_normalized(a):
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def compute_conv_and_weights(conv, temperature):
    shifted = conv / temperature
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    return weights


def compute_grad_from_weights(weights, a):
    n = len(a)
    # grad[i] = 2 * sum_k weights[k] * a[k-i] * [0<=k-i<n] - correction
    b = a[::-1].copy()
    conv_bw = np.convolve(b, weights, mode='full')
    grad = 2.0 * conv_bw[n-1:n-1+n]
    # Correction for k=2i
    for k in range(0, 2*n-1, 2):
        i = k // 2
        if i < n:
            grad[i] -= weights[k] * a[i]
    return grad


def stage1_gradient_descent(a, time_limit=10.0, verbose=True):
    """Smoothed gradient descent with Adam."""
    n = len(a)
    a = normalize_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    
    start = time.monotonic()
    temperature = 0.05  # initial temperature for softmax
    
    while time.monotonic() - start < time_limit:
        t += 1
        conv = np.convolve(a, a)
        current_score = 2.0 * n * conv.max()
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        # Decrease temperature over time
        progress = (time.monotonic() - start) / time_limit
        temperature = max(0.001, 0.05 * (1 - progress) + 0.001)
        
        # Compute gradient
        weights = compute_conv_and_weights(conv, temperature)
        grad = compute_grad_from_weights(weights, a)
        
        # Adam update
        lr = 0.1 / (1 + 0.01 * t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a = normalize_simplex(a)
    
    # Final check
    final_score = score_normalized(a)
    if final_score < best_score:
        best_a = a.copy()
        best_score = final_score
    
    if verbose:
        print(f"  Stage1 done: score={best_score:.8f}, iters={t}, time={time.monotonic()-start:.1f}s")
    
    return best_a, best_score


def stage2_subgradient(a, time_limit=10.0, verbose=True):
    """Subgradient method with Nesterov momentum."""
    n = len(a)
    a = normalize_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    
    # Nesterov momentum
    a_prev = a.copy()
    momentum = 0.9
    
    start = time.monotonic()
    it = 0
    stagnant = 0
    peak_frac = 0.05  # consider peaks within 5% of max
    
    while time.monotonic() - start < time_limit:
        it += 1
        conv = np.convolve(a, a)
        conv_max = conv.max()
        current_score = 2.0 * n * conv_max
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 500:
            break
        
        # Nesterov: extrapolate
        a_nest = a + momentum * (a - a_prev)
        a_nest = normalize_simplex(a_nest)
        
        # Compute subgradient using active peaks
        conv_nest = np.convolve(a_nest, a_nest)
        threshold = conv_nest.max() * (1 - peak_frac)
        active = conv_nest >= threshold
        if not active.any():
            active[np.argmax(conv_nest)] = True
        weights = conv_nest[active] ** 3  # weight by cubed value for emphasis
        weights = weights / weights.sum()
        
        # Compute subgradient at a_nest
        grad = np.zeros(n)
        active_indices = np.where(active)[0]
        for k in active_indices:
            w = weights[np.where(active_indices == k)[0][0]]
            # d(conv[k])/da[i] for valid i
            lo = max(0, k - (n - 1))
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        grad[i] += w * a_nest[j]
                    else:
                        grad[i] += w * (a_nest[j] + a_nest[j])  # = 2*w*a_nest[j]
        
        # Step size (Polyak or diminishing)
        step = 0.5 / (1 + 0.1 * it ** 0.5)
        a_prev = a.copy()
        a = a - step * grad
        a = normalize_simplex(a)
    
    final_score = score_normalized(a)
    if final_score < best_score:
        best_a = a.copy()
        best_score = final_score
    
    if verbose:
        print(f"  Stage2 done: score={best_score:.8f}, iters={it}, time={time.monotonic()-start:.1f}s")
    
    return best_a, best_score

import time
from scipy.optimize import linprog

    """Project onto {a >= 0, sum(a) = 1}."""

def score_normalized(a):

    n = len(a)

def score_sequence(a):
    """Full score computation matching evaluate_sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.2):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))

    return 0.54 - 0.46 * np.cos(2 * np.pi * t)

            ramp2 = np.linspace(1, 0, edge)
            a[end:end + edge] = ramp2
    return np.maximum(0, a)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def compute_grad_softmax(a, temperature):
    """Compute gradient of softmax(conv/temperature) for fast optimization."""
    n = len(a)
    conv = np.convolve(a, a)
    shifted = conv / temperature
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i] * [valid] - correction for k=2i
    b = a[::-1].copy()
    conv_bw = np.convolve(b, weights, mode='full')
    grad = 2.0 * conv_bw[n - 1:n - 1 + n]
    # Correction: for k=2i, derivative is a[i] not 2*a[i]
    even_ks = np.arange(0, min(2 * n - 1, 2 * n), 2)
    is = even_ks // 2
    valid = is < n
    is = is[valid]
    even_ks = even_ks[valid]
    grad[is] -= weights[even_ks] * a[is]
    return grad, conv.max()


def stage1_smoothed_descent(a, time_limit=15.0, verbose=True):
    """Adam-based smoothed gradient descent."""
    a = normalize_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    t = 0
    start = time.monotonic()
    base_temp = 0.08
    while time.monotonic() - start < time_limit:
        t += 1
        conv = np.convolve(a, a)
        curr = 2.0 * n * conv.max()
        if curr < best_score:
            best_score = curr
            best_a = a.copy()
        progress = (time.monotonic() - start) / time_limit
        temperature = max(0.002, base_temp * (1.0 - 0.9 * progress))
        grad, _ = compute_grad_softmax(a, temperature)
        lr = 0.15 / (1.0 + 0.005 * t)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = normalize_simplex(a)
    final = score_normalized(a)
    if final < best_score:
        best_a = a.copy()
        best_score = final
    if verbose:
        print(f"    Stage1: score={best_score:.8f}, iters={t}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score


def stage2_subgradient(a, time_limit=15.0, verbose=True):
    """Subgradient with Nesterov momentum on exact max."""
    a = normalize_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    a_prev = a.copy()
    mu = 0.85
    start = time.monotonic()
    it = 0
    stagnant = 0
    peak_frac = 0.08
    while time.monotonic() - start < time_limit:
        it += 1
        conv = np.convolve(a, a)
        curr = 2.0 * n * conv.max()
        if curr < best_score - 1e-12:
            best_score = curr
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 2000:
            break
        a_nest = a + mu * (a - a_prev)
        a_nest = normalize_simplex(a_nest)
        conv_n = np.convolve(a_nest, a_nest)
        thr = conv_n.max() * (1.0 - peak_frac)
        active = conv_n >= thr
        if not active.any():
            active[np.argmax(conv_n)] = True
        w_active = conv_n[active] ** 4
        w_sum = w_active.sum()
        if w_sum < 1e-30:
            break
        w_active = w_active / w_sum
        # Compute subgradient at a_nest for active peaks
        b = a_nest[::-1].copy()
        w_full = np.zeros(2 * n - 1)
        w_full[np.where(active)[0]] = w_active
        conv_bw = np.convolve(b, w_full, mode='full')
        grad = 2.0 * conv_bw[n - 1:n - 1 + n]
        even_ks = np.arange(0, 2 * n - 1, 2)
        is_even = even_ks // 2
        valid = is_even < n
        grad[is_even[valid]] -= w_full[even_ks[valid]] * a_nest[is_even[valid]]
        step = 0.3 / (1.0 + 0.05 * np.sqrt(it))
        a_new = a - step * grad
        a_new = normalize_simplex(a_new)
        new_score = score_normalized(a_new)
        if new_score <= curr + 0.01:
            a_prev = a.copy()
            a = a_new
        else:
            step2 = step * 0.1
            a_new = a - step2 * grad
            a_new = normalize_simplex(a_new)
            a_prev = a.copy()
            a = a_new
    final = score_normalized(a)
    if final < best_score:
        best_a = a.copy()
        best_score = final
    if verbose:
        print(f"    Stage2: score={best_score:.8f}, iters={it}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score


def build_conv_matrix_sparse(a):
    """Build sparse matrix M where M @ b = conv(a, b) for the first n entries."""
    n = len(a)
    n_rows = n
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, min(j + n, n), dtype=np.int32) for j in range(n)])
    data = []
    for j in range(n):
        valid_rows = np.arange(j, min(j + n, n), dtype=np.int32)
        data.extend(a[valid_rows - j].tolist())
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_refine(a, time_limit=60.0, verbose=True, active_frac=0.15):
    """LP-based active set refinement."""
    a = normalize_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    start = time.monotonic()
    stagnant = 0
    it = 0
    while time.monotonic() - start < time_limit:
        it += 1
        conv = np.convolve(a, a)
        M = conv.max()
        curr = 2.0 * n * M
        if curr < best_score - 1e-14:
            best_score = curr
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 80:
            break
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 5.0:
            break
        # Active set: peaks within active_frac of max
        thr = M * (1.0 - active_frac)
        active_ks = np.where(conv >= thr)[0]
        if len(active_ks) == 0:
            active_ks = np.array([np.argmax(conv)])
        # Build LP: max sum(b) s.t. (a*b)[k] <= M for k in active_ks, b >= 0
        # (a*b)[k] = sum_j a[j]*b[k-j] ... we need the full convolution
        # For full conv, row k of the matrix times b gives sum_j mat[k,j]*b[j] = (a*b)[k]
        # mat[k, j] = a[k-j] if 0 <= k-j < n else 0, PLUS a[j] * delta... 
        # Actually (a*b)[k] = sum_j a[j]*b[k-j] where b[m]=0 for m outside [0,n-1]
        # So (a*b)[k] = sum_{j: 0<=k-j<n} a[j]*b[k-j] = sum_{m: 0<=m<n, 0<=k-m<n} a[k-m]*b[m]
        # The coefficient of b[m] in (a*b)[k] is a[k-m] if 0 <= k-m < n, else 0.
        
        n_full = 2 * n - 1
        # Build A_ub: rows for active_ks, columns are b[0..n-1]
        # A_ub[k_row, m] = a[k - m] if 0 <= k-m < n else 0
        A_data = []
        A_row = []
        A_col = []
        for idx, k in enumerate(active_ks):
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    A_data.append(a[j])
                    A_row.append(idx)
                    A_col.append(m)
        A_ub = csr_matrix((A_data, (A_row, A_col)), shape=(len(active_ks), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_ks), M)
        bounds = [(0.0, None)] * n
        timeout = max(2.0, min(30.0, remaining * 0.4))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if not res.success or res.x is None:
                break
            g = np.maximum(0.0, res.x)
            sum_g = g.sum()
            if sum_g <= 1.0 + 1e-10:
                break
            g = g / sum_g
            alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = normalize_simplex(a_new)
            new_score = score_normalized(a_new)
            if new_score <= curr + 1e-10:
                a = a_new
            else:
                alpha2 = alpha * 0.25
                a_new = (1.0 - alpha2) * a + alpha2 * g
                a_new = normalize_simplex(a_new)
                a = a_new
        except Exception:
            break
    final = score_normalized(a)
    if final < best_score:
        best_a = a.copy()
        best_score = final
    if verbose:
        print(f"    LP: score={best_score:.8f}, iters={it}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score


def local_search(a, time_limit=30.0, rng=None, verbose=True):
    """Stochastic local search with multiplicative perturbations."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = normalize_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    start = time.monotonic()
    accepts = 0
    sigma = 0.08
    while time.monotonic() - start < time_limit:
        # Random segment perturbation
        width = max(3, int(rng.integers(n // 300, n // 6)))
        pos = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        candidate[pos:pos + width] *= np.exp(rng.normal(0.0, sigma, size=width))
        # Occasionally try global scaling of a sub-region
        if rng.random() < 0.2:
            w2 = max(5, n // 4)
            p2 = int(rng.integers(0, max(1, n - w2)))
            candidate[p2:p2 + w2] *= (1.0 + rng.normal(0, 0.05))
        # Symmetrize occasionally
        if rng.random() < 0.15:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_simplex(candidate)
        s = score_normalized(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            a = candidate.copy()
            accepts += 1
            sigma = max(0.01, sigma * 0.95)
        else:
            sigma = min(0.15, sigma * 1.01)
    if verbose:
        print(f"    LS: score={best_score:.8f}, accepts={accepts}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search for non-negative sequence minimizing 2*n*max(conv)/sum(a)^2.
    Uses smoothed gradient descent, subgradient with momentum, LP refinement,
    and local search across multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(30.0, budget_s - 20.0)

    best_score = float('inf')
    best_seq = None

    # Try to use the provided starting point
    init_sequences = []
    try:
        global height_sequence_1
        if height_sequence_1 is not None and len(height_sequence_1) > 10:
            hs = np.array(height_sequence_1, dtype=np.float64)
            hs = normalize_simplex(hs)
            init_sequences.append((len(hs), hs))
            print(f"Using height_sequence_1: n={len(hs)}, score={score_normalized(hs):.8f}")
    except Exception:
        pass

    # Define configurations: (n, init_type, time_frac)
    configs = []
    # Always try n=500 (previous best length) with raised cosine
    configs.append((500, 'raised_cosine', 0.12))
    configs.append((500, 'gaussian', 0.06))
    configs.append((500, 'blackman', 0.04))
    configs.append((750, 'raised_cosine', 0.10))
    configs.append((1000, 'raised_cosine', 0.10))
    configs.append((1000, 'gaussian', 0.06))
    configs.append((1500, 'raised_cosine', 0.08))
    configs.append((2000, 'raised_cosine', 0.08))
    configs.append((500, 'hamming', 0.04))
    configs.append((500, 'flat_top', 0.04))

    # If we have a starting sequence, put it first with extra time
    if init_sequences:
        n0, a0 = init_sequences[0]
        configs.insert(0, (n0, 'provided', 0.15))

    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        time_limit = min(time_frac * budget_s, remaining * 0.7)
        if time_limit < 10.0:
            continue

        print(f"\n  === n={n}, init={init_type}, alloc={time_limit:.0f}s ===")

        if init_type == 'provided':
            a = init_sequences[0][1].copy()
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.2)
        elif init_type == 'blackman':
            a = initial_blackman(n)
        elif init_type == 'hamming':
            a = initial_hamming(n)
        elif init_type == 'flat_top':
            a = initial_flat_top(n, frac=0.6)
        else:
            a = initial_raised_cosine(n)

        a = normalize_simplex(a)
        init_score = score_normalized(a)
        print(f"    Init: score={init_score:.8f}")

        # Stage 1: Smoothed gradient descent (use 40% of allocated time)
        t1 = min(time_limit * 0.35, remaining * 0.3)
        if t1 > 3.0:
            a, s1 = stage1_smoothed_descent(a, time_limit=t1, verbose=True)
            if s1 < best_score:
                best_score = s1
                best_seq = a.copy()
                print(f"    ** New best after Stage1: {best_score:.8f}")

        # Stage 2: Subgradient with momentum (use 30% of allocated time)
        remaining2 = deadline - time.monotonic()
        t2 = min(time_limit * 0.25, remaining2 * 0.3)
        if t2 > 3.0:
            a, s2 = stage2_subgradient(a, time_limit=t2, verbose=True)
            if s2 < best_score:
                best_score = s2
                best_seq = a.copy()
                print(f"    ** New best after Stage2: {best_score:.8f}")

        # Stage 3: LP refinement (use 25% of allocated time)
        remaining3 = deadline - time.monotonic()
        t3 = min(time_limit * 0.25, remaining3 * 0.3)
        if t3 > 8.0 and n <= 1200:
            a, s3 = lp_refine(a, time_limit=t3, verbose=True)
            if s3 < best_score:
                best_score = s3
                best_seq = a.copy()
                print(f"    ** New best after LP: {best_score:.8f}")

        # Stage 4: Local search (use 20% of allocated time)
        remaining4 = deadline - time.monotonic()
        t4 = min(time_limit * 0.15, remaining4 * 0.3)
        if t4 > 3.0:
            a, s4 = local_search(a, time_limit=t4, rng=rng, verbose=True)
            if s4 < best_score:
                best_score = s4
                best_seq = a.copy()
                print(f"    ** New best after LS: {best_score:.8f}")

    # Final polishing on the best sequence found
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n  === Final polishing (remaining={remaining:.0f}s) ===")
        
        # More aggressive LP on the best
        if remaining > 30 and len(best_seq) <= 1200:
            t_lp = min(remaining * 0.4, 120.0)
            a_pol, s_pol = lp_refine(best_seq, time_limit=t_lp, verbose=True, active_frac=0.1)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol.copy()
                print(f"  ** After final LP: {best_score:.8f}")
        
        # Local search polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_ls = min(remaining * 0.4, 60.0)
            a_pol, s_pol = local_search(best_seq, time_limit=t_ls, rng=rng, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol.copy()
                print(f"  ** After final LS: {best_score:.8f}")
        
        # Another gradient pass
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_g = min(remaining * 0.5, 30.0)
            a_pol, s_pol = stage1_smoothed_descent(best_seq, time_limit=t_g, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol.copy()
                print(f"  ** After final grad: {best_score:.8f}")
    else:
        # Fallback
        best_seq = normalize_simplex(initial_raised_cosine(500))
        best_score = score_normalized(best_seq)
        print(f"  Fallback: score={best_score:.8f}")

    # Verify
    verify = evaluate_sequence([float(x) for x in best_seq])
    print(f"\n  Final: score={best_score:.8f}, verified={verify:.8f}, n={len(best_seq)}")
    print(f"  Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
