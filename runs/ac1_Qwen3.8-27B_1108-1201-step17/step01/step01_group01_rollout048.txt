import numpy as np
import time

def _compute_b(a):
    """Compute autocorrelation (full convolution)."""
    return np.convolve(a, a)

def _compute_score_normalized(a):
    """Compute 2*n*max_b / sum_a^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = _compute_b(a)
    return 2.0 * n * np.max(b) / (s * s)

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0])
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _lsq_obj_and_grad(a, tau):
    """Compute log-sum-exp objective and its gradient."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    exp_b = np.exp(tau * (b - b_max))
    sum_exp = np.sum(exp_b)
    obj = (b_max + np.log(sum_exp)) / tau  # = (1/tau) * log(sum exp(tau*b))
    
    w = exp_b / sum_exp  # softmax weights
    
    # Gradient: grad_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    a_rev = a[::-1]
    c = np.convolve(a_rev, w, mode='full')
    grad = 2.0 * c[n-1:n-1+n]
    
    return obj, grad

def _adam_step(params, grad, t, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step. params is a dict with 'm', 'v'."""
    m = params['m']
    v = params['v']
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return update, {'m': m, 'v': v}

def _optimize_lsq(a_init, deadline, tau_start=5.0, tau_end=500.0, lr=0.01, seed=42):
    """
    Optimize using log-sum-exp smoothed objective with Adam.
    Anneals tau from tau_start to tau_end.
    """
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_score = _compute_score_normalized(best_a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    
    # Time budget for annealing
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        # Anneal tau: exponential schedule
        elapsed = time.monotonic() - start_time
        total_time = max(1.0, deadline - start_time)
        progress = min(1.0, elapsed / total_time)
        # Use exponential annealing
        tau = tau_start * (tau_end / tau_start) ** progress
        
        obj, grad = _lsq_obj_and_grad(a, tau)
        
        # Center gradient for sum constraint (project out the all-ones direction)
        grad = grad - np.mean(grad)
        
        # Adam step
        lr_t = lr / np.sqrt(n)  # Scale learning rate
        update, adam_state = _adam_step({'m': m, 'v': v}, grad, t, lr=lr_t)
        m, v = adam_state['m'], adam_state['v']
        t += 1
        
        a = a - update
        a = _project_simplex(a, 1.0)
        
        # Check score periodically
        if t % 50 == 0:
            score = _compute_score_normalized(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"  t={t}, tau={tau:.1f}, score={score:.6f}")
    
    return best_a, best_score

"""
Optimizes non-negative sequence to minimize 2*n*max(autocorr)/sum^2 using
log-sum-exp smoothing with multiplicative weights and Adam optimizer,
with diverse starting points and adaptive tau annealing.
"""

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (s * s))


def _get_lsq_weights(b, tau):
    """Numerically stable softmax weights."""
    b_max = np.max(b)
    exp_b = np.exp(tau * (b - b_max))
    w = exp_b / np.sum(exp_b)
    return w


def _lsq_gradient_from_weights(a, w):
    """
    Compute gradient of log-sum-exp objective given softmax weights w.
    grad_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    """
    n = len(a)
    a_rev = a[::-1]
    c = np.convolve(a_rev, w, mode='full')
    return 2.0 * c[n - 1:n - 1 + n]


def _make_cosine(n):
    """Cosine window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _make_triangle(n):
    """Triangle (Bartlett) window."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(2.0 * i / (n - 1) - 1.0)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S if S > 0 else np.ones(n) / n


def _make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return a / np.sum(a)


def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _make_flat_top(n, frac=0.6):
    """Flat-top window."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half_width = frac * n / 2.0
    a = np.ones(n)
    # Smooth edges
    left_edge = center - half_width
    right_edge = center + half_width
    a[:int(left_edge)] = 0.5 * (1 + np.cos(np.pi * (i[:int(left_edge)] - left_edge) / (left_edge))) if left_edge > 1 else 0.0
    # This is getting complicated, let me use a simpler approach
    # Just use a raised cosine at the edges
    edge = int((1 - frac) * n / 2)
    a[:edge] = 0.5 * (1 + np.cos(np.pi * np.arange(edge) / edge)) if edge > 1 else 0.0
    a[-edge:] = 0.5 * (1 + np.cos(np.pi * np.arange(edge) / edge)) if edge > 1 else 0.0
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _multiplicative_optimize(a_init, deadline, tau_start=5.0, tau_end=200.0, 
                              eta_init=0.5, eta_decay=0.999, min_eta=0.01,
                              reheat_interval=500, reheat_amount=0.5, seed=0):
    """
    Optimize using multiplicative weights with log-sum-exp objective.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_score = _compute_score(a)
    
    eta = eta_init
    t = 0
    last_improve = 0
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        # Adaptive tau: increase over time
        elapsed = time.monotonic() - start_time
        total_time = max(1.0, deadline - start_time)
        progress = min(1.0, elapsed / total_time)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)
        
        # Compute b and weights
        b = np.convolve(a, a)
        b_max = np.max(b)
        w = _get_lsq_weights(b, tau)
        
        # Compute gradient
        grad = _lsq_gradient_from_weights(a, w)
        
        # Multiplicative update
        # a_i *= exp(-eta * grad_i)
        # For numerical stability, subtract the mean of grad
        grad_centered = grad - np.mean(grad)
        a = a * np.exp(-eta * grad_centered)
        a = np.maximum(a, 1e-15)
        a = a / np.sum(a)
        
        t += 1
        
        # Decay eta
        if t % 10 == 0:
            eta = max(eta * eta_decay, min_eta)
        
        # Check score periodically
        if t % 100 == 0:
            score = 2.0 * n * b_max  # since sum(a) = 1
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                last_improve = t
                print(f"    MW t={t}, tau={tau:.1f}, score={score:.6f}")
            
            # Reheating: if no improvement for a while, reset eta
            if t - last_improve > reheat_interval:
                eta = max(eta, eta_init * reheat_amount)
                last_improve = t
    
    return best_a, best_score


def _adam_optimize(a_init, deadline, tau_start=10.0, tau_end=1000.0,
                   lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, seed=0):
    """
    Optimize using Adam with log-sum-exp objective.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_score = _compute_score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    last_improve = 0
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        # Adaptive tau
        elapsed = time.monotonic() - start_time
        total_time = max(1.0, deadline - start_time)
        progress = min(1.0, elapsed / total_time)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)
        
        # Compute b and weights
        b = np.convolve(a, a)
        b_max = np.max(b)
        w = _get_lsq_weights(b, tau)
        
        # Compute gradient
        grad = _lsq_gradient_from_weights(a, w)
        
        # Center gradient for sum constraint
        grad = grad - np.mean(grad)
        
        # Adam update
        lr_t = lr / np.sqrt(n)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a = _project_simplex(a, 1.0)
        
        # Check score periodically
        if t % 50 == 0:
            score = 2.0 * n * b_max  # since sum(a) ≈ 1
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                last_improve = t
                print(f"    Adam t={t}, tau={tau:.1f}, score={score:.6f}")
    
    return best_a, best_score
