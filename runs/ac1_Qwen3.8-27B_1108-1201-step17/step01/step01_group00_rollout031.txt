"""
LP-iterative optimization with gradient refinement and simulated annealing polish
for minimizing the normalized peak of linear autocorrelation of non-negative step functions.

Key innovation: Uses a linear programming step at each iteration to find the
"largest" sequence (max sum) whose cross-correlation with the current sequence
stays within the peak bound. This provides a strong descent direction that is
much faster than gradient-based methods alone. Combined with L-BFGS-B smooth-max
refinement and SA polishing, achieves scores below 1.50.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def peak_of_normalized(a):
    """Return max(conv(a,a)) assuming sum(a)=1."""
    c = fft_convolve(a, a)
    return float(np.max(c))


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def build_constraint_matrix(a):
    """Build (2n-1) x n matrix A where A[k,i] = a[i] if 0 <= k-i < n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        A[k, i_start:i_end + 1] = a[i_start:i_end + 1]
    return A


def solve_lp_step(a, M):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M (elementwise), b >= 0.
    Returns optimal b or None if infeasible.
    """
    n = len(a)
    A = build_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.success:
        return res.x
    return None


def lp_optimize(a_init, max_iters=2000, time_limit=60.0, t_step=1.0,
                adaptive=True, verbose=False):
    """
    LP-iterative optimization. Minimizes max(conv(a,a)) with sum(a)=1, a>=0.
    Uses LP to find best direction at each step.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 1e-6, None)
    a = a / np.sum(a)
    n = len(a)
    
    current_peak = peak_of_normalized(a)
    best_a = a.copy()
    best_peak = current_peak
    t0 = time.monotonic()
    t = t_step
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = solve_lp_step(a, current_peak)
        if b is None:
            break
        
        b_sum = np.sum(b)
        if b_sum < 1e-10:
            break
        
        b_norm = b / b_sum
        
        # Update
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.clip(a_new, 1e-10, None)
        a_new = a_new / np.sum(a_new)
        
        new_peak = peak_of_normalized(a_new)
        
        if adaptive and new_peak > current_peak * 1.05:
            # Peak increased too much, reduce step
            t = max(0.05, t * 0.5)
            a_new = (1.0 - t) * a + t * b_norm
            a_new = np.clip(a_new, 1e-10, None)
            a_new = a_new / np.sum(a_new)
            new_peak = peak_of_normalized(a_new)
        
        a = a_new
        current_peak = new_peak
        
        if current_peak < best_peak:
            best_peak = current_peak
            best_a = a.copy()
            stall_count = 0
            if verbose and it % 100 == 0:
                print(f"    LP iter {it}: peak={best_peak:.8f}, score={2*n*best_peak:.6f}")
        else:
            stall_count += 1
            if stall_count > 200:
                # Try to kick out of local region
                t = 0.3
                a = a * (1.0 + 0.01 * np.random.standard_normal(n))
                a = np.clip(a, 1e-6, None)
                a = a / np.sum(a)
                current_peak = peak_of_normalized(a)
                stall_count = 0
        
        if t < 0.01:
            t = 0.5  # Reset step size periodically
    
    return best_a, best_peak


def smooth_max_objective(theta, tau):
    """Smooth max objective with softmax parameterization."""
    a = softmax(theta)
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    c_shifted = (c - c_max) / tau
    exp_c = np.exp(c_shifted)
    val = 2.0 * n * (c_max + tau * np.log(np.sum(exp_c) / (2 * n - 1)))
    return val


def smooth_max_gradient(theta, tau):
    """Gradient of smooth max objective w.r.t. theta parameters."""
    a = softmax(theta)
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    c_shifted = (c - c_max) / tau
    exp_c = np.exp(c_shifted)
    w = exp_c / np.sum(exp_c)
    
    # Gradient of smooth_max w.r.t. a:
    # g_a[j] = 2 * sum_k w[k] * (a[k-j] + a[j-k])
    a_rev = a[::-1]
    conv1 = fft_convolve(a_rev, w)
    part1 = conv1[n - 1:2 * n - 1]
    conv2 = fft_convolve(a, w)
    part2 = conv2[0:n]
    g_a = 2.0 * (part1 + part2)
    
    # Chain rule through softmax: df/dtheta[j] = a[j] * (g_a[j] - sum(g_a*a))
    mu = np.dot(g_a, a)
    g_theta = a * (g_a - mu)
    return g_theta


def gradient_refine(a_init, time_limit=120.0, verbose=False):
    """L-BFGS-B refinement with annealed smooth max."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 1e-6, None)
    a = a / np.sum(a)
    n = len(a)
    theta = np.log(a + 1e-10)
    
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    tau_stages = [5.0 / n, 2.0 / n, 0.5 / n, 0.1 / n, 0.03 / n, 0.01 / n]
    n_stages = len(tau_stages)
    time_per_stage = time_limit / n_stages
    
    for tau in tau_stages:
        if time.monotonic() - t0 > time_limit:
            break
        stage_deadline = time.monotonic() + time_per_stage
        
        result = minimize(
            smooth_max_objective, theta,
            jac=smooth_max_gradient,
            args=(tau,),
            method='L-BFGS-B',
            options={'maxiter': 300, 'maxfun': 500,
                     'ftol': 1e-12, 'gtol': 1e-10, 'maxls': 50}
        )
        
        if result.success or result.fun < np.inf:
            theta = result.x
            a_check = softmax(theta)
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
                if verbose:
                    print(f"    Grad refine tau={tau:.4f}: score={score:.6f}")
    
    return best_a, best_score


def polish_sa(a_init, time_limit=120.0, rng=None):
    """Simulated annealing local search polish."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    current = np.asarray(a_init, dtype=np.float64).copy()
    n = len(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.0005
    cooling = 0.99998
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            max_i = min(n, peak_idx + 1)
            for i in range(max_i):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 10:
                n_inc = min(15, len(other) // 4)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(2, max(3, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.70:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.01, size=width))
        elif r < 0.85:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.003, size=n))
        else:
            # Global small
            candidate = current * np.exp(rng.normal(0.0, 0.003, size=n))
        
        candidate = np.clip(candidate, 1e-6, 1000.0)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score


def make_chirp(n, alpha):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)

def make_multi_tone(n, rng):
    """Multi-tone sinusoidal initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in [1, 2, 3, 5, 8, 13, 21]:

    return np.clip(a, 0.01, None)


def make_spectral_flat(n, rng, n_iter=100):
    """Iterative spectral flattening initialization."""
    a = np.ones(n) + 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    for _ in range(n_iter):
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        target_mag = np.sum(mag) / len(mag)
        A_new = A * (target_mag / mag)
        a_new = np.fft.irfft(A_new, size)[:n]
        a = np.clip(a_new, 0.01, None)
    return a


def resample_to_length(a, n_new):
    """Resample sequence to new length via interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

            break
        
        # Generate initializations
        inits = []
        if start_seq is not None:
            inits.append(('provided', resample_to_length(start_seq, n)))
        inits.append(('chirp_03', make_chirp(n, 0.3)))
        inits.append(('chirp_06', make_chirp(n, 0.6)))
        inits.append(('multi_tone', make_multi_tone(n, rng)))
        inits.append(('spectral', make_spectral_flat(n, rng, n_iter=80)))
        
        n_inits = len(inits)
        time_per_init = max(5.0, min(60.0, (deadline - time.monotonic() - 60) / (n_inits * 1.5)))
        
        for init_name, a_init in inits:
            if time.monotonic() > deadline - 30:
                break
            
            t0 = time.monotonic()
            a_lp, peak_lp = lp_optimize(
                a_init, max_iters=3000, time_limit=time_per_init,
                t_step=1.0, adaptive=True, verbose=False
            )
            score = 2.0 * n * peak_lp
            elapsed = time.monotonic() - t0
            
            if score < best_score:
                best_score = score
                best_seq = a_lp.copy()
                print(f"  n={n}, init={init_name}: score={score:.6f} (peak={peak_lp:.8f}) [{elapsed:.1f}s] *** NEW BEST ***")
            else:
                print(f"  n={n}, init={init_name}: score={score:.6f} [{elapsed:.1f}s]")
    
    print(f"\n{'='*60}")
    print(f"Phase 2: Gradient Refinement (L-BFGS-B + Smooth Max)")
    print(f"{'='*60}")
    
    if best_seq is not None:
        time_left = deadline - time.monotonic() - 30
        grad_time = min(180.0, max(20.0, time_left * 0.25))
        print(f"  Refining best (n={len(best_seq)}, score={best_score:.6f}) with {grad_time:.0f}s")
        
        a_grad, score_grad = gradient_refine(best_seq, time_limit=grad_time, verbose=True)
        if score_grad < best_score:
            best_score = score_grad
            best_seq = a_grad.copy()
            print(f"  After gradient: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After gradient: {score_grad:.6f} (no improvement)")
    
    print(f"\n{'='*60}")
    print(f"Phase 3: Additional LP runs from best (different params)")
    print(f"{'='*60}")
    
    if best_seq is not None and time.monotonic() < deadline - 60:
        # Try different step sizes and more iterations from the best
        for t_step in [0.5, 0.7, 0.3]:
            if time.monotonic() > deadline - 30:
                break
            n = len(best_seq)
            t0 = time.monotonic()
            a_lp2, peak_lp2 = lp_optimize(
                best_seq, max_iters=5000,
                time_limit=min(60.0, deadline - time.monotonic() - 30),
                t_step=t_step, adaptive=True, verbose=False
            )
            score2 = 2.0 * n * peak_lp2
            if score2 < best_score:
                best_score = score2
                best_seq = a_lp2.copy()
                print(f"  t={t_step}: score={score2:.6f} *** NEW BEST ***")
            else:
                print(f"  t={t_step}: score={score2:.6f}")
    
    print(f"\n{'='*60}")
    print(f"Phase 4: Simulated Annealing Polish")
    print(f"{'='*60}")
    
    if best_seq is not None:
        time_left = deadline - time.monotonic() - 20
        sa_time = min(400.0, max(30.0, time_left * 0.5))
        print(f"  Polishing with {sa_time:.0f}s, current best={best_score:.6f}")
        
        a_sa, score_sa = polish_sa(best_seq, time_limit=sa_time, rng=rng)
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"  After SA: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  After SA: {score_sa:.6f} (no improvement)")
    
    # Final gradient polish if time remains
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 15
        if remaining > 15:
            print(f"\n{'='*60}")
            print(f"Phase 5: Final Gradient Polish")
            print(f"{'='*60}")
            a_final_grad, score_final = gradient_refine(
                best_seq, time_limit=min(60.0, remaining), verbose=True
            )
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final_grad.copy()
                print(f"  Final gradient: {best_score:.6f} *** NEW BEST ***")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        elapsed_total = time.monotonic() - t_global
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}, total_time={elapsed_total:.1f}s")
        print(f"{'='*60}")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
