import numpy as np

def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        # Reject boolean types
        if isinstance(x, bool):
            return np.inf
        # Check for non-integer/non-float types (reject custom classes too)
        if not (isinstance(x, int) or isinstance(x, float)):
            return np.inf
        # Reject Not-a-Number (NaN) and infinity values
        if np.isnan(x) or np.isinf(x) or x < 0:
            return np.inf
    # Convert to float for consistency
    seq = np.array(sequence).astype(float)
    total = np.sum(seq)
    if total < 0.01:
        return np.inf
    # Max and avoid destructive normalizations
    seq = np.clip(seq, 0.0, 1000.0)
    conv = np.convolve(seq, seq, mode="full")
    max_out = np.max(conv)
    return float(2 * len(seq) * max_out / (total ** 2))
