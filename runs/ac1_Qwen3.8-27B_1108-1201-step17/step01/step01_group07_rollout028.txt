def compute_gradient_fft(w, a):
    """
    Compute grad_a[i] = 2 * sum_k w[k] * a[k-i] (valid indices only).
    w has length 2n-1, a has length n.
    """
    n = len(a)
    conv_len = 2 * n - 1
    # Need FFT size >= conv_len + n - 1 = 3n - 2
    fft_size = 1 << (conv_len + n - 2).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # Cross-correlation: IFFT(W * conj(A))
    # This gives: cross[i] = sum_j W[j] * A[j-i] (circular)
    # With zero-padding, this equals the linear cross-correlation for valid i

    # The i-th element is sum_j w[j] * a[j-i]...

    # With zero-padding to size N >= len(w) + len(a) - 1, this equals
    # sum_j w[j] * a[j-i] for the linear case.
    
    # But I want sum_k w[k] * a[k-i], which is sum_j w[j] * a[j-i] = cross[i]

The cross-correlation of sequences w and a is defined as:
(w ⋆ a)[i] = sum_j w[j] * conj(a[j-i]) = sum_j w[j] * a[j-i] (for real)

This can be computed as: IFFT(FFT(w) * conj(FFT(a)))

Product: sum_j sum_m w_j a_m exp(-2πijk/N) exp(2πikm/N) = sum_j sum_m w_j a_m exp(2πik(m-j)/N)

IDFT of product at position i: (1/N) sum_k [sum_j sum_m w_j a_m exp(2πik(m-j)/N)] exp(-2πiki/N)
= (1/N) sum_j sum_m w_j a_m sum_k exp(2πik(m-j-i)/N)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_grad_fft(w, a):
    """
    Compute grad_a[i] = 2 * sum_k w[k] * a[k-i] for valid (k-i).
    w has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    w_len = len(w)  # 2n-1
    # Need fft_size >= w_len + n - 1 = 3n - 2
    fft_size = 1 << (w_len + n - 2).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # cross[i] = sum_k w[k] * a[k-i] = IFFT(conj(W) * A)[i]
    cross = np.fft.irfft(np.conj(W) * A, fft_size)
    
    return 2.0 * cross[:n]


def softmax_weights(conv, t):
    """Compute softmax weights over convolution values with temperature t."""
    scaled = t * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    return w


def multiplicative_update(a, w, eta):
    """
    Perform one multiplicative weight update.
    a: current sequence (sum=1)
    w: softmax weights over convolution (length 2n-1)
    eta: learning rate
    Returns updated a (sum=1).
    """
    n = len(a)
    grad = compute_grad_fft(w, a)
    # Center the gradient (to preserve sum approximately)
    grad_centered = grad - grad.mean()
    # Multiplicative update
    log_factor = -eta * grad_centered
    # Clip for numerical stability
    log_factor = np.clip(log_factor, -50, 50)
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 1e-30)
    a_sum = a_new.sum()
    if a_sum < 1e-30:
        a_new = np.ones(n) / n
    else:
        a_new /= a_sum
    return a_new

def barker_init(n, seed=0):
    """Create a Barker-inspired initialization of length n."""
    rng = np.random.default_rng(seed)
    # Use a "block" structure inspired by Barker sequences
    # Barker: alternating blocks of 1s and 0s with specific pattern
    a = np.zeros(n)
    # Fill with a pseudo-Barker pattern
    block_size = max(1, n // 7)
    pattern = [1, 1, 1, 0, 1, 0, 0]  # Barker length 7 pattern
    pos = 0
    for block in range(0, n, block_size):
        for j in range(block_size):
            if block + j < n:
                a[block + j] = pattern[(block // block_size) % len(pattern)]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian (center)

    anti = 1.0 - np.exp(-0.5 * ((x - n*0.5) / (0.3*n))**2)

    inits['anti_center'] = anti / anti.sum()
    
    # From parent

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using annealed log-sum-exp smoothing
with FFT-based gradient computation. Primary optimizer: L-BFGS-B on softmax
parameterization. Explores multiple sequence lengths (64-1024) with diverse
initializations. Three-phase strategy: scan, deep optimize, polish.

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_grad_fft(w, a):
    """
    Compute grad_a[i] = 2 * sum_k w[k] * a[k-i] for valid (k-i).
    w has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    w_len = len(w)
    fft_size = 1 << (w_len + n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(W) * A, fft_size)
    return 2.0 * cross[:n]


def barker_init(n, seed=0):
    """Create a Barker-inspired initialization of length n."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    block_size = max(1, n // 7)
    pattern = [1, 1, 1, 0, 1, 0, 0]
    for block in range(0, n, block_size):
        for j in range(block_size):
            if block + j < n:
                a[block + j] = pattern[(block // block_size) % len(pattern)]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    g = np.exp(-0.5 * ((x - n * 0.5) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    e = np.exp(-0.5 * (x / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - n + 1) / (0.15 * n)) ** 2)
    inits['edges'] = e / e.sum()

    inits['barker'] = barker_init(n, seed=int(rng.integers(0, 10000)))

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    anti = 1.0 - np.exp(-0.5 * ((x - n * 0.5) / (0.3 * n)) ** 2)
    anti = np.maximum(anti, 0.1)
    inits['anti_center'] = anti / anti.sum()

    # Wigner-type distribution
    w_sig = 0.15 * n
    w = np.exp(-0.5 * ((x - n * 0.5) / w_sig) ** 2)
    w_alt = np.exp(-0.5 * ((x - n * 0.25) / w_sig) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / w_sig) ** 2)
    mixed = 0.5 * w + 0.5 * w_alt
    inits['wigner'] = mixed / mixed.sum()

    if parent is not None:
        if parent.size >= n:
            p = parent[:n].copy()
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                inits['parent'] = p / p.sum()
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                inits['parent_interp'] = p / p.sum()

    return inits

def optimize_lbfgsb(a0, time_limit, t_levels=None, max_iter_per_t=500):

        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break

        remaining = time_limit - elapsed
        if remaining < 2:
            break

        obj_grad, state = make_obj_grad(n, t)

        stop_flag = [False]
        t0 = time.monotonic()

        def callback(xk):
            if time.monotonic() - t0 > remaining * 0.9:
                stop_flag[0] = True
                return True
            return False

        try:
            result = minimize(
                obj_grad, x, method='L-BFGS-B', jac=True,
                options={'maxiter': max_iter_per_t, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxls': 20},
                callback=callback
            )
            x = result.x
        except Exception:
            pass

        # Convert back and evaluate
        x_shift = x - x.max()
        a_curr = np.exp(x_shift)

        a_curr = np.maximum(a_curr, 0)

        curr_score = 2.0 * n * fast_convolve(a_curr).max()

            best_score = curr_score
            best_a = a_curr.copy()

        # Update x for next iteration (warm start)
        a_safe = np.maximum(a_curr, 1e-15)
        x = np.log(a_safe)

    return best_a, best_score


def optimize_multiplicative(a0, time_limit, t_levels=None, eta0=0.6):
    """
    Optimize using multiplicative weights with annealed temperature.
    Returns (best_a, best_score_normalized).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    if t_levels is None:
        t_levels = [10.0, 30.0, 100.0, 300.0, 1000.0]

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t_start = time.monotonic()
    n_levels = len(t_levels)

    for level_idx, t in enumerate(t_levels):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break

        # Allocate remaining time evenly among remaining levels
        remaining_levels = n_levels - level_idx
        level_time = (time_limit - elapsed) / remaining_levels
        level_deadline = time.monotonic() + level_time

        eta = eta0 * (0.8 ** level_idx)
        no_improve = 0
        max_no_improve = 100

        while time.monotonic() < level_deadline:
            conv = fast_convolve(a)
            tc = t * conv
            tc_max = tc.max()
            exp_tc = np.exp(tc - tc_max)
            w = exp_tc / exp_tc.sum()

            grad = compute_grad_fft(w, a)
            grad_centered = grad - grad.mean()
            log_factor = -eta * grad_centered
            log_factor = np.clip(log_factor, -30, 30)
            a_new = a * np.exp(log_factor)
            a_new = np.maximum(a_new, 1e-30)
            a_sum = a_new.sum()
            if a_sum < 1e-30:
                a_new = np.ones(n) / n
            else:
                a_new /= a_sum
            a = a_new

            no_improve += 1
            if no_improve % 20 == 0:
                curr_score = 2.0 * n * conv.max()
                if curr_score < best_score - 1e-10:
                    best_score = curr_score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    if no_improve >= max_no_improve:
                        eta *= 0.6
                        no_improve = 0
                        if eta < 0.005:
                            break

    conv = fast_convolve(best_a)
    best_score = 2.0 * n * conv.max()
    return best_a, best_score

                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Polish (no jitter): {method}, score={score:.6f} [{elapsed():.0f}s]")

    # Final: try a few random restarts from the best if time permits
    if best_sequence is not None and remaining() > 40:
        print("Phase 4: Random restarts from best")
        n = len(best_sequence)
        for restart in range(5):
            if remaining() < 20:
                break
            # Create perturbed version
            noise_scale = 0.1 * (1.0 - restart * 0.15)
            perturbed = best_sequence + noise_scale * rng.standard_normal(n)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-10:
                continue
            perturbed /= ps

            t_alloc = min(30.0, remaining() - 15)
            if t_alloc < 5:
                break

            a_opt, _ = optimize_lbfgsb(perturbed, time_limit=t_alloc,
                                       t_levels=[50.0, 200.0, 1000.0, 5000.0],
                                       max_iter_per_t=300)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Restart {restart}: score={score:.6f} [{elapsed():.0f}s]")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    print(f"FINAL: score={final_score:.6f}, n={len(best_sequence)}, time={elapsed():.0f}s")
    return [float(x) for x in best_sequence]
