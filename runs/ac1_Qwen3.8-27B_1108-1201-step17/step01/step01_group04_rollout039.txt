"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based momentum gradient descent with adaptive softmax weighting,
LP refinement, coordinate descent polishing, diverse starting points,
and random restarts to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_score(a):
    """Compute the objective value: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, S_target):
    """Normalize a to have sum S_target."""
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a) * S_target / len(a)
    return a * (S_target / S)


def _fft_gradient(w, a):
    """
    Compute grad[j] = 2 * sum_k w[j+k] * a[k] for j=0..n-1.
    Uses FFT cross-correlation for O(n log n) computation.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    xcorr = np.fft.irfft(W * np.conj(A), N_fft)
    return 2.0 * xcorr[:n]


def _grad_descent_momentum(a0, deadline, S_target, lr_init=0.12, momentum=0.85,
                           tau_start=5.0, tau_end=80.0, max_iters=8000,
                           eval_every=30, patience=2000):
    """
    Gradient descent with Nesterov momentum and adaptive softmax.
    Minimizes max(convolve(a,a)) subject to sum(a) = S_target, a >= 0.
    """
    n = len(a0)
    a = np.clip(a0.copy(), 0, None)
    a = _normalize(a, S_target)

    score = _compute_score(a)
    best = a.copy()
    best_score = score
    best_it = 0

    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it > 0 and (it - best_it) > patience:
            break

        frac = it / max_iters
        tau = tau_start + (tau_end - tau_start) * frac * frac

        # Compute autocorrelation
        c = np.convolve(a, a)
        c_max = c.max()

        # Softmax weights
        c_shifted = c - c_max
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        w = w / w_sum

        # Compute gradient via FFT
        grad = _fft_gradient(w, a)

        # Project out mean (fixed-sum constraint)
        grad -= grad.mean()

        # Learning rate schedule (cosine annealing)
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, 0.002)

        # Momentum update
        v = momentum * v + grad
        a = a - lr * v
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_target)

        if (it + 1) % eval_every == 0:
            s = _compute_score(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                best_it = it

    s = _compute_score(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def _lp_refine(a, deadline, max_iters=100, S_target=None):
    """LP-based refinement: find g maximizing sum(g) s.t. conv(a,g) <= M, g>=0."""
    n = len(a)
    if n > 768:
        return a, _compute_score(a)
    if S_target is None:
        S_target = np.sqrt(2.0 * n)

    a = _normalize(a, S_target)
    c = np.convolve(a, a)
    M = float(c.max())
    score = _compute_score(a)
    best = a.copy()
    best_score = score

    # Build constraint matrix A: A[m,j] = a[m-j] if 0<=m-j<n
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if hi >= lo:
            A[m, lo:hi + 1] = a[hi - (hi - lo):0:-1] if False else a[m - hi:m - lo + 1][::-1]
    # Simpler: just fill directly
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        for j in range(max(0, m - n + 1), min(n, m + 1)):
            A[m, j] = a[m - j]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        # Try blending
        improved = False
        for t in [0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_target)
            s_new = _compute_score(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                M = float(np.convolve(a, a).max())
                b_ub = np.full(2 * n - 1, M)
                improved = True
                break

        if not improved:
            break

    return best, best_score


def _coord_descent_polish(a, deadline, max_sweeps=5, S_target=None):
    """
    Coordinate descent: optimize one coordinate at a time.
    For each j, find x >= 0 minimizing the 1D function.
    """
    n = len(a)
    if S_target is None:
        S_target = np.sqrt(2.0 * n)
    a = _normalize(a.copy(), S_target)
    score = _compute_score(a)
    best = a.copy()
    best_score = score

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved_any = False
        for j in range(n):
            if time.monotonic() > deadline:
                break

            a_old_j = a[j]
            S_rest = a.sum() - a_old_j

            # Compute c0[m] = conv(a_without_j, a_without_j) contribution
            # c[m] with a[j] = 0:
            a_j0 = a.copy()
            a_j0[j] = 0.0
            c0 = np.convolve(a_j0, a_j0)

            # b[m] = a[m-j] if 0<=m-j<n and m-j != j, else 0
            b = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                k = m - j
                if 0 <= k < n and k != j:
                    b[m] = a[k]
                # if k == j (m == 2j), the x^2 term applies
            has_quad = (2 * j < 2 * n - 1)

            # f(x) = 2*n * max_m(c0[m] + 2*x*b[m] + x^2*delta_m2j) / (S_rest + x)^2
            # Try a grid of x values around a_old_j
            # Also try x = 0
            x_candidates = [0.0, a_old_j]
            # Perturbations around current value
            for delta in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.5, -0.5]:
                x_c = a_old_j * (1.0 + delta)
                if x_c > 0:
                    x_candidates.append(x_c)
            # Set-based approaches
            for frac_x in [0.01, 0.02, 0.5, 1.0, 2.0, 5.0]:
                x_c = S_target * frac_x / n
                x_candidates.append(x_c)

            best_x = a_old_j
            best_f = _compute_score(a)

            for x_c in x_candidates:
                x_c = max(0.0, x_c)
                a_test = a.copy()
                a_test[j] = x_c
                f_val = _compute_score(a_test)
                if f_val < best_f:
                    best_f = f_val
                    best_x = x_c

            if best_x != a_old_j:
                a[j] = best_x
                improved_any = True

            a = _normalize(a, S_target)

        s = _compute_score(a)
        if s < best_score:
            best_score = s
            best = a.copy()

        if not improved_any:
            break

    return best, best_score


def _soft_gradient_descent(a0, deadline, S_target, max_iters=4000,
                           tau_start=3.0, tau_end=200.0, lr_init=0.08,
                           momentum=0.8, eval_every=50):
    """Softer gradient descent that focuses on top-K peaks."""
    n = len(a0)
    a = np.clip(a0.copy(), 0, None)
    a = _normalize(a, S_target)
    score = _compute_score(a)
    best = a.copy()
    best_score = score
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max_iters
        tau = tau_start + (tau_end - tau_start) * frac

        c = np.convolve(a, a)
        c_max = c.max()

        # Use a "top-K softmax" approach
        c_shifted = c - c_max
        w = np.exp(tau * c_shifted)
        w /= w.sum()

        # Also try a "hard top-K" variant: only weight the top K peaks
        K = max(1, int(0.05 * len(c)))
        top_indices = np.argsort(c)[-K:]
        w_hard = np.zeros_like(c)
        w_hard[top_indices] = 1.0 / K

        # Blend soft and hard weights
        alpha = 0.3 + 0.7 * frac
        w_blend = alpha * w_hard + (1.0 - alpha) * w

        grad = _fft_gradient(w_blend, a)
        grad -= grad.mean()

        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, 0.001)
        v = momentum * v + grad
        a = a - lr * v
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_target)

        if (it + 1) % eval_every == 0:
            s = _compute_score(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = _compute_score(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def _random_perturb(a, rng, scale=0.3):
    """Add random perturbation to escape local minima."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a_pert = a + scale * rng.random(n) * (a.max() * 0.5)
    a_pert = np.maximum(a_pert, 0.0)
    return _normalize(a_pert, S_target)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple starting sequences using
    momentum gradient descent, LP refinement, coordinate descent,
    and random restarts within the time budget.
    """
    global height_sequence_1
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def _update(a, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  New best: score={best_score:.8f}, n={len(a)}, "
                  f"t={time.monotonic()-t_start:.1f}s")

    # --- Generate starting sequences ---
    starts = []

    # Try to use the provided global sequence
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            for L in [64, 128, 256, 512, min(1024, len(gs))]:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
    except (NameError, Exception):
        pass

    # Diverse structured starts
    for n in [64, 128, 256, 512, 1024]:
        i = np.arange(n, dtype=np.float64)
        # Uniform
        starts.append(np.ones(n) * 0.5)
        # Various cosine modulations
        for freq in [1, 2, 3, 5, 7, 11]:
            starts.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * i / n))
        # Chirp
        starts.append(0.5 + 0.5 * np.cos(np.pi * (i / n) ** 2 * 4))
        # Linear ramp
        starts.append(np.linspace(0.2, 0.8, n))
        # Gaussian
        starts.append(np.exp(-2.0 * ((i - n / 2) / (n * 0.3)) ** 2))
        # Two-peak
        starts.append(np.exp(-3.0 * ((i - n * 0.25) / (n * 0.15)) ** 2) +
                      np.exp(-3.0 * ((i - n * 0.75) / (n * 0.15)) ** 2))
        # Random
        starts.append(rng.random(n) * 0.8 + 0.2)
        # Sparse-ish
        a_sparse = np.zeros(n)
        n_nonzero = max(4, n // 4)
        indices = rng.choice(n, size=n_nonzero, replace=False)
        a_sparse[indices] = rng.random(n_nonzero) + 0.3
        starts.append(a_sparse)

    # Limit starts to manage time
    max_starts = 40
    if len(starts) > max_starts:
        # Keep a diverse subset
        selected = []
        lengths = set()
        for s in starts:
            if len(s) not in lengths or len(selected) < 20:
                selected.append(s)
                lengths.add(len(s))
            if len(selected) >= max_starts:
                break
        starts = selected

    print(f"Starting optimization with {len(starts)} initial sequences, budget={budget_s}s")

    # --- Phase 1: Gradient descent on each start ---
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 60.0:
            break

        n = len(a0)
        S_target = np.sqrt(2.0 * n)
        t0 = time.monotonic()
        alloc = min(60.0, max(10.0, 1200.0 / len(starts)))
        sub_deadline = min(deadline - 60.0, t0 + alloc)

        a, score = _grad_descent_momentum(a0, sub_deadline, S_target,
                                          lr_init=0.12, momentum=0.85,
                                          tau_start=5.0, tau_end=60.0,
                                          max_iters=5000)
        _update(a, score)

        # Soft gradient variant
        if time.monotonic() < deadline - 60.0:
            sub_dl2 = min(deadline - 60.0, time.monotonic() + alloc)
            a2, score2 = _soft_gradient_descent(a0, sub_dl2, S_target,
                                                max_iters=3000,
                                                tau_start=3.0, tau_end=150.0)
            _update(a2, score2)

        elapsed = time.monotonic() - t0
        if (idx + 1) % 5 == 0:
            print(f"  [{idx+1}/{len(starts)}] n={n}, t={elapsed:.1f}s, "
                  f"best={best_score:.8f}")

    # --- Phase 2: LP refinement on best candidates ---
    if best_seq is not None and time.monotonic() < deadline - 30.0:

    if best_seq is not None and time.monotonic() < deadline - 20.0:

            sub_dl = min(deadline - 5.0, time.monotonic() + 120.0)
