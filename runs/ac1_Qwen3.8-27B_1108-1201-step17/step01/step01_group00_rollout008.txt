"""
LP-based iterative refinement for minimizing peak autocorrelation of non-negative sequences.
Uses linear programming to find optimal descent directions, combined with gradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize as scipy_minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    """Fast score computation."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def constraint_matrix(a):
    """Build the (2n-1) x n constraint matrix for conv(a, b) <= M."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            A[k, j] = a[k - j]
    return A

def lp_find_direction(a, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0. Returns b or None."""
    n = len(a)
    A = constraint_matrix(a)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    bnds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bnds, method='highs')
    if res.success:
        return res.x
    return None

def lp_iterate(a, n_iter, t_start=0.1, t_end=0.5, verbose=True, verbose_every=50):

    Returns best sequence found.
    """

        t = t_start + (t_end - t_start) * elapsed_frac
        
        # Compute current peak

        M = np.max(conv)
        
        # Solve LP
        b = lp_find_direction(a, M)
        if b is None or np.sum(b) < 1e-10:
            if verbose:
                print(f"  LP failed at iteration {it}")
            break
        
        # Rescale b to match sum of a
        sum_b = np.sum(b)
        sum_a = np.sum(a)
        b = b * (sum_a / sum_b)
        
        # Blend
        a_new = (1 - t) * a + t * b
        a_new = np.clip(a_new, 0.0, None)
        
        # Check score
        new_score = score_fast(a_new)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        
        # Accept or reject (simple: always accept for now, or use line search)
        a = a_new
        
        if verbose and it % verbose_every == 0:
            print(f"  iter={it}, t={t:.3f}, score={new_score:.6f}, best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

"""
LP-based iterative refinement for minimizing peak autocorrelation of non-negative sequences.
Uses linear programming to find optimal descent directions at each step, combined with
L-BFGS-B gradient polishing. Multiple initializations and sequence lengths are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize as scipy_minimize


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score computation matching evaluate_sequence formula."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_matrix(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    A = np.zeros((2 * n - 1, n))
    mask = (idx >= 0) & (idx < n)
    A[mask] = a[idx[mask]]
    return A


def lp_solve_direction(a, M):
    """Solve LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bnds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bnds, method='highs')
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def lp_iterate(a_init, n_iter=300, verbose=True, verbose_every=50, time_limit=None, t0=None):
    """
    Iterative LP-based refinement with backtracking line search.
    Returns (best_sequence, best_score).
    """
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    best_a = a.copy()
    best_score = score_fast(a)
    n = len(a)
    
    if t0 is None:
        t0 = time.monotonic()
    
    no_improve_count = 0
    A = build_constraint_matrix(a)
    
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(a, a)
        M = np.max(conv)
        
        # Solve LP for direction
        b = lp_solve_direction(a, M)
        if b is None or np.sum(b) < 1e-12:
            break
        
        sum_a = np.sum(a)
        sum_b = np.sum(b)
        b = b * (sum_a / sum_b)
        
        # Backtracking line search
        best_t = 0.0
        best_new_score = best_score
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
            a_candidate = (1 - t) * a + t * b
            a_candidate = np.clip(a_candidate, 0.0, None)
            s = score_fast(a_candidate)
            if s < best_score - 1e-10:
                best_t = t
                best_new_score = s
                break
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * b
            a = np.clip(a, 0.0, None)
            no_improve_count = 0
        else:
            # Try pure replacement
            s_replace = score_fast(b)
            if s_replace < best_score:
                a = b.copy()
                best_t = 1.0
                best_new_score = s_replace
                no_improve_count = 0
            else:
                no_improve_count += 1
                # Inject noise to escape stagnation

                    noise = np.exp(rng_random_normal(n, 0.0, 0.05))
                    a = a * noise
                    a = np.clip(a, 0.01, None)
        
        if best_new_score < best_score:
            best_score = best_new_score
            best_a = (1 - 0) * a.copy() if best_t < 1 else b.copy()
            best_a = a.copy()
        
        # Rebuild constraint matrix periodically (a changed)
        if it % 10 == 0:
            A = build_constraint_matrix(a)
        
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            cur_score = score_fast(a)
            print(f"    iter={it}, cur={cur_score:.6f}, best={best_score:.6f}, t={best_t:.3f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def rng_random_normal(n, mean, std):
    """Generate random normal array (top-level to avoid closure)."""
    return np.random.default_rng().normal(mean, std, size=n)


def gradient_polish(a_init, time_limit, tau_sequence=None):
    """
    L-BFGS-B polishing with smooth max approximation.
    Tries decreasing temperatures for progressively sharper optimization.
    """
    n = len(a_init)
    best_a = a_init.astype(np.float64).copy()
    best_score = score_fast(best_a)
    t0 = time.monotonic()
    
    if tau_sequence is None:
        tau_sequence = [1.0 / n, 0.5 / n, 0.2 / n, 0.1 / n, 0.05 / n]
    
    for tau in tau_sequence:
        if time.monotonic() - t0 > time_limit:
            break
        
        def objective(x):
            x = np.clip(x, 0.0, None)
            conv = fft_convolve(x, x)
            max_c = np.max(conv)
            shifted = (conv - max_c) / tau
            exp_shifted = np.exp(shifted)
            smooth_max = max_c + tau * np.log(np.sum(exp_shifted))
            sum_x = np.sum(x)
            if sum_x < 0.01:
                return 1e10
            return 2.0 * n * smooth_max / (sum_x ** 2)
        
        def gradient(x):
            x = np.clip(x, 0.0, None)
            conv = fft_convolve(x, x)
            max_c = np.max(conv)
            shifted = (conv - max_c) / tau
            exp_shifted = np.exp(shifted)
            sum_exp = np.sum(exp_shifted)
            w = exp_shifted / sum_exp
            smooth_max = max_c + tau * np.log(sum_exp)
            sum_x = np.sum(x)
            if sum_x < 0.01:
                return np.zeros(n)
            
            # Gradient of smooth_max w.r.t. x
            # d(conv[k])/dx[j] = 2*x[k-j] (if valid)
            # d(smooth_max)/dx[j] = 2*sum_k w[k]*x[k-j] = 2*(flip(x)*w)[n-1+j]
            flip_x = x[::-1]
            corr = fft_convolve(flip_x, w)
            g_smooth = 2.0 * corr[n - 1:2 * n - 1]
            
            # d(score)/dx[j] = (2n/sum_x^2)*(g_smooth[j] - 2*smooth_max/sum_x)
            grad = (2.0 * n / sum_x ** 2) * (g_smooth - 2.0 * smooth_max / sum_x)
            grad[x < 1e-12] = 0.0
            return grad
        
        try:
            max_iter = 3000 if tau > 0.2 / n else 5000
            res = scipy_minimize(
                objective, best_a.copy(),
                jac=gradient,
                method='L-BFGS-B',
                bounds=[(0, 1000)] * n,
                options={'maxiter': max_iter, 'maxfun': max_iter * 10,
                         'ftol': 1e-14, 'gtol': 1e-12}
            )
            a_res = np.clip(res.x, 0.0, None)
            s_res = score_fast(a_res)
            if s_res < best_score:
                best_score = s_res
                best_a = a_res.copy()
                print(f"    tau={tau:.4f}/{n:.0f}, score={s_res:.6f}")
        except Exception as e:
            print(f"    L-BFGS-B error at tau={tau:.4f}: {e}")
    
    return best_a, best_score


def simulated_annealing_polish(a_init, time_limit, rng):
    """Simulated annealing with targeted peak reduction."""
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = score_fast(a)
    n = len(a)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted: reduce peak contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.75:
            # Symmetrize partially
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.015, size=n))
        elif r < 0.9:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        else:
            # Global small
            candidate = a * np.exp(rng.normal(0.0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0.01, None)
        new_score = score_fast(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
        
        if new_score < score_fast(a):
            a = candidate
        elif rng.random() < np.exp(-(new_score - score_fast(a)) / max(temperature, 1e-12)):
            a = candidate
        
        temperature *= cooling
    
    return best_a, best_score


def make_initializer(n, init_type, rng, provided_seq=None):
    """Create initial sequence of given type and length."""
    if init_type == 'provided' and provided_seq is not None:
        if len(provided_seq) == n:
            return provided_seq.copy().astype(np.float64)
        else:
            x_old = np.linspace(0, 1, len(provided_seq))
            x_new = np.linspace(0, 1, n)
            return np.interp(x_new, x_old, provided_seq).astype(np.float64)
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)
        alpha = 0.3 + 0.5 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
        a = np.clip(a, 0.01, None)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'multi_tone':
        t = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for k in range(6):
            freq = rng.uniform(0.3, 2.0) * (k + 1)
            amp = 0.15 / (k + 1)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'random':
        a = 1.0 + 0.4 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'symmetric':
        half = n // 2
        a_half = 1.0 + 0.3 * rng.standard_normal(half)
        a_half = np.clip(a_half, 0.01, None)
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, [1.0], a_half[::-1]])
        return a
    else:
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-based iterative refinement + gradient polishing
    + simulated annealing. Explores multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    total_budget = max(10.0, float(budget_s))
    deadline = t_global + total_budget - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided_seq.copy()
    
    # Search configurations: (length, [(init_type, lp_iters), ...])
    # Time allocation weighted by sqrt(n) for computational cost
    configs = [
        (256, [('provided', 400), ('chirp', 300), ('multi_tone', 300), ('random', 250), ('symmetric', 250)]),
        (384, [('chirp', 300), ('multi_tone', 250), ('random', 200)]),
        (512, [('chirp', 250), ('multi_tone', 200)]),
        (768, [('chirp', 200)]),
        (1024, [('chirp', 150)]),
    ]
    
    # Phase 1: LP-based iterative refinement
    print("\n=== Phase 1: LP-based iterative refinement ===")
    for n, init_list in configs:
        if time.monotonic() > deadline - 120:
            break
        time_left = deadline - time.monotonic() - 90
        if time_left < 60:
            break
        
        # Estimate time per LP iteration based on n
        est_time_per_iter = 0.02 * (n / 256.0)  # rough estimate
        total_iters = sum(iters for _, iters in init_list)
        est_total_time = total_iters * est_time_per_iter
        scale = min(1.0, time_left / max(1, est_total_time))
        
        print(f"\n  --- n={n}, time_left={time_left:.0f}s, scale={scale:.2f} ---")
        
        for init_type, base_iters in init_list:
            if time.monotonic() > deadline - 60:
                break
            n_iters = max(50, int(base_iters * scale))
            time_per_run = time_left / 2.0
            
            a0 = make_initializer(n, init_type, rng, provided_seq)
            a0 = np.clip(a0, 0.01, None)
            a0 = a0 / np.sum(a0)  # normalize sum to 1
            s0 = score_fast(a0)
            
            t_run = time.monotonic()
            a_best, lp_score = lp_iterate(
                a0, n_iter=n_iters, verbose=(n == 256),
                verbose_every=100, time_limit=time_per_run, t0=t_run
            )
            elapsed = time.monotonic() - t_run
            
            print(f"    init='{init_type}', initial={s0:.6f}, lp={lp_score:.6f}, time={elapsed:.1f}s")
            
            if lp_score < best_score:
                best_score = lp_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}, '{init_type}') ***")
        
        print(f"  Best after n={n}: {best_score:.6f}")
    
    # Phase 2: Gradient polishing
    print(f"\n=== Phase 2: Gradient polishing (L-BFGS-B) ===")
    if best_seq is not None:
        polish_time = min(600.0, max(30.0, deadline - time.monotonic() - 30))
        print(f"  Polishing best (score={best_score:.6f}), time={polish_time:.0f}s")
        polished, polish_score = gradient_polish(best_seq, polish_time)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After gradient polish: {best_score:.6f}")
    
    # Phase 3: Additional LP iterations on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: Extra LP refinement ===")
        extra_time = min(300.0, max(20.0, deadline - time.monotonic() - 30))
        n_extra = int(extra_time / (0.05 * (len(best_seq) / 256.0)))
        t_extra = time.monotonic()
        a_extra, extra_score = lp_iterate(
            best_seq, n_iter=n_extra, verbose=True,
            verbose_every=100, time_limit=extra_time, t0=t_extra
        )
        if extra_score < best_score:
            best_score = extra_score
            best_seq = a_extra.copy()
            print(f"  After extra LP: {best_score:.6f}")
    
    # Phase 4: Final gradient + annealing polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 4: Final polish ===")
        remaining = deadline - time.monotonic() - 15
        
        # Gradient
        grad_time = remaining * 0.5
        if grad_time > 10:
            p1, s1 = gradient_polish(best_seq, grad_time)
            if s1 < best_score:
                best_score = s1
                best_seq = p1.copy()
                print(f"  After final gradient: {best_score:.6f}")
        
        # Annealing
        anneal_time = remaining * 0.4
        if anneal_time > 10:
            p2, s2 = simulated_annealing_polish(best_seq, anneal_time, rng)
            if s2 < best_score:
                best_score = s2
                best_seq = p2.copy()
                print(f"  After annealing: {best_score:.6f}")
        
        # One more LP pass
        if time.monotonic() < deadline - 10:
            final_lp_time = deadline - time.monotonic() - 5
            if final_lp_time > 10:
                n_final = int(final_lp_time / 0.05)
                p3, s3 = lp_iterate(best_seq, n_iter=n_final, verbose=False,
                                    time_limit=final_lp_time, t0=time.monotonic())
                if s3 < best_score:
                    best_score = s3
                    best_seq = p3.copy()
                    print(f"  After final LP: {best_score:.6f}")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
