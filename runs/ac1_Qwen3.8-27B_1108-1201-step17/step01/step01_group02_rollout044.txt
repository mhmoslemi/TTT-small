import numpy as np

def score_a(a):
    """Compute score = 2*n*max(conv)/sum^2"""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def normalize_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, len(v)+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    s = proj.sum()
    if s < 1e-30:
        return np.ones(len(v)) / len(v)
    return proj / s  # Should already sum to ~1, but normalize for safety

def compute_subgradient(a, k_star):
    """Compute subgradient of max_k (a*a)(k) at k_star with respect to a"""
    n = len(a)
    g = np.zeros(n)
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]
    return g

def smoothed_subgradient(a, T=10.0):
    """Compute smoothed subgradient using log-sum-exp trick"""
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax weights
    conv_shifted = conv - conv.max()
    weights = np.exp(T * conv_shifted)
    weights /= weights.sum()
    # Weighted average of subgradients
    g = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-10:
            continue
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g[mask] += weights[k] * 2.0 * a[idx[mask]]
    return g, conv.max()

"""
Minimize peak auto-convolution using subgradient descent + LP refinement.
Key insight: asymmetric sequences can achieve score < 2 (unlike symmetric ones).
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

# --- Helper functions ---

def score_a(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def normalize_nonneg(v):
    v = np.maximum(0.0, v)
    s = v.sum()
    if s < 1e-30:
        return np.ones(len(v)) / len(v)
    return v / s

def project_simplex(v):
    """Exact Euclidean projection onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    s = proj.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return proj / s

def subgradient_at_peak(a, k_star):
    """Subgradient of (a*a)(k_star) w.r.t. a"""
    n = len(a)
    g = np.zeros(n)
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]
    return g

def subgrad_descent(a_init, time_limit, lr_init=0.01, lr_decay=0.999, rho_switch=0.9):
    """
    Projected subgradient descent to minimize max(conv(a,a)) subject to sum(a)=1, a>=0.
    Returns best sequence and its score.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a_init.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()  # sum=1 so this is the score/2... 
    
    # Actually, score = 2*n*max_conv/sum^2. Since sum=1, score = 2*n*max_conv.
    best_obj = np.convolve(a, a).max()  # This is max_conv, score = 2*n*max_conv
    
    it = 0
    stagnant = 0
    while time.monotonic() - start_time < time_limit:
        conv = np.convolve(a, a)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 50:
            # Try a random restart from current
            a = a + 0.01 * np.random.randn(n)
            a = project_simplex(a)
            stagnant = 0
        
        k_star = np.argmax(conv)
        g = subgradient_at_peak(a, k_star)
        
        # AdaGrad-like learning rate
        lr = lr_init / np.sqrt(1 + it * lr_decay)
        # Clip lr to reasonable range
        lr = min(lr, 0.5)
        
        a_new = a - lr * g
        a = project_simplex(a_new)
        it += 1
    
    final_obj = np.convolve(a, a).max()
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    
    best_score = 2.0 * n * best_obj
    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(i, i+n, dtype=np.int32) for i in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))

def lp_step(a, M, timeout=30):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine(a, time_limit, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = np.convolve(a, a).max()
    best_score = 2.0 * n * best_obj
    
    stagnant = 0
    it = 0
    while time.monotonic() - start_time < time_limit:
        if stagnant > 30:
            break
        
        conv = np.convolve(a, a)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            best_score = 2.0 * n * obj
            stagnant = 0
            if verbose and it % 20 == 0:
                print(f"      LP iter {it}: obj={obj:.8f}, score={best_score:.6f}")
        else:
            stagnant += 1
        
        M = obj
        remaining = time_limit - (time.monotonic() - start_time)
        if remaining < 5:
            break
        timeout = max(2.0, min(30.0, remaining * 0.2))
        
        b = lp_step(a, M, timeout=timeout)
        if b is None:
            break
        
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-10:
            break
        
        b = b / sum_b
        alpha = 0.5
        a_new = (1 - alpha) * a + alpha * b
        a = normalize_nonneg(a_new)
        it += 1
    
    final_obj = np.convolve(a, a).max()
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
        best_score = 2.0 * n * best_obj
    
    if verbose:
        print(f"      LP done: score={best_score:.6f}, {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def local_search(a, time_limit, rng, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_a(a)
    
    accept = 0
    while time.monotonic() - start_time < time_limit:
        # Random perturbation
        width = int(rng.integers(max(2, n//100), max(3, n//5)))
        start = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        candidate[start:start+width] *= np.exp(rng.normal(0, 0.15, width))
        candidate = normalize_nonneg(candidate)
        s = score_a(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            a = candidate.copy()
            accept += 1
    
    if verbose:
        print(f"      LS: score={best_score:.6f}, {accept} accepts, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
