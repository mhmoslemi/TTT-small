```python
import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time

def _build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _solve_lp(a, M):
    """Solve max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = _build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x
```

```python
from scipy import sparse
```

```python
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""

    b = np.convolve(a, a)
    return 2.0 * n * float(np.max(b))


def _project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Convolve two sequences using FFT."""
    n = len(a) + len(b) - 1
    N = next_fast_len(n)
```

```python
            if reheat_count > max_reheats:
                break
            # Reheat: increase step, add perturbation
            eta *= 1.5
            temperature *= 0.5
            v *= 0.5
            # Small random perturbation
            noise = rng.standard_normal(n) * eta * 0.1
            a = _project_simplex(a + noise, 1.0)
            no_improve = 0
            continue

        # Adaptive temperature: focus more on peaks
        temperature *= 0.995
        temperature = max(temperature, 0.1)

        # Get top M peak locations with exponential weighting
        if M_peaks >= len(b):
            k_idx = np.arange(len(b))
            b_sel = b.copy()
        else:
            k_idx = np.argpartition(-b, M_peaks)[:M_peaks]

        b_sel = b[k_idx]
        wts = np.exp(-temperature * (b_sel - b_sel.min()) / max(b_max * 1e-10, 1e-15))
        # Actually use: wts proportional to exp(-temp * (b_max - b_sel))
        wts = np.exp(-temperature * 10.0 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)

        # Compute weighted gradient via FFT
        # grad = 2 * sum_k w_k * a_correlated_at_k
        # Build weighted "impulse response"
        # grad_j = 2 * sum_k w_k * a_{k-j} (if valid)
        # This is 2 * cross_correlate(w_extended, a)
        # Where w is placed at the right positions

        # Actually, let's compute it as:
        # grad_j = 2 * sum_k w_k * a_{k-j}
        # = 2 * sum_k w_k * a[k-j] for k where 0<=k-j<n
        # This is 2 * (w * a_reversed) offset... 
        # Let me just use the cross-correlate function
        # Define w_full of length 2n-1
        w_full = np.zeros(2 * n - 1)
        w_full[k_idx] = wts
        c = _cross_correlate(w_full, a)
        grad = 2.0 * c

        # Momentum update
        v = mu * v + (1.0 - mu) * grad
        a_new = _project_simplex(a - eta * v / (np.linalg.norm(v) + 1e-15), 1.0)

        # Check if function value worsened significantly (restart momentum)
        b_new = _conv(a_new, a_new)
        if np.max(b_new) > b_max * 1.01:
            v *= 0.5
            eta *= 0.8
            a_new = _project_simplex(a - eta * grad / (np.linalg.norm(grad) + 1e-15), 1.0)

        a = a_new
        eta *= 0.999  # slow cooldown

    return best, best_score


def _fw_lp_step(a, max_time=30.0, K_constraints=200):
    """
    One Frank-Wolfe step: solve LP to find best direction g.
    Uses top-K constraints for efficiency.
    Returns (g_direction, success).
    """
    n = len(a)
    b = _conv(a, a)
    M = np.max(b)

    if M < 1e-15:
        return None, False

    # Get top K constraint indices
    K = min(K_constraints, 2 * n - 1)
    top_k = np.argpartition(-b, K)[:K]

    # Build constraint matrix for top K rows: A[k,j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in top_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A_sparse = sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))

    # LP: max sum(g) s.t. A_sparse @ g <= M, g >= 0
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n

    t0 = time.monotonic()
    res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': max_time, 'presolve': True})
    solve_time = time.monotonic() - t0

    if res.status != 0 or res.x is None:
        return None, False

    g = res.x
    S_g = np.sum(g)
    if S_g < 1e-10:
        return None, False

    # Check feasibility against ALL constraints
    b_check = _conv(a, g)
    max_violation = np.max(b_check) - M
    if max_violation > 1e-6 * M:
        # Infeasible: scale down g
        scale = M / max(np.max(b_check), 1e-15)
        g *= scale
        S_g = np.sum(g)
        if S_g < 1e-10:
            return None, False

    return g / S_g, True


def _fw_line_search(a, g, max_iters=20):
    """
    Find best t in [0,1] minimizing max((a+t*(g-a)) * (a+t*(g-a))).
    Uses golden section search.
    """
    d = g - a
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    def obj(t):
        x = a + t * d
        b = _conv(x, x)
        return np.max(b)

    lo, hi = 0.0, 1.0
    c = hi - phi * (hi - lo)
    d_pt = lo + phi * (hi - lo)
    fc = obj(c)
    fd = obj(d_pt)

    for _ in range(max_iters):
        if fc < fd:
            hi = d_pt
            d_pt = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d_pt
            fc = fd
            d_pt = lo + phi * (hi - lo)
            fd = obj(d_pt)

    t_best = (lo + hi) / 2.0
    return t_best


def _make_two_level(n, eps):
    """Anti-symmetric two-level sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    S = np.sum(a)
    return a / S


def _make_smooth_profile(n, eps):
    """Smooth anti-symmetric profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def _make_random_smooth(n, rng, n_freqs=5):
    """Random smooth sequence via low-frequency sinusoids."""
    a = np.ones(n)
    i = np.arange(n, dtype=np.float64
    for _ in range(n_freqs):
        f = rng.integers(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a += amp * np.sin(2 * np.pi * f * i / n + phase)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization to minimize the normalized autocorrelation peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.1, float(budget_s) - 5.0)

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Normalize
    S = np.sum(parent)
    if S < 1e-10:
        parent = np.ones(n)
        S = n
    parent = parent / S

    global_best = parent.copy()
    global_best_score = _score(global_best)
    print(f"Initial score: {global_best_score:.6f} (n={n})")

    # ============ Phase 1: Multi-start quick optimization (120s) ============
    t_phase1_end = time.monotonic() + min(120.0, (deadline - time.monotonic()) * 0.15)
    print(f"\n--- Phase 1: Multi-start ({t_phase1_end - time.monotonic():.0f}s) ---")

    starts = [parent.copy()]
    starts.append(_make_two_level(n, 0.5))
    starts.append(_make_two_level(n, 0.8))
    starts.append(_make_smooth_profile(n, 0.7))
    starts.append(_make_random_smooth(n, rng, 5))
    starts.append(_make_random_smooth(n, rng, 3))

    # Add a "broadband" start: raise-cosine window with random perturbations
    i = np.arange(n, dtype=np.float64)
    a_rc = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    a_rc = np.maximum(a_rc, 1e-6)
    a_rc += 0.1 * rng.standard_normal(n)
    a_rc = np.maximum(a_rc, 1e-6)
    starts.append(a_rc / np.sum(a_rc))

    per_start_time = max(5.0, (t_phase1_end - time.monotonic()) / len(starts))
    for idx, a_s in enumerate(starts):
        if time.monotonic() >= t_phase1_end:
            break
        t_end = min(time.monotonic() + per_start_time, t_phase1_end)
        best_a, best_s = _subgrad_momentum(a_s, t_end, rng, M_peaks=5, eta_scale=0.2)
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  Start {idx}: NEW BEST {global_best_score:.6f}")
        else:
            print(f"  Start {idx}: {best_s:.6f} (best={global_best_score:.6f})")

    # ============ Phase 2: FISTA smooth optimization (largest chunk) ============
    remaining = deadline - time.monotonic()
    t_phase2_end = time.monotonic() + remaining * 0.4
    print(f"\n--- Phase 2: FISTA smooth ({t_phase2_end - time.monotonic():.0f}s) ---")

    best_a, best_s = _fista_smooth(global_best, t_phase2_end,
                                    beta_start=5.0, beta_end=3000.0, max_iters=100000)
    if best_s < global_best_score:
        global_best = best_a.copy()
        global_best_score = best_s
        print(f"  FISTA: NEW BEST {global_best_score:.6f}")
    else:
        print(f"  FISTA: {best_s:.6f} (best={global_best_score:.6f})")

    # ============ Phase 3: Frank-Wolfe with LP ============
    remaining = deadline - time.monotonic()
    t_phase3_end = time.monotonic() + remaining * 0.25
    print(f"\n--- Phase 3: Frank-Wolfe LP ({t_phase3_end - time.monotonic():.0f}s) ---")

    n_fw = 0
    while time.monotonic() < t_phase3_end:
        t0 = time.monotonic()
        g, success = _fw_lp_step(global_best, max_time=20.0, K_constraints=300)
        if not success:
            print(f"  FW: LP failed at iteration {n_fw}")
            break

        # Line search
        t_best = _fw_line_search(global_best, g)
        a_new = global_best + t_best * (g - global_best)
        a_new = _project_simplex(a_new, 1.0)

        new_score = _score(a_new)
        if new_score < global_best_score - 1e-10:
            global_best = a_new.copy()
            global_best_score = new_score
            print(f"  FW iter {n_fw}: NEW BEST {global_best_score:.6f} (t={t_best:.4f})")
        else:
            # Try smaller step
            for t_try in [0.5 * t_best, 0.1, 0.01]:
                a_try = _project_simplex(global_best + t_try * (g - global_best), 1.0)
                s_try = _score(a_try)
                if s_try < global_best_score - 1e-10:
                    global_best = a_try.copy()
                    global_best_score = s_try
                    print(f"  FW iter {n_fw}: NEW BEST {global_best_score:.6f} (t={t_try:.4f})")
                    break

        n_fw += 1
        if n_fw > 50:
            break

    print(f"  FW: {n_fw} iterations completed")

    # ============ Phase 4: Subgradient polishing ============
    remaining = deadline - time.monotonic()
    t_phase4_end = time.monotonic() + remaining * 0.4
    print(f"\n--- Phase 4: Subgradient polish ({t_phase4_end - time.monotonic():.0f}s) ---")

    best_a, best_s = _subgrad_momentum(global_best, t_phase4_end, rng,
                                        M_peaks=20, eta_scale=0.1)
    if best_s < global_best_score:
        global_best = best_a.copy()
        global_best_score = best_s
        print(f"  Polish: NEW BEST {global_best_score:.6f}")
    else:
        print(f"  Polish: {best_s:.6f} (best={global_best_score:.6f})")

    # ============ Phase 5: Final diversification ============
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n--- Phase 5: Diversification ({remaining:.0f}s) ---")
        n_div = max(1, int(remaining / 60))
        per_div = remaining / n_div

        for d in range(n_div):
            if time.monotonic() >= deadline:
                break

            # Different perturbation strategies
            strategy = d % 4
            if strategy == 0:
                # Smooth perturbation
                i = np.arange(n, dtype=np.float64)
                f = rng.integers(1, n // 4)
                noise = np.sin(2 * np.pi * f * i / n + rng.uniform(0, 2 * np.pi))
                a_pert = _project_simplex(global_best + 0.05 * noise, 1.0)
            elif strategy == 1:
                # Sparse perturbation
                idx = rng.choice(n, size=max(1, n // 100), replace=False)
                a_pert = global_best.copy()
                a_pert[idx[::2]] += 0.005
                a_pert[idx[1::2]] -= 0.005
                a_pert = _project_simplex(a_pert, 1.0)
            elif strategy == 2:
                # Blend with two-level
                a_2l = _make_two_level(n, rng.uniform(0.3, 0.9))
                a_pert = _project_simplex(0.8 * global_best + 0.2 * a_2l, 1.0)
            else:
                # Small random perturbation
                a_pert = _project_simplex(global_best + 0.02 * rng.standard_normal(n), 1.0)

            t_end = min(time.monotonic() + per_div, deadline)
            best_a, best_s = _subgrad_momentum(a_pert, t_end, rng,
                                                M_peaks=10, eta_scale=0.05)
            if best_s < global_best_score:
                global_best = best_a.copy()
                global_best_score = best_s
                print(f"  Div {d}: NEW BEST {global_best_score:.6f}")

    # Final: one more FISTA polish on the best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n--- Final FISTA polish ({remaining:.0f}s) ---")
        best_a, best_s = _fista_smooth(global_best, min(time.monotonic() + remaining, deadline),
                                        beta_start=100.0, beta_end=10000.0, max_iters=50000)
        if best_s < global_best_score:
            global_best = best_a
            global_best_score = best_s
            print(f"  Final polish: NEW BEST {global_best_score:.6f}")

    # Final evaluation
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f}")
    return result
```
