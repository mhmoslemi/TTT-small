Multi-phase optimizer for minimizing peak autocorrelation of non-negative step functions.
Combines adaptive gradient descent, coordinate descent, and simulated annealing.

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Compute the evaluation score."""
    n = len(a)
    s = np.sum(a)

    return 2.0 * n * np.max(c) / (s * s)

    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights, c

def gradient_descent_optimize(a_init, time_limit, rng):
    """Adam gradient descent with annealed soft-max in log-space."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    # Initialize theta in log-space
    a0 = np.clip(a_init, 1e-6, None)
    # Normalize to sum = S
    a0 = a0 / np.sum(a0) * S
    theta = np.log(a0 / np.sum(a0) + 1e-10)  # log of normalized proportions
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    best_theta = theta.copy()
    best_score = compute_score(a0)
    
    tau_start = 50.0 / n
    tau_end = 0.1 / n
    lr = 5.0
    
    t0 = time.monotonic()
    plateau = 0
    plateau_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        # Anneal temperature
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed * 0.8)
        
        # Get current a from theta
        t = theta - np.max(theta)
        e = np.exp(t)
        a = e / np.sum(e) * S
        
        # Compute gradient
        value, g_a, c = grad_soft_max_conv(a, tau)
        
        # Chain rule: d(value)/d(theta)
        # a = softmax(theta) * S
        # da_i/d(theta_j) = a_i * (delta_ij - softmax(theta)_j) = a_i/S * (delta_ij - e_j/sum(e))
        # Let p = softmax(theta) = e/sum(e), so a = p * S
        # da_i/d(theta_j) = S * p_i * (delta_ij - p_j)
        # d(value)/d(theta_j) = sum_i d(value)/d(a_i) * da_i/d(theta_j)
        # = sum_i g_a[i] * S * p_i * (delta_ij - p_j)
        # = S * (g_a[j] * p_j - p_j * sum_i g_a[i] * p_i)
        # = S * p_j * (g_a[j] - sum_i g_a[i] * p_i)
        
        p = a / S
        inner = np.dot(g_a, p)
        g_theta = S * p * (g_a - inner)
        
        # Also account for the /S^2 in the score
        # score = 2n * value / S^2 (since sum(a) = S, score = 2n*max_conv/S^2)
        # Actually score = 2n * value / (sum(a))^2 = 2n * value / S^2
        # d(score)/d(theta) = 2n/S^2 * d(value)/d(theta)
        # The 2n/S^2 factor is constant, so it doesn't affect the direction
        
        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        # Adaptive learning rate
        lr_eff = lr / (1 + step * 0.001)  # slight decay
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate true score
        a_check = softmax_normalize(theta) * S
        score = compute_score(a_check)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            plateau = 0
        else:
            plateau += 1
        
        if step % 5000 == 0:

"""
Multi-phase optimizer for minimizing peak autocorrelation of non-negative step functions.
Uses gradient descent with annealed soft-max, peak-focused coordinate descent,
and simulated annealing with diverse moves. Focused on fine-tuning from good starting points.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def soft_max_conv(a, tau):
    """Compute soft-max of conv(a,a) and softmax weights."""
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights, c


def grad_soft_max_conv(a, tau):
    """Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a."""
    n = len(a)
    value, w, c = soft_max_conv(a, tau)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)

    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in [1, 2, 3, 5, 8]:
        amp = 0.3 / k
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng):

def gradient_descent_optimize(a_init, time_limit, rng):
    """Adam gradient descent with annealed soft-max in log-space."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a0 = np.clip(a_init, 1e-6, None)
    a0 = a0 / np.sum(a0) * S
    theta = np.log(a0 / np.sum(a0) + 1e-10)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_theta = theta.copy()
    best_score = compute_score(a0)

    tau_start = 50.0 / n
    tau_end = 0.05 / n
    lr_base = 3.0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed_frac * 0.7)

        p = softmax_normalize(theta)
        a = p * S

        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(g_a, p)
        g_theta = S * p * (g_a - inner)

        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr_base / (1.0 + 0.0005 * step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_check = softmax_normalize(theta) * S
        score = compute_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

    a_best = softmax_normalize(best_theta) * S
    final_score = compute_score(a_best)
    return a_best, final_score

def pair_coordinate_descent(a_init, time_limit, rng):

def simulated_annealing(a_init, time_limit, rng):
    """
    Simulated annealing with diverse moves including peak-directed perturbations
    and sum-preserving swaps.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, 1000.0).copy()
    current_score = compute_score(a)
    if not np.isfinite(current_score):
        return a, np.inf

    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    temperature = max(0.001, (best_score - 1.45) * 0.1)
    cooling = 0.99995
    iterations = 0

    while time.monotonic() - t0 < time_limit and temperature > 1e-8:
        iterations += 1
        candidate = a.copy()
        r = rng.random()

        if r < 0.3:
            # Peak-directed: reduce top contributors to peak, increase others
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            contrib = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n, peak_idx + 1)
            for i in range(lo, hi):
                j = peak_idx - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(-contrib)[:top_k]
            candidate[top_idx] *= np.exp(rng.normal(-0.03 * temperature * 100, 0.01, size=top_k))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(5, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03 * temperature * 100, 0.01, size=n_inc))
        elif r < 0.5:
            # Sum-preserving swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-6:
                delta = a[i] * rng.uniform(0.05, 0.3) * (1 if rng.random() < 0.7 else -1)
                candidate[i] = max(0.0, a[i] - delta)
                candidate[j] = a[j] + delta
        elif r < 0.7:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width + 1))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, size=width))
        elif r < 0.85:
            # Small global noise
            candidate *= np.exp(rng.normal(0, 0.01, size=n))
        else:
            # Symmetrize with noise
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.02, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = compute_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

        temperature *= cooling

        if iterations % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA: iter={iterations}, score={current_score:.8f}, best={best_score:.8f}, T={temperature:.6f}, t={elapsed:.1f}s")

    return best, best_score

    # Try a slightly different length
    candidates.append(('chirp_384', 384, make_chirp_init(384, 0.4, rng)))

    print(f"\n=== Phase 1: Gradient descent ({len(candidates)} candidates) ===")
    gd_results = []
    gd_time_each = min(60.0, time_budget / (len(candidates) * 4))

    for name, n, a0 in candidates:
        if time.monotonic() > deadline - time_budget * 0.4:
            break
        a0 = np.clip(a0, 1e-6, 1000.0)
        t0 = time.monotonic()
        a_gd, s_gd = gradient_descent_optimize(a0, gd_time_each, rng)
        elapsed = time.monotonic() - t0
        print(f"  GD [{name}, n={n}]: score={s_gd:.8f}, time={elapsed:.1f}s")
        gd_results.append((a_gd, s_gd))

        if s_gd < best_score:
            best_score = s_gd
            best_seq = a_gd.copy()
            print(f"    *** New best: {best_score:.8f} ***")

    # Select the best GD result for detailed optimization
    gd_results.sort(key=lambda x: x[1])
    a_main = gd_results[0][0].copy() if gd_results else (start_seq.copy() if start_seq is not None else np.ones(256))
    n_main = len(a_main)
    print(f"\n=== Main optimization on n={n_main}, initial score={compute_score(a_main):.8f} ===")

    # Phase 2: Coordinate descent (main workhorse)
    print("\n=== Phase 2: Coordinate descent ===")
    cd_time = min(600.0, time_budget * 0.35)
    a_cd, s_cd = coordinate_descent(a_main, cd_time, rng, use_peak_focus=True)
    print(f"  After CD: score={s_cd:.8f}")
    if s_cd < best_score:
        best_score = s_cd
        best_seq = a_cd.copy()
        print(f"    *** New best: {best_score:.8f} ***")
    a_main = a_cd

    # Phase 3: Pair coordinate descent
    print("\n=== Phase 3: Pair coordinate descent ===")
    pair_time = min(200.0, time_budget * 0.12)
    a_pair, s_pair = pair_coordinate_descent(a_main, pair_time, rng)
    print(f"  After Pair CD: score={s_pair:.8f}")
    if s_pair < best_score:
        best_score = s_pair
        best_seq = a_pair.copy()
        print(f"    *** New best: {best_score:.8f} ***")
    a_main = a_pair

    # Phase 4: More coordinate descent
    print("\n=== Phase 4: Extended coordinate descent ===")
    cd_time2 = min(400.0, time_budget * 0.2)
    a_cd2, s_cd2 = coordinate_descent(a_main, cd_time2, rng, use_peak_focus=True)
    print(f"  After CD2: score={s_cd2:.8f}")
    if s_cd2 < best_score:
        best_score = s_cd2
        best_seq = a_cd2.copy()
        print(f"    *** New best: {best_score:.8f} ***")
    a_main = a_cd2

    # Phase 5: Simulated annealing
    print("\n=== Phase 5: Simulated annealing ===")
    sa_time = min(200.0, time_budget * 0.1)
    a_sa, s_sa = simulated_annealing(a_main, sa_time, rng)
    print(f"  After SA: score={s_sa:.8f}")
    if s_sa < best_score:
        best_score = s_sa
        best_seq = a_sa.copy()
        print(f"    *** New best: {best_score:.8f} ***")
    a_main = a_sa

    # Phase 6: Final coordinate descent polish
    print("\n=== Phase 6: Final coordinate descent ===")
    remaining = max(30.0, deadline - time.monotonic() - 10)
    cd_time3 = min(300.0, remaining)
    a_final, s_final = coordinate_descent(a_main, cd_time3, rng, use_peak_focus=True)
    print(f"  After final CD: score={s_final:.8f}")
    if s_final < best_score:
        best_score = s_final
        best_seq = a_final.copy()
        print(f"    *** New best: {best_score:.8f} ***")

    # Also try coordinate descent on other GD results if time permits
    if time.monotonic() < deadline - 30 and len(gd_results) > 1:
        print("\n=== Phase 7: Optimize other GD results ===")
        for i in range(1, min(3, len(gd_results))):
            if time.monotonic() > deadline - 20:
                break
            a_try = gd_results[i][0].copy()
            t_avail = min(60.0, deadline - time.monotonic() - 10)
            a_opt, s_opt = coordinate_descent(a_try, t_avail, rng, use_peak_focus=True)
            print(f"  GD result {i} (n={len(a_opt)}): score={s_opt:.8f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.8f} ***")

    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = compute_score(np.array(final_list))
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
