"""
Minimize 2*n*max(self-convolution)/(sum)^2 over non-negative sequences.

Strategy:

- Multiple diverse initializations: given best, random, flat-spectrum (CAZAC-like),

- Final polishing: fine-tuned gradient descent + random local search with

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_indices):
    """Weighted subgradient of max(c) using active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo <= hi:
            indices = k_int - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += w_val * 2.0 * a[indices]
    return grad


def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S} via water-filling."""
    n = len(v)
    if n == 0:
        return v.copy()
    v_clipped = np.maximum(v, 0.0)
    v_sum = v_clipped.sum()
    if v_sum <= 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    # Water-filling
    v_sorted = np.sort(v_clipped)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= v_sorted
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v_clipped - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a * (S / s)


def make_flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum (CAZAC-like)."""
    # Random phases -> flat magnitude spectrum
    phases = rng.uniform(0.0, 2.0 * np.pi, n)
    spectrum = np.exp(1j * phases)
    a_complex = ifft(spectrum) * n
    # Take absolute value of real part, add small positive offset
    a = np.abs(np.real(a_complex))
    # Add a small DC component to ensure non-negativity and avoid zeros
    a = a + 0.5 * a.max()
    return a


def make_multi_bump_init(n, rng):
    """Generate sequence as sum of shifted smooth bumps (anti-symmetric pattern)."""
    a = np.zeros(n, dtype=np.float64)
    n_bumps = rng.integers(3, 8)
    for _ in range(n_bumps):
        center = rng.integers(0, n)
        width = rng.uniform(n * 0.05, n * 0.3)
        height = rng.uniform(0.5, 2.0)
        x = np.arange(n)
        a += height * np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
    return np.maximum(a, 0.0)


def make_ramp_init(n, rng):
    """Linear ramp with noise."""
    x = np.linspace(0.0, 1.0, n)
    a = x ** rng.uniform(0.3, 3.0)
    a += rng.uniform(0.1, 0.5) * (1.0 - x)
    a += rng.normal(0, 0.1, n)
    return np.maximum(a, 0.0)


def make_legacy_init(n, rng):
    """Legendre-sequence-inspired: use quadratic residues pattern, mapped to non-negative."""
    if n % 2 == 0:
        n = n - 1  # need odd for Legendre
    # Simple pseudo-Legendre: a[i] = 1 if i is quadratic residue mod p, else 0
    # Then smooth and make non-negative
    a = np.zeros(n, dtype=np.float64)
    for i in range(n):
        if i == 0:
            a[i] = 0.0
        else:
            # Check if i is a QR mod n (approximate)
            a[i] = 1.0 if (int(np.sqrt(i)) ** 2 == i) else 0.3
    # Smooth with a small kernel
    kernel = np.ones(3) / 3.0
    a = np.convolve(a, kernel, mode='same')
    return np.maximum(a, 0.1)


def run_momentum_grad(a_init, n, S, max_time, rng, top_k_init=8, top_k_max=48):
    """Momentum gradient descent with projection, adaptive top-k, cosine LR."""
    a = safe_normalize(a_init, n)
    v = np.zeros(n, dtype=np.float64)
    momentum = 0.9
    S_target = S
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    lr_max = 2.0 * S / n
    lr_min = 0.01 * S / n
    patience = 0
    patience_limit = 5000

    while time.monotonic() - start < max_time:
        t += 1
        # Cosine annealing
        progress = min(1.0, float(t) / max(1.0, int(max(1.0, (max_time / 0.005)))))
        lr = lr_min + 0.5 * (lr_max - lr_min) * (1.0 + np.cos(np.pi * progress))

        c = fft_convolve_self(a)
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > patience_limit:
            # Perturb to escape local minimum
            noise_scale = 0.02 * S / n * (1.0 + patience / patience_limit)
            a = project_nonneg_sum(a + rng.normal(0, min(noise_scale, 0.1 * S / n), n), S_target)
            v *= 0.5
            patience = 0
            continue

        # Adaptive top-k
        top_k = min(top_k_max, max(top_k_init, int(top_k_init * (1.0 + 2.0 * progress))))
        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, c, top_indices)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm

        v = momentum * v + lr * grad
        a_new = a - v
        a = project_nonneg_sum(a_new, S_target)

        # Occasionally inject noise (annealed)
        if t % 1000 == 0 and rng.random() < 0.2 * (1.0 - progress):
            noise_scale = 0.01 * S / n * (1.0 - progress)
            a = project_nonneg_sum(a + rng.normal(0, noise_scale, n), S_target)
            v *= 0.3

    return best_local, best_score


def run_multiplicative(a_init, n, S, max_time, rng, eta_init=0.3, top_k=12):
    """Exponentiated gradient / multiplicative weights update."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    patience = 0

    while time.monotonic() - start < max_time:
        t += 1
        eta = eta_init / (1.0 + 0.002 * t)

        c = fft_convolve_self(a)
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > 3000:
            a = project_nonneg_sum(a + rng.normal(0, 0.01 * S / n, n), S)
            patience = 0
            continue

        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, c, top_indices)
        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        grad = grad / gmax

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)

    return best_local, best_score


def run_polish(a_best, n, S, max_time, rng):
    """Fine polishing: very small steps + random local search."""
    a = safe_normalize(a_best, n)
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    lr = 0.05 * S / n
    top_k = 16

    while time.monotonic() - start < max_time:
        phase1_left = max_time * 0.6 - (time.monotonic() - start)
        t += 1

        if phase1_left > 0:
            # Fine gradient step
            c = fft_convolve_self(a)
            cur = float(c.max())
            if cur < best_score:
                best_score = cur
                best_local = a.copy()

            top_indices = np.argsort(c)[::-1][:top_k]
            grad = compute_subgradient(a, c, top_indices)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                grad = grad / gnorm
                a_new = a - lr * grad
                a = project_nonneg_sum(a_new, S)
                lr *= 0.9999
        else:
            # Random local search
            if rng.random() < 0.5:
                # Small random perturbation
                delta = rng.normal(0, 0.005 * S / n, n)
                a_trial = project_nonneg_sum(a + delta, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best_score:
                    a = a_trial
                    best_score = float(c_trial.max())
                    best_local = a.copy()
            else:
                # Coordinate perturbation: randomly pick a few elements and adjust
                n_perturb = rng.integers(1, 6)
                indices = rng.integers(0, n, n_perturb)
                a_trial = a.copy()
                adjustments = rng.normal(0, 0.01 * S / n, n_perturb)
                a_trial[indices] += adjustments
                a_trial = project_nonneg_sum(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best_score:
                    a = a_trial
                    best_score = float(c_trial.max())
                    best_local = a.copy()
