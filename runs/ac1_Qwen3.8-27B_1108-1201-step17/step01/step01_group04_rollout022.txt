def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def grad_peak(a):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = a[m_star - j]
    return grad

def grad_softmax(a, tau=50.0):
    """Compute gradient of log-sum-exp of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        k_max = min(n, 2 * n - 1 - j)
        if k_max > 0:
            grad[j] = tau * np.dot(a[:k_max], w[j:j + k_max])
    return grad

def adam_step(a, grad, m, v, beta1, beta2, t, lr, eps=1e-8, target_sum=None):
    """Adam optimization step with projection onto non-negative and fixed-sum."""
    n = len(a)
    if target_sum is None:
        target_sum = a.sum()
    
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, target_sum)
    return a_new, m, v

    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 2048:
        return None

    A = np.zeros((2 * n - 1, n))

        hi = min(n - 1, m)

    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def line_search_blend(a, g, target_sum, n_evals=20):
    """Find best blend of a and g by 1D search."""
    best_alpha = 0.0
    best_score = compute_obj(a)
    # Golden ratio search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    
    def f_at(alpha):
        x = (1 - alpha) * a + alpha * g
        x = np.maximum(x, 0.0)
        x = normalize(x, target_sum)
        return compute_obj(x)
    
    f1 = f_at(a1)
    f2 = f_at(a2)
    
    for _ in range(n_evals):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = f_at(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = f_at(a2)
    
    best_alpha = (lo + hi) / 2
    x = (1 - best_alpha) * a + best_alpha * g
    x = np.maximum(x, 0.0)
    x = normalize(x, target_sum)
    return x, compute_obj(x)

Uses Adam optimizer with top-K subgradient, intensive LP-based refinement
with golden-section 1D line search, and simulated annealing restarts.

import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def grad_peak(a):
    """Subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = a[m_star - j]
    return grad


def grad_topk(a, K=3):
    """Subgradient using top-K peaks for smoother direction."""
    n = len(a)
    c = np.convolve(a, a)

    for m_star in top_k:
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):

        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def line_search_blend(a, g, target_sum, n_evals=25):
    """Find best blend of a and g via golden section search."""
    best_alpha = 0.0
    base_score = compute_obj(a)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    
    def f_at(alpha):
        x = (1.0 - alpha) * a + alpha * g
        x = np.maximum(x, 0.0)
        x = normalize(x, target_sum)
        return compute_obj(x)
    
    f1 = f_at(a1)
    f2 = f_at(a2)
    
    for _ in range(n_evals):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = f_at(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = f_at(a2)
    
    best_alpha = (lo + hi) / 2.0
    x = (1.0 - best_alpha) * a + best_alpha * g
    x = np.maximum(x, 0.0)
    x = normalize(x, target_sum)
    s = compute_obj(x)
    if s >= base_score:
        return a.copy(), base_score
    return x, s


def optimize_sequence(a0, deadline, max_adam_iters=2000, max_lp_iters=150, use_annealing=True):
    """Optimize a single sequence using Adam + LP refinement."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a, target_sum)
    
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    if score < 1.52:
        print(f"    Init score: {score:.6f}")
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 0
    
    # Annealing temperature
    temp = 0.05
    
    for it in range(max_adam_iters):
        if time.monotonic() > deadline - 40.0:
            break
        
        t += 1
        # Learning rate schedule
        if t <= 30:
            lr = 0.005 * (t / 30.0)
        else:
            lr = 0.005 * 30.0 / np.sqrt(t)
        
        # Gradient: alternate between softmax and topk
        if it % 60 < 30:
            tau = 40.0 + 60.0 * min(1.0, it / 500.0)
            grad = grad_softmax(a, tau)
        else:
            grad = grad_topk(a, K=3)
        
        # Project gradient (fixed sum)
        grad -= grad.mean()
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, target_sum)
        
        s_new = compute_obj(a_new)
        
        # Accept if better, or with probability if annealing
        accept = False
        if s_new < best_score:
            accept = True
        elif use_annealing and temp > 0.001:
            delta = s_new - best_score
            if np.random.random() < np.exp(-delta / temp):
                accept = True
        
        if accept:
            a = a_new
            if s_new < best_score:
                best_score = s_new
                best = a.copy()
        
        # Decay temperature
        temp *= 0.995
        
        # Check every 20 iterations
        if (it + 1) % 20 == 0:
            s_check = compute_obj(a)
            if s_check < best_score:
                best_score = s_check
                best = a.copy()
        
        # Periodic LP refinement
        if (it + 1) % 200 == 0 and n <= 1024:
            c = np.convolve(a, a)
            M = float(c.max())
            g = lp_solve(a, M)
            if g is not None:
                a_try, s_try = line_search_blend(a, g, target_sum)
                if s_try < best_score:
                    a = a_try
                    best = a.copy()
                    best_score = s_try
                    print(f"    LP@{it}: {best_score:.6f}")
    
    # Phase 2: Intensive LP refinement
    if n <= 1024 and time.monotonic() < deadline - 15.0:
        c = np.convolve(a, a)
        M = float(c.max())
        
        stall_count = 0
        for it in range(max_lp_iters):
            if time.monotonic() > deadline - 5.0:
                break
            
            g = lp_solve(a, M)
            if g is None:
                break
            
            a_try, s_try = line_search_blend(a, g, target_sum)
            if s_try < best_score:
                a = a_try
                best = a.copy()
                best_score = s_try
                stall_count = 0
                if it % 5 == 0:
                    print(f"    LP2@{it}: {best_score:.6f}")
            else:
                stall_count += 1
                
                # Try to escape: gradient step
                grad = grad_topk(a, K=2)
                grad -= grad.mean()
                a_pert = a - 0.01 * grad
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = normalize(a_pert, target_sum)
                s_pert = compute_obj(a_pert)
                if s_pert < best_score:
                    a = a_pert
                    best = a.copy()
                    best_score = s_pert
                    stall_count = 0
                elif stall_count >= 5:
                    # Random perturbation
                    a_pert = a + 0.01 * np.random.randn(n)
                    a_pert = np.maximum(a_pert, 0.0)
                    a_pert = normalize(a_pert, target_sum)
                    s_pert = compute_obj(a_pert)
                    if s_pert < best_score:
                        a = a_pert
                        best = a.copy()
                        best_score = s_pert
                        stall_count = 0
                    else:
                        break
            
            c = np.convolve(a, a)
            M = float(c.max())
    
    return best, best_score


def optimize_lp_only(a0, deadline, max_iters=300):
    """Pure LP-based optimization (no gradient)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a, target_sum)
    
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        g = lp_solve(a, M)
        if g is None:
            break
        
        a_try, s_try = line_search_blend(a, g, target_sum)
        if s_try < best_score:
            a = a_try
            best = a.copy()
            best_score = s_try
            stall = 0
            if it % 5 == 0:
                print(f"    LPonly@{it}: {best_score:.6f}")
        else:
            stall += 1
            # Perturb to escape
            if stall % 3 == 0:
                a_pert = a + 0.008 * np.random.randn(n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = normalize(a_pert, target_sum)
                a = a_pert
            if stall > 15:
                break
    
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses multiple strategies:
    1. Adam + LP from the previous best (primary)
    2. Pure LP refinement
    3. Alternative starting points
    Returns best sequence found within time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_optimize(a0, method, time_limit_frac=1.0):
        nonlocal best_seq, best_score
        if time.monotonic() > deadline - 30.0:
            return
        dl = t_start + time_limit_frac * (deadline - t_start)
        if method == 'adam_lp':
            seq, score = optimize_sequence(a0, dl)
        elif method == 'lp_only':
            seq, score = optimize_lp_only(a0, dl)
        else:
            return
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  ** New best: {best_score:.6f} (n={len(seq)}, method={method}) **")
    
    # Gather starting points
    starts = []
    
    # Primary: use the global best
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            starts.append(('adam_lp', gs.copy()))
            starts.append(('lp_only', gs.copy()))
            # Truncated versions for diversity
            if len(gs) > 256:
                starts.append(('adam_lp', gs[:256].copy()))
            if len(gs) > 128:
                starts.append(('adam_lp', gs[:128].copy()))
    except NameError:
        pass
    
    # Structured starts (different from previous)
    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        # Double chirp
        starts.append(('adam_lp', 0.3 + 0.7 * np.abs(np.cos(np.pi * i * 2.0 / n))))
        # Asymmetric raised cosine
        starts.append(('adam_lp', 0.2 + 0.8 * np.sin(np.pi * i / n) ** 1.5))
        # Triangle wave
        starts.append(('lp_only', 0.5 - np.abs(2.0 * i / n - 1.0)))
    
    # Random starts with structure
    for n in [384, 512]:
        base = 0.3 + 0.4 * np.random.randn(n)
        base = np.maximum(base, 0.05)
        # Add smooth component
        i = np.arange(n, dtype=np.float64)
        base += 0.3 * np.exp(-0.5 * ((i - n * 0.5) / (n * 0.3)) ** 2)
        starts.append(('adam_lp', base))
    
    # Sort: primary starts first
    starts.sort(key=lambda x: (0 if x[1].shape == (512,) and np.std(x[1]) > 0.3 else 1, -len(x[1])))
    
    print(f"=== Starting optimization with {len(starts)} initial sequences ===")
    print(f"    Budget: {budget_s}s, deadline in {deadline - t_start:.0f}s")
    
    # Track time spent per start
    total_time = deadline - t_start
    n_starts = len(starts)
    # Allocate more time to first (primary) start
    time_allocs = []
    remaining = 1.0
    for i in range(n_starts):
        if i == 0:
            frac = 0.35
        elif i == 1:
            frac = 0.25
        elif i < 4:
            frac = 0.10
        else:
            frac = remaining / max(1, n_starts - i) * 0.5
        time_allocs.append(frac)
        remaining -= frac
    
    if remaining > 0.05:
        time_allocs[0] += remaining
    
    for idx, (method, a0) in enumerate(starts):
        if time.monotonic() > deadline - 30.0:
            break
        t0 = time.monotonic()
        n_a = len(a0)
        print(f"  [{idx+1}/{n_starts}] n={n_a}, method={method}...")
        
        # Use allocated time fraction
        frac = time_allocs[idx] if idx < len(time_allocs) else 0.05
        dl = t_start + frac * total_time
        dl = min(dl, deadline - 20.0)
        
        if method == 'adam_lp':
            seq, score = optimize_sequence(a0, dl)
        else:
            seq, score = optimize_lp_only(a0, dl)
        
        elapsed = time.monotonic() - t0
        print(f"       score={score:.6f}, elapsed={elapsed:.1f}s")
        
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"       ** New global best: {best_score:.6f} **")
    
    # Final intensive refinement on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 20.0:
        remaining_time = deadline - 20.0 - time.monotonic()
        print(f"=== Final refinement ({remaining_time:.0f}s remaining) ===")
        print(f"    Current best: {best_score:.6f}")
        
        # Multiple passes with decreasing time
        for pass_i in range(4):
            if time.monotonic() > deadline - 15.0:
                break
            pass_dl = deadline - 10.0 - pass_i * 2.0
            cur_time = deadline - time.monotonic()
            pass_alloc = min(60.0 + 30.0 * (3 - pass_i), cur_time * 0.4)
            pass_dl = time.monotonic() + pass_alloc
            
            if pass_i % 2 == 0:
                seq, score = optimize_sequence(best_seq, pass_dl,
                                               max_adam_iters=500, max_lp_iters=100)
            else:
                seq, score = optimize_lp_only(best_seq, pass_dl, max_iters=100)
            
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    Pass {pass_i}: {best_score:.6f}")
            else:
                # Break early if no improvement
                if pass_i > 0:
                    break
    
    # Ultimate final LP push
    if best_seq is not None and time.monotonic() < deadline - 8.0:
        n = len(best_seq)
        if n <= 1024:
            a = best_seq.copy()
            target_sum = np.sqrt(2.0 * n)
            a = normalize(a, target_sum)
            # A few final LP steps
            for _ in range(20):
                if time.monotonic() > deadline - 3.0:
                    break
                c = np.convolve(a, a)
                M = float(c.max())
                g = lp_solve(a, M)
                if g is not None:
                    a_try, s_try = line_search_blend(a, g, target_sum)
                    if s_try < best_score:
                        a = a_try
                        best_seq = a.copy()
                        best_score = s_try
    
    # Final output
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"=== FINAL: n={len(best_seq)}, computed_score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
