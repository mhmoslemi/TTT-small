    Solve LP: maximize sum(g) subject to g >= 0, conv(f, g) <= M.
    Uses scipy.optimize.linprog.

    n = len(f)
    
    # Objective: minimize -sum(g) => c = -ones(n)
    c_obj = -np.ones(n)
    
    # Constraints: conv(f, g)[k] <= M for k = 0, ..., 2n-2

Multi-strategy optimizer for minimizing peak autocorrelation of non-negative sequences.
Combines annealed soft-max gradient descent, LP-based cutting plane updates,
and simulated annealing local search. Explores multiple sequence lengths
and initialization strategies to find the global minimum.

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score: 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def soft_max_value_and_weights(x, tau):
    """Compute soft-max value and corresponding weights."""
    xm = np.max(x)
    s = (x - xm) / tau
    e = np.exp(s)
    se = np.sum(e)
    w = e / se
    return xm + tau * np.log(se), w


def grad_softmax_conv(a, tau):
    """
    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    c = fft_convolve(a, a)
    val, w = soft_max_value_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return val, g, c

def lp_update(f, M):
    """
    Solve LP: maximize sum(g) subject to g >= 0, conv(f,g) <= M.
    Returns g or None if infeasible.
    """
    from scipy.optimize import linprog
    n = len(f)

    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)

    # Build constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n
    # conv(f,g)[k] = sum_j A[k,j]*g[j] <= M
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = f[0:k_end - k_start]

    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    if res.success:
        return np.clip(res.x, 0, None)
    return None

def make_chirp(n, alpha, rng):

    return np.clip(a, 0.01, None)


def make_random(n, rng, spread=0.3):

    deadline = t_global + max(1.0, budget_s - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(provided) > 0:
            s = local_score(provided)
            print(f"Provided: n={len(provided)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = provided.copy()

    # Phase 1: Gradient optimization at multiple lengths
    print("\n=== Phase 1: Gradient optimization ===")
    lengths = [256, 384, 512, 768, 1024]

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break

        remaining = deadline - time.monotonic()
        n_configs = lengths.index(n) + 2
        time_per_init = max(5.0, remaining / n_configs - 5)

        # Build initializations
        inits = []
        if provided is not None:
            x_old = np.linspace(0, 1, len(provided))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, provided))
        inits.append(make_chirp(n, 0.2 + 0.4 * rng.random(), rng))
        inits.append(make_random(n, rng, spread=0.15 + 0.4 * rng.random()))
        if n <= 512:
            inits.append(make_multitone(n, rng))

        for a0 in inits:
            if time.monotonic() > deadline - 30:
                break
            a0 = np.clip(a0, 0.01, 1000.0)
            if np.sum(a0) < 0.01:
                a0 = np.ones(n)

            t0 = time.monotonic()
            time_alloc = min(time_per_init, deadline - time.monotonic() - 30)
            if time_alloc < 3:
                continue

            a_opt, score = gradient_optimize(n, a0, time_alloc, rng)
            elapsed = time.monotonic() - t0
            print(f"  n={n}: grad score={score:.6f}, t={elapsed:.1f}s")

            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")

            # Try LP update on this result
            if time.monotonic() < deadline - 40 and score < best_score + 0.1:
                lp_time = min(20.0, deadline - time.monotonic() - 20)
                if lp_time > 3:
                    a_lp, lp_score = lp_iterate(a_opt, lp_time, rng)
                    if lp_score < best_score:
                        best_score = lp_score
                        best_seq = a_lp.copy()
                        print(f"    *** LP improved to {best_score:.6f} ***")

    # Phase 2: Intensive LP iteration on best
    print(f"\n=== Phase 2: LP iteration (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp_time = min(150.0, deadline - time.monotonic() - 30)
        a_lp, lp_score = lp_iterate(best_seq, lp_time, rng)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = a_lp.copy()
            print(f"  After LP: {best_score:.6f}")

    # Phase 3: More gradient optimization from LP result
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        grad_time = min(90.0, deadline - time.monotonic() - 30)
        if grad_time > 5:
            a_grad, grad_score = gradient_optimize(n, best_seq, grad_time, rng,
                                                   tau_start_ratio=20.0, tau_end_ratio=0.2,
                                                   lr_scale=0.8)
            if grad_score < best_score:
                best_score = grad_score
                best_seq = a_grad.copy()
                print(f"  After re-gradient: {best_score:.6f}")

    # Phase 4: Local search polish
    print(f"\n=== Phase 4: Local search polish (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        polish_time = min(120.0, deadline - time.monotonic() - 5)
        a_pol, pol_score = local_search_polish(best_seq, polish_time, rng)
        if pol_score < best_score:
            best_score = pol_score
            best_seq = a_pol.copy()
            print(f"  After polish: {best_score:.6f}")

    # Phase 5: One more LP + gradient cycle if time permits
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            n = len(best_seq)
            lp_time = remaining * 0.4
            a_lp, lp_score = lp_iterate(best_seq, lp_time, rng)
            if lp_score < best_score:
                best_score = lp_score
                best_seq = a_lp.copy()

            grad_time = remaining * 0.3
            if grad_time > 5:
                a_g, g_score = gradient_optimize(n, best_seq, grad_time, rng,
                                                 tau_start_ratio=10.0, tau_end_ratio=0.1,
                                                 lr_scale=0.5)
                if g_score < best_score:
                    best_score = g_score
                    best_seq = a_g.copy()

            pol_time = remaining * 0.2
            if pol_time > 5:
                a_p, p_score = local_search_polish(best_seq, pol_time, rng)
                if p_score < best_score:
                    best_score = p_score
                    best_seq = a_p.copy()

        print(f"  After final cycle: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
