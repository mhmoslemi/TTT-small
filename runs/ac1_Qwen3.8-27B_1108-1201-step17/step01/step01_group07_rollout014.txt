    """Compute smooth max of auto-convolution of a, and gradient w.r.t. a."""
    n = len(a)
    C = fast_convolve(a)
    C_scaled = C / kappa
    C_max = C_scaled.max()
    exp_C = np.exp(C_scaled - C_max)
    sum_exp = exp_C.sum()
    p = exp_C / sum_exp
    obj = 2.0 * n * kappa * (C_max + np.log(sum_exp))
    
    # Gradient: grad_a[j] = sum_k p[k] * (a[k-j] + a[j])
    # term1[j] = sum_m a[m] * p[m+j]
    a_rev = a[::-1]
    size = 2 * n - 1  # length of p
    conv_len = n + size - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_len)
    P = np.fft.rfft(p, fft_len)
    conv = np.fft.irfft(A_rev * P, fft_len)[:conv_len]
    term1 = conv[n-1:n-1+n]
    
    grad_a = term1 + a
    return obj, grad_a, p

def softmax_obj_grad(x, kappa):
    """Objective and gradient in log-parameterization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    a = a / a_sum
    
    obj, grad_a, p = compute_smooth_max_grad(a, kappa)
    
    # Chain rule: df/dx[i] = sum_j (df/da[j]) * (da[j]/dx[i])
    # da[j]/dx[i] = a[j]/a_sum * (delta[j,i] - a[i]/a_sum) ... 
    # Actually with a[j] = exp(x_j) / sum(exp(x)):
    # da[j]/dx[i] = a[j] * (delta[j,i] - a[i])
    
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x
