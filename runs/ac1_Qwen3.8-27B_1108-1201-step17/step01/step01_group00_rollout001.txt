        a_rev = a[::-1]

        grad = 2.0 * cross[n-1:2*n-1]  # elements n-1 to 2n-2, which give cross[j] for j=0 to n-1

            best_a = a_new.copy()

        tau *= 0.999
        # Decay learning rate
        if np.random.random() < 0.01:
            lr *= 0.95
    
    return best_a, best_M
