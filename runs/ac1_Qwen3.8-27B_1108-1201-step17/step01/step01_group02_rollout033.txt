```python
"""
Soft-max gradient descent + LP refinement to minimize peak auto-convolution.
Uses analytical gradients of the soft-max approximation of the peak,
combined with LP-based refinement and local search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_symmetric_cosine(n, num_periods=1):
    """Raised cosine that's also symmetric (palindrome)."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Symmetric raised cosine centered at the middle
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - center) / (n - 1)))
    return np.maximum(0, a)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    # Enforce symmetry
    a = 0.5 * (a + a[::-1])
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def softmax_gradient_descent(a, time_limit_s=300.0, verbose=True):
    """
    Minimize 2n*max(conv(a,a)) using soft-max gradient descent.
    a is normalized to sum=1.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    SGD start: score={best_score:.6f}, n={n}")
    
    # Temperature schedule: start moderate, increase over time
    T = n  # initial temperature
    lr = 0.5 / n  # initial learning rate
    
    stagnant = 0
    it = 0
    max_iters = 100000
    
    while time.monotonic() - start_time < time_limit_s and it < max_iters:
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max()  # since sum(a)=1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 500 == 0 or it < 10):
                print(f"    SGD iter {it:5d}: score={score:.6f}, T={T:.1f}, lr={lr:.2e}")
        else:
            stagnant += 1
        
        if stagnant > 2000:
            break
        
        # Compute softmax weights
        # For numerical stability, subtract max
        conv_shifted = conv - conv.max()
        exp_conv = np.exp(T * conv_shifted)
        p = exp_conv / exp_conv.sum()
        
        # Compute gradient: grad[j] = 4n * sum_k p[k] * a[k-j]
        # = 4n * correlate(p, a, 'full')[j] for j=0,...,n-1
        r = np.correlate(p, a, 'full')[:n]
        grad = 4.0 * n * r
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project onto probability simplex (approximate)
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a / n  # fallback to uniform
            s = 1.0
        a_new = a_new / s
        
        # Check if the step helped
        new_score = 2.0 * n * np.convolve(a_new, a_new).max()
        if new_score > score * 1.01:
            # Step was too large, reduce lr
            lr *= 0.5
            if lr < 1e-10:
                break
        elif new_score < score:
            # Good step
            a = a_new
            # Occasionally increase T to sharpen
            if it % 100 == 0:
                T = min(T * 1.05, 50 * n)
        else:
            a = a_new
            lr *= 0.8
        
        it += 1
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    SGD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=120.0, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    it = 0
    max_iters = 200
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose:
                print(f"    LP iter {it:3d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.3))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = 0.3

def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Try different perturbation strategies
        strategy = rng.random()
        
        if strategy < 0.4:
            # Multiply a segment by a random factor
            width = max(2, int(rng.integers(n // 100, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate = a.copy()
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.1, size=stop - start))
        elif strategy < 0.7:
            # Add a small Gaussian bump
            candidate = a.copy()
            center = float(rng.integers(0, n))
            width = float(rng.integers(max(1, n // 50), max(2, n // 10)))

            t = np.arange(n, dtype=np.float64)
            candidate += amplitude * np.exp(-0.5 * ((t - center) / width) ** 2)

            candidate = a.copy()
            kernel_size = int(rng.integers(3, min(11, n // 2)))
            kernel = np.ones(kernel_size) / kernel_size
            candidate = np.convolve(candidate, kernel, mode='same')
        
        # Enforce symmetry with some probability
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
```
