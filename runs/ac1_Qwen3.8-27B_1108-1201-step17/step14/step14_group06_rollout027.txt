def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution) of a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_seq(a, target_sum=None):
    """Normalize sequence to non-negative with given target sum (default sqrt(2n))."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a * target_sum / s


def lse_obj_grad(a, n, temp):
    """Compute LSE-smoothed objective and gradient.
    Objective: 2n * LSE(conv(a,a)) / (sum(a))^2
    where LSE(x) = (1/temp) * log(sum(exp(temp * (x - max(x))))) + max(x)
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    # LSE of b
    shifted = temp * (b - max_b)
    exp_vals = np.exp(shifted)
    lse_val = max_b + np.log(exp_vals.sum()) / temp
    w = exp_vals / exp_vals.sum()  # softmax weights
    
    # Gradient of LSE w.r.t. a: w convolved with a (truncated to n elements)
    # d(lse)/da[i] = sum_k w[k] * a[k-i] (for valid k-i)
    # This is the cross-correlation of w and a, truncated to length n
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    
    # Objective: 2n * lse / s^2
    obj = 2.0 * n * lse_val / (s * s)
    
    # Gradient: 2n * cross / s^2 - 2 * obj * a / s
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj * a / s
    
    return obj, grad

def build_T_matrix(a, n):
    """Build the constraint matrix T where (T @ b)[k] = sum_j T[k,j]*b[j] = conv(a,b)[k]."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

    if T is None:
        T = build_T_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=False, m_reduction=0.02):
    """Nesterov-accelerated LP best-response with adaptive M reduction."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    last_improve = t0
    stall = 0
    n_iters = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        
        # Adaptive M reduction
        eps = min(0.05, 0.005 * np.sqrt(t + 1))
        M_used = M_curr * (1.0 - eps)
        
        T = build_T_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None or b.sum() < 1e-15:
            b = lp_best_response(y, M_curr * 0.99, solver, T)
            if b is None or b.sum() < 1e-15:
                if stall > 10:
                    break
                stall += 1
                continue
        
        g = normalize_seq(b, norm_val)
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_seq(y_new, norm_val)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new, norm_val)
        
        y = y_new
        a = a_new
        n_avg += 1
        avg_a += a
        
        # Check improvement
        for cand in [y, a]:
            sc = fast_autocorr(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                last_improve = time.monotonic()
        
        if n_avg % 100 == 0:
            avg_check = normalize_seq(avg_a / n_avg, norm_val)
            sc = fast_autocorr(avg_check).max()
            if sc < best_score:
                best_score = sc
                best_a = avg_check.copy()
                last_improve = time.monotonic()
        
        if time.monotonic() - last_improve > 90.0:
            stall += 1
            if stall > 8:
                break
        else:
            stall = 0
        
        if verbose and n_iters % 200 == 0:
            print(f"    LP-Nest iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def peak_deflation(a0, time_limit, seed=42):
    """Peak deflation: iteratively reduce the largest autocorrelation peaks."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        M = b.max()
        
        # Get top peaks
        top_peaks = np.argsort(b)[::-1][:20]
        
        # Compute blame vector: for each element i, how much does it contribute to peaks?
        blame = np.zeros(n)
        for k in top_peaks:
            weight = max(0, b[k] - M * 0.95)  # Only blame elements for significant excess
            if weight < 1e-15:
                continue
            # Contribution of a[i] to b[k] is a[k-i] (if valid)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    blame[i] += a[j] * weight
        
        if blame.max() < 1e-15:
            stall += 1
            if stall > 50:
                break
            # Random perturbation
            a = a + rng.normal(0, 0.01 * a.mean(), n)
            a = normalize_seq(a, norm_val)
            continue
        
        # Reduce elements with high blame, increase elements with low blame
        lr = 0.1
        for _ in range(10):
            a_trial = a - lr * blame / (blame.max() + 1e-15)
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                lr *= 0.5
                continue
            a_trial = a_trial * norm_val / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
                break
            lr *= 0.5
        
        else:
            stall += 1
            if stall > 100:
                # Try random coordinate moves
                improved = False
                for _ in range(200):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = normalize_seq(a_trial, norm_val)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if not improved:
                    stall += 10
                    if stall > 200:
                        break
    
    return best_a, best_score


def coord_descent_refine(a0, time_limit, seed=42):
    """Targeted coordinate descent focusing on peak contributors."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_src = sorted_idx[:20]
        
        # Find low-value targets (elements far from peak)
        peak_mask = np.zeros(n, dtype=bool)
        top_peaks = np.argsort(b)[::-1][:10]
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        low_targets = non_peak[np.argsort(a[non_peak])[:20]]
        
        improved = False
        for src in top_src:
            if improved:
                break
            if a[src] < 1e-10:
                continue
            for dst in low_targets:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = normalize_seq(a_trial, norm_val)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
        
        if not improved:
            # Random moves
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize_seq(a_trial, norm_val)
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    s2 = f_interp.sum()
    if s2 < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s2


def structured_init(n, rng, init_type=None):
    """Generate diverse structured initializations."""
    if init_type is None:
        init_type = rng.integers(0, 8)
    
    if init_type == 0:
        # Cosine bell
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif init_type == 1:
        # Sine power
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif init_type == 2:
        # Multi-frequency
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif init_type == 3:
        # Double Gaussian
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif init_type == 4:
        # Quadratic chirp magnitude
        x = np.linspace(0.0, 1.0, n)
        phase = 2.0 * np.pi * (x ** 2)
        a = np.abs(np.cos(phase)) ** 0.5
    elif init_type == 5:
        # Asymmetric (biased left)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * x ** 2)
    elif init_type == 6:
        # Asymmetric (biased right)  
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (1.0 - x) ** 2)
    else:
        # Uniform with noise
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_state, candidate_a, label, t_global):
    """Try to update best state with candidate."""

    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def normalize_seq(a, target_sum=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a * target_sum / s


def lse_obj_grad(a, n, temp):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = temp * (b - max_b)
    exp_vals = np.exp(shifted)
    lse_val = max_b + np.log(exp_vals.sum()) / temp
    w = exp_vals / exp_vals.sum()
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    obj = 2.0 * n * lse_val / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj * a / s
    return obj, grad


def lse_optimize_bol(a0, time_limit, temp_start=50.0, temp_end=1.0, n_temps=4):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(temp_start, temp_end, n_temps)
    for ti, temp in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        a_curr = a.copy()
        stall = 0
        for it in range(10000):
            if time.monotonic() - t0 > time_limit - 5:
                break
            obj, grad = lse_obj_grad(a_curr, n, temp)
            if not np.isfinite(obj):
                break
            lr = 1.0
            accepted = False
            for _ in range(40):
                a_trial = a_curr - lr * grad
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    lr *= 0.5
                    continue
                a_trial = a_trial * norm_val / s_t
                score_trial = fast_autocorr(a_trial).max()
                if score_trial < best_score:
                    best_score = score_trial
                    best_a = a_trial.copy()
                if score_trial < best_score + 0.001:
                    a_curr = a_trial
                    accepted = True
                    stall = 0
                    break
                lr *= 0.5
            if not accepted:
                stall += 1
                if stall > 30:
                    break
            if it % 200 == 0:
                score_curr = fast_autocorr(a_curr).max()
                if score_curr < best_score:
                    best_score = score_curr
                    best_a = a_curr.copy()
        a = best_a.copy()
    return best_a, best_score


def build_T_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    if not HAS_CVXPY or solver is None:
        return None
    n = len(a)
    if T is None:
        T = build_T_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=False):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    last_improve = t0
    stall = 0
    n_iters = 0
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit - 5:
            break
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        # Adaptive M reduction: larger early, smaller later
        eps = min(0.06, 0.008 * np.sqrt(t + 1))
        M_used = M_curr * (1.0 - eps)
        T = build_T_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None or b.sum() < 1e-15:
            b = lp_best_response(y, M_curr * 0.995, solver, T)
            if b is None or b.sum() < 1e-15:
                stall += 1
                if stall > 15:
                    break
                continue
        g = normalize_seq(b, norm_val)
        n_iters += 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_seq(y_new, norm_val)
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new, norm_val)
        y = y_new
        a = a_new
        n_avg += 1
        avg_a += a
        for cand in [y, a]:
            sc = fast_autocorr(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                last_improve = time.monotonic()
                stall = 0
        if n_avg % 50 == 0:
            avg_check = normalize_seq(avg_a / n_avg, norm_val)
            sc = fast_autocorr(avg_check).max()
            if sc < best_score:
                best_score = sc
                best_a = avg_check.copy()
                last_improve = time.monotonic()
                stall = 0
        if time.monotonic() - last_improve > 120.0:
            stall += 1
            if stall > 10:
                break
        if verbose and n_iters % 200 == 0:
            print(f"    LP-Nest iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def peak_deflation_coord(a0, time_limit, seed=42):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = normalize_seq(a0, norm_val)

    t0 = time.monotonic()
    stall = 0
    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:25]
        # Compute blame vector
        blame = np.zeros(n)
        M = b.max()
        for k in top_peaks:
            weight = max(0.0, b[k] - M * 0.90)
            if weight < 1e-15:
                continue
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    blame[i] += a[j] * weight
        if blame.max() > 1e-10:
            lr = 0.1
            accepted = False
            for _ in range(10):
                a_trial = a - lr * blame / (blame.max() + 1e-15)
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    lr *= 0.5
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    accepted = True
                    stall = 0
                    break
                lr *= 0.5
            if accepted:
                continue
        # Spectral-guided coordinate moves
        # Identify elements contributing most to top peaks
        improved = False
        for k_peak in top_peaks[:8]:
            if improved:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            sorted_idx = idx[np.argsort(contrib)[::-1]]
            top_src = sorted_idx[:12]
            # Find low-peak targets
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.arange(n)
            low_targets = non_peak[np.argsort(a[non_peak])[:12]]
            for src in top_src:
                if improved:
                    break
                if a[src] < 1e-10:
                    continue
                for dst in low_targets:
                    if dst == src:
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.08, 0.15]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = normalize_seq(a_trial, norm_val)
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
        if not improved:
            # Random moves
            for _ in range(150):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.12)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize_seq(a_trial, norm_val)
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        if not improved:
            stall += 1
            if stall > 200:
                break
    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    s2 = f_interp.sum()
    if s2 < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s2


def structured_init(n, rng, init_type=None):
    if init_type is None:
        init_type = rng.integers(0, 8)
    if init_type == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif init_type == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif init_type == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif init_type == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif init_type == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.abs(np.cos(2.0 * np.pi * x ** 2)) ** 0.5
    elif init_type == 5:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * x ** 2)
    elif init_type == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (1.0 - x) ** 2)
    else:
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"restart{ri}", t_global)

    # Phase 7: Final polish (remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_cur} ({remaining:.0f}s) ===")
    if remaining > 15:
        half = remaining / 2.0
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = lse_optimize_bol(a_norm, half, temp_start=8.0, temp_end=0.5, n_temps=3)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "lse_polish", t_global)
        rem2 = time_budget - (time.monotonic() - t_global)
        if rem2 > 10:
            n_cur2 = len(best_seq)
            a_norm2 = normalize_seq(best_seq, np.sqrt(2.0 * n_cur2))
            a_out2, _ = peak_deflation_coord(a_norm2, rem2, seed=999)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out2, "coord_polish", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    return [float(x) for x in best_seq]
