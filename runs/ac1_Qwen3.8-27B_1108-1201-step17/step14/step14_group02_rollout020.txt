Now, let me think about the **coordinate descent** polishing step:

The key idea: for each pair (i, j) where we want to move mass from i to j, find the optimal amount to move.

If we move delta from position i to position j:
- conv[k] changes by: delta * (a[j][k-i] adjusted - a[i][k-j] adjusted)
  More precisely: conv_new[k] = conv[k] + delta * (a_shifted_j[k] - a_shifted_i[k])
  where a_shifted_i[k] = a[k-i] if 0 <= k-i < n else 0

The change in the max is complex, but we can approximate: if the current max is at lag k*, then the change is approximately:
delta * (a[k*-j] - a[k*-i]) (for valid indices)

To decrease the max, we want a[k*-j] < a[k*-i], i.e., move mass FROM positions that have high correlation at the max lag TO positions that have low correlation.

But we should also consider the second-order effects and the other near-max lags.

A practical approach:
1. Compute conv(a,a)
2. Identify the top K lags (K=5-10)
3. For each coordinate i, compute the "contribution score": s[i] = sum_{k in top_K} w_k * a[k-i]
   where w_k is the weight (based on how close to max)
4. Move mass from high-score to low-score coordinates
5. Use line search to find the optimal amount

def coordinate_polish(a0, time_limit, top_k=10, verbose=True):
    """Coordinate descent polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        c_max = c.max()
        
        # Top K lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        # Weights: how active is each constraint?
        w = (top_vals - c_max + c_max) / (top_k * c_max + 1e-30)  # normalized
        # Actually, just use the values directly as weights
        w = top_vals / top_vals.sum()
        
        # Compute contribution score for each coordinate
        # s[i] = sum_{k in top_idx} w * a[k-i]
        s = np.zeros(n)
        for ki, wi in zip(top_idx, w):
            ki = int(ki)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    s[i] += wi * a[j]
        
        # Move mass from high-s to low-s
        # Find the coordinate with highest s and lowest s
        src = np.argmax(s)
        dst = np.argmin(s)
        
        if a[src] < 1e-15 or s[src] <= s[dst]:
            # Random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                stall += 1
                continue
        
        # Line search: find optimal delta
        # Binary search on delta in [0, a[src]]
        lo, hi = 0.0, a[src]
        best_delta = 0.0
        best_obj = c_max
        
        for _ in range(20):
            mid = (lo + hi) / 2
            a_trial = a.copy()
            a_trial[src] -= mid
            a_trial[dst] += mid
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_obj:
                best_obj = m_trial
                best_delta = mid
                lo = mid
            else:
                hi = mid
        
        if best_delta > 1e-10 and best_obj < best_m - 1e-14:
            a[src] -= best_delta
            a[dst] += best_delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_m = best_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Nesterov-accelerated mirror descent with adaptive smooth max approximation
2. LP-based Frank-Wolfe with Nesterov acceleration
3. Coordinate descent polishing with exact line search
4. Multi-length scaling
The key innovation is the use of FFT-based smooth subgradients for very fast
iterations (O(n log n) per step), combined with the LP oracle for global
direction finding.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_max_grad(a, beta):
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(beta * c_shifted)
    w_sum = w.sum()
    w = w / w_sum
    # grad[i] = (1/beta) * sum_j a[j] * w[i+j]
    U_rev = a[::-1]
    V = w
    N = 1 << (3 * n - 2).bit_length()
    FFT_U = rfft(U_rev, N)
    FFT_V = rfft(V, N)
    conv_result = irfft(FFT_U * FFT_V, N)
    grad = conv_result[n-1 : 2*n-1]
    peak_frac = w.max()
    return grad / beta, c_max, peak_frac
