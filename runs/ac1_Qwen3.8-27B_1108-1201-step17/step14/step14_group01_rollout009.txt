    conv_y = fast_conv_self(a)
    
    # Select most active constraints
    active_idx = np.argsort(conv_y)[::-1][:max_constraints]
    
    T = build_T_rows(a, active_idx)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    b = np.maximum(val, 0.0)
    
    # Check if dropped constraints are violated
    check = fast_conv_cross(a, b)
    max_check = check.max()
    if max_check > M * (1 + 1e-3):
        # Add violated constraints and re-solve
        violated = np.where(check > M * 1.001)[0]
        all_active = np.unique(np.concatenate([active_idx, violated]))[:max_constraints * 2]
        T2 = build_T_rows(a, all_active)
        b_var2 = cp.Variable(n, nonneg=True)
        constraints2 = [T2 @ b_var2 <= M]
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
        try:
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            return b
        if prob2.status in ('optimal', 'optimal_inaccurate'):
            val2 = np.asarray(b_var2.value, dtype=np.float64)
            b = np.maximum(val2, 0.0)
    
    return b

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smoothed gradient descent for fast initial optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = np.geomspace(80.0, 0.5, 12)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(30, min(300, int(remaining * 8)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            
            shifted = np.maximum((conv - max_c) / T_val, -50.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient computation via FFT
            W = np.fft.rfft(w, fft_size)
            A = np.fft.rfft(a, fft_size)
            ca = np.fft.irfft(np.conj(A) * W, fft_size)
            aw = np.fft.irfft(A * W, fft_size)
            grad = 2.0 * (ca[:n] + aw[:n])
            
            # Adaptive step size with backtracking
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n))
            
            a_new = a.copy()
            for trial in range(8):
                a_new = np.maximum(a - step * grad, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.05 * T_val:
                    break
                step *= 0.5
            
            a = np.maximum(a - step * grad, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * norm / s_new
            else:
                a = a_new
            
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best_score={best_score:.8f}")

    return best_a, best_score

def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, n_active=35):
    """LP best-response with Nesterov acceleration and lazy constraints."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    
    # Track moving average for additional improvements
    avg_window = []
    window_size = 100
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        b = lp_best_response(y, M, solver, max_constraints=n_active)
        if b is None or b.sum() < 1e-15:
            continue
        
        b = b * norm / b.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            x_new = b.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm / s_x
        
        y = y_new
        a = x_new
        
        # Track best
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_x = fast_conv_self(x_new).max()
        if sc_x < best_score:
            best_score = sc_x
            best_a = x_new.copy()
        
        # Moving average
        avg_window.append(x_new.copy())
        if len(avg_window) > window_size:
            avg_window.pop(0)
        if len(avg_window) == window_size and t % 20 == 0:
            avg_a = np.mean(avg_window, axis=0)
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_a = avg_a * norm / s_avg
                sc_avg = fast_conv_self(avg_a).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_a.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")
        
        # Early stopping if converged
        if t > 500 and abs(sc_y - M) < 1e-10:
            break
    
    return best_a, best_score

def coordinate_descent_optimize(a0, time_limit, rng, verbose=True):
    """Fast coordinate descent with gradient-informed partner selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        # Find peak lags for gradient-informed selection
        peak_threshold = cur_max * 0.9
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            # Select candidate j's
            j_candidates = []
            # Random candidates
            for _ in range(5):
                j_candidates.append(int(rng.integers(n)))
            # Gradient-informed: j near peak_lags - i
            for kl in peak_lags[:3]:
                j1 = int((kl - i) % n)
                j2 = int((kl + i) % n)
                if j1 != i:
                    j_candidates.append(j1)
                if j2 != i:
                    j_candidates.append(j2)
            
            best_delta = 0.0
            best_v = cur_max
            best_j = -1
            
            for j in j_candidates:
                if j == i or j < 0 or j >= n:
                    continue
                aj = a[j]
                
                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Find optimal delta
                n_samples = 15
                deltas = np.linspace(0.01, 0.95, n_samples) * ai
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v - 1e-13:
                        best_v = v
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            # Perturb
            if no_improve == 2:
                n_perturb = max(10, n // 20)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0
        
        if verbose and idx % 200 == 0 and idx > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD it={idx} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def fine_polish(a0, time_limit, rng, verbose=True):
    """Very fine-grained polish: try small perturbations of individual coordinates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        i = int(rng.integers(n))
        ai = a[i]
        if ai < 1e-12:
            continue
        
        # Try small multiplicative perturbation
        for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]:
            new_ai = ai * factor
            if new_ai < 0:
                continue
            # Find where to put/remove the difference
            diff = new_ai - ai
            if abs(diff) < 1e-15:
                continue
            # Distribute to a random coordinate
            j = int(rng.integers(n))
            if j == i:
                continue
            if diff > 0 and a[j] + diff < 0:
                continue
            
            a[i] = new_ai
            a[j] = a[j] - diff
            if a[j] < 0:
                a[i] = ai
                a[j] = a[j] + diff
                continue
            s_new = a.sum()
            if s_new < 1e-15:
                a[i] = ai
                continue
            a = a * norm / s_new
            v = fast_conv_self(a).max()
            if v < best_score - 1e-13:
                best_score = v
                best_a = a.copy()
            else:
                a[i] = ai
                a[j] = a[j] + diff
                a = a * norm / a.sum()
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 3:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 4:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    else:
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            best_seq = p / p.sum()

            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")

        pass

        if a is None:
            return

        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)

            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global

    if best_seq is None:
