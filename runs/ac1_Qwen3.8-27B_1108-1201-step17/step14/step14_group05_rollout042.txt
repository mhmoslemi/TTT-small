def fast_autocorr(a):

    A = rfft(a, fft_size)

    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))

    c = -np.ones(n)
