    # This is cross-correlation of a and w at lags 0..n-1

    grad = 2.0 * cross_corr

def compute_grad_active(a, active_indices):

def smooth_gradient_descent(a0, time_limit, verbose=True):
    """Main optimization: smooth gradient descent with log-sum-exp."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    T_start = 5.0
    T_end = 0.3
    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    
    step_size = 0.1
    min_step = 1e-6
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Number of iterations for this temperature
        # More iterations at lower temperatures
        n_iters = max(50, min(2000, int(remaining * 8 * (si + 1) / n_stages)))
        
        step_size = 0.05 * (1.0 + 0.5 * si / n_stages)
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, conv, max_conv = compute_grad_lse(a, T)
            
            # Project gradient onto tangent space of sum constraint
            grad_mean = grad.mean()
            grad_proj = grad - grad_mean
            
            # Gradient step
            a_new = a - step_size * grad_proj / norm
            
            # Project to non-negative
            a_new = np.maximum(a_new, 0.0)
            
            # Renormalize
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            
            # Evaluate
            new_score = fast_autocorr(a_new).max()
            
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
            
            # Adaptive step size
            if new_score < conv.max() * (1 + 1e-6):
                # Score didn't get worse, can try bigger step
                if it % 50 == 0 and step_size < 0.5:
                    step_size *= 1.1
            else:
                # Score got worse, reduce step
                step_size *= 0.8
                if step_size < min_step:
                    break
                # Revert
                a_new = a.copy()
            
            a = a_new
            
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SGD T={T:.2f} iter={it} ({elapsed:.0f}s): best={best_score:.8f}, step={step_size:.4f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp surrogate + momentum
2. Active-set subgradient refinement on top peaks
3. Pair coordinate descent for local polish
4. Multi-start with different initializations
Uses FFT-based gradient for O(n log n) per iteration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolution(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_grad_lse(a, T):
    """
    Compute gradient of (1/T)*log(sum(exp(conv/T))) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) complexity.
    Returns (grad, conv, max_conv).
    """
    n = len(a)
    conv = fast_convolution(a)
    max_conv = conv.max()
    # Numerically stable log-sum-exp
    shifted = (conv - max_conv) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # grad[j] = 2 * sum_m a[m] * w[j+m] for 0<=m<n
    # This is cross-correlation of a and w at lags 0..n-1
    # Using FFT: cross_corr = ifft(fft(a_padded) * conj(fft(w_padded)))
    len_w = len(w)  # 2n-1
    fft_size = 1 << (n + len_w - 1 - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    cross_corr = irfft(A * np.conj(W), fft_size)[:n]
    grad = 2.0 * cross_corr
    return grad, conv, max_conv


def smooth_gd_optimize(a0, time_limit, T_start=5.0, T_end=0.1, n_stages=10,
                       step_init=0.1, momentum=0.5, verbose=True):
    """
    Smooth gradient descent with log-sum-exp smoothing and momentum.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_convolution(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    step = step_init
    v = np.zeros(n)  # momentum
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # More iterations for lower temperatures
        frac = (si + 1) / n_stages
        n_iters = max(30, int(remaining * 10 * frac))
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, conv, max_conv = compute_grad_lse(a, T)
            
            # Project gradient onto tangent space (sum constraint)
            grad_proj = grad - grad.mean()
            
            # Momentum update
            v = momentum * v + (1 - momentum) * grad_proj
            v = v - v.mean()  # Re-project after momentum
            
            # Step
            a_new = a - step * v / norm
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            
            # Evaluate
            new_conv = fast_convolution(a_new)
            new_score = new_conv.max()
            
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
            
            # Adaptive step: if score increased significantly, reduce step
            old_score = conv.max()
            if new_score > old_score * 1.01 + 1e-6:
                step = max(step * 0.7, 1e-6)
            elif it % 50 == 0 and step < 0.5:
                step = min(step * 1.05, 1.0)
            
            a = a_new
            
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SGD T={T:.3f} si={si} iter={it} ({elapsed:.0f}s): "
                      f"best={best_score:.8f}, step={step:.4f}")
    
    return best_a, best_score


def active_set_refine(a0, time_limit, n_active=20, verbose=True):
    """
    Subgradient method focusing on the top peaks of the convolution.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_convolution(a).max()
    t0 = time.monotonic()
    step = 0.005
    patience = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 200:
            break
        
        conv = fast_convolution(a)
        max_conv = conv.max()
        
        # Find active peaks
        # Top n_active peaks OR peaks within 0.5% of max
        sorted_idx = np.argsort(conv)[::-1]
        active = sorted_idx[:n_active]
        # Also include any peaks within 0.1% of max
        threshold = max_conv * 0.999
        active = np.union1d(active, np.where(conv >= threshold)[0])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            # d(conv[k])/d(a[j]) = 2*a[k-j] for 0<=k-j<n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                indices = np.arange(lo, hi + 1)
                grad[indices] += 2.0 * a[k - indices]
        
        # Project
        grad = grad - grad.mean()
        
        # Step
        a_new = a - step * grad / norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * norm / s
        
        new_score = fast_convolution(a_new).max()
        
        if new_score < best_score - 1e-10:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            step *= 0.95
            if step < 1e-8:
                break
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS iter={it} ({elapsed:.0f}s): best={best_score:.8f}, "
                  f"step={step:.6f}, patience={patience}")
    
    return best_a, best_score
