    return best_a, best_score


def gradient_guided_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent using gradient to select swap pairs."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_seq(a, n)

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    n_trials = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        M = conv.max()
        n_trials += 1

        if M < best_score - 1e-15:
            best_score = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            # Random restart
            for _ in range(50):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    delta = 0.1 * a[i]
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0.0)
                    a = normalize_seq(a, n)
            no_improve = 0
            continue

        # Find peak
        k_peak = int(np.argmax(conv))

        # Compute gradient direction for peak
        # g[p] = 2*a[k_peak - p] if 0 <= k_peak - p < n
        # This is equivalent to: g = 2 * (a shifted into conv space at position k_peak)
        g = np.zeros(n)
        lo = max(0, k_peak - n + 1)
        hi = min(n, k_peak + 1)
        for p in range(lo, hi):
            idx = k_peak - p
            if 0 <= idx < n:
                g[p] = 2.0 * a[idx]

        # Also add near-peak gradients for robustness
        near = np.where(conv > M * 0.97)[0]
        if len(near) > 1:
            for k in near:
                if k == k_peak:
                    continue
                lo2 = max(0, k - n + 1)
                hi2 = min(n, k + 1)
                for p in range(lo2, hi2):
                    idx = k - p
                    if 0 <= idx < n:
                        g[p] += 1.0 * a[idx] * 0.1  # weight near-peaks less

        # Find best swap: source = high g, destination = low g
        valid_src = np.where(a > 1e-8)[0]
        if len(valid_src) == 0:
            break

        # Rank by gradient
        src_rank = valid_src[np.argsort(-g[valid_src])]
        dst_rank = np.argsort(g)

        found_move = False
        for si in range(min(10, len(src_rank))):
            i_src = src_rank[si]
            if a[i_src] < 1e-10:
                continue
            for di in range(min(20, n)):
                i_dst = int(dst_rank[di])
                if i_dst == i_src:
                    continue
                if g[i_dst] >= g[i_src]:
                    break

                # Compute expected change direction
                # delta_conv[k] = delta * (2*a[k-i_dst] - 2*a[k-i_src]) for valid k
                # = delta * 2 * (a_shifted_dst - a_shifted_src)
                # a_shifted_dst[k] = a[k-i_dst] if 0<=k-i_dst<n
                # a_shifted_src[k] = a[k-i_src] if 0<=k-i_src<n

                # Compute max of (conv + delta*(shift_dst - shift_src))
                # Use a few candidate deltas
                best_d = 0
                best_val = M

                for frac in [0.05, 0.15, 0.3, 0.5, 0.8]:
                    d = frac * a[i_src]
                    # Compute conv change efficiently
                    # shift_dst - shift_src in conv space
                    # conv + d * 2 * (shift(a, i_dst) - shift(a, i_src))
                    # shift(a, i) has a at positions [i, i+n-1]
                    val = M
                    # Check at peak and nearby
                    check_range = range(max(0, k_peak - 5), min(size, k_peak + 6))
                    for kk in check_range:
                        contrib = 0.0
                        idx_dst = kk - i_dst
                        if 0 <= idx_dst < n:
                            contrib += 2.0 * d * a[idx_dst]
                        idx_src = kk - i_src
                        if 0 <= idx_src < n:
                            contrib -= 2.0 * d * a[idx_src]
                        v = conv[kk] + contrib
                        if v > val:
                            val = v
                    # Also check positions where the max might shift
                    # (positions of current top-5 conv values)
                    top5 = np.argsort(-conv)[:10]
                    for kk in top5:
                        kk = int(kk)
                        contrib = 0.0
                        idx_dst = kk - i_dst
                        if 0 <= idx_dst < n:
                            contrib += 2.0 * d * a[idx_dst]
                        idx_src = kk - i_src
                        if 0 <= idx_src < n:
                            contrib -= 2.0 * d * a[idx_src]
                        v = conv[kk] + contrib
                        if v > val:
                            val = v

                    if val < best_val - 1e-14:
                        best_val = val
                        best_d = d

                if best_d > 0:
                    a[i_src] -= best_d
                    a[i_dst] += best_d
                    a = np.maximum(a, 0.0)
                    a = normalize_seq(a, n)
                    conv = fast_autocorr(a)
                    found_move = True
                    if conv.max() < best_score:
                        best_score = conv.max()
                        best_a = a.copy()
                    break

            if found_move:
                break

        if not found_move and n_trials % 1000 == 0:
            # Try a random perturbation
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    delta = rng.uniform(0.01, 0.05) * a[i]
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0.0)
                    a = normalize_seq(a, n)
            conv = fast_autocorr(a)
            if conv.max() < best_score:
                best_score = conv.max()
                best_a = a.copy()
                no_improve = 0

        if verbose and n_trials % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GCD trial={n_trials} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def single_coord_cd(a0, time_limit, rng, verbose=True):
    """Single-coordinate descent with exact 1D optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_seq(a, n)

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        M = conv.max()
        k_peak = int(np.argmax(conv))

        if M < best_score - 1e-15:
            best_score = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        # For each coordinate, find optimal value (given others fixed)
        # This is a 1D quadratic problem for each coordinate
        # When we change a_i, conv_k changes by:
        #   2*a_i*a_{k-i} for the direct term, plus cross terms
        # Actually, let's use pair swaps which preserve sum

        # Pick a few random coordinates to optimize
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            if a[i] < 1e-10:
                continue
            j = int(rng.integers(n))
            if j == i:
                continue

            # Optimize: move delta from i to j
            # conv_k change = delta * (2*a[k-j] - 2*a[k-i]) + second order
            # Second order: at k=2i: -2*a[i]*delta + delta^2
            #               at k=2j: +2*a[j]*delta + delta^2
            #               at k=i+j: +2*delta*(a[j]-a[i])... this is getting complex
            # Just do a small line search

            best_d = 0
            best_val = M
            for frac in [0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                d = frac * a[i]
                # Approximate new max
                new_max = M
                # Check at peak and a few other positions
                for kk in [k_peak] + list(np.argsort(-conv)[:5]):
                    contrib = 0.0
                    idx_dst = int(kk) - j
                    if 0 <= idx_dst < n:
                        contrib += 2.0 * d * a[idx_dst]
                    idx_src = int(kk) - i
                    if 0 <= idx_src < n:
                        contrib -= 2.0 * d * a[idx_src]
                    # Second order at diagonal
                    if int(kk) == 2 * i:
                        contrib += -2.0 * a[i] * d + d * d
                    if int(kk) == 2 * j:
                        contrib += 2.0 * a[j] * d + d * d
                    v = conv[int(kk)] + contrib
                    if v > new_max:
                        new_max = v
                if new_max < best_val - 1e-14:
                    best_val = new_max
                    best_d = d

            if best_d > 0:
                a[i] -= best_d
                a[j] += best_d
                a = np.maximum(a, 0.0)
                a = normalize_seq(a, n)

    conv = fast_autocorr(a)
    if conv.max() < best_score:
        best_score = conv.max()
        best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng, style=0):
    """Generate structured initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    styles = [
        lambda: np.cos(np.pi * x / 2.0) ** 2,
        lambda: np.sin(np.pi * x) ** 1.5,
        lambda: 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(10 * np.pi * x),
        lambda: np.exp(-4.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-4.0 * (x - 0.7) ** 2),
        lambda: (1.0 - np.abs(2.0 * x - 1.0)) ** 2,
        lambda: 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5,
        lambda: 1.0 + 0.2 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x),
        lambda: np.sin(np.pi * x) ** 0.7 * (1 + 0.15 * np.cos(8 * np.pi * x)),
    ]
    idx = style % len(styles)
    a = styles[idx]()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    """Add multiplicative noise."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.2)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate():
    rng = np.random.default_rng(2024)
    t_start = time.monotonic()
    time_budget = 1740.0

    solver = get_solver()
    print(f"Solver: {solver}, time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng, style=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov (full constraints) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0, use_screening=False)
        do_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Gradient-guided CD ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Gradient CD n={n_cur}, budget={t2:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = gradient_guided_cd(a_norm, t2, rng, verbose=True)
        do_update(a_out, f"gcd_n{n_cur}")

    # === Phase 3: LP Nesterov with screening (faster, more iterations) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP Nesterov (screened) n={n_cur}, budget={t3:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=7, use_screening=True)
        do_update(a_out, f"lp3_n{n_cur}")

    # === Phase 4: Single coord CD ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Single Coord CD n={n_cur}, budget={t4:.0f}s ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = single_coord_cd(a_norm, t4, rng, verbose=True)
        do_update(a_out, f"sccd_n{n_cur}")

    # === Phase 5: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 120:
        n_long = n_cur + 200
        print(f"\n=== Phase 5: LP Nesterov n={n_long} ===")
        t5 = min(remaining * 0.20, 200.0)
        if t5 > 30:
            init_long = interpolate_sequence(best_seq, n_long)
            a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, seed=42, use_screening=False)
            do_update(a_out, f"lp_n{n_long}")
            # Follow up with CD
            remaining = time_budget - (time.monotonic() - t_start)
            if remaining > 60:
                t5b = min(remaining * 0.15, 100.0)
                if t5b > 20:
                    a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
                    a_out, _ = gradient_guided_cd(a_norm, t5b, rng, verbose=False)
                    do_update(a_out, f"gcd_long_n{len(best_seq)}")

    # === Phase 6: Perturbation + LP ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 100:
        print(f"\n=== Phase 6: Perturbation restart ===")
        n_cur = len(best_seq)
        pert = perturb_sequence(best_seq, rng, scale=0.04)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        t6 = min(remaining * 0.25, 200.0)
        if t6 > 30:
            a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, seed=99, use_screening=False)
            do_update(a_out, f"pert_n{n_cur}")

    # === Phase 7: Final gradient CD polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 7: Final Gradient CD n={n_cur} ===")
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = gradient_guided_cd(a_norm, t7, rng, verbose=True)
            do_update(a_out, f"final_cd_n{n_cur}")

    # === Final ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
