        # j ranges from 0 to n-1, valid when 0 <= k_star - j < n
        # i.e., k_star - n + 1 <= j <= k_star
        # So valid j: max(0, k_star-n+1) to min(n-1, k_star)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n, k_star + 1)  # exclusive
        # g[j] = 2*y[k_star - j] for j in [j_lo, j_hi)
        # We want to minimize g[j], which means minimize y[k_star - j]
        # k_star - j ranges from k_star - (j_hi-1) to k_star - j_lo
        # = k_star - min(n, k_star+1) + 1 to k_star - max(0, k_star-n+1)
        # Let's just find argmin of y in the relevant range
        idx_lo = k_star - (j_hi - 1)
        idx_hi = k_star - j_lo + 1
        idx_lo = max(0, idx_lo)
        idx_hi = min(n, idx_hi)
        if idx_hi > idx_lo:
            m = idx_lo + int(np.argmin(y[idx_lo:idx_hi]))
            j_min = k_star - m
        else:
            j_min = 0

        # FW direction: all mass at j_min
        v = np.zeros(n)
        v[j_min] = norm

        # Nesterov momentum
        if t == 1:
            beta = 0.0
        else:
            beta = (t - 1.5 + np.sqrt(2*t - 0.75)) / (t + 1.5)
            beta = min(beta, 0.95)

        y_new = x + beta * (x - prev_x)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x.copy()
        else:
            y_new = y_new * norm / s_y

        # FW step
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y_new + alpha * v
        x_new = np.maximum(x_new, 0.0)
        s_x = x_new.sum()
        if s_x < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new * norm / s_x

        prev_x = x
        x = x_new
        y = y_new

        # Check improvement
        score = fast_autocorr(x).max()
        if score < best_score:
            best_score = score
            best_a = x.copy()

        # Ergodic average
        avg_x += x
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_x / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose and t % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.1f}s): best={best_score:.10f}")

    return best_a, best_score


def greedy_cd(a0, time_limit, verbose=True, max_passes=200):
    """Greedy coordinate descent: move mass from high-gradient to low-gradient positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    for pass_num in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False

        # Find peak
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Compute gradient at peak: g[j] = 2*a[k_star - j] for valid j
        # We want to move mass from high g to low g
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n, k_star + 1)

        # For each source i (high g), find best target j (low g)
        # Try the top few sources
        sources_in_range = np.arange(j_lo, j_hi)
        grad_vals = 2.0 * a[max(0, k_star - (j_hi - 1)):min(n, k_star - j_lo + 1)]
        # Actually, let me just work with the full gradient vector
        g = np.zeros(n)
        for j in range(j_lo, j_hi):
            g[j] = 2.0 * a[k_star - j]

        # Find top sources (highest gradient) and bottom targets (lowest gradient)
        # Only consider positions with non-zero a
        nonzero = a > norm * 1e-8
        source_candidates = np.where(nonzero & (g > g.mean()))[0]
        target_candidates = np.where(g < g.mean())[0]

        if len(source_candidates) == 0 or len(target_candidates) == 0:
            break

        # Try top 5 sources, each with their best target
        source_candidates = source_candidates[np.argsort(g[source_candidates])[-5:]]

        for i in source_candidates:
            if time.monotonic() - t0 > time_limit:
                break

            # Best target: minimize g[j] (i.e., minimize a[k_star - j])
            # Among targets, prefer those with small a[k_star-j]
            target_scores = g[target_candidates]
            j = target_candidates[np.argmin(target_scores)]
            if j == i:
                continue

            # Line search: find optimal d to move from i to j
            # Change in conv: delta[k] = d*(g[j] - g[i]) approximately
            # More precisely, the first-order change in conv[k] is:
            #   d_conv[k] = -d * (2*a[k-i] valid) + d * (2*a[k-j] valid)
            # = d * (grad_at_j[k] - grad_at_i[k])
            # where grad_at_i[k] = 2*a[k-i] if 0<=k-i<n else 0

            # The new conv: new_conv[k] = conv[k] + d * h[k]
            # where h[k] = 2*a[k-j] (if valid) - 2*a[k-i] (if valid)
            # We want to minimize max(new_conv[k]) = max(conv[k] + d*h[k])
            # subject to d in [0, a[i]]

            # Build h vector
            h = np.zeros(size)
            lo_i = max(0, i)
            hi_i = min(size, i + n)
            src_i = 0
            for k in range(lo_i, hi_i):
                h[k] -= 2.0 * a[k - i]

            lo_j = max(0, j)
            hi_j = min(size, j + n)
            for k in range(lo_j, hi_j):
                h[k] += 2.0 * a[k - j]

            # Also second-order: if k = 2i, additional -d^2; if k = 2j, additional +d^2
            # For now, use first-order + diagonal correction

            # Find optimal d: minimize max_k (conv[k] + d*h[k])
            # The peak will decrease if h[k_star] < 0
            if h[k_star] >= 0:
                continue  # This move would increase the peak

            # The max can only increase at positions where h[k] > 0
            # The new max is max(conv[k] + d*h[k])
            # = max(max_{h[k]<=0} (conv[k] + d*h[k]), max_{h[k]>0} (conv[k] + d*h[k]))
            # The first part is maximized at d=0 (since h[k]<=0 means it decreases)
            #   Actually no: conv[k] + d*h[k] for h[k]<=0 is decreasing in d
            #   So the max over h[k]<=0 is at d=0, which is conv[k]
            #   Wait, we want the max over ALL k, so:
            #   For h[k] < 0: conv[k] + d*h[k] < conv[k] (decreasing)
            #   For h[k] = 0: conv[k]
            #   For h[k] > 0: conv[k] + d*h[k] (increasing)
            # So the new max = max(max_{h[k]<=0} conv[k] + d*h[k], max_{h[k]>0} conv[k] + d*h[k])
            # = max(conv_at_decreasing_positions, max_{h[k]>0}(conv[k] + d*h[k]))
            # The first term is <= conv.max() = peak_val
            # The second term increases with d
            # So we want the largest d such that max_{h[k]>0}(conv[k] + d*h[k]) <= peak_val
            # i.e., for all k with h[k] > 0: d <= (peak_val - conv[k]) / h[k]

            pos_h = h > 1e-12
            if np.any(pos_h):
                d_limits = (peak_val - conv[pos_h]) / h[pos_h]
                d_limits = d_limits[d_limits > 0]
                if len(d_limits) > 0:
                    d_max = d_limits.min()
                else:
                    d_max = a[i]  # No constraint from increasing positions
            else:
                d_max = a[i]

            # Also can't move more than available
            d_max = min(d_max, a[i])

            # Use a fraction of d_max (to be safe and allow further improvements)
            d = 0.5 * d_max
            if d < 1e-15:
                continue

            # Apply the move
            a[i] -= d
            a[j] += d

            # Renormalize
            s = a.sum()
            if s < 1e-15:
                a[i] += d
                a[j] -= d
                continue
            a = a * norm / s

            # Recompute conv
            conv = fast_autocorr(a)
            cur_max = conv.max()

            if cur_max < best_score - 1e-14:
                best_score = cur_max
                best_a = a.copy()
                pass_improved = True
            else:
                # Revert
                a[j] -= d
                a[i] += d
                a = a * norm / a.sum()
                conv = fast_autocorr(a)

        if not pass_improved:
            # Try a few random moves
            for _ in range(20):
                i = int(np.random.randint(n))
                j = int(np.random.randint(n))
                if i == j or a[i] < 1e-10:
                    continue
                d = a[i] * 0.1
                a[i] -= d
                a[j] += d
                a = np.maximum(a, 0)
                a = a * norm / a.sum()
                conv = fast_autocorr(a)
                if conv.max() < best_score:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[j] -= d
                    a[i] += d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
            if not pass_improved:
                break

        if verbose and (pass_num + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_num+1} ({elapsed:.1f}s): best={best_score:.10f}")

    return best_a, best_score


def build_conv_matrix_rows(a, k_indices):
    """Build rows of the convolution matrix for specific k indices."""
    n = len(a)
    rows = []
    for k in k_indices:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a[idx]
        rows.append(row)
    return np.array(rows)


def lp_best_response(a, M, solver, active_set=None):
    """Solve the LP best-response: maximize sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0:
        T = build_conv_matrix_rows(a, active_set)
    else:
        size = 2 * n - 1
        T = build_conv_matrix_rows(a, np.arange(size))
        active_set = np.arange(size)

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active_set

    return b_val, active_set


def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0):
    """Nesterov-accelerated LP best-response."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_x = a.copy()
    avg_x = np.zeros(n)
    n_avg = 0
    t = 0
    active_set = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if t == 1 or t % 50 == 0:
            tol = 0.05 + 0.05 * np.exp(-t / 200.0)
            active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
            if len(active_set) < 3:
                active_set = np.where(conv_y > M_curr * 0.7)[0]

        b, active_set_new = lp_best_response(y, M_curr, solver, active_set)

        if b is None:
            if t % 10 == 0:
                active_set = None
                b, active_set_new = lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue

        active_set = active_set_new
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        if t == 1:
            beta = 0.0
        else:
            beta = (t - 1.5 + np.sqrt(2*t - 0.75)) / (t + 1.5)
            beta = min(beta, 0.95)

        y_new = x + beta * (x - prev_x)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x.copy()
        else:
            y_new = y_new * norm / s_y

        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y_new + alpha * g
        x_new = np.maximum(x_new, 0.0)
        s_x = x_new.sum()
        if s_x < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new * norm / s_x

        prev_x = x
        x = x_new
        y = y_new

        # Check
        score = fast_autocorr(x).max()
        if score < best_score:
            best_score = score
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Ergodic average
        avg_x += x
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_x / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
