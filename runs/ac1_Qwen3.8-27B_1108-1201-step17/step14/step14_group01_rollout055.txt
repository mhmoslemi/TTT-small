def fast_conv_self(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    A = rfft(a, fft_size)

def build_T(a):

        T[m:m + n, m] = a

def lp_solve(a, M, solver, active=None, warm_start=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):

    norm = np.sqrt(2.0 * n)

    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    rng = np.random.default_rng(123)
    
    no_improve = 0
    n_since_improve = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Every 1000 iterations, try a full LP solve for verification
        conv_y = fast_conv_self(y)
        M = conv_y.max()

        # Adaptive constraint screening
        # Start with tight threshold, relax if too few constraints
        threshold_frac = max(0.7, 0.95 - 0.1 * (t / 5000.0))

            active = np.where(conv_y >= M * 0.5)[0]

        warm_start = y.copy()  # Use current as warm start
        b = lp_solve(y, M, solver, active, warm_start)

            b = lp_solve(y, M, solver, None, warm_start)

def pair_cd_optimize(a0, time_limit, rng, verbose=True):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            conv = fast_conv_self(a)
            cur_max = conv.max()

            # Find candidate j's
            j_candidates = set()
            # Random
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            # Structured: near the peak lags
            peak_lags = np.where(conv >= cur_max * 0.9)[0]
            for kl in peak_lags[:4]:
                j_struct = int((kl - i) % n)
                j_candidates.add(j_struct)
                j_struct2 = int((kl + i) % n)
                j_candidates.add(j_struct2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Build gain vector g where new_conv = conv + delta * g (approximately)
                # Exact: conv_new[k] = conv[k] + delta*(a[k-j]+a[j-k]-a[k-i]-a[i-k]) + quadratic terms
                # For small delta, the linear approximation is good; we'll verify exactly.
                
                # Sample delta values and check exactly
                best_d = 0.0
                best_v = cur_max
                n_samples = 20
                for d in np.linspace(0.01, 0.99, n_samples) * ai:
                    a[i] = ai - d
                    a[j] = aj + d
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a * norm / ss
                    v = fast_conv_self(a).max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                    a[i] = ai
                    a[j] = aj
                    a = a * norm / a.sum()

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] = ai - best_delta
                a[best_j] = a[best_j] + best_delta
                ss = a.sum()
                if ss > 1e-15:
                    a = a * norm / ss
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
            # Perturb
            if no_improve == 2:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= np.maximum(1.0 + 0.15 * rng.standard_normal(n_perturb), 0.1)
                a = np.maximum(a, 0.0)

def subgradient_optimize(a0, time_limit, verbose=True):
    """Direct subgradient descent on the actual objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    # Use Polyak step size
    f_star_est = best_score * 0.99  # optimistic estimate
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv_self(a)
        max_c = conv.max()
        k_star = np.argmax(conv)
        s_a = a.sum()
        
        # Subgradient of max(conv) at a: 2 * T[k_star, :]
        # T[k_star, i] = a[i] if k_star - n + 1 <= i <= k_star
        grad_max = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        grad_max[lo:hi] = 2.0 * a[lo:hi]
        
        # Projected subgradient step on the ratio
        # f = 2n * max_c / s_a^2
        # We want to decrease f. Moving in direction -grad_max decreases max_c
        # but we need to maintain normalization.
        # Simple approach: a_new = a - step * (grad_max - mean(grad_max) * a/s_a)
        # Then renormalize.
        
        step = f_star_est / it**0.7  # decreasing step size
        step = min(step, 0.5 * norm / (np.linalg.norm(grad_max) + 1e-15))
        
        # Average of subgradients (use a few random k values near the peak)
        n_sub = 5
        grad_avg = np.zeros(n)
        for _ in range(n_sub):
            kk = k_star + int(np.random.randint(-3, 4))
            kk = max(0, min(size-1, kk))
            g = np.zeros(n)
            lo = max(0, kk - n + 1)
            hi = min(n, kk + 1)
            g[lo:hi] = 2.0 * a[lo:hi]
            grad_avg += g
        grad_avg /= n_sub
        
        # Remove component along all-ones (to preserve sum)
        grad_avg -= np.mean(grad_avg)
        
        a_new = a - step * grad_avg
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * norm / s_new
            sc = fast_conv_self(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                f_star_est = best_score * 0.995
            a = a_new
        else:
            break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        # Flat with small ripple
        a = 1.0 + 0.1 * np.sin(2 * np.pi * 3 * x) + 0.05 * np.sin(2 * np.pi * 7 * x)
    else:
        # Two broad peaks
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.75) ** 2) + 0.3
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
