                print(f"    Reset at iter {n_iters}, t={time.monotonic()-t0:.0f}s")

        if verbose and n_iters % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ND iter={n_iters} ({elapsed:.0f}s): T={T:.2f}, best={best_score:.8f}")

    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, max_iters=500):
    """LP best-response with Nesterov acceleration (for polishing)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    a_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 20 == 0:
            print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Targeted coordinate descent for final peak reduction."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]
        improved = False

        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:5]]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    continue
                cand_sorted = candidates[np.argsort(a[candidates])][:(len(candidates) // 4) + 1]

                for dst in cand_sorted[:4]:
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 3 == 0:
                for _ in range(20):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.4) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(2.0 * x) + 0.1 * np.cos(4.0 * x)
    elif choice == 3:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.4 * np.sin(3.0 * np.pi * x) + 0.15 * np.sin(7.0 * np.pi * x)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine base length
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 0: Quick multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = sorted(set([n_base, max(1200, n_base - 200), n_base + 200]))
    scan_phase_time = min(time_budget * 0.10, 150.0)
    t_phase0 = time.monotonic()

    for n_scan in scan_lengths:
        if time.monotonic() - t_phase0 > scan_phase_time:
            break
        remaining = scan_phase_time - (time.monotonic() - t_phase0)
        if remaining < 10:
            break
        scan_time = min(remaining / 2.0, 100.0)

        init_seq = interpolate_sequence(best_seq, n_scan)
        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_gradient_descent(a_norm, scan_time,
                                                   T_start=60.0, T_end=5.0, verbose=False)
        # Evaluate with actual objective
        a_eval = a_out / a_out.sum()
        sc_eval = evaluate_safe(a_eval)
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
            best_seq = a_eval.copy()
            print(f"    n={n_scan}: NEW BEST score={sc_eval:.6f}")
        else:
            print(f"    n={n_scan}: max_conv={sc_out:.6f}, eval={sc_eval:.6f}")

    # Phase 1: Main Nesterov gradient descent (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t1 = remaining * 0.45
    print(f"\n=== Phase 1: Nesterov GD n={n_best} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, sc_out = nesterov_gradient_descent(a_norm, t1,
                                               T_start=100.0, T_end=1.0,
                                               verbose=True, use_topK_switch=True, K_switch=15)
    a_eval = a_out / a_out.sum()
    do_try_update(a_eval, f"nesterov_n{n_best}")

    # Phase 2: Second Nesterov pass with different parameters
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 2: Nesterov GD pass 2 n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        # Add small noise to escape local minima
        a_noisy = best_seq + rng.normal(0, 0.01 * best_seq.mean(), n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_norm = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = nesterov_gradient_descent(a_norm, t2,
                                              T_start=40.0, T_end=0.5,
                                              verbose=True, use_topK_switch=True, K_switch=10)
        a_eval = a_out / a_out.sum()
        do_try_update(a_eval, f"nesterov2_n{n_best}")

    # Phase 3: LP polishing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: LP polish n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t3, solver, verbose=True, max_iters=100)
        a_eval = a_out / a_out.sum()
        do_try_update(a_eval, f"lp_polish_n{n_best}")

    # Phase 4: Another Nesterov pass at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 4: Nesterov GD pass 3 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_gradient_descent(a_norm, t4,
                                              T_start=20.0, T_end=0.3,
                                              verbose=True, use_topK_switch=True, K_switch=8)
        a_eval = a_out / a_out.sum()
        do_try_update(a_eval, f"nesterov3_n{n_best}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=12345)
        a_eval = a_out / a_out.sum()
        do_try_update(a_eval, "coord_polish")

    # Phase 6: Final LP + coord
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.30, 300.0)
    print(f"\n=== Phase 6: Final LP + Coord n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, min(t6 * 0.5, 120), solver, verbose=False, max_iters=50)
        a_eval = a_out / a_out.sum()
        do_try_update(a_eval, "final_lp")

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coordinate_descent_polish(a_norm, remaining2 * 0.8, seed=777)
            a_eval = a_out / a_out.sum()
            do_try_update(a_eval, "final_coord")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
