The dual variables `y` give us the "shadow prices" of the constraints. Positions where `y > 0` are the binding constraints. This tells us which positions in the convolution are most important to reduce.

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)

def compute_lse_gradient(a, T):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    conv_shifted = conv - conv.max()
    conv_shifted = np.maximum(conv_shifted, -100.0)
    p = np.exp(conv_shifted / T)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(size) / size
    else:
        p /= p_sum
    P = rfft(p, fft_size)
    grad = 2.0 * irfft(np.conj(A) * P, fft_size)[:n]
    return grad, conv


def lse_nesterov_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

    t0 = time.monotonic()

    n_stages = 12
    temps = np.geomspace(80.0, 0.2, n_stages)

    y = a.copy()
    momentum = 0.0

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        if remaining < 3:
            break

        momentum = 0.0

            grad, conv = compute_lse_gradient(y, T_val)

            # Nesterov momentum

            else:
                y_new_momentum = y.copy()
                momentum = 1.0

            a = y_new_momentum

                        continue
                    a_try = a_try * norm_val / s_try
                    conv_try = fast_conv_self(a_try)

                    step *= 0.5

def build_T_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [cp.multiply(T[active, :], b_var) <= M]
    else:
        constraints = [cp.multiply(T, b_var) <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)
