"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with interpolation
2. LSE smooth approximation gradient descent for initialization
3. LP best-response with Nesterov acceleration (main engine)
4. Direct subgradient Nesterov on max-convolution
5. Vectorized pair coordinate descent for polishing
6. Adaptive time allocation based on progress
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    size = 2 * len(a) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def norm_seq(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a) * target_sum
    return a * target_sum / s

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = 8
    temps = np.geomspace(80.0, 0.5, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(10, min(300, int(remaining * 4)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)
            
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
            
            for trial in range(15):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = norm_seq(a_new, norm)
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.01 * T_val:
                    break
                step *= 0.5
            
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")
    
    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    
    max_iters = int(time_limit * 100)  # rough estimate
    t = 0
    while time.monotonic() - t0 < time_limit and t < max_iters:
        t += 1
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive screening
        threshold = M * (0.7 + 0.2 * min(t / 500.0, 1.0))
        active = np.where(conv_y > threshold)[0]
        if len(active) < 30:
            active = np.where(conv_y > M * 0.4)[0]
        
        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue
        
        b_sum = b.sum()

def subgradient_nesterov(a0, time_limit, rng, verbose=True):
    """Direct subgradient method on max-conv with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    t = 0
    step_base = norm / np.sqrt(n)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv_y = fast_conv_self(y)
        max_c = conv_y.max()
        argmax = np.argmax(conv_y)
        
        # Subgradient: for max_k conv(k), subgradient at a is 2*T[argmax,:] * a (approximately)
        # More precisely: d/d a_i max_k sum_j a[j]*a[k-j] = 2 * sum over k=argmax of (a[k-i] + a[i-k]) / 2
        # = a[argmax - i] + a[i - argmax] (with boundary handling)
        sg = np.zeros(n)
        k = argmax
        for i in range(n):
            vi = 0.0
            if 0 <= k - i < n:
                vi += a[k - i]
            if 0 <= i - k < n:
                vi += a[i - k]
            sg[i] = vi
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Adaptive step size
        step = step_base / (t + 1) ** 0.7
        step = min(step, max_c / sg_norm * 0.5)
        
        alpha = 2.0 / (t + 2.0)
        y_new = y - step * sg / max(max_c, 1e-10)
        y_new = np.maximum(y_new, 0.0)
        y_new = norm_seq(y_new, norm)
        
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = norm_seq(a_new, norm)
        
        y = y_new
        a = a_new
        
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): {best_score:.8f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

5. Vectorized pair coordinate descent with binary search for polishing
6. Adaptive time allocation and geometric averaging of iterates
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    size = 2 * len(a) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def norm_seq(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    n_stages = 10
    temps = np.geomspace(100.0, 0.3, n_stages)
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(300, int(remaining * 5)))
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
            for trial in range(15):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = norm_seq(a_new, norm)
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.01 * T_val:
                    break
                step *= 0.5
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")
    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    avg_a = np.zeros(n)
    n_avg = 0
    t = 0
    no_progress = 0
    last_best = best_score
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        # Adaptive screening: start loose, tighten over time
        progress_frac = min(t / 200.0, 1.0)
        threshold = M * (0.6 + 0.35 * progress_frac)
        active = np.where(conv_y > threshold)[0]
        if len(active) < 20:
            active = np.where(conv_y > M * 0.4)[0]
        b = lp_solve(y, M, solver, active)
        if b is None:
            b = lp_solve(y, M, solver, None)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        # Verify dropped constraints periodically
        if active is not None and len(active) < size and t % 15 == 0:
            check_conv = fast_conv_cross(y, g)
            if check_conv.max() > M * (1 + 5e-4):
                b_full = lp_solve(y, M, solver, None)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum
        # Nesterov update with standard schedule
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = norm_seq(y_new, norm)
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = norm_seq(a_new, norm)
        y = y_new
        a = a_new
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        # Track average iterate
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            avg_check = norm_seq(avg_check, norm)
            sc_avg = fast_conv_self(avg_check).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()
        # Detect stagnation and restart
        if t > 100 and best_score < last_best - 1e-8:
            no_progress = 0
            last_best = best_score
        else:
            no_progress += 1
        if no_progress > 300:
            # Warm restart: reset momentum
            y = best_a.copy()
            a = best_a.copy()
            no_progress = 0
            last_best = best_score
            if verbose:
                print(f"      [restart at t={t}]")
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def subgradient_optimize(a0, time_limit, rng, verbose=True):
    """Direct subgradient Nesterov on max-k conv(k). Subgrad at k*: sg[i] = 2*a[k*-i] if valid."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv_y = fast_conv_self(y)
        max_c = conv_y.max()
        if max_c < 1e-15:
            break
        k_star = int(np.argmax(conv_y))
        # Subgradient: sg[i] = 2*y[k_star - i] if 0 <= k_star-i < n
        sg = np.zeros(n)
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        if i_hi >= i_lo:
            sg[i_lo:i_hi + 1] = 2.0 * y[k_star - np.arange(i_lo, i_hi + 1)]
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        # Step size: O(1/sqrt(t)) adaptive
        step = norm / (max_c * np.sqrt(t + 1))
        step = min(step, 0.5 / sg_norm * norm)
        alpha = 2.0 / (t + 2.0)
        y_new = y - step * sg
        y_new = np.maximum(y_new, 0.0)
        y_new = norm_seq(y_new, norm)
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = norm_seq(a_new, norm)
        y = y_new
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): {best_score:.8f}")
    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = norm_seq(a, norm)
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            # Candidate j's: random + structured
            peak_lags = np.where(conv >= cur_max * 0.85)[0][:8]
            j_candidates = set()
            for _ in range(8):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j_candidates.add((kl - i) % n)
                j_candidates.add((kl + i) % n)
            j_candidates.add((i + 1) % n)
            j_candidates.add((i - 1) % n)
            j_candidates.discard(i)
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            for j in list(j_candidates)[:15]:
                if j == i:
                    continue
                aj = a[j]
                # Compute gain vector g where conv_new = conv + delta*g (linear approx)
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                # Binary search for optimal delta in (0, ai]
                lo_d = 0.0
                hi_d = ai
                best_d = 0.0
                best_v = cur_max
                # Include the exact second-order correction at peaks 2i and 2j
                for _ in range(14):
                    mid_d = (lo_d + hi_d) / 2.0
                    if mid_d < 1e-14:
                        lo_d = mid_d
                        continue
                    new_conv = conv + mid_d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2.0 * ai * mid_d + mid_d * mid_d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2.0 * aj * mid_d + mid_d * mid_d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = mid_d
                        lo_d = mid_d
                    else:
                        hi_d = mid_d
                if best_d > 1e-12 and best_v < best_val - 1e-13:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j
            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = norm_seq(a, norm)
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-14:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = norm_seq(a, norm)
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve == 3:
                n_perturb = max(5, n // 60)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                a = norm_seq(a, norm)
        else:
            no_improve = 0
        if verbose and total_improvements > 0 and total_improvements % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): {best_score:.8f}")
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sin(np.pi * x) ** 0.5 + 0.3 * np.sin(3 * np.pi * x) ** 2
    elif choice == 9:
        a = np.exp(-4.0 * (x - 0.25) ** 2) + np.exp(-4.0 * (x - 0.75) ** 2) + 0.3
    elif choice == 10:
        a = 0.4 + 0.4 * np.cos(np.pi * x) ** 2 + 0.2 * np.cos(3 * np.pi * x)
    else:
        a = np.exp(-2.0 * (x - 0.5) ** 2) * (1.0 + 0.3 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1650.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    best_seq = None
    best_score = float('inf')
    # Load parent
    parent_n = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_n = len(p)
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={parent_n}, score={best_score:.6f}")
    except Exception:
        pass
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = parent_n
    # Phase 1: Quick LSE GD
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE n={n_base} ({t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, f"lse_{n_base}")
    # Phase 2: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.35, 580.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} ({t2:.0f}s) ===")
    if t2 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_{n2}")
    # Phase 3: Subgradient exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: SubGrad n={n3} ({t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = subgradient_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"subgrad_{n3}")
    # Phase 4: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 4: Pair CD n={n4} ({t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd_{n4}")
    # Phase 5: Second LP round
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 5: LP round 2 n={n5} ({t5:.0f}s) ===")
    if t5 > 40:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver, verbose=True)
        try_update(a_out, f"lp2_{n5}")
    # Phase 6: Different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        print(f"\n=== Phase 6: Length exploration (rem={remaining:.0f}s) ===")
        n_cur = len(best_seq)
        candidates = [n_cur + 300, n_cur - 300, n_cur + 600, n_cur - 600]
        candidates = [c for c in candidates if 400 <= c <= 3500]
        for n_try in candidates:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            t_sub = min(remaining * 0.4, 150.0)
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = nesterov_lp_optimize(init_try, min(t_sub, 100.0), solver, verbose=False)
            try_update(a_out, f"lp_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 50 and len(best_seq) == n_try:
                a_out2, _ = pair_cd_optimize(best_seq, min(remaining * 0.3, 50.0), rng, verbose=False)
                try_update(a_out2, f"pcd_n{n_try}")
    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 7: Final polish (rem={remaining:.0f}s) ===")
        t_lse = min(remaining * 0.3, 50.0)
        a_out, _ = lse_gradient_descent(best_seq, t_lse, verbose=False)
        try_update(a_out, f"lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.6, 60.0), rng, verbose=False)
            try_update(a_out, f"pcd_final")
    # Final eval
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    return [float(x) for x in best_seq]
