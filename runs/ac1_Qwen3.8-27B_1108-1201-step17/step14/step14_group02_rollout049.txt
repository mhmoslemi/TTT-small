1. Nesterov-accelerated Frank-Wolfe with LP best-response (scipy HiGHS)

import numpy as np

from scipy.optimize import linprog
from scipy.linalg import toeplitz
from scipy import sparse

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_T(y):
    n = len(y)
    c = np.zeros(2 * n - 1)
    c[:n] = y
    r = np.zeros(n)
    r[0] = y[0]
    return toeplitz(c, r)

def lp_best_response(y, M):
    n = len(y)
    T = build_T(y)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_objective_bound': -1e10})
    if res.success:

    return None

        if b_sum < 1e-15:

        g = b / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)

            sx = x_new.sum()
            if sx < 1e-15:

            else:

        x = x_new

            best_a = y.copy()
