This is equivalent to finding the smallest t such that the LP:
`c_k + d*(a_{k-j} - a_{k-i}) + d^2*([k==2j] - [k==2i]) <= t` for all k
has a feasible (d, t) with d >= 0.

For a fixed t, the constraint on d from position k is:
`d^2*([k==2j] - [k==2i]) + d*(a_{k-j} - a_{k-i}) + (c_k - t) <= 0`

This is a quadratic in d (for most k it's linear since the coefficient of d^2 is 0). The feasible set for d is the intersection of half-lines (or half-planes for the quadratic cases).

This is getting complex. Let me just use a simpler approach: for each pair (i, j), do a golden section search or ternary search on d to find the optimal transfer amount.

Actually, the simplest effective approach is:
1. For each i (in random order), try to find the best j and best d.
2. For a given i and j, the optimal d can be found by:
   a. Identifying the "critical" k values (those near the current max)
   b. For each critical k, computing the d that would make `(a'*a')_k` equal to the current max
   c. Taking the smallest such d (that reduces the peak)
3. This is essentially a "water-filling" step.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to_norm(a, target_norm):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_norm / len(a)
    return a * target_norm / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_fast(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix_fast(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    norm_target = np.sqrt(2.0 * n)

def single_coord_optimize(a, conv, i, n, size, top_k_indices, num_candidates=20):
    S = a.sum()
    ai = a[i]
    if ai < 1e-15:
        return 0.0, None

    best_d = 0.0
    best_max = conv.max()

    for k in top_k_indices:
        if k < 0 or k >= size:
            continue
        ck = conv[k]
        if k - i >= 0 and k - i < n:
            s_k = a[k - i]
        else:
            s_k = 0.0
        C = 1.0 if k == 2 * i else 0.0
        B = 2.0 * s_k

        if C == 0.0:
            if abs(B) < 1e-15:
                continue
            d_opt = (2.0 * ck - B * S) / (2.0 * C * S - B) if (2.0 * C * S - B) != 0 else 0.0
        else:
            denom = 2.0 * C * S - B
            if abs(denom) < 1e-15:
                continue
            d_opt = (2.0 * ck - B * S) / denom

        if d_opt < -ai + 1e-15:
            d_opt = -ai + 1e-15
        if d_opt > 0:
            d_opt = d_opt * 0.99

        new_conv_est = conv[k] + d_opt * B + d_opt * d_opt * C
        if new_conv_est < 0:
            new_conv_est = 0.0

        new_max = 0.0
        for k2 in top_k_indices:
            if k2 < 0 or k2 >= size:
                continue
            if k2 - i >= 0 and k2 - i < n:
                s_k2 = a[k2 - i]
            else:
                s_k2 = 0.0
            C2 = 1.0 if k2 == 2 * i else 0.0
            B2 = 2.0 * s_k2
            val2 = conv[k2] + d_opt * B2 + d_opt * d_opt * C2
            if val2 > new_max:
                new_max = val2

        new_max = max(new_max, best_max + d_opt * 0.0)
        if new_max < best_max - 1e-13:
            best_max = new_max
            best_d = d_opt

    return best_d, best_max


def pair_move_optimize(a, conv, i, j, n, size, top_k_indices):
    S = a.sum()
    ai = a[i]
    if ai < 1e-15:
        return 0.0

    best_d = 0.0
    best_max = conv.max()

    d_samples = np.linspace(0.0, ai * 0.95, 30)
    for d in d_samples:
        if d <= 0:
            continue
        new_max = 0.0
        for k in top_k_indices:
            if k < 0 or k >= size:
                continue
            delta = 0.0
            if k - j >= 0 and k - j < n:
                delta += a[k - j]
            if k - i >= 0 and k - i < n:
                delta -= a[k - i]
            delta *= 2.0 * d
            C_j = 1.0 if k == 2 * j else 0.0
            C_i = 1.0 if k == 2 * i else 0.0
            delta2 = d * d * (C_j - C_i)
            val = conv[k] + delta + delta2
            if val > new_max:
                new_max = val

        if new_max < best_max - 1e-13:
            best_max = new_max
            best_d = d

    return best_d

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = normalize_to_norm(a0, norm_target)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_stages = 8
    temps = [20.0, 12.0, 7.0, 4.0, 2.5, 1.5, 0.8, 0.4]

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 3)))

        lr = norm_target / (T * n)
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            exp_arg = (conv - max_conv) / T
            exp_arg = np.clip(exp_arg, -50, 50)
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum < 1e-15:
                break
            w = w / w_sum

            A_fft = rfft(a, fft_size)
            W_fft = rfft(w, fft_size)
            cross = irfft(W_fft * np.conj(A_fft), fft_size)

            grad = np.zeros(n)
            for i in range(n):
                k_lo = max(0, i)
                k_hi = min(n, i + n)
                if k_hi > k_lo:
                    grad[i] = 2.0 * np.dot(w[k_lo:k_hi], a[k_lo - i:k_hi - i])
                k_lo2 = max(0, i - n + 1)
                k_hi2 = min(n, i + 1)
                if k_hi2 > k_lo2:
                    grad[i] += 2.0 * np.dot(w[k_lo2:k_hi2], a[i - np.arange(k_lo2, k_hi2)])

            step = lr / (1.0 + 0.05 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm_target / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def multi_start_explore(best_seq, time_limit, rng, solver, verbose=True):

    for trial in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        noise_level = 0.02 * (trial + 1)
        perturbed = best_a + noise_level * rng.standard_normal(n_best)
        perturbed = np.maximum(perturbed, 0.0)
        s = perturbed.sum()
        if s < 1e-15:
            continue
        perturbed = perturbed / s
        perturbed = perturbed * np.sqrt(2.0 * n_best) / perturbed.sum()

        t_lp = alloc * 0.6
        a_out, _ = nesterov_lp_optimize(perturbed, t_lp, solver, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    Multi-start {trial}: improved to {sc:.6f}")

        remaining2 = time_limit - (time.monotonic() - t0)
        if remaining2 > 15:
            a_out2, _ = pair_cd(a_out, min(remaining2 * 0.3, 40.0), rng, verbose=False, window=20)
            sc2 = evaluate_safe(a_out2)
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = sc2
                best_a = a_out2.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov on parent length (50% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.50
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Pair CD polish (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Pair CD n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t2, rng, verbose=True, window=30)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 3: LSE gradient (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: LSE gradient n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t3, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 4: Multi-start exploration (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Multi-start n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 60:
        a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        a_out, _ = multi_start_explore(a_norm, t4, rng, solver, verbose=True)
        do_try_update(a_out, "multi_start")

    # Phase 5: Try alternate length (5% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 90:
        n_alt = n_base + 200
        t5 = remaining * 0.5
        print(f"\n=== Phase 5: Alternate length n={n_alt} (budget={t5:.0f}s) ===")
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        t5a = min(t5 * 0.6, remaining * 0.35)
        a_out_alt, _ = nesterov_lp_optimize(a_norm_alt, t5a, solver, verbose=False)
        do_try_update(a_out_alt, f"lp_alt_n{n_alt}")
        remaining2 = time_budget - (time.monotonic() - t0)
        if remaining2 > 30 and best_seq is not None and len(best_seq) == n_alt:
            a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
            a_out_alt2, _ = pair_cd(a_norm_alt2, min(remaining2 * 0.5, 50.0), rng, verbose=False, window=25)
            do_try_update(a_out_alt2, f"pcd_alt_n{n_alt}")

    # Final pair CD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10 and best_seq is not None:
        n_best = len(best_seq)
        t_final = min(remaining * 0.3, 60.0)
        print(f"\n=== Final Polish: n={n_best} (budget={t_final:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t_final, rng, verbose=False, window=50)
        do_try_update(a_out, "final_pcd")

    # Verify final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
