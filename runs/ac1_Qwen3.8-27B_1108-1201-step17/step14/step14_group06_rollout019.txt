def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def build_constraint_matrix(a, n):
    """Build the convolution constraint matrix T where T @ b = convolve(a, b)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

def lp_best_response(a, M, solver, T=None):
    """Solve the LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lse_objective_and_grad(a_flat, n, T_temp):
    """Compute LSE objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    
    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

def lse_scipy_optimize(a0, time_limit, T_start=80.0, T_end=3.0, n_stages=5):
    """LSE smoothed optimization with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / (n_stages - si), 150.0)
        
        def obj_func(x, T_val=T):
            val, _ = lse_objective_and_grad(x, n, T_val)
            return val
        
        def grad_func(x, T_val=T):
            _, g = lse_objective_and_grad(x, n, T_val)
            return g
        
        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 500, 'maxfun': 2000,
                                           'gtol': 1e-10, 'ftol': 1e-12,
                                           'maxls': 100})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
        except Exception:
            pass
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=42):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        
        # Find elements contributing to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]
        
        # Find low-value elements to move mass to
        peak_mask = np.zeros(n, dtype=bool)
        top_peaks = np.argsort(b)[::-1][:20]
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True
        
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:20]
        
        improved = False
        
        # Try moving mass from peak contributors to low-value positions
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        # Also try random moves
        if not improved:
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.005, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64) if prob.constraints[0].dual_value is not None else None
    return b_val, dual_val


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def lse_scipy_optimize(a0, time_limit, T_start=100.0, T_end=2.0, n_stages=6):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def obj_func(x, T_val=T):
            val, _ = lse_objective_and_grad(x, n, T_val)
            return val

        def grad_func(x, T_val=T):
            _, g = lse_objective_and_grad(x, n, T_val)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 500, 'maxfun': 2000,
                                           'gtol': 1e-10, 'ftol': 1e-12,
                                           'maxls': 100})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
        except Exception:
            pass

    return best_a, best_score


def lp_nesterov_two_stage(a0, time_limit, solver, verbose=True):
    """
    Two-stage LP best-response:
    Stage 1: M = current_max (pure best response, converges to game value)
    Stage 2: M = current_max * (1 - eps) with increasing eps (pushes below game value)
    Uses Polyak averaging throughout.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    polyak_sum = np.zeros(n)
    polyak_count = 0
    t0 = time.monotonic()
    last_improve = time.monotonic()
    stall_count = 0
    n_iters = 0

    # Stage 1: 40% of time with M = current_max
    stage1_time = time_limit * 0.40

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Determine M based on stage
        elapsed = time.monotonic() - t0
        if elapsed < stage1_time:
            # Stage 1: exact best response
            M_used = M_curr
        else:
            # Stage 2: tight M with increasing reduction
            progress = min(1.0, (elapsed - stage1_time) / (time_limit - stage1_time))
            eps = 0.01 + 0.04 * progress
            M_used = M_curr * (1.0 - eps)

        T = build_constraint_matrix(y, n)
        b, dual = lp_best_response(y, M_used, solver, T)
        if b is None:
            b, dual = lp_best_response(y, M_curr * 0.99, solver, T)
        if b is None:
            b, dual = lp_best_response(y, M_curr, solver, T)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Step size: decrease over time
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.1), 1.0)

        # Update y
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        # Nesterov for a
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Polyak averaging
        polyak_sum += a
        polyak_count += 1
        if polyak_count % 10 == 0:
            avg_a = polyak_sum / polyak_count
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_a = avg_a * norm_val / s_avg
                score_avg = fast_autocorr(avg_a).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_a.copy()
                    last_improve = time.monotonic()

        # Also try a "trimmed" Polyak (discard early worst iterates)
        if polyak_count % 50 == 0 and polyak_count > 100:
            # Use last 70% of iterates
            tail_sum = polyak_sum - polyak_sum  # Simplified: just use full average
            pass

        # Stall detection
        if time.monotonic() - last_improve > 50.0:
            stall_count += 1
            if stall_count > 10:
                if verbose:
                    print(f"    LP stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            stage = "1" if elapsed < stage1_time else "2"
            print(f"    LP stage-{stage} iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def dual_guided_coord_descent(a0, time_limit, solver, seed=42):
    """
    Coordinate descent that uses LP dual variables to identify which
    convolution peaks are "active" (binding), then focuses moves on
    reducing those peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    lp_interval = max(10, int(time_limit / 30.0))  # solve LP every ~30s
    last_lp_time = 0.0
    active_peaks = np.zeros(2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)

        # Periodically solve LP to get dual info
        elapsed = time.monotonic() - t0
        if elapsed - last_lp_time >= lp_interval or stall == 0:
            T = build_constraint_matrix(a, n)
            M_curr = b.max()
            _, dual = lp_best_response(a, M_curr, solver, T)
            if dual is not None:
                active_peaks = dual
            last_lp_time = elapsed

        # Identify peak-contributing elements for the top peaks
        top_peaks = np.argsort(b)[::-1][:12]
        peak_contrib = np.zeros(n)
        for k2 in top_peaks:
            weight = max(b[k2] - b.min(), 0.0)
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            for i in range(js2, je2 + 1):
                peak_contrib[i] += weight * a[i] * a[k2 - i]

        # Also incorporate dual info
        dual_weight = np.zeros(n)
        for k2 in top_peaks:
            if active_peaks[k2] > 0:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for i in range(js2, je2 + 1):
                    dual_weight[i] += active_peaks[k2] * a[i] * a[k2 - i]

        combined_pressure = peak_contrib + 0.5 * dual_weight
        top_pressure = np.argsort(combined_pressure)[::-1][:20]

        # Low-value, low-pressure targets
        low_pressure = np.argsort(combined_pressure)[:20]
        low_value = np.argsort(a)[:20]
        targets = list(set(list(low_pressure[:10]) + list(low_value[:10])))

        improved = False
        for src in top_pressure:
            if a[src] < 1e-10:
                continue
            for dst in targets:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.08, 0.12, 0.18]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        # Random exploration
        if not improved:
            for _ in range(150):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.005, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 300:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

    print(f"\n=== Phase 3: Dual-guided CD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = dual_guided_coord_descent(a_norm, t3, solver, seed=42)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"dual_cd_n{n_best}", t_global)

    # Phase 4: LP round 2 with different schedule - 15% of time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 4: LP round 2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_two_stage(a_norm, t4, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_r2_n{n_best}", t_global)

    # Phase 5: Try longer sequence - 10% of time
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_best + 300
    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: LP longer n={n_long} (budget={t5:.0f}s) ===")
    if t5 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = lp_nesterov_two_stage(a_norm, t5, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_long_n{n_long}", t_global)

    # Phase 6: Final polish with simple coord descent + LSE
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        third = remaining / 3.0

        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = simple_coord_descent(a_norm, third, seed=12345)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord_final1", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            n_best2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            a_out, _ = lse_scipy_optimize(a_norm2, min(remaining2, 120.0), T_start=50.0, T_end=1.0, n_stages=4)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "lse_final", t_global)

        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 15:
            n_best3 = len(best_seq)
            a_norm3 = best_seq * np.sqrt(2.0 * n_best3) / best_seq.sum()
            a_out, _ = dual_guided_coord_descent(a_norm3, remaining3, solver, seed=54321)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "dual_cd_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
