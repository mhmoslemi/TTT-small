def lp_best_response(a, M, solver, active_set=None):

    size = 2 * n - 1

    else:

        T = build_conv_matrix_rows(a, active_set)

    try:
        prob.solve(solver=solver, verbose=False)

        a = a * norm / s0

    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    active_set = None
    stall_count = 0
    stall_threshold = 80

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 200 == 0:
            tol = 0.03 + 0.01 * np.exp(-t / 1000.0)
            active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
            if len(active_set) < 3:
                active_set = np.where(conv_y > M_curr * 0.7)[0]
            if len(active_set) == 0:
                active_set = np.arange(2 * n - 1)

        b, active_set_new, duals = lp_best_response(y, M_curr, solver, active_set)

        if b is not None and active_set_new is not None and len(active_set_new) < 2 * n - 1:
            all_k = np.setdiff1d(np.arange(2 * n - 1), active_set_new)
            if len(all_k) > 0:
                T_check = build_conv_matrix_rows(y, all_k)
                violations = T_check @ b
                viol_idx = np.where(violations > M_curr * (1 + 1e-6))[0]
                if len(viol_idx) > 0:
                    new_k = all_k[viol_idx]
                    active_set = np.union1d(active_set_new, new_k)
                    if len(active_set) > 0.9 * (2 * n - 1):
                        active_set = None
                    b, active_set_new, duals = lp_best_response(y, M_curr, solver, active_set)

        if b is None:
            if t % 50 == 0:
                active_set = None
                b, active_set_new, duals = lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue

        active_set = active_set_new
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 1
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Restart at iter {t} ({elapsed:.0f}s)")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            for attempt in range(6):
                j = int(rng.integers(n))
                if j == i:
                    continue

                g = np.zeros(size)
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    src = np.arange(max(0, lo1 - i), hi1 - i)
                    g[lo1:hi1] -= a[src]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n + i)
                if hi2 > lo2:
                    src2 = n - 1 - np.arange(lo2, hi2)
                    valid = (src2 >= 0) & (src2 < n)
                    if any(valid):
                        g[lo2:hi2][valid] -= a[src2[valid]]

                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    src3 = np.arange(max(0, lo3 - j), hi3 - j)
                    g[lo3:hi3] += a[src3]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n + j)
                if hi4 > lo4:
                    src4 = n - 1 - np.arange(lo4, hi4)
                    valid4 = (src4 >= 0) & (src4 < n)
                    if any(valid4):
                        g[lo4:hi4][valid4] += a[src4[valid4]]

                candidates = [0.0, ai * 0.05, ai * 0.1, ai * 0.2, ai * 0.35, ai * 0.5, ai * 0.7, ai * 0.9, ai]
                best_d = 0.0
                best_val = conv.max()

                for d in candidates:
                    if d <= 0:
                        continue
                    new_max = (conv + d * g).max()
                    di = 2 * i
                    if di < size:
                        new_max = max(new_max, conv[di] - 2 * a[i] * d + d * d)
                    dj = 2 * j
                    if dj < size:
                        new_max = max(new_max, conv[dj] + 2 * a[j] * d + d * d)
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        s = a.sum()
                        if s > 1e-15:
                            a = a * norm / s
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 12))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2.0 * 2 * np.pi * x) + 0.15 * np.cos(5.0 * 2 * np.pi * x) + 0.05 * np.cos(9.0 * 2 * np.pi * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    elif choice == 7:
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.2 * np.cos(6 * np.pi * x))
    elif choice == 8:
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3
    elif choice == 9:
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
        a = a - a.min() + 0.1
    elif choice == 10:
        a = 0.6 * np.sin(np.pi * x) + 0.4 * np.sin(2 * np.pi * x) ** 2
    else:
        a = np.cos(np.pi * (x - 0.5)) ** 4 + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def perturb_structured(a, rng, scale=0.05):
    n = len(a)
    a_pert = a.copy()
    n_bumps = int(rng.integers(3, 10))
    for _ in range(n_bumps):
        center = int(rng.integers(0, n))
        width = int(rng.integers(n // 50, n // 10))
        amp = scale * rng.standard_normal()
        x = np.arange(n)
        bump = amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a_pert += bump
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate():

            best_seq = p / p.sum()

            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")

            return

        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global

        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 1: MW exploration n={n_base} ===")
    t1 = min(remaining * 0.08, 80.0)

    # === Phase 3: CD polish (100s) ===

    remaining = time_budget - (time.monotonic() - t_global)

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
