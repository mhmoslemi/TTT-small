    #   Either way, if 0 <= k-i < n, sg[i] = 2*a[k-i].
    #   If k-i is not in [0,n), sg[i] = 0.
    
    # Wait, that can't be right. Let me check with a simple example.
    # a = [1, 2, 3], n=3
    # conv[0] = a[0]*a[0] = 1
    # conv[1] = a[0]*a[1] + a[1]*a[0] = 4
    # conv[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0] = 3+4+3 = 10
    # conv[3] = a[1]*a[2] + a[2]*a[1] = 12
    # conv[4] = a[2]*a[2] = 9
    
    # ∂conv[1]/∂a[0] = ?
    # conv[1] = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]
    # ∂/∂a[0] = 2*a[1] = 4. Using formula: k=1, i=0, k-i=1, a[1]=2, so 2*a[1]=4. ✓
    
    # ∂conv[2]/∂a[1] = ?
    # conv[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0]
    # ∂/∂a[1] = 2*a[1] = 4. Using formula: k=2, i=1, k-i=1, a[1]=2, so 2*a[1]=4. ✓
    
    # ∂conv[3]/∂a[0] = ?
    # conv[3] = a[1]*a[2] + a[2]*a[1] = 2*a[1]*a[2]
    # ∂/∂a[0] = 0. Using formula: k=3, i=0, k-i=3, 3 >= n=3, so sg=0. ✓
    
    # Great! So the formula is simply:
    # sg[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            sg[i] = 2.0 * a[j]
    return sg

def compute_subgrad(a, k):
    """Compute subgradient of conv[k] w.r.t. a."""
    n = len(a)
    sg = np.zeros(n)
    # conv[k] = sum_j a[j]*a[k-j] for valid j
    # d/d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    j = k - np.arange(n)
    valid = (j >= 0) & (j < n)
    sg[valid] = 2.0 * a[j[valid]]
    return sg


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def build_conv_matrix_rows(a, k_indices):
    """Build rows of convolution matrix for specific k indices."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(k_indices), n))
    for row_idx, k in enumerate(k_indices):
        # (T @ b)_row = sum_j a[k-j] * b[j] for valid k-j
        # i.e., T[row, j] = a[k-j] if 0 <= k-j < n
        j = k - np.arange(n)
        valid = (j >= 0) & (j < n)
        T[row_idx, valid] = a[j[valid]]
    return T


def lp_best_response(a, M, solver, constraint_indices, b_warm=None):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in constraint_indices, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix_rows(a, constraint_indices)
    constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    if s < 1e-15:

def peak_surgery(a0, time_limit, rng, verbose=True):
    """
    Direct peak reduction: identify the peak lag, find the "worst" contributing pairs,
    and transfer mass to reduce the peak while being careful not to increase other lags.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_rounds = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_rounds += 1
        conv = fast_autocorr(a)
        cur_max = conv.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            break
        
        # Find top 3 peak lags
        top_k = np.argsort(-conv)[:3]
        
        for k_star in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find contributing pairs
            j = k_star - np.arange(n)
            valid = (j >= 0) & (j < n)
            
            if not np.any(valid):
                continue
            
            # Contribution of each i to conv[k_star]
            contrib = np.zeros(n)
            contrib[valid] = a[np.arange(n)[valid]] * a[j[valid]]
            
            # Total at this lag
            total = contrib.sum()
            if total < 1e-15:
                continue
            
            # Find max and min contributors
            valid_indices = np.where(valid)[0]
            max_i = valid_indices[np.argmax(contrib[valid_indices])]
            min_i = valid_indices[np.argmin(contrib[valid_indices])]
            
            if max_i == min_i:
                continue
            
            # Try to transfer mass from max_i to min_i
            a_max = a[max_i]
            a_min = a[min_i]
            
            if a_max < 1e-10:
                continue
            
            # Try several transfer amounts
            best_delta = 0.0
            best_new_max = cur_max
            
            # After transfer: a[max_i] -= d, a[min_i] += d
            # Effect on conv[k_star]: decreases by d*(a[min_i] + a[k_star - min_i] - a[max_i] - a[k_star - max_i])
            # (approximately, for the peak lag)
            # But also affects other lags!
            
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                d = a_max * frac
                if d > a_max:
                    continue
                
                a_try = a.copy()
                a_try[max_i] -= d
                if a_try[max_i] < 0:
                    continue
                a_try[min_i] += d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                
                conv_try = fast_autocorr(a_try)
                new_max = conv_try.max()
                
                if new_max < best_new_max - 1e-12:
                    best_new_max = new_max
                    best_delta = d
            
            if best_delta > 0:
                a[max_i] -= best_delta
                if a[max_i] < 0:
                    a[max_i] = 0
                a[min_i] += best_delta
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
        
        # Also try random pair transfers
        for _ in range(10):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if a[i] < 1e-10:
                continue
            
            # Transfer from i to j, but only if it reduces the peak
            for frac in [0.01, 0.03, 0.05]:
                d = a[i] * frac
                a_try = a.copy()
                a_try[i] -= d
                if a_try[i] < 0:
                    continue
                a_try[j] += d
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                conv_try = fast_autocorr(a_try)
                if conv_try.max() < cur_max - 1e-12:
                    a = a_try
                    cur_max = conv_try.max()
                    break
        
        if verbose and n_rounds % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS round={n_rounds} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def refined_coordinate_descent(a0, time_limit, rng, verbose=True):
    """
    Coordinate descent that optimizes each coordinate individually
    using the exact effect on the max convolution.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    for epoch in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        epoch_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue
            
            # Try scaling a[i] by various factors
            best_scale = 1.0
            best_val = best_score
            
            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                d = (scale - 1.0) * ai
                if d == 0:
                    continue
                a_try = a.copy()
                a_try[i] = ai * scale
                if a_try[i] < 0:
                    continue
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                conv_try = fast_autocorr(a_try)
                val = conv_try.max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_scale = scale
            
            if best_scale != 1.0:
                a[i] *= best_scale
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                cur = fast_autocorr(a).max()
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_a = a.copy()
                    epoch_improved = True
        
        if verbose and (epoch + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD epoch={epoch+1} ({elapsed:.0f}s): best={best_score:.10f}")
        
        if not epoch_improved:
            # Try one more epoch with smaller steps
            improved2 = False
            perm2 = rng.permutation(n)
            for idx in range(min(n, 500)):
                if time.monotonic() - t0 > time_limit:
                    break
                i = perm2[idx]
                ai = a[i]
                if ai < norm * 1e-8:
                    continue
                for scale in [0.95, 0.98, 1.02, 1.05]:
                    a_try = a.copy()
                    a_try[i] = ai * scale
                    if a_try[i] < 0:
                        continue
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try * norm / s_try
                    val = fast_autocorr(a_try).max()
                    if val < best_score - 1e-13:
                        a = a_try
                        best_score = val
                        best_a = a.copy()
                        improved2 = True
                        break
            if not improved2 and not epoch_improved:
                break
    
    return best_a, best_score


def mixed_optimize(a0, time_limit, solver, verbose=True, seed=0):
    """
    Combine mirror descent and LP in an alternating fashion.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Alternate: 200s mirror descent, 100s LP, repeat
    phase_times = [150, 100, 150, 100, 150, 100]
    phase_idx = 0
    phase_seed = seed
    
    total_elapsed = 0
    for phase_time in phase_times:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        
        t_phase = min(phase_time, remaining - 10)
        if t_phase < 30:
            t_phase = remaining - 10
        if t_phase < 20:
            break
        
        if phase_idx % 2 == 0:
            # Mirror descent phase
            print(f"    Mixed: MD phase ({t_phase:.0f}s)")
            a_out, sc = mirror_nesterov_optimize(a, t_phase, verbose=False, seed=phase_seed)
        else:
            # LP phase
            print(f"    Mixed: LP phase ({t_phase:.0f}s)")
            a_out, sc = lp_nesterov_optimize(a, t_phase, solver, verbose=False, seed=phase_seed)
        
        if a_out is not None:
            a_out = np.maximum(a_out, 0.0)
            s = a_out.sum()
            if s > 1e-15:
                a_out = a_out * norm / s
                sc_actual = fast_autocorr(a_out).max()
                if sc_actual < best_score:
                    best_score = sc_actual
                    best_a = a_out.copy()
                a = a_out
        else:
            sc_actual = fast_autocorr(a).max()
            if sc_actual < best_score:
                best_score = sc_actual
                best_a = a.copy()
        
        phase_seed += 1
        phase_idx += 1
    
    return best_a, best_score


def structured_init(n, rng, init_type=None):
    """Generate various structured initial sequences."""
    if init_type is None:
        init_type = int(rng.integers(0, 10))
    
    if init_type == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif init_type == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif init_type == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif init_type == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif init_type == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif init_type == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif init_type == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.sin(8 * np.pi * x)
    elif init_type == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.15 * np.cos(4 * np.pi * x))
    elif init_type == 8:
        # "Anti-peak" shape: high in middle, low at edges
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - 0.4 * (2 * x - 1) ** 4
    else:
        # Smooth random
        a = np.ones(n)
        for freq in [np.pi * 2, np.pi * 4, np.pi * 6]:
            a += 0.1 * rng.standard_normal() * np.sin(freq * np.linspace(0, 1, n))
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
