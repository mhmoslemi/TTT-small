    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / 30.0, n / (T * 2.0))
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 300 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    best_at_stall = best_score
    restart_threshold = 2000

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0

        stall_count += 1
        if stall_count > restart_threshold and n_iters > 500:
            t = 1
            y = avg_a / n_avg
            s_y2 = y.sum()
            if s_y2 > 1e-15:
                y = y * norm / s_y2
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP-Nesterov RESTART at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_lag_shave(a0, time_limit, seed=42):
    """
    Novel spectral peak shaving that targets specific autocorrelation lags.
    Instead of reducing top frequency bins, it identifies the worst autocorrelation
    lag and modifies the power spectrum to reduce the corresponding sinusoidal component.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    pad_len = 1 << (2 * n - 1).bit_length()
    freq_indices = np.arange(pad_len // 2 + 1)

    while time.monotonic() - t0 < time_limit:
        a_pad = np.zeros(pad_len)
        a_pad[:n] = a
        A = rfft(a_pad)
        power = np.abs(A) ** 2
        ac = irfft(power, pad_len)[:2 * n - 1]

        peak_idx = np.argmax(ac)
        peak_val = ac[peak_idx]

        if peak_val < best_score + 1e-10:
            stall += 1
            if stall > 300:
                break
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            continue

        # Target the peak lag: reduce the sinusoidal component at that frequency
        # The autocorrelation peak at lag k corresponds to a cos(2*pi*k*m/L) component in power
        k = peak_idx
        # The contribution to ac[k] from the power spectrum is:
        # ac[k] = (1/L) * sum_m power[m] * cos(2*pi*k*m/L)
        # To reduce ac[k], we subtract a multiple of cos(2*pi*k*m/L) from power
        cos_component = np.cos(2 * np.pi * k * freq_indices / pad_len)
        # Only modify non-DC bins
        cos_component[0] = 0.0
        if pad_len % 2 == 0 and k == pad_len // 2:
            cos_component[-1] = 0.0

        for beta in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            power_new = power - beta * peak_val * (cos_component - cos_component.mean())
            power_new = np.maximum(power_new, 0.0)
            if power_new.max() < 1e-15:
                continue
            # Reconstruct A with modified magnitude, same phase
            new_magnitude = np.sqrt(power_new)
            old_magnitude = np.abs(A)
            phase_mask = old_magnitude > 1e-15
            A_new = np.zeros_like(A)
            A_new[phase_mask] = new_magnitude[phase_mask] * (A[phase_mask] / old_magnitude[phase_mask])
            # For bins where old magnitude was ~0, set new magnitude
            zero_mask = ~phase_mask
            A_new[zero_mask] = new_magnitude[zero_mask].astype(complex)

            a_new_full = irfft(A_new, pad_len)
            a_new = a_new_full[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new * norm / s_new
            sc = fast_autocorr(a_new).max()
            if sc < best_score - 1e-14:
                a = a_new
                best_score = sc
                best_a = a.copy()
                stall = 0
                break

        if stall > 500:
            break

    return best_a, best_score


def spectral_flatten(a0, time_limit, seed=123):
    """
    Gerchberg-Saxton variant: iteratively compress the dynamic range of the
    power spectrum to reduce autocorrelation peaks.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    pad_len = 1 << (2 * n - 1).bit_length()

    alpha = 0.3  # compression exponent

    while time.monotonic() - t0 < time_limit:
        a_pad = np.zeros(pad_len)
        a_pad[:n] = a
        A = rfft(a_pad)
        magnitude = np.abs(A)
        phase = np.angle(A)

        # Compress dynamic range: new_mag = mag^alpha (for mag > 0)
        max_mag = magnitude.max()
        if max_mag < 1e-15:
            break
        normalized_mag = magnitude / max_mag
        compressed = normalized_mag ** alpha
        new_magnitude = compressed * max_mag

        # Preserve DC component (sum is fixed)
        new_magnitude[0] = magnitude[0]

        A_new = new_magnitude * np.exp(1j * phase)
        a_new_full = irfft(A_new, pad_len)
        a_new = a_new_full[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            stall += 1
            continue
        a_new = a_new * norm / s_new
        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-14:
            a = a_new
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Try reducing alpha (less compression)
            alpha *= 0.85
            if alpha < 0.05:
                alpha = 0.5
                # Random kick
                if a.sum() > 1e-10:
                    idx = rng.integers(n)
                    amt = a[idx] * 0.05
                    a[idx] -= amt
                    jdx = rng.integers(n)
                    a[jdx] += amt
                    a = np.maximum(a, 0.0)
                    a = a * norm / a.sum()
            if stall > 400:
                break

    return best_a, best_score


def targeted_peak_shave(a0, time_limit, seed=12345):
    """Targeted peak reduction via mass redistribution from peak-contributing terms."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find positions contributing least to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_cand = max(1, len(candidates_sorted) // 4)
            dst = candidates_sorted[rng.integers(n_cand)]

            for frac in [0.0003, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 1000:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization combining LP, spectral, and targeted methods."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        # Use a better initial shape: U-shaped (high at edges, low in middle)
        # This helps reduce the center autocorrelation peak
        best_seq = 0.3 + 0.7 * np.cos(np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    if t1 > 15:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [20.0, 5.0, 1.5, 0.5, 0.15, 0.05]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP Nesterov (main workhorse - allocate most time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.45, 800.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Novel spectral lag shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Spectral lag shave n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = spectral_lag_shave(a_norm, t3, seed=42)
        try_update(a_out, f"lagshave_n{n_best}")

    # === Phase 4: Spectral flatten (GS variant) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Spectral flatten n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = spectral_flatten(a_norm, t4, seed=123)
        try_update(a_out, f"flatten_n{n_best}")

    # === Phase 5: LP Nesterov polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP polish n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t5, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 6: Targeted peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Targeted shave n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_peak_shave(a_norm, t6, seed=777)
        try_update(a_out, f"targeted_n{n_best}")

    # === Phase 7: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n2 = len(best_seq) + 400
        t7 = min(remaining * 0.15, 200.0)
        if t7 > 40:
            print(f"\n=== Phase 7: LP at n={n2} (budget={t7:.0f}s) ===")
            init_long = interpolate_sequence(best_seq, n2)
            a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
            a_out, _ = lp_nesterov(a_norm, t7, solver, verbose=True)
            try_update(a_out, f"lp_long_n{n2}")

    # === Phase 8: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t8 = min(remaining * 0.5, 150.0)
        if t8 > 20:
            print(f"\n=== Phase 8: Final LP n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t8, solver, verbose=False)
            try_update(a_out, "lp_final")

    # === Phase 9: Final spectral lag shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t9 = min(remaining * 0.4, 80.0)
        if t9 > 15:
            print(f"\n=== Phase 9: Final spectral lag shave n={n_best} (budget={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = spectral_lag_shave(a_norm, t9, seed=99999)
            try_update(a_out, "final_lagshave")

    # === Phase 10: Final targeted shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 10:
            print(f"\n=== Phase 10: Final targeted shave n={n_best} (budget={t10:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = targeted_peak_shave(a_norm, t10, seed=31415)
            try_update(a_out, "final_targeted")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
