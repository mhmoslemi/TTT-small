            a_new = (a_new + a_new[::-1]) / 2.0
            a_new = a_new * norm / a_new.sum()

        y = y_new
        a = a_new
        n_iters += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:

            rng = np.random.default_rng(n_iters)
            a_new = np.maximum(a + rng.normal(0, 0.02 * a.mean(), n), 0)
            a_new = (a_new + a_new[::-1]) / 2.0
            a_new = a_new * norm / a_new.sum()
            y = a_new.copy()
            prev_y = y.copy()
            stall = 0
    
    return best_a, best_score
