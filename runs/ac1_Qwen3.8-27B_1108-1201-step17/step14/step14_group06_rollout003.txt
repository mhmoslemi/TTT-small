def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)

def lp_best_response(a, M, solver, T=None):

def lp_best_response_pg(a, M, n_iters=50, alpha=None):
    """Projected gradient fallback for LP best response.
    Solves: max sum(b) s.t. T@b <= M, b >= 0
    Using a simple multiplicative update.
    """
    n = len(a)
    T = build_constraint_matrix(a, n)
    b = np.ones(n) * M / (T.sum() / n + 1e-15)
    b = np.maximum(b, 0)
    if alpha is None:
        alpha = 2.0 / (T.max() + 1e-15)
    for _ in range(n_iters):
        violations = np.maximum(T @ b - M, 0)
        grad = T.T @ violations
        b_new = b * np.maximum(1.0 - alpha * grad, 0.001)
        b_new = np.maximum(b_new, 0.0)
        s = b_new.sum()
        if s > 0:
            b = b_new
        else:
            break
    return b


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    np.clip(shifted, -500, 500, out=shifted)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def power_sum_obj_grad(a_flat, n, p):
    """Power-sum approximation: (sum b_k^p)^(1/p) approximates max(b_k)."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    b_p = np.power(np.maximum(b, 0), p)
    sum_b_p = b_p.sum()
    if sum_b_p < 1e-30:
        return 1e10, np.zeros(n)
    ps_val = sum_b_p ** (1.0 / p)
    w = b_p / sum_b_p
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    obj_val = 2.0 * n * ps_val / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

        b_sum = b.sum()

def gradient_structured_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent using the gradient structure:
    d(a*a)[k]/da[j] = 2*a[k-j]
    So moving mass from j to j' changes peak by ~2*(a[k*-j] - a[k*-j'])*delta
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        M = b.max()

        # Contributions to peak: a[j] * a[k_peak - j]
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]

        # Top contributors (should decrease)
        top_contrib = idx[np.argsort(contrib)[::-1][:20]]

        # Find positions where a[k_peak - j] is small (good targets to increase)
        # Moving from j to j' changes peak by 2*delta*(a[k_peak-j'] - a[k_peak-j])
        # We want a[k_peak-j'] < a[k_peak-j], so j' should have small a[k_peak-j']
        kp_ref = np.array([a[k_peak - j] for j in range(n) if 0 <= k_peak - j < n])
        kp_ref_full = np.zeros(n)
        valid = np.array([1 if 0 <= k_peak - j < n else 0 for j in range(n)])
        for j in range(n):
            if 0 <= k_peak - j < n:
                kp_ref_full[j] = a[k_peak - j]

        # Good targets: small kp_ref_full, small current value
        target_score = kp_ref_full + 0.1 * a
        sorted_targets = np.argsort(target_score)
        good_targets = sorted_targets[:20]

        improved = False
        for src in top_contrib:

                continue

                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:

                        best_score = score

                if improved:

            if improved:
                break

        if not improved:
            # Random perturbation
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.08)

                a_trial = a.copy()
                a_trial[i] -= amt

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)

        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov with tight M (main workhorse, 35% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest tight M n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.35
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_tight_M=True)
    best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_tight_n{n_best}", t_global)

    # Phase 2: LSE gradient with aggressive annealing (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE gradient n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=100.0, T_end=1.0, n_stages=6)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lse_n{n_best}", t_global)

    # Phase 3: Power-sum gradient (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Power-sum gradient n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = psum_scipy_optimize(a_norm, t3, p_start=20, p_end=300, n_stages=4)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"psum_n{n_best}", t_global)

    # Phase 4: Multi-length exploration (25% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Multi-length exploration (budget={remaining:.0f}s) ===")

    candidate_lengths = []
    for nl in [800, 1000, 1200, 1400, 1600, 1800, 2000, 2400, 2800, 3200]:
        if nl != n_best:
            candidate_lengths.append(nl)

    # Sort by proximity to current best length (try close ones first)
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    time_per_length = remaining / max(len(candidate_lengths), 1)
    for nl in candidate_lengths:
        if time.monotonic() - t_global > time_budget:
            break
        remaining_now = time_budget - (time.monotonic() - t_global)
        if remaining_now < 30:
            break
        t_alloc = min(time_per_length * 0.9, remaining_now * 0.6, 180.0)
        if t_alloc < 20:
            continue

        print(f"  Trying n={nl} (alloc={t_alloc:.0f}s)")
        init_ln = interpolate_sequence(best_seq, nl)
        a_norm_ln = init_ln * np.sqrt(2.0 * nl) / init_ln.sum()

        # Quick LP optimization
        a_out, _ = nesterov_lp_optimize(a_norm_ln, t_alloc * 0.6, solver, verbose=False, use_tight_M=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_n{nl}", t_global)

        # Quick LSE if time permits
        remaining_now = time_budget - (time.monotonic() - t_global)
        if remaining_now > 30 and a_out is not None:
            t_lse = min(remaining_now * 0.3, 60.0)
            if t_lse > 15:
                a_norm_ln2 = a_out * np.sqrt(2.0 * nl) / a_out.sum()
                a_out2, _ = lse_scipy_optimize(a_norm_ln2, t_lse, T_start=50.0, T_end=2.0, n_stages=3)
                best_seq, best_score = attempt_update((best_seq, best_score), a_out2, f"lse_n{nl}", t_global)

    # Phase 5: Gradient-structured coordinate descent on best (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Grad coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_structured_coord_descent(a_norm, t5, seed=42)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"gcoord_n{n_best}", t_global)

    # Phase 6: Final polish - alternate LP and LSE (5% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        # LP round
        t6a = remaining * 0.4
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6a, solver, verbose=False, use_tight_M=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "polish_lp", t_global)

        # LSE round
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            t6b = remaining2 * 0.4
            n_best2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            a_out2, _ = lse_scipy_optimize(a_norm2, t6b, T_start=30.0, T_end=0.5, n_stages=4)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out2, "polish_lse", t_global)

        # Coord descent
        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 15:
            n_best3 = len(best_seq)
            a_norm3 = best_seq * np.sqrt(2.0 * n_best3) / best_seq.sum()
            a_out3, _ = gradient_structured_coord_descent(a_norm3, remaining3 * 0.8, seed=12345)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out3, "polish_coord", t_global)

    # Final: try symmetry on best
    if best_seq is not None:
        sym = 0.5 * (best_seq + best_seq[::-1])
        s_sym = sym.sum()
        if s_sym > 0.01:
            sym = sym / s_sym
            sc_sym = evaluate_safe(sym)
            if np.isfinite(sc_sym) and sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(f"  *** SYMMETRY IMPROVEMENT: {sc_sym:.6f}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
