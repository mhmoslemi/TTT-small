def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution of a with itself)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_objective_and_grad(a_flat, n, T):
    """
    Compute the LSE-smoothed objective and gradient.
    Objective: 2n * LSE_T(conv(a,a)) / (sum(a))^2
    where LSE_T(c) = max(c) + (1/T)*log(sum(exp(T*(c-max(c)))))
    """
    a = np.maximum(a_flat.reshape(-1), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    c = fast_autocorr(a)
    max_c = c.max()
    if max_c < 1e-15:
        return 1e10, np.zeros(n)
    
    # LSE computation (numerically stable)
    shifted = T * (c - max_c)
    # For very negative values, exp will underflow to 0, which is fine
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)
    LSE = max_c + np.log(sum_exp) / T
    
    # Weights for gradient
    w = exp_shifted / sum_exp
    
    # Gradient of LSE w.r.t. a: T^T w
    # (T^T w)_j = sum_i a[i] * w[i+j] for i=0..n-1, with w[k]=0 for k outside [0,2n-2]

    conv_len = n + (2*n - 1) - 1  # = 3n - 2

    W = rfft(w, fft_size)
    cross_full = irfft(FA * W, fft_size)
    grad_lse = cross_full[n-1 : 2*n-1]
    
    # Full objective gradient
    obj = 2.0 * n * LSE / (s * s)
    grad = (2.0 * n / (s * s)) * grad_lse - (4.0 * n * LSE) / (s * s * s)
    
    return obj, grad
