def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, w):
    """Compute gradient of T*log(sum exp(c/T)) where w = softmax(c/T).
    grad_i = 2 * sum_k w_{i+k} * a_k = 2 * cross_corr(w, a)[i]
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * cross


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_optimize(a0, time_limit, verbose=True):
    """Minimize smooth approximation of max(conv(a,a)) using accelerated projected gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Temperature schedule
    temps = np.array([20.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.6, 0.3, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.012, 0.008, 0.005, 0.003, 0.002, 0.001])

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_steps = max(30, min(500, int(remaining * 3)))

        y = a.copy()
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(y)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            grad = compute_smooth_grad(y, w)

            # Adaptive step size
            step = norm / (n * T * (1.0 + 0.05 * it))
            step = min(step, norm / (n * 0.5))

            a_new = y - step * grad
            a_new = project_simplex(a_new, norm)

            # Nesterov update
            alpha = 2.0 / (it + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = project_simplex(y_new, norm)

            if it == 0:
                a_curr = a_new.copy()
            else:
                momentum = (it - 1.0 + 1.0) / (it + 2.0)
                a_curr = y_new + momentum * (y_new - y)
                a_curr = project_simplex(a_curr, norm)

            y = y_new
            a = a_curr

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if verbose and int(T * 100) % 3 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth T={T:.3f} step={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix_sparse(a):
    """Build the convolution matrix T where (T @ b)[j] = (a*b)[j] (full conv)."""
    n = len(a)
    size = 2 * n - 1
    # T is size x n, T[j, i] = a[j-i] if 0 <= j-i < n, else 0
    # Use scipy sparse or just build dense for moderate n
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T


def lp_best_response(a, M, solver, active_only=None):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix_sparse(a)

    if active_only is not None:
        T_use = T[active_only, :]
    else:
        T_use = T

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_use @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and constraint generation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Constraint generation: use only top constraints
        conv_y = fast_autocorr(y)
        threshold = 0.85 * M_curr
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 5:
            active = np.where(conv_y >= 0.5 * M_curr)[0]
        if len(active) < 3:
            active = np.arange(len(conv_y))

        b = lp_best_response(y, M_curr, solver, active_only=active)
        if b is None:
            # Fallback: use all constraints
            b = lp_best_response(y, M_curr, solver, active_only=None)
            if b is None:
                stuck_count += 1
                if stuck_count > 30:
                    break
                continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_greedy(a0, time_limit, rng, window=50, verbose=True):
    """Pair coordinate descent with greedy pair selection (prioritize peak-contributing pairs)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Find peak positions
        peak_threshold = 0.95 * conv.max()
        peak_positions = np.where(conv >= peak_threshold)[0]

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # Compute contribution of a[i] to the peak
            # a[i] contributes a[i]*a[j-i] to conv[j] for valid j-i
            # and a[i]*a[i-j] to conv[j] for valid i-j (i.e., conv[n-1 + (i-j)] in reversed)
            # For simplicity, just try moving to nearby positions
            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Quick evaluation: move small amount from i to j
                    # and check if max decreases
                    test_d = a[i] * 0.3
                    a_test = a.copy()
                    a_test[i] -= test_d
                    a_test[j] += test_d
                    s_test = a_test.sum()
                    a_test = a_test * norm / s_test
                    conv_test = fast_autocorr(a_test)
                    if conv_test.max() < best_score - 1e-12:
                        a = a_test.copy()
                        conv = conv_test
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        # Try smaller step
                        test_d2 = a[i] * 0.1
                        a_test2 = a.copy()
                        a_test2[i] -= test_d2
                        a_test2[j] += test_d2
                        a_test2 = a_test2 * norm / a_test2.sum()
                        conv_test2 = fast_autocorr(a_test2)
                        if conv_test2.max() < conv.max() - 1e-12:
                            a = a_test2.copy()
                            conv = conv_test2
                            if conv.max() < best_score:
                                best_score = conv.max()
                                best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD-Greedy pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_standard(a0, time_limit, rng, window=40, verbose=True):
    """Standard pair coordinate descent with line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i or a[i] < norm * 1e-10:
                        continue

                    # Binary search for best delta
                    lo, hi = 0.0, a[i] * 0.99
                    best_d = 0.0
                    best_val = conv.max()
                    for _ in range(8):
                        mid = (lo + hi) / 2
                        if mid <= 0:
                            lo = mid
                            continue
                        a_t = a.copy()
                        a_t[i] -= mid
                        a_t[j] += mid
                        a_t = a_t * norm / a_t.sum()
                        conv_t = fast_autocorr(a_t)
                        if conv_t.max() < best_val - 1e-12:
                            best_val = conv_t.max()
                            best_d = mid
                            lo = mid
                        else:
                            hi = mid

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                        if conv.max() < best_score - 1e-12:
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * (np.sqrt(2.0 * new_n) / new_n)
    return f_interp * np.sqrt(2.0 * new_n) / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov (45% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.45
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # === Phase 2: Smooth gradient optimization (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Smooth grad n={n_c} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"smooth_n{n_c}")

    # === Phase 3: Pair CD standard (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Pair CD n={n_c} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = pair_cd_standard(a_norm, t3, rng, window=40, verbose=True)
        do_try_update(a_out, f"pcd_n{n_c}")

    # === Phase 4: LP Nesterov round 2 (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_c} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_c}")

    # === Phase 5: Pair CD greedy (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Pair CD Greedy n={n_c} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = pair_cd_greedy(a_norm, t5, rng, window=30, verbose=True)
        do_try_update(a_out, f"pcd_greedy_n{n_c}")

    # === Phase 6: Smooth + Pair CD on perturbed restart (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_c = len(best_seq)
    t6 = remaining * 0.05
    print(f"\n=== Phase 6: Diversified restart n={n_c} (budget={t6:.0f}s) ===")
    if t6 > 30:
        # Perturb the best solution
        a_pert = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        noise = rng.normal(0, 0.1 * a_pert.std(), n_c)
        a_pert = a_pert + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert * np.sqrt(2.0 * n_c) / a_pert.sum()
        # Quick smooth optimization
        t6a = t6 * 0.5
        a_out, _ = smooth_optimize(a_pert, t6a, verbose=False)
        do_try_update(a_out, "restart_smooth")
        remaining6 = time_budget - (time.monotonic() - t_global)
        if remaining6 > 10 and best_seq is not None:
            n_r = len(best_seq)
            a_norm_r = best_seq * np.sqrt(2.0 * n_r) / best_seq.sum()
            a_out_r, _ = pair_cd_standard(a_norm_r, min(remaining6 * 0.5, 30.0), rng, window=60, verbose=False)
            do_try_update(a_out_r, "restart_pcd")

    # === Final: try a couple of alternate lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        n_base_final = len(best_seq)
        for n_alt in [n_base_final + 100, n_base_final - 100]:
            if n_alt < 100:
                continue
            remaining_sub = time_budget - (time.monotonic() - t_global)
            if remaining_sub < 30:
                break
            print(f"\n=== Alternate length n={n_alt} (budget={remaining_sub * 0.4:.0f}s) ===")
            t_alt = min(remaining_sub * 0.4, 120.0)
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_out_alt, _ = nesterov_lp(init_alt, t_alt, solver, verbose=False)
            do_try_update(a_out_alt, f"alt_n{n_alt}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
