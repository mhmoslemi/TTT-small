def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)
    return obj


def lse_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()

def spectral_init(n, rng):
    """Initialize using spectral domain considerations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        # Cosine-based (smooth, low sidelobe)
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        # Raised cosine
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 2:
        # Gaussian
        center = n / 2.0
        sigmas = [n / 3.0, n / 4.0, n / 5.0]
        sigmoid = rng.choice(len(sigmas))
        idx = np.arange(n) - center
        a = np.exp(-0.5 * (idx / sigmas[sigmoid]) ** 2)
    elif choice == 3:
        # Sum of gaussians
        centers = rng.uniform(0.2, 0.8, size=3) * n
        sigmas = rng.uniform(n / 8.0, n / 4.0, size=3)
        a = np.zeros(n)
        idx = np.arange(n)
        for c, s in zip(centers, sigmas):
            a += np.exp(-0.5 * ((idx - c) / s) ** 2)
    elif choice == 4:
        # Parabolic (polynomial)
        x = np.linspace(-1, 1, n)
        a = (1 - x ** 2) ** 2
    else:
        # Sinusoidal window
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t2 = min(remaining * 0.15, 180.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_cd(a_norm, t2, seed=99)
        do_try_update(a_out, f"subgrad_cd_n{n_best}")

    # Phase 3: LSE optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t3, T_start=50.0, T_end=0.5, n_stages=8)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 4: LP with spectral init (different starting point)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP spectral init n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 15:
        spec_init = spectral_init(n_best, rng)
        # Blend with current best
        blend = 0.5 * spec_init + 0.5 * best_seq
        blend = np.maximum(blend, 0)
        a_norm = blend * np.sqrt(2.0 * n_best) / blend.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=False, use_line_search=True)
        do_try_update(a_out, f"lp_spec_n{n_best}")

    # Phase 5: Noisy restart LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP noisy restart n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        noise_scale = 0.05 * best_seq.mean()
        a_noisy = best_seq + rng.normal(0, noise_scale, n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_norm = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False, use_line_search=True)
        do_try_update(a_out, f"lp_noisy_n{n_best}")

    # Phase 6: Another subgradient CD pass
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Subgrad CD pass 2 n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_cd(a_norm, t6, seed=12345)
        do_try_update(a_out, f"cd_pass2_n{n_best}")

    # Phase 7: Low-temperature LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 120.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t7, T_start=20.0, T_end=0.3, n_stages=6)
        do_try_update(a_out, f"lse_polish_n{n_best}")

    # Phase 8: Final CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t8 = remaining * 0.5
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_cd(a_norm, t8, seed=77777)
        do_try_update(a_out, "cd_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
