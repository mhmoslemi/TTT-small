def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Main LP (most of the time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.55, 900.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP-MinMax n={n_base} K=30 (budget={t1:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_loop(a_norm, t1, solver, K=30, verbose=True)
        try_update(a_out, f"lp1_K30_n{n_base}")

    # === Phase 2: LP with larger K (more constraints, slower but more accurate) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-MinMax n={n_best} K=80 (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_loop(a_norm, t2, solver, K=80, verbose=True)
        try_update(a_out, f"lp2_K80_n{n_best}")

    # === Phase 3: Peak coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Peak descent n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_coordinate_descent(a_norm, t3, seed=42)
        try_update(a_out, f"peak_desc_n{n_best}")

    # === Phase 4: LP polish after descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: LP polish n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_loop(a_norm, t4, solver, K=50, verbose=False)
        try_update(a_out, f"lp_polish_n{n_best}")

    # === Phase 5: Try longer n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 300
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP at n={n2} (budget={t5:.0f}s) ===")
        init_long = interpolate_seq(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_loop(a_norm, t5, solver, K=30, verbose=False)
        try_update(a_out, f"lp_long_n{n2}")

    # === Phase 6: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.05, 60.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Final LP n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_loop(a_norm, t6, solver, K=50, verbose=False)
        try_update(a_out, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
