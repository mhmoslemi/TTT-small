"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact 1D coordinate descent (vectorized golden-section-like search)
2. Gradient-based updates with Nesterov momentum for the max-subgradient
3. Multi-scale exploration (different lengths, interpolated transfers)
4. Simulated annealing coordinate swaps for escaping local minima
Key idea: exact per-coordinate optimization is cheaper and more effective
than heuristic moves, combined with momentum-based gradient steps for
faster convergence between CD passes.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))

def normalize(a, target_sum=None):
    """Normalize to non-negative with given target sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * 0.01
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a * target_sum / s

def exact_cd(a, time_limit, max_passes=5000, verbose=True, x_grid_size=60):
    """Exact coordinate descent: for each coordinate, find optimal value."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    S = a.sum()
    b = fast_conv(a)
    current_score = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    x_max_base = S / n * 3.0  # base upper bound for x
    
    for pass_num in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        S = a.sum()
        
        pass_improved = False
        for i in range(n):
            ai = a[i]
            S_base = S - ai
            
            # Build coupling vector g[k] = a[k-i] for valid k
            g = np.zeros(2 * n - 1)
            start = i
            end = min(i + n, 2 * n - 1)
            g[start:end] = a[:end - start]
            k_diag = 2 * i if 2 * i < 2 * n - 1 else -1
            
            # Range of x
            x_max = max(ai * 3.0, x_max_base, 0.01)
            xs = np.linspace(0.0, x_max, x_grid_size)
            
            # Vectorized: b_new[j, k] = b[k] + 2*(xs[j]-ai)*g[k] + diag correction
            delta = 2.0 * (xs[:, None] - ai) * g[None, :]
            b_new = b[None, :] + delta
            if k_diag >= 0:
                b_new[:, k_diag] += xs**2 - ai**2
            
            S_new = S_base + xs
            valid = S_new > 0.01
            if not valid.any():
                continue
            
            max_b_new = b_new.max(axis=1)
            f_vals = 2.0 * n * max_b_new / S_new**2
            f_vals[~valid] = np.inf
            
            best_idx = np.argmin(f_vals)
            best_x = xs[best_idx]
            best_f = f_vals[best_idx]
            
            if best_f < current_score - 1e-12:
                a[i] = best_x
                S = S_base + best_x
                current_score = best_f
                pass_improved = True
        
        if not pass_improved:
            if verbose and pass_num > 0:
                print(f"    CD converged at pass {pass_num}, score={current_score:.10f}")
            break
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        if verbose and (pass_num + 1) % 50 == 0:
            print(f"    CD pass {pass_num+1}: score={current_score:.10f}")
    
    # Final polish: refine with smaller grid around current values
    a = best_a
    return a, compute_score(a)

def gradient_nesterov(a, time_limit, verbose=True):
    """Gradient-based optimization with Nesterov momentum."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    y = a.copy()
    velocity = np.zeros(n)
    momentum = 0.9
    lr = 0.1
    lr_decay = 0.999
    
    for t in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        y_new = y + momentum * velocity
        
        b = fast_conv(y_new)
        S = y_new.sum()
        if S < 0.01:
            break
        B = b.max()
        k_star = np.argmax(b)
        
        # Gradient of B w.r.t. a[i]: a[k*-i] + a[i-k*]
        # (with 0 if out of range)
        grad_B = np.zeros(n)
        k = k_star
        # a[k-i] for i in range: valid when 0 <= k-i < n, i.e., max(0,k-n+1) <= i <= min(n-1,k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        grad_B[i_start:i_end+1] += y_new[k - np.arange(i_start, i_end+1)]
        # a[i-k]: valid when 0 <= i-k < n, i.e., k <= i <= k+n-1, intersect [0,n-1]
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            grad_B[i_start2:i_end2+1] += y_new[np.arange(i_start2, i_end2+1) - k]
        
        # Full gradient: 2n/S^2 * (grad_B - 2B/S)
        grad = 2.0 * n / S**2 * (grad_B - 2.0 * B / S)
        
        # Update
        velocity = momentum * velocity - lr * grad
        y = y_new + velocity
        y = np.maximum(y, 0.0)
        y = normalize(y)
        
        lr *= lr_decay
        
        score = compute_score(y)
        if score < best_score:
            best_score = score
            best_a = y.copy()
        
        if verbose and (t + 1) % 500 == 0:
            print(f"    GN iter {t+1}: score={best_score:.10f}, lr={lr:.4f}")
    
    return best_a, best_score

    a = np.maximum(a.copy(), 0.0)
    S = a.sum()
    b = fast_conv(a)
    current_score = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()
    best_score = current_score

        if time.monotonic() - t0 > time_limit:

        S_base = S  # will update as we go

            g[start:end] = a[:end - start]
            k_diag = 2 * i if 2 * i < 2 * n - 1 else -1

            # Range of x
            x_max = max(ai * 3.0, S / n * 3.0, 0.01)
            xs = np.linspace(0.0, x_max, x_grid_size)
            
            # b_new[j, k] = b[k] + 2*(xs[j]-ai)*g[k] + diag
            delta = 2.0 * (xs[:, None] - ai) * g[None, :]
