def _subgradient_polish(a0, time_limit, verbose=True):
    """Exponentiated gradient (mirror descent) with subgradient of max autocorrelation."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = _normalize(np.asarray(a0, dtype=np.float64), norm_val)
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 0.1  # initial step size
    eta_bb = eta  # Barzilai-Borwein step
    a_prev = a.copy()
    g_prev = None
    stall = 0
    last_improve = t0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = _fast_autocorr(a)
        M = b.max()
        
        # Find all lags achieving (approximately) the max
        threshold = M * 0.9999
        peak_lags = np.where(b >= threshold)[0]
        
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(b)])
        
        # Compute subgradient: avg of gradients at peak lags
        grad = np.zeros(n)
        for k in peak_lags:
            # db[k]/da[m] = 2*a[k-m] if 0 <= k-m < n
            valid = (k >= 0) & (k < 2*n - 1)
            if valid:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                for m in range(lo, hi + 1):
                    grad[m] += 2.0 * a[k - m]
        grad /= len(peak_lags)
        
        # Barzilai-Borwein step size
        if g_prev is not None:
            s_vec = a - a_prev
            y_vec = grad - g_prev
            s_norm = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            if s_norm > 1e-20 and abs(sy) > 1e-20:
                eta_bb = s_norm / abs(sy)
                eta_bb = np.clip(eta_bb, 0.001, 1.0)
        
        # Exponentiated gradient update
        a_new = a * np.exp(-eta_bb * grad / M)
        a_new = _normalize(a_new, norm_val)
        
        score = _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            last_improve = time.monotonic()
            stall = 0
            a = a_new
        else:
            # Try smaller step
            a_trial = a * np.exp(-eta_bb * 0.1 * grad / M)
            a_trial = _normalize(a_trial, norm_val)
            score_trial = _fast_autocorr(a_trial).max()
            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()
                last_improve = time.monotonic()
                a = a_trial

        g_prev = grad.copy()

            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration and adaptive line search
2. Exponentiated gradient (mirror descent) with subgradient of max autocorrelation
3. Gradient-guided pairwise coordinate descent for final polish
4. Multi-phase time allocation with adaptive restarts
Key improvements: cleaner subgradient method with BB step, more aggressive
pairwise moves targeting peak contributors, better LP warm-starting, and
multi-scale exploration.
"""

import numpy as np

from scipy.fft import rfft, irfft

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _eval_safe(a):

def _normalize(a, norm_val):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), norm_val / len(a))
    return a * norm_val / s


def _lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[:end - start]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver,
                   **({'eps_abs': 1e-8, 'eps_rel': 1e-8} if solver == cp.CLARABEL else {}))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = _normalize(a0, norm_val)
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0
    rng_local = np.random.default_rng(12345)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        g = b * norm_val / b.sum()
        n_iters += 1

        alpha_base = 2.0 / (t + 2.0)
        best_alpha = alpha_base
        cand_base = _normalize((1 - alpha_base) * y + alpha_base * g, norm_val)
        best_obj = _fast_autocorr(cand_base).max()
        for mult in [0.5, 1.5, 2.0, 3.0, 5.0]:
            trial_alpha = min(alpha_base * mult, 1.0)
            cand = _normalize((1 - trial_alpha) * y + trial_alpha * g, norm_val)
            sc = _fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial_alpha
        alpha = best_alpha

        y_new = _normalize((1.0 - alpha) * y + alpha * g, norm_val)
        if t == 1:
            a_new = y_new.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = _normalize(a_new, norm_val)

        y = y_new
        a = a_new

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
            stall = 0
        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
            stall = 0

        avg_a += a
        n_avg += 1
        if n_avg % 7 == 0:
            avg_check = _normalize(avg_a / n_avg, norm_val)
            score_avg = _fast_autocorr(avg_check).max()
            if score_avg < best_score:

        if time.monotonic() - last_improve > 90:

                a = y.copy()

                y = _normalize(best_a + noise, norm_val)
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _subgrad_step(a, b, norm_val, eta):
    """One exponentiated-gradient step given current a and autocorrelation b."""
    n = len(a)
    M = b.max()
    if M < 1e-15:
        return a
    threshold = M * (1.0 - 1e-6)
    peak_lags = np.where(b >= threshold)[0]
    if len(peak_lags) == 0:
        peak_lags = np.array([np.argmax(b)])
    grad = np.zeros(n)
    for k in peak_lags:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        valid_idx = np.arange(lo, hi + 1)
        grad[valid_idx] += 2.0 * a[k - valid_idx]
    grad /= len(peak_lags)
    scale = eta * M
    ratio = np.exp(-scale * grad / (M + 1e-30))
    a_new = a * ratio
    return _normalize(a_new, norm_val)


def _subgradient_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = _normalize(a0, norm_val)
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 0.5
    a_prev = a.copy()
    grad_prev = None
    stall = 0
    last_improve = t0

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        b = _fast_autocorr(a)
        M = b.max()
        if M < 1e-14:
            break

        threshold = M * (1.0 - 1e-6)
        peak_lags = np.where(b >= threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(b)])
        grad = np.zeros(n)
        for k in peak_lags:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            valid_idx = np.arange(lo, hi + 1)
            grad[valid_idx] += 2.0 * a[k - valid_idx]
        grad /= len(peak_lags)

        # BB step size
        if grad_prev is not None:
            s_vec = a - a_prev
            y_vec = grad - grad_prev
            s_norm = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            if s_norm > 1e-20 and abs(sy) > 1e-20:
                eta_new = s_norm / abs(sy)
                eta_new = np.clip(eta_new, 0.01, 5.0)
                eta = 0.5 * eta + 0.5 * eta_new

        # Try multiple step sizes
        best_trial = None
        best_trial_score = best_score
        for mult in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
            a_trial = _subgrad_step(a, b, norm_val, eta * mult)
            sc = _fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = a_trial

        if best_trial is not None:
            a = best_trial
            best_a = a.copy()
            best_score = best_trial_score
            last_improve = time.monotonic()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break

        a_prev = a.copy()
        grad_prev = grad.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")

def _try_pairwise(a, norm_val, best_score, src, dst, frac):
    """Try moving a fraction of a[src] to a[dst]. Returns (new_a, new_score, improved)."""
    amt = a[src] * frac
    if amt < 1e-15:
        return a, best_score, False
    trial = a.copy()
    trial[src] -= amt
    trial[dst] += amt
    trial = np.maximum(trial, 0.0)
    s_t = trial.sum()
    if s_t < 1e-10:
        return a, best_score, False
    trial = trial * norm_val / s_t
    score = _fast_autocorr(trial).max()
    if score < best_score - 1e-14:
        return trial, score, True
    return a, best_score, False

def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _structured_init(n, rng):

    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.2 * np.sin(6.0 * np.pi * x) + 0.1 * np.cos(10.0 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
