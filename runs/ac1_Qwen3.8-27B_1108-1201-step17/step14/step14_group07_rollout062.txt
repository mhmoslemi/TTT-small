    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_sparse_T(a, active_indices):
    n = len(a)
    m = len(active_indices)
    rows = []
    cols = []
    vals = []
    row_map_cache = {}
    for i, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            val = a[k - j]
            if val > 1e-15:
                rows.append(i)
                cols.append(j)
                vals.append(val)
    if len(vals) == 0:
        T = csr_matrix((0, n))
    else:
        T = csr_matrix((np.array(vals), (np.array(rows), np.array(cols))), shape=(m, n))
    return T


def lp_best_response(a, M, solver, active_frac=0.92, max_active=None):
    n = len(a)
    size = 2 * n - 1
    aa = fast_autocorr(a)
    threshold = M * active_frac
    active_mask = aa >= threshold
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        active_indices = np.array([n - 1])
    if max_active is not None and len(active_indices) > max_active:
        top_vals = aa[active_indices]
        order = np.argsort(top_vals)[::-1]
        active_indices = active_indices[order[:max_active]]

    T = build_sparse_T(a, active_indices)
    b_var = cp.Variable(n)
    rhs = M * np.ones(len(active_indices))
    constraints = [T @ b_var <= rhs, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_main(a0, time_limit, solver, verbose=True, active_frac=0.92,
                     max_active=200, aggr_line_search=True):
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
        s0 = 1.0
    a = a * norm_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    last_improve_time = t0

    y = a.copy()
    x_prev = a.copy()
    momentum coeff = 0.0
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver, active_frac=active_frac,
                             max_active=max_active)
        if b is None:
            stall_count += 1
            if stall_count > 50:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_target / b_sum
        n_iters += 1

        if t == 1:
            y_new = g.copy()
        else:
            base_alpha = 2.0 / (t + 2.0)
            if aggr_line_search:
                best_alpha = base_alpha
                best_obj = fast_autocorr(np.maximum((1 - base_alpha) * y + base_alpha * g, 0)).max()
                for trial in [base_alpha * 0.3, base_alpha * 0.6, base_alpha * 1.5, base_alpha * 2.5]:
                    trial = min(trial, 1.0)
                    cand = np.maximum((1 - trial) * y + trial * g, 0.0)
                    cs = cand.sum()
                    if cs < 1e-15:
                        continue
                    cand = cand * norm_target / cs
                    sc = fast_autocorr(cand).max()
                    if sc < best_obj:
                        best_obj = sc
                        best_alpha = trial
                alpha = best_alpha
            else:
                alpha = base_alpha
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm_target / s_y

            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm_target / s_x

        y = y_new
        a = x_new if t > 1 else y_new

        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            last_improve_time = time.monotonic()
            stall_count = 0

        if t > 1:
            sc_a = fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                last_improve_time = time.monotonic()
                stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 15 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_target / s_avg
                sc_avg = fast_autocorr(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
                    last_improve_time = time.monotonic()
                    stall_count = 0

        if t % 200 == 0:
            if time.monotonic() - last_improve_time > time_limit * 0.15:
                stall_count += 1
                if stall_count > 15:
                    if verbose:
                        print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                    break
                elif stall_count > 5:
                    y = best_a * norm_target / best_a.sum()
                    a = y.copy()
                    x_prev = y.copy()
                    avg_a = np.zeros(n)
                    n_avg = 0

        if verbose and n_iters % 50 == 0:
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=5000, verbose=False):
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    obj = 2.0 * n * best_score / (s0 * s0)
    eta = 0.5

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        M = b.max()
        peak_idx = np.argmax(b)
        S = a.sum()
        current_obj = 2.0 * n * M / (S * S)

        grad = np.zeros(n)
        for j in range(n):
            k_diff = peak_idx - j
            if 0 <= k_diff < n:
                grad[j] = 2.0 * a[k_diff]
        grad = grad / (S * S) - 2.0 * current_obj * a / S

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm_target / s_new
        sc = fast_autocorr(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            eta = max(eta * 0.9, 0.01)

        if step % 100 == 0 and verbose:
            print(f"    SG step={step}: best={best_score:.8f}, eta={eta:.4f}")

    return best_a, best_score


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    scaled = T_temp * (b - max_b)
    scaled = np.clip(scaled, -500, 500)
    exp_vals = np.exp(scaled)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    obj = 2.0 * n * log_sum / (s * s)

    w = exp_vals / exp_vals.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:size] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad


def lse_scipy_optimize(a0, time_limit, T_start=50.0, T_end=2.0, n_stages=5):
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / max(1, n_stages - si), 200.0)

        def obj_func(x):
            o, _ = lse_objective_and_grad(x, n, T)
            return o

        def grad_func(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'time_limit': stage_time})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm_target / s_c
                    sc = fast_autocorr(a_cand).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42, n_top_peaks=8,
                              n_top_contrib=5, fracs=None):
    if fracs is None:
        fracs = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:n_top_peaks]
        improved = False

        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:n_top_contrib]]

            safe_positions = np.where(~peak_mask)[0]
            if len(safe_positions) == 0:
                safe_positions = np.where(np.arange(n) != top_contrib[0])[0]
            safe_sorted = safe_positions[np.argsort(a[safe_positions])][
                :max(1, len(safe_positions) // 5)]

            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                for dst in safe_sorted[:8]:
                    for frac in fracs:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm_target / s_t
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_a = a_trial.copy()
                            best_score = sc
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 3 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_target / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = sc
                        stall = 0
                        improved = True
                        break

            if stall > 200:
                n_perturb = max(5, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.3, 2.0)
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial * norm_target / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = sc

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        a = np.ones(n) / n
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = x * (1 - x) + 0.1
    else:
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 3 + 0.2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def heavy_ball_lp(a0, time_limit, solver, active_frac=0.90, max_active=150):
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
        s0 = 1.0
    a = a * norm_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    momentum = 0.7
    n_iters = 0
    no_improve = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(a).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(a, M_curr, solver, active_frac=active_frac,
                             max_active=max_active)
        if b is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_target / b_sum
        n_iters += 1

        a_new = momentum * a + (1 - momentum) * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm_target / s_new
        sc = fast_autocorr(a_new).max()

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1
            if no_improve % 20 == 0:
                momentum = max(momentum * 0.95, 0.3)
            if no_improve > 100:
                break

        if n_iters % 100 == 0:
            print(f"    HB iter={n_iters} ({time.monotonic()-t0:.0f}s): "
                  f"best={best_score:.8f}, mom={momentum:.3f}")

    return best_a, best_score

                if moved:

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_parent = len(best_seq) if best_seq is not None else 1600

    # Phase 0: Quick multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = sorted(set([n_parent, n_parent - 200, n_parent + 200, n_parent + 400]))
    scan_lengths = [x for x in scan_lengths if x >= 400]
    for i_scan, n_scan in enumerate(scan_lengths):
        if time.monotonic() - t_global > time_budget * 0.08:
            break
        remaining_scan = time_budget * 0.08 - (time.monotonic() - t_global)
        if remaining_scan < 15:
            break
        scan_time = min(remaining_scan / max(1, len(scan_lengths) - i_scan), 80.0)
        if scan_time < 10:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_main(a_norm, scan_time, solver,
                                          verbose=False, active_frac=0.90,
                                          max_active=150, aggr_line_search=False)
        do_try_update(a_out, f"scan_n{n_scan}")
        print(f"    n={n_scan}: score={evaluate_safe(a_out):.6f} ({time.monotonic()-t_global:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Main LP optimization at base length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Deep LP n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_main(a_norm, t1, solver, verbose=True,
                                 active_frac=0.92, max_active=200,
                                 aggr_line_search=True)
    do_try_update(a_out, f"lp_deep_n{n_best}")

    # Phase 2: Heavy-ball LP (different dynamics)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Heavy-ball LP n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = heavy_ball_lp(a_norm, t2, solver, active_frac=0.90,
                                  max_active=150)
        do_try_update(a_out, f"hb_lp_n{n_best}")

    # Phase 3: LSE + scipy
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t3, T_start=40.0, T_end=2.0,
                                       n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 4: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t4, seed=12345,
                                              n_top_peaks=10, n_top_contrib=6)
        do_try_update(a_out, "coord_polish_1")

    # Phase 5: Multi-peak targeted moves
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-peak moves n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_targeted_moves(a_norm, t5, seed=77)
        do_try_update(a_out, "multi_peak")

    # Phase 6: Second LP pass with noise
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP noisy n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 20:
        noise_scale = 0.03 * best_seq.mean()
        a_noisy = best_seq + rng.normal(0, noise_scale, n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = nesterov_lp_main(a_noisy, t6, solver, verbose=False,
                                     active_frac=0.92, max_active=200,
                                     aggr_line_search=True)
        do_try_update(a_out, f"lp_noisy_n{n_best}")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.5
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=99999,
                                              n_top_peaks=12, n_top_contrib=8,
                                              fracs=[0.001, 0.003, 0.01, 0.02, 0.05])
        do_try_update(a_out, "coord_final")

    # Final subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, remaining * 0.3, n_steps=3000,
                                       verbose=False)
        do_try_update(a_out, "subgrad_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
