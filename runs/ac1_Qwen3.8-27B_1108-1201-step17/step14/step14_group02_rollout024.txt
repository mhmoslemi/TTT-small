def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(v):
    v = np.asarray(v, dtype=np.float64)
    v_shifted = v - v.max()
    e = np.exp(v_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(v) / len(v)
    return e / s

def norm_sum1(a):

    s = a.sum()
    if s < 1e-30:

def compute_smoothed_obj_grad(v, tau):
    a = softmax_fn(v)
    n = len(a)
    conv = fast_conv(a)
    max_val = conv.max()
    shifted = (conv - max_val) / tau
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    obj = max_val + tau * log_sum
    weights = exp_shifted / exp_shifted.sum()
    b = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(weights, fft_size)
    B = rfft(b, fft_size)
    conv_wb = irfft(W * B, fft_size)[:conv_len]
    g_a = 2.0 * conv_wb[n - 1:2 * n - 1]
    ga_dot_a = np.dot(g_a, a)
    g_v = a * g_a - a * ga_dot_a
    return obj, g_v

def lp_best_response(a, M_bound, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    constraints = [T @ b_var <= M_bound, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(0), constraints)
    # Actually we want to maximize sum(b) s.t. T@b <= M, b>=0
    # But if we normalize sum(b)=1, we just need feasibility
    # Better: maximize sum(b) s.t. T@b <= M, b>=0
    b_var2 = cp.Variable(n, nonneg=True)
    prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T @ b_var2 <= M_bound])
    try:
        prob2.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob2.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Initialization
    print("=== Phase 0: Initialization ===")
    candidates = []
    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            candidates.append(('parent', p / p.sum()))
            ps = evaluate_safe(p / p.sum())
            print(f"  Parent: n={len(p)}, score={ps:.6f}")
    except Exception:
        pass
    # Parametric families
    n_base = 1600
    x = np.linspace(0, 1, n_base)
    for p_pow in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]:
        a_ramp = (np.arange(n_base) + 1) ** p_pow
        a_ramp = a_ramp / a_ramp.sum()
        candidates.append((f'ramp_p{p_pow}', a_ramp))
    for p_pow in [1.0, 2.0, 3.0]:
        for q_pow in [1.0, 2.0, 3.0]:
            a_taper = (np.arange(n_base) + 1) ** p_pow * (1.0 - np.arange(n_base) / n_base) ** q_pow
            a_taper = np.maximum(a_taper, 0)
            if a_taper.sum() > 1e-10:
                a_taper = a_taper / a_taper.sum()
                candidates.append((f'ramp_taper_{p_pow}_{q_pow}', a_taper))
    # Sine/cosine bumps
    a_sin = np.sin(np.pi * x) ** 2
    candidates.append(('sin2', a_sin / a_sin.sum()))
    a_cos = np.cos(np.pi * (x - 0.5)) ** 2
    candidates.append(('cos2', a_cos / a_cos.sum()))
    # Evaluate all candidates
    best_init = None
    best_init_score = float('inf')
    for name, a_c in candidates:
        a_c = norm_sum1(a_c)
        sc = 2 * len(a_c) * fast_conv(a_c).max()
        if sc < best_init_score:
            best_init_score = sc
            best_init = a_c
            best_init_name = name
    print(f"  Best init: {best_init_name}, J={best_init_score:.6f}")
    best_seq = best_init.copy()
    best_score = evaluate_safe(best_seq)
    print(f"  Best init verified score: {best_score:.6f}")

    # Blend parent with best parametric if parent exists
    if len(candidates) > 0 and candidates[0][0] == 'parent':
        parent = candidates[0][1]
        if len(parent) != n_base:
            parent = interpolate_sequence(parent, n_base)
        for blend in [0.2, 0.3, 0.5, 0.7]:
            mix = blend * norm_sum1(parent) + (1 - blend) * norm_sum1(best_init)
            mix = norm_sum1(mix)
            sc = 2 * n_base * fast_conv(mix).max()
            if sc < best_score:
                best_score = sc
                best_seq = mix.copy()
                print(f"  Blend {blend}: score={sc:.6f}")

    solver = get_solver()
    print(f"  Solver: {solver}")

    # Phase 1: Smoothed L-BFGS at n=1600
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 400.0)
    if t1 > 30:
        a_out, _ = smoothed_lbfgs_optimize(best_seq, t1, verbose=True)
        try_update(a_out, f'lbfgs_n{n_base}')

    # Phase 2: LP-based FW refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: FW n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 200.0)
    if t2 > 20:
        a_out, _ = fw_backtracking(best_seq, t2, solver, max_iters=300, verbose=True)
        try_update(a_out, f'fw_n{len(best_seq)}')

    # Phase 3: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 100.0)
    if t3 > 10:
        a_out, _ = coordinate_polish(best_seq, t3, verbose=True)
        try_update(a_out, f'coord_n{len(best_seq)}')

    # Phase 4: Scale to 2400
    n2 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Scale to n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.30, 300.0)
    if t4 > 30:
        a_long = interpolate_sequence(best_seq, n2)
        a_out, _ = smoothed_lbfgs_optimize(a_long, t4 * 0.7, verbose=True)
        try_update(a_out, f'lbfgs_n{n2}')
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = coordinate_polish(best_seq, remaining * 0.5, verbose=True)
            try_update(a_out, f'coord_n{n2}')

    # Phase 5: Final L-BFGS polish at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.30, 200.0)
    if t5 > 20:
        a_out, _ = smoothed_lbfgs_optimize(best_seq, t5, verbose=True)
        try_update(a_out, f'final_lbfgs_n{n_best}')

    # Phase 6: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.40, 150.0)
    if t6 > 10:
        a_out, _ = coordinate_polish(best_seq, t6, verbose=True)
        try_update(a_out, f'final_coord_n{n_best}')

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
