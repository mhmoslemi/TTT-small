def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]

def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[start - i:end - i]
    return T


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    n = len(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    rng = np.random.default_rng(12345)

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0
    n_restarts = 0

        if time.monotonic() - t0 > time_limit:
            break

        M_curr = b_y.max()
        if M_curr < 1e-14:

        eps = min(0.05, 0.005 * np.sqrt(t + 1.0))
        M_used = M_curr * (1.0 - eps)

        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr * 0.98, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Compute step size with ratio-based adaptation
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.03), 1.0)

        # Nesterov update
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if n_restarts == 0 and t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Periodic random perturbation to escape local minima
        if t % 800 == 0 and t > 200:
            scale = best_score * 0.02 / max(y.std(), 1e-10)
            perturbation = rng.normal(0, scale, n)
            a_pert = a + perturbation
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-10:

                score_p = fast_autocorr(a_pert).max()
                if score_p < best_score * 1.005:
                    a = a_pert.copy()

                    y_prev = a_pert.copy()

        # Stall detection (relaxed)

            if stall_count > 6:

                break
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_objective_and_grad(a, n, T):
    """Compute LSE_T(conv(a,a)) and its gradient w.r.t. a."""
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 0.0, np.zeros(n)
    shifted = T * (b - max_b)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-30:
        return max_b, np.zeros(n)
    w = exp_vals / exp_sum

    # Gradient: grad_i = 2 * sum_{j=0}^{n-1} a_j * w_{i+j}
    # This is the valid cross-correlation of a and w
    # Use FFT: ifft(FFT(a, N) * conj(FFT(w, N)))[:n]
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.fft(a, N)
    W_fft = np.fft.fft(w, N)
    full_corr = np.fft.ifft(A_fft * np.conj(W_fft))
    grad = 2.0 * np.real(full_corr[:n])

    lse_val = max_b + np.log(exp_sum) / T
    return lse_val, grad


def nesterov_proximal_lse(a0, time_limit, T_start=30.0, T_end=0.8, n_temps=7):
    """Nesterov-accelerated proximal gradient with LSE annealing."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * target_sum / n
    else:
        a = a * target_sum / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_temps)
    for ti, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining, time_limit / n_temps * 1.2)
        stage_start = time.monotonic()

        # FISTA initialization
        x = a.copy()
        y = a.copy()
        t_k = 1.0
        eta = 1.0  # initial step size

        while time.monotonic() - stage_start < stage_time:
            # Compute gradient at y
            _, grad = lse_objective_and_grad(y, n, T_val)

            # Gradient step + projection
            z = y - eta * grad
            # Project onto {a >= 0, sum(a) = target_sum}
            # Use bisection for the threshold
            z_sorted = np.sort(z)[::-1]
            # Find tau such that sum(max(0, z - tau)) = target_sum
            cumsum_z = np.cumsum(z_sorted)
            # Number of positive elements after threshold
            n_pos = np.searchsorted(-cumsum_z, -target_sum, side='right')
            if n_pos == 0:
                x_new = np.maximum(z, 0.0)
                s_new = x_new.sum()
                if s_new > 0:
                    x_new = x_new * target_sum / s_new
                else:
                    x_new = np.ones(n) * target_sum / n
            else:
                tau = (z_sorted[n_pos - 1] + cumsum_z[n_pos - 1] - target_sum) / n_pos
                x_new = np.maximum(z - tau, 0.0)
                s_new = x_new.sum()
                if s_new < 0.01:
                    x_new = np.ones(n) * target_sum / n
                elif abs(s_new - target_sum) / target_sum > 1e-6:
                    x_new = x_new * target_sum / max(s_new, 1e-15)

            # Nesterov momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            beta = (t_k - 1.0) / t_new
            y = x_new + beta * (x_new - x)
            x = x_new
            t_k = t_new

            # Adaptive step size (simple heuristic)
            eta *= 0.9995
            if eta < 0.01:
                eta = 0.01

            # Check improvement
            score = fast_autocorr(x).max()
            if score < best_score:
                best_score = score
                best_a = x.copy()
                a = x.copy()

        a = x.copy()
        score_final = fast_autocorr(a).max()
        if score_final < best_score:
            best_score = score_final
            best_a = a.copy()

    return best_a, best_score

            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # One-sided (helps reduce central convolution peak)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-2.0 * x)
    else:
        # Remnant-like: alternating high/low
        x = np.linspace(0.0, 1.0, n)
        a = 0.7 + 0.3 * np.cos(2 * np.pi * x * 3)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search combining LP-Nesterov, Nesterov proximal LSE, and peak shaving."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP-Nesterov (main workhorse, 45% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.45
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp1_n{n_best}", t_global)

    # === Phase 2: Nesterov Proximal LSE (25% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Nesterov Prox-LSE n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_proximal_lse(a_norm, t2, T_start=30.0, T_end=0.8, n_temps=7)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"proxlse_n{n_best}", t_global)

    # === Phase 3: Peak-shaving coordinate descent (15%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shaving_coord_descent(a_norm, t3, seed=42)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"coord1_n{n_best}", t_global)

    # === Phase 4: LP round 2 from improved point (10%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp2_n{n_best}", t_global)

    # === Phase 5: Try slightly longer sequence (10%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_best + 200
    print(f"\n=== Phase 5: LP longer n={n_long} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_long_n{n_long}", t_global)

    # === Phase 6: Final polish (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 15:
        # Short LSE polish
        t6a = min(remaining * 0.4, 100.0)
        if t6a > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_proximal_lse(a_norm, t6a, T_start=20.0, T_end=0.5, n_temps=4)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "proxlse_final", t_global)

        # Final coordinate descent
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out, _ = peak_shaving_coord_descent(a_norm, remaining2, seed=99999)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
