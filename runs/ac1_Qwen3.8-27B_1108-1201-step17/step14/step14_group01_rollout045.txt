_solver = None

def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 12
    temps = np.geomspace(80.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 4)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)

            if s_new < 1e-15:

        elif frac < 0.6:

        if b_sum < 1e-15:
            continue

            a = a_new

    return best_a, best_score

def peak_shave_cd(a0, time_limit, rng, verbose=True):
    """Coordinate descent that specifically targets reducing the convolution peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score_safe(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Focus on positions that contribute to the peak
            conv = fast_conv_self(a)
            cur_max = conv.max()
            peak_set = np.where(conv >= cur_max * 0.98)[0]

            # Candidate j's: gradient-informed + random
            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_set[:4]:
                j_candidates.add((kl - i) % n)
                j_candidates.add((kl + i) % n)
                # Also try positions that would "spread out" mass
                j_candidates.add((kl // 2 + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in list(j_candidates)[:10]:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Optimize delta with finer grid near optimum
                n_samples = 40
                deltas = np.linspace(0.01, max(0.01, ai * 0.99), n_samples)
                # Also try very small transfer
                deltas = np.concatenate([[ai * 0.01], deltas])

                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    d = min(d, ai * 0.99)
                    if d <= 0:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-13:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve == 5:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave CD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def multiplicative_weight_update(a0, time_limit, verbose=True):
    """Mirror descent / multiplicative weights approach targeting the peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    eta = 0.3
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv_self(a)
        max_c = conv.max()
        if max_c < 1e-12:
            break

        # Use epsilon-soft peak set for stability
        eps = 0.05
        peak_set = np.where(conv >= max_c * (1 - eps))[0]

        # Subgradient
        grad = np.zeros(n)
        for k in peak_set:
            w = 1.0 / len(peak_set)
            for i in range(n):
                if 0 <= k - i < n:
                    grad[i] += w * a[k - i]
                if 0 <= i - k < n:
                    grad[i] += w * a[i - k]

        # Multiplicative update
        eta_curr = eta * max_c / (grad.max() + 1e-15)
        eta_curr = min(eta_curr, 0.5)
        a_new = a * np.exp(-eta_curr * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        sc = fast_conv_self(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and reduce eta
            eta *= 0.9

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        # W-shaped
        a = np.exp(-12.0 * (x - 0.2) ** 2) + np.exp(-12.0 * (x - 0.5) ** 2) + np.exp(-12.0 * (x - 0.8) ** 2)
    else:
        # Plateau
        a = np.ones(n) * 0.5
        center = n // 2
        width = n // 6
        a[center - width:center + width] = 1.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_score_safe(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    n = len(a)
    max_b = fast_conv_self(a).max()
    return 2.0 * n * max_b / (S * S)

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LSE gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 180.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_gradient_descent(best_seq.copy(), t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: LP Nesterov (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t2, solver, verbose=True, full_constraint_final=True)
        try_update(a_out, f"lp1_n{n2}")

    # === Phase 3: Peak-shaving CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Peak-shave CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = peak_shave_cd(best_seq.copy(), t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{n3}")

    # === Phase 4: MWU (multiplicative weight update) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: MWU n={n3} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = multiplicative_weight_update(best_seq.copy(), t4, verbose=True)
        try_update(a_out, f"mwu_n{n3}")

    # === Phase 5: LP Nesterov round 2 (from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    t5 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 5: LP Nesterov round 2 n={n5} (budget={t5:.0f}s) ===")
    if t5 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq.copy(), t5, solver, verbose=True, full_constraint_final=True)
        try_update(a_out, f"lp2_n{n5}")

    # === Phase 6: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.20, 150.0)
    print(f"\n=== Phase 6: Final CD polish n={n5} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = peak_shave_cd(best_seq.copy(), t6, rng, verbose=True)
        try_update(a_out, f"pcd_final_n{n5}")

    # === Phase 7: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = n_base + 200
        t7 = min(remaining * 0.4, 150.0)
        print(f"\n=== Phase 7: Try length n={n_long} (budget={t7:.0f}s) ===")
        if t7 > 60:
            init_long = interpolate_seq(best_seq, n_long)
            a_out, _ = nesterov_lp_optimize(init_long, min(t7, 100.0), solver, verbose=False, full_constraint_final=False)
            try_update(a_out, f"lp_long_{n_long}")

    # === Phase 8: Restart from perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        t8 = min(remaining * 0.4, 120.0)
        print(f"\n=== Phase 8: Restart from perturbation (budget={t8:.0f}s) ===")
        if t8 > 40:
            a_pert = best_seq.copy()
            n_pert = len(a_pert)
            idx_pert = rng.choice(n_pert, size=max(20, n_pert // 20), replace=False)
            a_pert[idx_pert] *= (1.0 + 0.3 * rng.standard_normal(len(idx_pert)))
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert / s_p
            a_out, _ = nesterov_lp_optimize(a_pert, t8, solver, verbose=False, full_constraint_final=True)
            try_update(a_out, f"restart_lp")

    # === Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Final polish (budget={remaining:.0f}s) ===")
        a_out, _ = peak_shave_cd(best_seq.copy(), min(remaining * 0.6, 60.0), rng, verbose=False)
        try_update(a_out, "final_cd")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
