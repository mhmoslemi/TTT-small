def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_rows(a, k_indices):
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            r = k - j
            if 0 <= r < n:
                rows[idx, j] = a[r]
    return rows


def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and len(active_set) > 0:
        T = build_conv_matrix_rows(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        # Full constraint
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j+n, j] = a
        constraints = [T @ b_var <= M]
        active_set = np.arange(size)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set, None
    
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    
    return b_val, active_set, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=0, max_active_frac=None):

    else:
        a = a * norm / s0

    stall_count = 0
    stall_threshold = 30
    n_iters = 0
    
    if max_active_frac is None:
        max_active_frac = 0.3
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        
        # Adaptive active set: use top fraction of convolution values
        conv_y = fast_autocorr(y)
        if active_set is None or t % 50 == 0:
            # Select constraints that are "close to active"
            threshold = M_curr * (1.0 - 0.05)
            active_set = np.where(conv_y >= threshold)[0]
            # Ensure minimum number of constraints
            if len(active_set) < max(10, int(0.05 * (2*n-1))):
                # Take top fraction
                k_sel = max(int(max_active_frac * (2*n-1)), 10)
                top_idx = np.argsort(conv_y)[-k_sel:]
                active_set = top_idx
        
        b, active_set_new, duals = lp_best_response(y, M_curr, solver, active_set)
        
        if b is None:
            active_set = None

            if b is None:
                continue
        
        active_set = active_set_new
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.99)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)

            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        if stall_count > stall_threshold:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def build_conv_matrix_rows(a, k_indices):
    """Build rows of convolution matrix for given lag indices."""
    n = len(a)
    k_indices = np.asarray(k_indices)
    rows = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            r = k - j
            if 0 <= r < n:
                rows[idx, j] = a[r]
    return rows


def build_conv_matrix_full(a):
    """Build full convolution matrix T where (T@b)_k = (a*b)_k."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0:
        T = build_conv_matrix_rows(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        T = build_conv_matrix_full(a)
        constraints = [T @ b_var <= M]
        active_set = np.arange(size)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set, None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set, None

    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass

    return b_val, active_set, duals

def coordinate_descent(a0, time_limit, verbose=True, seed=0):
    """Individual coordinate descent with strategic line search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a.copy(), float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Compute direction vector g: change in conv when a[i] increases by 1
            # conv_new[k] = conv[k] + delta * g[k] + delta^2 * (1 if k==2*i else 0)
            g = np.zeros(size)
            lo = max(0, i)
            hi = min(size, i + n)
            g[lo:hi] = 2.0 * a[:hi - lo]

            # Find best delta via strategic evaluation
            k_peak = int(np.argmax(conv))
            best_d = 0.0
            best_val = conv.max()

            # Generate candidates based on peak reduction
            if g[k_peak] > 0:
                # Need negative delta to reduce peak
                deltas = [-ai * f for f in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 0.95]]
            elif g[k_peak] < 0:
                # Positive delta might help
                deltas = [ai * f for f in [0.1, 0.25, 0.5, 0.75, 1.0]] + [norm * 0.01, norm * 0.05]
            else:
                deltas = [-ai * 0.3, ai * 0.3]

            for d in deltas:
                if d < -ai - 1e-12:
                    continue
                new_conv = conv + d * g
                if 2 * i < size:
                    new_conv[2 * i] = conv[2 * i] + d * g[2 * i] + d * d
                val = new_conv.max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d

            if best_d != 0.0:
                a[i] += best_d
                if a[i] < 0:
                    a[i] = 0.0
                s = a.sum()
                if s < 1e-15:
                    a[i] -= best_d
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur = conv.max()
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[i] -= best_d
                    if a[i] < 0:
                        a[i] = 0.0
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1

        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:

def pair_cd(a0, time_limit, verbose=True, seed=0):

    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Direction: move delta from i to j
                # g[k] = change in conv[k] per unit delta moved from i to j
                g = np.zeros(size)
                # Removing from i: conv changes by -delta * (a[k-i] * 2)
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                g[lo1:hi1] -= 2.0 * a[:hi1 - lo1]
                # Adding to j: conv changes by +delta * (a[k-j] * 2)
                lo2 = max(0, j)
                hi2 = min(size, j + n)
                g[lo2:hi2] += 2.0 * a[:hi2 - lo2]

                # Find best delta
                k_peak = int(np.argmax(conv))

                best_val = conv.max()

                candidates = [ai * f for f in [0.1, 0.25, 0.5, 0.75, 0.9]]
                for d in candidates:
                    new_conv = conv + d * g
                    # Second order: at k=2i, we get -d^2; at k=2j, we get +d^2
                    if 2 * i < size:
                        new_conv[2 * i] -= d * d
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    val = new_conv.max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur = conv.max()
                    if cur < best_score - 1e-14:
                        best_score = cur
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:

                        a[i] += best_d

                        conv = fast_autocorr(a)

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    """Add multiplicative noise and renormalize."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def gaussian_smooth(a, sigma_frac=0.01):
    """Apply light Gaussian smoothing to break symmetry."""
    n = len(a)
    sigma = sigma_frac * n
    if sigma < 0.5:
        return a.copy()
    from scipy.ndimage import gaussian_filter1d
    a_smooth = gaussian_filter1d(a, sigma=sigma)
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    return a_smooth / s

        pass

    return [float(x) for x in best_seq]
