import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve(a, M, solver, max_active=200):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    
    # Active set reduction
    for _ in range(4):
        circ = T @ b_val
        tol = 1e-7 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= min(size, max_active + 50) or n_active < 10:
            break
        if n_active > max_active:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
            active = np.zeros(size, dtype=bool)
            active[top_idx] = True
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if (T @ b_new).max() <= M * (1 + 1e-5):
            b_val = b_new
        else:
            break
    
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lse_gradient_descent_v2(a0, time_limit, verbose=True, seed=42):
    """
    Improved LSE-smoothed gradient descent with:
    - Adagrad adaptive step sizes
    - Nesterov momentum
    - Adaptive temperature schedule
    - Periodic restarts
    - Multi-peak soft-max weighting
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Adagrad state
    grad_sq_sum = np.zeros(n)
    lr = 1.0 / np.sqrt(n)
    
    # Nesterov momentum
    v = np.zeros(n)
    beta = 0.9
    
    # Temperature schedule
    T_start = best_score * 0.1
    T_end = best_score * 0.005
    
    # Precompute FFT size for gradient
    fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    
    stall_count = 0
    best_for_restart = best_score
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        # Anneal temperature
        T = T_start * (T_end / T_start) ** elapsed_frac
        T = max(T, 1e-10)
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        if max_b < best_score:
            best_score = max_b
            best_a = a.copy()
            best_for_restart = best_score
            stall_count = 0
        else:
            stall_count += 1
        
        # Soft-max weighting
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient via FFT: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
        # This is 2 * conv(w, a_rev)[n-1:2n-1]
        a_rev = a[::-1]
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        
        # Project gradient onto tangent space (sum = 0)
        grad -= np.mean(grad)
        
        # Adagrad update
        grad_sq_sum += grad * grad
        step = lr / np.sqrt(grad_sq_sum + 1e-10)
        
        # Nesterov: look-ahead
        a_lookahead = a + beta * v
        # Recompute gradient at lookahead (optional, expensive)
        # For efficiency, use gradient at current point
        
        v = beta * v - step * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1
        
        # Adaptive learning rate
        score_new = fast_autocorr(a).max()
        if score_new > best_score * 1.01:
            lr *= 0.5
            v *= 0.5
        elif score_new < best_score:
            lr = min(lr * 1.1, 1.0 / np.sqrt(n) * 2)
        
        # Restart if stalled
        if stall_count > 500:
            if elapsed_frac < 0.8:
                # Perturb
                noise_scale = 0.05 * a.mean()
                a = np.maximum(a + rng.normal(0, noise_scale, n), 0)
                a = a * norm / a.sum()
                v = np.zeros(n)
                grad_sq_sum = np.zeros(n)
                lr = 1.0 / np.sqrt(n)
                stall_count = 0
                if verbose:
                    print(f"    LSE-GD2 restart at step {steps}, T={T:.4f}")
        
        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD2 step={steps} ({elapsed:.0f}s): best={best_score:.8f}, max_b={max_b:.6f}, T={T:.4f}, lr={lr:.4f}")

    return best_a, best_score


def nesterov_lp_optimize_v2(a0, time_limit, solver, verbose=True, max_active=150):
    """Improved LP Nesterov with better parameter tuning."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.5  # Start with less aggressive momentum
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()
        
        b = lp_solve(y, M_curr, solver, max_active=max_active)
        if b is None:
            stall += 1
            if stall > 50:
                break
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        stall = 0

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.3)  # Cap step size for stability
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Adaptive gamma
        if n_iters % 200 == 0:
            oscillation = abs(fast_autocorr(y).max() - fast_autocorr(prev_y).max())
            if oscillation > 0.01 * best_score:
                gamma = max(1.0, gamma - 0.2)
            else:
                gamma = min(1.8, gamma + 0.1)

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, gamma={gamma:.2f}")

        # Restart
        if stall > 100:
            rng = np.random.default_rng(n_iters * 7 + 3)
            a = np.maximum(a + rng.normal(0, 0.02 * a.mean(), n), 0)
            a = a * norm / a.sum()
            y = a.copy()
            prev_y = a.copy()
            stall = 0
            gamma = 1.5

    return best_a, best_score


def peak_shave_v2(a0, time_limit, seed=42, verbose=True):
    """
    Improved peak shaving with:
    - Multi-source moves (not just single source)
    - Better destination selection
    - Temperature-annealed acceptance
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iter = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        if max_b < best_score:
            best_score = max_b
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top peaks
        top_k = min(3, 2 * n - 1)
        peak_indices = np.argsort(b)[::-1][:top_k]
        
        for p_star in peak_indices:
            # Find contributing pairs
            js = max(0, p_star - n + 1)
            je = min(n - 1, p_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[p_star - idx]
            
            if contrib.max() < 1e-18:
                continue

            # Source: position with highest contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Destination: position that contributes least to all top peaks
            # Score each position by how much it would contribute if we added mass there
            dst_scores = np.zeros(n)
            for p in peak_indices:
                # Contribution of position j to peak p: a[p-j] if valid
                j_range = np.arange(max(0, p - n + 1), min(n, p + 1))
                if len(j_range) > 0:
                    dst_scores[j_range] += a[np.clip(p - j_range, 0, n - 1)]
            
            # We want to move to positions with low dst_scores
            dst_scores[src] = np.inf  # Exclude source
            # Also exclude positions that are already 0 and would create new peaks
            low_score_positions = np.where(dst_scores < np.median(dst_scores))[0]
            
            if len(low_score_positions) == 0:
                low_score_positions = np.array([i for i in range(n) if i != src])
            
            dst = low_score_positions[rng.integers(len(low_score_positions))]

            # Try multiple fractions
            best_frac = 0
            best_trial = max_b
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                amt = src_val * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_trial:
                    best_trial = sc
                    best_frac = frac

            if best_frac > 0 and best_trial < max_b:
                a[src] -= src_val * best_frac
                a[dst] += src_val * best_frac
                a = np.maximum(a, 0)
                a = a * norm / a.sum()
                break  # Restart peak loop with updated a

        # Annealing acceptance when no improvement
        if stall > 20 and rng.random() < 0.1:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0:
                amt = a[i] * rng.uniform(0.02, 0.15)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    T = best_score * 0.002 * (1.0 - elapsed_frac)
                    if sc < best_score or rng.random() < np.exp(-(sc - best_score) / max(T, 1e-10)):
                        a = a_trial

        if stall > 500:
            # Larger perturbation
            n_pert = max(10, n // 30)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.1, 0.4)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                if fast_autocorr(a_trial).max() < best_score * 1.05:
                    a = a_trial
            stall = 0

        if verbose and n_iter % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave2 iter={n_iter} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

        return np.ones(new_n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Adaptive LSE-smoothed gradient descent with Adagrad + Nesterov momentum
2. LP-based Fessler-Parks Nesterov with progressive constraint management

Key improvements over previous: better gradient normalization,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve_fast(a, M, solver, max_active=100):
    """Fast LP with aggressive constraint reduction."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    # Start with only top constraints for speed
    # First pass: solve with relaxed bound to get warm start
    constraints = [T @ b_var <= M * 1.05]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try with full constraints
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None

    # Refine with active set
    for _ in range(3):
        circ = T @ b_val
        tol = 1e-6 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active < 5 or n_active > max_active + 30:
            if n_active > max_active:
                active_idx = np.where(active)[0]
                active_vals = circ[active_idx]
                top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
                active = np.zeros(size, dtype=bool)
                active[top_idx] = True
            else:
                break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if (T @ b_new).max() <= M * (1 + 1e-4):
            b_val = b_new
        else:
            break

    if b_val.sum() < 1e-15:
        return None
    return b_val


def lse_gd_improved(a0, time_limit, verbose=True, seed=42):
    """
    LSE-smoothed gradient descent with:
    - Adagrad adaptive step sizes
    - Nesterov momentum with damping
    - Adaptive temperature based on peak concentration
    - Gradient normalization for stability
    - Smart restart strategy
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Adagrad state
    grad_sq_sum = np.ones(n) * 1e-8
    base_lr = 50.0 / n

    # Momentum
    v = np.zeros(n)
    beta = 0.9

    # Temperature (starts high, decreases)
    T_init = best_score * 0.15
    T_min = max(1e-8, best_score * 0.003)

    # FFT size
    fft_size = 1 << (3 * n - 2).bit_length()

    stall = 0
    best_stall = 0
    n_restarts = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)

        # Adaptive temperature: lower when well-converged, higher when exploring
        T = T_init * (T_min / T_init) ** (elapsed_frac ** 0.7)
        T = max(T, T_min)

        b = fast_autocorr(a)
        max_b = b.max()

        if max_b < best_score:
            best_score = max_b
            best_a = a.copy()
            best_stall = steps
            stall = 0
        else:
            stall = steps - best_stall

        # Soft-max weights
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] for valid indices
        # = 2 * (w * a_rev)[n-1 : 2n-1] via FFT
        a_rev = a[::-1]
        W_fft = rfft(w, fft_size)
        AR_fft = rfft(a_rev, fft_size)
        conv_vals = irfft(W_fft * AR_fft, fft_size)
        grad = 2.0 * conv_vals[n - 1:2 * n - 1]

        # Normalize gradient for stability
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm

        # Project onto tangent space
        grad -= np.mean(grad)

        # Adagrad step
        step_sizes = base_lr / np.sqrt(grad_sq_sum)
        grad_sq_sum += (grad * grad) * 0.01

        # Nesterov momentum
        v = beta * v + step_sizes * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        # Adaptive learning rate
        score_new = fast_autocorr(a).max()
        if score_new > best_score * 1.02:
            base_lr *= 0.7
            v *= 0.3
        elif score_new < best_score * 0.999:
            base_lr = min(base_lr * 1.05, 200.0 / n)

        # Restart strategy
        if stall > 300 and n_restarts < 20:
            if elapsed_frac < 0.85:
                noise = rng.normal(0, 0.03 * a.mean(), n)
                a = np.maximum(a + noise, 0)
                a = a * norm / a.sum()
                v = np.zeros(n)
                grad_sq_sum = np.ones(n) * 1e-8
                base_lr = 50.0 / n
                stall = 0
                best_stall = steps
                n_restarts += 1
                if verbose:
                    print(f"    LSE restart {n_restarts} at step {steps}")

        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.5f}, lr={base_lr:.2f}")

    return best_a, best_score


def lp_nesterov_improved(a0, time_limit, solver, verbose=True, max_active=100):
    """
    LP Nesterov with:
    - Progressive alpha (starts larger, decays)
    - Adaptive gamma (momentum)
    - Fewer but more targeted restarts
    - Faster LP via warm-start-like constraint management
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.3
    n_restarts = 0

    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()

        if M_curr < 1e-12:
            break

        b = lp_solve_fast(y, M_curr, solver, max_active=max_active)
        if b is None:
            stall += 1
            if stall > 30:
                # Try again with more relaxed constraint
                b = lp_solve_fast(y, M_curr * 1.01, solver, max_active=max_active)
                if b is None:
                    if n_restarts < 10:
                        rng = np.random.default_rng(n_iters * 11 + 5)
                        y = np.maximum(y + rng.normal(0, 0.02 * y.mean(), n), 0)
                        y = y * norm / y.sum()
                        prev_y = y.copy()
                        n_restarts += 1
                        stall = 0
                        gamma = 1.3
                    else:
                        break
                    continue
            else:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        stall = 0

        # Fessler-Parks step with capped alpha
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.25)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t <= 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Adaptive gamma
        if n_iters % 100 == 0 and n_iters > 10:
            diff = abs(fast_autocorr(y).max() - fast_autocorr(prev_y).max())
            if diff > 0.005 * best_score:
                gamma = max(1.0, gamma - 0.15)
            elif diff < 0.001 * best_score:
                gamma = min(1.7, gamma + 0.05)

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, gamma={gamma:.2f}")

    return best_a, best_score


def peak_reduction(a0, time_limit, seed=42, verbose=True):
    """
    Principled peak reduction:
    - Identifies the dominant peak
    - Computes exact contribution of each position to that peak
    - Moves mass from high-contribution to low-contribution positions
    - Uses quadratic line search for optimal transfer amount
    - Considers multiple peaks simultaneously
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iter = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        if max_b < best_score:
            best_score = max_b
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top peaks
        n_peaks = min(3, 2 * n - 1)
        peak_order = np.argsort(b)[::-1][:n_peaks]

        improved_this_iter = False
        for p_star in peak_order:
            # Contribution of each position to peak p_star
            js = max(0, p_star - n + 1)
            je = min(n - 1, p_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[p_star - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-18:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Score all positions by their "peaking tendency"
            # A position j causes peaking if it's aligned with large values at other peaks
            dst_scores = np.zeros(n)
            for p in peak_order:
                jp = np.arange(max(0, p - n + 1), min(n, p + 1))
                if len(jp) > 0:
                    dst_scores[jp] += a[np.clip(p - jp, 0, n - 1)]
            dst_scores[src] = np.inf
            # Also penalize positions that would create a new peak at lag 0
            dst_scores += a  # Self-contribution

            # Pick destination from low-score positions
            valid_dst = np.where(dst_scores < np.inf)[0]
            if len(valid_dst) == 0:
                continue
            n_pick = min(5, len(valid_dst))
            candidates = valid_dst[np.argsort(dst_scores[valid_dst])][:n_pick]
            dst = int(candidates[rng.integers(n_pick)])

            # Quadratic line search: find delta that minimizes max autocorrelation
            # After moving delta from src to dst, the autocorrelation changes
            # c_new[k] = c_old[k] + delta*(a[|k-src|] - a[|k-dst|]) + O(delta^2)
            # For the target peak: delta * (a[p_star - src] - a[p_star - dst])
            # We want this negative, i.e., a[p_star-src] > a[p_star-dst]

            # Try a range of deltas
            best_delta = 0
            best_new_max = max_b
            for frac in [0.03, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
                delta = src_val * frac
                if delta < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[dst] += delta
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_new_max:
                    best_new_max = sc
                    best_delta = delta

            if best_delta > 0 and best_new_max < max_b:
                a[src] -= best_delta
                a[dst] += best_delta
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                improved_this_iter = True
                break  # Restart with updated a

        if not improved_this_iter:
            # Try random small moves with annealing
            if stall > 10:
                n_try = 5
                for _ in range(n_try):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    amt = a[i] * rng.uniform(0.02, 0.1)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        improved_this_iter = True
                        break
                    # Annealing
                    T_ann = best_score * 0.003 * (1.0 - elapsed_frac)
                    if T_ann > 1e-10 and rng.random() < np.exp(-(sc - best_score) / T_ann):
                        a = a_trial
                        break

        # Large perturbation if very stalled
        if stall > 300:
            n_pert = max(5, n // 40)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.1, 0.4)
                j = int(rng.integers(n))
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                if fast_autocorr(a_trial).max() < best_score * 1.03:
                    a = a_trial
            stall = 0

        if verbose and n_iter % 400 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed iter={n_iter} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: hybrid LSE-GD + LP Nesterov + peak reduction.
    Strategy: explore lengths quickly, then focus optimization on best length
    using alternating LSE-GD (fast exploration) and LP (precise exploitation).
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Quick length exploration (4% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_explore = min(remaining * 0.04, 50.0)
    if t_explore > 10:
        print(f"\n=== Phase 1: Length exploration (budget={t_explore:.0f}s) ===")
        lengths_to_try = [n_base - 100, n_base + 50, n_base + 100, n_base + 200]
        t_per = t_explore / len(lengths_to_try)
        for n_try in lengths_to_try:
            if n_try < 300 or time.monotonic() - t_global > time_budget - 100:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            a_norm = a_interp * np.sqrt(2.0 * n_try) / a_interp.sum()
            a_out, _ = lse_gd_improved(a_norm, t_per, verbose=False, seed=n_try)
            try_update(a_out, f"explore_n{n_try}")

    # === Phase 2: Main LSE-GD (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_main = min(remaining * 0.35, 550.0)
    if t_main > 30:
        print(f"\n=== Phase 2: LSE-GD n={n_cur} (budget={t_main:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gd_improved(a_norm, t_main, verbose=True, seed=123)
        try_update(a_out, f"lse_main_n{n_cur}")

    # === Phase 3: LP Nesterov (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_lp = min(remaining * 0.25, 400.0)
    if t_lp > 30:
        print(f"\n=== Phase 3: LP Nesterov n={n_cur} (budget={t_lp:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_improved(a_norm, t_lp, solver, verbose=True, max_active=100)
        try_update(a_out, f"lp_n{n_cur}")

    # === Phase 4: LSE-GD pass 2 (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_lse2 = min(remaining * 0.15, 250.0)
    if t_lse2 > 20:
        print(f"\n=== Phase 4: LSE-GD p2 n={n_cur} (budget={t_lse2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gd_improved(a_norm, t_lse2, verbose=True, seed=999)
        try_update(a_out, f"lse2_n{n_cur}")

    # === Phase 5: Peak reduction (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_peak = min(remaining * 0.10, 150.0)
    if t_peak > 15:
        print(f"\n=== Phase 5: PeakRed n={n_cur} (budget={t_peak:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_reduction(a_norm, t_peak, seed=314, verbose=True)
        try_update(a_out, f"peak_n{n_cur}")

    # === Phase 6: Final LP polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 6: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_improved(a_norm, remaining, solver, verbose=True, max_active=80)
        try_update(a_out, f"lp_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
