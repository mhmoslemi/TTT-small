def lse_obj_and_grad(x_flat, n, T):
    a = np.maximum(x_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = (b - max_b) / T
    exp_vals = np.exp(shifted)
    lse_val = max_b + T * np.log(exp_vals.sum())
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()

    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * lse_val / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def lse_optimize(a0, time_limit, T_start=100.0, T_end=2.0, n_stages=6):

    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    if s0 < 1e-15:

        a = a * norm_val / s0

    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        if remaining < 5:
            break

        def obj_func(x, T_val=T):
            val, _ = lse_obj_and_grad(x, n, T_val)
            return val

        def grad_func(x, T_val=T):
            _, g = lse_obj_and_grad(x, n, T_val)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                 jac=grad_func,

                        print(f"    LSE T={T:.1f}: score={score_c:.10f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score


def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M, cp.sum(b_var) >= 1e-6]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 3:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 6:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    else:
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search combining LSE, LP best-response, coordinate descent, and length exploration."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try constant sequence first (theoretical optimum for non-negative)
    n_const = 1600
    const_seq = np.ones(n_const)
    const_score = evaluate_safe(const_seq)
    print(f"Constant sequence (n={n_const}): score={const_score:.10f}")
    best_seq = const_seq.copy()
    best_score = const_score

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent sequence: n={len(parent)}, score={p_score:.10f}")
            if p_score < best_score:
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    # Phase 1: LSE optimization from parent (if different from constant)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 1: LSE optimization n={n_best} (budget={remaining:.0f}s) ===")
        t1 = min(remaining * 0.25, 300.0)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t1, T_start=200.0, T_end=1.0, n_stages=8)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lse_n{n_best}", t_global)

    # Phase 2: LP Nesterov with tight M
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 2: LP-Nest tight M n={n_best} (budget={remaining:.0f}s) ===")
        t2 = min(remaining * 0.30, 400.0)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, use_tight_M=True,
                                        tight_eps_scale=0.003, max_stall=45.0)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_n{n_best}", t_global)

    # Phase 3: Targeted coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 3: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
        t3 = min(remaining * 0.15, 150.0)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, t3, seed=12345)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"coord_n{n_best}", t_global)

    # Phase 4: Try different lengths with structured init
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 4: Multi-length exploration (budget={remaining:.0f}s) ===")
        lengths = [800, 1200, 2000, 2400]
        for n_try in lengths:
            if n_try == len(best_seq):
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_alloc = min(remaining * 0.2, 120.0)

            # Interpolated init
            init_seq = interpolate_sequence(best_seq, n_try)
            a_norm = init_seq * np.sqrt(2.0 * n_try) / init_seq.sum()

            # Quick LSE
            t_lse = t_alloc * 0.5
            a_out1, _ = lse_optimize(a_norm, t_lse, T_start=50.0, T_end=2.0, n_stages=4)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out1, f"lse_n{n_try}", t_global)

            # Quick LP
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                n_cur = len(best_seq)
                a_norm2 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                t_lp = min(remaining * 0.3, 100.0)
                a_out2, _ = nesterov_lp_optimize(a_norm2, t_lp, solver, verbose=False,
                                                 use_tight_M=True, tight_eps_scale=0.005,
                                                 max_stall=30.0)
                best_seq, best_score = attempt_update(best_seq, best_score, a_out2,
                                                      f"lp_n{n_cur}_r2", t_global)

    # Phase 5: Final LSE polish at very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 5: Final LSE polish n={n_best} (budget={remaining:.0f}s) ===")
        t5 = min(remaining * 0.4, 200.0)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t5, T_start=20.0, T_end=0.5, n_stages=5)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "lse_polish", t_global)

    # Phase 6: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 6: Final coord descent n={n_best} ===")
        t6 = remaining - 10
        if t6 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = targeted_coord_descent(a_norm, t6, seed=99999)
            best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(1600)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
