def project_simplex(v, target_sum=1.0):

    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return target_sum * v / s

        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f}: best={best_score:.8f}")
    return best_a, best_score


def fast_fw_nesterov(a0, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    n_iters = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv_y = fast_autocorr(y)
        k_peak = int(np.argmax(conv_y))
        M_y = conv_y[k_peak]
        if M_y < 1e-15:
            break

        # Gradient: grad[i] = 2*y[k_peak - i] for valid i
        grad = np.zeros(n)
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        if hi >= lo:
            seg = y[k_peak - lo:k_peak - hi + 1][::-1]
            grad[lo:hi + 1] = 2.0 * seg

        j_min = int(np.argmin(grad))
        alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y
        y_new[j_min] += alpha
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new / s_y

        sc = fast_autocorr(y_new).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = y_new.copy()
            stale = 0
        else:
            stale += 1

        y = y_new
        n_iters += 1

        if verbose and n_iters % 2000 == 0:
            print(f"    FW iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
        if stale > 5000:
            break

    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_print = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        elapsed = time.monotonic() - t0
        if verbose and elapsed - last_print > 30:
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
            last_print = elapsed

    return best_a, best_score


def spectral_flatten(a0, time_limit, seed=42):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    pad_len = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        a_pad = np.zeros(pad_len)
        a_pad[:n] = a
        A = rfft(a_pad)
        power = np.abs(A) ** 2

        # Find peak in autocorrelation domain
        ac = fast_autocorr(a)
        peak_idx = int(np.argmax(ac))
        peak_val = ac[peak_idx]
        avg_ac = ac.mean()
        if peak_val / max(avg_ac, 1e-15) < 1.005:
            stall += 1
            if stall > 300:
                break
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            continue

        # Reduce power at frequency bins that contribute to peak
        # The autocorrelation at lag k is (1/N) sum_m |A_m|^2 exp(2πimk/N)
        # To reduce ac[k_peak], reduce |A_m|^2 where cos(2πmk_peak/N) is large
        phase_alignment = np.cos(2 * np.pi * np.arange(len(A)) * peak_idx / pad_len)
        positive_bins = np.where(phase_alignment > 0.3)[0]
        positive_bins = positive_bins[positive_bins > 0]  # skip DC

        if len(positive_bins) > 0:
            # Scale down the top contributing bins
            top_contrib = np.argsort(power[positive_bins] * phase_alignment[positive_bins])[::-1][:8]
            for idx in top_contrib:
                bi = positive_bins[idx]
                current_mag = np.abs(A[bi])
                if current_mag < 1e-15:
                    continue
                reduction = 1.0 - 0.03
                A[bi] *= reduction

        a_new_full = irfft(A, pad_len)
        a_new = a_new_full[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            stall += 1
            continue
        a_new = a_new * norm / s_new
        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-13:
            a = a_new
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

    return best_a, best_score


def local_refine(a0, time_limit, seed=12345):
    """Coordinate descent + random perturbation local refinement."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:8]

        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if len(contrib) == 0 or contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find good destination
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            # Sort by current value (move from high to low)
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_cand = max(1, len(candidates_sorted) // 10)
            n_cand = min(n_cand, 20)

            for _ in range(n_cand):
                dst = candidates_sorted[rng.integers(len(candidates_sorted))]
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-18 or a[src] - amt < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = a_trial * norm / a_trial.sum()
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random pair swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                if a[i] - amt >= 0:
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = a_trial * norm / a_trial.sum()
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 2000:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_best = len(best_seq)

    # Phase 1: Fast Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 80.0)
    if t1 > 15:
        print(f"\n=== Phase 1: Fast FW n={n_best} (budget={t1:.0f}s) ===")
        a_out, _ = fast_fw_nesterov(best_seq, t1, verbose=True)
        try_update(a_out, f"fw1_n{n_best}")

    # Phase 2: LSE-FISTA
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 100.0)
    if t2 > 15:
        print(f"\n=== Phase 2: LSE-FISTA n={n_best} (budget={t2:.0f}s) ===")
        T_sched = [15.0, 4.0, 1.0, 0.3, 0.1]
        a_out, _ = lse_fista(best_seq, T_sched, t2, verbose=True)
        try_update(a_out, f"lse_n{n_best}")

    # Phase 3: LP-Nesterov main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.35, 550.0)
    if t3 > 60:
        print(f"\n=== Phase 3: LP-Nesterov n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t3, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # Phase 4: Spectral flattening
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Spectral n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = spectral_flatten(a_norm, t4, seed=77)
        try_update(a_out, f"spectral_n{n_best}")

    # Phase 5: Local refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Local refine n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = local_refine(a_norm, t5, seed=4242)
        try_update(a_out, f"local_n{n_best}")

    # Phase 6: LP again (post spectral + local)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.25, 400.0)
    if t6 > 60:
        print(f"\n=== Phase 6: LP-Nesterov 2 n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t6, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # Phase 7: Try different n values
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_candidates = [len(best_seq) - 200, len(best_seq) + 200, len(best_seq) + 400]
        for nc in n_candidates:
            if nc < 400 or nc > 2500:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t7 = min(remaining * 0.3, 200.0)
            if t7 < 30:
                continue
            print(f"\n=== Phase 7: LP at n={nc} (budget={t7:.0f}s) ===")
            init_long = interpolate_sequence(best_seq, nc)
            a_norm = init_long * np.sqrt(2.0 * nc) / init_long.sum()
            a_out, _ = lp_nesterov(a_norm, t7, solver, verbose=False)
            try_update(a_out, f"lp_n{nc}")

    # Phase 8: Final local refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.4, 100.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final local n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = local_refine(a_norm, t8, seed=99999)
            try_update(a_out, "final_local")

    # Phase 9: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t9 = min(remaining * 0.6, 150.0)
        if t9 > 30:
            print(f"\n=== Phase 9: Final LP n={n_best} (budget={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t9, solver, verbose=False)
            try_update(a_out, "lp_final")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
