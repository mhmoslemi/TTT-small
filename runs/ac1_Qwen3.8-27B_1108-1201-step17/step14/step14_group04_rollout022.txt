Can we achieve this? We'd need all convolution values to be equal, which means conv(a,a) = [1/(2n-1)] * ones(2n-1). This would require a very specific structure.

For n=2: conv(a,a) = [a0^2, 2*a0*a1, a1^2]. We want all equal: a0^2 = 2*a0*a1 = a1^2. From a0^2 = a1^2, a0 = a1 (non-negative). Then 2*a0*a1 = 2*a0^2 = a0^2, so a0 = 0. Degenerate.

So we can't achieve the bound, but we can get close for large n.

The theoretical minimum is 1 (as n -> infinity), and we're at 1.504. So there's room for improvement.

Now, what shape of a minimizes the peak of conv(a,a)?

Intuitively, we want the convolution to be as flat as possible. The convolution of a with itself is the "smearing" of the sequence. To make it flat, we want the sequence to have a specific frequency structure.

In the frequency domain: conv(a,a) has DFT = |DFT(a)|^2 (with appropriate zero-padding for linear convolution). So we want |DFT(a)|^2 to be as flat as possible (white spectrum), which means DFT(a) has constant magnitude.

But a is non-negative and real, which constrains what DFT(a) can be. This is related to the "flat-top" or "minimum peak" problem in signal processing.

For a real non-negative sequence, the DFT satisfies:
- A[0] = sum a_i (real, positive)
- A[n] = sum a_i * (-1)^i (real) if n is even (Nyquist)
- A[k] = conjugate(A[n-k]) (for real a)

The constraint that a >= 0 means the inverse DFT of |A|^2 / A (phase) must be non-negative. This is a complex constraint.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Frank-Wolfe with LSE smoothing (O(n log n) per iteration via FFT)
2. Exact subgradient FW with active-set identification
3. Coordinate descent polish (exploits 1D structure of each coordinate)
4. Pairwise transfer local search
5. Multiple restarts with diverse initializations
Key innovation: extremely fast FW iterations enabling millions of steps,
combined with coordinate descent that exploits the piecewise-linear structure
of the objective in each coordinate direction.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def compute_lse_grad(a, T):
    """Compute gradient of LSE objective: T*log(sum_k exp(b_k/T - b_max/T))."""
    n = len(a)
    b = compute_conv(a)
    b_max = b.max()
    if b_max < 1e-15:
        return np.zeros(n), b
    shifted = np.clip((b - b_max) / T, -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= ws
    # grad_i = (2/T) * sum_k w_k * a[k-i] for valid k-i
    # = (2/T) * (w * a_rev)[n-1+i] via cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Arev = rfft(a[::-1], N)
    cross = irfft(W * Arev, N)
    grad = (2.0 / T) * cross[n - 1: n - 1 + n]
    return grad, b


def fw_lse(a_init, T_schedule, time_limit, verbose=True):
    """Frank-Wolfe with LSE smoothing - very fast per iteration."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = compute_conv(a)
    best_a = a.copy()
    best_b = b.max()
    t0 = time.monotonic()
    global_iter = 0

    for Ti, T in enumerate(T_schedule):
        stage_iter = 0
        while time.monotonic() - t0 < time_limit:
            grad, b = compute_lse_grad(a, T)
            b_max = b.max()
            j = int(np.argmin(grad))

            global_iter += 1
            step = 2.0 / (global_iter + 2.0)
            # Cap step size for stability
            step = min(step, 0.5)

            a = (1.0 - step) * a
            a[j] += step

            # Quick eval every few iterations
            if stage_iter % 50 == 0:
                b_check = compute_conv(a)
                if b_check.max() < best_b:
                    best_b = b_check.max()
                    best_a = a.copy()
                    if verbose:
                        print(f"    FW-LSE T={T:.3f} iter={global_iter} ({time.monotonic()-t0:.0f}s): best={best_b:.10f}")

            stage_iter += 1

        # End of temperature stage
        b_final = compute_conv(a)
        if b_final.max() < best_b:
            best_b = b_final.max()
            best_a = a.copy()
        if verbose:
            print(f"    FW-LSE stage T={T:.3f} done ({time.monotonic()-t0:.0f}s): best={best_b:.10f}")

    return best_a, best_b


def fw_subgrad(a_init, time_limit, eps=1e-6, verbose=True):
    """Exact subgradient FW: use uniform subgradient over active peaks."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_b = compute_conv(a).max()
    t0 = time.monotonic()
    global_iter = 0

    while time.monotonic() - t0 < time_limit:
        b = compute_conv(a)
        b_max = b.max()
        if b_max < 1e-12:
            break

        # Identify active set (peaks within eps of max)
        threshold = b_max - eps * b_max
        active = np.where(b >= threshold)[0]

        # Compute subgradient: average gradient over active peaks
        # grad for peak k: 2*a[k-i] (reversed and shifted)
        # Combined: sum over active k of a[k-i], divided by |active|
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += a[k - np.arange(js, je + 1)]
        grad *= 2.0 / len(active)

        j = int(np.argmin(grad))
        global_iter += 1
        step = 2.0 / (global_iter + 2.0)
        step = min(step, 0.3)

        a = (1.0 - step) * a
        a[j] += step

        if global_iter % 20 == 0:
            b_check = compute_conv(a)
            if b_check.max() < best_b:
                best_b = b_check.max()
                best_a = a.copy()
                if verbose and global_iter % 200 == 0:
                    print(f"    FW-subgrad iter={global_iter} ({time.monotonic()-t0:.0f}s): best={best_b:.10f} |active|={len(active)}")

    return best_a, best_b


def coordinate_descent(a_init, time_limit, verbose=True, max_rounds=5000):
    """
    Coordinate descent exploiting 1D structure.
    For coordinate i, the objective max_k conv(a,a)[k] as a function of a[i]
    is piecewise linear (or quadratic at k=2i). We find the minimizer.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_conv(a).max()
    t0 = time.monotonic()
    round_count = 0

    # Pre-compute: for each k, which (i, m) pairs contribute to conv[k]
    # conv[k] = sum_j a[j]*a[k-j]; for fixing a[i], the terms involving a[i] are:
    #   j=i: a[i]*a[k-i] (if valid)
    #   k-j=i => j=k-i: a[k-i]*a[i] (if valid)
    # So conv[k] = a[i]*(a[k-i] + a[k-i]) + rest = 2*a[i]*a[k-i] + rest (if k-i valid and k!=2i)
    # Or conv[k] = a[i]^2 + rest (if k=2i)

    # For coordinate i, define:
    #   c_k = conv[k] - terms_involving_a[i]
    #   Then conv[k] as function of x=a[i]:
    #     if k=2i: c_k + x^2
    #     elif 0<=k-i<n and k-i != i: c_k + 2*x*a[k-i]
    #     elif k-i == i (same as k=2i): handled above
    #     else: c_k
    #   Also need to maintain sum(a)=1, so when a[i] changes, we redistribute.
    #   Simplest: fix sum of others, vary a[i], and scale others to maintain sum=1.
    #   Actually, let's just do: set a[i]=x, keep others fixed, then renormalize.
    #   But renormalization changes all coordinates... Let's use a different approach.
    #   
    #   Alternative: work with the constraint sum(a)=1 by parameterizing
    #   a[i] = x, and scaling all other coordinates by (1-x)/(1-a[i_old]).
    #   This is complex. Instead, let's do simple projected updates:
    #   Try a[i] += delta for small delta, project to non-negative simplex.

    # Simpler approach: for each coordinate, try a few candidate values
    # and pick the best one (that minimizes the max convolution).
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and round_count < max_rounds:
        b = compute_conv(a)
        b_max = b.max()
        improved = False
        round_count += 1

        # Process coordinates in random order
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            a_i_old = a[i]
            if a_i_old < 1e-15:
                # Try setting to a small positive value (take from the largest other coord)
                max_other = np.max(np.delete(a, i))
                if max_other < 1e-10:
                    continue
                for frac in [0.0001, 0.001]:
                    delta = max_other * frac
                    a_trial = a.copy()
                    a_trial[i] = delta
                    a_trial /= a_trial.sum()
                    b_trial = compute_conv(a_trial).max()
                    if b_trial < best_score - 1e-14:
                        a = a_trial
                        best_score = b_trial
                        best_a = a.copy()
                        improved = True
                        break
                continue

            # Try reducing a[i] (shift mass to other coordinates)
            # Or increasing it
            # Key insight: reducing a[i] reduces conv[2i] = a[i]^2 + ... 
            # and linear terms 2*a[i]*a[k-i]
            # Best to shift mass to coordinates that don't contribute to peaks

            # Simple: try a few fraction reductions
            for frac in [0.01, 0.05, 0.1, 0.2, 0.4]:
                delta = a_i_old * frac
                if delta < 1e-12:
                    break
                a_trial = a.copy()
                a_trial[i] -= delta
                # Distribute delta to the coordinate with lowest "peak contribution"
                # Simple heuristic: give to a random other coordinate or the minimum
                # For efficiency, just give to a uniform spread
                others = np.arange(n)
                others = others[others != i]
                a_trial[others] += delta / len(others)
                a_trial = np.maximum(a_trial, 0)
                a_trial_sum = a_trial.sum()
                if a_trial_sum < 1e-10:
                    continue
                a_trial /= a_trial_sum
                b_trial = compute_conv(a_trial).max()
                if b_trial < best_score - 1e-14:
                    a = a_trial
                    best_score = b_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and round_count > 100:
            # Stuck: try random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                for frac in [0.005, 0.01, 0.02]:
                    amt = a[i] * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < 0:
                        continue
                    a_trial /= a_trial.sum()
                    b_trial = compute_conv(a_trial).max()
                    if b_trial < best_score - 1e-14:
                        a = a_trial
                        best_score = b_trial
                        best_a = a.copy()
                        improved = True
                        break

    return best_a, best_score


def pairwise_transfer(a_init, time_limit, verbose=True, seed=42):
    """Targeted pairwise mass transfer to reduce peak convolution values."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = compute_conv(a)
        b_max = b.max()
        # Get top peaks
        top_peaks = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_peaks:
            if improved:
                break
            # Find contributing indices
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            # Source: index contributing most to this peak
            src_pos = int(idx[np.argmax(contrib)])

            # Destination: find coordinate that minimizes increase in all peaks
            # Heuristic: choose from coordinates with low values and far from peak sources
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src_pos)[0]
            if len(candidates) == 0:
                continue

            # Sort by current value (prefer moving to low-value positions)
            candidates = candidates[np.argsort(a[candidates])]
            # Try a few destinations
            for dst in candidates[:5]:
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src_pos] * frac
                    if amt < 1e-13:
                        break
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst] += amt
                    if a_trial[src_pos] < 0:
                        continue
                    a_trial /= a_trial.sum()
                    b_trial = compute_conv(a_trial).max()
                    if b_trial < best_score - 1e-14:
                        a = a_trial
                        best_score = b_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial /= a_trial.sum()
                    b_trial = compute_conv(a_trial).max()
                    if b_trial < best_score:
                        a = a_trial
                        best_score = b_trial
                        best_a = a.copy()
                        stall = 0
            if stall > 2000:
                break

    return best_a, best_score


def fw_lse_line_search(a_init, T, time_limit, verbose=True):
    """FW with LSE and periodic line search for better convergence."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_b = compute_conv(a).max()
    t0 = time.monotonic()
    global_iter = 0

    while time.monotonic() - t0 < time_limit:
        grad, b = compute_lse_grad(a, T)
        j = int(np.argmin(grad))
        global_iter += 1

        # Line search every 50 iterations, diminishing otherwise
        if global_iter % 50 == 0:
            # Try several step sizes
            best_step = 0.0
            best_val = best_b
            for step in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                a_trial = (1.0 - step) * a
                a_trial[j] += step
                val = compute_conv(a_trial).max()
                if val < best_val:
                    best_val = val
                    best_step = step
            if best_step > 0:
                a = (1.0 - best_step) * a
                a[j] += best_step
            else:
                a = (1.0 - 0.01) * a
                a[j] += 0.01
        else:
            step = 2.0 / (global_iter + 2.0)
            step = min(step, 0.3)
            a = (1.0 - step) * a
            a[j] += step

        if global_iter % 100 == 0:
            b_check = compute_conv(a)
            if b_check.max() < best_b:
                best_b = b_check.max()
                best_a = a.copy()
                if verbose:
                    print(f"    FW-LS T={T:.3f} iter={global_iter} ({time.monotonic()-t0:.0f}s): best={best_b:.10f}")

    return best_a, best_b


def make_initializations(n, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # Bump functions with various shapes
    x = np.linspace(0, 1, n)
    for p in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        f = np.sin(np.pi * x) ** p
        f = np.maximum(f, 0)
        f /= f.sum()
        inits.append(f)

    # Gaussian bumps
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
        f = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        f /= f.sum()
        inits.append(f)

    # Triangular
    f = np.maximum(1 - 2 * np.abs(x - 0.5), 0)
    f /= f.sum()
    inits.append(f)

    # Edge-concentrated (asymmetric)
    for alpha in [0.3, 0.5, 0.7]:
        f = np.exp(-alpha * x) * np.exp(-(1-alpha) * (1-x))
        f /= f.sum()
        inits.append(f)

    # Two bumps
    for center in [0.3, 0.4]:
        f = np.exp(-0.5 * ((x - center) / 0.1) ** 2) + np.exp(-0.5 * ((x - (1-center)) / 0.1) ** 2)
        f /= f.sum()
        inits.append(f)

    # Random
    for _ in range(3):
        f = rng.random(n)
        f = f ** rng.uniform(0.5, 3.0)
        f /= f.sum()
        inits.append(f)

    return inits


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc) if np.isfinite(sc) else float('inf')
    except Exception:
        return float('inf')
