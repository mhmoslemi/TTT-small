    """
    Compute the LSE objective value and gradient in log-space.
    a: non-negative vector with sum = s = sqrt(2n)
    Returns: (objective_value, grad_logspace)
    where grad_logspace is the gradient of the LSE objective w.r.t. log(a).

def fast_autocorr(a):
    """Compute linear autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_objective_and_grad(a, T):
    """
    Compute LSE objective and gradient w.r.t. a.
    Uses stride tricks for the cross-correlation (O(n^2) but very fast in practice).
    """
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)

    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)

    shifted = T * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = exp_vals.sum()
    if log_sum < 1e-300:
        return max_b, np.zeros(n)
    lse_val = max_b + np.log(log_sum) / T
    w = exp_vals / log_sum

    # grad_a[j] = 2 * sum_m w[j+m] * a[m]  (j=0..n-1)
    # Use stride tricks: windows[j, m] = w[j+m]
    windows = as_strided(w, (n, n), (w.strides[0], w.strides[0]))
    cross = windows.dot(a)
    grad_a = 2.0 * cross

    return lse_val, grad_a


def grad_to_logspace(grad_a, a, s):
    """Convert gradient from a-space to log-space (x = log(a) parameterization)."""
    ga_dot_a = np.dot(grad_a, a) / s
    return a * (grad_a - ga_dot_a)


def lse_adam_optimize(a0, time_limit, T_schedule=None, lr=0.05, beta1=0.9, beta2=0.999,
                      eps=1e-8, n_params=None, verbose=True):
    """
    Optimize LSE objective using Adam in log-space.
    a0: initial sequence (will be normalized)
    T_schedule: list of (T_value,_num_iters) pairs, or a single T value
    """
    n = len(a0)
    if n_params is None:
        n_params = n
    s = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a * s / a.sum()

    # Initialize log-space
    x = np.log(a + 1e-30)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_count = 0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    if T_schedule is None:
        T_schedule = [(10.0, None)]  # None = use remaining time

    for T_val, n_iters in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        stage_start = time.monotonic()

        for it in range(1000000):
            if n_iters is not None and it >= n_iters:
                break
            if time.monotonic() - t0 > time_limit:
                break

            # Recompute a from x (softmax)
            x_shifted = x - x.max()
            exp_x = np.exp(x_shifted)
            a = s * exp_x / exp_x.sum()

            # Compute gradient
            lse_val, grad_a = lse_objective_and_grad(a, T_val)
            grad_x = grad_to_logspace(grad_a, a, s)

            # Adam update
            t_count += 1
            m_adam = beta1 * m_adam + (1 - beta1) * grad_x
            v_adam = beta2 * v_adam + (1 - beta2) * grad_x ** 2
            m_hat = m_adam / (1 - beta1 ** t_count)
            v_hat = v_adam / (1 - beta2 ** t_count)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Periodic evaluation
            if (it + 1) % 50 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if verbose and (it + 1) % 500 == 0:
                        print(f"      LSE-Adam T={T_val:.1f} it={it}: score={score:.8f}")

        # End of stage: update a from final x
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = s * exp_x / exp_x.sum()
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def subgradient_mirror_descent(a0, time_limit, eta0=None, decay=0.9995, verbose=True):
    """
    Subgradient mirror descent on the simplex.
    Uses the entropy barrier (softmax update) which naturally maintains constraints.
    """
    n = len(a0)
    s = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a * s / a.sum()

    if eta0 is None:
        eta0 = 1.0 / np.sqrt(2.0 * n)

    eta = eta0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    last_improve = t0
    stall = 0

    it = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Subgradient: g = 2 * (R_{k_star} a)
        # (R_{k_star} a)[j] = a[k_star - j] if 0 <= k_star - j < n, else 0
        g = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_start <= j_end:
            g[j_start:j_end + 1] = 2.0 * a[k_star - j_start:k_star - j_end:-1]

        # Mirror descent update: a_new = a * exp(-eta * g) / Z
        log_ratio = -eta * g
        log_ratio -= log_ratio.max()  # for numerical stability
        weights = np.exp(log_ratio)
        Z = np.dot(a, weights)
        if Z < 1e-30:
            break
        a = a * weights / Z
        # Ensure sum is exactly s
        a = a * s / a.sum()

        it += 1
        if it % 100 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve = time.monotonic()
            elif time.monotonic() - last_improve > 30:
                stall += 1
                if stall > 20:
                    break
            if verbose and it % 2000 == 0:
                print(f"      SubGrad it={it}: score={best_score:.8f}, eta={eta:.6f}")

        eta *= decay

    return best_a, best_score


def build_constraint_matrix(a, n):
    """Build the matrix T such that T @ b = conv(a, b)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T_mat=None):
    """Solve the LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T_mat is None:
        T_mat = build_constraint_matrix(a, n)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
