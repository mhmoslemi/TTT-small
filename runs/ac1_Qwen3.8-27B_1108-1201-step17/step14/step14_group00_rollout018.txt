            # conv_k = sum_{j: 0<=j<n AND 0<=k-j<n} a_j * a_{k-j}
            # d/d(a_i) conv_k = sum_{j: valid} [1(j=i)*a_{k-j} + a_j*1(k-j=i)]
            # = a_{k-i} * I(0<=k-i<n AND 0<=i<n) + a_i * I(0<=i<n AND 0<=k-i<n)
            # = (a_{k-i} + a_i) * I(0<=k-i<n)  [since 0<=i<n is always true]
            pass

        # Redo gradient properly
        grad = np.zeros(n)
        valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
        grad[valid] = y[np.arange(n)[valid]] + y[k_star - np.arange(n)[valid]]
        # For invalid indices, gradient is 0

        # FW: v = e_{argmin grad} scaled
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = 1.0  # We work with sum=1, so v is a unit vector

        # Step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new /= s_y

        if t == 1:
            a_cur = v.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.99)
            a_cur = y_new + momentum * (y_new - y)
            a_cur = np.maximum(a_cur, 0.0)
            s_a = a_cur.sum()
            if s_a < 1e-15:
                a_cur = y_new.copy()
            else:
                a_cur /= s_a

        y = y_new
        a = a_cur

        score = fast_autocorr(a).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            sc_avg = fast_autocorr(avg_check).max()
            if sc_avg < best_score - 1e-13:
                best_score = sc_avg
                best_a = avg_check.copy()
                stall = 0

        if stall > 2000:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall = 0

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): score={2*n*score:.8f}")

    return best_a, 2 * n * best_score


def structured_cd(a0, time_limit, rng, verbose=True, max_passes=500):
    """Structured coordinate descent: move mass from high-gradient to low-gradient coords."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a /= s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    for pnum in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break

        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

            # Find the best partner j (where a[j] is large and gradient contribution is favorable)
            # Try a few candidates
            best_delta = 0.0
            best_new_max = best_score
            best_j = -1

            # Sample candidates: prefer positions with high a[j]
            candidates = rng.choice(n, size=min(20, n), replace=False)
            # Also add i itself's neighbors for locality
            for j in candidates:
                if j == i:
                    continue
                aj = a[j]
                if aj < 1e-12:
                    continue

                # Compute the gain vector: change in conv when moving delta from j to i
                # First order: 2*(a[k-i] - a[k-j]) for valid k
                # We want to minimize max(conv_k + delta * g_k + so_k * delta^2)
                
                # Compute g_k = 2*(a[k-i] - a[k-j]) for k=0,...,size-1
                # a[k-i] is valid when 0<=k-i<n, i.e., i<=k<=i+n-1
                # a[k-j] is valid when 0<=k-j<n, i.e., j<=k<=j+n-1
                
                g = np.zeros(size)
                lo1, hi1 = max(0, i), min(size, i + n)
                g[lo1:hi1] += 2.0 * a[lo1 - i:hi1 - i]
                lo2, hi2 = max(0, j), min(size, j + n)
                g[lo2:hi2] -= 2.0 * a[lo2 - j:hi2 - j]

                # Try a few delta values
                max_delta = ai  # Can't move more than a[i]... wait, we move FROM j TO i
                # Actually we move delta from j to i: a[j]-=delta, a[i]+=delta
                # So delta <= a[j]
                max_delta = aj
                fracs = [0.25, 0.5, 0.75, 0.9, 0.99]
                for frac in fracs:
                    d = frac * max_delta
                    if d < 1e-15:
                        continue
                    new_conv_est = conv + d * g
                    # Second order corrections
                    di = 2 * i
                    if di < size:
                        new_conv_est[di] += d * d
                    dj = 2 * j
                    if dj < size:
                        new_conv_est[dj] -= d * d
                    dij = i + j
                    if dij < size:
                        new_conv_est[dij] -= 2.0 * d * d

                    val = new_conv_est.max()
                    if val < best_new_max - 1e-14:
                        best_new_max = val
                        best_delta = d
                        best_j = j

            if best_delta > 0 and best_j >= 0:
                a[i] += best_delta
                a[best_j] -= best_delta
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a[i] -= best_delta
                    a[best_j] += best_delta
                    continue
                a /= s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[i] -= best_delta
                    a[best_j] += best_delta
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and pnum % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pnum} ({elapsed:.0f}s): score={2*n*best_score:.8f}")

    return best_a, 2 * n * best_score


def single_coord_descent(a0, time_limit, rng, verbose=True):
    """Single-coordinate descent: try adjusting one coordinate at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a /= s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_passes = 200

    for pnum in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        perm = rng.permutation(n)

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx
            ai = a[i]
            if ai < 1e-12:
                continue

            # Try moving mass from i to a random j
            j = int(rng.integers(n))
            if j == i:
                continue

            # Compute gain
            g = np.zeros(size)
            lo1, hi1 = max(0, i), min(size, i + n)
            g[lo1:hi1] -= 2.0 * a[lo1 - i:hi1 - i]
            lo2, hi2 = max(0, j), min(size, j + n)
            g[lo2:hi2] += 2.0 * a[lo2 - j:hi2 - j]

            max_d = ai
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                d = frac * max_d
                new_conv_est = conv + d * g
                di = 2 * i
                if di < size:
                    new_conv_est[di] -= d * d
                dj = 2 * j
                if dj < size:
                    new_conv_est[dj] += d * d
                dij = i + j
                if dij < size:
                    new_conv_est[dij] += 2.0 * d * d
                val = new_conv_est.max()
                if val < best_score - 1e-14:
                    a[i] -= d
                    a[j] += d
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    conv = fast_autocorr(a)
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and pnum % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD pass={pnum} ({elapsed:.0f}s): score={2*n*best_score:.8f}")

    return best_a, 2 * n * best_score


def structured_init(n, rng):
    """Generate structured initial sequence."""
    choice = int(rng.integers(0, 10))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.2 * np.cos(6 * np.pi * x))
    elif choice == 8:
        # Frequency-flat inspired
        a = np.ones(n)
        a += 0.1 * np.cos(2 * np.pi * np.arange(n) / n * 7)
        a += 0.05 * np.cos(2 * np.pi * np.arange(n) / n * 13)
    else:
        # Bimodal
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20*(x-0.25)**2) + np.exp(-20*(x-0.75)**2) + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def fourtier_perturb(a, rng, n_modes=10, scale=0.03):
    """Perturb using Fourier modes for smoother variations."""
    n = len(a)
    x = np.arange(n)
    pert = np.zeros(n)
    for _ in range(n_modes):
        freq = rng.integers(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0) * scale
        pert += amp * np.cos(2 * np.pi * freq * x / n + phase)
    noise = 1.0 + pert
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate():
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 1: Smoothed Mirror Descent ===
    print(f"\n=== Phase 1: Smoothed Mirror Descent n={n_base} ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = smoothed_mirror_descent(best_seq, t1, verbose=True, seed=0)
        try_update(a_out, f"smd1_n{n_base}")

    # === Phase 2: FW Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: FW Nesterov n={n_best} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, _ = fw_nesterov(best_seq, t2, verbose=True, seed=1)
        try_update(a_out, f"fw1_n{n_best}")

    # === Phase 3: Structured CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Structured CD n={n_best} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_out, _ = structured_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"scd1_n{n_best}")

    # === Phase 4: Fourier perturbation + SMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Fourier perturb + SMD ===")
    n_restarts = 3
    t4_total = min(remaining * 0.20, 250.0)
    t4_each = t4_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t4 = min(t4_each, remaining - 50)
        if t4 < 25:
            break
        n_cur = len(best_seq)
        pert = fourtier_perturb(best_seq, rng, n_modes=5 + 5 * r, scale=0.02 + 0.01 * r)
        a_out, _ = smoothed_mirror_descent(pert, t4, verbose=False, seed=100 + r)
        try_update(a_out, f"fp4_{r}_n{n_cur}")

    # === Phase 5: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 200
    print(f"\n=== Phase 5: SMD+CD n={n2} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40:
        init_long = interpolate_sequence(best_seq, n2)
        a_out, _ = smoothed_mirror_descent(init_long, t5 * 0.5, verbose=True, seed=7)
        try_update(a_out, f"smd_n{n2}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = structured_cd(a_out, min(remaining * 0.3, 60.0), rng, verbose=False)
            try_update(a_out2, f"scd_n{n2}")

    # === Phase 6: Try shorter length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(1200, len(best_seq) - 200)
    if remaining > 80 and n_short != len(best_seq):
        print(f"\n=== Phase 6: SMD+CD n={n_short} ===")
        t6 = min(remaining * 0.15, 120.0)
        if t6 > 30:
            init_short = interpolate_sequence(best_seq, n_short)
            a_out, _ = smoothed_mirror_descent(init_short, t6 * 0.6, verbose=True, seed=99)
            try_update(a_out, f"smd_n{n_short}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out2, _ = structured_cd(a_out, min(remaining * 0.3, 40.0), rng, verbose=False)
                try_update(a_out2, f"scd_n{n_short}")

    # === Phase 7: Final polish with CD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD polish n={n_best} ===")
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 15:
        a_out, _ = structured_cd(best_seq, t7, rng, verbose=True)
        try_update(a_out, f"cd_final_n{n_best}")

    # === Phase 8: Single coord descent final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final single CD n={n_best} ===")
        t8 = min(remaining * 0.30, 100.0)
        if t8 > 10:
            a_out, _ = single_coord_descent(best_seq, t8, rng, verbose=False)
            try_update(a_out, f"scd_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
