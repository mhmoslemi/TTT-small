def fast_autocorr(a):

def build_T_pattern(n):
    """Pre-build the sparsity pattern for the convolution constraint matrix."""
    size = 2 * n - 1
    row_parts = []
    col_parts = []
    for i in range(n):
        k_end = min(i + n, size)
        cnt = k_end - i
        row_parts.append(np.arange(i, k_end))
        col_parts.append(np.full(cnt, i))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    return rows, cols


def sparse_conv_matrix(a, pattern):
    """Build sparse matrix T where T@b = conv(a,b) (truncated to length 2n-1)."""
    rows, cols = pattern
    n = len(a)
    size = 2 * n - 1
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def lp_best_response_fast(a, M, pattern):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0 using scipy linprog."""
    n = len(a)
    size = 2 * n - 1
    T = sparse_conv_matrix(a, pattern)
    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            b_val = np.maximum(res.x, 0.0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val, res
        return None, None
    except Exception:
        return None, None

def lse_objective_and_grad(a_flat, n, T_temp):
    """Log-sum-exp smoothed objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalized sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp2_n{n_best}", t_global)

    # Phase 5: Try slightly longer sequence
    remaining = time_budget - (time.time() - t_global)
    n_long = n_best + 300
    pattern_long = build_T_pattern(n_long)
    print(f"\n=== Phase 5: Longer sequence (n={n_long}, budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, pattern_long, verbose=True, periodic_shake=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_long_n{n_long}", t_global)

    # Phase 6: Final polish
    remaining = time_budget - (time.time() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, budget={remaining:.0f}s) ===")
    if remaining > 30:
        pattern_fin = build_T_pattern(n_best)
        # Quick LP round
        t6a = min(remaining * 0.4, 120.0)
        if t6a > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t6a, pattern_fin, verbose=False, periodic_shake=False)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "lp_final", t_global)

        # Coord descent polish
        remaining2 = time_budget - (time.time() - t_global)
        t6b = min(remaining2 * 0.4, 100.0)
        if t6b > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = peak_targeted_coord_descent(a_norm, t6b, seed=99999, max_rounds=20000)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_final", t_global)

        # LSE final
        remaining3 = time_budget - (time.time() - t_global)
        t6c = min(remaining3 * 0.5, 80.0)
        if t6c > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_scipy_optimize(a_norm, t6c, T_start=30.0, T_end=1.0, n_stages=3)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "lse_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.time()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
