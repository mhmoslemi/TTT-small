import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def subgradient_at_max(a, c, top_k=1):
    """Compute subgradient of max_k c_k at point a.
    Returns gradient vector of length n."""
    n = len(a)
    # Get top k lags
    top_indices = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_indices:
        # grad[j] += 2 * a[k-j] if 0 <= k-j <= n-1
        js = np.arange(0, n)
        ks = k - js
        valid = (ks >= 0) & (ks < n)
        grad[valid] += 2.0 * a[ks[valid]]
    return grad

def mirror_descent_optimize(a0, time_limit, n_top=1, eta0=0.5, eta_decay=0.9999, restart_interval=500):
    """Mirror descent (exponential gradient) for min max_k c_k(a) s.t. sum(a)=S, a>=0."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    eta = eta0
    avg_a = np.zeros(n)
    count = 0
    stall = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        score = c.max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute subgradient
        grad = subgradient_at_max(a, c, top_k=n_top)
        
        # Mirror descent update: a *= exp(-eta * grad)
        # For numerical stability: a *= exp(-eta * (grad - grad.mean()))
        eta_eff = eta * S  # scale
        log_update = -eta_eff * grad
        # Center for stability
        log_update -= log_update.mean()
        # Clip for numerical stability
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a * S / s
        
        # Decay learning rate
        eta *= eta_decay
        
        # Averaging
        avg_a += a
        count += 1
        if count % 50 == 0:
            avg_check = avg_a / count
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * S / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        # Restart if stalled
        if stall > restart_interval:
            if eta < eta0 * 0.01:
                # Full restart with perturbation
                a = best_a + np.random.randn(n) * 0.01 * best_a.mean()
                a = np.maximum(a, 0)
                s = a.sum()
                a = a * S / s
                eta = eta0
                avg_a = np.zeros(n)
                count = 0
                stall = 0
            else:
                eta = max(eta, eta0 * 0.1)
                stall = 0
        
        iter_count += 1
        if iter_count % 1000 == 0:
            print(f"    MD iter={iter_count}: best={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score

def softmax_subgradient(a, c, temperature):
    """Compute smoothed subgradient using softmax over lags."""
    n = len(a)
    size = len(c)
    c_shifted = c - c.max()
    w = np.exp(c_shifted / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        k_max = int(np.argmax(c))
        w = np.zeros(size)
        w[k_max] = 1.0
    else:
        w = w / w_sum
    # grad[j] = 2 * sum_k w[k] * a[k-j] if 0 <= k-j < n
    # This is a weighted combination of shifted/reversed copies of a
    # Use FFT for efficiency: grad = 2 * convolve(w, a_reversed) truncated
    a_rev = a[::-1]
    fft_size = 1 << (n + size - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_wa = irfft(W * AR, fft_size)
    # The k=0 element of conv_wa corresponds to sum_k w[k]*a_rev[k-j] for j=0..n-1
    # a_rev[k-j] = a[n-1-(k-j)] = a[n-1-k+j]
    # We want: grad[j] = 2 * sum_k w[k] * a[k-j] (if 0<=k-j<n)
    # Let m = k-j, so k = j+m, and we need 0<=m<n
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is (w * a)[j] in the cross-correlation sense
    # Let's compute it directly for simplicity when n is not too large
    # Actually let's use the direct formula
    grad = np.zeros(n)
    for j in range(n):
        ks = np.arange(j, j + n)
        valid = (ks >= 0) & (ks < size)
        grad[j] = 2.0 * np.dot(w[ks[valid]], a[valid])
    return grad


def softmax_subgradient_fast(a, c, temperature, top_k=None):
    """Faster version using only top-k lags."""
    n = len(a)
    size = len(c)
    if top_k is None or top_k >= size:
        top_k = size
    c_shifted = c - c.max()
    w_full = np.exp(c_shifted / temperature)
    w_full = w_full / w_full.sum()
    # Use only top_k for efficiency
    if top_k < size:
        top_indices = np.argsort(w_full)[::-1][:top_k]
        w = np.zeros(size)
        w[top_indices] = w_full[top_indices]
        w = w / w.sum()
    else:
        w = w_full
    # Direct computation: grad[j] = 2 * sum_k w[k] * a[k-j] if valid
    grad = np.zeros(n)
    for j in range(n):
        ks = np.arange(j, j + n)
        valid = (ks >= 0) & (ks < size)
        if valid.any():
            grad[j] = 2.0 * np.dot(w[ks[valid]], a[valid])
    return grad


def exponential_gradient_optimize(a0, time_limit, T_start=5.0, T_end=0.5, eta_start=0.3, eta_decay=0.99995, top_k=8, restart_stall=2000, verbose=True):
    """Exponential gradient (mirror descent) with softmax-subgradient."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    eta = eta_start
    avg_a = np.zeros(n)
    count = 0
    stall = 0
    iter_count = 0
    total_time = _global_time_budget - (time.monotonic() - _global_t0)
    time_limit = min(time_limit, total_time)

    # Anneal temperature
    total_iters_est = max(100, int(time_limit / 0.005))
    T = T_start

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** min(1.0, elapsed_frac * 1.5)

        c = fast_autocorr(a)
        score = c.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Compute subgradient
        grad = softmax_subgradient_fast(a, c, T, top_k=top_k)

        # Mirror descent: a *= exp(-eta * grad), centered for stability
        eta_eff = eta
        log_update = -eta_eff * grad
        log_update -= log_update.mean()
        log_update = np.clip(log_update, -30.0, 30.0)
        a = a * np.exp(log_update)

        # Renormalize
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a * S / s

        # Decay
        eta *= eta_decay
        if eta < eta_start * 0.05:
            eta = eta_start * 0.1

        # Averaging
        avg_a += a
        count += 1
        if count % 100 == 0:
            avg_check = avg_a / count
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * S / s_avg
                sc_avg = fast_autocorr(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
                    stall = 0

        # Restart logic
        if stall > restart_stall:
            rng_local = np.random.default_rng(iter_count)
            a = best_a + rng_local.normal(0, 0.05 * best_a.mean(), n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a * S / s
            eta = eta_start
            avg_a = np.zeros(n)
            count = 0
            stall = 0

        iter_count += 1
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EG iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.3f}, eta={eta:.5f}")

    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def coordinate_descent_polish(a0, time_limit, seed=42):
    """Aggressive coordinate descent targeting spectral peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    total_time = _global_time_budget - (time.monotonic() - _global_t0)
    time_limit = min(time_limit, total_time)

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:16]
        improved = False

        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:8]]

            # Build mask of indices involved in top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-12:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) < 5:
                    candidates = np.where(a > 1e-12)[0]
                if len(candidates) == 0:
                    continue
                candidates_sorted = candidates[np.argsort(-a[candidates])[:max(5, len(candidates) // 8)]]

                for dst in candidates_sorted[:8]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbations at various stall levels
            if stall % 3 == 0:
                for _ in range(50):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.2)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break

            if stall % 20 == 0:
                n_perturb = max(5, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.3, 2.0)
                try_update(a_trial)

            if stall > 500:
                break

    return best_a, best_score


def interpolated_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def diverse_init(n, rng):
    """Generate diverse asymmetric initializations."""
    choice = int(rng.integers(0, 10))
    if choice == 0:
        # Left-shifted Gaussian
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2)
    elif choice == 1:
        # Right-shifted Gaussian
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 2:
        # Two peaks asymmetric
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-10.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-10.0 * (x - 0.7) ** 2)
    elif choice == 3:
        # Ramp with oscillation
        x = np.linspace(0.0, 1.0, n)
        a = x * (1 - x) * (1 + 0.3 * np.sin(6 * np.pi * x))
    elif choice == 4:
        # Powers of position
        x = np.linspace(0.0, 1.0, n)
        beta = rng.uniform(0.5, 3.0)
        a = x ** beta
    elif choice == 5:
        # Inverse powers
        x = np.linspace(0.01, 1.0, n)
        beta = rng.uniform(0.5, 2.0)
        a = x ** (-beta)
    elif choice == 6:
        # Sine-based asymmetric
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** rng.uniform(1.0, 4.0)
    elif choice == 7:
        # Random smooth
        k = int(rng.integers(3, 15))
        coeffs = rng.normal(0, 1, k)
        x = np.linspace(0.0, 2 * np.pi, n)
        a = np.ones(n)
        for i in range(k):
            a += coeffs[i] * np.cos((i + 1) * x)
        a = np.maximum(a, 0.0)
    elif choice == 8:
        # Exponential decay
        x = np.arange(n) / n
        a = np.exp(-rng.uniform(2, 8) * x)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = a.ravel()
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - _global_t0
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Exponential gradient from parent (main fast optimizer) ===
    print("\n=== Phase 0: Exponential gradient from parent ===")
    n_base = len(best_seq) if best_seq is not None else 1600
    t0_phase = time.monotonic()
    t_phase0 = _global_time_budget * 0.30

    if best_seq is not None:
        a_init = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, sc_out = exponential_gradient_optimize(
            a_init, t_phase0,
            T_start=3.0, T_end=0.3,
            eta_start=0.2, eta_decay=0.99997,
            top_k=12, restart_stall=1500,
            verbose=True
        )
        do_try_update(a_out, f"EG_n{n_base}")

    # === Phase 1: LP Nesterov from EG result ===
    remaining = _global_time_budget - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 1: LP Nesterov n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 250.0)
    n_best = len(best_seq)
    if t1 > 30:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_init, t1, solver, verbose=True)
        do_try_update(a_out, f"LP_n{n_best}")

    # === Phase 2: Coordinate descent polish ===
    remaining = _global_time_budget - (time.monotonic() - _global_t0)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_init, t2, seed=12345)
        do_try_update(a_out, "cd_polish1")

    # === Phase 3: EG with different temperature schedule ===
    remaining = _global_time_budget - (time.monotonic() - _global_t0)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: EG different params n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20:
        # Perturb and run EG with different params
        a_perturbed = best_seq + rng.normal(0, 0.03 * best_seq.mean(), n_best)
        a_perturbed = np.maximum(a_perturbed, 0.0)
        a_init = a_perturbed * np.sqrt(2.0 * n_best) / a_perturbed.sum()
        a_out, sc_out = exponential_gradient_optimize(
            a_init, t3,
            T_start=1.0, T_end=0.1,
            eta_start=0.1, eta_decay=0.99999,
            top_k=20, restart_stall=3000,
            verbose=False
        )
        do_try_update(a_out, f"EG_perturbed_n{n_best}")

    # === Phase 4: Try different length ===
    remaining = _global_time_budget - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 4: Different length (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 30:
        n_alt = len(best_seq) + 200
        init_alt = interpolated_sequence(best_seq, n_alt)
        a_init = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = exponential_gradient_optimize(
            a_init, t4,
            T_start=2.0, T_end=0.2,
            eta_start=0.15, eta_decay=0.99998,
            top_k=10, restart_stall=2000,
            verbose=False
        )
        do_try_update(a_out, f"EG_altlen_n{n_alt}")
        if best_seq is not None and len(best_seq) != n_alt:
            # Also try slightly shorter
            n_alt2 = len(best_seq) - 200
            if n_alt2 > 500:
                remaining2 = _global_time_budget - (time.monotonic() - _global_t0)
                t4b = min(remaining2 * 0.05, 60.0)
                if t4b > 15:
                    init_alt2 = interpolated_sequence(best_seq, n_alt2)
                    a_init2 = init_alt2 * np.sqrt(2.0 * n_alt2) / init_alt2.sum()
                    a_out2, _ = exponential_gradient_optimize(
                        a_init2, t4b,
                        T_start=2.0, T_end=0.2,
                        eta_start=0.15, eta_decay=0.99998,
                        top_k=10, restart_stall=1000,
                        verbose=False
                    )
                    do_try_update(a_out2, f"EG_altlen2_n{n_alt2}")

    # === Phase 5: Final LP polish ===
    remaining = _global_time_budget - (time.monotonic() - _global_t0)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 120.0)
    if t5 > 20:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_init, t5, solver, verbose=False)
        do_try_update(a_out, f"LP_final_n{n_best}")

    # === Phase 6: Final coordinate descent (use all remaining time) ===
    remaining = _global_time_budget - (time.monotonic() - _global_t0)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t6 = remaining * 0.5
    if t6 > 10:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_init, t6, seed=99999)
        do_try_update(a_out, "cd_final")

    # === Phase 7: One more short CD with different seed ===
    remaining = _global_time_budget - (time.monotonic() - _global_t0)
    n_best = len(best_seq)
    t7 = min(remaining * 0.3, 60.0)
    if t7 > 5:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_init, t7, seed=55555)
        do_try_update(a_out, "cd_final2")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-_global_t0:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
