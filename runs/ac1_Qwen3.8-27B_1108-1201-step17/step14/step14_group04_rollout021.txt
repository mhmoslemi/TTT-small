def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        k_max = np.where(b >= M_curr * 0.999)[0]  # near-peak lags
        
        improved_this_pass = False
        
        # Random coordinate ordering
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute the linear and quadratic coefficients for a[i]
            # conv(a,a)[k] contribution from a[i]:
            # For each k, the coefficient of a[i] (linear) is:
            #   sum over j where j=i or k-j=i of a[k-j] or a[j]
            # Specifically:
            #   From j=i: a[i]*a[k-i] (if 0 <= k-i < n)
            #   From k-j=i, j=k-i: a[k-i]*a[i] (if 0 <= k-i < n and k-i != i)
            # Quadratic: a[i]^2 when k = 2i
            
            # For peak lags, compute the upper bound on a[i]
            ai = a[i]
            if ai < 1e-15:
                continue
            
            max_allowed = np.inf
            
            for k in k_max:
                # conv(a,a)[k] = fixed + linear_coeff * a[i] + quad_coeff * a[i]^2
                # fixed = b[k] - current contribution of a[i]
                # Current contribution of a[i] to b[k]:
                contrib = 0.0
                # j = i: a[i] * a[k-i] if 0 <= k-i < n
                if 0 <= k - i < n:
                    contrib += a[i] * a[k - i]
                # j = k-i (different from i): a[k-i] * a[i] if 0 <= k-i < n and k-i != i
                if 0 <= k - i < n and k - i != i:
                    contrib += a[k - i] * a[i]
                
                fixed = b[k] - contrib
                # Constraint: fixed + linear_coeff * a[i] + quad_coeff * a[i]^2 <= M_curr
                # linear_coeff and quad_coeff:
                lin_coeff = 0.0
                quad_coeff = 0.0
                if 0 <= k - i < n:
                    lin_coeff += a[k - i]  # from j=i term
                if 0 <= k - i < n and k - i != i:
                    lin_coeff += a[k - i]  # from j=k-i term
                if k == 2 * i:
                    quad_coeff = 1.0  # a[i]^2 term
            
            # Solve: quad_coeff * x^2 + lin_coeff * x + (fixed - M_curr) <= 0
            # Find max x >= 0 satisfying this
            c0 = fixed - M_curr
            if quad_coeff > 0:
                # a[i]^2 + lin_coeff * a[i] + c0 <= 0 (after dividing by quad_coeff)
                disc = lin_coeff * lin_coeff - 4 * c0
                if disc >= 0:
                    x_max = (-lin_coeff + np.sqrt(disc)) / 2.0
                    max_allowed = min(max_allowed, max(x_max, 0.0))
            else:
                if lin_coeff > 1e-15:
                    x_max = -c0 / lin_coeff
                    max_allowed = min(max_allowed, max(x_max, 0.0))
                # else: no constraint from this k
            
            if max_allowed < ai:
                a[i] = max(max_allowed, 0.0)
                improved_this_pass = True
        
        n_passes += 1
        
        # Normalize and check score
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a * norm / s_a
        score = fast_autocorr(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stall = 0
        elif not improved_this_pass:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): score={score:.8f}, best={best_score:.8f}")
    
    return best_a, best_score


def spectral_refine(a0, time_limit, seed=42):
    """Frequency-domain refinement: iteratively reduce spectral peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    pad_len = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()
        
        # Get the phase of the FFT at peak frequency
        a_pad = np.zeros(pad_len)
        a_pad[:n] = a
        A = rfft(a_pad)
        
        # The autocorrelation peak corresponds to the max of |A|^2 in a specific way
        # Actually for linear convolution, it's more complex. Let's use a simpler approach.
        
        # Find which lag k is the peak
        k_peak = np.argmax(b)
        
        # The peak is contributed by pairs (j, k_peak - j)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        
        # Move mass from high-contributing positions to low-contributing ones
        if contrib.max() > 0:
            src = idx[np.argmax(contrib)]
            # Find destination: positions not involved in any near-peak lag
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in np.argsort(b)[::-1][:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) > 0:
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
                dst = candidates_sorted[rng.integers(len(candidates_sorted))]
                
                for frac in [0.001, 0.003, 0.007, 0.015]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
                else:
                    stall += 1
                    if stall > 300:
                        break
            else:
                stall += 1
                if stall > 300:
                    break
        
        if stall > 300:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_initialization(n, kind='sin'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'sin':
        a = np.sin(np.pi * x) ** 1.5
    elif kind == 'sin2':
        a = np.sin(np.pi * x) ** 2.0
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'bump':
        a = np.exp(-5.0 * (x - 0.5) ** 2)
    elif kind == 'tri':
        a = 1.0 - np.abs(2.0 * x - 1.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a
