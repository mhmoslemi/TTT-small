def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def soft_subgrad(a, ac, top_k=30, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def nesterov_subgrad(a0, time_limit, top_k=30, beta=5.0, verbose=True):
    """Nesterov-accelerated mirror descent with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov variables
    y = a.copy()  # momentum point
    x_old = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        g = soft_subgrad(y, ac, top_k=top_k, beta=beta)
        
        # Mirror descent step
        step = 0.5 / (np.sqrt(t) + 1.0)
        log_y = np.log(y + 1e-30)
        log_new = log_y - step * g
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov update
        gamma_t = (1.0 + np.sqrt(1.0 + 4.0 * (t-1)**2)) / 2.0
        y_new = (1.0 + 1.0/gamma_t) * a_new - (1.0/gamma_t) * x_old
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        x_old = a_new
        a = a_new
        y = y_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        # Check y too
        M_y = fast_autocorr(y).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      NS t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def active_set_lp_fw(a0, time_limit, solver, verbose=True, tol=1e-6):
    """Frank-Wolfe with active-set LP: only include near-binding constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Active set: constraints within tol*M_curr of the max
        threshold = M_curr * (1.0 - tol)
        active = np.where(ac >= threshold)[0]
        
        # Always include the top few
        top_n = min(50, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_n]
        active = np.unique(np.concatenate([active, top_idx]))
        
        # Build reduced LP
        b_var = cp.Variable(n, nonneg=True)
        
        # Build only active constraint rows
        T_active = np.zeros((len(active), n))
        for row_idx, k in enumerate(active):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            T_active[row_idx, idx] = y[j]
        
        constraints = [T_active @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()

            best_a = x.copy()

            n_active = len(active)
            print(f"      AS-LP t={t} ({elapsed:.0f}s, active={n_active}): J={J:.6f}")

from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_subgrad(a, ac, top_k=40, beta=6.0):
    """Soft subgradient with exponential weighting of top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        # Handle the k=2i case correctly
        for ii in range(len(idx)):
            i_val = idx[ii]
            j_val = j[ii]
            if k_int == 2 * i_val:
                grad[i_val] += 2.0 * wi * a[i_val]
            else:
                grad[i_val] += 2.0 * wi * a[j_val]
    return grad

def nesterov_subgrad_opt(a0, time_limit, top_k=40, beta=6.0, verbose=True):
    """Nesterov-accelerated mirror descent with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_old = a.copy()
    gamma0 = 0.8

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        g = compute_subgrad(y, ac, top_k=top_k, beta=beta)

        step = gamma0 / (np.sqrt(t) + 1.0)
        log_y = np.log(y + 1e-30)
        log_new = log_y - step * g
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        # Nesterov momentum
        beta_t = (1.0 - np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_t = max(beta_t, -1.0)
        y_new = (1.0 + beta_t) * a_new - beta_t * x_old
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        x_old = a_new
        a = a_new
        y = y_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        M_y = fast_autocorr(y).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      NS t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

def cutting_plane_lp_best_response(y, M_curr, n, solver, max_sub_iters=10, init_active=None):
    """LP best response with cutting-plane: start with few constraints, add violations."""
    if init_active is not None:
        active = init_active.copy()
    else:
        ac = fast_autocorr(y)
        top_n = min(60, len(ac))
        active = np.argsort(ac)[::-1][:top_n]

    for sub_iter in range(max_sub_iters):
        b_var = cp.Variable(n, nonneg=True)

        # Build constraint matrix for active set
        T_active = np.zeros((len(active), n))
        for row_idx, k in enumerate(active):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            T_active[row_idx, idx] = y[j]

        constraints = [T_active @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None, active

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, active

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            return None, active
        g = b_val / b_sum

        # Check for violated constraints
        ac_check = fast_autocorr(g)
        violations = np.where(ac_check > M_curr + 1e-10)[0]

        if len(violations) == 0:
            return g, active

        # Add violated constraints (top 20)
        viol_sorted = violations[np.argsort(ac_check[violations])[::-1]][:20]
        active = np.unique(np.concatenate([active, viol_sorted]))
        # Keep only top 200 to limit LP size
        if len(active) > 200:
            ac_full = fast_autocorr(y)
            active_scores = np.zeros(len(ac_full))
            active_set = set(active.tolist())
            for ai in active:
                active_scores[ai] = ac_full[ai]
            active = np.argsort(active_scores)[::-1][:200]

    return g, active

def active_set_fw_optimize(a0, time_limit, solver, verbose=True):

    n = len(a0)

            break

        # LP best response with cutting plane

        )
        if g is None:
            g = a.copy()

        # Nesterov Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      FW t={t} ({elapsed:.0f}s, |A|={len(active_set)}): J={J:.6f}")

def batch_coordinate_polish(a0, time_limit, verbose=False):

            je = min(n - 1, k_int)

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination using peak sensitivity
            peak_indices = top_k[:5]
            best_dst = -1
            best_score = float('inf')

                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            if len(candidates) > 200:
                candidates = candidates[rng.permutation(len(candidates))[:200]]

                if dst == src:
                    continue
                score = 0.0
                for k2 in peak_indices:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    d1s = k2i - src
                    d2s = src - k2i
                    rem = 0.0
                    if 0 <= d1s < n:
                        rem += a[d1s]
                    if 0 <= d2s < n:
                        rem += a[d2s]
                    score += (add - rem)

                if score < best_score:
                    best_score = score
                    best_dst = dst

            if best_dst < 0:
                continue

            for frac in [0.02, 0.05, 0.1, 0.2, 0.35]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                if a_trial[src] < -1e-14:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      Coord ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def freq_init(n, seed=0):
    """Frequency-domain initialization with good autocorrelation properties."""
    rng = np.random.default_rng(seed)
    best = None
    best_score = float('inf')

    for trial in range(15):
        fft_size = 1 << (2 * n - 1).bit_length()
        freqs = np.arange(fft_size // 2 + 1)

        # Smooth amplitude with random modulation
        cutoff = rng.uniform(0.3, 0.8) * fft_size
        amp = np.exp(-0.5 * (freqs / cutoff) ** 2)
        mod_freq = rng.uniform(1, 5)
        amp *= 1.0 + 0.3 * np.sin(2 * np.pi * mod_freq * freqs / fft_size)
        amp = np.maximum(amp, 0.05)

        phase = 2 * np.pi * rng.random(fft_size // 2 + 1)
        phase[0] = 0.0

        X = amp * np.exp(1j * phase)
        X_full = np.zeros(fft_size, dtype=complex)
        X_full[:fft_size // 2 + 1] = X
        for i in range(1, fft_size // 2 + 1):
            X_full[fft_size - i] = np.conj(X[i])

        a = irfft(X_full, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.05:
            continue
        a = a / s
        m = fast_autocorr(a).max()
        score = 2.0 * n * m
        if score < best_score:
            best_score = score
            best = a.copy()

    if best is None:
        x = np.linspace(0, 1, n)
        best = np.sin(np.pi * x) ** 2
        best = best / best.sum()

    return best

def barrier_aided_optimize(a0, time_limit, solver, verbose=True):
    """Optimize with a barrier: minimize max(C[k] - M) for decreasing M."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    M = fast_autocorr(a).max()
    decay = 0.995

    for outer in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break

        # Inner: minimize max_k (C[k] - M) using subgradient
        a_cur = a.copy()
        for inner in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            ac = fast_autocorr(a_cur)
            # Subgradient of max(C[k] - M)
            peak_lags = np.where(ac >= ac.max() - 1e-6)[0]
            if len(peak_lags) == 0:
                peak_lags = np.array([np.argmax(ac)])

            g = np.zeros(n)
            n_peaks = len(peak_lags)
            for k in peak_lags:
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                idx = np.arange(i_start, i_end)
                j = k_int - idx
                for ii in range(len(idx)):
                    i_val = idx[ii]
                    j_val = j[ii]
                    if k_int == 2 * i_val:
                        g[i_val] += 2.0 * a_cur[i_val] / n_peaks
                    else:
                        g[i_val] += 2.0 * a_cur[j_val] / n_peaks

            step = 0.3 / (np.sqrt(inner + 1) + 1.0)
            a_new = a_cur - step * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            else:
                a_new = a_cur.copy()
            a_cur = a_new

        a = a_cur
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        # Decay barrier
        M = max(M * decay, M_new * 0.99)

        if verbose and outer % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Barrier outer={outer} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization combining:
    - Cutting-plane active-set LP Frank-Wolfe (Nesterov accelerated)
    - Nesterov soft-subgradient mirror descent
    - Barrier-aided optimization
    - Batch coordinate descent
    - Multi-scale exploration (1600 -> 2000 -> 2400 -> 3200)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}) t={elapsed:.0f}s")

    if best_seq is None:
        n_base = 1600
        best_seq = freq_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Freq init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: FW at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FW n={n0} ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, _ = active_set_fw_optimize(norm_sum1(best_seq), t1, solver, verbose=True)
        try_update(a_out, f"fw_n{n0}")

    # === Phase 2: Nesterov subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 2: Nesterousubgrad n={n_curr} ===")
    t2 = min(remaining * 0.15, 150.0)
    if t2 > 20:
        a_out, _ = nesterov_subgrad_opt(norm_sum1(best_seq), t2, top_k=40, beta=6.0, verbose=True)
        try_update(a_out, f"nsg_n{n_curr}")

    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 3: Coord n={n_curr} ===")
    t3 = min(remaining * 0.08, 80.0)
    if t3 > 10:
        a_out, _ = batch_coordinate_polish(norm_sum1(best_seq), t3, verbose=True)
        try_update(a_out, f"coord_n{n_curr}")

    # === Phase 4: Scale to 2000 ===
    n1 = 2000
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Scale {n1} ===")
    if remaining > 50:
        a_long = interp_seq(best_seq, n1)
        t4 = min(remaining * 0.15, 150.0)
        if t4 > 25:
            a_out, _ = nesterov_subgrad_opt(norm_sum1(a_long), t4, top_k=40, beta=6.0, verbose=True)
            try_update(a_out, f"nsg_n{n1}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and len(best_seq) <= 2100:
            t4b = min(remaining * 0.12, 120.0)
            a_out, _ = active_set_fw_optimize(norm_sum1(best_seq), t4b, solver, verbose=True)
            try_update(a_out, f"fw2_n{len(best_seq)}")

    # === Phase 5: Scale to 2400 ===
    n2 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Scale {n2} ===")
    if remaining > 50:
        a_long2 = interp_seq(best_seq, n2)
        t5 = min(remaining * 0.15, 130.0)
        if t5 > 25:
            a_out, _ = nesterov_subgrad_opt(norm_sum1(a_long2), t5, top_k=50, beta=5.0, verbose=True)
            try_update(a_out, f"nsg_n{n2}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = batch_coordinate_polish(norm_sum1(best_seq), min(remaining * 0.15, 80.0), verbose=True)
            try_update(a_out, f"coord2_n{len(best_seq)}")

    # === Phase 6: Barrier-aided at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Barrier n={n_best} ===")
    if remaining > 40:
        t6 = min(remaining * 0.15, 120.0)
        if t6 > 20:
            a_out, _ = barrier_aided_optimize(norm_sum1(best_seq), t6, solver, verbose=True)
            try_update(a_out, f"barrier_n{n_best}")

    # === Phase 7: Final FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final FW n={n_best} ===")
    if n_best <= 2500 and remaining > 40:
        t7 = min(remaining * 0.25, 200.0)
        if t7 > 25:
            a_out, _ = active_set_fw_optimize(norm_sum1(best_seq), t7, solver, verbose=True)
            try_update(a_out, f"final_fw_n{n_best}")

    # === Phase 8: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Coord n={n_best} ===")
    if remaining > 20:
        t8 = min(remaining * 0.35, 120.0)
        if t8 > 15:
            a_out, _ = batch_coordinate_polish(norm_sum1(best_seq), t8, verbose=True)
            try_update(a_out, f"final_coord_n{n_best}")

    # === Phase 9: Alternative init exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Alt init n={n_best} ===")
        alt = freq_init(n_best, seed=123)
        blend = 0.4 * norm_sum1(best_seq) + 0.6 * alt
        blend = norm_sum1(blend)
        t9 = min(remaining * 0.3, 80.0)
        if t9 > 20:
            a_out, _ = nesterov_subgrad_opt(blend, t9, top_k=50, beta=5.0, verbose=True)
            try_update(a_out, "alt_nsg")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
