def smooth_max_optimize(a0, time_limit, max_iters=50000, verbose=True):
    """Optimize using smooth max with adaptive beta + projected Adam."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr0 = 0.05
    betas = (0.9, 0.999)
    eps = 1e-8
    
    # Beta schedule: start at 1, increase to max_beta
    max_beta = max(50.0, n / 20.0)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and t < max_iters:
        # Determine current beta
        progress = min(1.0, t / max_iters)
        # Exponential schedule
        beta = np.exp(np.log(1.0) * (1 - progress) + np.log(max_beta) * progress)
        
        grad, b = smooth_max_grad(a, beta)
        
        t += 1
        m = betas[0] * m + (1 - betas[0]) * grad
        v = betas[1] * v + (1 - betas[1]) * grad**2
        m_hat = m / (1 - betas[0]**t)
        v_hat = v / (1 - betas[1]**t)
        
        lr = lr0 / (1.0 + 0.01 * t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SM t={t} beta={beta:.1f} ({elapsed:.1f}s): J={J:.6f}")
        
        if stall > max_stall:
            break
    
    return best_a, best_m
