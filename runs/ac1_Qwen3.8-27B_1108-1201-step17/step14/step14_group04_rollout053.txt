def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

            step = 1.0 / L

            staging = a_new.copy()

    return best_a, best_score


def build_conv_matrix(seq, n):
    """Build the (2n-1) x n convolution matrix T where (T @ x)_k = (seq * x)_k."""
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        js = max(0, j - (n - 1))
        je = min(2 * n - 2, j + (n - 1))
        for k in range(js, je + 1):
            T_mat[k, j] = seq[k - j]
    return T_mat


def lp_min_max_given_other(other, n, solver):
    """
    Given 'other' (length n), solve:
      min t
      s.t. T_other @ a <= t * 1
           sum(a) = 1
           a >= 0
    where T_other is the convolution matrix built from 'other'.
    Returns (a_opt, t_opt).
    """
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T_mat = build_conv_matrix(other, n)
    constraints = [
        T_mat @ a_var <= t_var,
        cp.sum(a_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    a_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_val.sum()
    if s < 1e-15:
        return None, None
    a_val = a_val / s
    t_val = float(t_var.value) if t_var.value is not None else fast_autocorr(a_val).max()
    return a_val, t_val


def lp_max_sum_given_bound(fixed, M, n, solver):
    """
    Given 'fixed' (length n) and bound M, solve:
      max sum(b)
      s.t. T_fixed @ b <= M
           b >= 0
    Returns b_opt.
    """
    b_var = cp.Variable(n, nonneg=True)
    T_mat = build_conv_matrix(fixed, n)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def minimax_lp_nesterov(a0, time_limit, solver, verbose=True):
    """
    Bidirectional minimax LP with Nesterov acceleration and ergodic averaging.
    At each step:
      1. Given current b, solve LP to find optimal a (minimizes max (a*b)_k)
      2. Given new a, solve LP to find optimal b (maximizes sum(b) s.t. (a*b)_k <= t)
    Use Nesterov momentum on the a-sequence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    # Initialize b
    b = a.copy()

    y_a = a.copy()  # Nesterov momentum for a
    avg_a = np.zeros(n)
    n_avg = 0

    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    prev_score = best_score + 1.0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Given b, find optimal a via LP
        a_new, t_val = lp_min_max_given_other(b, n, solver)
        if a_new is None:
            a_new = a.copy()
            t_val = fast_autocorr(a).max()

        # Step 2: Given a_new, find optimal b via LP
        M_curr = fast_autocorr(a_new).max()
        if M_curr < 1e-12:
            break
        b_new = lp_max_sum_given_bound(a_new, M_curr, n, solver)
        if b_new is None:
            b_new = a_new.copy()
        b = b_new / b_new.sum()

        # Nesterov update on a
        alpha = 2.0 / (t + 2.0)
        y_a_new = (1.0 - alpha) * y_a + alpha * a_new
        y_a_new = project_simplex(y_a_new, 1.0)

        if t == 1:
            a_mom = a_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_mom = y_a_new + momentum * (y_a_new - y_a)
            a_mom = project_simplex(a_mom, 1.0)

        y_a = y_a_new
        a = a_mom
        n_iters += 1

        # Track ergodic average
        avg_a += a
        n_avg += 1

        # Check scores
        if n_iters % 3 == 0:
            sc_a = fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            if n_avg % 10 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check / s_avg
                    sc_avg = fast_autocorr(avg_check).max()
                    if sc_avg < best_score:
                        best_score = sc_avg
                        best_a = avg_check.copy()
                        stall = 0

        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MM-LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, t_val={t_val:.8f}")

        if stall > 200 and n_iters > 50:
            # Restart from ergodic average
            if n_avg > 10:
                a = (avg_a / n_avg).copy()
                a = project_simplex(a, 1.0)
                y_a = a.copy()
                b = a.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                stall = 0

    return best_a, best_score


def frank_wolfe_polish(a0, time_limit, verbose=True):
    """
    Frank-Wolfe (conditional gradient) for polishing.
    At each step:
      - Compute subgradient: 2 * P_{k*} * a where k* = argmax autocorr
      - Find vertex e_i minimizing <grad, e_i> = grad_i
      - Update with line search
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        k_star = np.argmax(ac)
        M = ac[k_star]

        # Subgradient: 2 * P_{k*} * a
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[k_star - j]

        # Frank-Wolfe: find vertex e_i with minimum grad_i
        i_min = np.argmin(grad)
        e_i = np.zeros(n)
        e_i[i_min] = 1.0

        # Line search: minimize f((1-gamma)*a + gamma*e_i)
        best_gamma = 0.0
        best_trial_score = M
        for gamma in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = (1.0 - gamma) * a + gamma * e_i
            a_trial = project_simplex(a_trial, 1.0)
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_gamma = gamma

        if best_gamma == 0.0:
            stall += 1
            if stall > 500:
                break
            # Try different starting point
            i_alt = np.argmin(grad[:n//2]) if n > 2 else 0
            e_alt = np.zeros(n)
            e_alt[i_alt] = 1.0
            for gamma in [0.01, 0.05, 0.1]:
                a_trial = (1.0 - gamma) * a + gamma * e_alt
                a_trial = project_simplex(a_trial, 1.0)
                sc = fast_autocorr(a_trial).max()
                if sc < best_score:
                    best_trial_score = sc
                    best_gamma = gamma
                    break
            if best_gamma == 0.0 and stall > 2000:
                break
            continue

        a = (1.0 - best_gamma) * a + best_gamma * e_i
        a = project_simplex(a, 1.0)
        n_iters += 1

        sc = fast_autocorr(a).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    t0 = time.monotonic()
    t_iter = 1
    n_iters = 0
    stall = 0
    prev_best = best_score

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at y
        ac = fast_autocorr(y)
        k_star = np.argmax(ac)

        # Subgradient: 2 * P_{k*} * y
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = 2.0 * y[k_star - j]

        # Step size: 2 / (L * t) where L is subgradient norm bound
        # ||grad||_2 = 2 * ||y||_2 <= 2 (since sum(y)=1, y>=0)
        L = 2.5  # Slightly larger for safety
        step = 2.0 / (L * (t_iter + 2.0))

        # Nesterov update
        z = y - step * grad
        a_new = project_simplex(z, 1.0)

        if t_iter == 1:
            y_new = a_new.copy()
        else:
            momentum = (t_iter - 1.0) / (t_iter + 2.0)
            y_new = a_new + momentum * (a_new - a_mom)
            y_new = project_simplex(y_new, 1.0)

        a_mom = a_new
        y = y_new
        a = a_new
        t_iter += 1
        n_iters += 1

        # Periodic check
        if n_iters % 50 == 0:
            sc = fast_autocorr(y).max()
            if sc < best_score:
                best_score = sc
                best_a = y.copy()
                stall = 0
            else:
                stall += 1

            sc_a = fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                stall = 0

            if stall > 500:
                # Slow down
                t_iter = max(1, t_iter // 2)
                stall = 0

        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def targeted_local_search(a0, time_limit, seed=42):
    """
    Targeted local search: find peak-contributing pairs and redistribute mass.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find low-contribution destination
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Try multiple destinations
            cand_vals = a[candidates]
            sorted_idx = np.argsort(cand_vals)[:min(10, len(candidates))]

            for ci in sorted_idx:
                dst = candidates[ci]
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 600:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining LSE-FISTA, bidirectional LP,
    subgradient Nesterov, Frank-Wolfe, and targeted local search.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up (8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [15.0, 4.0, 1.0, 0.3, 0.1]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: Subgradient Nesterov (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        print(f"\n=== Phase 2: SubGrad-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_out, _ = subgrad_nesterov_fast(best_seq, t2, verbose=True)
        try_update(a_out, f"subgrad_n{n_best}")

    # === Phase 3: Bidirectional Minimax LP (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.35, 500.0)
    if t3 > 60:
        print(f"\n=== Phase 3: Minimax LP n={n_best} (budget={t3:.0f}s) ===")
        a_out, _ = minimax_lp_nesterov(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"mm_lp_n{n_best}")

    # === Phase 4: Frank-Wolfe Polish (12% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Frank-Wolfe n={n_best} (budget={t4:.0f}s) ===")
        a_out, _ = frank_wolfe_polish(best_seq, t4, verbose=True)
        try_update(a_out, f"fw_n{n_best}")

    # === Phase 5: Targeted local search (8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 10:
        print(f"\n=== Phase 5: Targeted search n={n_best} (budget={t5:.0f}s) ===")
        a_out, _ = targeted_local_search(best_seq, t5, seed=12345)
        try_update(a_out, f"targeted_n{n_best}")

    # === Phase 6: Try longer sequence with LP (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Minimax LP n={n2} (budget={t6:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_out, _ = minimax_lp_nesterov(init_long, t6, solver, verbose=True)
        try_update(a_out, f"mm_lp_n{n2}")

    # === Phase 7: Final Frank-Wolfe on best (7% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.07, 80.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Final FW n={n_best} (budget={t7:.0f}s) ===")
        a_out, _ = frank_wolfe_polish(best_seq, t7, verbose=True)
        try_update(a_out, f"fw_final_n{n_best}")

    # === Phase 8: Final targeted search (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.05, 60.0)
    if t8 > 5:
        print(f"\n=== Phase 8: Final targeted n={n_best} (budget={t8:.0f}s) ===")
        a_out, _ = targeted_local_search(best_seq, t8, seed=99999)
        try_update(a_out, "final_targeted")

    # === Phase 9: If time remains, more LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t9 = min(remaining * 0.5, 150.0)
        if t9 > 30:
            print(f"\n=== Phase 9: Extra LP n={n_best} (budget={t9:.0f}s) ===")
            a_out, _ = minimax_lp_nesterov(best_seq, t9, solver, verbose=False)
            try_update(a_out, "lp_extra")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            # Try one more FW polish at the very end if time permits
            remaining_final = time_budget - (time.monotonic() - t_global)
            if remaining_final > 15:
                a_out, _ = frank_wolfe_polish(best_seq, min(remaining_final * 0.5, 30.0), verbose=False)
                try_update(a_out, "ultimate_fw")

        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
