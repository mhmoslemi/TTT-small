**Idea: Alternating Direction Method (ADMM)**

The problem: minimize `max_k (a*a)_k` subject to `sum(a) = 1, a >= 0`.

Introduce variable `M >= (a*a)_k for all k`. Then:

ADMM would alternate between:
1. Given `a`, update `M = max((a*a).max(), M_old)` (trivial)
2. Given `M`, update `a = argmin (a*a)_k penalty + quadratic penalty for sum(a)=1 + non-negativity`

Step 2 is a non-convex QP (minimize the max of quadratics). Not directly solvable.

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=0):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    full_size = 2 * n - 1

    y = a.copy()
    y_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    active_set = None
    stall_count = 0
    stall_threshold = 40
    n_iters = 0

    for t in range(1, 5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive constraint screening
        if active_set is None or t % 50 == 0:
            tol = 0.05 + 0.02 * np.exp(-t / 300.0)
            active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
            if len(active_set) < 10:
                active_set = np.where(conv_y > M_curr * 0.7)[0]
            # Always include center and a few neighbors
            center = n - 1
            for c in range(max(0, center - 3), min(full_size, center + 4)):
                if c not in active_set:
                    active_set = np.append(active_set, c)

        # Solve LP
        b, duals = lp_best_response(y, M_curr, solver, active_set)

        # Violation check and constraint addition
        if b is not None and len(active_set) < full_size:
            all_k = np.arange(full_size)
            missing_k = np.setdiff1d(all_k, active_set, assume_unique=False)
            if len(missing_k) > 0:
                viol_vals = convolve_with(y, b, missing_k)
                viol_idx = np.where(viol_vals > M_curr * (1.0 + 1e-7))[0]
                if len(viol_idx) > 0:
                    new_k = missing_k[viol_idx]
                    active_set = np.union1d(active_set, new_k)
                    if len(active_set) > 0.9 * full_size:
                        active_set = all_k.copy()
                    b, duals = lp_best_response(y, M_curr, solver, active_set)
                    if b is None:
                        b, duals = lp_best_response(y, M_curr, solver, all_k)

        if b is None:
            if t % 10 == 0:
                b, duals = lp_best_response(y, M_curr, solver, np.arange(full_size))
            if b is None:
                # Fall back to subgradient step
                k_star = int(np.argmax(conv_y))
                grad = np.zeros(n)
                for m in range(n):
                    idx = k_star - m
                    if 0 <= idx < n:
                        grad[m] = 2.0 * y[idx]
                gn = np.linalg.norm(grad)
                if gn > 1e-15:
                    y_new2 = y - (2.0 / (np.sqrt(t) + 1.0)) * grad / gn
                    y_new2 = np.maximum(y_new2, 0.0)
                    ss = y_new2.sum()
                    if ss > 1e-15:
                        y_new2 = y_new2 * norm / ss
                        y_prev = y
                        y = y_new2
                continue

        b_sum = b.sum()
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.92)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0
            if verbose:
                print(f"    Restart at iter {t} ({elapsed:.0f}s), score={best_score:.10f}")

        if verbose and n_iters % 300 == 0:
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))
        M_curr = conv_y[k_star]

        # Subgradient of max_k (a*a)_k at k_star
        grad = np.zeros(n)
        for m in range(n):
            idx = k_star - m
            if 0 <= idx < n:
                grad[m] = 2.0 * y[idx]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        eta = 2.0 * norm / (np.sqrt(t) + 1.0)
        y_new = y - eta * grad / gn
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = min((t - 0.5) / (t + 1.5), 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t % 10000 == 0:
            print(f"    SG iter={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_coord_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(perm[idx])
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Gain vector for moving mass from i to j
                g = np.zeros(size)
                # Remove from i
                for k in range(max(0, i), min(size, i + n)):
                    src = k - i
                    if 0 <= src < n:
                        g[k] -= a[src]
                for k in range(max(0, n - 1 - i), min(size, n + i)):
                    src = n - 1 - k + i
                    if 0 <= src < n:
                        g[k] -= a[src]
                # Add to j
                for k in range(max(0, j), min(size, j + n)):
                    src = k - j
                    if 0 <= src < n:
                        g[k] += a[src]
                for k in range(max(0, n - 1 - j), min(size, n + j)):
                    src = n - 1 - k + j
                    if 0 <= src < n:
                        g[k] += a[src]

                # Try candidate deltas
                candidates = [ai * 0.3, ai * 0.5, ai * 0.7, ai]
                best_d = 0.0
                best_val = conv.max()

                for d in candidates:
                    if d <= 0:
                        continue
                    new_max = (conv + d * g).max()
                    # Diagonal corrections
                    di = 2 * i
                    if di < size:
                        new_max = max(new_max, conv[di] - 2 * a[i] * d + d * d)
                    dj = 2 * j
                    if dj < size:
                        new_max = max(new_max, conv[dj] + 2 * a[j] * d + d * d)
                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                    break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 20 == 0:
            print(f"    PCD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

    remaining = time_budget - (time.monotonic() - t_global)

        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_base))

def perturb_normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s
