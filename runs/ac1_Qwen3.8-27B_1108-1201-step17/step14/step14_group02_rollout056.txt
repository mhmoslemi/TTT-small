        log_a = np.log(a + 1e-30)

            log_a_new = log_a - step * g
            a_new = np.exp(log_a_new - log_a_new.max())
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            ac_new = fast_autocorr(a_new)
            val_new = ac_new.max()
            if val_new < best_trial_val:
                best_trial_val = val_new
                best_trial = a_new

        if best_trial is not None:
            a = best_trial
            if best_trial_val < best_ac_max:
                best_ac_max = best_trial_val
                best_a = a.copy()
        else:
            # Fallback: small fixed step
            step = gamma / (np.sqrt(t) + 1.0) * 0.5
            log_a_new = log_a - step * g
            a = np.exp(log_a_new - log_a_new.max())
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = np.maximum(avg_a / n_avg, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_ac_max:
                    best_ac_max = M_avg
                    best_a = avg_check.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_ac_max
            print(f"    MD t={t} ({elapsed:.1f}s): score={score:.6f}, beta={beta:.1f}, topk={top_k}")

    return best_a, best_ac_max


def coordinate_polish(a0, time_limit, verbose=True):
    """Subgradient-guided coordinate polish."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_m = ac.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(10, len(ac))]

        improved = False
        # Get subgradient to guide source selection
        g = compute_soft_subgradient(a, ac, top_k=20, beta=8.0)

        # Try moving from high-gradient positions to low-gradient positions
        src_candidates = np.argsort(g)[::-1][:min(50, n)]
        dst_candidates = np.argsort(g)[:min(50, n)]

        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates:
                if abs(int(src) - int(dst)) < 2:
                    continue
                # Try moving a fraction
                for frac in [0.001, 0.005, 0.01, 0.03, 0.08]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

    if verbose:
        score = 2.0 * n * best_m
        print(f"    Coord polish ({time.monotonic()-t0:.1f}s): score={score:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses fast mirror descent with adaptive soft subgradient,
    coordinate polish, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')
    best_n = 0

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_n = len(parent)
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score, best_n
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            best_n = len(a)
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n0 = 1600
        x = np.linspace(0, 1, n0)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        best_n = n0
        print(f"Default init: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent at parent length (45% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror descent n={best_n} ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_adaptive(a_in, t1, verbose=True, seed_beta=12.0, seed_topk=80)
        try_update(a_out, f"md_n{best_n}")

    # === Phase 2: Coordinate polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coord polish n={best_n} ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t2, verbose=True)
        try_update(a_out, f"coord_n{best_n}")

    # === Phase 3: Mirror descent with different parameters (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: MD alt params n={best_n} ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_adaptive(a_in, t3, verbose=False, seed_beta=6.0, seed_topk=150)
        try_update(a_out, f"md_alt_n{best_n}")

    # === Phase 4: Try longer length (15% time) ===
    n_new = min(best_n + 800, 3200)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Scale to n={n_new} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_long = interpolate_sequence(best_seq, n_new)
        a_in = norm_sum1(a_long)
        a_out, _ = mirror_descent_adaptive(a_in, t4 * 0.7, verbose=True, seed_beta=10.0, seed_topk=100)
        try_update(a_out, f"md_n{n_new}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, min(remaining * 0.4, 60.0), verbose=False)
            try_update(a_out, f"coord_long_n{len(a_out)}")

    # === Phase 5: Try shorter length (10% time) ===
    n_short = max(best_n // 2, 800)
    remaining = time_budget - (time.monotonic() - t_global)
    if best_n > 1000:
        print(f"\n=== Phase 5: Scale to n={n_short} ===")
        t5 = min(remaining * 0.10, 120.0)
        if t5 > 20:
            a_short = interpolate_sequence(best_seq, n_short)
            a_in = norm_sum1(a_short)
            a_out, _ = mirror_descent_adaptive(a_in, t5, verbose=False, seed_beta=14.0, seed_topk=60)
            try_update(a_out, f"md_short_n{n_short}")

    # === Phase 6: Different initialization (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Alt init n={n_best} ===")
        t6 = min(remaining * 0.10, 120.0)
        if t6 > 20:
            # Try raised cosine init
            x = np.linspace(0, 1, n_best)
            alt = (1.0 + np.cos(2.0 * np.pi * (x - 0.5))) / 2.0
            alt = np.maximum(alt, 0.0)
            alt = alt / alt.sum()
            # Blend with current best
            blend = 0.4 * norm_sum1(best_seq) + 0.6 * alt
            blend = norm_sum1(blend)
            a_out, _ = mirror_descent_adaptive(blend, t6 * 0.6, verbose=False, seed_beta=10.0, seed_topk=80)
            try_update(a_out, "alt_blend_md")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, min(remaining * 0.5, 40.0), verbose=False)
                try_update(a_out, "coord_alt")

    # === Phase 7: Final refinement (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final refinement n={n_final} ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent_adaptive(a_in, t7, verbose=True, seed_beta=8.0, seed_topk=120)
            try_update(a_out, f"final_md_n{n_final}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, min(remaining * 0.6, 80.0), verbose=True)
            try_update(a_out, f"final_coord_n{len(a_out)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
