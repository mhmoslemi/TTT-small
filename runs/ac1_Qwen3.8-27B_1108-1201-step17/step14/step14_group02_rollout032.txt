"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated mirror descent with softmax-weighted subgradient
2. LP-based Frank-Wolfe with warm starts
3. Coordinate descent polishing with optimal step selection
4. Multi-length scaling and multi-start exploration

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation (convolution with self) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Project onto probability simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad(a, ac, top_k=30, beta=6.0):
    """Compute softmax-weighted subgradient of max autocorrelation.
    
    Parameters:
    - a: sequence (length n)
    - ac: autocorrelation (length 2n-1)
    - top_k: number of active lags to consider
    - beta: temperature for softmax weighting
    
    Returns:
    - subgradient vector (length n)
    - max value of ac
    """
    n = len(a)
    K = min(top_k, len(ac))
    
    # Select top K lags
    top_idx = np.argsort(ac)[::-1][:K]
    top_vals = ac[top_idx]
    
    # Softmax weights
    max_val = top_vals[0]
    w = np.exp(beta * (top_vals - max_val))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K) / K
    else:
        w = w / w_sum
    
    # Compute subgradient
    # For lag k: d(conv[k])/d(a[i]) = 2*a[k-i] (if 0<=k-i<n and i!=k-i) or 2*a[i] (if i=k/2)
    # Simplified: for 0<=k<n, the gradient is 2*a[k-i] for i in [0,k]
    # For n<=k<=2n-2, the gradient is 2*a[k-i] for i in [k-n+1, n-1]
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        # For i in [i_start, i_end): grad[i] += 2*wi*a[k-i]
        # But we need to be careful: a[k-i] is valid iff 0 <= k-i < n
        # i_start = max(0, k-n+1) ensures k-i <= n-1
        # i_end = min(n, k+1) ensures k-i >= 0
        indices = np.arange(i_start, i_end)
        grad[indices] += 2.0 * wi * a[k - indices]
    
    return grad

def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution with self) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Project onto probability simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgrad(a, ac, top_k=30, beta=6.0):
    """Compute softmax-weighted subgradient of max autocorrelation.
    
    For lag k: d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n, else 0.
    """
    n = len(a)
    K = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:K]
    top_vals = np.maximum(ac[top_idx], 0.0)
    
    max_val = top_vals[0]
    if max_val < 1e-30:
        return np.zeros(n)
    
    w = np.exp(beta * (top_vals - max_val))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K) / K
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        indices = np.arange(i_start, i_end)
        grad[indices] += 2.0 * wi * a[k - indices]
    
    return grad

        if M < 1e-14:
            break

def nesterov_md_optimize(a0, time_limit, top_k=30, beta=6.0, verbose=False):
    """Nesterov-accelerated mirror descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # lookahead point
    gamma = 1.5
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M = ac.max()
        if M < 1e-14:
            break
        
        g = compute_subgrad(y, ac, top_k=top_k, beta=beta)
        step = gamma / (np.sqrt(t) * (1.0 + M * n))
        if step > 2.0:
            step = 2.0
        
        # Mirror step from y
        log_y = np.log(y + 1e-30)
        log_y_new = log_y - step * g
        log_y_new -= log_y_new.max()
        s = np.exp(log_y_new)
        s = norm_sum1(s)
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * s
        y_new = norm_sum1(y_new)
        
        # Extrapolated point
        if t == 1:
            x_new = s.copy()
        else:
            beta_nr = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_nr * (y_new - y)
            x_new = norm_sum1(x_new)
        
        # Evaluate both
        my = fast_autocorr(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()
        
        mx = fast_autocorr(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()
        
        y = y_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      NMD t={t} ({elapsed:.1f}s): J={J:.6f}")

            i_end = min(n, k + 1)

            if contrib.max() < 1e-20:
                continue

            best_dst = -1
            best_delta = 1e-10  # threshold for improvement

            n_cand = min(n, 300)

            for dst in candidates:

                    continue
                
                # Compute delta in autocorrelation at top lags

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving the integral."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a_out, _ = spectral_flatten(a_in, t1, verbose=True)
        try_update(a_out, f"spectral_n{n0}")
    
    # === Phase 2: Nesterov mirror descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: NMD n={n0} ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_md_optimize(a_in, t2, top_k=40, beta=5.0, verbose=True)
        try_update(a_out, f"nmd_n{n0}")
    
    # === Phase 3: Mirror descent with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_now = len(best_seq)
    print(f"\n=== Phase 3: MD2 n={n_best_now} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t3, top_k=50, beta=8.0, gamma_scale=0.8, verbose=True)
        try_update(a_out, f"md2_n{n_best_now}")
    
    # === Phase 4: LP Frank-Wolfe (if n is manageable) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_now = len(best_seq)
    if n_best_now <= 3000:
        print(f"\n=== Phase 4: LP n={n_best_now} ===")
        t4 = min(remaining * 0.20, 250.0)
        if t4 > 30:
            solver = get_lp_solver()
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True)
            try_update(a_out, f"lp_n{n_best_now}")
    
    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_now = len(best_seq)
    print(f"\n=== Phase 5: Coord n={n_best_now} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True)
        try_update(a_out, f"coord_n{n_best_now}")
    
    # === Phase 6: Scale to larger length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_target = max(n0 + 800, 3200)
    print(f"\n=== Phase 6: Scale to n={n_target} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        a_long = interpolate_sequence(best_seq, n_target)
        a_in = norm_sum1(a_long)
        a_out, _ = nesterov_md_optimize(a_in, t6, top_k=40, beta=6.0, verbose=True)
        try_update(a_out, f"nmd_n{n_target}")
    
    # === Phase 7: Final polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_now = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best_now} ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        a_in = norm_sum1(best_seq)
        # Alternate MD and coord
        t_md = t7 * 0.5
        t_coord = t7 * 0.5
        a_out, _ = mirror_descent_optimize(a_in, t_md, top_k=30, beta=4.0, gamma_scale=0.5, verbose=False)
        try_update(a_out, f"final_md_n{n_best_now}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, remaining * 0.7, verbose=True)
            try_update(a_out, f"final_coord_n{n_best_now}")
    
    # === Phase 8: Try alternative initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best_now = len(best_seq)
        print(f"\n=== Phase 8: Alt init n={n_best_now} ===")
        # Try a "U-shaped" initialization blended with current best
        x = np.linspace(0, 1, n_best_now)
        u_shape = (1.0 + np.cos(2 * np.pi * x)) / 2.0
        u_shape = u_shape / u_shape.sum()
        blend = 0.3 * norm_sum1(best_seq) + 0.7 * u_shape
        blend = norm_sum1(blend)
        
        t8 = min(remaining * 0.3, 80.0)
        if t8 > 10:
            a_out, _ = nesterov_md_optimize(blend, t8, top_k=30, beta=5.0, verbose=False)
            try_update(a_out, f"alt_n{n_best_now}")
    
    # === Phase 9: Ultra-fine coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_now = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Ultra coord n={n_best_now} ===")
        t9 = remaining * 0.8
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t9, verbose=True)
        try_update(a_out, f"ultra_coord_n{n_best_now}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
