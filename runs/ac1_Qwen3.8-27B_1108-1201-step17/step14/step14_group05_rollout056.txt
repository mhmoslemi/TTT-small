One more thought: the LP formulation. The standard best response LP is:
max sum(b) subject to (a*b)_k <= M for all k, b >= 0

But I can also formulate it as:
min t subject to (a*b)_k <= t for all k, b >= 0, sum(b) = 1

These are equivalent (the first maximizes the "gain" while keeping the output bounded, the second minimizes the peak while normalizing the input).

For the Nesterov iteration, the idea is:
1. Current point: y (normalized to sum = sqrt(2n))
2. Compute M = max(y*y)
3. Solve LP: find g that maximizes sum(g) subject to (y*g)_k <= M, g >= 0
4. Normalize g to sum = sqrt(2n)
5. Update: y_new = (1-alpha)*y + alpha*g (with Nesterov momentum)

The key insight is that if we could solve the LP exactly (with all constraints), the iteration would monotonically decrease M. With constraint reduction, it might not be monotone, but it should still make progress.

Let me also consider: what if instead of the LP, I use a "power iteration" style update?
g_j = M / max_k (T_k @ e_j) ... this is the component-wise best response.

Actually, the component-wise best response for the constraint (a*b)_k <= M is:
For each j, b_j can be at most M / max_k T[k,j] = M / max_k a[k-j]

So b_j <= M / max_k a[k-j] = M / max(a) (since the max of a[k-j] over k is just max of a).

This gives b_j <= M / max(a) for all j, which is a uniform upper bound. Not very useful.

A better approach: for each j, the tightest constraint is the one where a[k-j] is largest, i.e., k = argmax_k a[k-j] = argmax_i a[i] + j. So the binding constraint for variable b_j is:
b_j * a[argmax_i a[i]] + (other terms) <= M

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv(a, b):
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lse_gradient(a, temp):
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n), max_b
    shifted = np.clip(b - max_b, -50.0, 0.0)
    w = np.exp(shifted / temp)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), max_b
    w = w / w_sum
    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad, max_b

def fast_lp(a, M, K=10, solver=None):

    b_val = None

        try:

            prob.solve(solver=solver, verbose=False)

            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        if b_var.value is None:
            return None

            active = list(all_cand[np.argsort(vals)[::-1][:2 * K]])

def peak_shave(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:

    else:
        a = a * norm / s0

    t0 = time.monotonic()
    stall = 0
    n_improve = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        p = int(np.argmax(b))
        js = max(0, p - n + 1)
        je = min(n - 1, p)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p - idx]
        if contrib.max() < 1e-18:
            stall += 1
            if stall > 200:
                break
            continue
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        non_contrib[src] = True
        nc_idx = np.where(non_contrib)[0]
        if len(nc_idx) == 0:
            nc_idx = np.array([i for i in range(n) if i != src])
        # Pick lowest-value non-contributing position
        dst = nc_idx[np.argmin(a[nc_idx])]
        best_frac = 0
        best_trial = best_M

            amt = src_val * frac

            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial:
                best_trial = sc
                best_frac = frac
        if best_frac > 0:
            a_trial = a.copy()
            amt = a_trial[src] * best_frac
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial * norm / a_trial.sum()
            sc = fast_autocorr(a_trial).max()
            if sc < best_M:
                a = a_trial
                best_M = sc
                best_a = a.copy()
                n_improve += 1
                stall = 0
            else:
                stall += 1
                # Annealing
                elapsed_frac = (time.monotonic() - t0) / time_limit
                Tanneal = best_M * 0.0005 * (1.0 - elapsed_frac)
                if Tanneal > 1e-12 and rng.random() < np.exp(-(sc - best_M) / Tanneal):
                    a = a_trial
                    stall += 1
        else:
            stall += 1
        if stall > 100:
            # Random perturbation escape
            n_pert = max(3, n // 100)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * rng.uniform(0.05, 0.3)
                    a[i] -= amt
                    a[j] += amt

                best_a = a.copy()
            stall = 0
        if verbose and n_improve > 0 and n_improve % 200 == 0:
