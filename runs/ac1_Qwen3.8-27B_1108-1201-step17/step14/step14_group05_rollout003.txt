def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s

            continue

    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T_subset(a, k_indices):
    """Build rows of the (2n-1) x n convolution matrix for given lag indices."""
    n = len(a)
    T_sub = np.zeros((len(k_indices), n))
    for i, k in enumerate(k_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T_sub[i, js:je + 1] = a[k - np.arange(js, je + 1)]
    return T_sub


def lp_best_response(a, M, solver, active_idx, max_active=200):
    """
    Solve the LP: max sum(b) s.t. T b <= M, b >= 0
    using only the specified active constraints.
    Returns b (or None on failure).
    """
    n = len(a)
    T_sub = build_T_subset(a, active_idx)
    M_vec = np.full(len(active_idx), M)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M_vec]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def find_active_constraints(a, M, tol_frac=0.05, max_active=200):
    """Find constraint indices where the convolution is close to M."""
    b = fast_autocorr(a)
    tol = M * (1.0 - tol_frac)
    active = b >= tol
    n_active = int(active.sum())
    if n_active > max_active:
        active_idx = np.where(active)[0]
        vals = b[active_idx]
        top = active_idx[np.argsort(vals)[::-1][:max_active]]
        active = np.zeros(len(b), dtype=bool)
        active[top] = True
    return np.where(active)[0], b.max()

    y = a.copy()
    prev_y = a.copy()

        if len(active_idx) < 5:

        # Solve LP
        b = lp_best_response(y, M_curr, solver, active_idx, max_active=max_active)
        if b is None:
            # Fallback: use y itself as a valid direction
            b = y.copy()
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = y.copy()
            b_sum = b.sum()
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        t = n_iters + 1
        alpha = 2.0 / (t + 2.0 - cur_gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - cur_gamma) / (t + 2.0 - cur_gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if stall > 0:
            stall += 1

        # Adaptive gamma
        if len(M_hist) > 50:
            recent = M_hist[-30:]
            diffs = np.abs(np.diff(recent))
            osc = diffs.mean() / max(recent[0], 1e-15)

            js = max(0, p_star - n + 1)

def exact_subgradient_nesterov(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated exact subgradient method for min max_k (conv(a,a))_k
    subject to sum(a) = sqrt(2n), a >= 0.
    Uses exact subgradient: grad_j = a[k*-j] + a[j] for the maximizing k*.
    """

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stag = 1.0  # Step size scale
    warm = 0

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        t = n_iters

        b = fast_autocorr(y)
        max_b = b.max()

        # Find all near-max peaks (within 1% of max)
        tol = 0.01 * max_b
        peak_indices = np.where(b >= max_b - tol)[0]
        # Sample at most 10 peaks for averaging
        if len(peak_indices) > 10:
            rng = np.random.default_rng(t)
            peak_indices = peak_indices[rng.choice(len(peak_indices), 10, replace=False)]

        # Compute exact subgradient (average over near-max peaks)
        grad = np.zeros(n)
        for p in peak_indices:
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            for j in range(js, je + 1):
                grad[j] += (y[p - j] + y[j])
        grad /= len(peak_indices)
        # Project onto tangent space of {sum = norm}
        grad -= np.mean(grad)

        # Adaptive step size (Nesterov with Polyak-like)
        # f(y) - f(a_best) as gap estimate
        gap = max(max_b - best_score, 1e-10)
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        step = stag * gap / grad_norm_sq
        step = min(step, 2.0 * n / max_b)  # Cap

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * (y - step * grad)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = t / (t + 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stag = min(stag * 1.02, 5.0 * n / max_b)
        else:
            warm += 1
            if warm % 100 == 0:
                stag *= 0.9

        # Periodic restart
        if n_iters % 500 == 0:
            rng = np.random.default_rng(n_iters)
            perturb = rng.normal(0, 0.01 * y.mean(), n)
            y = np.maximum(y + perturb, 0)
            s_y = y.sum()
            if s_y > 1e-15:

            prev_y = y.copy()

def mirror_descent(a0, time_limit, verbose=True):
    """
    Mirror descent (exponential weights) for min max_k (conv(a,a))_k
    subject to sum(a) = sqrt(2n), a >= 0.
    Update: a_new[j] = a[j] * exp(-eta * grad[j])
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    # Ensure strictly positive for mirror descent
    a = np.maximum(a, norm / n * 0.01)
    a = a * norm / a.sum()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    eta = 0.1

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        n_iters += 1

        b = fast_autocorr(a)
        max_b = b.max()

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(b)[::-1][:top_k]

        # Softmax-weighted subgradient
        T_temp = max(0.01 * max_b, 1e-10)
        w = np.exp((b[peak_indices] - max_b) / T_temp)
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, p in zip(w, peak_indices):
            if wi < 1e-6:
                continue
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            for j in range(js, je + 1):
                grad[j] += wi * (a[p - j] + a[j])
        grad -= np.mean(grad)

        # Mirror descent update: a * exp(-eta * grad)
        # Scale eta so the update is reasonable
        grad_max = np.abs(grad).max()
        if grad_max < 1e-20:
            break
        eta_eff = eta / grad_max
        a_new = a * np.exp(-eta_eff * grad)
        a_new = np.maximum(a_new, norm / n * 0.001)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            eta = min(eta * 1.01, 10.0)
        else:
            eta *= 0.95

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={max_b:.6f}, eta={eta:.4f}")

    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Coordinate ascent peak shaving: move mass from high-contributing to low-contributing positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Contributing indices
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        max_contrib_idx = idx[np.argmax(contrib)]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 50:
                break
            continue

        # Try moving from max_contrib_idx to various destinations
        src = max_contrib_idx
        src_val = a[src]
        if src_val < 1e-15:
            n_stall += 1
            continue

        # Find best destination
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        non_contrib[src] = True
        nc_idx = np.where(non_contrib)[0]

        if len(nc_idx) == 0:
            nc_idx = np.array([i for i in range(n) if i != src])

        # Try a few random destinations and a few low-value ones
        candidates = list(set(list(nc_idx[:5]) + list(nc_idx[rng.integers(len(nc_idx), size=min(10, len(nc_idx)))])))
        candidates = [c for c in candidates if c != src]

        best_trial = max_b
        best_move = None
        for dst in candidates:
            for frac in [0.1, 0.2, 0.3]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_trial:
                    best_trial = sc
                    best_move = (src, dst, amt)

        if best_move is not None:
            src, dst, amt = best_move
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0)
            s_t = a.sum()
            if s_t > 1e-15:
                a = a * norm / s_t
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                n_stall += 1
                # Annealing
                T = best_score * 0.0005 * (1.0 - elapsed_frac)
                if rng.random() < np.exp(-(sc - best_score) / max(T, 1e-15)):
                    n_stall += 1
                else:
                    # Revert
                    a[dst] -= amt
                    a[src] += amt
                    n_stall += 1
        else:
            n_stall += 1

        # Escape: larger random perturbation
        if n_stall > 100:
            n_pert = max(3, n // 100)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * rng.uniform(0.05, 0.3)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            s_t = a.sum()
            if s_t > 1e-15:
                a = a * norm / s_t
            sc = fast_autocorr(a).max()
            if sc < best_score * 1.05:
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        # Raised cosine squared
        best_seq = (0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP-Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: LP-Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, max_active=150, gamma=1.8)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Exact subgradient Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: SubGrad-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc_out = exact_subgradient_nesterov(a_norm, t2, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 3: LP-Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP-Nesterov r2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, max_active=120, gamma=1.9)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Mirror Descent n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc_out = mirror_descent(a_norm, t4, verbose=True)
        do_try_update(a_out, f"mirror_n{n_cur}")

    # === Phase 5: Try longer n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = n_base + 100
    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: LP-Nesterov n={n2} (budget={t5:.0f}s) ===")
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, max_active=130, gamma=1.8)
        do_try_update(a_out, f"lp_n{n2}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: PeakShave n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc_out = peak_shave(a_norm, t6, seed=12345, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t7, solver, verbose=True, max_active=100, gamma=2.0)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc_out = peak_shave(a_norm, t8, seed=99999, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
