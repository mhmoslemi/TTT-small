def lp_best_response(a, M, solver):

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:

    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0

        a = a_new

        if score_y < best_score:

            iter_since_improve = 0

            # We want g[i] = sum_{m=0}^{n-1} a[m] * w[m+i] (with w having length 2n-1)
            # This is the "upper triangular" part of the outer product summed.
            # Equivalently, g = ifft(FFT(a_padded) * conj(FFT(w_padded))) ... no.

            # Simplest: g[i] = sum_m a[m] * w[m+i]
            # Let's define w_full of length 2n-1. Then g is the cross-correlation
            # of a (length n) with w (length 2n-1) evaluated at lags 0 to n-1.
            # crosscorr[lag] = sum_m a[m] * w[m+lag]
            # In FFT terms: ifft(conj(A_fft) * W_fft) gives the cross-correlation
            # where crosscorr[lag] = sum_m a[m] * conj(w[m-lag+N]) ...

            # I'll just compute it directly using the property that:
            # sum_m a[m] * w[m+i] = (a * w_rev_full)[n-1+i] where w_rev_full is reversed
            # Actually, let me just use a straightforward numpy approach for clarity,
            # then optimize with FFT if needed.

            # For n=1600, the direct computation is O(n*(2n-1)) = O(2n^2) ~ 5M ops per step.
            # That's fine for a few hundred steps.
            # But let me use FFT:
            # g[i] = sum_{m=0}^{n-1} a[m] * w[m+i]
            # = sum_{m=0}^{n-1} a[m] * w[m+i]
            # Let v = w of length 2n-1, pad to length 4n-2 (next power of 2)
            # g = ifft(FFT(a_padded) * FFT(v_padded)) ... no, that gives convolution.

            # OK let me think again. We want the "anti-convolution":
            # g[i] = sum_m a[m] * w[m+i]
            # This equals sum_m a[m] * w[m+i] = (a * w_rev)[2n-2-i] where w_rev[k] = w[2n-2-k]
            # because (a * w_rev)[k] = sum_m a[m] * w_rev[k-m] = sum_m a[m] * w[2n-2-k+m]
            # Set k = 2n-2-i: = sum_m a[m] * w[2n-2-(2n-2-i)+m] = sum_m a[m] * w[i+m]
            # But we need the sum to be over valid m (0<=m<n and 0<=i+m<2n-1).
            # Since 0<=i<n and 0<=m<n, we have 0<=i+m<=2n-2, so it's always valid!
            # So g[i] = (a * w_rev)[2n-2-i] where w_rev[k] = w[2n-2-k]

            w_rev = w[::-1]
            awr_fft = rfft(a, fft_size) * rfft(w_rev, fft_size)
            conv_awr = irfft(awr_fft, fft_size)[:size]
            g = (2.0 / eps) * conv_awr[size - 1 - np.arange(n)]

            # Projected gradient step
            # We need to minimize f(a) subject to sum(a) = norm, a >= 0
            # Use backtracking line search
            best_step = 1.0
            for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
                step = best_step * step_mult / (1.0 + 0.01 * it)
                a_trial = a - step * g
                a_trial = np.maximum(a_trial, 0.0)
                s_tr = a_trial.sum()
                if s_tr < 1e-15:
                    continue
                a_trial = a_trial * norm / s_tr
                trial_conv = fast_autocorr(a_trial)
                trial_max = trial_conv.max()
                if trial_max < max_conv:
                    a = a_trial
                    break
            else:
                # No improvement, try smaller step
                step = 0.01 / (1.0 + 0.1 * it)
                a_new = a - step * g
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new * norm / s_new

            conv = fast_autocorr(a)
            cur_max = conv.max()
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGr grad stage {si+1}/{n_stages} eps={eps:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_polish(a0, time_limit, rng, verbose=True):
    """Fine coordinate/pair descent for final polish."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            # Try moving mass from i to neighbors
            for offset in range(1, 15):
                j = (i + offset) % n
                if j == i:
                    continue
                if a[i] < norm * 1e-12:
                    break

                # Compute delta vector for moving small mass d from i to j
                # New conv = old conv + d * (delta)
                # delta accounts for the change in all convolution terms
                delta_vec = np.zeros(size)

                # Terms involving a[i] decrease by d:
                # c[k] has term a[i]*a[k-i] and a[k-i]*a[i] (if valid)
                # Decrease: -d * a[k-i] for each k where k-i is valid (0<=k-i<n)
                # i.e., for k in [i, i+n-1]: delta -= d * a[k-i]
                lo1 = i
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    delta_vec[lo1:hi1] -= a[:hi1 - lo1]

                # Terms involving a[j] increase by d:
                # for k in [j, j+n-1]: delta += d * a[k-j]
                lo2 = j
                hi2 = min(size, j + n)
                if hi2 > lo2:
                    delta_vec[lo2:hi2] += a[:hi2 - lo2]

                # Double-count correction: at k where both i and j contribute
                # Actually the above is correct: c[k] = sum_m a[m]*a[k-m]
                # After change: c_new[k] = sum_m a_new[m]*a_new[k-m]
                # = c_old[k] - d*a[k-i] - d*a[i-(k)] ... 

                # Wait, let me be more careful.
                # a_new[i] = a[i] - d, a_new[j] = a[j] + d, others same.
                # c_new[k] = sum_m a_new[m] * a_new[k-m]
                # = c_old[k] + d*(a_new[k-j] - a[k-j]) + (-d)*(a_new[k-i] - a[k-i]) + cross terms
                # = c_old[k] + d*a[k-j] - d*a[k-i] + d^2 * (terms with both i and j)
                # The d^2 terms: if i = k-j and j = k-i (i.e., k=i+j), then we have d^2
                # if i = k-i (i.e., k=2i), the -d term becomes -2d*a[i]... 

                # Actually for small d, ignoring d^2:
                # c_new[k] ≈ c_old[k] + d*a[k-j] - d*a[k-i] (with a[k-·] = 0 if out of range)
                # So delta_vec[k] = a[k-j] - a[k-i] (where defined)

                # Let me redo:
                delta_vec = np.zeros(size)
                # Contribution from a[j] increasing by d:
                # c[k] gets +d * a_new[k-j] ≈ +d * a[k-j] for k in [j, j+n-1]
                lo2 = j
                hi2 = min(size, j + n)
                if hi2 > lo2:
                    delta_vec[lo2:hi2] += a[:hi2 - lo2]
                # Contribution from a[i] decreasing by d:
                # c[k] gets -d * a_new[k-i] ≈ -d * a[k-i] for k in [i, i+n-1]
                lo1 = i
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    delta_vec[lo1:hi1] -= a[:hi1 - lo1]

                # For diagonal terms (k=2i or k=2j), there's an additional -d^2 or +d^2
                # but for small d we ignore this.

                # Find best d > 0 that reduces max(conv + d*delta_vec)
                best_d = 0.0
                best_val = conv.max()
                n_samples = 10
                max_d = min(a[i] * 0.99, norm * 0.1)
                if max_d < 1e-15:
                    continue
                deltas = np.linspace(0, max_d, n_samples)
                for d in deltas:
                    if d <= 0:
                        continue
                    new_conv = conv + d * delta_vec
                    # Correct diagonal terms (second order)
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                    new_max = new_conv.max()
                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        # Undo
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPolish pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LP Nesterov - main workhorse (55% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = remaining * 0.55
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Smooth gradient (25% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Smooth Gradient n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smooth_gradient_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"smooth_n{n_best}")

    # Phase 3: Coordinate polish (12% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.12
    print(f"\n=== Phase 3: Coord Polish n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_polish(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"coord_n{n_best}")

    # Phase 4: Second LP Nesterov round (refine after polish)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.30
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Final smooth + coord
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        t5a = min(remaining * 0.3, 60.0)
        t5b = min(remaining * 0.3, 60.0)
        print(f"\n=== Phase 5: Final polish n={n_best} ===")
        if t5a > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = smooth_gradient_optimize(a_norm, t5a, verbose=False)
            do_try_update(a_out, f"smooth2_n{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)
        if t5b > 10 and remaining > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = coord_polish(a_norm, min(t5b, remaining - 5), rng, verbose=False)
            do_try_update(a_out, f"coord2_n{n_best}")

    # Phase 6: Try alternate length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = n_base + 200
        t6 = min(remaining * 0.6, 120.0)
        print(f"\n=== Phase 6: Alternate length n={n_alt} (budget={t6:.0f}s) ===")
        if t6 > 30:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            a_out_alt, _ = nesterov_lp_optimize(a_norm_alt, t6 * 0.7, solver, verbose=False)
            do_try_update(a_out_alt, f"lp_n{n_alt}")
            if a_out_alt is not None:
                remaining2 = time_budget - (time.monotonic() - t_global)
                if remaining2 > 20:
                    a_norm_alt2 = a_out_alt * np.sqrt(2.0 * n_alt) / a_out_alt.sum()
                    a_out_alt2, _ = smooth_gradient_optimize(a_norm_alt2, min(remaining2 * 0.5, 30.0), verbose=False)
                    do_try_update(a_out_alt2, f"smooth_n{n_alt}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
