def simplex_project(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum} via bisection on threshold."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = n - 1
    for k in range(n - 1, 0, -1):
        if (v_sorted[k] > 0) and (cumsum[k] - k * v_sorted[k] > target_sum):
            rho = k
            break
    if v_sorted[rho] <= 0:
        theta = target_sum / n
    else:
        theta = (cumsum[rho] - target_sum) / (rho + 1)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 1e-15:
        result *= target_sum / s
    return result

def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(size):
        # (a*b)(k) = sum_j a[j] * b[k-j] where 0<=j<n and 0<=k-j<n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            expr = sum(a[j] * b_var[k - j] for j in range(j_lo, j_hi + 1))
            constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_fast(a, M, solver):

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP Nesterov with adaptive restarts and iterate averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    n_lp_calls = 0
    stuck_count = 0
    last_improve_iter = 0
    iterate_history = []
    history_window = 200

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        M_curr = float(fast_autocorr(y).max())
        if M_curr < 1e-12:
            break

        b = lp_best_response_fast(y, M_curr, solver)
        n_lp_calls += 1
        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y_prev)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        # Track best from y and a
        score_y = float(fast_autocorr(y).max())
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve_iter = n_iters

        score_a = float(fast_autocorr(a).max())
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve_iter = n_iters

        # Maintain iterate history for averaging
        iterate_history.append(y.copy())
        if len(iterate_history) > history_window:
            iterate_history.pop(0)

        # Periodic averaging check
        if n_iters % 50 == 0:
            avg_y = np.mean(iterate_history, axis=0)
            s_avg = avg_y.sum()
            if s_avg > 1e-15:
                avg_y = avg_y * norm / s_avg
                score_avg = float(fast_autocorr(avg_y).max())
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_y.copy()
                    last_improve_iter = n_iters

        # Adaptive restart: if no improvement for many iterations, restart
        if n_iters - last_improve_iter > 200:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            iterate_history = []
            last_improve_iter = n_iters

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} lp={n_lp_calls} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final averaging
    if len(iterate_history) > 10:
        avg_y = np.mean(iterate_history, axis=0)
        s_avg = avg_y.sum()
        if s_avg > 1e-15:
            avg_y = avg_y * norm / s_avg
            score_avg = float(fast_autocorr(avg_y).max())
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_y.copy()

    return best_a, best_score


def compute_fft_gradient(a, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    max_conv = conv.max()
    shifted = (conv - max_conv) / T
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 2 * (a_rev * w)[n-1+i] where a_rev = a[::-1]
    a_rev = a[::-1]
    # Use FFT for the convolution
    fft_size = 1 << (n + size - 1 - 1).bit_length()
    A_fft = rfft(a_rev, fft_size)
    W_fft = rfft(w, fft_size)
    conv_aw = irfft(A_fft * W_fft, fft_size)[:n + size - 1]
    # conv_aw has length n + (2n-1) - 1 = 3n-2
    # grad[i] = 2 * conv_aw[n-1+i]
    grad = 2.0 * conv_aw[n - 1:2 * n - 1]
    return grad, conv, w, float(max_conv)


def smoothed_gd_optimize(a0, time_limit, verbose=True):
    """Smoothed gradient descent with Adam optimizer and temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    # Temperature schedule: start high, anneal to low
    n_stages = 8
    temps = np.geomspace(5.0, 0.1, n_stages)
    steps_per_stage = 50

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        # Adjust steps per stage based on remaining time
        stages_left = n_stages - si
        steps_this = max(10, min(100, int(remaining / max(stages_left, 1) * 1.5)))

        for it in range(steps_this):
            if time.monotonic() - t0 > time_limit:
                break

            grad, conv, w, max_conv = compute_fft_gradient(a, T)

            # Adam step
            lr = norm / (n * 0.5) / (1.0 + 0.05 * it)
            t_adam = it + 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Project onto simplex
            a_new = simplex_project(a_new, norm)

            # Check improvement
            score = float(fast_autocorr(a_new).max())
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SGD stage T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent with grid-based 1D optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # Try several j candidates within window
            best_delta = 0.0
            best_j = -1
            best_new_max = best_score + 1e-10
            n_grid = 8

            for offset in range(1, window + 1):
                j = (i + offset) % n
                if j == i:
                    continue
                if best_new_max < conv.max() - 0.1 * (conv.max() - norm / n):
                    break  # Already improved a lot, stop early

                # Compute direction: delta_conv[k] = 2*delta*(a[k-j] - a[k-i])
                # For k in [0, 2n-2]:
                dir_vec = np.zeros(size)
                # Contribution from removing mass at i: -2*a[k-i] for valid k-i
                lo1 = i
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    dir_vec[lo1:hi1] -= a[0:hi1 - lo1]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n - 1 + i + 1)
                if hi2 > lo2:
                    src = n - 1 - np.arange(hi2 - lo2)
                    dir_vec[lo2:hi2] -= a[src]

                # Contribution from adding mass at j: +2*a[k-j] for valid k-j
                lo3 = j
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    dir_vec[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n - 1 + j + 1)
                if hi4 > lo4:
                    src4 = n - 1 - np.arange(hi4 - lo4)
                    dir_vec[lo4:hi4] += a[src4]

                dir_vec *= 2.0

                # Grid search for best delta
                max_delta = a[i] * 0.99
                for d_frac in np.linspace(0.1, 1.0, n_grid):
                    d = max_delta * d_frac
                    new_conv = conv + d * dir_vec
                    # Add delta^2 correction
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] + d * dir_vec[2 * i] + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + d * dir_vec[2 * j] - d * d
                    new_max = float(new_conv.max())
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_delta = d
                        best_j = j

            if best_delta > 0 and best_j >= 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = float(conv.max())
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        # Undo
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improve:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def local_polish(a0, time_limit, verbose=True):
    """Aggressive local polish: try small random perturbations + reoptimize."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()

    # Try random perturbations
    n_tries = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        # Random perturbation
        scale = norm * 0.02 / (1.0 + 0.1 * n_tries)
        noise = np.random.randn(n) * scale
        a_pert = a + noise
        a_pert = simplex_project(a_pert, norm)
        score = float(fast_autocorr(a_pert).max())
        if score < best_score:
            best_score = score
            best_a = a_pert.copy()
            a = a_pert.copy()
        else:
            # Try smaller step
            a_pert2 = a + noise * 0.3
            a_pert2 = simplex_project(a_pert2, norm)
            score2 = float(fast_autocorr(a_pert2).max())
            if score2 < best_score:
                best_score = score2
                best_a = a_pert2.copy()
                a = a_pert2.copy()
        n_tries += 1

        if verbose and n_tries % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish try={n_tries} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    time_budget = min(float(budget_s) - 90.0, 1700.0)

    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov - main workhorse (50% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = min(remaining * 0.50, 700.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Smoothed GD with Adam (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: Smoothed GD n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smoothed_gd_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"sgd_n{n_best}")

    # Phase 3: Pair CD (20% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Pair CD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_optimize(a_norm, t3, rng, window=25, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 4: LP Nesterov round 2 (20% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 40:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Final pair CD polish (20% of remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 200.0)
    print(f"\n=== Phase 5: Final Pair CD n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_optimize(a_norm, t5, rng, window=40, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # Phase 6: Local polish with random perturbations
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t6 = min(remaining * 0.3, 100.0)
        print(f"\n=== Phase 6: Local polish n={n_best} (budget={t6:.0f}s) ===")
        if t6 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = local_polish(a_norm, t6, verbose=False)
            do_try_update(a_out, f"polish_n{n_best}")

    # Phase 7: One more LP Nesterov if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        t7 = min(remaining * 0.5, 200.0)
        print(f"\n=== Phase 7: LP Nesterov final n={n_best} (budget={t7:.0f}s) ===")
        if t7 > 30:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=False)
            do_try_update(a_out, f"lp3_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
