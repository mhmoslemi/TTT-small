def get_solver():

        return _global_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_weights(b, beta):
    """Compute softmax weights for LSE."""
    b_max = b.max()
    exp_vals = np.exp(beta * (b - b_max))
    return exp_vals / exp_vals.sum()

def lse_value(b, beta):
    """Compute LSE value."""
    b_max = b.max()
    return b_max + np.log(np.exp(beta*(b - b_max)).sum()) / beta

def lse_grad_wrt_a(a, w):
    """Compute gradient of sum_k w[k]*b[k] w.r.t. a.
    Returns grad[i] = 2 * sum_j a[j] * w[i+j] (for valid indices)
    """
    n = len(a)
    # w has length 2n-1
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[j+i]  (where j+i < 2n-1)
    # This is: 2 * np.correlate(w, a, 'valid')
    c = np.correlate(w, a, mode='valid')
    return 2.0 * c

def lp_best_response(a, M, solver):
    """Solve: max sum(b) s.t. T@b <= M, b >= 0.
    T is the (2n-1) x n matrix where T[k,i] = a[k-i] (valid indices).
    """
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build sparse T matrix
    rows = []
    cols = []
    data = []
    for i in range(n):
        for k in range(i, min(i + n, size)):
            rows.append(k)
            cols.append(i)
            data.append(a[k - i])
    
    T = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(size, n))
    M_vec = np.full(size, M)
    
    constraints = [T @ b_var <= M_vec]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val
