import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Global solver cache
_solver_cache = None

def get_solver():

    if _solver_cache is not None:
        return _solver_cache
    # Try solvers in order of preference
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                return s
        except:
            pass
    _solver_cache = cp.CLARABEL
    return _solver_cache

def autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_score(a):
    """Compute J(a) = 2*n*max(ac)/(sum(a))^2"""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = autocorr(a)
    return 2.0 * len(a) * ac.max() / (s * s)

def lp_solve(a, M, solver):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0"""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # Build Toeplitz matrix T (2n-1 x n)
    # T[k,j] = a[|k-j|] if |k-j| < n, else 0
    # Actually T[k,j] = a[k-j] if 0 <= k-j < n
    # (a*b)_k = sum_j a_{k-j} * b_j for valid indices
    # So T[k,j] = a[k-j] if 0 <= k-j <= n-1
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(b.value, dtype=np.float64), 0)

def nesterov_lp(a_init, time_limit, solver, n_exp=2000):
    """LP-Nesterov with exponential averaging."""
    n = len(a_init)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a_init, 0) * norm / max(a_init.sum(), 1e-15)
    y = a.copy()
    best_a = a.copy()
    best_score = autocorr(a).max()
    
    # Exponential moving average
    ema = a.copy()
    ema_weight = 0.95  # will anneal
    n_ema = 0
    
    t0 = time.monotonic()
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = autocorr(y).max()
        if M < 1e-12:
            break
        
        b = lp_solve(y, M, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b * norm / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = np.maximum((1 - alpha) * y + alpha * g, 0)
        y_new = y_new * norm / max(y_new.sum(), 1e-15)
        
        if t == 1:
            a_new = g.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = np.maximum(y_new + mom * (y_new - y), 0)
            a_new = a_new * norm / max(a_new.sum(), 1e-15)
        
        y = y_new
        a = a_new
        
        # Update EMA
        ema_weight = max(0.5, 0.999 * (1 - t / 50000))
        ema = ema_weight * ema + (1 - ema_weight) * a
        
        # Check scores
        for candidate in [y, a, ema]:
            sc = autocorr(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
    
    return best_a, best_score

def acorr(a):
    n = len(a)
    sz = 2*n - 1
    fs = 1 << (sz-1).bit_length()
    A = rfft(a, fs)
    return irfft(A*A, fs)[:sz]

def interp_seq(f, new_n):
    old_n = len(f)

    s = f.sum()

    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    return fi/ss if ss > 1e-15 else np.ones(new_n)/new_n

    return best_a, best_sc
