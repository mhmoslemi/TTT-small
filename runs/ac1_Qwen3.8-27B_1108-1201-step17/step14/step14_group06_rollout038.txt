T_k is a symmetric matrix. Is it PSD? The eigenvalues of T_k... For k = n-1, T_k is the anti-identity matrix, which has eigenvalues ±1. So it's NOT PSD, meaning c[k] is not convex.

So set B is not convex, and POCS doesn't directly apply.

However, we can still use a "projection" onto set B in a heuristic sense: given a current point a, find the point in set B that's closest to a (in some metric). This is a non-convex optimization problem, but we can solve it approximately.

**Alternative: Use the LP as a "projection"**

The LP best-response is essentially a "projection" onto set B in the L1 sense (maximizing sum(b) is like finding the point in B with the largest L1 norm, which for non-negative points is the "furthest" point from the origin in the L1 sense).

Actually, it's not exactly a projection. It's more like a "best response" in a game-theoretic sense.

Let me just go with a practical implementation that combines the best ideas:

1. LP best-response with Nesterov (main workhorse)
2. Mirror descent (exploration)
3. LSE gradient (polish)
4. Coordinate descent (final touch)
5. Good averaging strategy

One more thought: the previous program uses `build_constraint_matrix` which creates a (2n-1) x n matrix. For n=1600, this is 3199 x 1600 = about 5 million entries. This is fine memory-wise.

But constructing this matrix at each iteration is wasteful if `a` doesn't change much. Actually, the matrix depends on `a` (the entries are values of `a`), so we need to rebuild it each time. But we can do it efficiently.

Actually, looking at the code more carefully, the constraint matrix T has T[k, i] = a[k-i] (for valid indices). This is a "convolution matrix" or "Toeplitz-like" matrix. We don't actually need to form it explicitly if we use a solver that supports sparsity or matrix-free operations. But cvxpy needs the explicit matrix.

For n=1600, forming the matrix takes O(n^2) = 2.5 million operations, which is fast. The LP solve is the bottleneck.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. LP best-response with Nesterov acceleration and adaptive M threshold
2. Mirror descent (multiplicative weights) for exploration
3. LSE-smoothed gradient descent with L-BFGS-B for refinement
4. Targeted coordinate descent with line search for final polish
5. Polyak-Ruppert averaging for improved final iterates
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_solver_cache = None

def get_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.ECOS
    print(f"  Solver (fallback): ECOS")
    return _solver_cache

def fast_conv(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally (faster than calling evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def build_conv_matrix(a, n):
    """Build the (2n-1) x n convolution constraint matrix."""
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        start = i
        end = min(i + n, 2 * n - 1)
        T[start:end, i] = a[i:end - i]
    return T

def lp_best_response(a, M, solver, T=None):
    """Solve the LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lse_obj_grad(x, n, T_val):
    """Log-sum-exp smoothed objective and gradient."""
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_val * (c - max_c)
    exp_vals = np.exp(shifted)
    lse = max_c + np.log(exp_vals.sum()) / T_val
    w = exp_vals / exp_vals.sum()
    
    # Gradient via FFT: grad[i] = (2/n) * sum_k w[k] * a[k-i]
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    W = rfft(w_full, fft_size)
    A = rfft(a_full, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    
    obj = 2.0 * n * lse / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj * a / s
    return obj, grad

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive M."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    y = a.copy()
    avg = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    last_improve = t0
    stall = 0
    iters = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c_y = fast_conv(y)
        M_curr = c_y.max()
        
        # Adaptive epsilon: start aggressive, become conservative
        eps = min(0.05, 0.005 * np.sqrt(t + 1))
        M_used = M_curr * (1.0 - eps)
        
        T_mat = build_conv_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T_mat)
        if b is None:
            b = lp_best_response(y, M_curr * 0.99, solver, T_mat)
        if b is None:
            if verbose:
                print(f"    LP failed at iter {iters}")
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum
        iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            mu = (t - 1.0) / (t + 2.0)
            a_new = y_new + mu * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = fast_conv(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        sc_a = fast_conv(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            last_improve = time.monotonic()
        
        # Polyak averaging
        avg += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 20 == 0:
            avg_c = avg / n_avg
            s_avg = avg_c.sum()
            if s_avg > 1e-15:
                avg_c = avg_c * norm / s_avg
                sc_avg = fast_conv(avg_c).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_c.copy()
                    last_improve = time.monotonic()
        
        # Stall detection
        if time.monotonic() - last_improve > 45:
            stall += 1
            if stall > 8:
                if verbose:
                    print(f"    Stalled at iter={iters}, best={best_score:.8f}")
                break
        else:
            stall = 0
        
        if verbose and iters % 50 == 0:
            print(f"    LP-Nest iter={iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) for exploration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Start with a moderate step size and decay
    eta = 1.0 / np.sqrt(n)
    decay = 0.999
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Find peak positions (within 99.5% of max)
        threshold = M * 0.995
        peaks = np.where(c >= threshold)[0]
        
        # Compute subgradient: g[i] = (1/|peaks|) * sum_{k in peaks} a[k-i]
        g = np.zeros(n)
        for k in peaks:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += a[j]
        g /= len(peaks)
        g *= 2.0  # factor of 2 from the quadratic
        
        # Mirror descent update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        eta *= decay
        
        if verbose and (it + 1) % 100 == 0:
            print(f"    Mirror iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score

def lse_optimize(a0, time_limit, T_start=100.0, T_end=2.0, n_stages=6):
    """LSE-smoothed gradient optimization with L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        def obj(x, T_val=T):
            v, _ = lse_obj_grad(x, n, T_val)
            return v
        
        def grad(x, T_val=T):
            _, g = lse_obj_grad(x, n, T_val)
            return g
        
        try:
            res = scipy_minimize(obj, a.copy(), method='L-BFGS-B',
                                  jac=grad,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'gtol': 1e-10, 'ftol': 1e-12,
                                           'maxls': 30})
            if res.x is not None:
                a_c = np.maximum(res.x, 0.0)
                s_c = a_c.sum()
                if s_c > 1e-10:
                    a_c = a_c * norm / s_c
                    sc = fast_conv(a_c).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_c.copy()
                        a = a_c.copy()
                        print(f"    LSE T={T:.1f}: score={sc:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")
    
    return best_a, best_score

def coord_descent(a0, time_limit, seed=42):
    """Targeted coordinate descent with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        k_peak = int(np.argmax(c))
        
        # Find top contributors to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        top_src = idx[np.argsort(contrib)[::-1]][:20]
        
        # Find low-value targets (positions not contributing to top peaks)
        top_peaks = np.argsort(c)[::-1][:20]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            if je2 >= js2:
                peak_mask[js2:je2 + 1] = True
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        low_targets = non_peak[np.argsort(a[non_peak])][:20]
        
        improved = False
        # Try moving mass from top contributors to low targets
        for src in top_src:
            if a[src] < 1e-10:
                continue
            for dst in low_targets:
                if dst == src:
                    continue
                best_frac = 0
                best_sc = best_score
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-10:
                        continue
                    a_t = a_t * norm / s_t
                    sc = fast_conv(a_t).max()
                    if sc < best_sc:
                        best_sc = sc
                        best_frac = frac
                if best_frac > 0:
                    amt = a[src] * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    a = a * norm / s_a
                    best_score = best_sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a_t = a.copy()
                a_t[i] -= amt
                a_t[j] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-10:
                    continue
                a_t = a_t * norm / s_t
                sc = fast_conv(a_t).max()
                if sc < best_score - 1e-12:
                    a = a_t.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall > 100:
                break
    
    return best_a, best_score

def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3*np.cos(2*x) + 0.15*np.cos(5*x) + 0.05*np.cos(9*x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5*(x-0.3)**2) + 0.7*np.exp(-5*(x-0.7)**2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def interpolate(f, new_n):
    """Interpolate sequence to new length, preserving relative proportions."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_n)
    f_new = np.maximum(f_new, 0)
    s_new = f_new.sum()
    if s_new < 1e-15:
        f_new = np.ones(new_n) / new_n
    else:
        f_new = f_new / s_new
    return f_new

def try_update(best_seq, best_score, candidate, label, t0):
    """Try to update best with candidate."""
    if candidate is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    sc = evaluate_local(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t0
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score
