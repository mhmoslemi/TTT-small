def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)

def build_sparse_constraints(a, n, frac=0.25):
    b = fast_autocorr(a)
    max_b = b.max()
    threshold = max_b * (1.0 - frac)
    active = np.where(b >= threshold)[0]
    T = np.zeros((len(active), n))
    for row_idx, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[row_idx, idx] = a[k - idx]
    return T

    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[k, idx] = a[k - idx]
    return T


def lp_solve(a, M, solver, T):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None or len(b_val) != n:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Progressive M reduction: start gentle, get more aggressive
        progress = min(1.0, t / 300.0)
        eps = 0.015 + 0.025 * progress
        M_used = M_curr * (1.0 - eps)

        T = build_sparse_constraints(y, n, frac=0.25)
        b = lp_solve(y, M_used, solver, T)
        if b is None:
            b = lp_solve(y, M_curr, solver, T)
            if b is None:
                T_full = build_full_constraints(y, n)
                b = lp_solve(y, M_curr, solver, T_full)
                if b is None:
                    break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 80 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score - 1e-10:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Stall detection and restart
        if time.monotonic() - last_improve > 40.0:
            stall_count += 1
            if stall_count > 4:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}. Restarting.")
                y = best_a.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                stall_count = 0
                last_improve = time.monotonic()
                # Perturb to escape
                rng = np.random.default_rng((t * 7919 + 13) % (2**31))
                perturbation_scale = 0.005 + 0.01 * min(stall_count, 3)
                a_pert = y + rng.normal(0, perturbation_scale * max(y.std(), 1e-6), n)
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a_pert = a_pert * norm_val / s_pert
                    score_pert = fast_autocorr(a_pert).max()
                    if score_pert <= best_score + 0.005:
                        y = a_pert
        else:
            stall_count = 0

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M_curr={M_curr:.8f}")

    return best_a, best_score


def targeted_peak_reduction(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 300:
            break

        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k_star in top_k:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_star - idx]
            sorted_pos = np.argsort(contribs)[::-1]
            top_srcs = idx[sorted_pos[:12]]
            low_dsts = idx[sorted_pos[-12:]]

            for src in top_srcs:
                if a[src] < 1e-10:
                    continue
                for dst in low_dsts:
                    if dst == src:
                        continue
                    for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm_val / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial.copy()
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve % 40 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                        break
                    elif score < best_score + 0.002:
                        a = a_trial.copy()

        if verbose_peak_reduction and (time.monotonic() - t0) % 30 < 0.1:
            print(f"    Peak reduction ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a


verbose_peak_reduction = False


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    global verbose_peak_reduction
    verbose_peak_reduction = False

    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    rng = np.random.default_rng(seed)
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 0.01:
            parent = p / p.sum()
            best_score = evaluate_safe(parent)
            best_seq = parent.copy()
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Main sparse LP Nesterov (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.45
    print(f"\n=== Phase 1: Sparse LP-Nest n={n_base} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_sparse_lp(a_norm, t1, solver, verbose=True)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp1_n{n_base}", t_global)

    # Phase 2: Second round from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 40:
        print(f"\n=== Phase 2: LP round 2 n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_sparse_lp(a_norm, t2, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp2_n{n_best}", t_global)

    # Phase 3: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_best + 400
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 60:
        print(f"\n=== Phase 3: LP longer n={n_long} (budget={t3:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_sparse_lp(a_norm, t3, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp_long_n{n_long}", t_global)

    # Phase 4: Targeted peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Peak reduction n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out = targeted_peak_reduction(a_norm, t4, seed=54321)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"peak_n{n_best}", t_global)

    # Phase 5: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Final polish n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_sparse_lp(a_norm, t5, solver, verbose=True)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"final_n{n_best}", t_global)

    # Final peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Final: Peak reduction n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out = targeted_peak_reduction(a_norm, remaining, seed=99999)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
