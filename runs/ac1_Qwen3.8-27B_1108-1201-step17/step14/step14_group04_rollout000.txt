def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_nonneg_sum(v, target_sum):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return target_sum * v / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lp_best_response(a, M, solver):
    """Solve max sum(b) s.t. b>=0, a*b <= M (elementwise)."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix: (a*b)[k] = sum_j a[k-j]*b[j] for valid j
    # Use sparse representation
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    T_mat = cp.cvxpy.reductions.solvers.defenseless_dispatch.sparse_matrix
    from scipy.sparse import coo_matrix
    T_sp = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    constraints = [T_sp @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_restart(a0, time_limit, solver, verbose=True, restart_interval=100):
    """LP Nesterov with periodic restarts from ergodic average."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    best_score_full = evaluate_safe(a)
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_restart = 0
    stall_count = 0
    stall_threshold = 300

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Ergodic averaging
        avg_a += a
        n_avg += 1

        # Periodic check of ergodic average and restart
        if n_avg % restart_interval == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                # Restart from ergodic average if it's good
                if score_avg < best_score + 1e-10:
                    y = avg_check.copy()
                    a = avg_check.copy()
                    avg_a = np.zeros(n)
                    n_avg = 0
                    last_restart = t
                    stall_count = 0

        # Stall detection: if no improvement for stall_threshold iters, restart
        if n_iters - last_restart > stall_threshold and n_iters % 50 == 0:
            # Try restarting from best_a directly
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            last_restart = n_iters
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-R iter={n_iters} ({elapsed:.0f}s): best_peak={best_score:.10f}")

    return best_a, best_score, evaluate_safe(best_a)


def coordinate_search(a0, time_limit, seed=42, n_trials_per_iter=50):
    """Fine-grained coordinate-level local search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    # Precompute which indices contribute to top peaks
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:4]

        # Find indices that contribute to top peaks
        peak_indices = set()
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                if a[j] * a[k - j] > 0:
                    peak_indices.add(j)

        # Find "safe" destination indices (not contributing to any top peak)
        safe_indices = [j for j in range(n) if j not in peak_indices]
        if not safe_indices:
            safe_indices = list(range(n))

        improved = False
        for _ in range(n_trials_per_iter):
            # Pick source from peak-contributing indices
            if peak_indices:
                src = rng.choice(list(peak_indices))
            else:
                src = rng.integers(n)

            if a[src] < norm * 1e-8:
                continue

            # Pick destination from safe indices
            if len(safe_indices) > 1:
                dst = safe_indices[rng.integers(len(safe_indices))]
            else:
                dst = rng.integers(n)
                if dst == src:
                    continue

            # Try several fraction sizes
            for frac in [0.0001, 0.0003, 0.001, 0.003, 0.01]:
                amt = a[src] * frac
                if amt < norm * 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial * norm / a_trial.sum()
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if a.max() > norm * 1e-10:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial * norm / a_trial.sum()
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 200:
                break

    return best_a, best_score, evaluate_safe(best_a)


def multi_peak_lp(a0, time_limit, solver, verbose=True):
    """LP method that focuses on balancing the top few peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_curr = fast_autocorr(y)
        # Use a slightly relaxed bound: 95th percentile of top peaks
        sorted_peaks = np.sort(b_curr)[::-1]
        # Use max but with a tiny relaxation to allow LP to find feasible solution
        M_curr = sorted_peaks[0]
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = (avg_a / n_avg) * norm / (avg_a / n_avg).sum()
            sc_avg = fast_autocorr(avg_check).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()
                # Restart
                y = avg_check.copy()
                a = avg_check.copy()
                avg_a = np.zeros(n)
                n_avg = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MP-LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score, evaluate_safe(best_a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_smooth(a0, time_limit, seed=99):
    """Frequency-domain smoothing to reduce spectral peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    pad_len = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        if peak_val < 1e-12:
            break

        # Get FFT
        a_pad = np.zeros(pad_len)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = np.abs(A)**2

        # Find top non-DC power bins
        power[0] = 0  # Exclude DC
        top_bins = np.argsort(power)[::-1][:3]

        improved = False
        for bin_idx in top_bins:
            if bin_idx == 0 or power[bin_idx] < 1e-10:
                continue
            # Reduce this frequency component
            for reduction in [0.01, 0.03, 0.05, 0.1]:
                A_new = A.copy()
                mag = np.abs(A_new[bin_idx])
                if mag < 1e-15:
                    break
                phase = np.angle(A_new[bin_idx])
                new_mag = mag * (1.0 - reduction)
                A_new[bin_idx] = new_mag * np.exp(1j * phase)

                a_new_full = np.fft.irfft(A_new, pad_len)
                a_new = np.maximum(a_new_full[:n], 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    break
                a_new = a_new * norm / s_new
                sc = fast_autocorr(a_new).max()
                if sc < best_score - 1e-14:
                    a = a_new
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.01)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 100:
                break

    return best_a, best_score, evaluate_safe(best_a)


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search combining restartable LP, coordinate search, and spectral methods."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Brief LP to confirm parent quality ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    if t1 > 20:
        print(f"\n=== Phase 1: LP warm-up n={n_base} (budget={t1:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _, _ = lp_nesterov_restart(a_norm, t1, solver, verbose=False, restart_interval=50)
        try_update(a_out, "lp_warmup")

    # === Phase 2: Main LP Nesterov with restarts (bulk of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP Nesterov+restart n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _, _ = lp_nesterov_restart(a_norm, t2, solver, verbose=True, restart_interval=80)
        try_update(a_out, "lp_main")

    # === Phase 3: Coordinate search polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Coordinate search n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _, _ = coordinate_search(a_norm, t3, seed=42)
        try_update(a_out, "coord1")

    # === Phase 4: Another LP round (post coordinate search) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP round 2 n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _, _ = lp_nesterov_restart(a_norm, t4, solver, verbose=True, restart_interval=60)
        try_update(a_out, "lp_r2")

    # === Phase 5: Spectral smoothing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Spectral smooth n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _, _ = spectral_smooth(a_norm, t5, seed=77)
        try_update(a_out, "spectral")

    # === Phase 6: Coordinate search round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Coordinate search 2 n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _, _ = coordinate_search(a_norm, t6, seed=12345)
        try_update(a_out, "coord2")

    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for n_try in [n_best + 2, n_best - 2, n_best + 4]:
            if n_try < 100 or time.monotonic() - t_global > time_budget - 30:
                break
            t_try = min(remaining * 0.12, 80.0)
            if t_try < 20:
                break
            print(f"\n=== Phase 7: Try n={n_try} (budget={t_try:.0f}s) ===")
            init_try = interpolate_sequence(best_seq, n_try)
            a_norm_try = init_try * np.sqrt(2.0 * n_try) / init_try.sum()
            a_out, _, _ = lp_nesterov_restart(a_norm_try, t_try, solver, verbose=False, restart_interval=50)
            try_update(a_out, f"len_{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.4, 150.0)
        if t8 > 20:
            print(f"\n=== Phase 8: Final LP n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _, _ = lp_nesterov_restart(a_norm, t8, solver, verbose=False, restart_interval=40)
            try_update(a_out, "lp_final")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.5, 60.0)
        if t9 > 10:
            print(f"\n=== Phase 9: Final coord polish n={n_best} (budget={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _, _ = coordinate_search(a_norm, t9, seed=99999, n_trials_per_iter=100)
            try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
