def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        # Estimate Lipschitz constant
        grad0 = lse_grad(y, T)
        g_norm = np.linalg.norm(grad0)
        if g_norm < 1e-15:
            L = 1.0 / T
        else:
            # Conservative estimate: L ~ n/T for the smoothed max
            L = max(n / (T * 0.5), g_norm / (np.linalg.norm(y) * 0.1 + 1e-10))
        
        t_fista = 1
        staging = a.copy()
        n_iter = 0
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            n_iter += 1
            if n_iter % 200 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s, {n_iter} iters): best={best_score:.8f}")

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def spectral_peak_shave(a0, time_limit, seed=42):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()

    pad_len = 1 << (2 * n - 1).bit_length()

            stall = 0

        A = rfft(a_pad)

            j = rng.integers(n)

            if ki > pad_len // 2:
                continue

            if current_magnitude < 1e-15:
                continue

            new_magnitude = current_magnitude * 0.98

        a_new = a_new_full[:n]

        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        if sc < best_score - 1e-12:
            a = a_new
            best_score = sc
            best_a = a.copy()

                i = rng.integers(n)
                j = rng.integers(n)

                        sc_t = fast_autocorr(a_trial).max()
                        if sc_t < best_score:

                            best_a = a.copy()

def targeted_peak_shave(a0, time_limit, seed=12345):

    rng = np.random.default_rng(seed)

                if a_trial[src] < 0:
                    continue

                if s_t < 1e-15:

def lse_fista_adaptive(a0, time_limit, verbose=True):
    """
    LSE-FISTA with adaptive temperature and step size.

    """

    T_end = 0.05

    t_fista = 1

    L = n / 50.0

    last_improve = t0

        progress = n_iter / n_total

        z = y - step * grad
        a_new = project_simplex(z, 1.0)
        
        if t_fista == 1:
            y_new = a_new.copy()
        else:
            mom = (t_fista - 1.0) / (t_fista + 2.0)
            y_new = a_new + mom * (a_new - staging)
            y_new = project_simplex(y_new, 1.0)
        
        staging = a_new.copy()
        a = a_new
        y = y_new

        # Adaptive L: grow L as T decreases (gradient becomes steeper)
        if n_iter % 100 == 0:
            L = max(n / (T * 2.0), L * 1.05)

            sc = fast_autocorr(a).max()
            sc_y = fast_autocorr(y).max()
            sc = min(sc, sc_y)
            if sc < best_score:
                best_score = sc
                best_a = (a if sc == fast_autocorr(a).max() else y).copy()
                last_improve = time.monotonic()
            
            if verbose:
                elapsed = time.monotonic() - t0

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _global_solver


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def subgradient_max(y, threshold_frac=0.99):
    """Compute subgradient of max(b_k) using all near-peak positions."""
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n), 0.0
    threshold = ac_max * threshold_frac
    peak_idx = np.where(ac >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(ac)])
    w = np.zeros(2 * n - 1)
    w[peak_idx] = 1.0 / len(peak_idx)
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad, ac_max


def lse_fista_adaptive(a0, time_limit, verbose=True):
    """LSE-FISTA with adaptive temperature continuation and restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    T_schedule = [15.0, 5.0, 1.5, 0.5, 0.15, 0.05, 0.015, 0.005]
    time_per_stage = time_limit / len(T_schedule)

    for stage_i, T in enumerate(T_schedule):
        t_fista = 1
        staging = a.copy()
        L = n / max(T * 0.8, 0.3)
        n_iter = 0
        stage_start = time.monotonic()
        last_improve = stage_start

        while time.monotonic() - stage_start < time_per_stage:
            if time.monotonic() - t0 >= time_limit:
                break
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            n_iter += 1

            if n_iter % 300 == 0:
                sc_a = fast_autocorr(a).max()
                sc_y = fast_autocorr(y).max()
                sc = min(sc_a, sc_y)
                if sc < best_score:
                    best_score = sc
                    best_a = (a if sc_a < sc_y else y).copy()
                    last_improve = time.monotonic()
                if time.monotonic() - last_improve > 30:
                    t_fista = 1
                    staging = a.copy()
                    y = a.copy()
                    last_improve = time.monotonic()

        sc = fast_autocorr(a).max()
        sc_y2 = fast_autocorr(y).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if sc_y2 < best_score:
            best_score = sc_y2
            best_a = y.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adapt stage={stage_i} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgrad_fista(a0, time_limit, verbose=True):
    """
    FISTA using the subgradient of max(b) directly (no smoothing).
    Uses all near-peak positions for a robust gradient direction.
    Adaptive step size with backtracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    t_fista = 1
    staging = a.copy()
    L = n / 30.0
    n_iter = 0
    last_improve = t0
    stall_restarts = 0

    while time.monotonic() - t0 < time_limit:
        grad, ac_max = subgradient_max(y, threshold_frac=0.99)
        step = 1.0 / L
        z = y - step * grad
        a_new = project_simplex(z, 1.0)

        if t_fista == 1:
            y_new = a_new.copy()
        else:
            mom = (t_fista - 1.0) / (t_fista + 2.0)
            y_new = a_new + mom * (a_new - staging)
            y_new = project_simplex(y_new, 1.0)

        staging = a_new.copy()
        a = a_new
        y = y_new
        t_fista += 1
        n_iter += 1

        if n_iter % 50 == 0:
            sc_a = fast_autocorr(a).max()
            sc_y = fast_autocorr(y).max()
            sc = min(sc_a, sc_y)
            if sc < best_score:
                best_score = sc
                best_a = (a if sc_a < sc_y else y).copy()
                last_improve = time.monotonic()

            # Adaptive step: grow L if no progress
            if time.monotonic() - last_improve > 45:
                stall_restarts += 1
                L *= 0.7
                if stall_restarts > 5:
                    t_fista = 1
                    staging = a.copy()
                    y = a.copy()
                    stall_restarts = 0
                    last_improve = time.monotonic()
            else:
                stall_restarts = 0

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SubGrad-FISTA iter={n_iter} ({elapsed:.0f}s): best={best_score:.8f}, L={L:.1f}")

        if n_iter > 200000:
            break

    sc = fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration and ergodic averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if time.monotonic() - last_improve > 240 and n_iters > 100:
            break

    return best_a, best_score


def targeted_local_search(a0, time_limit, seed=12345):
    """
    Local search with diverse move types:
    - Pair moves (transfer mass between two positions)
    - Block moves (transfer mass from a block to another)
    - 3-point moves (redistribute among three positions)
    - Peak-targeted moves (reduce specific autocorrelation peaks)
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    move_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        improved = False

        # Move type 1: Peak-targeted pair move
        for k in top_k[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue
            cand_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 5)]
            dst = cand_sorted[rng.integers(len(cand_sorted))]

            for frac in [0.0005, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    move_count += 1
                    break
            if improved:
                break

        if improved:
            continue

        # Move type 2: Random pair move
        for _ in range(3):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < norm * 1e-10:
                continue
            frac = rng.uniform(0.002, 0.03)
            amt = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            if a_trial[i] < 0:
                continue
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                move_count += 1
                break

        # Move type 3: 3-point redistribution
        if not improved and rng.random() < 0.3:
            i = rng.integers(n)
            j = rng.integers(n)
            k3 = rng.integers(n)
            unique_idx = list(set([i, j, k3]))
            if len(unique_idx) < 3:
                pass
            else:
                i, j, k3 = unique_idx
                if a[i] > norm * 1e-8 and a[j] > norm * 1e-8:
                    fract1 = rng.uniform(0.005, 0.03)
                    fract2 = rng.uniform(0.005, 0.03)
                    amt1 = a[i] * fract1
                    amt2 = a[j] * fract2
                    a_trial = a.copy()
                    a_trial[i] -= amt1
                    a_trial[j] -= amt2
                    a_trial[k3] += amt1 + amt2
                    if a_trial[i] >= 0 and a_trial[j] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                move_count += 1

        if not improved:
            stall += 1
            if stall > 500:
                break

    return best_a, best_score


def spectral_peak_shave(a0, time_limit, seed=42):
    """Frequency-domain peak shaving."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    pad_len = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        a_pad = np.zeros(pad_len)
        a_pad[:n] = a
        A = rfft(a_pad)
        power = np.abs(A) ** 2
        peak_idx = np.argmax(power)

        if peak_idx == 0:
            stall += 1
            if stall > 200:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            continue

        top_k = np.argsort(power)[::-1][:4]
        for ki in top_k:
            if ki == 0 or ki > pad_len // 2:
                continue
            cmag = np.abs(A[ki])
            if cmag < 1e-15:
                continue
            phase = np.angle(A[ki])
            A[ki] = cmag * 0.985 * np.exp(1j * phase)

        a_new_full = irfft(A, pad_len)
        a_new = a_new_full[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-12:
            a = a_new
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np
