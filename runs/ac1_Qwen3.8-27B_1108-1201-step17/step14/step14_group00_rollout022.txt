"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth optimization via log-sum-exp with multiplicative (mirror) updates
2. LP best-response with adaptive active set for polish
3. Pair coordinate descent for final refinement
4. Multi-start with structured initializations

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_conv(a):
    """Full linear convolution of a with a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def normalize(a, target_sum=None):
    """Normalize a to have the desired sum (default: 1)."""
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * (target_sum or 1.0) / len(a)
    if target_sum is None:
        return a / s
    return a * target_sum / s

def logsumexp_and_grad(a, tau):
    """
    Compute log-sum-exp(tau * c) / tau and its gradient, where c = conv(a,a).
    Returns (logsumexp_value, gradient, c_max).
    """
    n = len(a)
    c = fast_conv(a)
    tc = tau * c
    tc_max = tc.max()
    # Numerically stable
    exp_tc = np.exp(tc - tc_max)
    sum_exp = exp_tc.sum()
    logsumexp_val = (tc_max + np.log(sum_exp)) / tau
    w = exp_tc / sum_exp  # softmax weights
    
    # Gradient: g[i] = 2 * sum_j a[j] * w[i+j]
    # Using convolution: g = 2 * convolve(w, a[::-1])[n-1 : 2n-1]
    conv_w = np.convolve(w, a[::-1])
    g = 2.0 * conv_w[n-1 : 2*n-1]
    
    return logsumexp_val, g, c.max()

def smooth_optimize(a0, time_limit, verbose=True, seed=42):
    """
    Smooth optimization using log-sum-exp approximation with multiplicative updates.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = normalize(a, 1.0)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule: start moderate, increase over time
    tau_start = n * 0.5
    tau_end = n * 5.0
    n_total = 200000  # planned iterations
    
    stall_count = 0
    stall_threshold = 500
    
    for t in range(1, n_total + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        frac = t / n_total
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Compute objective and gradient
        obj, g, c_max = logsumexp_and_grad(a, tau)
        
        if c_max < best_score * (1 - 1e-10):
            best_score = c_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Multiplicative update with adaptive step size
        g_max = g.max()
        if g_max < 1e-15:
            break
        
        # Try a few step sizes
        base_eta = 1.0 / g_max
        etas = [0.5 * base_eta, base_eta, 2.0 * base_eta, 4.0 * base_eta]
        
        best_new_obj = obj + 1e-10  # Slightly worse than current
        best_a_new = a.copy()
        
        for eta in etas:
            a_trial = a * np.exp(-eta * g)
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial = normalize(a_trial, 1.0)
            trial_obj, _, _ = logsumexp_and_grad(a_trial, tau)
            if trial_obj < best_new_obj:
                best_new_obj = trial_obj
                best_a_new = a_trial.copy()
        
        a = best_a_new
        
        # If stalled too long, perturb
        if stall_count > stall_threshold:
            noise = 1.0 + 0.02 * rng.standard_normal(n)
            noise = np.maximum(noise, 0.5)
            a = a * noise
            a = normalize(a, 1.0)
            stall_count = 0
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            cur_max = fast_conv(a).max()
            print(f"    Smooth t={t} ({elapsed:.0f}s): tau={tau:.1f}, best_max={best_score:.10f}, cur={cur_max:.10f}")
    
    return best_a, best_score

def lp_best_response(a, M, active_set):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix for active set
    T = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            T[idx, j] = a[k - j]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    
    return b_val

def lp_optimize(a0, time_limit, verbose=True, seed=0):
    """LP best-response optimization with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = normalize(a, 1.0)
    
    best_a = a.copy()
    best_conv = fast_conv(a)
    best_score = best_conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    active_set = None
    stall_count = 0
    stall_threshold = 30
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv(a)
        M = conv_y.max()
        
        # Update active set
        if active_set is None or t % 50 == 0:
            tol = 0.05
            active_set = np.where(conv_y > M * (1.0 - tol))[0]
            if len(active_set) < 10:
                top_k = min(50, size)
                active_set = np.argsort(conv_y)[-top_k:]
        
        b = lp_best_response(a, M, active_set)
        if b is None:
            # Try with full constraint set
            if t % 10 == 0:
                active_set = np.arange(size)
                b = lp_best_response(a, M, active_set)
            if b is None:
                continue
        
        b = normalize(b, 1.0)
        
        # Check if solution violates non-active constraints
        if len(active_set) < size:
            full_conv_b = fast_conv_b(a, b)
            non_active = np.setdiff1d(np.arange(size), active_set)
            if len(non_active) > 0 and full_conv_b[non_active].max() > M * 1.001:
                active_set = np.union1d(active_set, non_active[full_conv_b[non_active] > M])
                continue
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, 1.0)
        
        score = fast_conv(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        if stall_count > stall_threshold:
            a = best_a.copy()
            active_set = None
            stall_count = 0
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def fast_conv_b(a, b):
    """Compute convolution of a and b (different arrays) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def structured_init(n, rng):
    """Generate various structured initial sequences."""
    choices = []
    x = np.linspace(0, 1, n)
    
    # Hanning window
    choices.append(0.5 - 0.5 * np.cos(2 * np.pi * x))
    # Hamming window
    choices.append(0.54 - 0.46 * np.cos(2 * np.pi * x))
    # Triangular
    choices.append(np.maximum(0, 1 - np.abs(2*x - 1)))
    # Quadratic
    choices.append(np.maximum(0, 1 - (2*x - 1)**2))
    # Sine
    choices.append(np.sin(np.pi * x)**2)
    # Sine^4
    choices.append(np.sin(np.pi * x)**4)
    # Flat-top
    choices.append(0.2156 + 0.4160*np.cos(2*np.pi*x) + 0.2776*np.cos(4*np.pi*x) 
                   + 0.0836*np.cos(6*np.pi*x) + 0.0069*np.cos(8*np.pi*x))
    # Smooth bump
    choices.append(np.exp(-10*(x-0.3)**2) + 0.8*np.exp(-10*(x-0.7)**2))
    # Power law
    choices.append(x**0.5 * (1-x)**0.5)
    # Asymmetric
    choices.append(np.maximum(0, 1 - np.abs((x-0.3)/0.7)) + 0.5*np.maximum(0, 1 - np.abs((x-0.8)/0.2)) * 0.3)
    
    best = None
    best_score = float('inf')
    for c in choices:
        c = np.maximum(c, 0)
        s = c.sum()
        if s < 1e-10:
            continue
        c = c / s
        sc = fast_conv(c).max()
        if sc < best_score:
            best_score = sc
            best = c
    
    return best, best_score
