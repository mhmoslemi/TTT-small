        return None
    return np.maximum(val, 0.0)

def spectrum_flatten(a, n_flatten_iters=3):
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * norm / s
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(n_flatten_iters):
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        target_mag = np.median(mag[mag > 1e-15]) if np.any(mag > 1e-15) else 1.0
        blend = 0.3
        new_mag = (1.0 - blend) * mag + blend * target_mag
        A_new = new_mag * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new * norm / s_new
    return a

def lp_nesterov_main(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_max_conv = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    total_iters = 0
    last_flatten_iter = 0

    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
            s_y = a.sum()
        y = y * norm / s_y

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        progress = min(1.0, elapsed / max(time_limit * 0.85, 1.0))
        threshold_frac = 0.55 + 0.44 * progress
        threshold = M * threshold_frac

        active = np.where(conv_y >= threshold)[0]
        if len(active) < 15:
            active = np.where(conv_y >= M * 0.3)[0]
        max_active = 250
        if len(active) > max_active:
            top_idx = np.argsort(conv_y[active])[::-1][:max_active]
            active = active[top_idx]

        b = lp_best_response(y, M, active, solver)
        if b is None:
            active2 = np.where(conv_y >= M * 0.9)[0]
            b = lp_best_response(y, M, active2, solver)
        if b is None:
            b = np.ones(n)

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        if len(active) < 2 * n - 1:
            check = fast_conv_cross(y, g)
            violated = np.where(check > M * (1 + 1e-3))[0]
            violated_not_in_active = np.setdiff1d(violated, active)
            if len(violated_not_in_active) > 0:
                new_active = np.union1d(active, violated_not_in_active)
                if len(new_active) > 350:
                    new_active = np.where(conv_y >= M * 0.95)[0]
                    if len(new_active) > 400:
                        new_active = np.argsort(conv_y)[::-1][:400]
                b2 = lp_best_response(y, M, new_active, solver)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue
                    g = b * norm / b_sum

        alpha = 2.0 / (total_iters + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new

        if total_iters == 1:
            a = g.copy()
        else:
            momentum = (total_iters - 1.0) / (total_iters + 2.0)
            a_new = y + momentum * (y - a)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new * norm / s_a
            a = a_new

        improved = False
        for candidate in [y, a]:
            sc = fast_conv_self(candidate).max()
            if sc < best_max_conv - 1e-10:
                best_max_conv = sc
                best_a = candidate.copy()
                improved = True

        if total_iters - last_flatten_iter >= 200 and improved:
            y_fl = spectrum_flatten(y, n_flatten_iters=2)
            sc_fl = fast_conv_self(y_fl).max()
            if sc_fl < best_max_conv - 1e-10:
                best_max_conv = sc_fl
                best_a = y_fl.copy()
            y = y_fl
            last_flatten_iter = total_iters
            a = best_a.copy()

        if verbose and total_iters % 40 == 0:
            print(f"    LP-Nav iter={total_iters} ({elapsed:.0f}s): max_conv={best_max_conv:.8f}, thresh={threshold_frac:.3f}, active={len(active)}")

    return best_a, best_max_conv

def lp_polish(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_max_conv = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
            s_y = a.sum()
        y = y * norm / s_y

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        active = np.where(conv_y >= M * 0.99)[0]
        if len(active) < 20:
            active = np.where(conv_y >= M * 0.95)[0]
        if len(active) > 500:
            active = np.argsort(conv_y)[::-1][:500]

        b = lp_best_response(y, M, active, solver)
        if b is None:
            b = lp_best_response(y, M, np.arange(2 * n - 1), solver)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        check = fast_conv_cross(y, g)
        violated = np.where(check > M * (1 + 1e-4))[0]
        if len(violated) > 0:
            new_active = np.setdiff1d(violated, active)
            if len(new_active) > 0:
                active2 = np.union1d(active, violated)
                if len(active2) > 600:
                    active2 = np.argsort(conv_y)[::-1][:600]
                b2 = lp_best_response(y, M, active2, solver)
                if b2 is not None:
                    b = b2
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        g = b * norm / b_sum

        alpha = 2.0 / (total_iters + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new

        if total_iters == 1:
            a = g.copy()
        else:
            momentum = (total_iters - 1.0) / (total_iters + 2.0)
            a_new = y + momentum * (y - a)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new * norm / s_a
            a = a_new

        for candidate in [y, a]:
            sc = fast_conv_self(candidate).max()
            if sc < best_max_conv - 1e-11:
                best_max_conv = sc
                best_a = candidate.copy()

        if verbose and total_iters % 20 == 0:
            print(f"    LP-Polish iter={total_iters} ({elapsed:.0f}s): max_conv={best_max_conv:.8f}, active={len(active)}")

    return best_a, best_max_conv

def coord_descent_polish(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_max_conv = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    no_improve_rounds = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            conv = fast_conv_self(a)
            cur_max = conv.max()

            peak_threshold = cur_max * 0.97
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j1 = int((kl - i) % n)
                j2 = int((kl + i) % n)
                if 0 <= j1 < n:
                    j_candidates.add(j1)
                if 0 <= j2 < n:
                    j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                n_samples = 25
                deltas = np.linspace(0.0, ai, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    if d < 1e-15:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v - 1e-14:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_max_conv - 1e-12:
                        best_max_conv = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 8:
                break
            if no_improve_rounds == 4:
                n_perturb = max(3, n // 30)
                idx_p = rng.choice(n, size=n_perturb, replace=False)
                a[idx_p] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve_rounds = 0

        if verbose and total_improvements % 50 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD impr={total_improvements} ({elapsed:.0f}s): max_conv={best_max_conv:.8f}")

    return best_a, best_max_conv

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 1:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 2:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 3:
        a = 0.4 + 0.4 * np.sin(np.pi * x) ** 1.2 + 0.2 * np.cos(4 * np.pi * x) ** 2
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = np.exp(-4.0 * (x - 0.2) ** 2) + 0.8 * np.exp(-4.0 * (x - 0.8) ** 2)
    elif choice == 6:
        a = 0.3 + 0.5 * np.sin(np.pi * x) ** 0.8
    elif choice == 7:
        a = np.exp(-5.0 * (x - 0.5) ** 2) + 0.3
    elif choice == 8:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x) ** 2
    else:
        a = np.exp(-7.0 * (x - 0.35) ** 2) + 0.65 * np.exp(-7.0 * (x - 0.65) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 1: LP Nesterov main n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_out, _ = lp_nesterov_main(best_seq, t1, solver, verbose=True)
        try_update(a_out, f"lp_main_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 2: LP Polish n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_polish(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_polish_n{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 3: Coordinate Descent n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = coord_descent_polish(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"cd_n{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_nesterov_main(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 5: CD polish round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = coord_descent_polish(best_seq, t5, rng, verbose=True)
        try_update(a_out, f"cd2_n{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 200
    t6 = min(remaining * 0.4, 150.0)
    print(f"\n=== Phase 6: Longer length n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 80:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = lp_nesterov_main(init_long, min(t6, 100.0), solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(1200, n_base - 200)
    t7 = min(remaining * 0.3, 100.0)
    print(f"\n=== Phase 7: Shorter length n={n_short} (budget={t7:.0f}s) ===")
    if t7 > 20 and remaining > 50:
        init_short = interpolate_seq(best_seq, n_short)
        a_out, _ = lp_nesterov_main(init_short, min(t7, 80.0), solver, verbose=False)
        try_update(a_out, f"lp_short_{n_short}")

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t8a = min(remaining * 0.5, 100.0)
        a_out, _ = lp_polish(best_seq, t8a, solver, verbose=False)
        try_update(a_out, f"lp_final")

            a_out, _ = coord_descent_polish(best_seq, t8b, rng, verbose=False)
            try_update(a_out, f"cd_final")

    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
