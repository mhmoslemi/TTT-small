        cols.extend([i] * (end - start))
        vals.extend(a[idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_T_sparse(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lse_obj_grad(a, n, T_temp):
    """Compute LSE-smoothed objective and gradient."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def lse_optimize(a0, time_limit, T_start=60.0, T_end=2.0, n_stages=4):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    a = a * norm_val / max(s0, 1e-15)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 20:
            break

        def obj_func(x, T_val=T):
            val, _ = lse_obj_grad(x, n, T_val)
            return val

        def grad_func(x, T_val=T):
            _, g = lse_obj_grad(x, n, T_val)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': 200, 'maxfun': 800,
                                          'gtol': 1e-9, 'ftol': 1e-12, 'maxls': 40})
            if res.x is not None:
                a_c = np.maximum(res.x, 0.0)
                s_c = a_c.sum()
                if s_c > 1e-10:
                    a_c = a_c * norm_val / s_c
                    score_c = fast_autocorr(a_c).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_c.copy()
                        a = a_c.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception:
            pass

    return best_a, best_score


def nesterov_lp_multi(a0, time_limit, solver, eps_schedule='adaptive', verbose=True):
    """
    LP best-response with Nesterov acceleration.
    eps_schedule: 'adaptive' (increasing), 'fixed_003', 'fixed_005', 'fixed_008'
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    a = a * norm_val / max(s0, 1e-15)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    stall = 0
    stall_thresh = 300

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Determine M target
        if eps_schedule == 'adaptive':
            eps = min(0.06, 0.004 * (1.0 + 0.05 * t))
        elif eps_schedule == 'fixed_003':
            eps = 0.003
        elif eps_schedule == 'fixed_005':
            eps = 0.005
        elif eps_schedule == 'fixed_008':
            eps = 0.008
        else:
            eps = 0.005

        M_used = M_curr * (1.0 - eps)

        T = build_T_sparse(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum

        # Adaptive step size based on peak ratio
        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 1e-12 and M_new > 1e-12:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.03), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        y_new = y_new * norm_val / s_y if s_y > 1e-15 else y_new

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            a_new = a_new * norm_val / s_a if s_a > 1e-15 else y_new

        y = y_new
        a = a_new

        # Track best (check both y and a)
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Polyak-Ruppert averaging (exponentially weighted)
        avg_a = 0.95 * avg_a + 0.05 * a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall = 0

        # Stall detection
        if score_y >= best_score and score_a >= best_score:
            stall += 1
        else:
            stall = 0

        if stall > stall_thresh:
            if verbose:
                print(f"    Stalling at t={t} ({time.monotonic()-t0:.0f}s), best={best_score:.8f}")
            break

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): M={M_curr:.6f}, best={best_score:.8f}, eps={eps:.4f}")

    return best_a, best_score


def quadratic_coord_descent(a0, time_limit, seed=42):
    """
    Coordinate descent where each coordinate is optimized by solving a 1D
    problem analytically (the peak is piecewise quadratic in each coordinate).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    a = a * norm_val / max(s0, 1e-15)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        rng.shuffle(order)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                    break
            if a[i] < 1e-15:
                continue

            # Compute contribution of a[i] to each autocorrelation bin
            # c_k = sum_j a[j]*a[k-j], the terms involving a[i] are:
            # For k >= i: a[i] * a[k-i] (if k-i < n)
            # For k < i: a[i] * a[i-k] (if i-k < n, i.e., k > i-n)
            # So c_k = base_k + a[i] * (a[k-i] + a[i-k]) where terms are 0 if index out of range
            # (for k != 0, both terms exist; for k=0 only a[0]^2 is the self-term)

            # Let's compute the "cross" contribution
            # For coordinate i, the autocorrelation terms involving a[i] are:
            # c_k gets contribution a[i]*a[k-i] for valid k-i, and a[k-i]*a[i] for valid k-i
            # Actually c_k = sum_j a[j]*a[k-j], so a[i] appears when j=i (giving a[i]*a[k-i])
            # or when k-j=i, i.e., j=k-i (giving a[k-i]*a[i])
            # These are the same term a[i]*a[k-i] (counted once in the sum)
            # EXCEPT when k-i = i, i.e., k=2i, where it's a[i]^2 (the self-term)

            # So: c_k = base_k + 2*a[i]*a[k-i] for k != 2i (both j=i and j=k-i contribute)
            #       c_k = base_k + a[i]^2 for k = 2i
            # where base_k excludes all terms involving a[i]

            # More precisely:
            # c_k = sum_{j: j != i, k-j != i} a[j]*a[k-j] + [i<n terms]
            # Let's just compute it directly:
            # c_k includes a[i]*a[k-i] if 0<=k-i<n (from j=i)
            # and a[k-i]*a[i] if 0<=k-i<n (from j=k-i)
            # If k-i = i (i.e., k=2i), these are the same term a[i]^2
            # If k-i != i, they are two different terms each equal to a[i]*a[k-i]

            # For the optimization: fix all a[j] for j != i, vary a[i] = x
            # c_k(x) = base_k + 2*x*a[k-i] for k != 2i (if 0 <= k-i < n)
            # c_k(x) = base_k + x^2 for k = 2i
            # c_k(x) = base_k for other k (if k-i out of range and i out of range for that k)

            # We want to minimize max_k c_k(x) subject to x >= 0
            # This is a 1D problem. The feasible x is [0, infinity) but we normalize
            # after, so effectively [0, some_max].

            # For simplicity, let's try a few candidate values:
            # Current value, and perturbations
            a_save = a[i]
            candidates = [a_save * 0.5, a_save * 0.7, a_save * 0.8, a_save * 1.2,
                          a_save * 1.4, a_save * 1.6, a_save * 0.3]
            # Also try moving mass to a random other coordinate
            j = rng.integers(n)
            if j != i:
                for frac in [0.1, 0.2, 0.3]:
                    amt = a_save * frac
                    a[i] = a_save - amt
                    a[j] = a_save + amt
                    a[a < 0] = 0
                    s_t = a.sum()
                    if s_t > 1e-10:
                        a = a * norm_val / s_t
                        score = fast_autocorr(a).max()
                        if score < best_score - 1e-12:
                            best_score = score
                            best_a = a.copy()
                            improved = True
                    a[i] = a_save
                    a[j] = a[0]  # reset (we'll fully reset below)

            # Reset a properly
            a = best_a.copy()
            a[i] = a_save

            # Try simple rescaling of a[i]
            for factor in [0.6, 0.75, 0.9, 1.1, 1.3]:
                a[i] = a_save * factor
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-10:
                    a_trial = a * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
            a = best_a.copy()

        if not improved:
            stall += 1
            if stall > 100:
                break
        else:
            stall = 0

    return best_a, best_score


def peak_targeted_moves(a0, time_limit, seed=123):
    """Targeted moves that specifically reduce peak-contributing pairs."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    a = a * norm_val / max(s0, 1e-15)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        # Find top contributing pairs for the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_idx = sorted_idx[:10]

        # Find low-value coordinates to receive mass
        low_coords = np.argsort(a)[:20]

        improved = False
        for src in top_idx:
            if a[src] < 1e-10:
                continue
            for dst in low_coords:
                if dst == src:
                    continue
                for frac in [0.03, 0.05, 0.08, 0.12]:
                    a_trial = a.copy()
                    amt = a[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        # Also try random moves
        if not improved:
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.02, 0.1)
                a_trial = a.copy()
                a_trial[i] -= a[i] * frac
                a_trial[j] += a[i] * frac
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    f_norm = f / max(s, 1e-15)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-3.0 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 / (1.0 + 10.0 * x * (1.0 - x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-M LP Nesterov + LSE gradient + targeted coord descent + multi-length."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def try_update(candidate_a, label):
        nonlocal best_seq, best_score
        if candidate_a is None:
            return
        a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LP Nesterov with adaptive M (main)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest adaptive M, n={n_cur}, budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_multi(a_norm, t1, solver, eps_schedule='adaptive', verbose=True)
        try_update(a_out, f"lp_adapt_n{n_cur}")

    # Phase 2: LP Nesterov with fixed aggressive M
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-Nest fixed M=0.008, n={n_cur}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.2, 300.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_multi(a_norm, t2, solver, eps_schedule='fixed_008', verbose=False)
        try_update(a_out, f"lp_aggr_n{n_cur}")

    # Phase 3: LSE gradient optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LSE gradient, n={n_cur}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t3, T_start=50.0, T_end=2.0, n_stages=4)
        try_update(a_out, f"lse_n{n_cur}")

    # Phase 4: LP Nesterov round 2 (conservative)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-Nest conservative, n={n_cur}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_multi(a_norm, t4, solver, eps_schedule='fixed_003', verbose=False)
        try_update(a_out, f"lp_cons_n{n_cur}")

    # Phase 5: Peak-targeted coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Peak-targeted moves, n={n_cur}, budget={remaining:.0f}s ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_targeted_moves(a_norm, t5, seed=777)
        try_update(a_out, f"peaktarg_n{n_cur}")

    # Phase 6: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 6: Longer sequence n={n_long}, budget={remaining:.0f}s ===")
    t6 = min(remaining * 0.25, 400.0)
    if t6 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_multi(a_norm, t6, solver, eps_schedule='adaptive', verbose=True)
        try_update(a_out, f"lp_long_n{n_long}")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final polish, n={n_cur}, budget={remaining:.0f}s ===")
    if remaining > 30:
        half = remaining / 2.0
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_multi(a_norm, half, solver, eps_schedule='fixed_005', verbose=False)
        try_update(a_out, "lp_final")

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            n_cur2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_cur2) / best_seq.sum()
            a_out2, _ = peak_targeted_moves(a_norm2, remaining2, seed=99999)
            try_update(a_out2, "peak_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
