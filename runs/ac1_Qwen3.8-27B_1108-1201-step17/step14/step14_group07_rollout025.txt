        # b[j] appears in conv at positions m = j, j+1, ..., j+n-1
        # with coefficient a[m-j] = a[0], a[1], ..., a[n-1]
        start = j
        end = j + n
        actual_end = min(end, size)
        length = actual_end - start
        if length > 0:
            rows.extend(range(start, actual_end))
            cols.extend([j] * length)
            vals.extend(a[:length])
    T = cp.Problems  # just a placeholder, we'll use numpy sparse
    from scipy.sparse import csr_matrix
    T_mat = csr_matrix((vals, (rows, cols)), shape=(size, n))
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_dense(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_line_search=True, max_iters=500000):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    no_improve_count = 0
    best_score_hist = []

    use_sparse = n > 800

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        if use_sparse:
            b = lp_best_response_sparse(y, M_curr, solver)
        else:
            b = lp_best_response_dense(y, M_curr, solver)
        if b is None:
            stall_count += 1
            if stall_count > 50:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        stall_count = 0

        # Line search with more trial points
        if use_line_search:
            alpha_base = 2.0 / (t + 2.0)
            best_alpha = alpha_base
            trial_alphas = [alpha_base * 0.25, alpha_base * 0.5, alpha_base,
                            alpha_base * 1.5, alpha_base * 2.0, alpha_base * 3.0]
            best_obj = float('inf')
            for ta in trial_alphas:
                ta = min(ta, 1.0)
                cand = (1.0 - ta) * y + ta * g
                cand = np.maximum(cand, 0.0)
                sc = fast_autocorr(cand).max()
                if sc < best_obj:
                    best_obj = sc
                    best_alpha = ta
            alpha = best_alpha
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            no_improve_count = 0

        # Average tracking
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    no_improve_count = 0

        # Stall detection and restart
        if t % 300 == 0:
            if no_improve_count > 1000:
                if verbose:
                    print(f"    Restart at iter={n_iters}, best={best_score:.8f}")
                y = best_a * norm / best_a.sum()
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                no_improve_count = 0
            elif no_improve_count > 500:
                # Perturb
                rng_loop = np.random.default_rng(t)
                n_pert = max(5, n // 50)
                idx_pert = rng_loop.choice(n, size=n_pert, replace=False)
                y_pert = y.copy()
                for ip in idx_pert:
                    y_pert[ip] *= rng_loop.uniform(0.7, 1.3)
                y_pert = np.maximum(y_pert, 0.0)
                sp = y_pert.sum()
                if sp > 1e-15:
                    y_pert = y_pert * norm / sp
                    sp_score = fast_autocorr(y_pert).max()
                    if sp_score < best_score * 1.05:
                        y = y_pert
                        a = y_pert.copy()
                        no_improve_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    shift = b - max_b
    max_shift = shift.max()
    if max_shift > 50.0 / T_temp:
        exp_vals = np.exp(T_temp * shift)
        lse = max_b + np.log(exp_vals.sum()) / T_temp
    else:
        exp_vals = np.exp(T_temp * shift)
        log_sum = max_shift + np.log(np.exp(T_temp * (shift - max_shift)).sum())
        lse = max_b + log_sum / T_temp
    obj = 2.0 * n * lse / (s * s)
    return obj


def lse_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shift = b - max_b
    max_shift = shift.max()
    exp_vals = np.exp(T_temp * (shift - max_shift))
    w = exp_vals / exp_vals.sum()
    lse = max_b + (max_shift + np.log(exp_vals.sum())) / T_temp

    # Gradient of LSE w.r.t. a: sum_m w[m] * db[m]/da[i]
    # db[m]/da[i] = 2*a[m-i] (with a[j]=0 for j out of [0,n-1])
    # So grad_lse[i] = 2 * sum_m w[m] * a[m-i]
    # This is 2 * cross_correlation(w, a)[i]
    fft_size = 1 << (3 * n - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    # cross[i] = sum_m a[m] * w[m+i] ... need to check
    # Actually: ifft(fft(a) * conj(fft(w)))[i] = sum_m a[m] * conj(w[m-i])
    # We want sum_m w[m] * a[m-i]
    # = sum_k w[k+i] * a[k] (substituting m=k+i)
    # This is the cross-correlation
    cross = irfft(np.conj(A_fft) * W_fft, fft_size)[:n]
    # cross[i] = sum_k a[k+i] * w[k] = sum_m a[m] * w[m-i] ... 
    # We want sum_m w[m] * a[m-i] which is the same by symmetry of the sum
    # Actually let me just verify: we want grad[i] = 2*sum_m w[m]*a[m-i]
    # cross[i] from ifft(conj(fft(a))*fft(w)) = sum_k a[k]*w[k+i]... 
    # Let's just use: grad_conv[i] = 2 * (w * a reversed correlation)
    # Simpler: use the fact that d(conv(a,a)[m])/da[i] = a[m-i] + a[m-i] = 2*a[m-i]
    # So sum_m w[m]*2*a[m-i] = 2 * sum_m w[m]*a[m-i]
    # Let v[m] = w[m], u[m] = a[m]
    # sum_m v[m]*u[m-i] for i=0..n-1
    # = (v * u)[i] in the "cross-correlation" sense
    # Using FFT: ifft(fft(v) * conj(fft(u)))[i] gives sum_k v[k]*conj(u[k-i])
    # Since everything is real: sum_k v[k]*u[k-i]
    # Substituting m=k: sum_m v[m]*u[m-i] ✓
    cross2 = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    conv_grad = 2.0 * cross2
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = remaining / max(1, n_stages - si)

        x0 = a.copy()

        try:
            res = scipy_minimize(lse_objective, x0, args=(n, T), method='L-BFGS-B',
                                 jac=lambda x: lse_grad(x, n, T),
                                 bounds=[(0, None)] * n,
                                 options={'maxiter': 200, 'maxfun': 500,
                                          'time_limit': min(stage_time, 120.0),
                                          'ftol': 1e-10, 'gtol': 1e-8})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def gradient_guided_cd(a0, time_limit, seed=42):
    """Coordinate descent guided by gradient of smoothed objective."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    T_smooth = 20.0  # temperature for gradient

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Compute softmax weights on autocorrelation
        shift = b - max_b
        exp_vals = np.exp(T_smooth * (shift - shift.max()))
        w = exp_vals / exp_vals.sum()

        # Gradient: grad[i] = 2 * sum_m w[m] * a[m-i]
        fft_size = 1 << (3 * n - 1).bit_length()
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        w_pad = np.zeros(fft_size)
        w_pad[:2 * n - 1] = w
        cross = irfft(rfft(w_pad, fft_size) * np.conj(rfft(a_pad, fft_size)), fft_size)[:n]
        grad = 2.0 * cross

        # Normalize gradient for direction
        grad_max = np.abs(grad).max()
        if grad_max < 1e-15:
            stall += 1
            if stall > 200:
                break

        # Strategy: move mass from high-grad positions to low-grad positions
        # High positive grad means increasing a[i] increases the objective
        # So we want to decrease a at high-grad positions and increase at low-grad
        # But we need to maintain sum(a) = norm

        # Find top contributors to the peak
        top_peaks = np.argsort(b)[::-1][:8]
        peak_mask = np.zeros(n, dtype=bool)
        for kp in top_peaks:
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            peak_mask[js:je + 1] = True

        improved = False

        # Try moving mass from high-gradient, in-peak positions to low-gradient, out-of-peak positions
        in_peak_grad = grad[peak_mask].copy()
        in_peak_idx = np.where(peak_mask)[0]
        out_peak_grad = grad[~peak_mask].copy()
        out_peak_idx = np.where(~peak_mask)[0]

        if len(in_peak_idx) > 0 and len(out_peak_idx) > 0:
            # Sort by gradient (highest first for source, lowest first for dest)
            src_order = in_peak_idx[np.argsort(in_peak_grad)[::-1][:10]]
            dst_order = out_peak_idx[np.argsort(out_peak_grad)[:10]]

            for src in src_order:
                if a[src] < 1e-10:
                    continue
                for dst in dst_order:
                    found = False
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial.copy()
                            best_a = a_trial.copy()
                            best_score = score
                            stall = 0
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if improved:
                    break

        if not improved:
            # Random exploration
            stall += 1
            if stall % 3 == 0:
                for _ in range(20):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial.copy()
                        best_a = a_trial.copy()
                        best_score = score
                        stall = 0
                        break

            # Periodic perturbation
            if stall % 30 == 0 and stall > 0:
                n_pert = max(3, n // 80)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for ip in idx_pert:
                    a_trial[ip] *= rng.uniform(0.6, 1.4)
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial.copy()
                        best_a = a_trial.copy()
                        best_score = score
                        stall = 0

            if stall > 500:
                break

        # Adapt temperature
        if stall % 50 == 0:
            T_smooth *= 0.9  # cool down

    return best_a, best_score


def peak_shaving_cd(a0, time_limit, seed=99):
    """Targeted peak shaving: for each peak, find optimal mass redistribution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:6]

        improved = False

        for k_peak in top_peaks:
            # Find which pairs (i, j) with i+j = k_peak contribute most
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_src = idx[np.argsort(contrib)[::-1][:8]]

            # Build mask of "danger zone" (positions that affect top peaks)
            danger_mask = np.zeros(n, dtype=bool)
            for kp2 in top_peaks[:4]:
                js2 = max(0, kp2 - n + 1)
                je2 = min(n - 1, kp2)
                danger_mask[js2:je2 + 1] = True

            # Safe positions (not in danger zone)
            safe_idx = np.where(~danger_mask)[0]
            if len(safe_idx) == 0:
                safe_idx = np.where(np.arange(n) != top_src[0] if len(top_src) > 0 else True)[0]

            # Try moving mass from top contributors to safe positions
            for src in top_src:
                if a[src] < 1e-12:
                    continue
                # Sort safe positions by current value (move to lower-value positions)
                safe_sorted = safe_idx[np.argsort(a[safe_idx])][:15]

                for dst in safe_sorted:
                    if dst == src:
                        continue
                    found = False
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.12, 0.25]:
                        amt = a[src] * frac
                        if amt < 1e-13:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial.copy()
                            best_a = a_trial.copy()
                            best_score = score
                            stall = 0
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random swap
            if stall % 4 == 0:
                for _ in range(15):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.003, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_a = a_trial.copy()
                        best_score = score
                        stall = 0
                        break

            # Larger perturbation every 50 stalls
            if stall % 50 == 0 and stall > 0:
                n_pert = max(5, n // 40)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for ip in idx_pert:
                    a_trial[ip] *= rng.uniform(0.5, 1.5)
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.01:
                        a = a_trial.copy()
                        if score < best_score:
                            best_a = a_trial.copy()
                            best_score = score
                        stall = 0

            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def spectral_init(n, rng):
    """Initialize using spectral methods - aim for flat power spectrum."""
    choice = rng.integers(0, 6)
    if choice == 0:
        # Raised cosine
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif choice == 1:
        # Gaussian
        center = n / 2.0
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 2:
        # Sinc-like (inverse of rectangular spectrum)
        center = n // 2
        idx = np.arange(n) - center
        a = np.sinc(idx / (n * 0.3))
        a = np.abs(a)
    elif choice == 3:
        # Cosine bell
        x = np.linspace(-np.pi / 2, np.pi / 2, n)
        a = np.cos(x) ** 2
    elif choice == 4:
        # Hybrid: cosine + small Gaussian bumps
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * np.cos(np.pi * x) ** 2
        a += 0.2 * np.exp(-20 * (x - 0.3) ** 2)
        a += 0.2 * np.exp(-20 * (x - 0.7) ** 2)
    else:
        # Flat with small structured perturbation
        a = np.ones(n)
        freq = rng.uniform(1, 5)
        amp = rng.uniform(0.1, 0.3)
        x = np.linspace(0, 1, n)
        a += amp * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def two_stage_refine(a0, time_limit, solver, seed=42):
    """Two-stage: coarse LP then fine LP with tighter tolerance."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    # Stage 1: Coarser (shorter time, less line search)
    t1 = time_limit * 0.6
    a_out, sc_out = nesterov_lp_optimize(a, t1, solver, verbose=False, use_line_search=True)
    if sc_out < best_score:
        best_score = sc_out
        best_a = a_out.copy()

    # Stage 2: Finer (longer time, more line search)
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a_out2, sc_out2 = nesterov_lp_optimize(best_a, remaining, solver, verbose=True, use_line_search=True)
        if sc_out2 < best_score:
            best_score = sc_out2
            best_a = a_out2.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    # Phase 0: Quick multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    n_parent = len(best_seq) if best_seq is not None else 1600
    # Try a few lengths around the parent
    scan_lengths = sorted(set([
        n_parent,
        max(1000, n_parent - 200),
        n_parent + 200,
        n_parent + 400
    ]))
    scan_results = {}

    phase0_budget = time_budget * 0.10
    for i_scan, n_scan in enumerate(scan_lengths):
        if time.monotonic() - t_global > time_budget * 0.12:
            break
        remaining_scan = time_budget * 0.12 - (time.monotonic() - t_global)
        if remaining_scan < 15:
            break
        scan_time = min(remaining_scan / max(1, len(scan_lengths) - i_scan), 80.0)
        if scan_time < 8:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = spectral_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False, use_line_search=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, _ = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = spectral_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Main LP-Nesterov at best length (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_line_search=True)
    do_try_update(a_out, f"lp_main_n{n_best}")

    # Phase 2: Gradient-guided coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Grad-guided CD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.15
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_guided_cd(a_norm, t2, seed=777)
        do_try_update(a_out, f"grad_cd_n{n_best}")

    # Phase 3: Peak shaving CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak shaving n={n_best} (budget={remaining:.0f}s) ===")
    t3 = remaining * 0.15
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shaving_cd(a_norm, t3, seed=1234)
        do_try_update(a_out, f"peak_shave_n{n_best}")

    # Phase 4: LSE polish with adaptive temperatures
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t4 = remaining * 0.12
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t4, T_start=80.0, T_end=2.0, n_stages=8)
        do_try_update(a_out, f"lse_polish_n{n_best}")

    # Phase 5: Second LP pass with noisy restart
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP noisy restart n={n_best} (budget={remaining:.0f}s) ===")
    t5 = remaining * 0.15
    if t5 > 20:
        noise_scale = 0.03 * best_seq.mean()
        a_noisy = best_seq + rng.normal(0, noise_scale, n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = nesterov_lp_optimize(a_noisy, t5, solver, verbose=False, use_line_search=True)
        do_try_update(a_out, f"lp_noisy_n{n_best}")

    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")

    # Split remaining time between peak shaving and gradient CD
    t6a = remaining * 0.5
    t6b = remaining * 0.5

    if t6a > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shaving_cd(a_norm, t6a, seed=55555)
        do_try_update(a_out, "peak_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        a_out, _ = gradient_guided_cd(a_norm, remaining, seed=88888)
        do_try_update(a_out, "grad_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
