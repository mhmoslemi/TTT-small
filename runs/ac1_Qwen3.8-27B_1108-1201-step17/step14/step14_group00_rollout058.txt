    n = len(a)
    size = 2 * n - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))

            worst = violated[np.argsort(yb[violated])[-5:]]
        else:
            worst = violated
        for k in worst:
            if k not in active:
                active.append(k)
    
    return b

def normalize(a, norm_val):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * norm_val / len(a)
    return a * norm_val / s

def score_of(a):
    """Compute the objective value."""
    n = len(a)
    conv = fast_conv(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)

def nesterov_lp_optimize(a0, time_limit, seed=0, verbose=True):
    """Nesterov-accelerated LP best-response optimization."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0, norm_val)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    active_set = None
    stall = 0
    stall_thresh = 30
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_conv(y).max()
        if M < 1e-12:
            break
        
        # Solve LP
        b = lp_best_response_cuting_plane(y, M, seed_constraints=active_set)
        if b is None:
            if t % 50 == 0:
                b = lp_best_response_cuting_plane(y, M, seed_constraints=None)
            if b is None:
                continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        y_new = y_new * norm_val / s_y if s_y > 1e-15 else y.copy()
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            beta = min(beta, 0.99)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            x_new = x_new * norm_val / s_x if s_x > 1e-15 else y_new.copy()
        
        y = y_new
        x = x_new
        
        # Check best
        sc_y = fast_conv(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        sc_x = fast_conv(x).max()
        if sc_x < best_score:
            best_score = sc_x
            best_a = x.copy()
            stall = 0
        
        # Update active set hint
        conv_y = fast_conv(y)
        tol = 0.05
        active_set = list(np.where(conv_y > M * (1.0 - tol))[0])
        
        # Restart if stalled
        if stall > stall_thresh:
            y = best_a.copy()
            x = best_a.copy()
            active_set = None
            stall = 0
            if verbose:
                print(f"    [NLP] restart at t={t}, best={best_score:.10f}")
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [NLP] t={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Pair coordinate descent polishing."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0, norm_val)
    conv = fast_conv(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    size = 2 * n - 1
    
    no_improve = 0
    for iteration in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-10:
                continue
            
            # Try moving mass from i to a random j
            for _ in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                # Try a few fractions
                for frac in [0.3, 0.5, 0.7]:
                    d = ai * frac
                    a[i] -= d
                    a[j] += d
                    a = a * norm_val / a.sum()
                    new_conv = fast_conv(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                    else:
                        a[j] -= d
                        a[i] += d
                        a = a * norm_val / a.sum()
                        conv = fast_conv(a)
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def structured_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    if choice == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.8 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(10 * np.pi * x)
    elif choice == 3:
        a = (1 - 2 * np.abs(x - 0.5)) ** 2
    elif choice == 4:
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.3 * np.cos(6 * np.pi * x))
    else:
        a = 0.4 + 0.6 * np.exp(-10 * (x - 0.5) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_to_length(a, new_n):
    old_n = len(a)
    s = a.sum()
    fn = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, fn)
    result = np.maximum(result, 0.0)
    rs = result.sum()
    if rs < 1e-15:
        return np.ones(new_n) / new_n
    return result / rs

    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
