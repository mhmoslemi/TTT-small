def build_conv_matrix(a):

        T[j:j + n, j] = a
    return T

def simplex_project(x, c):
    """Project x onto {a >= 0, sum(a) = c}."""
    n = len(x)
    if c <= 0:
        return np.zeros(n)
    # Simple projection: clip to non-negative, then rescale
    x_clipped = np.maximum(x, 0)
    s = x_clipped.sum()
    if s < 1e-30:
        return np.full(n, c / n)
    return x_clipped * c / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        return b_val
    if prob.status not in ('optimal', 'optimal_inaccurate') or b_var.value is None:

    if circ_new.max() <= M * 1.001:

    else:

        violated = circ_new > M

        active2 = np.zeros(size, dtype=bool)

def interpolate_sequence(f, new_n):

    if old_n == new_n:

    s = f.sum()
    if s < 1e-15:

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

    f_interp = np.maximum(f_interp, 0.0)

def lp_solve_constrained(a, M, solver, constraint_indices=None, verbose_lp=False):
    """Solve LP: max sum(b) s.t. T_a[b, active] <= M, b >= 0."""
    n = len(a)
    T = build_conv_matrix(a)
    size = 2 * n - 1

        constraint_indices = np.arange(size)

    b_var = cp.Variable(n, nonneg=True)
    rhs = M * np.ones(len(constraint_indices))
    constraints = [T[constraint_indices, :] @ b_var <= rhs]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    if b_var.value is None:
        return None, None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None

    # Try to get dual variables for constraint selection
    dual_vars = None
    try:
        dual_vars = constraints[0].dual_value
    except Exception:
        pass

    return b_val, dual_vars

            warm_constraint_idx = None
            gamma = 1.7

    return best_a, best_score


def high_precision_polish(a0, time_limit, solver, verbose=True):
    """Final polishing with full-constraint LP and very small steps."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(a).max()
        if M_curr < 1e-12:
            break

        # Full LP (no reduction) for maximum precision
        b, _ = lp_solve_constrained(a, M_curr, solver, constraint_indices=None)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum

        # Very small blend (conservative)
        for blend in [0.05, 0.1, 0.2, 0.3]:
            a_trial = (1 - blend) * a + blend * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                break

        n_iters += 1

        if verbose and n_iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

        if n_iters > 50:
            break

    return best_a, best_score


def quadratic_peak_redistribute(a0, time_limit, seed=42, verbose=True):
    """Global peak redistribution using quadratic approximation."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find contributing pairs to the peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 200:
                break
            if n_stall % 50 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * rng.uniform(0.05, 0.15)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            n_improve += 1
                            n_stall = 0
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]

        # Find good destinations (positions that don't contribute to top peaks)
        # Compute "peak contribution map" for this position
        # Position j contributes to peaks at lags j+l for l in [0, n-1]
        # We want positions that have LOW total contribution to top peaks

        # Top 5 peaks
        sorted_lags = np.argsort(b)[::-1][:5]
        peak_set = set(sorted_lags.tolist())

        # Score each destination: how much it contributes to top peaks
        dst_scores = np.zeros(n)
        for l in sorted_lags:
            lj = max(0, l - n + 1)
            lj_end = min(n - 1, l)
            for j in range(lj, lj_end + 1):
                dst_scores[j] += b[l] * max(a[j] * a[l - j], 0)

        dst_scores[src] = np.inf  # exclude source
        # Prefer low-score destinations
        n_dst = min(20, n - 1)
        dst_candidates = np.argsort(dst_scores)[:n_dst]
        dst = dst_candidates[rng.integers(len(dst_candidates))]

        # Quadratic line search: try multiple fractions
        best_trial_score = max_b
        best_frac = 0
        for frac in [0.03, 0.05, 0.08, 0.12, 0.18, 0.25]:
            amt = src_val * frac
            if amt < 1e-18:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_frac = frac

        if best_frac > 0:
            a_trial = a.copy()
            a_trial[src] -= src_val * best_frac
            a_trial[dst] += src_val * best_frac
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()

            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                # Annealing
                T = best_score * 0.0005 * (1.0 - elapsed_frac)
                if T > 1e-10 and rng.random() < np.exp(-(sc - best_score) / T):
                    a = a_trial
                n_stall += 1
        else:
            n_stall += 1

        # Larger perturbation if stuck
        if n_stall > 100:
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for ip in idx_pert:
                if a_trial[ip] > 0:
                    amt = a_trial[ip] * rng.uniform(0.1, 0.4)
                    j_dst = rng.integers(n)
                    a_trial[ip] -= amt
                    a_trial[j_dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.01:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        n_improve += 1
            n_stall = 0

        if verbose and n_improve % 200 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRedist improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_scale_optimize(a0, time_limit, solver, verbose=True):
    """Optimize at small n, interpolate to larger n, continue optimizing."""
    n0 = len(a0)
    best_a = a0.copy()
    best_score = fast_autocorr(a0).max()
    t0 = time.monotonic()

    # Try a few different target lengths
    target_lengths = []
    if n0 <= 1000:
        target_lengths = [n0, n0 + 200, n0 + 400, n0 + 600]
    else:
        target_lengths = [n0, n0 + 100, n0 + 200, n0 + 300]

    for tl in target_lengths:
        if time.monotonic() - t0 >= time_limit * 0.85:
            break

        # Interpolate to target length
        a_t = interpolate_sequence(best_a, tl)
        norm_t = np.sqrt(2.0 * tl)
        a_t = a_t * norm_t / a_t.sum()

        # Quick smooth-max optimization
        t_sub = min(60.0, (time_limit * 0.85) - (time.monotonic() - t0))
        if t_sub > 10:
            a_out, sc_out = smooth_max_nesterov(a_t, t_sub, verbose=False, n_epochs=5)
            if sc_out < best_score:
                best_score = sc_out
                best_a = a_out.copy()
                if verbose:
                    print(f"    MultiScale n={tl}: score={sc_out:.8f}")

        # Quick LP
        t_sub2 = min(60.0, (time_limit * 0.85) - (time.monotonic() - t0))
        if t_sub2 > 15:
            a_out, sc_out = nesterov_lp_optimize(a_t, t_sub2, solver, verbose=False, max_keep=50)
            if sc_out < best_score:
                best_score = sc_out
                best_a = a_out.copy()
                if verbose:
                    print(f"    MultiScale LP n={tl}: score={sc_out:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies:
    1. Multi-scale exploration with smooth-max gradient
    2. LP Nesterov with dual-guided constraint selection
    3. High-precision polishing
    4. Quadratic peak redistribution
    """
    global _global_start_time
    _global_start_time = time.monotonic()

    rng = np.random.default_rng(seed)
    time_budget = min(float(budget_s) - 80.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - _global_start_time
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_init = 1200
        x = np.linspace(0.0, 1.0, n_init)
        # Use a raised-cosine center bump
        best_seq = np.exp(-((x - 0.5) / 0.15) ** 2)
        best_seq += 0.05 * np.exp(-((x - 0.3) / 0.08) ** 2)
        best_seq += 0.05 * np.exp(-((x - 0.7) / 0.08) ** 2)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_cur = len(best_seq)

    # === Phase 1: Smooth-max Nesterov (fast, no LP) ===
    remaining = time_budget - (time.monotonic() - _global_start_time)
    print(f"\n=== Phase 1: SmoothMax Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        norm = np.sqrt(2.0 * n_cur)
        a_in = best_seq * norm / best_seq.sum()
        a_out, _ = smooth_max_nesterov(a_in, t1, verbose=True, n_epochs=10)
        do_try_update(a_out, f"smmax_n{n_cur}")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - _global_start_time)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        norm = np.sqrt(2.0 * n_cur)
        a_in = best_seq * norm / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_in, t2, solver, verbose=True, max_keep=60)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 3: Smooth-max at potentially different n ===
    remaining = time_budget - (time.monotonic() - _global_start_time)
    print(f"\n=== Phase 3: MultiScale (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        norm = np.sqrt(2.0 * len(best_seq))
        a_in = best_seq * norm / best_seq.sum()
        a_out, _ = multi_scale_optimize(a_in, t3, solver, verbose=True)
        do_try_update(a_out, f"multiscale_n{len(a_out) if a_out is not None else 0}")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - _global_start_time)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        norm = np.sqrt(2.0 * n_cur)
        a_in = best_seq * norm / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_in, t4, solver, verbose=True, max_keep=50)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Peak redistribution ===
    remaining = time_budget - (time.monotonic() - _global_start_time)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: PeakRedistribute n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        norm = np.sqrt(2.0 * n_cur)
        a_in = best_seq * norm / best_seq.sum()
        a_out, _ = quadratic_peak_redistribute(a_in, t5, seed=777, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 6: Smooth-max final ===
    remaining = time_budget - (time.monotonic() - _global_start_time)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: SmoothMax Final n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 15:
        norm = np.sqrt(2.0 * n_cur)
        a_in = best_seq * norm / best_seq.sum()
        a_out, _ = smooth_max_nesterov(a_in, t6, verbose=True, n_epochs=8)
        do_try_update(a_out, f"smmax2_n{n_cur}")

    # === Phase 7: High-precision polish ===
    remaining = time_budget - (time.monotonic() - _global_start_time)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: HighPrec Polish n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.13, 150.0)
    if t7 > 10:
        norm = np.sqrt(2.0 * n_cur)
        a_in = best_seq * norm / best_seq.sum()
        a_out, _ = high_precision_polish(a_in, t7, solver, verbose=True)
        do_try_update(a_out, f"polish_n{n_cur}")

    # === Phase 8: Final peak redistribution ===
    remaining = time_budget - (time.monotonic() - _global_start_time)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final PeakRedist n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.10, 100.0)
    if t8 > 10:
        norm = np.sqrt(2.0 * n_cur)
        a_in = best_seq * norm / best_seq.sum()
        a_out, _ = quadratic_peak_redistribute(a_in, t8, seed=31415, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-_global_start_time:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
