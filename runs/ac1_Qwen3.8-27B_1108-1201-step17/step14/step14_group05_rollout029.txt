    prev_y = a.copy()

    while time.monotonic() - t0 < time_limit:
        t = iters + 1
        M_curr = fast_autocorr(y).max()
        M_hist.append(M_curr)
        if M_curr < 1e-12:
            break

        # Solve LP (use reduced version for speed)
        b = solve_best_response_reduced(y, M_curr, max_active=max_active)
        if b is None:
            b = solve_best_response_lp(y, M_curr)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        improved = False
        for candidate in [y, a]:
            sc = fast_autocorr(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1

        # Adaptive gamma
        if len(M_hist) > 80:
            recent = M_hist[-40:]
            osc = np.sum(np.abs(np.diff(recent))) / (40 * max(recent[0], 1e-15))
            if osc > 0.03:
                gamma = max(1.2, gamma - 0.05)
            elif osc < 0.005:
                gamma = min(2.0, gamma + 0.03)

        # Adaptive restart
        if stall > 200:
            rng = np.random.default_rng(iters * 13 + 7)
            perturb_scale = 0.04 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.9
            M_hist = []

        if verbose and iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, gamma={gamma:.2f}, stall={stall}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=True, seed=42):
    """
    Efficient coordinate descent via pair moves with quadratic line search.
    For each peak in the autocorrelation, move mass from the highest-contributing
    position to a low-contributing position, using exact line search.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b = fast_autocorr(a)
        max_b = b.max()

        # Find top peaks
        top_k = min(3, 2 * n - 1)
        peak_indices = np.argpartition(b, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(b[peak_indices])[::-1]]

        improved_this_iter = False

        for p in peak_indices:
            if a[p if p < n else 0] < 1e-15:
                continue

            # Find contributing indices to peak p
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[p - idx]

            if contrib.max() < 1e-18:
                continue

            # Source: highest contributor
            src_local = np.argmax(contrib)
            src = idx[src_local]
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Find destinations: positions with low contribution to peak p
            # Non-contributing positions for peak p
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je+1] = False
            non_contrib[src] = False  # exclude source

            if non_contrib.any():
                # Among non-contributing, pick low-value ones
                nc_idx = np.where(non_contrib)[0]
                # Prefer positions that also don't contribute to other peaks
                dst_candidates = nc_idx[np.argsort(a[nc_idx])[:min(20, len(nc_idx))]]
            else:
                dst_candidates = np.array([i for i in range(n) if i != src])

            if len(dst_candidates) == 0:
                continue

            # Try several destinations
            n_try = min(5, len(dst_candidates))
            chosen = rng.choice(len(dst_candidates), size=n_try, replace=False)
            
            for ci in chosen:
                dst = dst_candidates[ci]
                
                # Quadratic line search: move delta from src to dst
                # c_k changes linearly in delta for k != 0 (approximately)
                # Actually c_p is quadratic in delta due to a_dst * a_dst term
                # Let's just try several values
                best_delta = 0
                best_trial = max_b
                
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = src_val * frac
                    if delta < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        break
                    a_trial = a_trial * norm / st
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_trial:
                        best_trial = sc
                        best_delta = delta

                if best_delta > 0 and best_trial < best_score:
                    a[src] -= best_delta
                    a[dst] += best_delta
                    a = np.maximum(a, 0)
                    a = a * norm / a.sum()
                    best_score = best_trial
                    best_a = a.copy()
                    n_improve += 1
                    improved_this_iter = True
                    break  # Move to next peak

        if not improved_this_iter:
            n_stall += 1
        else:
            n_stall = 0

        # Escape mechanism
        if n_stall > 100:
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            for idx in idx_pert:
                if a[idx] > 1e-10:
                    amt = a[idx] * rng.uniform(0.1, 0.4)
                    j = rng.integers(n)
                    a[idx] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            if a.sum() > 1e-15:
                a = a * norm / a.sum()
                sc = fast_autocorr(a).max()
                if sc < best_score * 1.05:
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def log_param_gradient(a0, time_limit, verbose=True):
    """
    Gradient descent in log-parameterization space.
    a = exp(theta), minimize smoothed max autocorrelation.
    Uses the fact that log(c_k) is convex in theta.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s = a.sum()
    a = a * norm / s
    theta = np.log(a)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Temperature for log-sum-exp smoothing
    T0 = max(0.1, best_score * 0.2)
    T_factor = 0.01  # T ends at T0 * T_factor

    # Learning rate schedule
    lr0 = 0.5 / np.sqrt(n)
    lr_decay = 0.999

    # Momentum
    velocity = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.99
    lr = lr0

    while time.monotonic() - t0 < time_limit:
        steps += 1
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T0 * (T_factor ** elapsed_frac)
        lr *= lr_decay

        # Current a from theta (apply normalization)
        log_a = theta - np.logaddexp.reduce(theta)  # softmax-like normalization
        a_curr = np.exp(log_a) * norm
        if a_curr.sum() < 1e-15:
            break

        # Compute autocorrelation
        b = fast_autocorr(a_curr)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Log-sum-exp weights
        log_b = np.log(np.maximum(b, 1e-30))
        log_b_max = log_b.max()
        w = np.exp((log_b - log_b_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Compute gradient of smoothed objective w.r.t. theta
        # f = (1/T) * log(sum_k exp((1/T) * log(b_k)))  -- wait, let me redo
        # f = logsumexp(b; T) = (1/T) * log(sum_k exp(b_k / T))
        # df/dtheta_j = sum_k (exp(b_k/T)/sum_m exp(b_m/T)) * (1/T) * db_k/dtheta_j
        # db_k/dtheta_j = a_j * (a_{j-k} + a_{j+k}) for valid indices
        # So df/dtheta_j = sum_k w_k * (1/T) * a_j * (a_{j-k} + a_{j+k})
        #                = a_j * (1/T) * sum_k w_k * (a_{j-k} + a_{j+k})

        # Compute h_j = sum_k w_k * (a_{j-k} + a_{j+k}) efficiently
        h = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            # a_{j+k} for j in [0, n-1-k]
            end_b = n - k
            if end_b > 0:
                h[:end_b] += wk * a_curr[:end_b]
            # a_{j-k} for j in [k, n-1]
            if k < n:
                h[k:] += wk * a_curr[:n-k]

        grad = a_curr * h / T

        # Handle constraint: in the softmax parameterization, the gradient
        # should be projected to be orthogonal to the "all-ones" direction
        # (since adding a constant to theta doesn't change a)
        grad -= np.mean(grad)

        # Adam-like update
        velocity = beta1 * velocity + (1 - beta1) * grad
        v_hat = velocity / (1 - beta1 ** steps)
        # Skip second moment for simplicity (Adam vs RMSProp)
        theta -= lr * v_hat / (np.sqrt(np.mean(v_hat ** 2)) + 1e-8)

        # Actually let's use a simpler step
        # theta -= lr * grad / (np.linalg.norm(grad) + 1e-10)

        # Evaluate
        log_a = theta - np.logaddexp.reduce(theta)
        a_new = np.exp(log_a) * norm
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_curr.copy()
        else:
            a_new = a_new * norm / s_new

        a_curr = a_new
        score = fast_autocorr(a_curr).max()
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LogParam step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def peak_shave_refine(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with multi-destination moves and annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find contributing pairs
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        if js > je:
            n_stall += 1
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 200:
                break
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]

        # Non-contributing positions
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        non_contrib[src] = False

        if non_contrib.any():
            nc_idx = np.where(non_contrib)[0]
            sorted_dst = nc_idx[np.argsort(a[nc_idx])[:min(15, len(nc_idx))]]
        else:
            sorted_dst = np.array([i for i in range(n) if i != src])

        if len(sorted_dst) == 0:
            n_stall += 1
            continue

        # Try multiple destinations
        n_try = min(8, len(sorted_dst))
        chosen = rng.choice(len(sorted_dst), size=n_try, replace=False)

        best_move = None
        best_move_score = best_score

        for ci in chosen:
            dst = sorted_dst[ci]
            for frac in [0.1, 0.2, 0.35, 0.5]:
                amt = src_val * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial * norm / st
                sc = fast_autocorr(a_trial).max()
                if sc < best_move_score:
                    best_move_score = sc
                    best_move = (src, dst, amt)

        if best_move is not None:
            src_m, dst_m, amt_m = best_move
            a[src_m] -= amt_m
            a[dst_m] += amt_m
            a = np.maximum(a, 0)
            a = a * norm / a.sum()

            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                # Annealing
                T_anneal = best_score * 0.002 * (1.0 - elapsed_frac)
                if rng.random() < np.exp(-(sc - best_score) / max(T_anneal, 1e-10)):
                    n_stall += 1
                else:
                    # Revert
                    a[src_m] += amt_m
                    a[dst_m] -= amt_m
                    a = np.maximum(a, 0)
                    a = a * norm / a.sum()
                    n_stall += 1
        else:
            n_stall += 1

        # Large perturbation when stuck
        if n_stall > 200:
            n_pert = max(5, n // 50)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            for idx in idx_pert:
                if a[idx] > 1e-10:
                    amt = a[idx] * rng.uniform(0.1, 0.5)
                    j = rng.integers(n)
                    a[idx] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            if a.sum() > 1e-15:
                a = a * norm / a.sum()
                sc = fast_autocorr(a).max()
                if sc < best_score * 1.03:
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 1000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 540.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, verbose=True, max_active=200)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: CoordDescent n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t2, verbose=True, seed=123)
        do_try_update(a_out, f"coord_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 (refined start) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, verbose=True, max_active=150)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Log-parameterization gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LogParam n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = log_param_gradient(a_norm, t4, verbose=True)
        do_try_update(a_out, f"logparam_n{n_cur}")

    # === Phase 5: Try longer length ===
    n2 = n_base + 100
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, verbose=True, max_active=150)
        do_try_update(a_out, f"lp_n{n2}")

    # === Phase 6: Peak shaving on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t6, seed=999, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, verbose=True, max_active=100)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final CoordDescent n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.10, 100.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t8, verbose=True, seed=777)
        do_try_update(a_out, f"coord_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
