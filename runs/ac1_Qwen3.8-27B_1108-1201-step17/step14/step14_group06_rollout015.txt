def fast_autocorr(a):
    """Compute full convolution conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    r = np.arange(1, n + 1)
    cond = mu - (cv - 1) / r > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond))
    theta = (cv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)

def build_constraint_matrix(a, n):
    """Build the convolution constraint matrix T where (T @ b) = conv(a, b)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    if T is None:
        T = build_constraint_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

            break

            b_lp = lp_best_response(a, M_target, solver, T_mat)
            if b_lp is None:
                b_lp = lp_best_response(a, current_max, solver, T_mat)
            if b_lp is not None:
                b_lp_norm = b_lp / b_lp.sum()
                # Blend with current
                alpha = min(0.3, 0.1 * np.sqrt(n_iters / use_lp_every))
                a_blend = (1 - alpha) * a + alpha * b_lp_norm
                a_blend = project_simplex(a_blend)
                b_blend = fast_autocorr(a_blend)
                if b_blend.max() < best_score:
                    best_score = b_blend.max()
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    last_improve = time.monotonic()
        
        # Stall detection
        if time.monotonic() - last_improve > 30:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0
        
        if n_iters % 500 == 0:
            print(f"    SubGrad iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def lse_objective_and_grad(a_flat, n, T_temp):
    """LSE-smoothed objective and gradient for L-BFGS."""
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()
    
    # Gradient: sum_k w[k] * a[k-i] (for each i)
    # This is the same as the subgradient but with softmax weights on ALL lags
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    
    # The LSE objective: (1/T) * log(sum exp(T * b_k))
    # grad of this = sum_k (exp(T*b_k)/sum) * (db/d a_i) = sum_k w_k * a[k-i]
    # But we also need the chain rule for the fact that b depends on a.
    # db_k/da_i = a[k-i] (for valid k-i)
    # So grad_i = sum_k w_k * a[k-i] (which is what 'cross' computes)
    
    obj_val = log_sum
    grad = cross  # This is the gradient of the LSE-smoothed max_conv
    
    return obj_val, grad


def lse_bfgs_optimize(a0, time_limit, T_start=100.0, T_end=2.0, n_stages=4):
    """L-BFGS optimization on the LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0  # normalize to simplex
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)
        
        def obj_func(x, T_val=T):
            a_pos = np.maximum(x, 0)
            s = a_pos.sum()
            if s < 1e-10:
                return 1e10
            a_norm = a_pos / s
            b = fast_autocorr(a_norm)
            max_b = b.max()
            if max_b < 1e-15:
                return 1e10
            shifted = T_val * (b - max_b)
            exp_vals = np.exp(shifted)
            return max_b + np.log(exp_vals.sum()) / T_val
        
        def grad_func(x, T_val=T):
            a_pos = np.maximum(x, 0)
            s = a_pos.sum()
            if s < 1e-10:
                return np.zeros(n)
            a_norm = a_pos / s
            b = fast_autocorr(a_norm)
            max_b = b.max()
            if max_b < 1e-15:
                return np.zeros(n)
            shifted = T_val * (b - max_b)
            exp_vals = np.exp(shifted)
            w = exp_vals / exp_vals.sum()
            
            fft_size = 1 << (2 * n - 1).bit_length()
            w_full = np.zeros(fft_size)
            w_full[:2 * n - 1] = w
            a_full = np.zeros(fft_size)
            a_full[:n] = a_norm
            W_fft = rfft(w_full, fft_size)
            A_fft = rfft(a_full, fft_size)
            cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
            return cross
        
        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'gtol': 1e-7, 'ftol': 1e-9,
                                           'maxls': 40})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            pass
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_tight_M=True):
    """Original LP Nesterov approach (kept as fallback)."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break
        
        if use_tight_M:
            eps = min(0.03, 0.003 * np.sqrt(t + 1))
            M_used = M_curr * (1.0 - eps)
        else:
            M_used = M_curr
        
        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
        
        if time.monotonic() - last_improve > 60.0:
            stall_count += 1
            if stall_count > 5:
                break
        else:
            stall_count = 0
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
