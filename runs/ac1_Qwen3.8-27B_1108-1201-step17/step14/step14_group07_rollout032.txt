"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smoothed gradient descent (log-sum-exp surrogate for max)
2. Adaptive temperature annealing for progressive peak sharpening
3. Peak-targeted coordinate descent for final polish
4. Multi-length exploration with interpolation
5. Multiple restarts with structured and perturbed initializations
Key: Fast per-iteration cost via FFT enables thousands of gradient steps
within the time budget, outperforming LP-based methods in total iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lse_objective_and_grad(a, n, T):
    """
    Compute log-sum-exp(T * b) and its gradient w.r.t. a,
    where b = autocorrelation(a).
    Also handles the (sum a)^2 normalization.
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    # Log-sum-exp for numerical stability
    shifted = T * (b - max_b)
    log_sum = max_b + np.log(np.exp(shifted).sum()) / T
    lse_val = T * max_b + np.log(np.exp(shifted).sum())
    
    # Softmax weights
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient of LSE w.r.t. a:
    # ∂LSE/∂a[j] = T * Σ_k w[k] * ∂b[k]/∂a[j]
    # ∂b[k]/∂a[j] = a[k-j] + a[j] (when j ≠ k-j) or a[j] (when j = k-j)
    # This is 2 * (a ⋆ w)[j] approximately (cross-correlation)
    
    # More precisely: ∂b[k]/∂a[j] = a[k-j] (from i=j) + a[k-i where k-i=j, i.e., a[j] from i=k-j)
    # = a[k-j] + a[j] if k-j ≠ j, i.e., k ≠ 2j
    # = a[j] if k = 2j
    # Summing over k: Σ_k w[k] * (a[k-j] + a[k-j]) for k≠2j, and Σ_k w[k]*a[j] for k=2j
    # ≈ 2 * Σ_k w[k] * a[k-j] (the k=2j correction is small)
    
    # Compute cross-correlation of w with a using FFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:size] = w
    
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    
    # Cross-correlation: (a ⋆ w)[j] = Σ_i a[i+j] * w[i] = Σ_k a[k] * w[k-j]
    # In frequency: IFFT(FFT(a) * conj(FFT(w)))
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    
    # Gradient of LSE (before normalization)
    grad_lse = 2.0 * T * cross
    
    # Now account for normalization: we optimize f(a) = 2n * LSE / s^2
    # But actually, we can fix the normalization and just optimize LSE
    # Let's work with the normalized version: fix sum(a) = sqrt(2n)
    # Then the objective is just max_b (or LSE approximation)
    # And the constraint is sum(a) = sqrt(2n), handled by projection
    
    return lse_val, grad_lse


def project_nonneg_sum(a, target_sum):
    """Project a onto {x ≥ 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s


def adam_smoothed_descent(a0, time_limit, T_stages=None, verbose=True):
    """
    Adam optimizer on the smoothed (log-sum-exp) autocorrelation objective.
    The sequence is kept normalized to sum = sqrt(2n).
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = project_nonneg_sum(a, target_sum)
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5]
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr_base = 0.5
    step_count = 0
    
    for stage_i, T in enumerate(T_stages):
        stage_start = time.monotonic()
        # Time allocation: give more time to later (colder) stages
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_remaining_stages = len(T_stages) - stage_i
        stage_time = min(remaining / n_remaining_stages, 120.0)
        
        if verbose:
            print(f"    Stage {stage_i}: T={T}, time_budget={stage_time:.0f}s")
        
        iter_count = 0
        stagnant = 0
        last_best = best_score
        
        while time.monotonic() - t0 < time_limit:
            if time.monotonic() - stage_start > stage_time:
                break
            
            # Reset Adam state at each stage for fresh optimization
            if iter_count == 0:
                m = np.zeros(n)
                v = np.zeros(n)
                step_count = 0
            
            _, grad = lse_objective_and_grad(a, n, T)
            
            step_count += 1
            lr = lr_base / (1.0 + 0.01 * step_count)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**step_count)
            v_hat = v / (1 - beta2**step_count)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project to non-negative with correct sum
            a_new = project_nonneg_sum(a_new, target_sum)
            
            # Check for improvement
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                stagnant = 0
            else:
                stagnant += 1
            
            a = a_new
            iter_count += 1
            
            if stagnant > 500:
                break
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Stage {stage_i} done ({elapsed:.0f}s total): best={best_score:.8f}, iters={iter_count}")
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Peak-targeted coordinate descent: identify top autocorrelation peaks,
    move mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = project_nonneg_sum(a, target_sum)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * target_sum / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top peaks
        top_k = min(8, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:top_k]
        
        improved = False
        
        for k_peak in top_peaks:
            # Find contributing pairs for this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            
            # Build peak mask (positions involved in top peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:4]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if je2 >= js2:
                    peak_mask[js2:je2 + 1] = True
            
            # Try moving mass from high-contributing positions to low ones
            top_contrib = idx[np.argsort(contrib)[::-1][:4]]
            
            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                # Find target positions (not in peak mask, with low values)
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Sort by a value (move to positions with least impact on peaks)
                candidates_sorted = candidates[np.argsort(a[candidates])][:20]
                
                for dst in candidates_sorted[:10]:
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 3 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break
            
            # Larger perturbation every 50 stalls
            if stall % 50 == 0 and stall > 0:
                n_perturb = max(5, n // 50)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.3, 2.0)
                try_update(a_trial)
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    """Generate structured initial sequences with good potential."""
    choice = rng.integers(0, 10)
    if choice == 0:
        # Smooth bump
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 1:
        # Two bumps
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 2:
        # Asymmetric step (key insight: asymmetric can reduce center peak)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.8) ** 2)
    elif choice == 3:
        # Cosine window
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x / 2.0) ** 2
    elif choice == 4:
        # Linear ramp (asymmetric)
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - x + 0.1
    elif choice == 5:
        # Quadratic
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - (2 * x - 1) ** 2
    elif choice == 6:
        # Flat with boundary emphasis
        a = np.ones(n) * 0.5
        a[0] = 1.0
        a[-1] = 1.0
        a[1] = 0.8
        a[-2] = 0.8
    elif choice == 7:
        # Sine-modulated
        x = np.linspace(0.0, 1.0, n)
        a = 0.8 + 0.2 * np.cos(2 * np.pi * x)
    elif choice == 8:
        # Heavy tail (power law)
        x = np.linspace(0.01, 1.0, n)
        a = x ** (-0.5)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

Wait, I need to double-check the gradient computation. Let me be more careful.

We have b[k] = Σ_{i} a[i] * a[k-i] where the sum is over all i such that 0 ≤ i < n and 0 ≤ k-i < n.

∂b[k]/∂a[j] = Σ_i [δ_{i,j} * a[k-i] + a[i] * δ_{k-i,j}]
= a[k-j] (from the first term, when i=j, valid if 0≤k-j<n)
+ a[k-j] (from the second term, when k-i=j, i.e., i=k-j, valid if 0≤k-j<n)
= 2*a[k-j] (when j ≠ k-j, i.e., 2j ≠ k)
= a[j] (when 2j = k, since both terms give a[j])

So the gradient of LSE = Σ_k f(b[k]) where f(x) = exp(T*x)/Σexp(T*b[k]) (softmax weights) is:
∂LSE/∂a[j] = T * Σ_k w[k] * ∂b[k]/∂a[j]
= T * Σ_k w[k] * 2*a[k-j] (approximately, ignoring the 2j=k correction)
= 2T * Σ_k w[k] * a[k-j]

The last sum is the cross-correlation (w ⋆ a)[j] = Σ_k w[k] * a[k-j].

In terms of FFT: if we define the cross-correlation as C[j] = Σ_k w[k] * a[k-j], this is equivalent to the cross-correlation of w and a at lag j.

Using the FFT-based approach:
- Pad a and w to length L ≥ 2n-1

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smoothed gradient descent (log-sum-exp surrogate) with Adam
2. Adaptive temperature annealing (hot→cold) for progressive peak sharpening
3. Peak-targeted coordinate descent for final polish
4. Multi-length exploration with interpolation transfer
5. Multiple restarts with structured and perturbed initializations
Key advantage over LP-based methods: O(n log n) per iteration enables
thousands of gradient steps in the time budget, exploring the landscape
more thoroughly than expensive LP best-response iterations.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation of a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lse_grad(a, n, T):
    """
    Compute gradient of LSE(T * autocorr(a)) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) cost.
    """
    a = np.maximum(a, 0.0)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)

    # Softmax weights (numerically stable)
    shifted = T * (b - max_b)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    # Cross-correlation: G[j] = sum_k w[k] * a[k-j]
    # Computed via: ifft(fft(w) * conj(fft(a)))
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()

    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w

    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)

    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    return 2.0 * T * cross


def project_nonneg_sum(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.full(n, target_sum / n)
    return a * target_sum / s


def adam_smoothed_descent(a0, time_limit, T_stages=None, verbose=True):
    """
    Adam optimizer on smoothed (log-sum-exp) autocorrelation objective.
    Sequence is normalized to sum = sqrt(2n) at each step.
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = project_nonneg_sum(a, target_sum)

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.5]

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    total_iters = 0
    for stage_i, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_stages_left = len(T_stages) - stage_i
        stage_time = min(remaining / n_stages_left, 150.0)
        stage_start = time.monotonic()

        if verbose:
            print(f"    Stage {stage_i}: T={T:.1f}, budget={stage_time:.0f}s")

        # Fresh Adam state per stage
        m = np.zeros(n)
        v = np.zeros(n)
        beta1 = 0.8
        beta2 = 0.999
        eps = 1e-8
        step = 0
        stagnant = 0
        last_score = best_score

        while time.monotonic() - t0 < time_limit:
            if time.monotonic() - stage_start > stage_time:
                break

            grad = lse_grad(a, n, T)
            step += 1
            total_iters += 1

            # Adaptive learning rate
            lr = 0.8 / (1.0 + 0.005 * (step % 5000))

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad ** 2
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = project_nonneg_sum(a_new, target_sum)
            a = a_new

            # Check score
            score = fast_autocorr(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1

            # Early exit if very stagnant
            if stagnant > 2000:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Stage {stage_i} done (t={elapsed:.0f}s): best={best_score:.8f}")

def gradient_descent_momentum(a0, time_limit, T, lr_init=0.3, verbose=False):
    """
    Simpler gradient descent with Polyak momentum.
    Good for very high T where the landscape is smooth.
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = project_nonneg_sum(a, target_sum)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    v = np.zeros(n)
    momentum = 0.9
    step = 0

    while time.monotonic() - t0 < time_limit:
        grad = lse_grad(a, n, T)
        step += 1
        lr = lr_init / (1.0 + 0.01 * step)

        v = momentum * v + lr * grad
        a_new = a - v
        a = project_nonneg_sum(a_new, target_sum)

        score = fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        if step % 5000 == 0 and verbose:
            print(f"      GD iter={step}, score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Peak-targeted coordinate descent: move mass from positions contributing
    to top autocorrelation peaks toward positions with low peak involvement.
    """

    rng = np.random.default_rng(seed)
    target_sum = np.sqrt(2.0 * n)

    a = project_nonneg_sum(a, target_sum)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * target_sum / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-12:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

        b = fast_autocorr(a)
        top_k = min(10, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:top_k]

        improved = False

        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]

            # Build mask of positions involved in top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if je2 >= js2:
                    peak_mask[js2:je2 + 1] = True

            top_contrib = idx[np.argsort(contrib)[::-1][:5]]

            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates_sorted = candidates[np.argsort(a[candidates])][:30]

                for dst in candidates_sorted[:15]:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.08, 0.12]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-14:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 3 == 0:
                for _ in range(80):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.12)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break

            if stall % 40 == 0 and stall > 0:
                n_perturb = max(5, n // 40)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.3, 2.5)
                try_update(a_trial)

            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.5) ** 2)
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 2:
        # Asymmetric: concentrated on one side (key for reducing center peak)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.15) ** 2) + 0.3 * np.exp(-4.0 * (x - 0.85) ** 2)
    elif choice == 3:
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x / 2.0) ** 1.5
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - x) ** 0.7 + 0.1
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(8.0 * np.pi * x)
    elif choice == 6:
        a = np.ones(n) * 0.4
        a[0] = 1.0
        a[-1] = 1.0
        for i in range(min(20, n // 10)):
            a[i] = 1.0 - i * 0.02
            a[-1 - i] = 1.0 - i * 0.02
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
    elif choice == 8:
        # Triangle wave
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 9:
        # Power law decay from center
        center = n / 2.0
        idx = np.arange(n) - center
        a = np.exp(-0.3 * np.abs(idx) / center)
    elif choice == 10:
        # Step-like (explores asymmetric structures)
        a = np.ones(n)
        a[:n // 3] = 1.5
        a[2 * n // 3:] = 0.5
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: smoothed gradient descent with adaptive temperature,
    multi-length exploration, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_parent = len(best_seq) if best_seq is not None else 1600

    # === Phase 1: Main smoothed gradient descent at parent length ===
    print(f"\n=== Phase 1: Smoothed GD n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_init = best_seq * np.sqrt(2.0 * n_parent) / best_seq.sum()
        T_stages = [40.0, 20.0, 12.0, 7.0, 4.0, 2.5, 1.8]
        a_out, _ = adam_smoothed_descent(a_init, t1, T_stages, verbose=True)
        do_update(a_out, f"smoothed_n{n_parent}")

    # === Phase 2: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_parent + 400
    print(f"\n=== Phase 2: Smoothed GD n={n_long} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_init = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        T_stages = [40.0, 20.0, 12.0, 7.0, 4.0, 2.5]
        a_out, _ = adam_smoothed_descent(a_init, t2, T_stages, verbose=True)
        do_update(a_out, f"smoothed_long_n{n_long}")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} ===")
    t3 = min(remaining * 0.25, 300.0)
    if t3 > 20:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_init, t3, seed=12345)
        do_update(a_out, f"coord_n{n_best}")

    # === Phase 4: Perturbed restart with smoothed GD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Perturbed restart n={n_best} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        noise_scale = 0.04 * best_seq.mean()
        a_pert = best_seq + rng.normal(0, noise_scale, n_best)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert * np.sqrt(2.0 * n_best) / a_pert.sum()
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0]
        a_out, _ = adam_smoothed_descent(a_pert, t4, T_stages, verbose=False)
        do_update(a_out, f"perturbed_n{n_best}")

    # === Phase 5: Fresh structured initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Fresh init exploration ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 20:
        n_fresh = len(best_seq)
        n_attempts = 4
        for attempt in range(n_attempts):
            if time.monotonic() - t_global > time_budget - 20:
                break
            remaining_attempts = n_attempts - attempt
            attempt_time = min((time_budget - 20 - (time.monotonic() - t_global)) / remaining_attempts, 80.0)
            if attempt_time < 10:
                break
            a_fresh = structured_init(n_fresh, rng)
            a_fresh = a_fresh * np.sqrt(2.0 * n_fresh) / a_fresh.sum()
            T_stages = [25.0, 12.0, 6.0, 3.0, 2.0]
            a_out, _ = adam_smoothed_descent(a_fresh, attempt_time, T_stages, verbose=False)
            do_update(a_out, f"fresh_{attempt}_n{n_fresh}")

    # === Phase 6: Two-sided coordinate descent (alternating seeds) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final coord descent n={n_best} ===")
    t6 = remaining * 0.7
    if t6 > 15:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_init, t6, seed=77777)
        do_update(a_out, f"coord_final_n{n_best}")

    # === Phase 7: Quick re-optimization of best with lower T ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final polish T=1.0 n={n_best} ===")
        t7 = min(remaining * 0.8, 120.0)
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        T_stages = [3.0, 2.0, 1.5, 1.0]
        a_out, _ = adam_smoothed_descent(a_init, t7, T_stages, verbose=False)
        do_update(a_out, f"polish_T1_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
