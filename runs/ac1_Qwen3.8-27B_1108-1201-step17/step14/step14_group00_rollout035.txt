def build_T(a):

def lp_solve_full(a, M, solver):

def nesterov_lp(a0, time_limit, solver, verbose=True, seed=0, max_iters=5000):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    stall_threshold = 80

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        else:

def smooth_optimize(a0, time_limit, verbose=True):
    """Optimize using logsumexp smooth surrogate with projected gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    beta = 50.0  # Sharpness of max approximation
    lr = 50.0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        # Softmax weights
        bc = beta * conv
        bc_max = bc.max()
        w = np.exp(bc - bc_max)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(size) / size
            w_sum = size
        w = w / w_sum

        # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j] for i+j < size
        # Compute via FFT
        fft_size = 1 << (2 * size - 1).bit_length()
        w_r = w[::-1]
        A = rfft(a, fft_size)
        Wr = rfft(w_r, fft_size)
        conv_awr = irfft(A * Wr, fft_size)[:size]
        # C[i] = conv_awr[size-1-i] for i=0,...,n-1
        # grad[i] = 2 * C[i] = 2 * conv_awr[size-1-i]
        idx = size - 1 - np.arange(n)
        grad = 2.0 * conv_awr[idx]

        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            lr *= 0.5
        else:
            a_new = a_new * norm / s_new
            a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Adaptive learning rate
        if t % 100 == 0:
            beta = min(beta * 1.1, 500.0)
            lr *= 0.95

        if verbose and (t + 1) % 5000 == 0:
            print(f"    [Smooth] t={t+1} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}, beta={beta:.0f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def perturb_freq(a, rng, scale):
    n = len(a)
    a_f = rfft(a, 2 * n)
    # Add random noise to high frequencies
    for k in range(n // 4, n):
        a_f[k] += scale * rng.standard_normal()
    a_new = irfft(a_f, 2 * n)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def perturb_swap(a, rng, block_size):
    n = len(a)
    a_new = a.copy()
    n_blocks = n // (2 * block_size)
    if n_blocks < 1:
        return a_new
    b1 = rng.integers(0, n_blocks)
    b2 = rng.integers(0, n_blocks)
    if b1 == b2:
        return a_new
    s1 = b1 * block_size
    e1 = s1 + block_size
    s2 = b2 * block_size
    e2 = s2 + block_size
    tmp = a_new[s1:e1].copy()
    a_new[s1:e1] = a_new[s2:e2]
    a_new[s2:e2] = tmp
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP Nesterov at base length ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=0, max_iters=3000)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: CD polish n={n_best} ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t2, verbose=True, max_passes=1000, n_evals_per_coord=30)
        do_try_update(a_out, f"cd1_n{n_best}")

    # === Phase 3: Multiple perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Perturbation restarts ===")
    n_restarts = 5
    t3_total = min(remaining * 0.30, 400.0)
    t3_each = t3_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t3 = min(t3_each, remaining - 40)
        if t3 < 30:
            break

        n_cur = len(best_seq)
        # Different perturbation strategies
        if r == 0:
            pert = perturb_mult(best_seq, rng, 0.03)
        elif r == 1:
            pert = perturb_mult(best_seq, rng, 0.05)
        elif r == 2:
            pert = perturb_freq(best_seq, rng, 0.02)
        elif r == 3:
            pert = perturb_swap(best_seq, rng, max(10, n_cur // 20))
        else:
            pert = perturb_mult(best_seq, rng, 0.08)

        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        # Quick LP
        t_lp = t3 * 0.6
        a_out, _ = nesterov_lp(a_norm, t_lp, solver, verbose=False, seed=200 + r, max_iters=2000)
        do_try_update(a_out, f"pert{r}_lp_n{n_cur}")

        # Quick CD polish on result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and a_out is not None:
            n_r = len(a_out)
            a_norm2 = a_out * np.sqrt(2.0 * n_r) / a_out.sum()
            a_out2, _ = coord_descent(a_norm2, min(t3 * 0.4, remaining - 20), verbose=False, max_passes=500, n_evals_per_coord=25)
            do_try_update(a_out2, f"pert{r}_cd_n{n_r}")

    # === Phase 4: Smooth optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Smooth optimization n={n_best} ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smooth_optimize(a_norm, t4, verbose=True)
        do_try_update(a_out, f"smooth_n{n_best}")

    # === Phase 5: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    print(f"\n=== Phase 5: Longer length n={n2} ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 40:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        t_lp = t5 * 0.6
        a_out, _ = nesterov_lp(a_norm, t_lp, solver, verbose=True, seed=7, max_iters=2000)
        do_try_update(a_out, f"lp_long_n{n2}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and a_out is not None:
            a_norm2 = a_out * np.sqrt(2.0 * n2) / a_out.sum()
            a_out2, _ = coord_descent(a_norm2, min(t5 * 0.4, remaining - 10), verbose=False, max_passes=500, n_evals_per_coord=25)
            do_try_update(a_out2, f"cd_long_n{n2}")

    # === Phase 6: Try shorter length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = max(800, len(best_seq) - 400)
    print(f"\n=== Phase 6: Shorter length n={n3} ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 30:
        init_short = interpolate_sequence(best_seq, n3)
        a_norm = init_short * np.sqrt(2.0 * n3) / init_short.sum()
        a_out, _ = nesterov_lp(a_norm, t6 * 0.6, solver, verbose=False, seed=99, max_iters=1500)
        do_try_update(a_out, f"lp_short_n{n3}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and a_out is not None:
            a_norm2 = a_out * np.sqrt(2.0 * n3) / a_out.sum()
            a_out2, _ = coord_descent(a_norm2, min(t6 * 0.4, remaining - 10), verbose=False, max_passes=400, n_evals_per_coord=25)
            do_try_update(a_out2, f"cd_short_n{n3}")

    # === Phase 7: Final LP Nesterov on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP Nesterov n={n_best} ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t7, solver, verbose=True, seed=42, max_iters=2000)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final CD n={n_best} ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t8, verbose=True, max_passes=2000, n_evals_per_coord=35)
        do_try_update(a_out, f"cd_final_n{n_best}")

    # === Phase 9: One more perturbation + quick optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Final perturbation ===")
        pert = perturb_mult(best_seq, rng, 0.02)
        a_norm = pert * np.sqrt(2.0 * n_best) / pert.sum()
        a_out, _ = nesterov_lp(a_norm, min(remaining * 0.5, 60.0), solver, verbose=False, seed=777, max_iters=1000)
        do_try_update(a_out, "pert_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and a_out is not None:
            a_norm2 = a_out * np.sqrt(2.0 * n_best) / a_out.sum()
            a_out2, _ = coord_descent(a_norm2, min(remaining - 10, 50.0), verbose=False, max_passes=300, n_evals_per_coord=25)
            do_try_update(a_out2, "pert_final_cd")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
