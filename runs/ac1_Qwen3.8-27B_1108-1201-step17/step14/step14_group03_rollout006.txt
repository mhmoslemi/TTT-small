    return best_a, best_score


def softmax_subgradient(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(100.0, 0.05, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(100, min(2000, int(remaining * 5)))

        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        base_lr = norm / (T * np.sqrt(n))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            w = np.exp((conv - max_conv) / T)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(size)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

            w_padded = np.zeros(fft_size)
            w_padded[:min(size, fft_size)] = w[:min(size, fft_size)]
            W_fft = rfft(w_padded, fft_size)
            A_fft = rfft(a, fft_size)

            conv_wa = irfft(W_fft * A_fft, fft_size)[:size]
            B = np.zeros(fft_size)
            B[:n] = np.flip(a)
            B_fft = rfft(B, fft_size)
            conv_wb = irfft(W_fft * B_fft, fft_size)
            cross = conv_wb[n - 1:n - 1 + n]
            grad = cross + conv_wa[:n]

            decay = it / max(n_steps, 1)
            lr = base_lr * (1.0 - 0.5 * decay)

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            if it % 20 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SoftMax T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_adaptive(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        window = min(50, 10 + n_passes * 2)

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        delta_vec[lo1:hi1] += a[:hi1 - lo1]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        src2 = n - 1 - np.arange(lo2, hi2)
                        valid = (src2 >= 0) & (src2 < n)
                        delta_vec[lo2:hi2][valid] += a[src2[valid]]

                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        delta_vec[lo3:hi3] -= a[:hi3 - lo3]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        src4 = n - 1 - np.arange(lo4, hi4)
                        valid4 = (src4 >= 0) & (src4 < n)
                        delta_vec[lo4:hi4][valid4] -= a[src4[valid4]]

                    delta_vec *= 2.0

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 20
                    deltas = np.linspace(0.0, a[i] * 0.95, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-12:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-12:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def perturbation_restart(a, rng, strength=0.05):
    n = len(a)
    noise = rng.random(n) * strength
    a_perturbed = a + noise
    a_perturbed = np.maximum(a_perturbed, 0.0)
    s = a_perturbed.sum()
    if s < 1e-15:
        return a.copy()
    return a_perturbed * np.sqrt(2.0 * n) / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def cosine_init(n):
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2.0) ** 2
    s = a.sum()
    return a * np.sqrt(2.0 * n) / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === ROUND 1: Main optimization ===

    # Phase 1: LP Nesterov (35% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n_base}")

    # Phase 2: SoftMax subgradient with annealing (25% of remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: SoftMax Grad n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = softmax_subgradient(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"sm1_n{n_base}")

    # Phase 3: Pair CD (15% of remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Pair CD n={n_base} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = pair_cd_adaptive(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"pcd1_n{n_base}")

    # === ROUND 2: Perturbation + re-optimize ===

    # Phase 4: Perturbed LP Nesterov (15% of remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Perturbed LP Nesterov n={n_base} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_pert = perturbation_restart(best_seq * best_seq.sum() / best_seq.sum(), rng, strength=0.03)
        a_pert_norm = a_pert * np.sqrt(2.0 * n_base) / a_pert.sum()
        a_out, _ = nesterov_lp_optimize(a_pert_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_pert_n{n_base}")

    # Phase 5: SoftMax from perturbed starting point (10% of remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: SoftMax perturbed n={n_base} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_pert2 = perturbation_restart(best_seq * best_seq.sum() / best_seq.sum(), rng, strength=0.08)
        a_pert2_norm = a_pert2 * np.sqrt(2.0 * n_base) / a_pert2.sum()
        a_out, _ = softmax_subgradient(a_pert2_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"sm_pert_n{n_base}")

    # === ROUND 3: Alternate length exploration ===

    # Phase 6: Try longer sequence (10% of remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: Longer sequence exploration (budget={t6:.0f}s) ===")
    if t6 > 60:
        n_alt = n_base + 200
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_alt_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        t6a = min(t6 * 0.7, remaining * 0.4)
        a_out_alt, _ = nesterov_lp_optimize(a_alt_norm, t6a, solver, verbose=False)
        do_try_update(a_out_alt, f"lp_n{n_alt}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20 and best_seq is not None:
            a_out_alt2 = pair_cd_adaptive(
                best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum(),
                min(remaining2 * 0.4, 30.0), rng, verbose=False)
            do_try_update(a_out_alt2, f"pcd_alt_n{len(best_seq)}")

    # Phase 7: Final LP polish on best (remaining time, capped)
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.5, 120.0)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 7: Final LP polish n={n_best} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Phase 8: Final pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.6, 60.0)
    n_best = len(best_seq) if best_seq is not None else n_base
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_adaptive(a_norm, t8, rng, verbose=False)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
