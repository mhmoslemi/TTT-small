2. Block-coordinate primal-dual with FISTA inner solves (replaces slow LP)

Key innovation: O(n log n) primal-dual game replaces O(LP) iterations; intensive local search.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.full(n, target_sum / n)
    return w * target_sum / s


def lse_grad(a, T):
    """Gradient of log-sum-exp smoothed autocorrelation peak."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    if b_max < 1e-15:
        return np.zeros(n)
    shifted = (b - b_max) / T
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    # Gradient: 2 * sum_k w_k * (d(a*a)_k / da_j) = 2 * sum_k w_k * a_{k-j} * 1[valid]
    # = 2 * (w_cross a)_j where (w_cross a)_j = sum_l a_l * w_{j+l}
    # Use FFT: reverse a, convolve with w
    a_rev = a[::-1]
    conv_size = n + len(w) - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv_full = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv_full[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):
    """LSE-smoothed FISTA optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), 1.0)
        t_fista = 1
        staging = a.copy()
        while time.monotonic() - t0 < time_limit:
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if t_fista % 300 == 0:
                sc = fast_conv(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_conv(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            print(f"    LSE-FISTA T={T:.2f}: best={best_score:.8f}")
    return best_a, best_score
