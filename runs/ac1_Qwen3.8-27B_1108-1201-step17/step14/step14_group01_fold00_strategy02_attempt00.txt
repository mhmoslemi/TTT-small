    a = perturbed.copy() if perturbed is not None else np.ones(1000) / 1000.0
    n = np.random.randint(100, 1001)  # 100 to 1000 base length
    num_lengths_to_try = 4
    length_trials = [get_optimal_length() + i * length_step for i in range(num_lengths_to_try)]
    improvement_thresholds = [0.001, 0.0005, 0.0001]  # For score minimization (lower bound is better)

    for i in range(iteration_max):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget:
            print("Time budget exceeded, returning best candidate.")
            break

        # Adjust time allocation (from budget if time is free)
        remaining_time = time_budget - elapsed
        if remaining_time > 10:
            time_multiplier = 1.0 + (remaining_time / 3600)  # Extra time encourages exploration
        else:
            time_multiplier = 1.0

        # Determine the length for this iteration
        current_length = length_trials[i % num_lengths_to_try]
        # Resample the sequence to the new length
        a = interpolate_seq(np.array([float(x) for x in a]), current_length)  # Requires writing interpolate_seq function

        # Compute score and derivative
        score_here = evaluate_sequence(a.tolist())
        if np.isinf(score_here):
            # Reset gradient step to random
            a = np.random.rand(current_length) * 1000 / np.sum(a)
            continue

        # Gradient descent step defined earlier, optimize locally
        # Adapt gradient descent for potential local improvements
        grad_step, improved = gradient_descent_step(a, iterations=10, time_limit=time_multiplier)
        if improved and score_here < best_score:
            best_score = score_here
            best_known = a.copy()

        # Occasional mutation to escape local optima
        if np.random.rand() < 0.01:  # Mutation rate
            mutated_a = mutate(a)
            new_score = evaluate_sequence(mutated_a.tolist())
            if new_score < best_score * 0.999:  # Small improvement condition
                best_score = new_score
                best_known = mutated_a  # Update if beneficial

    # Fallback if no improvement
    if best_known is None and perturbed is not None:
        best_known = perturbed
        try:
            best_score = float(evaluate_sequence(best_known.tolist()))
        except Exception:

    # Final candidate is the best_known of small length if achiev
