Now for the polishing:
- We want to minimize max_k (a*a)_k subject to a >= 0, sum(a) = S
- At a local min, the subgradient of the active constraint should be orthogonal to the feasible set (modulo the normal cone of a >= 0)
- This means: for all i with a_i > 0, g_i - lambda = 0 for some lambda
- I.e., for all i with a_i > 0: 2*a_{k*-i} = lambda (where k* is the peak lag)

A practical algorithm:
1. Find k* = argmax (a*a)_k
2. Compute g = 2 * reverse(a) shifted by k* (i.e., g[i] = 2*a[k*-i] if valid)
3. We want to find lambda such that:

   - For i with a_i = 0: g_i <= lambda
4. Update: a_i = max(0, a_i - alpha * (g_i - lambda))
5. Renormalize to sum = S

This is essentially a projected subgradient step. The choice of lambda is important. A good choice is the average of g_i over the support of a:
lambda = mean(g_i for i where a_i > 0)

Or better: lambda = the value that makes sum(a) = S after the step.

    t0 = time.monotonic()
    
    # Adaptive step size
    alpha = norm * 0.1
    
    for t in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break

        k_star = int(np.argmax(b))
        
        # Compute subgradient
        g = np.zeros(n)
        for i in range(n):

            if 0 <= j < n:

        support = np.where(a > norm * 1e-8)[0]

            lambda_val = g[support].mean()
            g_centered = g - lambda_val
        else:
            g_centered = g - g.mean()

        a_new = project_simplex(a_new, norm)

        if sc_new < best_score - 1e-15:

            # Increase step slightly if making progress
            alpha *= 1.001
        else:
            # Decrease step
            alpha *= 0.995
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={t} ({elapsed:.0f}s): best={best_score:.8f}, alpha={alpha:.6f}")
    
    return best_a, best_score

def peak_shave_polish(a0, time_limit, seed=12345):
    """Targeted peak reduction via mass redistribution from peak-contributing terms."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False
        
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find destination: positions that contribute least to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            # Prefer low-value positions
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]
            
            for frac in [0.0002, 0.0005, 0.001, 0.003, 0.007, 0.015]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def project_simplex(v, target_sum):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def get_lp_solver():
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): CLARABEL")
    return cp.CLARABEL


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lse_fista(a0, T_schedule, time_limit):

        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    alpha = norm_val * 0.05
    best_alpha = alpha
    stall = 0

    for t in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Compute subgradient: g[i] = 2*a[k*-i] if 0<=k*-i<n else 0
        g = np.zeros(n)
        valid_start = max(0, k_star - (n - 1))
        valid_end = min(n - 1, k_star)
        for i in range(valid_start, valid_end + 1):
            g[i] = 2.0 * a[k_star - i]

        # Center the gradient
        support = np.where(a > norm_val * 1e-8)[0]
        if len(support) > 0:
            lambda_val = g[support].mean()
        else:
            lambda_val = g.mean()
        g_centered = g - lambda_val

        # Step
        a_new = a - alpha * g_centered
        a_new = project_simplex(a_new, norm_val)

        sc_new = fast_autocorr(a_new).max()
        if sc_new < best_score - 1e-15:
            a = a_new
            best_score = sc_new
            best_a = a.copy()
            best_alpha = alpha
            stall = 0
        else:
            stall += 1
            alpha *= 0.99

        # Adaptive alpha
        if stall > 100:
            alpha = best_alpha * (0.5 + 0.5 * np.exp(-stall / 200.0))
            stall = 0

        if t % 2000 == 0 and t > 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={t} ({elapsed:.0f}s): best={best_score:.8f}, alpha={alpha:.6f}")

    return best_a, best_score


def peak_shave_polish(a0, time_limit, seed_val=12345):
    rng = np.random.default_rng(seed_val)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find destination: positions contributing least to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.0001, 0.0003, 0.001, 0.003, 0.007, 0.015, 0.03]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j2 = rng.integers(n)
            if i != j2 and a[i] > norm_val * 1e-10:
                frac = rng.uniform(0.0005, 0.008)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j2] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm_val / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            stall = 0
            if stall > 600:
                break

        if stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit):
    """Local search: try pair swaps and small perturbations systematically."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    # Sort indices by value for systematic exploration
    order = np.argsort(a)

    while time.monotonic() - t0 < time_limit:
        improved_any = False
        # Try moving mass from high-valued to low-valued positions
        for trial in range(200):
            i = order[trial % n]
            j2 = order[(n - 1 - trial) % n]
            if a[i] < norm_val * 1e-10:
                continue
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j2] += amt
                if a_trial[i] < 0:
                    break
                a_trial = a_trial * norm_val / a_trial.sum()
                sc = fast_autocorr(a_trial).max()
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved_any = True
                    break
        # Also try symmetric perturbations
        for trial in range(100):
            i = trial % n
            j2 = (n - 1 - trial) % n
            if i == j2:
                continue
            dm = (a[i] - a[j2]) * 0.01
            a_trial = a.copy()
            a_trial[i] -= dm
            a_trial[j2] += dm
            if a_trial[i] < 0 or a_trial[j2] < 0:
                continue
            a_trial = a_trial * norm_val / a_trial.sum()
            sc = fast_autocorr(a_trial).max()
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved_any = True

        if not improved_any:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1680.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Helper to try updating best (inline to avoid closure)
    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 50.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [10.0, 3.0, 1.0, 0.3, 0.1]
        a_out, _ = lse_fista(best_seq, T_sched, t1)
        do_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.42, 650.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t2, solver)
        do_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.10, 100.0)
    if t3 > 10:
        print(f"\n=== Phase 3: Subgrad polish n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t3)
        do_update(a_out, f"subgrad_n{n_best}")

    # === Phase 4: Peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.08, 80.0)
    if t4 > 10:
        print(f"\n=== Phase 4: Peak shave n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave_polish(a_norm, t4, seed_val=777)
        do_update(a_out, f"peakshave_n{n_best}")

    # === Phase 5: LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP round 2 n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_enhanced(a_norm, t5, solver)
        do_update(a_out, f"lp2_n{n_best}")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.05, 50.0)
    if t6 > 5:
        print(f"\n=== Phase 6: Coordinate polish n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_polish(a_norm, t6)
        do_update(a_out, f"coord_n{n_best}")

    # === Phase 7: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n2 = len(best_seq) + 200
        t7 = min(remaining * 0.12, 130.0)
        if t7 > 30:
            print(f"\n=== Phase 7: LP at n={n2} (budget={t7:.0f}s) ===")
            init_long = interpolate_sequence(best_seq, n2)
            a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
            a_out, _ = lp_nesterov_enhanced(a_norm, t7, solver)
            do_update(a_out, f"lp3_n{n2}")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.4, 50.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final peak shave n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_shave_polish(a_norm, t8, seed_val=99999)
            do_update(a_out, "final_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
