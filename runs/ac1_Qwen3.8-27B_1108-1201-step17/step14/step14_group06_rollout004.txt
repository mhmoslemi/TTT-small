def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(100):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_state, candidate_a, label, t_global):
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: Cutting-plane LP Nesterov + corrected LSE + subgradient CD + multi-length."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov with cutting plane (main workhorse, 42% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest cutting plane n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.42
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_tight_M=True, use_cutting=True)
    best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_cut_n{n_best}", t_global)

    # Phase 2: LSE gradient with CORRECT cooling (smooth to sharp)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE corrected cooling n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.14, 180.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=5.0, T_end=100.0, n_stages=6)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lse_corr_n{n_best}", t_global)

    # Phase 3: LP Nesterov round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-Nest round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=False, use_tight_M=True, use_cutting=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_r2_n{n_best}", t_global)

    # Phase 4: Try longer sequence n+400
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_best + 400
    print(f"\n=== Phase 4: LP-Nest longer n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, use_tight_M=True, use_cutting=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_long_n{n_long}", t_global)

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_coord_descent(a_norm, t5, seed=42)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord1", t_global)

    # Phase 6: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        half = remaining / 2.0

        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, half, solver, verbose=False, use_tight_M=True, use_cutting=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "lp_final", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15:
            n_best2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            a_out, _ = subgradient_coord_descent(a_norm2, remaining2, seed=12345)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
