import numpy as np

def fast_autocorr(a):
    """Compute full linear convolution (autocorrelation) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def normalize_seq(a, S):
    """Clip to non-negative and rescale to sum S."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def lse_gradient_step(a, T, fft_size, n):
    """Compute the LSE gradient using FFT cross-correlation.
    
    Returns grad where grad[i] = (2/T) * sum_j w[i+j] * a[j]
    """
    size = 2 * n - 1
    conv_a = fast_autocorr(a)
    max_conv = conv_a.max()
    
    # Softmax weights (numerically stable)
    w = np.exp((conv_a - max_conv) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[int(conv_a.argmax())] = 1.0
    else:
        w = w / w_sum
    
    # Cross-correlation via FFT: corr[i] = sum_j w[i+j] * a[j]
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(A) * W, fft_size)[:n]
    
    grad = (2.0 / T) * corr
    return grad, conv_a, max_conv

                for k in check_positions:
                    if k not in top_positions:
                        vi = 2.0 * a[k - i] if (0 <= k - i < n) else 0.0
                        vj = 2.0 * a[k - j] if (0 <= k - j < n) else 0.0
                        dc_k = vj - vi
                        if dc_k > 1e-15:
                            slack = best_score - conv[k]
                            if slack < 0:
                                max_delta = 0
                                break
                            delta_k = slack / dc_k
                            if delta_k < max_delta:
                                max_delta = delta_k
                
                if max_delta < 1e-12:
                    continue
                
                # Use a fraction of max_delta for safety
                delta = 0.95 * max_delta
                if delta > a[i] * 0.5:
                    delta = a[i] * 0.5
                
                # Verify and apply
                a[i] -= delta
                a[j] += delta
                a = a * S / a.sum()
                
                new_conv = fast_autocorr(a)
                new_max = new_conv.max()
                
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    conv = new_conv
                    pass_improved = True
                    pass_improvement += best_score  # track
                else:
                    # Undo
                    a[i] += delta
                    a[j] -= delta
                    a = a * S / a.sum()
                    conv = fast_autocorr(a)
            
            # Early break if no improvement in this pass
            if not pass_improved:
                break
        
        if not pass_improved:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
        else:
            no_improve_count = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def single_cd_refine(a0, time_limit, rng, verbose=True):
    """Single-coordinate descent: try moving all weight from one coordinate
    to distribute across several others."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * S / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < S * 1e-8:
                continue
            
            # Try reducing a[i] and distributing to neighbors
            # The best target minimizes the increase in max conv
            # Try moving to a single best target
            best_j = -1
            best_delta = 0.0
            
            # Try a few targets: neighbors and random
            targets = list(set([
                (i - 1) % n, (i + 1) % n,
                (i - 2) % n, (i + 2) % n,
                int(rng.integers(n)),
                int(np.argmin(a)),
            ]))
            
            for j in targets:
                if j == i:
                    continue
                # Compute max allowable delta
                max_delta = a[i]
                # Check all positions where conv increases
                for k in range(size):
                    vi = 2.0 * a[k - i] if (0 <= k - i < n) else 0.0
                    vj = 2.0 * a[k - j] if (0 <= k - j < n) else 0.0
                    dc = vj - vi
                    if dc > 1e-15:
                        slack = best_score - conv[k]
                        if slack < 0:
                            max_delta = 0
                            break
                        d = slack / dc
                        if d < max_delta:
                            max_delta = d
                
                if max_delta > best_delta:
                    best_delta = max_delta
                    best_j = j
            
            if best_j >= 0 and best_delta > 1e-12:
                delta = 0.9 * best_delta
                a[i] -= delta
                a[best_j] += delta
                a = a * S / a.sum()
                new_conv = fast_autocorr(a)
                new_max = new_conv.max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    conv = new_conv
                    pass_improved = True
                else:
                    a[i] += delta
                    a[best_j] -= delta
                    a = a * S / a.sum()
                    conv = fast_autocorr(a)
        
        if not pass_improved:
            break
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def multi_start_lse(a0, time_limit, rng, n_restarts=3, verbose=True):
    """Multiple restarts of LSE from perturbed versions of a0."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    best_a = a0.copy()
    best_score = fast_autocorr(a0).max()
    t0 = time.monotonic()
    
    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        time_per_restart = remaining / (n_restarts - r)
        
        # Perturb: mix with random
        noise = rng.random(n) * 0.1
        a_perturbed = a0 * (1.0 + noise)
        a_perturbed = normalize_seq(a_perturbed, S)
        
        a_out, _ = lse_momentum_descent(
            a_perturbed, time_per_restart * 0.8,
            T_start=2.0, T_end=0.2, n_phases=5, verbose=False
        )
        
        sc = fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    Multi-start {r}: improved to {best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-phase optimization to minimize the score."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    n = len(best_seq)
    
    # === Phase 1: LSE Gradient Descent (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LSE-GD n={n} (budget={t1:.0f}s) ===")
    if t1 > 20:
        S = np.sqrt(2.0 * n)
        a_init = best_seq * S / best_seq.sum()
        a_out, _ = lse_momentum_descent(a_init, t1, T_start=5.0, T_end=0.1, n_phases=12, verbose=True)
        do_try_update(a_out, f"lse1_n{n}")
    
    # === Phase 2: Pair CD (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Pair CD n={n} (budget={t2:.0f}s) ===")
    if t2 > 15:
        S = np.sqrt(2.0 * n)
        a_init = best_seq * S / best_seq.sum()
        a_out, _ = pair_cd_descent(a_init, t2, rng, top_k=8, verbose=True)
        do_try_update(a_out, f"pcd1_n{n}")
    
    # === Phase 3: Second LSE with tighter schedule (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LSE-GD round 2 n={n} (budget={t3:.0f}s) ===")
    if t3 > 15:
        S = np.sqrt(2.0 * n)
        a_init = best_seq * S / best_seq.sum()
        a_out, _ = lse_momentum_descent(a_init, t3, T_start=2.0, T_end=0.05, n_phases=10, verbose=True)
        do_try_update(a_out, f"lse2_n{n}")
    
    # === Phase 4: Pair CD round 2 (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Pair CD round 2 n={n} (budget={t4:.0f}s) ===")
    if t4 > 10:
        S = np.sqrt(2.0 * n)
        a_init = best_seq * S / best_seq.sum()
        a_out, _ = pair_cd_descent(a_init, t4, rng, top_k=15, verbose=True)
        do_try_update(a_out, f"pcd2_n{n}")
    
    # === Phase 5: Multi-start exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Multi-start LSE n={n} (budget={t5:.0f}s) ===")
    if t5 > 30:
        S = np.sqrt(2.0 * n)
        a_init = best_seq * S / best_seq.sum()
        a_out, _ = multi_start_lse(a_init, t5, rng, n_restarts=2, verbose=True)
        do_try_update(a_out, f"ms_n{n}")
    
    # === Phase 6: Final polish CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n = len(best_seq)
    t6 = remaining * 0.30
    print(f"\n=== Phase 6: Final CD n={n} (budget={t6:.0f}s) ===")
    if t6 > 10:
        S = np.sqrt(2.0 * n)
        a_init = best_seq * S / best_seq.sum()
        a_out, _ = pair_cd_descent(a_init, t6, rng, top_k=20, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n}")
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
