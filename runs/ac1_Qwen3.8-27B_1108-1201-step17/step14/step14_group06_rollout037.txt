    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

        b = b * np.exp(-lr * grad_b / max(b.max(), 1e-10))
        b = np.maximum(b, 0)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.MOSEK]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def build_T(a, n):
    """Build constraint matrix T where T[k,i] = a[k-i] if valid."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[start - i : end - i]
    return T

def lp_best_response(a, M, solver, T=None):

        return None

        return None

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, mode=0):

def fast_autocorr(a):

    n = len(a)

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T(a, n):
    """Build constraint matrix T where T[k,i] = a[k-i] if 0<=k-i<n."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[start - i:end - i]
    return T

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        # Two-bump
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + 0.7 * np.exp(-20 * (x - 0.7) ** 2)
    a = np.maximum(a, 1e-8)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def try_update(best_seq, best_score, candidate, label, t_global):
    """Try to update best state. Returns (best_seq, best_score)."""
    if candidate is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-length LP min-peak best-response with Nesterov,
    targeted coordinate descent, and running-average exploitation.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        best_seq = structured_init(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov on base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 1: LP-Nest n={n_base} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
    best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp_n{n_base}", t_global)

    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.10, 130.0)
    print(f"\n=== Phase 2: Coord descent n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t2, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"cd1_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov round 2 (warmed start) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 3: LP-Nest r2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp_r2_n{n_cur}", t_global)

    # === Phase 4: Try longer sequences ===
    longer_lengths = []
    if n_cur < 2000:
        longer_lengths.append(2000)
    if n_cur < 2800:
        longer_lengths.append(2800)
    if n_cur < 3600:
        longer_lengths.append(3600)

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        t_phase = min(remaining * 0.22, 300.0)
        if t_phase < 60:
            break

        print(f"\n=== Phase 4: LP-Nest n={new_n} (budget={t_phase:.0f}s) ===")
        init = interpolate_to_length(best_seq, new_n)
        a_norm = init * np.sqrt(2.0 * new_n) / init.sum()
        a_out, _ = nesterov_lp(a_norm, t_phase, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp_long_n{new_n}", t_global)

        # Quick coord descent on the longer sequence
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == new_n:
            t_cd = min(remaining * 0.08, 80.0)
            if t_cd > 15:
                a_norm2 = best_seq * np.sqrt(2.0 * new_n) / best_seq.sum()
                a_out2, _ = coord_descent(a_norm2, t_cd, seed=777)
                best_seq, best_score = try_update(best_seq, best_score, a_out2, f"cd_long_n{new_n}", t_global)

    # === Phase 5: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.35, 250.0)
    print(f"\n=== Phase 5: Final polish n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 20:
        half = t5 / 2.0
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, half, solver, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15:
            n_cur2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_cur2) / best_seq.sum()
            a_out2, _ = coord_descent(a_norm2, remaining2, seed=99999)
            best_seq, best_score = try_update(best_seq, best_score, a_out2, "cd_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
