        return np.inf
    try:
        seq_length_ok = isinstance(x, (list, np.ndarray)) and 10 <= len(x) <= n_max_possible
    except:
        return np.inf
    x_floats = [float(y) for y in x]
    x_np = np.array(x_floats, dtype=float)
    for val in x_np:
        if val < 0 or np.isinf(val):
            return np.inf
            
    # We clip the values in the evaluate sequence function
    x_clipped = np.maximum(x_np, 0.0)
    x_clipped = np.minimum(x_np * 1.0, 1000.0 * len(x_np) / sum(x_np))
    # But for simplicity, we don't clip here and assume optimizer takes care
    
    # Normalize the x array to have a sum of 0.01
    total = np.sum(x_clipped)
    if abs(total) < 1e-5:
        x_clipped *= 0.01 / (len(x_clipped) * 1e-10)
    # Clip to [0, 0.01] but use the real normalization instead
    obj_val = 2 * len(x_clipped) * compute_conv(x_clipped).max() / (total ** 2)
    
    # We have to check if nan is returned
    if np.isnan(obj_val):
        return np.inf
        
    return float(obj_val)

# Convex constraints: we'll use projected gradient descent
def nonneg_constraint(x):
    return np.maximum(x, 0.0)

def optimize_sequence(time_budget=1800):

            break

        perturbation = np.random.normal(0, 0.05, n_candidates)  # Small random perturbation

            new_seq = new_seq * (0.02 / compute_objective(new_seq)[0] if compute_objective(new_seq) < 2.0 else 1)

        elapsed_remaining = time_budget - (time.time()-start_time)

This implementation of `optimize_sequence`:
1. Starts with a global sequence if available, otherwise uses (1,n)-normalized weights.
2. Uses a combination of L-BFGS-B and SLSQP optimizers to handle the non-convex optimization problem.
3. The objective is scaled by the square of the sum, which is automatically handled.
4. We clip negative values (handled by the constraint) and ensure the values don't exceed 1000.

The search strategy:
- Iterates candidate lengths exponentially (doubling or halving).
- Uses perturbation and gradient descent for fine tuning.
- Handles local minimization based on objective values.

This should find a candidate that wins the bound of about 2 (theoretical lower bound). The returned sequence is stored explicitly and validated.
