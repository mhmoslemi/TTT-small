import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]

        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_max = rho[-1]
    theta = (cssv[rho_max] - 1) / (rho_max + 1)
    return np.maximum(v - theta, 0)

def smooth_objective_and_grad(a, tau):
    """Compute the log-sum-exp smooth approximation and its gradient."""
    ac = fast_autocorr(a)
    max_ac = ac.max()
    shifted = tau * (ac - max_ac)
    # For numerical stability
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / w_sum
    
    obj = max_ac + np.log(w_sum) / tau
    
    # Gradient: grad_j = 2 * sum_k w_k * a[|k-j|]
    n = len(a)
    # Vectorized computation
    k_idx = np.arange(2 * n - 1)
    j_idx = np.arange(n)
    diff = np.abs(k_idx[None, :] - j_idx[:, None])  # (n, 2n-1)
    valid = (diff < n).astype(np.float64)
    # a[diff] where diff < n, else 0
    a_vals = np.zeros_like(diff, dtype=np.float64)
    mask = diff < n
    a_vals[mask] = a[diff[mask]]
    grad = 2.0 * (a_vals * valid * w[None, :]).sum(axis=1)
    
    return obj, grad

def fista_smooth(a0, time_limit, tau_schedule, verbose=True):
    """
    FISTA with log-sum-exp smooth approximation.
    tau_schedule: list of (tau, n_iters) tuples.
    """

    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()

    a_prev = a.copy()

            break
        
        # Estimate Lipschitz constant
        # L ≈ 2 * tau * max(w) * 2 * max(a) (rough bound)
        # Safer: use backtracking
        step = 1.0 / (2.0 * tau * max(a.max(), 1e-6))
        # Actually let's use a more conservative step
        step = min(step, 1.0 / (2.0 * tau))
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Evaluate objective and gradient at y
            obj_y, grad_y = smooth_objective_and_grad(y, tau)
            
            # Gradient step
            z = y - step * grad_y
            
            # Project onto simplex
            a_new = project_simplex(z)
            
            # Momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0
            y_new = a_new + (t_mom - 1.0) / t_new * (a_new - a_prev)
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = a_new.copy()
            else:
                # Re-normalize (y doesn't need to be on the simplex, but keeping it bounded helps)
                # Actually in FISTA, y is just the "extrapolated" point, it doesn't need to be on the constraint set
                pass
            
            a_prev = a_new
            a_new2 = a_new
            y = y_new
            t_mom = t_new
            
            # Check improvement
            M_curr = fast_autocorr(a_new2).max()
            if M_curr < best_obj:
                best_obj = M_curr
                best_a = a_new2.copy()
            
            # Adaptive step size (backtracking)
            if it > 0 and obj_y > fista_prev_obj + 1e-10:
                step *= 0.5
            fista_prev_obj = obj_y
        
        # Reset momentum between tau phases
        y = best_a.copy()
        a_prev = best_a.copy()
        t_mom = 1.0
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj
            print(f"      FISTA tau={tau} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_obj

def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:min(8, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            peak_indices = top_k
            best_dst = -1
            best_score = float('inf')
            
            # Sample candidates
            n_candidates = min(n, 100)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peak_indices:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2+1] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) < n_candidates:
                candidates = np.where(rng.random(n) < 0.5)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:n_candidates]]
            
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in peak_indices:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens += add
                net_sens = sens
                if net_sens < best_score:
                    best_score = net_sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_obj - 1e-14:
                    a = a_trial
                    best_obj = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with log-sum-exp smooth approximation (adaptive temperature)
2. Multi-start with parametric initializations

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum 1, ensure non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1)
    rho = np.where(cond > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_max = rho[-1]
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(a, w):
    """
    Compute gradient of log-sum-exp objective.
    grad_j = 2 * sum_k w_k * a[|k-j|]
    Uses O(n^2) direct computation (fast enough for n <= 3000).
    """
    n = len(a)
    # Vectorized: for each j, sum over k of w[k] * a[|k-j|]
    k_idx = np.arange(2 * n - 1)
    j_idx = np.arange(n)
    diff = np.abs(k_idx[None, :] - j_idx[:, None])
    valid = diff < n
    # Build a_values: a_values[j, k] = a[|k-j|] if valid, else 0
    a_vals = np.where(valid, a[np.minimum(diff, n-1)], 0.0)
    grad = 2.0 * (a_vals * w[None, :]).sum(axis=1)
    return grad

def fista_smooth(a0, time_limit, tau_vals, iters_per_tau=2000, verbose=True):
    """
    FISTA with log-sum-exp smooth approximation.
    tau_vals: list of tau values to use (in order).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_ac_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for tau in tau_vals:
        if time.monotonic() - t0 > time_limit:
            break
        
        y = a.copy()
        a_prev = a.copy()
        t_mom = 1.0
        step = 1.0 / (2.0 * tau * max(a.max(), 1e-10))
        
        for it in range(iters_per_tau):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient at y
            ac_y = fast_autocorr(y)
            max_ac_y = ac_y.max()
            shifted = tau * (ac_y - max_ac_y)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(ac_y)) / len(ac_y)
            else:
                w = w / w_sum
            
            grad = compute_smooth_grad(y, w)
            
            # Gradient step + projection
            z = y - step * grad
            a_new = project_simplex(z)
            
            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
            y_new = a_new + (t_mom - 1.0) / t_new * (a_new - a_prev)
            y_new = np.maximum(y_new, 0.0)
            
            a_prev = a_new
            a = a_new
            y = y_new
            t_mom = t_new
            
            # Track best
            M_curr = fast_autocorr(a).max()
            if M_curr < best_ac_max:
                best_ac_max = M_curr
                best_a = a.copy()
            
            # Adapt step: try to increase if making progress
            if it > 0 and it % 200 == 0:
                # Check if we've been making progress
                # Simple heuristic: increase step by 5%
                step *= 1.05
        
        a = best_a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_ac_max
            print(f"      FISTA tau={tau} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_ac_max

def coordinate_polish_fast(a0, time_limit, verbose=False):

def make_bell(n, width):
