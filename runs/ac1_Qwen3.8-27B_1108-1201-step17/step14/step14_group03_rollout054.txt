Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

3. Multi-length exploration

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        lo = j
        hi = min(size, j + n)
        length = hi - lo
        T[lo:hi, j] = a[:length]
    return T

def lp_best_response(a, M):
    """Solve max 1^T b s.t. T_a @ b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    """LP Nesterov optimization with restart."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    n_iters = 0

    stagnation_count = 0

        if M_curr < 1e-12:

        b = lp_best_response(y, M_curr)

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:

        else:

        else:

                stagnation_count = 0

            stagnation_count = 0

def lp_best_response(a, M):

    size = 2 * n - 1
    T = build_T(a)
    c = -np.ones(n)

    bounds = [(0.0, None)] * n
    try:

def nesterov_lp_optimize(a0, time_limit, verbose=True):

        sy = y_new.sum()
        if sy > 1e-15:

                    # conv_new = conv + delta * lin + delta^2 * quad
                    # lin_k = 2*a_{k-j} - 2*a_{k-i} (where valid)
