def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smoothed_gradient(a, w_full):
    """
    Compute gradient of smoothed objective using FFT.
    w_full: softmax weights of length 2n-1
    Returns gradient of length n.
    """
    n = len(a)
    w_rev = w_full[::-1]
    # convolve(a, w_rev) has length 3n-2
    # grad_i = 2 * conv(a, w_rev)[2n-2-i]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    WR = rfft(w_rev, fft_size)
    conv_result = irfft(A * WR, fft_size)
    # We need elements at indices 2n-2, 2n-3, ..., n-1
    # i.e., conv_result[2n-2-i] for i = 0, ..., n-1
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_result[idx]
    return grad


def smoothed_gradient_descent(a0, time_limit, T_start=None, T_end=None, verbose=True):
    """
    Smoothed gradient descent with adaptive learning rate.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    if T_start is None:
        T_start = best_score * 0.1
    if T_end is None:
        T_end = max(0.001, best_score * 0.001)
    
    # Learning rate schedule
    lr = 0.01 / n
    lr_min = 1e-10 / n
    momentum = 0.9
    v = np.zeros(n)  # momentum buffer
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        c = fast_autocorr(a)
        max_c = c.max()
        
        # Compute softmax weights (shift to avoid overflow)
        shifted = (c - max_c) / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = compute_smoothed_gradient(a, w)
        
        # Project gradient onto tangent space of {sum(a) = norm}
        grad -= np.mean(grad)
        
        # Adam-style update
        m = momentum * v + (1 - momentum) * grad
        v = m
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # Slightly increase learning rate on improvement
            lr = min(lr * 1.01, 1.0 / n)
        else:
            # Decrease learning rate
            lr = max(lr * 0.995, lr_min)
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}, lr={lr:.2e}")
        
        # Reset momentum periodically
        if steps % 10000 == 0:
            v = np.zeros(n)
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def peak_shave_fast(a0, time_limit, seed=42, verbose=True):
    """Fast peak shaving with batch evaluations."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find contributing positions
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 100:
                break
            continue
        
        # Move mass from top contributor to a non-contributing position
        src = idx[np.argmax(contrib)]
        
        # Find best destination
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        non_contrib[src] = True
        dst_candidates = np.where(non_contrib)[0]
        
        if len(dst_candidates) == 0:
            dst_candidates = np.array([i for i in range(n) if i != src])
        
        # Pick from low-value destinations
        n_select = min(5, len(dst_candidates))
        sorted_dst = dst_candidates[np.argsort(a[dst_candidates])][:n_select]
        dst = sorted_dst[rng.integers(len(sorted_dst))]
        
        # Try a few fractions
        best_trial_score = max_b
        best_frac = 0
        src_val = a[src]
        for frac in [0.1, 0.2, 0.3]:
            amt = src_val * frac
            if amt < 1e-18:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_frac = frac
        
        if best_frac > 0:
            a[src] -= src_val * best_frac
            a[dst] += src_val * best_frac
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                n_stall += 1
        else:
            n_stall += 1
        
        if n_stall > 200:
            # Random perturbation
            n_pert = max(3, n // 100)

                i = rng.integers(n)
                j = rng.integers(n)
                if i != j:
                    amt = a[i] * rng.uniform(0.05, 0.3)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            n_stall = 0
        
        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, target_sum):
    """Normalize non-negative array to have the target sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def compute_smoothed_gradient_fft(a, w_full):
    """
    Compute gradient of smoothed objective T*logsumexp(c/T) where c = autocorr(a).
    w_full: softmax weights of length 2n-1.
    grad_i = 2 * sum_{j=0}^{n-1} a_j * w_{i+j} (with w_k=0 for k>=2n-1)
    """
    n = len(a)
    # w_rev[k] = w_full[2n-2-k]
    w_rev = w_full[::-1]
    # grad_i = 2 * conv(a, w_rev)[2n-2-i] for i=0,...,n-1
    # conv(a, w_rev) has length n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    WR = rfft(w_rev, fft_size)
    conv_result = irfft(A * WR, fft_size)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_result[idx]
    return grad


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        valid_k = np.arange(j, j + n)
        T[valid_k, j] = a
    return T


def smoothed_gradient_descent(a0, time_limit, T_start=None, T_end=None, verbose=True, gamma=1.8):
    """
    Nesterov-accelerated smoothed gradient descent.
    Minimizes the log-sum-exp smoothed autocorrelation peak.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)

    c0 = fast_autocorr(a)
    best_score = c0.max()
    best_a = a.copy()

    if T_start is None:
        T_start = best_score * 0.08
    if T_end is None:
        T_end = max(0.0005, best_score * 0.0005)

    t0 = time.monotonic()
    steps = 0
    lr = 0.5 / n
    v = np.zeros(n)  # momentum velocity
    y = a.copy()  # Nesterov lookahead

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        c = fast_autocorr(y)
        max_c = c.max()

        # Softmax weights
        shifted = (c - max_c) / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum

        # Gradient
        grad = compute_smoothed_gradient_fft(y, w)
        grad -= np.mean(grad)

        # Nesterov step
        t_step = steps + 1
        alpha = 2.0 / (t_step + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        # Gradient step
        y_new = y - lr * grad
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to(y_new, norm)

        # Nesterov momentum
        if steps == 0:
            a_new = y_new
        else:
            momentum = (steps + 1.0 - gamma) / (steps + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - a)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, norm)

        y = y_new
        a = a_new
        steps += 1

        # Adapt learning rate
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            lr = min(lr * 1.02, 5.0 / n)
        else:
            lr = max(lr * 0.99, 0.01 / n)

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AGD step={steps} ({elapsed:.0f}s): T={T:.4e}, best={best_score:.8f}, lr={lr:.2e}")

        # Periodic reset to avoid divergence
        if steps % 5000 == 0:
            a = np.maximum(a + 0.01 * np.random.default_rng(steps).normal(0, a.mean(), n), 0)
            a = normalize_to(a, norm)
            y = a.copy()
            lr = min(lr * 2.0, 5.0 / n)

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_active=200):
    """
    Fessler-Parks accelerated LP with adaptive constraint reduction.
    """
    if not HAS_CVXPY:
        return a0, fast_autocorr(a0).max()

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.8

    # Pre-build full constraint matrix
    T_full = build_conv_matrix(y)

    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1

        c = fast_autocorr(y)
        M_curr = c.max()
        if M_curr < 1e-12:
            break

        # Rebuild T for current y
        T_full = build_conv_matrix(y)

        # Solve LP with all constraints
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_full @ b_var <= M_curr])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        if b_var.value is None:
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        s = b_val.sum()
        if s < 1e-15:
            continue

        # Constraint reduction: keep only active constraints
        circ = cross_convolve(y, b_val)
        for red_iter in range(4):
            tol = 1e-6 * M_curr + 1e-12
            active = circ >= M_curr - tol
            n_active = int(active.sum())

            if n_active < 5:
                break
            if n_active >= max_active + 100:
                # Keep top max_active active
                active_idx = np.where(active)[0]
                active_vals = circ[active_idx]
                top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
                active = np.zeros(2 * n - 1, dtype=bool)
                active[top_idx] = True
            else:
                # Re-solve with only active constraints
                active_idx = np.where(active)[0]
                if len(active_idx) < n:  # worthwhile reduction
                    T_sub = T_full[active_idx, :]
                    b_var2 = cp.Variable(n, nonneg=True)
                    prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_sub @ b_var2 <= M_curr])
                    try:
                        prob2.solve(solver=solver, verbose=False)
                    except Exception:
                        break
                    if prob2.status in ('optimal', 'optimal_inaccurate') and b_var2.value is not None:
                        b_new = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                        circ_new = cross_convolve(y, b_new)
                        if circ_new.max() <= M_curr * (1 + 1e-4):
                            b_val = b_new
                            circ = circ_new
                        else:
                            break
                    else:
                        break
                break

        s = b_val.sum()
        if s < 1e-15:
            continue

        g = b_val * norm / s
        n_iters += 1

        # Fessler-Parks update
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to(y_new, norm)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, norm)

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if null_check(stall):
            pass  # placeholder

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, gamma={gamma:.2f}, stall={stall}")

        # Adaptive restart
        if stall > 150:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = normalize_to(y, norm)
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = normalize_to(a, norm)
            prev_y = y.copy()
            stall = 0
            gamma = 1.8

        # Adaptive gamma based on progress
        if n_iters % 100 == 0:
            recent_improvement = (M_curr / max(best_score, 1e-12)) - 1.0
            if recent_improvement < 1e-5:
                gamma = max(1.2, gamma - 0.1)
            else:
                gamma = min(2.0, gamma + 0.05)

    return best_a, best_score


def null_check(x):
    """Trivial helper to avoid closure issues."""
    return False


def get_solver():
    """Find the best available LP solver."""
    if not HAS_CVXPY:
        return None
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): SCS")
    return cp.SCS


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def peak_shave_fast(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with multi-destination moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find contributing positions to this peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 100:
                break
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]

        # Find non-contributing positions (don't add to this peak)
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        non_contrib[src] = True
        dst_candidates = np.where(non_contrib)[0]

        if len(dst_candidates) == 0:
            dst_candidates = np.array([i for i in range(n) if i != src])

        # Pick from low-value non-contributing positions
        n_select = min(8, len(dst_candidates))
        sorted_dst = dst_candidates[np.argsort(a[dst_candidates])][:n_select]
        dst = sorted_dst[rng.integers(len(sorted_dst))]

        # Try several fractions
        best_trial_score = max_b
        best_frac = 0.0
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            amt = src_val * frac
            if amt < 1e-18:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = normalize_to(a_trial, norm)
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_frac = frac

        if best_frac > 0.0:
            a = a.copy()
            amt = src_val * best_frac
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0)
            a = normalize_to(a, norm)
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                n_stall += 1
        else:
            n_stall += 1

        # Escape: random multi-point perturbation
        if n_stall > 100:
            n_pert = max(5, n // 80)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-15:
                    amt = a_trial[i] * rng.uniform(0.05, 0.4)
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = normalize_to(a_trial, norm)
            sc = fast_autocorr(a_trial).max()
            if sc < best_score * 1.01:
                a = a_trial
            n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_peak_subgradient(a0, time_limit, verbose=True):
    """
    Subgradient method using weighted average of top peaks.
    Faster than full gradient when only a few peaks matter.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step_size = 0.05 / n

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        # Top K peaks
        K = min(8, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:K]
        peak_vals = b[top_peaks]

        # Softmax weighting with annealed temperature
        T_w = max(0.005, 0.05 * (1.0 - progress))
        w = np.exp((peak_vals - max_b) / T_w)
        w /= w.sum()

        # Compute gradient from weighted peaks
        grad = np.zeros(n)
        for m in range(K):
            if w[m] < 1e-6:
                continue
            p = top_peaks[m]
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            j_arr = np.arange(js, je + 1)
            grad[js:je + 1] += 2.0 * w[m] * a[p - j_arr]

        # Project to tangent space
        grad -= np.mean(grad)

        # Step
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to(a_new, norm)
        score_new = fast_autocorr(a_new).max()

        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            step_size = min(step_size * 1.03, 1.0 / n)
        else:
            step_size *= 0.95
            if step_size < 1e-8 / n:
                # Random restart
                rng = np.random.default_rng(steps)
                a = np.maximum(a + rng.normal(0, 0.02 * a.mean(), n), 0)
                a = normalize_to(a, norm)
                step_size = 0.05 / n

        steps += 1

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MSPG step={steps} ({elapsed:.0f}s): best={best_score:.8f}, step={step_size:.2e}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: hybrid smoothed GD + LP Nesterov + peak shaving.
    Explores multiple sequence lengths with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, cvxpy available: {HAS_CVXPY}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Smoothed gradient descent (fast, many iterations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smoothed GD n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 20:
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = smoothed_gradient_descent(a_norm, t1, verbose=True, gamma=1.8)
        do_try_update(a_out, f"sgd1_n{n_cur}")

    # === Phase 2: LP Nesterov (main refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30 and HAS_CVXPY:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, max_active=200)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 3: Multi-peak subgradient (different geometry) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: MSPG n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = multi_peak_subgradient(a_norm, t3, verbose=True)
        do_try_update(a_out, f"mspg_n{n_cur}")

    # === Phase 4: Fine smoothed GD (low temperature) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Fine GD n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        c0 = fast_autocorr(a_norm).max()
        a_out, _ = smoothed_gradient_descent(a_norm, t4, T_start=c0 * 0.02, T_end=c0 * 0.0005, verbose=True, gamma=1.5)
        do_try_update(a_out, f"sgd2_n{n_cur}")

    # === Phase 5: LP Nesterov round 2 (refined start) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and HAS_CVXPY:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, max_active=150)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 6: Try alternate length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_base + 128
    print(f"\n=== Phase 6: Explore n={n_alt} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 20:
        init_long = interpolate_sequence(best_seq, n_alt)
        a_norm = normalize_to(init_long, np.sqrt(2.0 * n_alt))
        if HAS_CVXPY:
            a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, max_active=150)
            do_try_update(a_out, f"lp_n{n_alt}")
        else:
            a_out, _ = smoothed_gradient_descent(a_norm, t6, verbose=True)
            do_try_update(a_out, f"sgd_n{n_alt}")

    # === Phase 7: Peak shaving (final polish) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.13, 180.0)
    if t7 > 10:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = peak_shave_fast(a_norm, t7, seed=31415, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
