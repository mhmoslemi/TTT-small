def lse_objective_and_grad(a, T):
    """Compute LSE-smoothed max autocorrelation and its gradient w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    b_max = b.max()
    if b_max < 1e-15:
        return 0.0, np.zeros(n)
    shifted = (b - b_max) / T
    exp_vals = np.exp(shifted)
    w_sum = exp_vals.sum()
    if w_sum < 1e-30:
        return b_max, np.zeros(n)
    w = exp_vals / w_sum
    lse_val = b_max + T * np.log(w_sum) / T  # = b_max + log(w_sum)
    # Actually: LSE = b_max + T*log(sum(exp((b-b_max)/T))) = b_max + T*log(w_sum)
    lse_val = b_max + T * np.log(w_sum)

    # Gradient: dLSE/da_i = T * sum_k w_k * dc_k/da_i / sum_k w_k
    # dc_k/da_i = 2*a_{k-i} (for valid k-i)
    # So dLSE/da_i = 2*T * sum_k w_k * a_{k-i} / w_sum
    # = 2*T * (cross_corr(w, a))_i / w_sum

    fft_size = 1 << (3 * n - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w

    A_f = rfft(a_pad, fft_size)
    W_f = rfft(w_pad, fft_size)
    cross = irfft(A_f * np.conj(W_f), fft_size)[:n]
    grad = 2.0 * T * cross / w_sum
    return lse_val, grad

        n_iters += 1

def lse_gradient_descent(a0, time_limit, T_start=50.0, T_end=2.0, n_stages=5):
    """LSE-smoothed gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    lr = 0.01
    momentum = 0.9
    velocity = np.zeros(n)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        stage_time = (time_limit - (time.monotonic() - t0)) / max(1, n_stages - si)
        stage_end = time.monotonic() + stage_time
        iters_in_stage = 0

        while time.monotonic() < stage_end:
            lse_val, grad = lse_objective_and_grad(a, T)
            s = a.sum()
            if s < 1e-10:
                break

            # Full gradient of objective = 2*n*lse/s^2
            obj_grad = 2.0 * n * grad / (s * s) - 4.0 * n * lse_val * a / (s * s * s)

            # Normalized gradient (since scaling doesn't matter)
            grad_norm = np.linalg.norm(obj_grad)
            if grad_norm < 1e-15:
                break

            # Adam-like update
            velocity = momentum * velocity - lr * obj_grad / grad_norm
            a_new = a + velocity
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new

            sc = fast_autocorr(a_new).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
            iters_in_stage += 1

            # Decay learning rate
            if iters_in_stage % 50 == 0:
                lr *= 0.9

        if verbose_flag:
            print(f"    LSE stage {si+1}/{n_stages} T={T:.1f}: best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=42):
    """Targeted coordinate descent: move mass from high-autocorrelation contributors."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:16]
        improved = False

        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_src = idx[np.argsort(contrib)[::-1][:8]]

            # Find safe targets (away from all top peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            safe_dst = np.where(~peak_mask)[0]
            if len(safe_dst) < n // 4:
                safe_dst = np.where(np.arange(n) != 0)[0]
            safe_dst_sorted = safe_dst[np.argsort(-a[safe_dst])][:max(2, len(safe_dst) // 6)]

            for src in top_src:
                if a[src] < 1e-12:
                    continue
                for dst in safe_dst_sorted[:6]:
                    if dst == src:
                        continue
                    for frac in [0.003, 0.008, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        trial = a.copy()
                        trial[src] -= amt
                        trial[dst] += amt
                        if trial[src] < -1e-14:
                            continue
                        if try_update(trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 10 == 0:
                # Random restarts
                for _ in range(50):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.12)
                    amt = a[i] * frac
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    if try_update(trial):
                        break

            if stall % 50 == 0:
                # Perturbation
                n_pert = max(5, n // 80)
                indices = rng.choice(n, size=n_pert, replace=False)
                trial = a.copy()
                for idx in indices:
                    trial[idx] *= rng.uniform(0.5, 1.5)
                try_update(trial)

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 10))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.5) ** 2) + 0.3
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.25 * np.sin(6.0 * np.pi * x) + 0.1 * np.sin(12.0 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.abs(np.sin(3.0 * np.pi * x)) ** 0.7
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * x) + 0.8 * np.exp(-3.0 * (1.0 - x))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    n_parent = len(best_seq) if best_seq is not None else 1600
    scan_lengths = sorted(set([n_parent, max(1000, n_parent - 200), n_parent + 300, n_parent + 600]))
    scan_results = {}

    for i_scan, n_scan in enumerate(scan_lengths):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget * 0.10:
            break
        remaining_scan = time_budget * 0.10 - elapsed
        if remaining_scan < 20:
            break
        scan_time = min(remaining_scan / max(1, len(scan_lengths) - i_scan), 80.0)
        if scan_time < 15:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Main deep LP-Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Deep LP n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.38
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_deep_n{n_best}")

    # Phase 2: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE grad n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t2, T_start=60.0, T_end=3.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 3: LP at longer length
    n_long = n_best + 500
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP n={n_long} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 40:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # Phase 4: Noisy restart LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Noisy LP n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_noisy = best_seq + rng.normal(0, 0.015 * best_seq.mean(), n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = nesterov_lp_optimize(a_noisy, t4, solver, verbose=False)
        do_try_update(a_out, f"lp_noisy_n{n_best}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t5, seed=99999)
        do_try_update(a_out, "coord1")

    # Phase 6: LSE polish (low temp)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t6, T_start=15.0, T_end=1.0, n_stages=4)
        do_try_update(a_out, "lse_polish")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.6
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t7, seed=31337)
        do_try_update(a_out, "coord_final")

    # Phase 8: One more short LP if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, remaining * 0.8, solver, verbose=False)
        do_try_update(a_out, "lp_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
