        if M < 1e-14:
            break
        
        # Find active lags (top few)
        top_k = min(20, len(c))
        active = np.argsort(c)[::-1][:top_k]
        
        b_var = cp.Variable(n, nonneg=True)
        
        # Build constraint matrix T: T[k,j] = y[k-j] if valid
        # For efficiency, only use active constraints
        T_active = np.zeros((len(active), n))
        for idx, k in enumerate(active):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                T_active[idx, j] = y[k_int - j]
        
        constraints = [T_active @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
