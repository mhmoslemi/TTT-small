    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def nesterov_lp_optimize_v2(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated FW with improved per-constraint LP bounds
    and backtracking line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Solve improved LP
        T = build_T(y)
        rhs = (M_curr + ac) / 2.0
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Check both y and x
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPv2 t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def adaptive_fw_optimize(a0, time_limit, solver, verbose=True):
    """
    FW with adaptive step size (backtracking line search) 
    and per-constraint LP bounds.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # LP for direction
        T = build_T(a)
        rhs = (M_curr + ac) / 2.0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        # Backtracking line search
        best_alpha = 1.0 / (t + 1.0)
        best_obj = M_curr
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 1.0/(t+1), 0.5/(t+1)]:
            a_trial = (1.0 - alpha) * a + alpha * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AFW t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def mirror_descent_optimize(a0, time_limit, top_k=20, beta=4.0, verbose=True):
    """Mirror descent with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    gamma = 0.5
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = gamma / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(a)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]
        
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_seq(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_mask = u * idx > (cssv - 1)
    if not rho_mask.any():
        return np.ones(n) / n
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_T(y):
    """Build convolution matrix: T[k,j] = y[k-j] if 0<=k-j<n, else 0."""
    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T


def nesterov_lp_v2(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated FW with per-constraint linearized LP bounds.
    Key improvement: T_k(b) <= (M + ac_k)/2 instead of T_k(b) <= M.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build T from y
        T = build_T(y)
        # Per-constraint bounds (tighter than uniform M)
        rhs = (M_curr + ac) / 2.0

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPv2 t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def grad_proj_optimize(a0, time_limit, verbose=True):
    """
    Gradient descent with smoothing + simplex projection.
    Fast per-iteration, good for exploring different basins.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adaptive temperature for smoothing
    tau0 = best_m * 0.3
    lr0 = 0.1

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Smoothed subgradient
        tau = max(tau0 * (0.995 ** t), 1e-10)
        shifted = ac - ac.max()
        w = np.exp(shifted / tau)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(ac)) / len(ac)
        else:
            w = w / w_sum

        # Compute gradient: grad_i = sum_k w_k * 2 * a[k-i] (if valid)
        grad = np.zeros(n)
        # For efficiency, use FFT-like approach or direct computation
        # Direct: for each k, add 2*w_k*a[k-i] to grad[i] for valid i
        # This is: grad = 2 * (w convolved with a reflected) essentially
        # Let's do it via FFT
        # w has length 2n-1, a has length n
        # We want grad[i] = sum_k w[k] * 2*a[k-i] for 0 <= k-i < n
        # = 2 * sum_k w[k] * a[k-i] where a[j]=0 for j outside [0,n-1]
        # This is a cross-correlation of w and a
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # Cross-correlation: ifft(conj(W) * A) gives sum_k w[k] * a[k+i]... 
        # Let's just do it directly for clarity (n is at most ~4000)
        # Actually for n=1600 and many iterations, let's use the FFT trick properly
        # grad[i] = 2 * sum_{k=0}^{2n-2} w[k] * a[k-i] (with a[j]=0 outside [0,n-1])
        # = 2 * (w * a_rev)[n-1-i] where a_rev is reversed a... 
        # Simpler: just loop over top w's
        top_m = min(50, len(ac))
        top_idx = np.argsort(w)[::-1][:top_m]
        for k in top_idx:
            k_int = int(k)
            if w[k] < 1e-15:
                continue
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * w[k_int] * a[j]

        # Gradient step with adaptive learning rate
        lr = lr0 / (1.0 + 0.01 * t)
        a_new = a - lr * grad
        # Project onto simplex
        a_new = project_simplex(a_new)

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve learning rate if we didn't improve
            lr = lr * 0.5
            a_new = a - lr * grad
            a_new = project_simplex(a_new)
            a = a_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GP t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def mirror_desc_optimize(a0, time_limit, top_k=25, beta=5.0, verbose=True):
    """Mirror descent with soft subgradient (exponential weights on top peaks)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    gamma = 0.5
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        step = gamma / (np.sqrt(t) + 1.0)

        ac = fast_autocorr(a)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coord_polish(a0, time_limit, verbose=False):
    """
    Coordinate polish: move mass from high-contributing positions to low ones.
    Uses peak sensitivity analysis to choose destinations.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        improved = False

        # Find top contributing lags
        top_k = np.argsort(ac)[::-1][:min(10, len(ac))]

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination using peak sensitivity
            peak_indices = top_k
            n_candidates = min(n, 150)
            candidates = rng.choice(n, size=n_candidates, replace=False)

            best_dst = -1
            best_score = float('inf')

            for dst in candidates:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in peak_indices[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst

            if best_dst < 0:
                continue

            for frac in [0.003, 0.008, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random kick
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = norm_sum1(a_out)
            print(f"  *** NEW BEST [lpv2_n2400]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 6: MD at current best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: MD n={n_cur} ===")
    t6 = min(remaining * 0.07, 70.0)
    if t6 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_desc_optimize(a_in, t6, top_k=30, beta=6.0, verbose=False)
        sc = eval_seq(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = norm_sum1(a_out)
            print(f"  *** NEW BEST [md2]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 7: Final LP polish (if length allows) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_cur} ===")
    if n_cur <= 2000 and remaining > 30:
        t7 = min(remaining * 0.25, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_lp_v2(a_in, t7, solver, verbose=True)
            sc = eval_seq(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = norm_sum1(a_out)
                print(f"  *** NEW BEST [lp_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final Coord n={n_cur} ===")
    if remaining > 15:
        t8 = min(remaining * 0.35, 120.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_polish(a_in, t8, verbose=True)
            sc = eval_seq(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = norm_sum1(a_out)
                print(f"  *** NEW BEST [coord_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 9: Alternative initialization + quick polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_cur = len(best_seq)
        print(f"\n=== Phase 9: Alt init n={n_cur} ===")
        # Raised cosine blend
        x = np.linspace(0, 1, n_cur)
        alt = (1 + np.cos(2 * np.pi * x)) / 2.0
        alt = np.maximum(alt, 0)
        alt = alt / alt.sum()
        blend = 0.4 * norm_sum1(best_seq) + 0.6 * alt
        blend = norm_sum1(blend)

        t9 = min(remaining * 0.3, 50.0)
        if t9 > 10:
            a_out, _ = mirror_desc_optimize(blend, t9, top_k=25, beta=5.0, verbose=False)
            sc = eval_seq(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = norm_sum1(a_out)
                print(f"  *** NEW BEST [alt]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s}")

    # Final validation
    if best_seq is not None:
        fs = eval_seq(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
