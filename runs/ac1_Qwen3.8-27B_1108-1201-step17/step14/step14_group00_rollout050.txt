```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_LP_SOLVER = None


def _init_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = s
                print(f"  LP solver: {s}")
                return _LP_SOLVER
        except Exception:
            continue
    _LP_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _LP_SOLVER


def _acorr(a):
    n = len(a)
    sz = 2 * n - 1
    N = 1 << (sz - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:sz]


def _obj(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * _acorr(a).max() / (s * s)


def _ev(a):
    try:
        r = evaluate_sequence([float(x) for x in a])
        return float(r)
    except Exception:
        return float('inf')


def _conv_yb(y, b):
    n = len(y)
    sz = 2 * n - 1
    N = 1 << (sz - 1).bit_length()
    Y = rfft(y, N)
    B = rfft(b, N)
    return irfft(Y * B, N)[:sz]


def _build_T(y, ks):
    n = len(y)
    m = len(ks)
    T = np.zeros((m, n))
    for ii in range(m):
        k = ks[ii]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            js = np.arange(lo, hi)
            T[ii, js] = y[k - js]
    return T


def _lp_br(y, M, solver, ks):
    n = len(y)
    b = cp.Variable(n, nonneg=True)
    T = _build_T(y, ks)
    cons = [T @ b <= M]
    p = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    p.solve(solver=solver, verbose=False)
    if p.status not in ('optimal', 'optimal_inaccurate'):
        return None, ks
    bv = np.maximum(np.asarray(b.value, dtype=float), 0.0)
    # Check violations via FFT
    cv = _conv_yb(y, bv)
    viol = np.where(cv > M * (1.0 + 1e-9))[0]
    if len(viol) > 0:
        ks2 = np.unique(np.concatenate([ks, viol]))
        if len(ks2) > 0.85 * (2 * n - 1):
            ks2 = np.arange(2 * n - 1)
        T2 = _build_T(y, ks2)
        cons2 = [T2 @ b <= M]
        p2 = cp.Problem(cp.Maximize(cp.sum(b)), cons2)
        p2.solve(solver=solver, verbose=False)
        if p2.status in ('optimal', 'optimal_inaccurate'):
            bv = np.maximum(np.asarray(b.value, dtype=float), 0.0)
            ks = ks2
    return bv, ks


def _norm1(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _fw_step(a, solver, t0, budget, verbose=True):
    """Fast subgradient Frank-Wolfe (normalized to sum=1)."""
    n = len(a)
    a = _norm1(np.maximum(a, 0.0))
    best = a.copy()
    best_s = _obj(a)
    y = a.copy()
    x = a.copy()
    clock = time.monotonic()
    
    for t in range(1, 10000000):
        if time.monotonic() - clock > budget:
            break
        ca = _acorr(y)
        k = int(np.argmax(ca))
        M = ca[k]
        # Subgradient: grad[j] = y[(k-j) % valid] + y[(j-k) % valid]
        # For linear conv: grad[j] = y[k-j] (if 0<=k-j<n) + y[j-k] (if 0<=j-k<n)
        grad = np.zeros(n)
        # Forward: (a*a)_k = sum_i a_i * a_{i+k}, d/d a_j = a_{j+k} (if j+k < n)
        idx1 = k + np.arange(n)
        m1 = idx1 < n
        grad[m1] += y[idx1[m1]]
        # Backward: d/d a_j = a_{j-k} (if j-k >= 0)
        idx2 = np.arange(n) - k
        m2 = idx2 >= 0
        grad[m2] += y[idx2[m2]]
        
        jmin = int(np.argmin(grad))
        v = np.zeros(n)
        v[jmin] = 1.0
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        if t == 1:
            x_new = v
        else:
            mom = (t - 1.0) / (t + 2.0)
            x_new = y_new + mom * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
        
        y = _norm1(y_new)
        x = _norm1(x_new)
        
        for cand in [y, x]:
            sc = _obj(cand)
            if sc < best_s:
                best_s = sc
                best = cand.copy()
    
    if verbose:
        print(f"  FW done: best={best_s:.8f}, n={n}")
    return best, best_s


def _nesterov_lp(a, time_budget, solver, verbose=True, seed=0):

    rng = np.random.default_rng(seed)
    n = len(a)
    a = _norm1(np.maximum(a, 0.0))
    best = a.copy()
    best_s = _obj(a)
    y = a.copy()
    x = a.copy()
    avg = np.zeros(n)
    navg = 0
    clock = time.monotonic()
    active = np.array([0])
    stall = 0
    stall_thresh = 80
    
    for t in range(1, 500000):
        if time.monotonic() - clock > time_budget:
            break
        
        ca = _acorr(y)
        M = ca.max()
        if M < 1e-15:
            break
        
        # Screen active set periodically
        if t % 50 == 0 or t == 1:
            tol = 0.05 + 0.03 * np.exp(-t / 300.0)
            active = np.where(ca > M * (1.0 - tol))[0]
            if len(active) < 3:
                active = np.where(ca > M * 0.85)[0]
        
        bv, active = _lp_br(y, M, solver, active)
        if bv is None:
            active = np.array([0])
            bv, active = _lp_br(y, M, solver, active)
            if bv is None:
                continue
        
        s_b = bv.sum()
        if s_b < 1e-15:
            continue
        g = bv / s_b  # normalize to sum=1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        if t == 1:
            x_new = g.copy()
        else:
            mom = min((t - 1.0) / (t + 2.0), 0.98)
            x_new = y_new + mom * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
        
        y = _norm1(y_new)
        x = _norm1(x_new)

            sc = _obj(cand)
            if sc < best_s:
                best_s = sc
                best = cand.copy()
                stall = 0
            else:
                stall += 1
        
        avg += x
        navg += 1
        if navg % 50 == 0:
            av = _norm1(avg / navg)
            sc = _obj(av)
            if sc < best_s:
                best_s = sc
                best = av.copy()
                stall = 0
        
        if stall > stall_thresh:
            y = best.copy()
            x = best.copy()
            avg = np.zeros(n)
            navg = 0
            active = np.array([0])
            stall = 0
        
        if verbose and t % 200 == 0:
            print(f"    NL t={t} ({time.monotonic()-clock:.0f}s): best={best_s:.10f} | active={len(active)}")
    
    return best, best_s


def _pair_cd(a, time_budget, rng, verbose=True):
    """Greedy pair coordinate descent (sum=1 normalization)."""
    n = len(a)
    a = _norm1(np.maximum(a, 0.0))
    best = a.copy()
    best_s = _obj(a)
    clk = time.monotonic()
    sz = 2 * n - 1
    no_imp = 0
    
    while time.monotonic() - clk < time_budget:
        improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - clk > time_budget:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue
            
            # Try moving mass from i to a random j
            for _ in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue
                aj = a[j]
                
                # Try several step sizes
                best_d = 0.0
                best_val = best_s
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9, 1.0]:
                    d = ai * frac
                    a2 = a.copy()
                    a2[i] -= d
                    a2[j] += d
                    if a2.sum() < 1e-15:
                        continue
                    a2 = _norm1(a2)
                    sc = _obj(a2)
                    if sc < best_val - 1e-14:
                        best_val = sc
                        best_d = d
                
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = _norm1(a)
                    if best_val < best_s:
                        best_s = best_val
                        best = a.copy()
                        improved = True
                    no_imp = 0
        
        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break
        elif verbose:
            pass  # skip per-pass message for speed
    
    if verbose:
        print(f"  PCD done: best={best_s:.8f}")
    return best, best_s


def _interp(f, new_n):
    on = len(f)
    if on == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0, 1, on)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    return _norm1(fi)


def _perturb(a, rng, scale=0.05):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    return _norm1(np.maximum(a, 0) * noise)
```

```python
def _targeted_cd(a, time_budget, rng):
    n = len(a)
    a = _norm1(np.maximum(a, 0))
    best = a.copy()
    best_s = _obj(a)
    clk = time.monotonic()
    
    while time.monotonic() - clk < time_budget:
        ca = _acorr(a)
        k = int(np.argmax(ca))
        
        # Find coordinates contributing to the peak
        # (a*a)_k = sum_i a_i * a_{i+k}
        # The main contributors are where a_i * a_{i+k} is large
        contrib = np.zeros(n)
        for i in range(min(n, n-k)):
            contrib[i] = a[i] * a[i+k]
        
        # Find the coordinate with the largest contribution
        i_big = int(np.argmax(contrib))
        
        # Try moving mass from i_big to several random positions
        best_d = 0.0
        best_val = best_s
        n_tries = 20
        for _ in range(n_tries):
            j = int(rng.integers(n))
            if j == i_big:
                continue
            for frac in [0.1, 0.3, 0.5, 0.7]:
                d = a[i_big] * frac
                a2 = a.copy()
                a2[i_big] -= d
                a2[j] += d
                if a2.min() < -1e-15:
                    continue
                a2 = _norm1(a2)
                sc = _obj(a2)
                if sc < best_val:
                    best_val = sc
                    best_d = d
                    best_j = j
        
        if best_d > 0:
            a[i_big] -= best_d
            a[best_j] += best_d
            a = _norm1(a)
            best_s = best_val
            best = a.copy()
        else:
            # Don't improve by targeted move, try random
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                d = a[i] * 0.3
                a2 = a.copy()
                a2[i] -= d
                a2[j] += d
                a2 = _norm1(a2)
                sc = _obj(a2)
                if sc < best_s:
                    a = a2
                    best_s = sc
                    best = a.copy()
            else:
                break  # No improvement possible
    
    return best, best_s
```

```python
def _param_search(a_init, time_budget, rng):
    """Parametric search: represent a as sum of Gaussians + DC."""
    n = len(a_init)
    a0 = _norm1(np.maximum(a_init, 0))
    x = np.linspace(0, 1, n)
    
    # Initialize with 4 Gaussians + DC
    n_gauss = 4
    params = np.zeros(3 * n_gauss + 1)  # (center, width, amplitude) * n_gauss + DC
    for i in range(n_gauss):
        params[3*i] = 0.1 + 0.8 * i / (n_gauss - 1)  # centers
        params[3*i+1] = 0.05 + 0.1 * rng.random()  # widths
        params[3*i+2] = 0.5 + 0.5 * rng.random()  # amplitudes
    params[-1] = 0.3  # DC
    
    def model(params):
        a = np.zeros(n)
        for i in range(n_gauss):
            c = params[3*i]
            s = max(params[3*i+1], 0.01)
            amp = max(params[3*i+2], 0.0)
            a += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
        a += max(params[-1], 0.0)
        return _norm1(a)
    
    def objective(params):
        a = model(params)
        return _obj(a)
    
    # Use scipy.optimize
    from scipy.optimize import minimize
    best_params = params.copy()
    best_val = objective(params)
    clk = time.monotonic()
    
    while time.monotonic() - clk < time_budget:
        res = minimize(objective, best_params, method='Nelder-Mead', 
                      options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
        if res.fun < best_val:
            best_val = res.fun
            best_params = res.x
        else:
            # Perturb and try again
            best_params = best_params + 0.05 * rng.standard_normal(len(best_params))
    
    return model(best_params), best_val
```
