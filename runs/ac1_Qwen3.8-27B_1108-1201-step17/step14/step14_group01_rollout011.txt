def fast_conv_self(a):
    n = len(a)
    if n <= 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 12
    temps = np.geomspace(80.0, 0.15, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 6)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]

            grad = 2.0 * (g1 + g2)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.05 * it))
            for trial in range(8):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = fast_conv_self(a_new).max()
                if sc <= max_c + 0.02 * T_val:
                    break
                step *= 0.5

            a = a - step * grad
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = best_a.copy()
            else:
                a = a * norm / s_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 2 == 0:
            print(f"    LSE stage {si}/{n_stages} T={T_val:.2f}: best={best_score:.8f}", flush=True)

    return best_a, best_score

def lp_solve(a, M, solver, active, warm_start=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)

    if warm_start is not None:
        b_var.value = warm_start

    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    prev_b = None
    patience = 0
    max_patience = 50
    momentum = 0.9

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        if t < 5:
            active = None
        else:
            frac = max(0.3, 0.8 - 0.01 * np.log(1 + t / 10.0))
            threshold = M * frac
            active = np.where(conv_y > threshold)[0]
            if len(active) < 10:
                active = np.where(conv_y > M * 0.3)[0]
            if len(active) == 0:
                active = None

        b = lp_solve(y, M, solver, active, warm_start=prev_b)
        if b is None:
            b = lp_solve(y, M, solver, None, warm_start=prev_b)
        if b is None:
            patience += 1
            if patience > max_patience:
                break
            continue
        patience = 0

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        # Check if dropped constraints are violated
        if active is not None and len(active) < size:
            check_conv = fast_conv_cross(y, b)
            violated = np.where(check_conv > M * (1 + 0.01))[0]
            if len(violated) > 0:
                b_full = lp_solve(y, M, solver, None, warm_start=b)
                if b_full is not None:
                    b = b_full
                    b_sum = b.sum()
                    if b_sum < 1e-15:
                        continue

        g = b * norm / b_sum

        # Convergence check
        if prev_b is not None:
            b_diff = np.linalg.norm(g - prev_b * norm / prev_b.sum()) / (norm + 1e-15)
            if b_diff < 1e-6:
                print(f"    LP converged at t={t}, diff={b_diff:.2e}", flush=True)
                break

        prev_b = b.copy()

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        momentum = 0.95 * momentum + 0.05 * (2.0 / (np.sqrt(t) + 1.0))
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f} M={M:.8f}", flush=True)

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            conv = fast_conv_self(a)
            cur_max = conv.max()

            # Select j candidates: random + structured
            j_candidates = set()
            n_rand = max(6, n // 200)
            for _ in range(n_rand):
                j_candidates.add(int(rng.integers(n)))

            peak_lags = np.where(conv >= cur_max * 0.92)[0]
            for kl in peak_lags[:8]:
                j_struct = (kl - i) % n
                j_candidates.add(j_struct)
                j_candidates.add((kl + i) % n)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta using ternary-like search
                n_samples = 20
                deltas = np.linspace(0.0, ai * 0.95, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    if d < 1e-15:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-13:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve >= 3:
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}", flush=True)

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        # Asymmetric: sharp rise, slow decay
        a = np.where(x < 0.2, x / 0.2, np.exp(-4.0 * (x - 0.2)))
    else:
        # Two asymmetric peaks
        a = np.exp(-12.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.75) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, starting at t=0", flush=True)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}", flush=True)
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)", flush=True)

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}", flush=True)
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===", flush=True)
    if t1 > 30:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.38, 550.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===", flush=True)
    if t2 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n2}")

    # Phase 3: Pair CD on current best
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 3: Pair CD n={n3} (budget={t3:.0f}s) ===", flush=True)
    if t3 > 30:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{n3}")

    # Phase 4: Second LP round from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.20, 280.0)
    print(f"\n=== Phase 4: LP round 2 n={n4} (budget={t4:.0f}s) ===", flush=True)
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=False)
        try_update(a_out, f"lp2_n{n4}")

    # Phase 5: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 300
    t5 = min(remaining * 0.25, 200.0)
    print(f"\n=== Phase 5: Longer n={n_long} (budget={t5:.0f}s) ===", flush=True)
    if t5 > 50 and remaining > 80:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, min(t5 * 0.6, 80.0), solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out2, _ = pair_cd_optimize(best_seq if len(best_seq) == n_base else init_long,
                                         min(remaining * 0.5, 50.0), rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 6: Try shorter length (sometimes shorter is better due to 2n factor)
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(800, n_base - 400)
    t6 = min(remaining * 0.20, 120.0)
    print(f"\n=== Phase 6: Shorter n={n_short} (budget={t6:.0f}s) ===", flush=True)
    if t6 > 30 and remaining > 50:
        init_short = interpolate_seq(best_seq, n_short)
        a_out, _ = lse_gradient_descent(init_short, min(t6 * 0.4, 30.0), verbose=False)
        try_update(a_out, f"lse_short_{n_short}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = nesterov_lp_optimize(init_short, min(remaining * 0.5, 50.0), solver, verbose=False)
            try_update(a_out2, f"lp_short_{n_short}")

    # Phase 7: Final LP + CD alternation
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_final} (budget={remaining:.0f}s) ===", flush=True)
    if remaining > 40:
        t_lp = remaining * 0.5
        a_out, _ = nesterov_lp_optimize(best_seq, min(t_lp, 100.0), solver, verbose=False)
        try_update(a_out, f"lp_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 80.0), rng, verbose=False)
            try_update(a_out, f"pcd_final")

    # Phase 8: One more LSE pass with very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_out, _ = lse_gradient_descent(best_seq, min(remaining * 0.6, 40.0), verbose=False)
        try_update(a_out, "lse_final2")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s", flush=True)
    else:
        print("\nERROR: No valid sequence found!", flush=True)
        return [1.0]

    return [float(x) for x in best_seq]

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
