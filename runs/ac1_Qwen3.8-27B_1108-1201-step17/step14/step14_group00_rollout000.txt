        lp_start = time.monotonic()
        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break

        b, _, duals = lp_solve_full(y, M_curr, solver, warm_b)
        lp_elapsed = time.monotonic() - lp_start
        lp_times.append(lp_elapsed)

        if b is None:
            warm_b = None
            b, _, duals = lp_solve_full(y, M_curr, solver, None)
            if b is None:
                continue

        warm_b = b.copy()
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b * norm / b_sum
        n_avg += 1

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            a_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.97)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_conv(y).max()
        if score_y < best_score - 1e-15:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_conv(a).max()
        if score_a < best_score - 1e-15:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average (checked periodically)
        avg_a += a
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_conv(avg_check).max()
                if score_avg < best_score - 1e-15:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0

        # Stall detection with restart
        restart_threshold = 80
        if stall_count > restart_threshold:
            # Reset to best and try different momentum
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            stall_count = 0
            warm_b = None
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      Restart at t={t} ({elapsed:.0f}s), best={best_score:.10f}")

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-20:]) if len(lp_times) >= 20 else np.mean(lp_times)
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.10f} (LP avg={avg_lp*1000:.0f}ms)")

    return best_a, best_score, init_score


def coordinate_descent_polish(a0, time_limit, rng, verbose=True, n_sweeps=100):
    """Individual coordinate descent: optimize each a_i one at a time."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, fast_conv(a).max()
    a = a * norm / s0

    conv = fast_conv(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    prev_score = best_score

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        perm = rng.permutation(n)
        sweep_improved = False

        for idx in perm:
            if time.monotonic() - t0 > time_limit:
                    break
            i = idx
            ai = a[i]
            if ai < norm * 1e-10:
                # Try adding a small amount
                candidates = [norm * 0.001, norm * 0.01]
            else:
                # Try removing some amount
                candidates = [0.0, ai * 0.1, ai * 0.3, ai * 0.5, ai * 0.7, ai * 0.9]

            best_delta = 0.0
            best_local_score = conv.max()

            for delta in candidates:
                new_ai = max(0.0, ai - delta) if ai > 0 else delta
                # Compute change in convolution
                # Removing (ai - new_ai) from position i
                removed = ai - new_ai
                if abs(removed) < 1e-15:
                    new_conv_max = conv.max()
                else:
                    # conv change: -removed * (a shifted by i) in two directions
                    # For position i in the sequence, conv_k changes by:
                    # -removed * a[k-i] for the "forward" part
                    # -removed * a[n-1-(k-i)] for the "backward" part (if we think of it as correlation)
                    # Actually for full convolution: (a*a)_k = sum_j a_j * a[k-j]
                    # If we change a_i by delta, conv_k changes by delta * a[k-i] + delta * a[k-i] 
                    # (since a appears on both sides) ... no, that's not right either.
                    # Let's just compute directly for the affected range.
                    
                    # The change in conv when a_i changes by d (a_i -> a_i + d) is:
                    # dconv_k = d * a[k-i] + d * a[k-i] ... no
                    # Actually: conv_k = sum_j a_j * a_{k-j}
                    # d(conv_k)/d(a_i) = a_{k-i} (from the first a) + a_{k-i} (from the second a)
                    # Wait: conv_k = sum_j a_j * a_{k-j}
                    # If we differentiate w.r.t. a_i:
                    # The term a_i appears as a_j when j=i: gives a_{k-i}
                    # The term a_i appears as a_{k-j} when k-j=i, i.e., j=k-i: gives a_i... no
                    # j=k-i: term is a_{k-i} * a_i, derivative w.r.t. a_i is a_{k-i}
                    # So d(conv_k)/d(a_i) = a_{k-i} (if 0<=k-i<n) + a_{k-i} (if 0<=k-i<n)
                    # Wait that's the same thing counted twice? No:
                    # conv_k = sum_j a_j * a_{k-j}
                    # Let u = j, v = k-j. Then conv_k = sum_{u+v=k} a_u * a_v
                    # d/d(a_i) conv_k = sum_{u+v=k} [delta_{u,i} * a_v + a_u * delta_{v,i}]
                    #               = a_{k-i} (if 0<=k-i<n) + a_{k-i} (if 0<=k-i<n)
                    #               = 2 * a_{k-i} (if 0<=k-i<n)
                    # So the gradient is 2*a_{k-i} for valid k-i.
                    
                    # For small changes, new_conv_k ~= conv_k + removed * 2 * a[k-i]
                    # But for large changes, we need to include the second-order term: removed^2 at k=2i
                    
                    affected_lo = max(0, i)
                    affected_hi = min(size, 2 * i + 1)
                    
                    # Get the relevant slices of a
                    # k ranges from i to 2*i (for the case where k-i is in [0,n-1])
                    # Actually k-i is valid when 0 <= k-i <= n-1, i.e., i <= k <= i+n-1
                    lo = max(0, i)
                    hi = min(size, i + n)
                    
                    if hi > lo:
                        valid_idx = np.arange(lo, hi)
                        a_vals = a[valid_idx - i]  # a[k-i] for k in [lo, hi)
                        new_conv_seg = conv[lo:hi] + 2.0 * removed * a_vals
                        # Add second-order term at k=2i
                        if 2 * i < size:
                            new_conv_seg[2 * i - lo] += removed * removed
                        new_conv_max = max(new_conv_seg.max(), 
                                           np.delete(conv, np.arange(lo, hi)).max() if hi - lo < size else -np.inf)
                        # Simpler: just take max of new segment and old unaffected part
                        if hi - lo < size:
                            unaffected = np.concatenate([conv[:lo], conv[hi:]])
                            new_conv_max = max(new_conv_seg.max(), unaffected.max())
                        else:
                            new_conv_max = new_conv_seg.max()
                
                if new_conv_max < best_local_score - 1e-14:
                    best_local_score = new_conv_max
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(0.0, ai - best_delta) if (ai > 0 and best_delta > 0) else ai + best_delta
                s = a.sum()
                if s < 1e-15:
                    a[i] = ai  # revert
                    continue
                a = a * norm / s
                conv = fast_conv(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            # Try random pair swaps as a perturbation
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                i, j = rng.choice(n, size=2, replace=False)
                d = a[i] * rng.uniform(0.05, 0.3)
                if d > a[j] * 0.1 or d < 1e-10:
                    continue
                a[i] -= d
                a[j] += d
                s = a.sum()
                if s < 1e-15:
                    a[i] += d
                    a[j] -= d
                    continue
                a = a * norm / s
                conv = fast_conv(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    sweep_improved = True
                else:
                    a[j] -= d
                    a[i] += d
                    a = a * norm / a.sum()
                    conv = fast_conv(a)
            if not sweep_improved:
                break

        if verbose and (sweep + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD sweep={sweep+1} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, method=0):
    """Generate structured initial sequences with various methods."""
    if method == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif method == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif method == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif method == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif method == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif method == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif method == 6:
        # Symmetric bump
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20.0 * (x - 0.5) ** 2)
    elif method == 7:
        # Cosine taper
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif method == 8:
        # Flat-top with taper
        x = np.linspace(0.0, 1.0, n)
        a = np.where(np.abs(x - 0.5) < 0.3, 1.0, 0.3 + 0.7 * np.exp(-50*(np.abs(x-0.5)-0.3)**2))
    elif method == 9:
        # Optimal low-PAPR inspired
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) - 0.2 * np.cos(2 * np.pi * 7 * x)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
