import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_max(b, temperature):
    """Compute LSE approximation of max."""
    m = b.max()
    if m < 0:
        m = 0
    return m + np.log(np.sum(np.exp(temperature * (b - m)))) / temperature

def lse_max_grad(b, temperature):
    """Compute LSE and its gradient (weights)."""
    m = b.max()
    exp_vals = np.exp(temperature * (b - m))
    w = exp_vals / exp_vals.sum()
    lse = m + np.log(exp_vals.sum()) / temperature
    return lse, w

    # = sum_i a[i] * w[i+j]

def lse_objective_with_grad(a, temperature):
    """Compute LSE objective (scaled) and its gradient."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    lse_val, w = lse_max_grad(b, temperature)
    
    # Objective: 2*n*lse / S^2
    obj = 2.0 * n * lse_val / (S * S)
    
    # Gradient:
    # d(obj)/da[j] = 2*n/(S^2) * d(lse)/da[j] - 2*obj*a[j]/S
    # d(lse)/da[j] = sum_k w[k] * 2*a[k-j] (as derived above)
    
    fft_size = 1 << (3 * n - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2*n-1] = w
    
    A_fft = rfft(a_pad, fft_size)
    W_fft = rfft(w_pad, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    
    conv_grad = 2.0 * cross
    grad = 2.0 * n * conv_grad / (S * S) - 2.0 * obj * a / S
    
    return obj, grad

def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8, verbose=True):
    """
    Adam-based optimization of LSE-smoothed objective with decreasing temperature.
    """
    n = len(a0)
    S_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S_target / n
    else:
        a = a * S_target / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        lr = 0.01 * (T_start / T)  # Larger lr at higher temperature
        t_adam = 0
        
        stage_start = time.monotonic()
        max_stage_time = min(remaining / max(1, n_stages - si), 180.0)
        
        while time.monotonic() - stage_start < max_stage_time:
            t_adam += 1
            
            obj, grad = lse_objective_with_grad(a, T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t_adam)
            v_hat = v / (1 - beta2**t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project: clamp and rescale
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()  # Revert
            else:
                a_new = a_new * S_target / s_new
            
            # Check improvement
            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            a = a_new
            
            if verbose and t_adam % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      Stage {si+1}/{n_stages} (T={T:.1f}), iter {t_adam}: score={score:.8f} (elapsed={elapsed:.0f}s)")
    
    return best_a, best_score

def subgradient_momentum_optimize(a0, time_limit, verbose=True):
    """
    Subgradient method with momentum, targeting the top peaks.
    Much faster per iteration than LSE-BFGS.
    """
    n = len(a0)
    S_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S_target / n
    else:
        a = a * S_target / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    v = np.zeros(n)  # Momentum
    beta = 0.9  # Momentum coefficient
    t0 = time.monotonic()
    lr = 0.5
    lr_decay = 0.9999
    n_top = 5  # Number of top peaks to target
    
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        
        # Find top peaks
        top_k = np.argsort(b)[::-1][:n_top]
        # Weight peaks by how close they are to the max
        weights = np.exp(-(b[top_k] - peak_val) * 5.0)  # Soft weighting
        weights /= weights.sum()
        
        # Compute gradient: sum of weighted peak gradients
        # grad[j] = sum_m weights[m] * 2 * a[top_k[m] - j] (windowed)
        grad = np.zeros(n)
        for mi, k in enumerate(top_k):
            w = weights[mi] * 2.0
            # a[k-j] for valid j: j in [max(0, k-n+1), min(k, n-1)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                grad[js:je+1] += w * a[k - np.arange(js, je+1)]
        
        # Momentum update
        v = beta * v - lr * grad
        a_new = a + v  # Note: v already has the negative sign
        
        # Project
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * S_target / s_new
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:

def coordinate_descent_v2(a0, time_limit, seed=42):
    """
    Improved coordinate descent that targets peak reduction.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)

    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S_target / n
    else:
        a = a * S_target / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * S_target / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-12:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        
        # Target the top 3 peaks
        top_peaks = np.argsort(b)[::-1][:3]
        
        improved = False
        for k_peak in top_peaks:
            # Elements that contribute to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            
            # Move mass from high-contributing elements
            top_idx = idx[np.argsort(contrib)[::-1][:8]]
            
            # Find good destinations (elements far from peak)
            peak_zone = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_zone[js2:je2+1] = True
            
            safe_zones = np.where(~peak_zone)[0]
            if len(safe_zones) == 0:
                safe_zones = np.where(np.arange(n) != k_peak)[0]
            
            for src in top_idx:
                if a[src] < 1e-12:
                    continue
                # Try moving to safe zones
                dests = safe_zones[np.argsort(-a[safe_zones])[:10]]
                for dst in dests:
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        trial = a.copy()
                        trial[src] -= amt
                        trial[dst] += amt
                        if try_update(trial):

                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random exploration

                i = rng.integers(n)
                j = rng.integers(n)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def structured_init(n, rng):

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a combination of:
1. Fast subgradient momentum method with adaptive step-size (very fast per iteration)
2. LSE-smoothed objective + Adam optimizer with multi-temperature cooling
3. Peak-targeted coordinate descent for final polish
4. Multi-length exploration with interpolation transfer
Key differences from prior approach: no LP solver needed, orders of magnitude more
iterations, Adam-based adaptive learning rates, and multi-peak gradient targeting.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_max_and_weights(b, temperature):
    """Compute LSE approximation of max and softmax weights."""
    m = b.max()
    exp_vals = np.exp(temperature * (b - m))
    w = exp_vals / exp_vals.sum()
    lse = m + np.log(exp_vals.sum()) / temperature
    return lse, w


def compute_gradient_lse(a, temperature):
    """
    Compute gradient of LSE approximation of max(conv(a,a)) w.r.t. a.
    Returns (gradient, lse_value).
    The gradient formula: grad[j] = 2 * sum_k w[k] * a[k-j] (windowed).
    This is 2 * cross_correlation(w, a) evaluated at lags 0..n-1.
    """
    n = len(a)
    b = fast_autocorr(a)
    lse_val, w = lse_max_and_weights(b, temperature)

    # Compute cross-correlation: grad[j] = 2 * sum_i a[i] * w[i+j]
    # Using FFT: ifft(FFT(a_pad) * conj(FFT(w_pad)))
    fft_size = 1 << (3 * n - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w

    A_fft = rfft(a_pad, fft_size)
    W_fft = rfft(w_pad, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    return 2.0 * cross, lse_val


def lse_objective_grad(a, temperature):
    """
    Compute the scaled LSE objective and its gradient.
    Objective: 2*n*LSE / S^2
    """
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    conv_grad, lse_val = compute_gradient_lse(a, temperature)
    obj = 2.0 * n * lse_val / (S * S)
    grad = 2.0 * n * conv_grad / (S * S) - 2.0 * obj * a / S
    return obj, grad


def subgradient_peak_gradient(a, n_top=5, sharpness=8.0):
    """
    Compute a weighted combination of gradients at the top peaks.
    Much faster than LSE when we only care about a few peaks.
    Returns (gradient, peak_value).
    """
    n = len(a)
    b = fast_autocorr(a)
    peak_val = b.max()

    # Find top peaks
    top_k = np.argsort(b)[::-1][:n_top]
    # Weight by proximity to max (sharper = more focused on the top peak)
    weights = np.exp(-sharpness * (peak_val - b[top_k]))
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(n_top) / n_top
    else:
        weights /= wsum

    # Compute gradient: grad[j] = sum_m weights[m] * 2 * a[top_k[m] - j]
    grad = np.zeros(n)
    for mi, k in enumerate(top_k):
        w = weights[mi] * 2.0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            j_range = np.arange(js, je + 1)
            grad[js:je + 1] += w * a[k - j_range]

    return grad, peak_val


def subgradient_momentum_optimize(a0, time_limit, verbose=True, n_top=5, sharpness=8.0, beta_mom=0.85):
    """
    Subgradient method with momentum, targeting top peaks.
    Very fast per iteration - ideal for large numbers of steps.
    """

    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n * S_target
    else:
        a = a * S_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = 1.0
    lr_decay = 0.99995
    stall = 0
    iter_count = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        grad, peak_val = subgradient_peak_gradient(a, n_top, sharpness)

        # Nesterov-like momentum
        v = beta_mom * v - lr * grad
        a_new = a + v

        # Project onto simplex {a >= 0, sum(a) = S_target}
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * S_target / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
            a = a_new
        else:
            a = a_new
            stall += 1
            if stall > 200:
                # Adaptive: reduce lr, dampen momentum
                lr *= 0.9
                v *= 0.3
                stall = 0
                # Occasionally restart from best

                    a = best_a.copy()
                    v = np.zeros(n)
                    lr = 0.5
                    stall = 0
            if lr < 1e-4:
                break

        lr = max(lr * lr_decay, 1e-5)
        iter_count += 1

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        Subgrad iter={iter_count}: score={score:.8f} (t={elapsed:.0f}s, lr={lr:.5f})")

def adam_lse_optimize(a0, time_limit, T_start=40.0, T_end=1.0, n_stages=8, verbose=True):
    """
    Adam optimizer on LSE-smoothed objective with geometric temperature cooling.
    """
    n = len(a0)
    S_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S_target / n
    else:
        a = a * S_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Adam hyperparameters (tuned for this problem)
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.75, 0.999, 1e-8
        # Learning rate scales with temperature (larger T = smoother = can use larger lr)
        lr = 0.005 * np.sqrt(T_start / T)
        t_adam = 0

        stage_time = min(remaining / max(1, n_stages - si), 200.0)
        stage_start = time.monotonic()
        stall_count = 0

        while time.monotonic() - stage_start < stage_time:
            t_adam += 1
            obj, grad = lse_objective_grad(a, T)

            if not np.all(np.isfinite(grad)):
                break

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Project
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new * S_target / s_new

            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                stall_count = 0
            else:
                stall_count += 1

            a = a_new

            # Adaptive: if stalling, reduce lr
            if stall_count > 300:
                lr *= 0.8
                stall_count = 0
                if lr < 1e-6:
                    break

            if verbose and t_adam % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"        LSE+Adam stage={si+1}/{n_stages} (T={T:.1f}) iter={t_adam}: "
                      f"score={score:.8f} (t={elapsed:.0f}s)")

    return best_a, best_score


def coordinate_descent_peak(a0, time_limit, seed=42):
    """
    Coordinate descent that specifically targets peak reduction.
    Moves mass from elements contributing to peaks toward elements that don't.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    S_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S_target / n
    else:
        a = a * S_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()

        # Target top 4 peaks
        top_peaks = np.argsort(b)[::-1][:4]

        improved = False
        for k_peak in top_peaks:
            # Elements contributing to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]

            # Sort by contribution (highest first)
            order = np.argsort(contrib)[::-1]
            top_idx = idx[order[:10]]

            # Find safe destinations (far from all top peaks)
            peak_zone = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_zone[js2:je2 + 1] = True

            safe = np.where(~peak_zone)[0]
            if len(safe) < 10:
                safe = np.where(np.arange(n) != k_peak)[0]

            # Try moving from high-contribution sources to safe destinations
            for src in top_idx:
                if a[src] < 1e-12:
                    continue
                # Try several fraction sizes
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    # Move to a few best safe destinations
                    dest_candidates = safe[np.argsort(-a[safe])[:5]]
                    for dst in dest_candidates:
                        trial = a.copy()
                        trial[src] -= amt
                        trial[dst] += amt
                        trial = np.maximum(trial, 0.0)
                        s_t = trial.sum()
                        if s_t < 1e-10:
                            continue
                        trial = trial * S_target / s_t
                        score = fast_autocorr(trial).max()
                        if score < best_score - 1e-13:
                            a = trial
                            best_a = trial.copy()
                            best_score = score
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbations to escape
            if stall % 10 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.15)
                    amt = a[i] * frac
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0.0)
                    s_t = trial.sum()
                    if s_t < 1e-10:
                        continue
                    trial = trial * S_target / s_t
                    score = fast_autocorr(trial).max()
                    if score < best_score:
                        a = trial
                        best_a = trial.copy()
                        best_score = score
                        stall = 0
                        break

            if stall > 800:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate structured initial sequences with various shapes."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 1.0 / (1.0 + np.exp(-10.0 * (x - 0.5)))
    elif choice == 5:
        a = 1.0 - 0.4 * np.sin(6.0 * np.pi * x)
    elif choice == 6:
        a = np.abs(np.sin(2.0 * np.pi * x)) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def multi_peak_lse_optimize(a0, time_limit, verbose=True):
    """
    Hybrid optimizer that alternates between:
    - LSE gradient with Adam (smooth, global)
    - Peak-targeted subgradient with momentum (fast, local)
    This combines the strengths of both approaches.
    """
    n = len(a0)
    S_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S_target / n
    else:
        a = a * S_target / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Phase A: LSE + Adam (warm-up, explore landscape)
    t_a = min(time_limit * 0.4, 300.0)
    if t_a > 10:
        a, best_score = _adam_phase(a, best_score, best_a, t_a, T_start=30.0, T_end=3.0, n_stages=4)
        best_a = a.copy()

    # Phase B: Subgradient momentum (refine)
    remaining = time_limit - (time.monotonic() - t0)
    t_b = min(remaining * 0.4, 300.0)
    if t_b > 10:
        a, best_score = _subgrad_phase(a, best_score, best_a, t_b, verbose=False)
        best_a = a.copy()

    # Phase C: LSE + Adam (final polish with low temperature)
    remaining = time_limit - (time.monotonic() - t0)
    t_c = remaining * 0.9
    if t_c > 10:
        a, best_score = _adam_phase(a, best_score, best_a, t_c, T_start=8.0, T_end=0.5, n_stages=4)
        best_a = a.copy()

    return best_a, best_score


def _adam_phase(a, best_score, best_a, time_limit, T_start, T_end, n_stages):
    """Internal Adam phase."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.75, 0.999, 1e-8
        lr = 0.004 * np.sqrt(T_start / T)
        t_adam = 0
        stage_time = min(remaining / max(1, n_stages - si), 150.0)
        stage_start = time.monotonic()

        while time.monotonic() - stage_start < stage_time:
            t_adam += 1
            obj, grad = lse_objective_grad(a, T)
            if not np.all(np.isfinite(grad)):
                break

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new * S_target / s_new

            score = fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new

    return a, best_score


def _subgrad_phase(a, best_score, best_a, time_limit, verbose=False):
    """Internal subgradient phase."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = 0.8
    beta = 0.85
    stall = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        grad, peak_val = subgradient_peak_gradient(a, n_top=4, sharpness=10.0)
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * S_target / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.9
                v *= 0.3
                stall = 0
                if lr < 0.01:
                    a = best_a.copy()
                    v = np.zeros(n)
                    lr = 0.5

        a = a_new
        lr = max(lr * 0.99998, 1e-5)
        iters += 1
        if lr < 1e-4:
            break

    return a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine base length
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Hybrid LSE+Subgrad at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: Hybrid LSE+Subgrad n={n_base} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = multi_peak_lse_optimize(a_norm, t1, verbose=True)
    do_try_update(a_out, f"hybrid_n{n_base}")

    # === Phase 2: Pure subgradient momentum (fast, many iterations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: Subgrad momentum n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_momentum_optimize(a_norm, t2, verbose=True, n_top=4, sharpness=10.0, beta_mom=0.82)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # === Phase 3: LSE + Adam with fine temperature ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LSE+Adam fine n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = adam_lse_optimize(a_norm, t3, T_start=15.0, T_end=1.0, n_stages=6, verbose=True)
        do_try_update(a_out, f"lse_fine_n{n_best}")

    # === Phase 4: Try longer length ===
    n_long = n_best + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Longer length n={n_long} (budget={t4:.0f}s) ===")
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = multi_peak_lse_optimize(a_norm, t4, verbose=False)
        do_try_update(a_out, f"hybrid_long_n{n_long}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_peak(a_norm, t5, seed=99999)
        do_try_update(a_out, "coord_polish")

    # === Phase 6: Final hybrid polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = remaining * 0.8
    print(f"\n=== Phase 6: Final hybrid n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_lse_optimize(a_norm, t6, verbose=False)
        do_try_update(a_out, "hybrid_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time_used={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
