def lp_best_response(a, M, solver, n):
    b_var = cp.Variable(n, nonneg=True)
    # Build convolution matrix efficiently using scipy
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f} stall={stall_count}")

def spectral_flatten(a0, time_limit, seed=42):
    """Spectral flattening: compress the power spectrum to reduce autocorrelation peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = project_nonneg_normalize(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    pad_len = 1 << (2 * n - 1).bit_length()
    
    for alpha in [0.7, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05]:
        while time.monotonic() - t0 < time_limit and stall < 50:
            a_pad = np.zeros(pad_len)
            a_pad[:n] = a
            A = rfft(a_pad)
            power = A.real ** 2 + A.imag ** 2
            magnitudes = np.sqrt(power)
            
            # Compress the spectrum: mag_new = mag^alpha (in log domain: log_mag_new = alpha * log_mag)
            # But we need to be careful with zeros
            log_mag = np.log(magnitudes + 1e-15)
            mean_log = np.mean(log_mag)
            # Compress: bring all magnitudes closer to the geometric mean
            log_mag_new = np.exp(alpha * np.log(magnitudes + 1e-15))
            # Actually, let's do: log_mag_new = alpha * log_mag + (1-alpha) * mean_log
            log_mag_new = alpha * log_mag + (1.0 - alpha) * mean_log
            mag_new = np.exp(log_mag_new)
            
            # Reconstruct with same phase
            phases = A / (magnitudes + 1e-30)
            A_new = mag_new * phases
            
            a_new_full = irfft(A_new, pad_len)
            a_new = a_new_full[:n]
            a_new = project_nonneg_normalize(a_new, norm)
            
            sc = fast_autocorr(a_new).max()
            if sc < best_score - 1e-12:
                a = a_new
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    break  # Move to next alpha
    
    # Add random perturbations to escape local minima
    while time.monotonic() - t0 < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        frac = rng.uniform(0.001, 0.01)
        amt = a[i] * frac
        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        if a_trial[i] < 0:
            continue
        a_trial = project_nonneg_normalize(a_trial, norm)
        sc = fast_autocorr(a_trial).max()
        if sc < best_score:
            a = a_trial
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def targeted_refine(a0, time_limit, seed=12345):
    """Targeted peak reduction via mass redistribution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = project_nonneg_normalize(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find the peak (excluding lag 0)
        b_no_dc = b.copy()
        b_no_dc[0] = -1
        peak_k = np.argmax(b_no_dc)
        
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find best destination
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            # Sort by current value (prefer moving to low-value positions)
            candidates_sorted = candidates[np.argsort(a[candidates])]
            dst = candidates_sorted[rng.integers(min(10, len(candidates_sorted)))]

            for frac in [0.0003, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = project_nonneg_normalize(a_trial, norm)
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = project_nonneg_normalize(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 600:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            avg_check = project_nonneg_normalize(avg_check, norm_val)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f} stall={stall_count}")

        if stall_count > stall_threshold:
            perturb = rng.normal(0, 0.005 * norm_val / np.sqrt(n), n)
            y_pert = y + perturb
            y_pert = project_nonneg_normalize(y_pert, norm_val)
            score_pert = fast_autocorr(y_pert).max()
            if score_pert < best_score:
                best_score = score_pert
                best_a = y_pert.copy()
            y = y_pert
            stall_count = 0

    return best_a, best_score


def spectral_flatten(a0, time_limit, seed=42):
    """Spectral flattening: compress the power spectrum to reduce autocorrelation peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = project_nonneg_normalize(a0, norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    pad_len = 1 << (2 * n - 1).bit_length()

    alpha_list = [0.7, 0.6, 0.5, 0.4, 0.35, 0.3, 0.25, 0.2, 0.15, 0.1]
    for alpha in alpha_list:
        if time.monotonic() - t0 > time_limit:
            break
        stall = 0
        while time.monotonic() - t0 < time_limit and stall < 30:
            a_pad = np.zeros(pad_len)
            a_pad[:n] = a
            A = rfft(a_pad)
            magnitudes = np.abs(A)
            log_mag = np.log(magnitudes + 1e-15)
            mean_log = np.mean(log_mag)
            log_mag_new = alpha * log_mag + (1.0 - alpha) * mean_log
            mag_new = np.exp(log_mag_new)
            phases = A / (magnitudes + 1e-30)
            A_new = mag_new * phases
            a_new_full = irfft(A_new, pad_len)
            a_new = a_new_full[:n]
            a_new = project_nonneg_normalize(a_new, norm_val)
            sc = fast_autocorr(a_new).max()
            if sc < best_score - 1e-12:
                a = a_new
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    break

    # Random perturbations to escape local minima
    while time.monotonic() - t0 < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        frac = rng.uniform(0.001, 0.008)
        amt = a[i] * frac
        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        if a_trial[i] < 0:
            continue
        a_trial = project_nonneg_normalize(a_trial, norm_val)
        sc = fast_autocorr(a_trial).max()
        if sc < best_score:
            a = a_trial
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def targeted_refine(a0, time_limit, seed=12345):
    """Targeted peak reduction via mass redistribution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = project_nonneg_normalize(a0, norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_candidates = min(10, len(candidates_sorted))
            dst = candidates_sorted[rng.integers(n_candidates)]

            for frac in [0.0003, 0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = project_nonneg_normalize(a_trial, norm_val)
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm_val * 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = project_nonneg_normalize(a_trial, norm_val)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
