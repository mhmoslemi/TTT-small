"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective value 2*n*max_b/S^2."""
    b = fast_autocorr(a)
    max_b = b.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2 * len(a) * max_b / (S * S))


def build_conv_T(a):
    """Build the convolution matrix T where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def lp_best_response_cutting_plane(a, M, solver, max_constraints=300, max_rounds=10):
    """
    Solve the LP best response using cutting plane method.
    max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_T(a)
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Start with constraints at the most violated positions
    # First, try with just the peak constraint
    peak_idx = np.arange(size)
    # Start with a small set
    initial_set = np.linspace(0, size-1, min(20, size), dtype=int)
    
    active_mask = np.zeros(size, dtype=bool)
    active_mask[initial_set] = True
    
    b_val = None
    
    for round_idx in range(max_rounds):
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active_mask, :] @ b_var <= M[active_mask]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        
        # Check which constraints are violated
        if b_val.sum() < 1e-15:
            break
            
        circ = T @ b_val
        tol = 1e-6 * M.max()
        violated = circ > M + tol
        
        if not violated.any():
            break  # Feasible!
        
        # Add violated constraints (up to max_constraints total)
        current_count = active_mask.sum()
        if current_count >= max_constraints:
            # Keep the most violated
            v_idx = np.where(violated)[0]
            v_sev = circ[v_idx] - M[v_idx]
            # Add the worst violators
            n_add = min(50, max(0, max_constraints - current_count + 20))
            worst = v_idx[np.argsort(v_sev)[::-1][:n_add]]
            active_mask[worst] = True
            # If still too many, remove least violated active ones
            if active_mask.sum() > max_constraints:
                all_active = np.where(active_mask)[0]
                circ_active = circ[all_active]
                # Keep the ones with highest circ values
                order = np.argsort(circ_active)[::-1][:max_constraints]
                active_mask = np.zeros(size, dtype=bool)
                active_mask[all_active[order]] = True
        else:
            active_mask[violated] = True
    
    if b_val is None or b_val.sum() < 1e-15:
        return None
    return b_val


def lp_best_response_fixed(a, M, solver, active_indices=None):
    """Solve LP with a fixed set of constraints (for warm-starting in Nesterov)."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_T(a)
    
    if active_indices is None:
        active_indices = np.arange(size)
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_indices, :] @ b_var <= M[active_indices]]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """
    Fessler-Parks accelerated LP Nesterov with cutting-plane LP solves.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = eval_score(a)

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.8
    
    # Determine constraint budget based on time per LP solve
    # Start with moderate constraint count
    max_constraints = min(200, 2*n-1)
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr_vec = fast_autocorr(y)
        M_scalar = M_curr_vec.max()
        
        if M_scalar < 1e-12:
            break

        # Solve LP best response
        b = lp_best_response_cutting_plane(y, M_curr_vec, solver, 
                                            max_constraints=max_constraints,
                                            max_rounds=8)
        if b is None:
            # Fallback: use uniform
            b = np.ones(n)
            n_iters += 1
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            n_iters += 1
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum step
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = eval_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = eval_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if stall > 100:
            gamma = max(1.2, gamma - 0.05)
        elif stall == 0 and gamma < 1.8:
            gamma = min(1.8, gamma + 0.02)

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_scalar:.6f}, gamma={gamma:.2f}, stall={stall}")

        # Adaptive restart
        if stall > 400:
            rng = np.random.default_rng(n_iters * 31 + 17)
            perturb_scale = 0.05 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.8
            print(f"    [Restart at iter {n_iters}]")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, verbose=True):
    """
    Mirror descent (multiplicative weight) for the non-negative constrained problem.
    Uses LSE-smoothed gradient with adaptive step size.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    steps = 0
    
    # Initial temperature
    b0 = fast_autocorr(a)
    T0 = max(0.1, (b0.max() - b0.min()) * 0.5)
    T_factor = 0.97  # Cooling rate
    
    # Adaptive step size
    eta = 1.0 / (n * 0.5)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T0 * (T_factor ** steps)
        T = max(T, best_score * 0.001)
        
        b = fast_autocorr(a)
        b_max = b.max()
        
        # LSE-smoothed gradient
        w = np.exp((b - b_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient: grad_j = 2 * sum_k w[k] * a[k-j] (shifted)
        # Using FFT for efficiency
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        # grad[j] = 2 * sum_{k} w[k] * a[k-j] for valid k,j
        # This is conv(w, a_rev)[offset]
        a_rev = a[::-1]
        conv = irfft(W * rfft(a_rev, fft_size), fft_size)
        grad = 2.0 * conv[n-1:2*n-1]
        
        # Project gradient (zero-sum constraint for norm preservation)
        grad -= np.mean(grad)
        
        # Step size: use (gap / ||grad||^2) as Polyaak-type
        gap = b_max - best_score * 0.999
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq > 1e-30:
            polyak_eta = max(gap, 0) / grad_norm_sq
            eta = min(eta * 1.01, max(polyak_eta, 1e-8/n))
        eta = min(eta, 5.0 / n)
        
        # Mirror descent update: a_new = a * exp(-eta * grad) / Z
        log_update = -eta * grad
        # Clamp for numerical stability
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)  # Avoid exact zeros
        s_new = a_new.sum()
        a_new = a_new * norm / s_new
        
        steps += 1
        
        score = eval_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        elif score < b_max * 1.005:
            # Accept small worsening
            accept_prob = np.exp(-(score - best_score) / max(0.002 * best_score, 1e-10))
            if np.random.random() < accept_prob * (1.0 - 0.5 * progress):
                a = a_new
            else:
                eta *= 0.7
        else:
            eta *= 0.5
            if eta < 1e-10 / n:
                # Reset with perturbation
                rng = np.random.default_rng(steps)
                a = np.maximum(a + rng.normal(0, 0.02 * a.mean(), n), 0)
                a = a * norm / a.sum()
                eta = 1.0 / (n * 0.3)
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc step={steps} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.4f}, eta={eta:.2e}")
    
    return best_a, best_score


def top_k_shave(a0, time_limit, K=5, verbose=True):
    """
    Directly attack the top K peaks of the autocorrelation.
    For each peak, compute the subgradient and do a projected gradient step.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    steps = 0
    step_size = 0.01 / n
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Get top K peaks
        top_k = min(K, len(b))
        peak_indices = np.argsort(b)[::-1][:top_k]
        peak_vals = b[peak_indices]
        
        # Equal weighting (or softmax)
        w = np.ones(top_k) / top_k
        
        # Compute combined subgradient
        grad = np.zeros(n)
        for m, p in enumerate(peak_indices):
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[idx] += w[m] * 2.0 * a[p - idx]
        
        # Project onto sum=0 subspace
        grad -= np.mean(grad)
        
        # Adaptive step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Try several step sizes
        best_trial = b_max
        best_ss = step_size
        for ss in [step_size * 0.5, step_size, step_size * 2, step_size * 4]:
            a_trial = a - ss * grad
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial:
                best_trial = sc
                best_ss = ss
        
        a_new = a - best_ss * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        score = eval_score(a_new)
        steps += 1
        
        if score < best_score:
            a = a_new
            best_score = score
            best_a = a.copy()
            step_size = min(step_size * 1.1, 1.0/n)
        else:
            step_size *= 0.7
            if step_size < 1e-12 / n:
                # Restart
                rng = np.random.default_rng(steps)
                a = np.maximum(a + rng.normal(0, 0.03 * a.mean(), n), 0)
                a = a * norm / a.sum()
                step_size = 0.01 / n
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK step={steps} ({elapsed:.0f}s): best={best_score:.8f}, step={step_size:.2e}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def diverse_initializations(n, rng):
    """Generate diverse initial sequences for exploration."""
    inits = []
    
    # Sinusoidal
    x = np.linspace(0, 1, n)
    inits.append(np.sin(np.pi * x) ** 1.5)
    
    # Triangular
    tri = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    inits.append(tri)
    
    # Bimodal
    bimodal = np.exp(-50 * (x - 0.3)**2) + np.exp(-50 * (x - 0.7)**2)
    inits.append(bimodal)
    
    # Random positive
    inits.append(np.abs(rng.normal(1, 0.5, n)))
    
    # Cosine raised
    inits.append(np.maximum(0, np.cos(np.pi * (x - 0.5))) ** 2)
    
    # Plateau
    plateau = np.ones(n)
    plateau[:n//10] = 0
    plateau[-n//10:] = 0
    # Smooth the edges
    from scipy.ndimage import uniform_filter1d
    plateau = uniform_filter1d(plateau, size=max(2, n//20))
    inits.append(plateau)
    
    # Power law
    inits.append(x ** 0.5 * (1-x) ** 0.5)
    
    for i in range(len(inits)):
        s = inits[i].sum()
        if s > 1e-15:
            inits[i] = inits[i] / s
        else:
            inits[i] = np.ones(n) / n
    
    return inits
