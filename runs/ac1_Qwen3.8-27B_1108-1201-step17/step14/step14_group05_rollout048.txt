import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Fast linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_gradient(w, a):
    """Compute g[j] = sum_i a[i] * w[j+i] via correlation.
    w has length 2n-1, a has length n, result has length n.
    """
    return np.correlate(w, a, mode='valid')


def project_normalize(a, norm_target):
    """Project onto {a >= 0, sum(a) = norm_target}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, norm_target / len(a))
    return a * norm_target / s


def softmax_weights(conv_result, T):
    """Compute softmax weights for the convolution result."""
    max_val = conv_result.max()
    if max_val < 1e-30:
        w = np.zeros_like(conv_result)
        w[np.argmax(conv_result)] = 1.0
        return w
    w = np.exp((conv_result - max_val) / T)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv_result)
        w[np.argmax(conv_result)] = 1.0
    else:
        w /= w_sum
    return w

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_parametric_init(n, shape='sine', seed=0):

        a = rng.dirichlet(np.ones(n) * 0.5)
    else:
        a = np.sin(np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies:
    1. Nesterov-accelerated gradient on LSE-smoothed objective (fast, FFT-based)
    2. Multi-peak subgradient with annealing
    3. Greedy exchange-based local search
    4. Multi-start with different n and shapes
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        print("No parent found, using default initialization")
        for n_try in [1200, 1400, 1600, 1800]:
            for shape in ['sine', 'gaussian', 'bimodal']:
                a_try = make_parametric_init(n_try, shape, seed=seed + n_try)
                sc = evaluate_safe(a_try)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_try.copy()
                    n_base = n_try
        print(f"Best default: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Multi-start accelerated gradient at parent n ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Accelerated Gradient n={n_base} (budget={remaining:.0f}s) ===")

    # Try multiple temperature schedules
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()

        # T schedule 1: wide
        T_start = max(a_norm.max() * n_cur * 0.1, 0.5)
        T_end = max(a_norm.max() * n_cur * 0.001, 0.01)
        a_out, _ = accelerated_optimize(a_norm, t1 * 0.5, T_start, T_end, verbose=True, seed=seed)
        try_update(a_out, f"ag1_n{n_cur}")

        # T schedule 2: narrow (refinement)
        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.10, 120.0)
        if t1b > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            cur_max = fast_conv(a_norm).max()
            T_start2 = max(cur_max * 0.05, 0.1)
            T_end2 = max(cur_max * 0.002, 0.01)
            a_out, _ = accelerated_optimize(a_norm, t1b, T_start2, T_end2, verbose=True, seed=seed + 1)
            try_update(a_out, f"ag2_n{n_cur}")

    # === Phase 2: Multi-peak subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-peak SubGrad n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_refine(a_norm, t2, verbose=True, seed=seed + 2)
        try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 3: Local search (exchange moves) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Local Search n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = local_search_exchange(a_norm, t3, verbose=True, seed=seed + 3)
        try_update(a_out, f"ls_n{n_cur}")

    # === Phase 4: Try slightly different n values ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Try different n (budget={remaining:.0f}s) ===")

    n_candidates = []
    for delta in [50, 100, 150, -50, -100]:
        n_try = n_base + delta
        if 400 <= n_try <= 4000:
            n_candidates.append(n_try)

    for n_try in n_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_try = min(remaining * 0.12, 100.0)
        if t_try < 15:
            break

        print(f"  Trying n={n_try} (t={t_try:.0f}s)")
        a_interp = interpolate_sequence(best_seq, n_try)
        a_norm = a_interp * np.sqrt(2.0 * n_try) / a_interp.sum()

        # Quick gradient optimization
        cur_max = fast_conv(a_norm).max()
        T_s = max(cur_max * 0.1, 0.1)
        T_e = max(cur_max * 0.003, 0.01)
        a_out, _ = accelerated_optimize(a_norm, t_try * 0.6, T_s, T_e, verbose=False, seed=seed + n_try)
        if a_out is not None:
            # Quick local search
            a_out2, _ = local_search_exchange(a_out, t_try * 0.3, verbose=False, seed=seed + n_try + 1)
            try_update(a_out2, f"n{n_try}")
            try_update(a_out, f"n{n_try}_ag")

    # === Phase 5: Refine best with multiple methods ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Refine best n={len(best_seq)} (budget={remaining:.0f}s) ===")

    # 5a: Another AG round with tighter T
    t5a = min(remaining * 0.20, 200.0)
    if t5a > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        cur_max = fast_conv(a_norm).max()
        T_s = max(cur_max * 0.03, 0.05)
        T_e = max(cur_max * 0.001, 0.005)
        a_out, _ = accelerated_optimize(a_norm, t5a, T_s, T_e, verbose=True, seed=seed + 10)
        try_update(a_out, f"refine_ag_n{n_cur}")

    # 5b: Local search
    remaining = time_budget - (time.monotonic() - t_global)
    t5b = min(remaining * 0.20, 180.0)
    if t5b > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = local_search_exchange(a_norm, t5b, verbose=True, seed=seed + 20)
        try_update(a_out, f"refine_ls_n{n_cur}")

    # 5c: Subgradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t5c = min(remaining * 0.15, 120.0)
    if t5c > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_refine(a_norm, t5c, verbose=True, seed=seed + 30)
        try_update(a_out, f"refine_sg_n{n_cur}")

    # === Phase 6: Final alternating refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final alternating (budget={remaining:.0f}s) ===")

    if remaining > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()

        # Alternate between AG and LS for the remaining time
        n_rounds = max(2, int(remaining // 60))
        for rnd in range(n_rounds):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            t_rnd = min(remaining * 0.4, 80.0)

            # AG round
            cur_max = fast_conv(a_norm).max()
            T_s = max(cur_max * 0.02, 0.03)
            T_e = max(cur_max * 0.001, 0.005)
            a_out, _ = accelerated_optimize(a_norm, t_rnd * 0.5, T_s, T_e, verbose=False, seed=seed + 100 + rnd)
            if a_out is not None and fast_conv(a_out).max() < cur_max:
                a_norm = a_out * np.sqrt(2.0 * n_cur) / a_out.sum()
                try_update(a_out, f"alt_ag_{rnd}")

            # LS round
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            a_out2, _ = local_search_exchange(a_norm, t_rnd * 0.4, verbose=False, seed=seed + 200 + rnd)
            if a_out2 is not None:
                try_update(a_out2, f"alt_ls_{rnd}")
                a_norm = a_out2 * np.sqrt(2.0 * n_cur) / a_out2.sum()

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
