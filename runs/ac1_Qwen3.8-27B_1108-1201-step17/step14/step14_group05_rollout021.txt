def fast_autocorr(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_objective_and_grad(a, T):
    """Compute LSE-smoothed objective and gradient w.r.t. a.
    Returns (lse_value, gradient) where gradient is d/d a of the LSE surrogate.
    """
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return max_b, np.zeros_like(a)
    w = w / w_sum
    lse_val = max_b + T * np.log(w_sum)
    # Gradient: 2 * cross-correlation of w with a
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # (w * a)[i] = sum_k w[k] * a[k-i] ... we need sum_k w[k] * a[i+k-n+1] type thing
    # Actually: grad_i = sum_k w_k * 2 * a[k-i] for valid k-i in [0,n-1]
    # This is 2 * (flipped_w convolve a) evaluated at position n-1
    w_flip = w[::-1]
    grad_conv = irfft(rfft(w_flip, fft_size) * A, fft_size)
    grad = 2.0 * grad_conv[n - 1:2 * n - 1]
    return lse_val, grad


def project_simplex(v, S):
    """Project vector v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    # Use bisection on the lagrange multiplier
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cumsum[-1] - S))[0]
    if len(rho) == 0:
        # All zero
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cumsum[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_best_response(a, M, solver, max_constr=3000):

    cols = []
    vals = []
    for k in range(size):

        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    return b_val

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive step and Newton-like acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.time()
    steps = 0
    
    # Temperature schedule: start high, end very low
    b0 = fast_autocorr(a)
    T_start = max(0.1, (b0.max() - b0.min()) * 0.5)
    T_end = max(0.0001, best_score * 0.001)
    
    # Adaptive step size
    step_size = 0.05 / n
    best_step = step_size
    grad_hist = []
    
    while time.time() - t0 < time_limit:
        progress = min(1.0, (time.time() - t0) / time_limit)
        # Exponential cooling
        T = T_start * (T_end / T_start) ** progress
        # In final phase, use even lower T
        if progress > 0.8:
            T = T_end * (1.0 + 0.5 * (1.0 - (progress - 0.8) / 0.2))
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Compute LSE weights
        b_shifted = b - max_b
        max_shift = b_shifted.max()
        if max_shift > 50:
            b_shifted = b_shifted - max_shift
            T_eff = T * np.exp(max_shift / T) if max_shift < 700 else T
        else:
            T_eff = T
        
        exp_vals = np.exp(b_shifted / T)
        w_sum = exp_vals.sum()
        if w_sum < 1e-30:
            break
        w = exp_vals / w_sum
        
        # Compute gradient via FFT
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # grad_i = 2 * sum_k w[k] * a[k-i] (valid indices)
        w_flip = w[::-1]
        grad_raw = irfft(rfft(w_flip, fft_size) * A, fft_size)
        grad = 2.0 * grad_raw[n - 1:2 * n - 1]
        
        # Project gradient onto tangent space (sum = 0)
        grad -= np.mean(grad)
        
        # Adaptive step size with backtracking
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Newton-like step: step = (f - f_opt_lower) / ||grad||^2
        # Lower bound is approximately the mean of b
        f_lower = np.mean(b)
        polyak_step = max(0.01 * (max_b - f_lower), 1e-10) / (grad_norm * grad_norm + 1e-20)
        step = min(step_size * 3.0, max(polyak_step, 1e-8 / n))

        accepted = False
        for _ in range(5):
            a_trial = a - alpha_try * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                alpha_try *= 0.5

            score_trial = fast_autocorr(a_trial).max()

                a = a_trial
                accepted = True
                if score_trial < best_score:
                    best_score = score_trial
                    best_a = a.copy()
                    step_size = min(step_size * 1.1, 1.0 / n)
                    best_step = alpha_try
                break
            alpha_try *= 0.5

            # Occasionally try a larger random step

            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.8f}, max_b={max_b:.6f}")
        
        # Escape mechanism
        if steps % 500 == 0 and step_size < 1e-10 / n:
            rng = np.random.default_rng(steps)
            n_pert = max(10, n // 20)
            idx = rng.choice(n, n_pert, replace=False)
            a[idx] += rng.normal(0, 0.01 * a.mean(), n_pert)
            a = np.maximum(a, 0)
            a = a * norm_val / a.sum()
            step_size = 0.02 / n
    
    return best_a, best_score


def lse_newton_polish(a0, time_limit, verbose=True):
    """Final polish using LSE with very fine temperature and careful line search."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.time()
    steps = 0
    
    b0 = fast_autocorr(a)
    spread = b0.max() - np.median(b0)
    T = max(spread * 0.1, 0.01)
    
    prev_grad = None
    
    while time.time() - t0 < time_limit:
        progress = (time.time() - t0) / time_limit
        T = max(T * 0.95, 0.0001)  # Cool down each iteration
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # LSE weights
        b_shifted = b - max_b
        T_eff = T
        exp_vals = np.exp(b_shifted / T_eff)
        w_sum = exp_vals.sum()
        if w_sum < 1e-30:
            break
        w = exp_vals / w_sum
        
        # Gradient
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        w_flip = w[::-1]
        grad_raw = irfft(rfft(w_flip, fft_size) * A, fft_size)
        grad = 2.0 * grad_raw[n - 1:2 * n - 1]
        grad -= np.mean(grad)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Nesterov-like momentum
        if prev_grad is not None:
            beta = 0.9
            grad_mom = grad + beta * prev_grad
            grad_mom -= np.mean(grad_mom)
        else:
            grad_mom = grad
        prev_grad = grad
        
        # Step size
        step = 0.1 / (grad_norm * np.linalg.norm(grad_mom) + 1e-20)
        step = min(step, 1.0 / n)
        
        # Backtracking line search
        alpha = step
        best_trial_score = max_b
        best_trial_a = a
        for _ in range(10):
            a_trial = a - alpha * grad_mom
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial * norm_val / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_a = a_trial
            if sc < max_b * 0.999:
                break
            alpha *= 0.5
        
        a = best_trial_a
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        steps += 1
        if verbose and steps % 100 == 0:
            elapsed = time.time() - t0
            print(f"    Polish step={steps} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, seed=42):
    """Generate a structured initial sequence with good autocorrelation properties."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Superposition of smooth basis functions
    a = np.ones(n) * 0.5
    for freq in range(1, 6):
        a += 0.1 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization to minimize normalized peak autocorrelation."""
    t_global = time.time()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            s_p = p.sum()
            best_seq = p / s_p
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, seed)
        best_score = evaluate_safe(best_seq)
        print(f"Structured init: n={n_base}, score={best_score:.6f}")
    
    n_base = len(best_seq)
    
    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.time() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.time() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        update_best(a_out, f"lp1_n{n_cur}")
    
    # === Phase 2: LSE gradient descent ===
    remaining = time_budget - (time.time() - t_global)

        a_out, _ = lse_gradient_descent(a_norm, t2, verbose=True)
        update_best(a_out, f"lse1_n{n_cur}")
    
    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.time() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True)
        update_best(a_out, f"lp2_n{n_cur}")
    
    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.time() - t_global)
    n_cur = len(best_seq)
    # Try a few different lengths
    length_options = [n_cur + 50, n_cur + 100, n_cur - 50]

        update_best(a_out, f"lse_final_n{n_cur}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

        x = np.linspace(0, 1, 1600)
        best_score = evaluate_safe([float(v) for v in np.sin(np.pi * x)])
        return [float(v) for v in np.sin(np.pi * x)]
    
    return [float(x) for x in best_seq]
