Actually, one more important consideration: **the `sliding_window_view` approach for the gradient**.

`w` has length `2n-1`. `sliding_window_view(w, n)` gives a view of shape `(2n-1-n+1, n) = (n, n)` where `view[j, m] = w[j+m]`.

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_grad(a, n, tau):
    """Compute LSE objective value and gradient w.r.t. a."""
    b = fast_conv(a)
    S = a.sum()
    if S < 1e-10:
        return None, None
    max_b = b.max()
    scaled = (b - max_b) / tau
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    if Z < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(b)] = 1.0
        lse_val = max_b
    else:
        w = exp_scaled / Z
        lse_val = max_b + tau * np.log(Z)
    windows = sliding_window_view(w, n)
    h = windows @ a
    dLSE = 2.0 * h
    F_val = 2.0 * n * lse_val / (S * S)
    grad = 2.0 * n * dLSE / (S * S) - 4.0 * n * lse_val * a / (S**3)
    return F_val, grad

def lse_grad_normalized(a, n, tau):
    """Gradient assuming sum(a) = sqrt(2n), so objective = max_b approx."""
    b = fast_conv(a)
    max_b = b.max()
    scaled = (b - max_b) / tau
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    if Z < 1e-300:

        w[np.argmax(b)] = 1.0
        lse_val = max_b
    else:
        w = exp_scaled / Z
        lse_val = max_b + tau * np.log(Z)
    windows = sliding_window_view(w, n)
    h = windows @ a
    dLSE = 2.0 * h
    return lse_val, dLSE

def adam_lse_optimize(a0, n, time_limit, T_start=40.0, T_min=0.5, lr=0.05, verbose=True):
    """Adam optimizer on LSE objective with decreasing temperature."""
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_true = fast_conv(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    t0 = time.monotonic()
    stall = 0
    best_t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        T = max(T_min, T_start * (0.95 ** (t / 20)))
        
        lse_val, grad = lse_grad_normalized(a, n, T)
        if grad is None:
            break
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        current_lr = lr / (1.0 + 0.05 * t)
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new
        
        true_score = fast_conv(a).max()
        if true_score < best_true:
            best_true = true_score
            best_a = a.copy()
            stall = 0
            best_t = t
        else:
            stall += 1
        
        if stall > 500:
            # Restart with perturbation
            rng = np.random.default_rng(t)
            a_pert = a + rng.normal(0, 0.01 * norm / n, n)
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-10:
                a = a_pert * norm / s_p
                m = np.zeros(n)
                v = np.zeros(n)
                stall = 0
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam iter={t} T={T:.2f} lr={current_lr:.6f} best={best_true:.8f} t={elapsed:.0f}s")
    
    return best_a, best_true

def nesterov_lp(a0, n, time_limit, solver, verbose=True):

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    last_improve_iter = 0
    
    for step in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Line search
        alpha_base = 2.0 / (step + 2.0)
        best_alpha = alpha_base
        best_obj = fast_conv(np.maximum((1 - alpha_base) * y + alpha_base * g, 0)).max()
        for trial in [alpha_base * 0.5, alpha_base * 1.5, alpha_base * 2.0, alpha_base * 0.25]:
            trial = min(trial, 1.0)
            if trial <= 0:
                continue
            cand = np.maximum((1 - trial) * y + trial * g, 0)
            s_c = cand.sum()
            if s_c < 1e-15:
                continue
            cand = cand * norm / s_c
            sc = fast_conv(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial
        alpha = best_alpha
        
        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Nesterov extrapolation
        if step == 1:
            a_cur = g.copy()
        else:
            momentum = (step - 1.0) / (step + 2.0)
            a_cur = y_new + momentum * (y_new - y)
            a_cur = np.maximum(a_cur, 0.0)
            s_a = a_cur.sum()
            if s_a < 1e-15:
                a_cur = y_new.copy()
            else:
                a_cur = a_cur * norm / s_a
        
        y = y_new
        
        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve_iter = n_iters
        
        score_a = fast_conv(a_cur).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_cur.copy()
            last_improve_iter = n_iters
        
        stall = n_iters - last_improve_iter
        if stall > 100:
            if stall % 200 == 100:
                y = best_a * norm / best_a.sum()
            if stall > 500:
                break
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f} M={M_curr:.6f}")
    
    return best_a, best_score

def gradient_coord_descent(a0, n, time_limit, seed=42):
    """Coordinate descent guided by LSE gradient."""
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        
        # Compute gradient at low temperature
        tau = 1.0
        scaled = (b - b.max()) / tau
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        windows = sliding_window_view(w, n)
        h = windows @ a
        grad = 2.0 * h  # gradient of LSE (up to constant)
        
        # Normalize gradient
        grad_mean = grad.mean()
        grad_std = grad.std()
        if grad_std > 1e-15:
            grad_norm = (grad - grad_mean) / grad_std
        else:
            grad_norm = np.zeros(n)
        
        # Coordinates to decrease (positive gradient = bad)
        dec_candidates = np.where(grad_norm > 0.3)[0]
        # Coordinates to increase (negative gradient = good)
        inc_candidates = np.where(grad_norm < -0.3)[0]
        
        improved = False
        
        # Try transfers from dec to inc
        if len(dec_candidates) > 0 and len(inc_candidates) > 0:
            # Sort by magnitude
            dec_sorted = dec_candidates[np.argsort(-grad_norm[dec_candidates])]
            inc_sorted = inc_candidates[np.argsort(grad_norm[inc_candidates])]
            
            for src in dec_sorted[:10]:
                if a[src] < 1e-12:
                    continue
                for dst in inc_sorted[:10]:
                    if src == dst:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_conv(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_a = a.copy()
                            best_score = score
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Random perturbation
            stall += 1
            if stall % 10 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_a = a.copy()
                        best_score = score
                        stall = 0
                        break
            
            if stall > 500:
                # Bigger perturbation
                n_pert = max(5, n // 100)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.3, 2.0)
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score:

def fine_coord_descent(a0, n, time_limit):

        improved = False
