def fast_autocorr(a):
    """Compute full autocorrelation (convolution) of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-30:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_gradient(y, T):
    """Compute gradient of the log-sum-exp approximation to max(autocorr)."""
    n = len(y)
    y_pos = np.maximum(y, 0.0)
    ac = fast_autocorr(y_pos)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    exp_sum = exp_s.sum()
    if exp_sum < 1e-30:
        w = np.zeros(len(ac))
        w[np.argmax(ac)] = 1.0
    else:
        w = exp_s / exp_sum
    # Gradient: sum_k w[k] * d(conv(a,a)[k])/da[i]
    # d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n (accounting for symmetry)
    # Use FFT-based cross correlation for efficiency
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    Y_rev = rfft(y_pos[::-1], N)
    cross = irfft(W * Y_rev, N)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

            break
        stage_start = time.monotonic()

        stage_budget = min(stage_budget, time_limit - (time.monotonic() - t0))
        if stage_budget < 5:

def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. (a⊛b)[k] <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix (Toeplitz/convolution structure)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def multipass_targeted_shave(a0, time_limit, seed=42):
    """Multi-peak targeted shaving: reduce top-K peaks simultaneously."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    K = 5  # number of top peaks to consider

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top K peaks
        top_peaks = np.argsort(b)[::-1][:K]
        peak_vals = [b[k] for k in top_peaks]
        max_peak = max(peak_vals)
        improved = False

        for k in top_peaks:
            # Identify contributing pairs for this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            # Find the highest contributing pair
            src_pos = idx[np.argmax(contrib)]
            partner_pos = k - src_pos

            # Build mask of positions that contribute to ANY top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            # Candidate destinations: positions NOT in any peak's range, or low-value positions
            candidates = np.where(
                (~peak_mask) | (a < np.median(a))
            )[0]
            candidates = candidates[candidates != src_pos]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src_pos)[0]

            # Pick destination: prefer low-value positions far from peak contributors
            # Score: low a-value, far from partner positions
            dist_to_partner = np.abs(candidates.astype(np.float64) - partner_pos)
            scores = -a[candidates] + 0.1 * dist_to_partner / n
            top_candidates = candidates[np.argsort(scores)[::-1][:max(1, len(candidates) // 10)]]
            dst = top_candidates[rng.integers(len(top_candidates))]

            # Try multiple transfer fractions
            for frac in [0.0003, 0.0007, 0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src_pos] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[dst] += amt
                if a_trial[src_pos] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                # Check: does this improve the max peak without worsening others too much?
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            stall += 1
            if stall > 500:
                # Larger perturbation
                n_perturb = rng.integers(3, 10)
                sources = rng.choice(n, n_perturb, replace=False)
                a_trial = a.copy()

                        amt = a_trial[s_idx] * rng.uniform(0.01, 0.05)
                        a_trial[s_idx] -= amt

                a_trial = np.maximum(a_trial, 0.0)

            if stall > 2000:

def dual_guided_subgradient(a0, time_limit, verbose=False):
    """Subgradient method guided by the peak structure with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    x_prev = a.copy()
    t_iter = 0

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        if t_iter == 1:
            y = a.copy()
        else:
            beta = (t_iter - 1.0) / (t_iter + 2.0)
            y = (1.0 + beta) * a - beta * x_prev
            y_pos = np.maximum(y, 0.0)
            s_y = y_pos.sum()
            if s_y > 1e-15:
                y = y_pos / s_y

        k_star = np.argmax(ac)

        # Subgradient: g[i] = 2 * y[k*-i] if 0 <= k*-i < n, else 0
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        g[lo:hi] = 2.0 * y_pos[k_star - (np.arange(lo, hi))]

        # Frank-Wolfe style: find min of g
        min_g_idx = np.argmin(g)
        # But use a "soft" version: put mass proportional to exp(-g/T_fw)
        T_fw = max(0.01, 0.1 * g.max())
        w = np.exp(-g / T_fw)
        w_sum = w.sum()
        if w_sum < 1e-15:
            s_fw = np.zeros(n)
            s_fw[min_g_idx] = 1.0
        else:
            s_fw = w / w_sum

        gamma = 2.0 / (t_iter + 2.0)
        x_prev = a.copy()
        a = (1.0 - gamma) * a + gamma * s_fw
        a = project_simplex(a, 1.0)

        if t_iter % 200 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if verbose and t_iter % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Subgrad iter={t_iter} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
