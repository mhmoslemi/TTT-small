def fast_autocorr(a):
    """Compute full linear autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_best_response_truncated(a, M, solver, top_k=None):
    """
    LP best response with only top-k lag constraints.
    Still uses the full dense matrix but only for active constraints.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    
    # Compute which lags are "active" (near the peak)
    a_autocorr = fast_autocorr(a)
    # We need conv(a, b) <= M for all lags, but b is variable.
    # The binding constraints will be those where a has large values.
    # For now, use all constraints but with a sparser structure.
    
    # Build the constraint matrix T where (T @ b)_k = sum_i a[i] * b[k-i] (with valid indices)
    # This is a Toeplitz-like matrix
    # T has shape (2n-1, n)
    # T[k, i] = a[i] if k-i is in [0, n-1], else 0
    # i.e., T[k, i] = a[k-i] if 0 <= k-i < n and 0 <= i < n
    
    # For efficiency with large n, use a different approach:
    # conv(a, b)_k = sum_{i=0}^{n-1} a[i] * b[k-i] (with b[j]=0 for j out of [0,n-1])
    
    # Let's build T as a sparse matrix
    from scipy.sparse import lil_matrix, csr_matrix
    
    T = lil_matrix((size, n), dtype=np.float64)
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[i:end-start]
    
    T = T.tocsr()
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def fw_best_response(a, M, n_top=50):
    """
    Frank-Wolfe style best response using gradient of a surrogate.
    Much faster than full LP.
    
    We approximate: minimize max_k conv(a,b)_k over b>=0, sum(b)=1
    by using the LSE surrogate and a greedy coordinate approach.
    """
    n = len(a)
    
    # The gradient of conv(a,b) with respect to b at b=e_j is the column of T
    # T[:, j] = convolution of a with a delta at position j
    # = a shifted by j (truncated)
    
    # For FW: minimize <grad_LSE(a_curr, a_curr), b> over b in simplex
    # where grad_LSE is the gradient of the LSE of conv(a_curr, .)
    
    # Compute current autocorrelation
    b_curr = fast_autocorr(a)
    max_b = b_curr.max()
    
    # LSE gradient weights
    T_temp = 20.0
    exp_vals = np.exp(T_temp * (b_curr - max_b))
    w = exp_vals / exp_vals.sum()
    
    # Gradient of LSE wrt b_new: grad_j = sum_k w_k * T[k,j] = sum_k w_k * conv(a, e_j)[k]
    # conv(a, e_j)[k] = a[k-j] if 0 <= k-j < n, else 0
    # So grad_j = sum_{k=j}^{j+n-1} w[k] * a[k-j] (with bounds)
    # = sum_{i=0}^{n-1} w[j+i] * a[i] (with bounds on j+i < 2n-1)
    
    # This is a cross-correlation of w and a!
    grad = np.zeros(n)
    for j in range(n):
        # grad_j = sum_i a[i] * w[j+i] where j+i < 2n-1
        max_i = min(n, 2*n-1-j)
        if max_i > 0:
            grad[j] = np.dot(a[:max_i], w[j:j+max_i])
    
    # FW step: b = e_{argmin grad}
    j_min = np.argmin(grad)
    b = np.zeros(n)
    b[j_min] = 1.0
    
    return b


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_line_search=True, use_truncated_lp=False):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_truncated(y, M_curr, solver)
        if b is None:
            # Fallback to uniform
            b = np.ones(n) / n
            n_iters += 1

        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
            b_sum = 1.0
        g = b * norm / b_sum
        n_iters += 1

        if use_line_search:
            alpha_base = 2.0 / (t + 2.0)
            best_alpha = alpha_base
            best_obj = fast_autocorr(np.maximum((1 - alpha_base) * y + alpha_base * g, 0)).max()
            for trial_alpha in [alpha_base * 0.5, alpha_base * 1.5, alpha_base * 2.0, alpha_base * 0.25]:
                trial_alpha = min(trial_alpha, 1.0)
                cand = np.maximum((1 - trial_alpha) * y + trial_alpha * g, 0.0)
                sc = fast_autocorr(cand).max()
                if sc < best_obj:
                    best_obj = sc
                    best_alpha = trial_alpha
            alpha = best_alpha
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg

                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if t % 200 == 0:
            stall_count += 1
            if stall_count > 30:
                break
            elif stall_count > 8:
                y = best_a * norm / best_a.sum()
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

def spectral_smooth(a, alpha=0.01):

def coordinate_descent_polish(a0, time_limit, seed=42):
    """Targeted coordinate descent focusing on peak-reducing moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-14:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:16]

        improved = False

        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:8]]

            # Build mask of indices contributing to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if je2 >= js2:
                    peak_mask[js2:je2+1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-12:
                    continue
                # Try moving mass from src to positions that don't contribute to peaks
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Prefer moving to positions with small current value
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 3)]

                for dst in candidates_sorted[:8]:
                    for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbations to escape
            if stall % 3 == 0:
                n_perturb = max(5, n // 100)
                a_trial = a.copy()
                pert_indices = rng.choice(n, size=n_perturb, replace=False)
                for idx in pert_indices:
                    a_trial[idx] *= rng.uniform(0.7, 1.3)
                a_trial = np.maximum(a_trial, 0)
                try_update(a_trial)

            # Single random swap
            if stall % 2 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue

                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x) + 0.1 * np.sin(8.0 * np.pi * x)
    elif choice == 7:
        # Wigner semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(1 - x**2)
    elif choice == 8:
        # Bessel-like
        x = np.linspace(0, 1, n)
        a = np.abs(np.sinc(3 * (x - 0.5)))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def mirror_descent_step(a, eta, n_top_peaks=10):
    """
    Multiplicative (mirror) descent step targeting peak reduction.
    a_new[i] = a[i] * exp(-eta * g_i) where g_i is the "contribution to peaks"
    """
    n = len(a)
    b = fast_autocorr(a)
    top_peaks = np.argsort(b)[::-1][:n_top_peaks]
    
    # Compute gradient of peak contribution for each coordinate
    grad = np.zeros(n)
    for k in top_peaks:
        weight = b[k] - b.mean()
        if weight <= 0:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            j = k - i
            grad[i] += weight * a[j]
            if j != i:
                grad[j] += weight * a[i]
    
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-15)
    return a_new
