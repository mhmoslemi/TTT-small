def build_T(y):
    """Build constraint matrix T where T[k,j] = y[k-j] for valid indices.
    Shape: (2n-1) x n
    """
    n = len(y)
    # Vectorized construction
    ks = np.arange(2 * n - 1)[:, None]  # (2n-1, 1)
    js = np.arange(n)[None, :]  # (1, n)
    diff = ks - js  # (2n-1, n)
    T = np.where((diff >= 0) & (diff < n), y[diff], 0.0)
    return T

def lp_solve(T, M, n):
    """Solve: max sum(b) s.t. T @ b <= M, b >= 0.
    Returns the optimal b (normalized to sum=1).
    """
    # linprog minimizes, so we negate the objective
    c = -np.ones(n)
    A_ub = T
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    b = np.maximum(res.x, 0.0)
    s = b.sum()
    if s < 1e-15:
        return None
    return b / s
