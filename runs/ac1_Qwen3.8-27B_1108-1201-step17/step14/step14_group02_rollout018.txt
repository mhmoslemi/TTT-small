    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T

def subgradient_at_lag(a, k):
    """Compute ∇conv(a,a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    return grad

def soft_subgradient(a, c, top_k=20, beta=3.0):
    """Compute soft subgradient using top-k autocorrelation values."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = np.maximum(c[top_idx], 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        grad += wi * subgradient_at_lag(a, int(k))
    return grad

        ac = fast_autocorr(a)
        g = soft_subgradient(a, ac, top_k=top_k, beta=beta)

        log_a_new = log_a - step * g
        a_new = np.exp(log_a_new - log_a_new.max())

        if s < 1e-15:

        if m_new < best_m:

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                m_avg = fast_autocorr(avg_check).max()
                if m_avg < best_m:
                    best_m = m_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 2000 == 0:

            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
