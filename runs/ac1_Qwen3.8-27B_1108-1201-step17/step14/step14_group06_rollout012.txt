    best_a = a.copy()
    best_score = fast_autocorr_vec(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Beta schedule
    betas = np.geomspace(beta_start, beta_end, n_betas)

    # AdaGrad accumulator
    adagrad_acc = np.zeros(n)
    adagrad_lr = 0.5

    total_iters = 0
    stagnation = 0

    for bi, beta in enumerate(betas):
        beta_time = 180.0  # seconds per beta level (will be cut by time_limit)
        local_t0 = time.monotonic()
        local_iters = 0
        local_best_score = best_score
        local_stag = 0

        while time.monotonic() - local_t0 < beta_time:
            if time.monotonic() - t0 > time_limit:
                break

            b = fast_autocorr_vec(a)
            max_b = b.max()

            # Softmax weights
            shifted = b - max_b
            exp_vals = np.exp(beta * shifted)
            w = exp_vals / exp_vals.sum()

            # Compute subgradient
            g = weighted_subgradient(a, w)

            # AdaGrad step size
            adagrad_acc += g * g
            eta_t = adagrad_lr / (np.sqrt(adagrad_acc) + 1e-10)

            # Multiplicative update
            # a_new = a * exp(-eta * g / max(g)) for stability
            g_max = g.max()
            if g_max < 1e-15:
                break
            scaled_g = g / g_max
            a = a * np.exp(-eta_t * scaled_g)
            a = np.maximum(a, 1e-30)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
                s = 1.0
            a = a / s

            local_iters += 1
            total_iters += 1

            # Check score every few iterations
            if local_iters % 50 == 0:
                score = fast_autocorr_vec(a).max()
                if score < local_best_score:
                    local_best_score = score
                    local_stag = 0
                else:
                    local_stag += 1

        # End of beta level
        final_score = fast_autocorr_vec(a).max()
        if final_score < best_score:
            best_score = final_score
            best_a = a.copy()
            if verbose:
                print(f"    SGD beta={beta:.1f} ({local_iters} iters): score={best_score:.10f}")
        elif verbose:
            print(f"    SGD beta={beta:.1f} ({local_iters} iters): no improve ({final_score:.10f})")

        # If stagnating at this beta, try a random perturbation
        if local_stag > 60:
            pert = np.ones(n) * rng.uniform(0.05, 0.15)
            a = a * (1 + pert)
            a = np.maximum(a, 1e-30)
            a = a / a.sum()

    return best_a, best_score


def build_constraint_matrix(a, n, peak_indices=None):
    """Build constraint matrix T where T[k,j] = a[k-j] if valid, else 0."""
    size = 2 * n - 1
    if peak_indices is None:
        peak_indices = np.arange(size)
    T = np.zeros((len(peak_indices), n))
    for idx, k in enumerate(peak_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T[idx, js:je+1] = a[k-j: k-js+1 if k-js >= 0 else None]
            # More carefully:
            T[idx, js:je+1] = a[np.arange(k-je, k-js+1)]
    return T


def lp_best_response_sparse(a, M, solver, top_k=150):
    """LP best response using only top_k peak constraints."""
    n = len(a)
    b = fast_autocorr_vec(a)
    peak_indices = np.argsort(b)[::-1][:top_k]

    T = build_constraint_matrix(a, n, peak_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None
    return b_val / sv


def chebyshev_center_sparse(a, solver, top_k=150):
    """Find the probability vector b that minimizes max peak."""
    n = len(a)
    b = fast_autocorr_vec(a)
    peak_indices = np.argsort(b)[::-1][:top_k]

    T = build_constraint_matrix(a, n, peak_indices)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None, None
    b_val = b_val / sv
    t_val = float(t_var.value)
    return b_val, t_val


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True, top_k=150,
                          use_cheby=False):
    """LP best-response with Nesterov acceleration and progressive M."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr_vec(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr_vec(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Progressive M target
        if use_cheby:
            b_cheby, t_cheby = chebyshev_center_sparse(y, solver, top_k)
            if b_cheby is not None:
                M_used = t_cheby
                g = b_cheby
            else:
                eps = min(0.05, 0.01 * np.sqrt(t + 1))
                M_used = M_curr * (1.0 - eps)
                g = lp_best_response_sparse(y, M_used, solver, top_k)
        else:
            eps = min(0.05, 0.01 * np.sqrt(t + 1))
            M_used = M_curr * (1.0 - eps)
            g = lp_best_response_sparse(y, M_used, solver, top_k)
            if g is None:
                M_used = M_curr * (1.0 - eps * 0.5)
                g = lp_best_response_sparse(y, M_used, solver, top_k)

        if g is None:
            stall_count += 5
            if stall_count > 20:
                break
            continue

        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        M_g = fast_autocorr_vec(g).max()
        if M_curr > 0 and M_g > 0:
            ratio = min(M_g / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.05), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        # Nesterov extrapolation
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr_vec(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
            stall_count = 0

        score_a = fast_autocorr_vec(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
            stall_count = 0

        if time.monotonic() - last_improve > 90.0:
            stall_count += 1
            if stall_count > 10:
                break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr_vec(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr_vec(a)
        k_peak = int(np.argmax(b))

        # Find top contributing indices to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:20]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2+1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:20]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-12:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-12:
                        continue
                    a_trial = a_trial / s_t
                    score = fast_autocorr_vec(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random moves
            for _ in range(100):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.01, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-12:
                    continue
                a_trial = a_trial / s_t
                score = fast_autocorr_vec(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break
            # Perturb to escape
            if stall % 50 == 0:
                a = a * (1 + rng.normal(0, 0.05, n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Flat with small perturbation
        a = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
    else:
        # Periodic structure
        a = 1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(2, 8) * np.arange(n) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def local_score(a):
    """Compute the objective without calling evaluate_sequence (for speed during optimization)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr_vec(a)
    max_b = b.max()
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining softmax subgradient mirror descent, LP best-response,
    Chebyshev center guidance, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_score(parent)
            print(f"Parent: n={len(parent)}, local_score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = local_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def attempt_update(candidate_a, label):
        nonlocal best_seq, best_score
        if candidate_a is None:
            return
        a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Softmax subgradient mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Softmax SGD n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.30
    if t1 > 30:
        a_init = best_seq.copy()
        a_init = a_init / a_init.sum()
        a_out, _ = softmax_subgradient_descent(
            a_init, t1, beta_start=3.0, beta_end=80.0,
            eta_start=0.8, n_betas=25, verbose=True)
        attempt_update(a_out, f"sgd_n{n_base}")

    # === Phase 2: LP Nesterov with sparse constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-Nest sparse n={n_best} (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.25
    if t2 > 60:
        a_init = best_seq / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_init, t2, solver, verbose=True,
                                         top_k=120, use_cheby=False)
        attempt_update(a_out, f"lp_n{n_best}")

    # === Phase 3: More SGD with higher beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: SGD high-beta n={n_best} (budget={remaining:.0f}s) ===")
    t3 = remaining * 0.25
    if t3 > 30:
        a_init = best_seq / best_seq.sum()
        a_out, _ = softmax_subgradient_descent(
            a_init, t3, beta_start=40.0, beta_end=300.0,
            eta_start=0.5, n_betas=15, verbose=True)
        attempt_update(a_out, f"sgd_hi_n{n_best}")

    # === Phase 4: LP with Chebyshev center target ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP+Cheby n={n_best} (budget={remaining:.0f}s) ===")
    t4 = remaining * 0.15
    if t4 > 60:
        a_init = best_seq / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_init, t4, solver, verbose=True,
                                         top_k=100, use_cheby=True)
        attempt_update(a_out, f"lpcheby_n{n_best}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = remaining * 0.10
    if t5 > 20:
        a_init = best_seq / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_init, t5, seed=42)
        attempt_update(a_out, f"coord1_n{n_best}")

    # === Phase 6: Final SGD + coord alternation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        half = remaining / 2.0
        a_init = best_seq / best_seq.sum()
        a_out, _ = softmax_subgradient_descent(
            a_init, half, beta_start=100.0, beta_end=500.0,
            eta_start=0.3, n_betas=8, verbose=False)
        attempt_update(a_out, "sgd_final")

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            a_init = best_seq / best_seq.sum()
            a_out, _ = targeted_coord_descent(a_init, remaining2, seed=99999)
            attempt_update(a_out, "coord_final")

    # === Phase 7: Try a different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_alt = n_best + 200
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_out, _ = softmax_subgradient_descent(
            init_alt, min(remaining, 200), beta_start=5.0, beta_end=60.0,
            eta_start=0.7, n_betas=10, verbose=False)
        attempt_update(a_out, f"alt_n{n_alt}")

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 60:
            a_init = best_seq / best_seq.sum()
            a_out, _ = lp_nesterov_optimize(a_init, remaining2, solver, verbose=False,
                                             top_k=100, use_cheby=False)
            attempt_update(a_out, f"lp_final_n{len(a_init)}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
