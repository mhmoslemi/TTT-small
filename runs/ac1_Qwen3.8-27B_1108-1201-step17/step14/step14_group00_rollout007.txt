import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_onto_simplex(b, target_sum):
    """Project b onto {x >= 0, sum(x) = target_sum}"""
    n = len(b)
    # Water-filling
    b_sorted = np.sort(b)[::-1]
    cumsum = np.cumsum(b_sorted)
    rho = np.where(b_sorted > (cumsum - target_sum) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    tau = (cumsum[rho] - target_sum) / (rho + 1)
    return np.maximum(b - tau, 0)

def subgradient_optimize(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Momentum
    v = np.zeros(n)
    beta = 0.9
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        k_star = int(np.argmax(c))
        M = c[k_star]
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            g = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                g += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                g += a[j2]
            grad[i] = 2.0 * g

        # f(a) - f* where f* is a lower bound
        # Lower bound: 2n/(2n-1) * norm^2 / norm^2 = 2n/(2n-1) (in normalized form)
        # In our normalization (sum = norm), the lower bound for max autocorr is norm^2/(2n-1)
        lb = norm * norm / (2 * n - 1)
        f_gap = M - lb
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:

        alpha = f_gap / grad_norm_sq

def project_onto_simplex(b, target_sum):
    n = len(b)
    b = np.asarray(b, dtype=np.float64).copy()

        return np.ones(n) / n

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def perturb_sequence(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
