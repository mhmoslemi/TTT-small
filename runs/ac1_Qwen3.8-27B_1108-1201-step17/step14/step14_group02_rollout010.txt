    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    
    x = a.copy()
    y = a.copy()
    t_momentum = 1.0
    
    t0 = time.monotonic()
    step_base = 1.0 / (2.0 * n)  # Lipschitz constant estimate for the gradient
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        b_y = fast_autocorr(y)
        k_max = min(top_k, len(b_y))
        top_idx = np.argsort(b_y)[::-1][:k_max]
        top_b = np.maximum(b_y[top_idx], 0)

        w_full = np.zeros(2*n-1)
        w_full[top_idx] = w_sparse

        step = step_base / (1.0 + beta)  # soften the step

        conv_result = np.convolve(y, v_flip, mode='full')
        grad = 2.0 * conv_result[n-1:2*n-1][::-1]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with soft-max smoothing (primary new method)
2. LP-based Frank-Wolfe with Nesterov acceleration
3. Aggressive coordinate pair optimization
4. Multi-length scaling (1600 -> 2400 -> 3200)
5. Spectral analysis for initialization

import numpy as np
import time
from scipy.fft import rfft, irfft

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False

def fast_conv_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def get_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    b = fast_conv_autocorr(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize_nonneg(a):
    """Clamp to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def fista_soft_max(a0, time_limit, beta=5.0, verbose=True):
    """
    FISTA optimization of soft-max approximation of max(conv(a,a))
    on the probability simplex.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_obj = get_objective(a)
    
    x = a.copy()
    y = a.copy()
    t_mom = 1.0
    
    # Step size: conservative based on Lipschitz estimate
    step = 1.0 / (2.0 + 4.0 * beta)
    
    t0 = time.monotonic()
    len_b = 2 * n - 1
    
    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 5000 == 0 and time.monotonic() - t0 > time_limit:
            break
        
        # Compute soft-max weights at y
        b_y = fast_conv_autocorr(y)
        # For numerical stability, center b_y
        b_centered = b_y - b_y.max()
        max_b = b_y.max()
        
        # Softmax
        exp_b = np.exp(beta * b_centered)
        w = exp_b / exp_b.sum()
        
        # Compute gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
        # Using FFT: grad[i] = 2 * conv(flipped_w, y)[len_b-1-i]
        w_flip = w[::-1]
        # conv(y, w_flip) has length n + len_b - 1 = 3n - 3
        # We want positions [n-1, ..., 2n-2] reversed
        # Use FFT for speed
        fft_size = 1 << (n + len_b - 1).bit_length()
        Y = rfft(y, fft_size)
        Wf = rfft(w_flip, fft_size)
        conv_res = irfft(Y * Wf, fft_size)[:n + len_b - 1]
        grad = 2.0 * conv_res[n - 1 : 2 * n - 1][::-1]
        
        # FISTA update
        z = y - step * grad
        x_new = project_simplex(z)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y_new = normalize_nonneg(y_new)
        
        x = x_new
        y = y_new
        t_mom = t_new
        
        # Track best
        if it % 20 == 0:
            obj_x = get_objective(x)
            if obj_x < best_obj:
                best_obj = obj_x
                best_a = x.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it} ({elapsed:.1f}s): obj={best_obj:.8f}")
    
    # Final check
    obj_final = get_objective(x)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = x.copy()
    
    return best_a, best_obj
