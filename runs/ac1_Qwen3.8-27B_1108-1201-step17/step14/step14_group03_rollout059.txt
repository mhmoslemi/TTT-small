Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed Nesterov with continuation (log-sum-exp smoothing of max autocorrelation)

3. Fast pair coordinate descent with incremental autocorrelation updates

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad_soft(a, T):
    """Compute gradient of (1/T)*logsumexp(conv(a)/T)"""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    max_c = conv.max()
    w = np.exp((conv - max_c) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(size) / size
    else:
        w /= w_sum
    # grad[i] = (2/T) * sum_j w[i+j] * a[j] (cross-correlation)
    # Use FFT: cross = irfft(FFT(w) * conj(FFT(a)))
    fft_size = 1 << (size + n - 1 - 1).bit_length()  # Need size + n - 1 for linear cross-corr
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    grad = (2.0 / T) * cross
    return grad, conv, w

def project_nonneg_sum(x, target_sum):
    """Project x onto {y >= 0, sum(y) = target_sum}"""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    # Find lambda such that sum(max(x - lambda, 0)) = target_sum
    # max(x - lambda, 0) is decreasing in lambda
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.maximum(x - mid, 0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    result = np.maximum(x - lam, 0.0)
    # Correct for rounding
    s = result.sum()
    if s > 0:
        result = result * target_sum / s
    return result

def smoothed_nesterov(a0, time_limit, T_start=5.0, T_end=0.05, n_total_iters=50000, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov state
    x = a.copy()  # Current iterate
    y = a.copy()  # Nesterov extrapolation point
    t_nest = 1.0
    
    # Temperature schedule
    n_temps = 10
    temps = np.geomspace(T_start, T_end, n_temps)
    
    total_iters = 0
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Reset Nesterov state for new temperature (or continue)
        y = x.copy()
        t_nest = 1.0
        
        # Estimate step size: L ~ n / T^2 (conservative)
        # Actually, let's use L = 2 * n / T (empirical)
        L = 4.0 * n / T
        step = 1.0 / L
        
        # Number of iterations at this temperature
        iters_this_T = max(200, n_total_iters // n_temps)
        
        for it in range(iters_this_T):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient at y
            grad, conv_y, w_y = compute_grad_soft(y, T)
            
            # Gradient norm for monitoring
            # (skip for speed)
            
            # Nesterov step
            x_old = x.copy()
            step_size = step / (1.0 + 0.01 * it)  # Slight decay
            x_new = x - step_size * grad
            x_new = project_nonneg_sum(x_new, norm)
            
            # Nesterov momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            momentum = (t_nest - 1.0) / t_new
            y_new = x_new + momentum * (x_new - x_old)
            y_new = project_nonneg_sum(y_new, norm)
            
            t_nest = t_new
            x = x_new
            y = y_new
            total_iters += 1
            
            # Check improvement (every 10 iterations)
            if total_iters % 10 == 0:
                score = fast_autocorr(x).max()
                if score < best_score:
                    best_score = score
                    best_a = x.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            score_x = fast_autocorr(x).max()
            print(f"  Smooth T={T:.3f} ({elapsed:.0f}s): x_score={score_x:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

The second term (1/T) * sum_k p_k * H_{g_k} has spectral norm <= (1/T) * 2 = 2/T.

The first term: J_g = ∇g = ∇conv/T. Each ∇conv[k] has entries that are 2*a[|k-i|] (bounded by 2*max(a)). So ||J_g||_F^2 <= M * (2*max(a)/T)^2 * n (roughly).

Actually, a simpler approach: since we're using Nesterov with projection, and the problem is convex, we can use the fact that the function is self-concordant-like and use a step size of order T/n (which is conservative but safe).

Or even simpler: use an adaptive step size with backtracking.

Let me just use step = T / n as a starting point and adjust empirically.

Actually, you know what, let me just try L = 2*n/T which is more conservative (since the Hessian of the full logsumexp involves cross-terms). With n=1600 and T=1, L ≈ 3200, step ≈ 0.0003. That might be too small.

Let me try a different approach: use the smoothed gradient but with a "fused" step size that adapts.

OK, I think I'm overcomplicating this. Let me just implement it with a backtracking line search and see how it goes. The key advantage of the smoothed approach is that each iteration is O(n log n) regardless of the step size.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_soft_grad(a, T):
    """Compute gradient of (1/T)*logsumexp(conv(a)/T) using FFT.
    Returns (grad, conv, weights)."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    max_c = conv.max()
    exp_conv = np.exp((conv - max_c) / T)
    w_sum = exp_conv.sum()
    if w_sum < 1e-300:
        w = np.ones(size) / size
    else:
        w = exp_conv / w_sum
    # grad[i] = (2/T) * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = (2/T) * cross_correlation(w, a)[i]
    # cross_correlation(w, a)[i] = sum_j w[i+j] * a[j]
    # = irfft(FFT(w) * conj(FFT(a)))[i] with proper zero-padding
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    grad = (2.0 / T) * cross
    return grad, conv, w

def project_nonneg_sum(x, target_sum):
    """Project x onto {y >= 0, sum(y) = target_sum} using bisection."""
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.maximum(x - mid, 0.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.maximum(x - lam, 0.0)
    s = result.sum()
    if s > 1e-15:
        result *= target_sum / s
    return result

def smoothed_nesterov(a0, time_limit, T_start=5.0, T_end=0.05, verbose=True):
    """Smoothed Nesterov with continuation on temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    y = a.copy()
    t_nest = 1.0
    total_iters = 0
    
    # Temperature schedule (geometric)
    n_temps = 12
    temps = np.geomspace(T_start, T_end, n_temps)
    
    for Ti, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Reset Nesterov for new temperature
        if Ti > 0:
            y = x.copy()
            t_nest = 1.0
        
        # Step size: L ≈ 4/T (empirical, may need tuning)
        step = T / 4.0
        # Cap step to avoid huge jumps
        step = min(step, norm / (2.0 * n))

        # More iterations for smaller T (harder problem)
        iters_this = max(500, int(2000 * (T_start / T) ** 0.3))
        iters_this = min(iters_this, 20000)
        
        stagnation = 0
        prev_score = best_score
        
        for it in range(iters_this):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Gradient at y
            grad, conv_y, w_y = compute_soft_grad(y, T)
            
            # Step
            x_old = x.copy()
            x_new = x - step * grad
            x_new = project_nonneg_sum(x_new, norm)
            
            # Nesterov momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            momentum = (t_nest - 1.0) / t_new
            y_new = x_new + momentum * (x_new - x_old)
            y_new = project_nonneg_sum(y_new, norm)
            
            t_nest = t_new
            x = x_new
            y = y_new
            total_iters += 1
            
            # Check improvement periodically
            if total_iters % 20 == 0:
                score = fast_autocorr(x).max()
                if score < best_score:
                    best_score = score
                    best_a = x.copy()
                    stagnation = 0
                else:
                    stagnation += 1
                if stagnation > 50:
                    break
        
        if verbose:
            elapsed = time.monotonic() - t0
            score_x = fast_autocorr(x).max()
            print(f"  Smooth T={T:.4f} ({elapsed:.0f}s, iter={total_iters}): x={score_x:.8f}, best={best_score:.8f}")
    
    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve LP: maximize sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    # Build constraint matrix T (size x n)
    # T[k, j] = a[k-j] if 0<=k-j<n else a[j-k] if 0<=j-k<n else 0
    # Actually T[k, j] is the coefficient of b[j] in (a*b)[k]
    # (a*b)[k] = sum_j a[k-j] * b[j] (valid j) + ... 
    # Wait, conv(a,b)[k] = sum_j a[j] * b[k-j] = sum_j a[k-j] * b[j]
    # So T[k,j] = a[k-j] if 0<=k-j<n and 0<=j<n, else 0
    
    T_mat = np.zeros((size, n))
    for j in range(n):
        # (a*b)[k] includes a[k-j]*b[j] for k in [j, j+n-1]
        k_lo = j
        k_hi = min(size, j + n)
        T_mat[k_lo:k_hi, j] = a[:k_hi - j]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
