def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def lse_grad_fft(y, T):
    """Compute gradient of log-sum-exp smoothed max of autocorrelation via FFT."""
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    Y = rfft(y, fft_size)
    cross = irfft(np.conj(Y) * W, fft_size)
    grad = 2.0 * cross[:n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        a = a * norm / s
    y = a.copy()

    t0 = time.monotonic()
    L_init = n / 20.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 6.0), n / (T * 1.2))
        t_fista = 1
        staging = a.copy()

def lp_best_response_full(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_best_response_reduced(a, M, solver, num_active=500):
    """Solve LP with only the most active (tightest) constraints for speed."""
    n = len(a)
    b = fast_autocorr(a)
    # Find the tightest constraints (smallest margins)
    margins = M - b
    # Take the top num_active tightest
    k = min(num_active, 2 * n - 1)
    active_idx = np.argsort(margins)[:k]
    b_var = cp.Variable(n, nonneg=True)
    T_sub = np.zeros((k, n))
    for i, j in enumerate(active_idx):
        js = max(0, j - n + 1)
        je = min(n - 1, j)
        T_sub[i, js:je + 1] = a[j - je: j - js + 1][::-1]
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True, use_reduced=True, num_active=800):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        if use_reduced:
            b = lp_best_response_reduced(y, M_curr, solver, num_active)
        else:
            b = lp_best_response_full(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()

                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1

            s_avg = avg_check.sum()

                score_avg = fast_autocorr(avg_check).max()

                    best_a = avg_check.copy()

            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

def fft_subgradient_nesterov(a0, time_limit, verbose=False):
    """Fast FFT-based min-max subgradient with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    # FFT size for subgradient computation
    fft_size = 1 << (3 * n - 2).bit_length()

    # Adaptive step size
    base_L = norm * 4.0
    step = 1.0 / base_L

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M = ac.max()
        if M < 1e-14:
            break

        # Find near-active set
        eps = 0.01 * M
        S = (ac >= M - eps).astype(np.float64)
        S_sum = S.sum()
        if S_sum < 1:
            S[np.argmax(ac)] = 1.0
            S_sum = 1.0
        S = S / S_sum

        # Compute subgradient via FFT
        S_pad = np.zeros(fft_size)
        size_ac = 2 * n - 1
        S_pad[:min(size_ac, fft_size)] = S
        Y = rfft(y, fft_size)
        S_fft = rfft(S_pad)
        cross = irfft(np.conj(Y) * S_fft, fft_size)
        grad = 2.0 * cross[:n]

        # Update with Nesterov
        alpha_t = 2.0 / (t + 2.0)
        step_t = step * (1.0 + 0.5 / np.sqrt(t + 1.0))

        y_new = y - step_t * grad
        y_new = project_simplex(y_new, norm)

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new, norm)

        y = y_new
        a = a_new
        n_iters += 1

        if n_iters % 100 == 0:
            sc = fast_autocorr(y).max()
            if sc < best_score:
                best_score = sc
                best_a = y.copy()
            sc_a = fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            sc_avg = fast_autocorr(avg_check).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check * norm / avg_check.sum()

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FFT-Subgrad iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.0003, 0.0007, 0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_perturb = rng.integers(2, 6)
            for _ in range(n_perturb):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.001, 0.015)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break
            if stall > 500:
                break

    return best_a, best_score


def smooth_sequence(a, sigma_frac=0.5):
    """Apply mild Gaussian smoothing to the sequence."""
    n = len(a)
    sigma = max(1.0, sigma_frac)
    kernel_size = int(2 * np.ceil(3 * sigma)) | 1
    if kernel_size >= n:
        return a.copy()
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    # Apply circular convolution then take first n
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    smoothed = np.convolve(a_pad, kernel, mode='same')[:n]
    smoothed = np.maximum(smoothed, 0.0)
    s = smoothed.sum()
    if s < 1e-15:
        return a.copy()
    return smoothed * (a.sum() / s)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining LSE-FISTA, LP-Nesterov (full and reduced),
    FFT subgradient, and targeted peak shaving with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [30.0, 8.0, 2.0, 0.6, 0.2, 0.08]
        a_out, _ = lse_fista(best_seq, T_sched, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP Nesterov with reduced constraints (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-Nesterov (reduced) n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t2, solver, verbose=True, use_reduced=True, num_active=600)
        try_update(a_out, f"lp_red_n{n_best}")

    # === Phase 3: FFT subgradient (many fast iterations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20:
        print(f"\n=== Phase 3: FFT-Subgradient n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fft_subgradient_nesterov(a_norm, t3, verbose=True)
        try_update(a_out, f"fftsub_n{n_best}")

    # === Phase 4: LP Nesterov full (polish) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP-Nesterov (full) n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t4, solver, verbose=True, use_reduced=False)
        try_update(a_out, f"lp_full_n{n_best}")

    # === Phase 5: Targeted peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Targeted peak shave n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_peak_shave(a_norm, t5, seed=42)
        try_update(a_out, f"targeted_n{n_best}")

    # === Phase 6: Smoothing + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        # Try mild smoothing
        a_smooth = smooth_sequence(best_seq, sigma_frac=0.7)
        try_update(a_smooth, f"smooth_n{n_best}")
        a_smooth2 = smooth_sequence(best_seq, sigma_frac=0.4)
        try_update(a_smooth2, f"smooth2_n{n_best}")

        # LP on smoothed
        t6 = min(remaining * 0.10, 80.0)
        if t6 > 15:
            print(f"\n=== Phase 6: LP on smoothed n={n_best} (budget={t6:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t6, solver, verbose=False, use_reduced=True, num_active=400)
            try_update(a_out, f"lp6_n{n_best}")

    # === Phase 7: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = len(best_seq) + 200
        t7 = min(remaining * 0.12, 100.0)
        if t7 > 20:
            print(f"\n=== Phase 7: LP at n={n_alt} (budget={t7:.0f}s) ===")
            init_long = interpolate_sequence(best_seq, n_alt)
            a_norm = init_long * np.sqrt(2.0 * n_alt) / init_long.sum()
            a_out, _ = lp_nesterov(a_norm, t7, solver, verbose=False, use_reduced=True, num_active=600)
            try_update(a_out, f"lp_n{n_alt}")

    # === Phase 8: Final targeted shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.08, 60.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final peak shave n={n_best} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_peak_shave(a_norm, t8, seed=99999)
        try_update(a_out, "final_shave")

    # === Phase 9: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 15:
            print(f"\n=== Phase 9: Final LP n={n_best} (budget={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov(a_norm, t9, solver, verbose=False, use_reduced=True, num_active=300)
            try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
