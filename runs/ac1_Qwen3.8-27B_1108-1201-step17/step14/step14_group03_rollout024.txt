import numpy as np

import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        stuck_count = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_active(a0, time_limit, verbose=True, max_passes_limit=5000):
    """Fast coordinate descent with active-set 1D minimax."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    conv = fast_autocorr(a)
    best_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0
    range_shrink = 1.0

    while time.monotonic() - t0 < time_limit and n_passes < max_passes_limit:
        n_passes += 1
        pass_improved = False
        max_conv = conv.max()
        sum_a = a.sum()
        obj = 2.0 * n * max_conv / (sum_a * sum_a)

        # Active set: lags contributing to max
        threshold = max_conv * 0.995
        active_idx = np.where(conv >= threshold)[0]
        if len(active_idx) > 300:
            active_idx = np.argsort(conv)[-300:]
        elif len(active_idx) < 5:
            active_idx = np.argsort(conv)[-20:]

        active_size = len(active_idx)
        conv_act = conv[active_idx]

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]

            # p[k] at active indices: a[k-i] for k in active_idx
            p_act = np.zeros(active_size)
            for idx_j, k_val in enumerate(active_idx):
                j = k_val - i
                if 0 <= j < n:
                    p_act[idx_j] = a[j]

            # q at active indices
            q_act = np.zeros(active_size)
            q_full = 2 * i
            if q_full < size:
                mask = (active_idx == q_full)
                if mask.any():
                    q_act[mask] = 1.0

            # Also need full p for conv update
            p_full = np.zeros(size)
            lo = i
            hi = min(size, i + n)
            if hi > lo:
                p_full[lo:hi] = a[lo - i:hi - i]

            if ai < norm * 1e-12:
                # Try small positive values
                d_candidates = np.array([norm * 1e-8, norm * 1e-7, norm * 1e-6, norm * 1e-5])
                best_d = d_candidates[0]
                best_val = (conv_act + 2.0 * best_d * p_act + best_d ** 2 * q_act).max()
                for d_c in d_candidates[1:]:
                    val = (conv_act + 2.0 * d_c * p_act + d_c ** 2 * q_act).max()
                    if val < best_val:
                        best_val = val
                        best_d = d_c
                d = best_d
            else:
                # Adaptive range: shrink over time for finer search
                d_lo = -ai * 0.99 * range_shrink
                d_hi = ai * 1.99 * range_shrink
                if d_hi <= d_lo:
                    d_hi = d_lo + 1e-10
                n_steps = 50
                deltas = np.linspace(d_lo, d_hi, n_steps)

                # Vectorized evaluation over active set
                incr = 2.0 * deltas[None, :] * p_act[:, None]
                incr += (deltas[None, :] ** 2) * q_act[:, None]
                new_maxes = (conv_act[:, None] + incr).max(axis=0)

                best_idx_d = np.argmin(new_maxes)
                best_d = deltas[best_idx_d]

                # Fine search
                if 0 < best_idx_d < n_steps - 1:
                    f_lo = deltas[best_idx_d - 1]
                    f_hi = deltas[best_idx_d + 1]
                    f_deltas = np.linspace(f_lo, f_hi, 20)
                    f_incr = 2.0 * f_deltas[None, :] * p_act[:, None]
                    f_incr += (f_deltas[None, :] ** 2) * q_act[:, None]
                    f_maxes = (conv_act[:, None] + f_incr).max(axis=0)
                    best_d = f_deltas[np.argmin(f_maxes)]

                d = best_d

            if abs(d) < 1e-16:
                continue

            # Apply update
            a[i] += d
            conv += 2.0 * d * p_full
            if q_full < size:
                conv[q_full] += d * d

            new_sum = sum_a + d
            if new_sum < 0.01:
                a[i] -= d
                conv -= 2.0 * d * p_full
                if q_full < size:
                    conv[q_full] -= d * d
                continue

            new_obj = 2.0 * n * conv.max() / (new_sum * new_sum)

            if new_obj < obj - 1e-15:
                sum_a = new_sum
                obj = new_obj
                pass_improved = True
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
            else:
                # Revert
                a[i] -= d
                conv -= 2.0 * d * p_full
                if q_full < size:
                    conv[q_full] -= d * d

        if pass_improved:
            no_improve = 0
            range_shrink = max(0.3, range_shrink * 0.995)
        else:
            no_improve += 1
            range_shrink = min(1.0, range_shrink * 1.1)
            if no_improve >= 5:
                break

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): obj={best_obj:.8f}")

    # Final verification with FFT
    conv_final = fast_autocorr(best_a)
    sum_final = best_a.sum()
    final_obj = 2.0 * n * conv_final.max() / (sum_final * sum_final)
    return best_a, final_obj


def lse_polish(a0, time_limit, verbose=True):
    """Log-sum-exp smoothed gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(8.0, 0.1, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            max_conv = conv.max()
            if max_conv < 1e-12:
                break
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            # Gradient computation
            A_fft = rfft(a, fft_size)
            W_fft = rfft(w, fft_size)

            # g1[i] = sum_k w[k] * a[k-i] for valid k
            g1 = np.zeros(n)
            g2 = np.zeros(n)
            for k in range(n):
                # w[k] contributes to g1[i] when k-i is valid, i.e., i in [k-n+1, k]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_hi >= i_lo:
                    g1[i_lo:i_hi + 1] += w[k] * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
                # w[k] contributes to g2[i] when k < i, a[i-k] valid
                # Actually g2[i] = sum_k w[k] * a[i-k] for i-k in [0,n-1]
                # = sum_k w[k] * a[i-k] where 0<=i-k<n => k in [i-n+1, i]
                j_lo = max(0, i_lo)
                # Let me just compute directly
            # Simpler: use FFT-based cross-correlation
            cross1 = irfft(W_fft * np.conj(A_fft), fft_size)
            cross2 = irfft(A_fft * np.conj(W_fft), fft_size)

            g1 = np.zeros(n)
            g2 = np.zeros(n)
            for i in range(n):
                # g1[i] = sum_k w[k]*a[k-i] where k-i in [0,n-1], k in [i, i+n-1]
                k_lo = i
                k_hi = min(size, i + n)
                if k_hi > k_lo:
                    g1[i] = np.dot(w[k_lo:k_hi], a[k_lo - i:k_hi - i])
                # g2[i] = sum_k w[k]*a[i-k] where i-k in [0,n-1], k in [i-n+1, i]
                k_lo2 = max(0, i - n + 1)
                k_hi2 = min(size, i + 1)
                if k_hi2 > k_lo2:
                    g2[i] = np.sum(w[k_lo2:k_hi2] * a[i - np.arange(k_lo2, k_hi2)])

            grad = 2.0 * (g1 + g2)

            step = norm / (T * n * (1.0 + 0.05 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        update_best(a_out, f"lp2_n{n_best}")

    # Phase 4: CD final (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: CD Final n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent_active(a_norm, t4, verbose=True)
        update_best(a_out, f"cd2_n{n_best}")

    # Phase 5: Perturbation restarts (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.15
    print(f"\n=== Phase 5: Perturbation restarts n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 30:
        n_restarts = max(2, min(5, int(t5 / 15)))
        per_restart = t5 / n_restarts
        for ri in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 5:
                break
            # Add random perturbation
            noise_scale = np.sqrt(2.0 * n_best) / n_best * (0.05 + 0.05 * ri)
            perturbed = best_seq.copy() * np.sqrt(2.0 * n_best) / best_seq.sum()
            perturbed += rng.normal(0, noise_scale, size=n_best)
            perturbed = np.maximum(perturbed, 0.0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed * np.sqrt(2.0 * n_best) / ps

            # Short CD on perturbed
            rt = min(per_restart * 0.6, time_budget - (time.monotonic() - t_global))
            if rt < 5:
                break
            a_out_r, _ = coord_descent_active(perturbed, rt, verbose=False)
            update_best(a_out_r, f"pert_{ri}_n{n_best}")

            # If improved, do a short LP
            remaining_r = time_budget - (time.monotonic() - t_global)
            if remaining_r > 20 and len(best_seq) == n_best:
                a_norm_r = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
                a_out_lp, _ = nesterov_lp_optimize(a_norm_r, min(remaining_r * 0.4, 60.0), solver, verbose=False)
                update_best(a_out_lp, f"pert{ri}_lp_n{n_best}")

    # Phase 6: Alternate length (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_alt = n_best + 100
        print(f"\n=== Phase 6: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out_alt, _ = nesterov_lp_optimize(a_norm_alt, remaining * 0.5, solver, verbose=False)
        update_best(a_out_alt, f"alt_n{n_alt}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20 and best_seq is not None and len(best_seq) == n_alt:
            a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
            a_out_alt2, _ = coord_descent_active(a_norm_alt2, min(remaining2 * 0.5, 40.0), verbose=False)
            update_best(a_out_alt2, f"alt_cd_n{n_alt}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
