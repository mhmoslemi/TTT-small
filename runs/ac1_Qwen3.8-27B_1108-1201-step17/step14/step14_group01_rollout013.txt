"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive constraint screening + Nesterov acceleration
2. Multiplicative weight updates (MWU) for alternative exploration
3. Pair coordinate descent for fine-tuning

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
from scipy.sparse import csr_matrix

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T_sparse(a):
    n = len(a)
    size = 2 * n - 1
    rows_list = np.add.outer(np.arange(n), np.arange(n)).flatten('F')
    cols_list = np.repeat(np.arange(n), n)
    vals_list = np.tile(a, n)
    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(size, n))

def eval_fn(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def lp_solve_full(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_T_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def lp_solve_screened(a, M, solver, active_indices):
    n = len(a)
    size = 2 * n - 1
    T = build_T_sparse(a)
    T_sub = T[active_indices, :]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1
    rng = np.random.default_rng(12345)

    # Adaptive screening fraction
    screen_frac = 0.3
    min_active = 50
    perturb_interval = 40
    since_perturb = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Adaptive constraint screening
        threshold = M * (0.6 + 0.3 * min(1.0, t / 100.0))
        active = np.where(conv_y >= threshold)[0]
        if len(active) < min_active:
            active = np.where(conv_y >= M * 0.4)[0]
        if len(active) < min_active:
            active = np.arange(size)

        # Solve LP
        if len(active) < size * screen_frac:
            b = lp_solve_screened(y, M, solver, active)
            if b is not None:
                # Verify no major violations
                check = fast_conv_cross(y, b)
                viol = check.max()
                if viol > M * 1.05:
                    b = lp_solve_full(y, M, solver)
        else:
            b = lp_solve_full(y, M, solver)

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Periodic perturbation to escape local optima
        since_perturb += 1
        if since_perturb >= perturb_interval and t > 20:
            since_perturb = 0
            # Small perturbation proportional to elapsed time (heating)
            heat = min(1.0, 5.0 / t)
            noise = rng.standard_normal(n) * 0.02 * heat
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert * norm / s_p
                sc_p = fast_conv_self(a_pert).max()
                # Accept if not too bad (simulated annealing style)
                if sc_p < best_score * 1.1 or rng.random() < 0.3:
                    a = a_pert
                    y = a_pert.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def mwu_optimize(a0, time_limit, solver, verbose=True):
    """Multiplicative Weight Update approach using LP dual information."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    eta = 0.1  # Learning rate

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fast_conv_self(a)
        M = conv_a.max()
        if M < 1e-12:
            break

        # Solve LP to get best response
        b = lp_solve_full(a, M, solver)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum

        # Compute "loss" for each coordinate using gradient of max conv
        # grad_i = 2 * sum_{k: conv[k]=M} (a[k-i] + a[i-k])
        # Approximate with soft-max weighting
        temp = 0.1 * M
        w = np.exp((conv_a - M) / temp)
        w = w / w.sum()

        # Gradient computation
        g1 = np.zeros(n)
        g2 = np.zeros(n)
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        ca = irfft(np.conj(A) * W, fft_size)
        g1 = ca[:n]
        aw = irfft(A * W, fft_size)
        g2 = aw[:n]
        grad = 2.0 * (g1 + g2)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad / max(M, 1e-10))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        # Blend with best response
        blend = min(1.0, 0.1 * t / 100.0)
        a_mix = (1.0 - blend) * a_new + blend * g
        a_mix = np.maximum(a_mix, 0.0)
        s_mix = a_mix.sum()
        if s_mix > 1e-15:
            a_mix = a_mix * norm / s_mix
        a = a_mix

        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Find good j candidates
            peak_threshold = cur_max * 0.90
            peak_lags = np.where(conv >= peak_threshold)[0]

            j_candidates = set()
            for _ in range(6):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j_struct = (kl - i) % n
                j_candidates.add(j_struct)
                j_struct2 = (kl + i) % n
                j_candidates.add(j_struct2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Find optimal delta
                n_samples = 20
                deltas = np.linspace(0.0, ai, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas:
                    if d == 0:
                        continue
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve == 3:
                n_perturb = max(5, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.05 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 50 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(30.0, 0.5, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(10, min(150, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
                for trial in range(8):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= conv.max() + 0.01 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.1f}: best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 7: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish (budget={remaining:.0f}s) ===")
    if remaining > 30 and best_seq is not None:
        t_half = remaining * 0.5
        a_out, _ = lse_gradient_descent(best_seq, min(t_half, 50.0), verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 50.0), rng, verbose=False)
            try_update(a_out, "pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 100

    return [float(x) for x in best_seq]
