    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    # grad[i] = 2 * sum_k w_k * a[k-i] (where a[k-i] is valid)
    # This is equivalent to: grad = 2 * convolve weights with a (in a specific way)
    
    # For each peak at position k, the subgradient contribution at position i is 2*w_k*a[k-i]
    # So grad[i] = 2 * sum_{k in peaks} w_k * a[k-i] (if 0 <= k-i < n)
    
    # This is a "deconvolution" type operation. Let's just do it directly for small K.
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:

        idx = np.arange(i_start, i_end)
        j = k_int - idx  # these are the indices into a
        grad[idx] += 2.0 * wi * a[j]
    return grad

from scipy.fft import rfft, irfft

_global_solver = None

def fast_autocorr(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in a]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_soft_subgrad(a, ac, top_k=40, beta=6.0):
    """Compute weighted subgradient using top-K peaks with geometric weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def mirror_descent_optimize(a0, time_limit, top_k=40, beta=6.0, verbose=False):
    """Mirror descent with averaging and adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    gamma = 1.0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        g = compute_soft_subgrad(a, ac, top_k=top_k, beta=beta)

        step = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, max_constraints=None):
    """LP-based Frank-Wolfe with line search and constraint pruning."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if max_constraints is None:
        max_constraints = min(2 * n - 1, 200)

    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < 1e-14:
            break

        # Select top constraints for the LP (pruning for speed)
        n_cons = min(max_constraints, 2 * n - 1)
        top_k_idx = np.argsort(ac_y)[::-1][:n_cons]

        # Build constraint matrix for selected constraints
        b_var = cp.Variable(n, nonneg=True)

        # For each selected constraint k, build the row: sum_j b[j] * y[k-j] <= M_curr
        T_selected = np.zeros((n_cons, n))
        for ii, k_int in enumerate(top_k_idx):
            k_int = int(k_int)
            for j in range(n):
                kj = k_int - j
                if 0 <= kj < n:
                    T_selected[ii, j] = y[kj]

        constraints = [T_selected @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum

        # Line search: try multiple step sizes
        alpha_fw = 2.0 / (t + 2.0)
        candidates = [alpha_fw, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]
        # Remove duplicates and sort
        candidates = list(set([min(c, 1.0) for c in candidates]))
        candidates.sort(reverse=True)

        best_alpha = alpha_fw
        best_trial_m = fast_autocorr(y).max()
        for al in candidates:
            if al <= 0:
                continue
            trial = (1.0 - al) * y + al * g
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            mt = fast_autocorr(trial).max()
            if mt < best_trial_m:
                best_trial_m = mt
                best_alpha = al

        alpha = best_alpha

        # Nesterov update
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def pair_swap_polish(a0, time_limit, verbose=False, max_pairs_per_pass=50):
    """Coordinate descent via pair swaps with bisection for optimal transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall_count = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall_count < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Compute contribution of each position to top peaks
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        contrib = np.zeros(n)
        for k in top_k:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            contrib[idx] += a[j]

        # Select candidate source positions (high contribution) and dest (low)
        if contrib.max() < 1e-20:
            break

        high_thresh = np.percentile(contrib, 80)
        low_thresh = np.percentile(contrib, 30)
        sources = np.where(contrib > high_thresh)[0]
        dests = np.where(contrib < low_thresh)[0]

        if len(sources) == 0:
            sources = np.argsort(contrib)[::-1][:max_pairs_per_pass]
        if len(dests) == 0:
            dests = np.argsort(contrib)[:max_pairs_per_pass]

        improved = False
        n_tried = 0

        # Try random pairs from sources and dests
        n_attempts = min(max_pairs_per_pass * 3, len(sources) * len(dests))
        for _ in range(n_attempts):
            if time.monotonic() - t0 > time_limit:
                break
            i = sources[rng.integers(len(sources))]
            j = dests[rng.integers(len(dests))]
            if i == j:
                continue
            n_tried += 1

            # Compute sensitivity vector D[k] = a[k-i] - a[k-j]
            D = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                ki = k - i
                if 0 <= ki < n:
                    val += a[ki]
                kj = k - j
                if 0 <= kj < n:
                    val -= a[kj]
                D[k] = val

            # Bisection for optimal eps in [-a[i], a[j]]
            eps_lo = -a[i]
            eps_hi = a[j]
            if eps_hi - eps_lo < 1e-15:
                continue

            # Find optimal eps via ternary search (f is approximately unimodal)
            best_eps = 0.0
            best_val = M_curr
            lo = eps_lo
            hi = eps_hi

            for _bis in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                # Evaluate at m1
                val1 = np.max(ac + 2.0 * m1 * D)
                val2 = np.max(ac + 2.0 * m2 * D)

                if val1 < val2:
                    hi = m2
                else:
                    lo = m1

                # Track best
                for ev in [m1, m2]:
                    v = np.max(ac + 2.0 * ev * D)
                    if v < best_val:
                        best_val = v
                        best_eps = ev

            # Apply if improved
            if best_val < best_m - 1e-14:
                a[i] += best_eps
                a[j] -= best_eps
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a = best_a.copy()
                    break
                a = a / s
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    stall_count = 0
                    ac = fast_autocorr(a)

        if not improved:
            stall_count += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall_count = 0
                else:
                    a = best_a.copy()

        if verbose and stall_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CS stall={stall_count} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def sine_init(n, alpha=2.0):
    """Create a sine-based initialization."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** alpha
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def raised_cosine_init(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s
