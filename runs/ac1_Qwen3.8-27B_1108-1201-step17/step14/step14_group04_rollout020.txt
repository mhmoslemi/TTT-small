def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):

def lp_best_response(y, M, n, solver):
    b_var = cv.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = y
    constraints = [T_mat @ b_var <= M]
    prob = cv.Problem(cv.Maximize(cv.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_polyak(a0, time_limit, solver, upper_bound=None, verbose=True):
    """LP-Nesterov with Polyak adaptive step size and exponential averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    best_J = 2.0 * n * best_score / (norm ** 2)

    y = a.copy()
    exp_avg = a.copy()
    exp_w = 0.98
    exp_avg_curr = exp_avg * exp_w
    w_sum = exp_w
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    last_improve = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, n, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Polyak step size
        M_y = fast_autocorr(y).max()
        gap = M_y - M_curr  # gap in objective (approximate)
        diff = y - g
        diff_norm_sq = np.dot(diff, diff)

        if upper_bound is not None and diff_norm_sq > 1e-20:
            alpha = min(1.0, max(0.01, gap / (2.0 * diff_norm_sq / norm ** 2)))
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Exponential average
        exp_avg_curr = exp_w * exp_avg_curr + (1 - exp_w) * a
        w_sum = exp_w * w_sum + (1 - exp_w)
        exp_avg = exp_avg_curr / w_sum

        # Check all candidates
        for candidate in [y, a, exp_avg]:
            s_c = candidate.sum()
            if s_c < 1e-15:
                continue
            cand_norm = candidate * norm / s_c
            sc = fast_autocorr(cand_norm).max()
            J_c = 2.0 * n * sc / (norm ** 2)
            if J_c < best_J:
                best_J = J_c
                best_score = sc
                best_a = cand_norm.copy()
                last_improve = n_iters

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Polyak iter={n_iters} ({elapsed:.0f}s): J={best_J:.8f} M={M_curr:.6f} alpha={alpha:.4f}")

        # Stall detection
        if n_iters - last_improve > 500:
            stall_count += 1
            if stall_count > 5:
                # Perturb and restart momentum
                n_iters = max(1, n_iters // 2)
                stall_count = 0
                if verbose:
                    print(f"    Restart at iter {n_iters}")

    return best_a, best_J


def lse_fista_adaptive(a0, time_limit, verbose=True):
    """LSE-FISTA with adaptive temperature and backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    best_J = 2.0 * n * best_score / (norm ** 2)
    t0 = time.monotonic()

    # Temperature schedule: geometric decrease
    T_start = 10.0
    T_end = 0.01
    n_phases = 8
    T_schedule = np.geomspace(T_start, T_end, n_phases)

    for phase, T in enumerate(T_schedule):
        phase_time = time_limit / n_phases
        phase_start = time.monotonic()
        L = n / (T * 0.5)  # Lipschitz estimate
        backtracks = 0
        t_fista = 1
        staging = a.copy()

        while time.monotonic() - phase_start < phase_time:
            # Compute LSE gradient
            ac = fast_autocorr(y)
            ac_max = ac.max()
            if ac_max < 1e-15:
                break
            shifted = (ac - ac_max) / T
            shifted = np.clip(shifted, -50.0, 50.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()

            # Gradient of LSE: weighted sum of subgradients
            N_fft = 1 << (3 * n - 2).bit_length()
            W = rfft(w, N_fft)
            Y_rev = rfft(y[::-1], N_fft)
            cross = irfft(W * Y_rev, N_fft)
            grad = 2.0 * cross[n - 1: n - 1 + n] / T

            step = 1.0 / L

            # Backtracking
            z = y - step * grad
            for bt in range(10):
                a_new = project_simplex(z, 1.0)
                # Check sufficient decrease (simplified)
                if bt < 3:
                    break
                step *= 0.5
                z = y - step * grad
            backtracks += bt

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if t_fista % 100 == 0:
                sc = fast_autocorr(a).max()
                J_c = 2.0 * n * sc / (norm ** 2)
                if J_c < best_J:
                    best_J = J_c
                    best_score = sc
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA phase={phase} T={T:.3f} ({elapsed:.0f}s): J={best_J:.8f}")

    return best_a, best_J


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: move mass between pairs to reduce peak autocorrelation."""
    rng = np.random.default_rng(42)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    best_J = 2.0 * n * best_score / (norm ** 2)
    t0 = time.monotonic()
    stall = 0

    # Precompute which indices contribute to which lags
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        sorted_lags = np.argsort(b)[::-1]
        top_lags = sorted_lags[:min(10, len(sorted_lags))]

        improved = False
        for k in top_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find a good destination: low contribution to all top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_lags[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue
            # Pick from low-value candidates
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_try = min(5, len(candidates_sorted))
            dst_candidates = candidates_sorted[:n_try]

            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    J_t = 2.0 * n * sc / (norm ** 2)
                    if J_t < best_J - 1e-12:
                        a = a_trial
                        best_J = J_t
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    J_t = 2.0 * n * sc / (norm ** 2)
                    if J_t < best_J:
                        a = a_trial
                        best_J = J_t
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            if not improved:
                stall += 1
                if stall > 200:
                    break

    return best_a, best_J


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Try diverse initializations ===
    if parent is None:
        n_base = 1600
        inits = [
            make_init_linear_ramp(n_base),
            make_init_exp_decay(n_base, 2.0),
            make_init_exp_decay(n_base, 4.0),
            make_init_two_bump(n_base, 0.7),
            make_init_bell_asym(n_base, 2.0),
            make_init_sinc_like(n_base, 3.0),
        ]
        best_init = None
        best_init_score = float('inf')
        for i, init in enumerate(inits):
            sc = evaluate_safe(init)
            print(f"  Init {i}: score={sc:.6f}")
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init.copy()
        best_score = best_init_score
        n_base = len(best_seq)
    else:
        n_base = len(best_seq)
        # Also try some fresh inits at this length
        inits = [
            make_init_linear_ramp(n_base),
            make_init_exp_decay(n_base, 3.0),
            make_init_two_bump(n_base, 0.6),
            make_init_bell_asym(n_base, 1.5),
        ]
        for i, init in enumerate(inits):
            sc = evaluate_safe(init)
            if sc < best_score:
                best_score = sc
                best_seq = init.copy()
                print(f"  Fresh init {i} better: {sc:.6f}")

    print(f"Starting n={n_base}, score={best_score:.6f}")

    # === Phase 1: LSE-FISTA warm-up (fast, gets close) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 180.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lse_fista_adaptive(a_norm, t1, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP-Nesterov with Polyak (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov-Polyak n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_polyak(a_norm, t2, solver, upper_bound=best_score, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: LSE-FISTA polish (fast, fine-tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: LSE-FISTA polish n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_fista_adaptive(a_norm, t3, verbose=True)
        try_update(a_out, f"lse2_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Coord descent n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t4, verbose=True)
        try_update(a_out, f"cd_n{n_best}")

    # === Phase 5: LP-Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP round 2 n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_polyak(a_norm, t5, solver, upper_bound=best_score, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 6: Multi-length exploration ===")
        n_current = len(best_seq)
        length_options = []
        for delta in [-200, -100, 100, 200]:
            nl = n_current + delta
            if 800 <= nl <= 3000:
                length_options.append(nl)

        t_per = remaining / max(1, len(length_options)) * 0.7
        for nl in length_options:
            if time.monotonic() - t_global > time_budget - 60:
                break
            if nl == n_current:
                continue
            t_alloc = min(t_per, remaining - (time.monotonic() - t_global))
            if t_alloc < 30:
                continue
            print(f"  Trying n={nl} (budget={t_alloc:.0f}s)")
            init_l = interpolate_seq(best_seq, nl)
            a_norm = init_l * np.sqrt(2.0 * nl) / init_l.sum()

            # Short LSE-FISTA
            t_lse = t_alloc * 0.4
            a_out, _ = lse_fista_adaptive(a_norm, t_lse, verbose=False)
            try_update(a_out, f"lse_n{nl}")

            # Short LP
            t_lp = t_alloc * 0.5
            if t_lp > 20:
                n_cur = len(best_seq)
                a_norm2 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_out2, _ = lp_nesterov_polyak(a_norm2, t_lp, solver, upper_bound=best_score, verbose=False)
                try_update(a_out2, f"lp_n{nl}")

    # === Phase 7: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.3, 100.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Final CD polish (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t7, verbose=True)
        try_update(a_out, "final_cd")

    # === Phase 8: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        t8 = min(remaining * 0.7, 120.0)
        if t8 > 20:
            print(f"\n=== Phase 8: Final LP (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lp_nesterov_polyak(a_norm, t8, solver, upper_bound=best_score, verbose=False)
            try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
