            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step with line search
        t = n_iters
        alpha_base = 2.0 / (t + 2.0)
        
        # Try a few step sizes
        best_alpha = alpha_base
        best_obj = fast_autocorr(np.maximum(y + alpha_base * (g - y), 0)).max()
        for trial in [0.5, 0.75, 1.5, 2.0]:
            trial_alpha = min(alpha_base * trial, 1.0)
            cand = np.maximum(y + trial_alpha * (g - y), 0.0)
            sc = fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial_alpha
        alpha = best_alpha

        y_new = np.maximum(y + alpha * (g - y), 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if n_iters % 200 == 0:
            stall_count += 1
            if stall_count > 30:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
            elif stall_count > 10:
                y = best_a.copy()
                a = best_a.copy()

        if verbose and n_iters % 50 == 0:
            print(f"    NLP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def subgradient_optimize(a0, time_limit, T_start=5.0, T_end=0.5, eta_start=0.1, verbose=True):
    """Subgradient method with softmax smoothing of the max."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iter = 0
    stall = 0

    T = T_start
    eta = eta_start
    best_T_score = best_score

    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        
        # Compute autocorrelation
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Softmax weights
        T_curr = max(T_end, T * (T_end / T_start) ** (n_iter / max(1, time_limit / 2)))
        exp_b = np.exp((b - max_b) / T_curr)
        w = exp_b / exp_b.sum()
        
        # Compute subgradient: grad of sum_k w[k] * b[k] w.r.t. a
        # d(b[k])/d(a[i]) = a[k-i] (if 0<=k-i<n, k!=2i) + 2*a[i] (if k=2i)
        # More efficiently: grad[i] = sum_k w[k] * a[k-i] 
        # = (w_padded * a_padded)[first n] ... actually it's the cross-correlation
        # Let me use FFT: if W = rfft(w_padded), A = rfft(a_padded), then the "cross" is irfft(W * conj(A))
        
        fft_size = 1 << (3 * n - 2).bit_length()
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        w_pad = np.zeros(fft_size)
        w_pad[:2*n-1] = w
        
        A_f = rfft(a_pad, fft_size)
        W_f = rfft(w_pad, fft_size)
        
        # cross[i] = sum_k w[k] * a[k-i] for the relevant range
        # This is the linear cross-correlation of w and a
        cross = irfft(W_f * np.conj(A_f), fft_size)[:n]
        
        # But we also need the factor of 2 for diagonal terms
        # Actually for the quadratic form, d(sum_k w[k] b[k])/d(a[i]) = 2 * (w * a)[i] (circular-ish)
        # Let me just use: grad = 4 * cross (factor of 2 for quadratic, factor of 2 for symmetry)
        # Actually let me derive it properly.
        # b[k] = sum_i a[i] a[k-i]
        # sum_k w[k] b[k] = sum_k w[k] sum_i a[i] a[k-i] = sum_i a[i] sum_k w[k] a[k-i]
        # d/d(a[i]) = sum_k w[k] a[k-i] + sum_k w[k] a[k-i] (by symmetry, since a[i] appears as both factors)
        # = 2 * sum_k w[k] a[k-i]
        # The sum_k w[k] a[k-i] is the cross-correlation at lag i
        # For linear convolution: (w * a_reversed) or something...
        
        # Let me just use the direct formula with FFT
        # sum_k w[k] * a[k-i] = (w cross-correlate a) at lag i
        # If we define (w ⊛ a)[i] = sum_k w[k] a[k+i], that's not quite right either.
        
        # Actually: sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] = (w_shifted * a)[...] 
        # This is getting confusing. Let me just compute it directly for small n or use a different formula.
        
        # Alternative: use the fact that the gradient of the "smoothed max" is:
        # grad_i = 2 * sum_k w[k] * (d b[k] / d a[i])
        # d b[k] / d a[i] = sum_j [i+j=k] a[j] = a[k-i] (if valid)
        # So grad_i = 2 * sum_k w[k] * a[k-i] (where a[k-i] = 0 if k-i not in [0,n))
        # = 2 * (full_cross_corr(w, a))[i]
        
        # The full cross-correlation can be computed as:
        # (w ⊛ a)[i] = sum_k w[k] a[k-i] = sum_j w[j+i] a[j]
        # = first n elements of (w_reversed_padded * a_padded) in FFT domain
        # Or: = irfft(FFT(w_padded) * conj(FFT(a_padded))) but with the right indexing
        
        # Let me use a simpler approach: just compute the gradient numerically or use the FFT trick
        # Actually, the correct FFT formula for sum_k w[k] a[k-i] (linear, with zero extension):
        # Let W = rfft(w, N), A = rfft(a, N) where N >= 3n-2
        # Then sum_k w[k] a[k-i] = irfft(W * conj(A)) at index i... 
        # Hmm, this is the circular cross-correlation. For linear, we need N large enough.
        
        # Let me just do it more carefully:
        # Define v[i] = sum_k w[k] a[k-i] for i = 0, ..., n-1
        # = sum_k w[k] a[k-i]
        # Let j = k-i, so k = j+i:
        # = sum_j w[j+i] a[j]  (where w[j+i] = 0 if j+i >= 2n-1 or j+i < 0)
        # = sum_j w[j+i] a[j] for j = 0, ..., n-1
        # This is the cross-correlation of w (length 2n-1) and a (length n) at lag i
        # = first n values of the cross-correlation
        
        # Using FFT: cross_corr[i] = sum_j w[j+i] a[j]
        # = sum_j w_rev[2n-2-(j+i)] a[j] ... this is getting complicated.
        
        # Simplest: just compute it as a sum using the structure
        # v[i] = sum_{j=0}^{n-1} a[j] * w[j+i]
        # This is like a "sliding dot product"
        # Can be computed as: v = the first n elements of the linear convolution of a with w_reversed
        # w_reversed[j] = w[2n-2-j]
        # (a * w_rev)[k] = sum_j a[j] w_rev[k-j] = sum_j a[j] w[2n-2-k+j]
        # Hmm, not quite.
        
        # Let me just do: v[i] = sum_j a[j] w[j+i]
        # = sum_j a[j] * w[i+j]
        # This is (a * w)[i] where * is NOT standard convolution but a "shifted correlation"
        # Actually: (a * w)[k] = sum_j a[j] w[k-j]
        # v[i] = sum_j a[j] w[i+j] = sum_j a[j] w[-(j-i)] ... no
        # v[i] = sum_j a[j] w[i+j]
        # Let m = i+j, j = m-i: = sum_m a[m-i] w[m] = (a_reversed * w)[i + n - 1] ... 
        
        # OK I'll just use the FFT approach that I know works:
        # If I define the "cross" as: cross[i] = sum_k w[k] * a[k-i] (with a extended by 0)
        # Then: cross = irfft(rfft(w, N) * conj(rfft(a, N)), N) at the right indices
        # where N >= 3n-2 to avoid circular aliasing
        
        # Testing: for a = [1, 2, 3], w = [1, 0, 0]
        # cross[0] = w[0]*a[0] + w[1]*a[1] + w[2]*a[2] = 1
        # cross[1] = w[1]*a[0] + w[2]*a[1] + ... = 0
        # cross[2] = 0
        # Using FFT: W = [1, 0, 0], A = [1, 2, 3]
        # W * conj(A) = [1, 0, 0]
        # IFFT = [1, 0, 0] ✓ for cross[0]
        # But we need cross[1] which should be 0. The IFFT gives [1,0,0] so cross[1] = 0 ✓
        
        # For a = [1, 2, 3], w = [0, 1, 0]:
        # cross[0] = 0*a[0] + 1*a[1] + 0*a[2] = 2
        # cross[1] = 0*a[0] + 1*a[0] + 0*a[1] = 1 (wait, this doesn't seem right)
        # Let me recompute: cross[i] = sum_k w[k] * a[k-i]
        # cross[0] = w[0]*a[0] + w[1]*a[1] + w[2]*a[2] = 0 + 2 + 0 = 2
        # cross[1] = w[0]*a[-1] + w[1]*a[0] + w[2]*a[1] = 0 + 1 + 0 = 1
        # cross[2] = w[0]*a[-2] + w[1]*a[-1] + w[2]*a[0] = 0 + 0 + 0 = 0
        
        # Using FFT with zero-padding to length N>=5:
        # w_padded = [0, 1, 0, 0, 0], a_padded = [1, 2, 3, 0, 0]
        # W = FFT([0,1,0,0,0]), A = FFT([1,2,3,0,0])
        # W * conj(A) = ?
        # IFFT(W * conj(A)) = cross-correlation of w and a
        # The cross-correlation (w ⊛ a)[lag] = sum_k w[k+lag] * a[k]
        # (w ⊛ a)[0] = sum w[k]a[k] = 2
        # (w ⊛ a)[1] = sum w[k+1]a[k] = w[1]a[0] + w[2]a[1] = 1
        # (w ⊛ a)[2] = w[2]a[0] + w[3]a[1] = 0
        
        # So IFFT(W * conj(A)) gives [(w⊛a)[0], (w⊛a)[1], ...] at the right indices
        # For our definition cross[i] = sum_k w[k]a[k-i] = (w⊛a)[i] (with appropriate indexing)
        # Actually (w⊛a)[i] = sum_k w[k+i]a[k] which is NOT the same as sum_k w[k]a[k-i]
        # sum_k w[k]a[k-i] = let j=k-i: sum_j w[j+i]a[j] = (w⊛a)[i] with the definition (w⊛a)[i] = sum_j w[j+i]a[j]
        # Hmm, depends on convention. Let me just verify numerically in code.
        
        # Actually, I'll just use a direct computation for the gradient since n=1600 is manageable:
        # grad[i] = 2 * sum_k w[k] * a[k-i]
        # = 2 * sum_j w[j+i] * a[j] for j=0..n-1 (with w extended by 0)
        
        # For efficiency, I'll use the FFT cross-correlation:
        # (w ⊛ a)[i] = sum_j w[j+i] a[j] = IFFT(FFT(w) * conj(FFT(a))) at index i (for linear, need padding)
        
        # With N >= 3n-2 padding, the first n values of IFFT(FFT(w_pad) * conj(FFT(a_pad))) give us (w⊛a)[0..n-1]
        
        # Wait, I need to check the convention. Let me just use scipy.signal.correlate or compute directly.
        
        # For n=1600, direct computation is O(n^2) = 2.56M operations per iteration, which is fine.
        # But let's use FFT for safety.
        
        # Let me use a different approach: compute the gradient using the "Toeplitz" structure
        # grad[i] = 2 * sum_{k=0}^{2n-2} w[k] * a[k-i] (where a[m]=0 if m<0 or m>=n)
        # = 2 * sum_{k=max(i,0)}^{min(i+n-1, 2n-2)} w[k] * a[k-i]
        
        # I'll compute this using FFT:
        # Define u = w (length 2n-1), v = a (length n)
        # We want result[i] = sum_k u[k] * v[k-i] for i = 0, ..., n-1
        # = sum_k u[k] * v[k-i]
        # Let p = k-i, so k = p+i: sum_p u[p+i] * v[p]
        # = (u_shifted_by_i) dotted with v
        # This is the cross-correlation at lag i
        
        # Using FFT: cross[i] = IFFT(FFT(u) * conj(FFT(v)))[i] (circular)
        # For linear, pad to length >= 3n-2
        
        # Let me just do it:
        N_fft = 1 << (3 * n - 2).bit_length()
        u = np.zeros(N_fft)
        u[:2*n-1] = w
        v = np.zeros(N_fft)
        v[:n] = a
        
        U = rfft(u, N_fft)
        V = rfft(v, N_fft)
        
        # cross-correlation: IFFT(FFT(u) * conj(FFT(v)))
        cross = irfft(U * np.conj(V), N_fft)[:n]
        
        # grad = 2 * cross (factor of 2 because b[k] is quadratic in a)
        grad = 2.0 * cross
        
        # But wait, we also need to account for the normalization constraint sum(a) = norm
        # The projected gradient in the simplex is: grad_proj = grad - mean(grad)
        # Actually for the constraint sum(a) = norm, the projected subgradient step is:
        # a_new = a - eta * (grad - sum(grad)/n) ... no, that's not right either
        # For the constraint sum(a) = S: the projection of (a - eta*grad) onto {a: sum(a)=S, a>=0}
        # is: a_new = max(a - eta*grad - tau, 0) for tau chosen so sum(a_new) = S
        
        # Actually, for simplicity, let's normalize after the step:
        a_candidate = a - eta * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        s_c = a_candidate.sum()
        if s_c < 1e-15:
            continue
        a_candidate = a_candidate * norm / s_c
        score = fast_autocorr(a_candidate).max()
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            stall = 0
        else:
            stall += 1
        
        # Decay eta
        eta *= 0.999
        if stall > 100:
            eta *= 0.5
            stall = 0
        if stall > 500:
            break

        if verbose and n_iter % 100 == 0:
            print(f"    Subgrad iter={n_iter} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=42):
    """Aggressive coordinate descent targeting the top peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Focus on top peaks
        top_k = min(20, 2*n-1)
        top_peaks = np.argsort(b)[::-1][:top_k]
        
        improved = False
        
        for k_peak in top_peaks[:8]:
            # Find which (i, j) pairs contribute most to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            # Sort by contribution
            top_contrib = idx[np.argsort(contrib)[::-1][:max(2, len(idx)//8)]]
            
            # For each top contributor, try moving mass to a position that doesn't affect this peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            for src in top_contrib:
                if a[src] < 1e-10:
                    continue
                # Find good destinations (not in peak mask, and with low current value)
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    continue
                # Sort by a value (prefer moving to low-value positions)
                candidates = candidates[np.argsort(a[candidates])[:10]]
                
                for dst in candidates:
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_a = a_trial.copy()
                            best_score = score
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbations when stalled
            if stall % 3 == 0:
                for _ in range(20):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.02, 0.15)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_a = a_trial.copy()
                        best_score = score
                        stall = 0
                        break
            
            if stall > 200:
                # Larger perturbation
                n_pert = max(5, n // 50)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.5, 1.5)
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-10:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.01:  # Allow slight degradation for escape
                        a = a_trial
                        stall = 0
            
            if stall > 500:
                break

        if (stall // 10) % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    
    # Phase 1: Quick LP at base length (warm up)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP warm-up n={n_base} ===")
    t1 = min(remaining * 0.15, 180.0)
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_warmup_n{n_base}")

    # Phase 2: Subgradient optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Subgradient n={n_best} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_optimize(a_norm, t2, T_start=10.0, T_end=0.5, eta_start=0.05, verbose=True)

1. Sparse LP best-response with Nesterov acceleration and adaptive line search
2. Softmax-smoothed subgradient descent (complementary to LP)
3. Aggressive peak-targeted coordinate descent for final polish
4. Multi-length exploration with interpolation transfer
5. Momentum restarts and ensemble tracking
Key improvements over previous: sparse LP matrices, dual subgradient method,
more aggressive CD with better peak targeting, and refined time allocation.

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def build_sparse_T(a):
    """Build sparse matrix T where (T@b)[k] = (a*b)[k]."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        row_vals = a[:end - i]
        for j, (r, v) in enumerate(zip(idx, row_vals)):
            rows.append(r)
            cols.append(i)
            vals.append(v)
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = build_sparse_T(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with adaptive line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0

    # Pre-build sparse T (will be updated if a changes)
    T_cache = build_sparse_T(a / a.sum())
    T_cache_sum = a.sum()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Recompute T if y has changed significantly
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            b = np.ones(n) / n

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive step size with line search
        alpha_base = 2.0 / (t + 2.0)
        best_alpha = alpha_base
        cand = np.maximum(y + alpha_base * (g - y), 0.0)
        s_cand = cand.sum()
        if s_cand > 1e-15:
            cand = cand * norm / s_cand
        best_obj = fast_autocorr(cand).max()

        for trial_mult in [0.5, 0.75, 1.5, 2.0, 3.0]:
            trial_alpha = min(alpha_base * trial_mult, 1.0)
            cand = np.maximum(y + trial_alpha * (g - y), 0.0)
            s_cand = cand.sum()
            if s_cand > 1e-15:
                cand = cand * norm / s_cand
            sc = fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial_alpha
        alpha = best_alpha

        y_new = np.maximum(y + alpha * (g - y), 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if n_iters % 300 == 0:
            stall_count += 1
            if stall_count > 25:
                if verbose:

                break
            elif stall_count > 8:
                y = best_a.copy()
                a = best_a.copy()

        if verbose and n_iters % 100 == 0:
            print(f"    NLP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, T_start=10.0, T_end=0.3, eta_start=0.05, verbose=True):
    """Softmax-smoothed subgradient descent with adaptive learning rate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iter = 0
    stall = 0
    eta = eta_start
    best_ever_score = best_score

    while time.monotonic() - t0 < time_limit:
        n_iter += 1

        b = fast_autocorr(a)
        max_b = b.max()

        # Adaptive temperature
        frac_done = min(1.0, (time.monotonic() - t0) / time_limit)
        T_curr = T_start * (T_end / T_start) ** (frac_done ** 0.7)
        T_curr = max(T_curr, T_end)

        # Softmax weights
        exp_b = np.exp((b - max_b) / T_curr)
        w = exp_b / exp_b.sum()

        # Compute gradient via cross-correlation using FFT
        fft_size = 1 << (2 * (2 * n - 1) - 1).bit_length()
        u = np.zeros(fft_size)
        u[:2 * n - 1] = w
        v = np.zeros(fft_size)
        v[:n] = a

        U = rfft(u, fft_size)
        V = rfft(v, fft_size)
        cross = irfft(U * np.conj(V), fft_size)[:n]
        grad = 2.0 * cross

        # Projected subgradient step
        a_candidate = a - eta * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        s_c = a_candidate.sum()
        if s_c < 1e-15:
            eta *= 0.5
            stall += 1
            continue
        a_candidate = a_candidate * norm / s_c
        score = fast_autocorr(a_candidate).max()

        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            stall = 0
            a = a_candidate
        else:
            stall += 1
            # Only partially update if close
            if score < best_score * 1.001:
                a = 0.5 * a + 0.5 * a_candidate

        # Adaptive eta
        eta *= 0.9995
        if stall > 50:
            eta *= 0.7
            stall = 0
        if stall > 300:
            # Restart from best
            a = best_a.copy()
            eta = eta_start * 0.5
            stall = 0
        if stall > 800:
            break

        if verbose and n_iter % 200 == 0:
            print(f"    Subgrad iter={n_iter} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score

def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)
    return obj


def lse_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        print(f"\n=== Phase 6: LP n={n_alt} (budget={min(remaining*0.15, 200):.0f}s) ===")
        t6 = min(remaining * 0.15, 200.0)
        if t6 > 20:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=False)
            do_try_update(a_out, f"lp_alt_n{n_alt}")

    # ===== Phase 7: Final coordinate descent =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, remaining - 5.0, seed=12345)
        do_try_update(a_out, "cd_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
