def lp_best_response(a, M, solver, max_inner=4, max_active=300):
    """
    Compute LP best response with constraint generation.
    Returns b that maximizes sum(b) subject to (a*b)[k] <= M[k], b >= 0.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Start with key constraints
    key_lags = [0, n-1, n//2, n//4, 3*n//4, 1, n-2, size//2]
    active_idx = sorted(set(k for k in key_lags if 0 <= k < size))
    
    b_val = None
    for inner in range(max_inner):
        b_val = solve_lp_reduced(a, M, solver, active_idx)
        if b_val is None:
            # Fallback to full LP
            b_val = solve_lp_full(a, M, solver)
            if b_val is None:
                return None
            return b_val
        
        # Check feasibility with all constraints
        T = build_conv_matrix(a)
        circ = T @ b_val
        violated = circ > M + 1e-7
        
        if not violated.any():
            break  # Feasible, we're done
        
        # Add most violated constraints
        viol_idx = np.where(violated)[0]
        if len(viol_idx) < max_active:
            active_idx = sorted(set(active_idx) | set(viol_idx.tolist()))
        else:
            # Keep existing + most violated
            new_idx = viol_idx[np.argsort(circ[viol_idx])[::-1]][:max_active - len(active_idx)]
            active_idx = sorted(set(active_idx) | set(new_idx.tolist()))
            
            if len(active_idx) > max_active:
                # Keep top max_active by violation
                all_idx = np.array(active_idx)
                circ_vals = circ[all_idx]
                keep = all_idx[np.argsort(circ_vals)[::-1][:max_active]]
                active_idx = sorted(keep.tolist())
    
    return b_val

    prev_y = a.copy()
    gamma = 1.8

        M_curr = fast_autocorr(y).max()

        if b_sum < 1e-15:
            n_iters += 1
            continue
        
        g = b * norm_target / b_sum
        n_iters += 1
        
        # Fessler-Parks step
        alpha = 2.0 / (t_iter + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)

        else:
            momentum = (t_iter + 1.0 - gamma) / (t_iter + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:

        a = a_new

def lse_adam(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with Adam optimizer."""
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_target / n
    else:
        a = a * norm_target / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t0 = time.monotonic()
    steps = 0
    
    # Temperature schedule
    M0 = best_M
    T_start = M0 * 0.1
    T_end = M0 * 0.005

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        lr = 0.01 * (1.0 - 0.5 * progress)

        max_b = b.max()

        # This is 2 * (w_conv * a_rev)[n-1 : 2n-1]... 
        # Actually: grad[j] = 2 * sum_k w[k] * a[k-j] where 0 <= k-j < n
        # = 2 * (w * a_rev_cross)[j] ... 
        # Let me think: (w * a)[k] = sum_j w[k-j] * a[j]
        # We want: grad[j] = 2 * sum_k w[k] * a[k-j]
        # Let m = k-j, so k = j+m: grad[j] = 2 * sum_m w[j+m] * a[m]
        # This is 2 * (cross-correlation of w and a at lag j)
        # = 2 * sum_m a[m] * w[j+m]
        # = 2 * (a_rev * w)[n-1-j] ... 
        # Simpler: use FFT
        # grad = 2 * ifft(fft(a) * conj(fft(w)))... 
        # Actually the simplest: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m] (with w out of range = 0)
        # This is the cross-correlation of a and w.
        # iff we compute fft of a (padded) and fft of w (padded), then:
        # cross[j] = ifft(fft(a) * conj(fft(w)))[j]
        # But we need to be careful about the exact definition.
        
        # Let me just do it directly with the structure:
        # grad[j] = 2 * sum_{m: 0<=j+m<2n-1} a[m] * w[j+m]
        # = 2 * (a * w_full)[n-1-j] where w_full is w zero-padded to length 2n-1
        # Actually: sum_m a[m] * w[j+m] = (a cross w)[j] (cross-correlation)
        # In terms of convolution: (a cross w)[j] = (a * w_rev)[n-1-j + ...]
        # This is getting confusing. Let me just compute it with FFT properly.
        
        # sum_m a[m] * w[j+m] for m=0..n-1, j+m in 0..2n-2
        # = sum_m a[m] * w_full[j+m] where w_full has length 2n-1
        # This is the cross-correlation: (a ⋆ w_full)[j]
        # = (a * w_full_rev)[(2n-2) - j + n - 1]... 
        # I'll just use the direct FFT formula:
        # cross[j] = ifft(FFT(a_padded) * conj(FFT(w_padded)))[j]
        # where a_padded and w_padded are zero-padded to length >= 2n-1
        
        fft_size = 1 << (2 * n - 1 - 1).bit_length()  # >= 2n-1
        A_fft = rfft(a, fft_size)
        W_fft = rfft(w, fft_size)
        cross = irfft(A_fft * np.conj(W_fft), fft_size)
        grad = 2.0 * cross[:n]
        
        # Project to tangent space of sum(a) = norm_target
        grad -= np.mean(grad)
        
        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm_target / s_new
        
        a = a_new
        M_new = fast_autocorr(a).max()
        
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2 * n * best_M / (norm_target ** 2)
            print(f"    LSE-Adam step={steps} ({elapsed:.0f}s): obj={score:.6f}, T={T:.4f}")
    
    score = 2 * n * best_M / (norm_target ** 2)
    return best_a, score


def fw_method(a0, time_limit, verbose=True):
    """Frank-Wolfe method for the convex relaxation."""
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_target / n
    else:
        a = a * norm_target / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p = int(np.argmax(b))
        
        # Subgradient of max at peak p:
        # grad[j] = 2 * a[p-j] for j in [max(0,p-n+1), min(n-1,p)]
        grad = np.zeros(n)
        js = max(0, p - n + 1)
        je = min(n - 1, p)
        grad[js:je+1] = 2.0 * a[p - np.arange(js, je+1)]
        
        # Linear minimization over simplex: pick j with smallest grad
        j_star = int(np.argmin(grad))
        
        # Step size: tau / (1 + tau) where tau = step counter
        # Or use strong step: tau = 2/(t+2)
        t = steps + 1
        step = 2.0 / (t + 2.0)
        
        e = np.zeros(n)
        e[j_star] = 1.0
        a_new = (1 - step) * a + step * e * norm_target
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm_target / s_new
        
        a = a_new
        steps += 1
        
        M_new = fast_autocorr(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
        
        if verbose and steps % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2 * n * best_M / (norm_target ** 2)
            print(f"    FW step={steps} ({elapsed:.0f}s): obj={score:.6f}")
    
    score = 2 * n * best_M / (norm_target ** 2)
    return best_a, score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_target / n
    else:
        a = a * norm_target / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        p = int(np.argmax(b))
        max_b = b[p]
        
        # Find contributing pairs to peak p
        js = max(0, p - n + 1)
        je = min(n - 1, p)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p - idx]
        
        if contrib.max() < 1e-20:
            stall += 1
            if stall > 50:
                break
            continue
        
        # Source: position with largest contribution
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        # Destination: position not contributing to peak, with low value
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        non_contrib[src] = True
        
        if non_contrib.sum() > 0:
            dst_candidates = np.where(non_contrib)[0]
            dst = dst_candidates[np.argmin(a[dst_candidates])]
        else:
            dst = (src + 1) % n
            if dst == src:
                dst = (src + 2) % n
        
        # Try several fractions
        best_frac = 0
        best_trial_M = max_b
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = src_val * frac
            if amt < 1e-20:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm_target / s_t
            M_t = fast_autocorr(a_trial).max()
            if M_t < best_trial_M:
                best_trial_M = M_t
                best_frac = frac
        
        if best_frac > 0:
            a[src] -= src_val * best_frac
            a[dst] += src_val * best_frac
            a = np.maximum(a, 0)
            s_a = a.sum()
            a = a * norm_target / s_a
            M_new = fast_autocorr(a).max()
            
            if M_new < best_M:
                best_M = M_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Random perturbation when stalling
        if stall > 30:
            n_pert = max(3, n // 100)
            srcs = rng.choice(n, size=n_pert, replace=False)
            dsts = rng.choice(n, size=n_pert, replace=False)
            for i, j in zip(srcs, dsts):
                if a[i] > 0 and i != j:
                    amt = a[i] * 0.2
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm_target / s_a
            stall = 0
        
        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            score = 2 * n * best_M / (norm_target ** 2)
            print(f"    PeakShave ({elapsed:.0f}s): obj={score:.6f}, stall={stall}")
    
    score = 2 * n * best_M / (norm_target ** 2)
    return best_a, score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
