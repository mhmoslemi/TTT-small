def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = max(0.1, best_score * 0.02)
    T_end = max(0.001, best_score * 0.002)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Softmax weights
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient via FFT: grad = 2 * (w_rev * a) [n-1 : 2n-1]
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        
        # Project gradient to maintain sum constraint
        grad -= np.mean(grad)
        
        # Step size
        step = 2.0 / (T * n * 0.5)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with random moves and line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_stall = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find contributing positions
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        
        if len(idx) == 0:
            n_stall += 1
            if n_stall > 200:
                break
            continue
        
        contrib = a[idx] * a[p_star - idx]
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        if src_val < 1e-18:
            n_stall += 1
            continue
        
        # Choose destination: prefer positions not contributing to the peak
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        non_contrib[src] = True
        
        if non_contrib.sum() > 0:
            dst_candidates = np.where(non_contrib)[0]
        else:
            dst_candidates = np.array([i for i in range(n) if i != src])
        
        dst = dst_candidates[rng.integers(len(dst_candidates))]
        
        # Try different transfer fractions
        best_trial = max_b
        best_frac = 0
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            amt = src_val * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial:
                best_trial = sc
                best_frac = frac
        
        if best_frac > 0:
            a_trial = a.copy()
            a_trial[src] -= src_val * best_frac
            a_trial[dst] += src_val * best_frac
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial * norm / a_trial.sum()
            sc = fast_autocorr(a_trial).max()
            
            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                n_stall = 0
            else:
                # Annealing
                T = best_score * 0.002 * (1.0 - elapsed_frac)
                if rng.random() < np.exp(-(sc - best_score) / max(T, 1e-10)):
                    a = a_trial
                n_stall += 1
        else:
            n_stall += 1
        
        if n_stall > 100:
            # Larger random perturbation
            n_pert = max(5, n // 100)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-15:
                    amt = a[i] * rng.uniform(0.05, 0.3)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            n_stall = 0
        
        if verbose and n_stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave stall={n_stall} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def solve_best_response_penalty(a, M, time_limit=0.1, verbose=False):
    """Solve the best response LP using penalty gradient descent."""
    n = len(a)
    T = build_conv_matrix(a)
    size = T.shape[0]
    
    # Initialize
    b = np.ones(n)
    
    # Penalty parameter
    mu = 1.0  # Can tune this
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        # Compute (a*b) via FFT
        ab = fast_autocorr_b(a, b)  # cross-correlation
        
        # Penalty gradient
        excess = np.maximum(0, ab - M)
        # grad_j = sum_k 2 * excess_k * a_{k-j}
        # This is the cross-correlation of excess and a_rev
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        E = rfft(excess, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(E * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Negative of objective gradient (we want to maximize sum(b) - penalty)
        # min -sum(b) + mu/2 * sum(excess^2)
        # grad of this is: -1 + mu * 2 * excess * (a*b gradient)
        # = -1 + mu * grad
        total_grad = -1.0 + mu * grad
        
        # Projected gradient step
        step = 1.0 / (mu * np.max(grad) + 1) if np.max(grad) > 0 else 0.01
        b_new = b - step * total_grad
        b_new = np.maximum(b_new, 0)
        
        # Check convergence
        if np.max(np.abs(b_new - b)) < 1e-8:
            b = b_new
            break
        
        b = b_new
    
    return b

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def cross_corr(a, b):
    """Compute full cross-correlation a * b (convolution) via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a

def solve_lp_cutting_plane(a, M, solver, T, max_cp_iters=15, initial_k=80):
    """Solve max sum(b) s.t. (a*b)_k <= M, b >= 0 using cutting plane."""
    n = len(a)
    size = T.shape[0]

    # Initial active set: top-k constraints by current (a*a) values
    aa = T @ a
    init_idx = np.argsort(aa)[::-1][:min(initial_k, size)]
    active_set = list(init_idx)
    active_set_arr = np.array(active_set)

    b_var = cp.Variable(n, nonneg=True)
    b_val = None

    for cp_it in range(max_cp_iters):
        constraints = [T[active_set_arr, :] @ b_var <= M[active_set_arr]]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_val.sum() < 1e-15:
            break

        # Check all constraints via FFT (fast)
        circ = cross_corr(a, b_val)
        tol = 1e-7 * M + 1e-10
        violated = circ > M + tol

        if not violated.any():
            break

        # Add most violated constraints (up to 5 per round)
        viol_idx = np.where(violated)[0]
        viol_severity = circ[viol_idx] - M
        worst = viol_idx[np.argsort(viol_severity)[::-1][:5]]
        active_set_set = set(active_set)
        added = False
        for w in worst:
            if w not in active_set_set:
                active_set.append(int(w))
                active_set_set.add(int(w))
                added = True
        if not added:
            break
        active_set_arr = np.array(active_set)

        # Cap active set size
        if len(active_set) > 800:
            circ_cur = cross_corr(a, b_val)
            all_idx = np.argsort(circ_cur)[::-1][:800]
            active_set = list(all_idx)
            active_set_arr = np.array(active_set)

    if b_val is None:
        return None
    return b_val


def fessler_parks_lp(a0, time_limit, solver, verbose=True):
    """Fessler-Parks accelerated LP fixed-point with cutting-plane LP."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    T = build_conv_matrix(a)  # Will be rebuilt when a changes

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.8
    M_history = []

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        M_history.append(M_curr)
        if M_curr < 1e-12:
            break

        # Rebuild T for current y
        T = build_conv_matrix(y)

        M_vec = np.full(2 * n - 1, M_curr)
        b = solve_lp_cutting_plane(y, M_curr, solver, T)

        if b is None or b.sum() < 1e-15:
            stall += 1
            if stall > 100:
                rng = np.random.default_rng(n_iters * 7 + 3)
                y = np.maximum(y + rng.normal(0, 0.02 * y.mean(), n), 0)
                y = y * norm / y.sum()
                prev_y = y.copy()
                stall = 0
            continue

        g = b * norm / b.sum()
        n_iters += 1

        # Fessler-Parks step
        t = n_iters
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma based on oscillation
        if len(M_history) > 50:
            recent = M_history[-30:]
            diffs = np.diff(recent)
            sign_changes = np.sum(np.sign(diffs[:-1]) * np.sign(diffs[1:]) < 0)
            osc_ratio = sign_changes / max(len(diffs) - 1, 1)
            if osc_ratio > 0.6:
                gamma = max(1.0, gamma - 0.15)
            elif osc_ratio < 0.2 and stall == 0:
                gamma = min(2.0, gamma + 0.05)

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, g={gamma:.2f}, stall={stall}")

        # Restart
        if stall > 150:
            rng = np.random.default_rng(n_iters * 13 + 7)
            scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.8
            M_history = []
            if verbose:
                print(f"    [Restart iter={n_iters}]")

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive temperature and momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.05, best_score * 0.03)
    T_end = max(0.001, best_score * 0.001)
    v = np.zeros(n)  # momentum
    beta = 0.9

    a_rev = None

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient via FFT
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]

        # Zero-mean projection
        grad -= np.mean(grad)

        # Adam-like adaptive step
        step = 2.0 / (T * np.sqrt(n) * 0.5)
        v = beta * v + (1 - beta) * grad
        v_hat = v / (1 - beta ** (steps + 1))
        a_new = a - step * v_hat
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Multi-peak shaving with adaptive moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_stall = 0
    n_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        # Find top-3 peaks
        top_k = min(3, 2 * n - 1)
        peak_indices = np.argsort(b)[::-1][:top_k]

        improved = False
        for p_star in peak_indices:
            p_star = int(p_star)
            js = max(0, p_star - n + 1)
            je = min(n - 1, p_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[p_star - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-18:
                continue

            # Destination: non-contributing to this peak, low value
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je + 1] = False
            non_contrib[src] = True
            if non_contrib.sum() > 0:
                dst_cands = np.where(non_contrib)[0]
            else:
                dst_cands = np.array([i for i in range(n) if i != src])
            dst = dst_cands[rng.integers(len(dst_cands))]

            # Try fractions
            best_trial = max_b
            best_frac = 0
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_trial:
                    best_trial = sc
                    best_frac = frac

            if best_frac > 0 and best_trial < best_score:
                a.copy()
                a[src] -= src_val * best_frac
                a[dst] += src_val * best_frac
                a = np.maximum(a, 0)
                a = a * norm / a.sum()
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                    improved = True
                    break

        if not improved:
            # Annealing move
            p_star = int(peak_indices[0])
            js = max(0, p_star - n + 1)
            je = min(n - 1, p_star)
            idx = np.arange(js, je + 1)
            if len(idx) > 0:
                contrib = a[idx] * a[p_star - idx]
                src = idx[np.argmax(contrib)]
                src_val = a[src]
                if src_val > 1e-15:
                    dst = rng.integers(n)
                    if dst != src:
                        for frac in [0.05, 0.1, 0.2]:
                            a_trial = a.copy()
                            a_trial[src] -= src_val * frac
                            a_trial[dst] += src_val * frac
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial * norm / a_trial.sum()
                            sc = fast_autocorr(a_trial).max()
                            if sc < best_score:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                n_improve += 1
                                n_stall = 0
                                break
                        else:
                            n_stall += 1
                    else:
                        n_stall += 1
                else:
                    n_stall += 1
            else:
                n_stall += 1

        # Large perturbation when stuck
        if n_stall > 100:
            n_pert = max(5, n // 80)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-15:
                    amt = a[i] * rng.uniform(0.05, 0.4)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            n_stall = 0

        if verbose and n_improve > 0 and n_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    base = np.sin(np.pi * x) ** 1.5
    noise = rng.normal(0, 0.05, n)
    a = base + noise
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase LP Nesterov with multi-length and polishing."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = make_init(n_base, seed)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Quick multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-length exploration (budget={remaining:.0f}s) ===")
    candidate_lengths = [n_base, n_base + 200, n_base + 500]
    # Deduplicate and filter
    candidate_lengths = sorted(set([l for l in candidate_lengths if 500 <= l <= 5000]))

    for n_try in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        t_alloc = min(remaining * 0.15, 120.0)
        if t_alloc < 20:
            break
        print(f"  Trying n={n_try} (budget={t_alloc:.0f}s)")
        if n_try == n_base:
            init = best_seq.copy()
        else:
            init = interpolate_sequence(best_seq, n_try)
        a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
        a_out, _ = fessler_parks_lp(a_norm, t_alloc, solver, verbose=False)
        update_best(a_out, f"explore_n{n_try}")

    # === Phase 2: Main LP Nesterov on best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Main LP Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.50, 800.0)
    if t2 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = fessler_parks_lp(a_norm, t2, solver, verbose=True)
        update_best(a_out, f"main_n{n_cur}")

    # === Phase 3: Continued LP on best (with fresh start if needed) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Round 2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        # Add small perturbation to avoid exact same trajectory
        rng2 = np.random.default_rng(12345)
        a_norm = np.maximum(a_norm + rng2.normal(0, 0.005 * a_norm.mean(), n_cur), 0)
        a_norm = a_norm * np.sqrt(2.0 * n_cur) / a_norm.sum()
        a_out, _ = fessler_parks_lp(a_norm, t3, solver, verbose=True)
        update_best(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Try a longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 4: LP longer n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = fessler_parks_lp(a_norm, t4, solver, verbose=True)
        update_best(a_out, f"long_n{n_long}")

    # === Phase 5: LSE gradient polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LSE-GD n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t5, verbose=True)
        update_best(a_out, f"lse_n{n_cur}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t6, seed=777, verbose=True)
        update_best(a_out, f"peak_n{n_cur}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = fessler_parks_lp(a_norm, t7, solver, verbose=True)
        update_best(a_out, f"lpfin_n{n_cur}")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.10, 80.0)
    if t8 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t8, seed=31415, verbose=False)
        update_best(a_out, f"peakfin_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
