import numpy as np

def project_to_simplex(sequence, n=None):
    """Rescales a sequence of non-negative numbers to have sum 1.0.
     If the sequence is already non-negative and sum>=0.01, direct normalization occurs.
     Else, if sum < 0.01, returns a scaled version using factors from 0 to 1000 that increases the sum to 1 while preserving ratios.
     If n is given, uses interpolation to match the desired length before normalization."""
    if n is not None and len(sequence) != n:
        new_seq = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(sequence)), sequence)
        new_seq = np.maximum(new_seq, 0.0)
        s_val = new_seq.sum()
        if s_val > 1e-10:
            scale_factor = 1.0 / s_val
            new_seq = np.clip(new_seq * scale_factor, 0.0, 100.0)  # Apply maximum scaling here
        else:
            new_seq = np.ones(n) * 0.0005  # Fallback
        return new_seq
    else:
        arr = np.array(sequence, dtype=np.float64)
        s_val = arr.sum()
        arr = np.maximum(arr, 0.0)
        if s_val > 1e-10:
            scale_factor = 0.01 / s_val  # Ensure sum>=0.01 by scaling first
            if scale_factor > 1.0 or np.any(arr*scale_factor > 1000.0):
                scale_factor = 1.0 / s_val
            arr = np.clip(arr * scale_factor, 0.0, 1000.0)
        else:
            arr = np.ones(len(sequence)) * 0.0005 / 0.0005  # Normalize to smallest possible with sum 0.0005 if all zero        
        return arr
