    # Or use power iteration on Q_y to estimate lambda_max
    # For simplicity, start with a safe step size

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Von Neumann minimax iteration with FISTA inner solver
2. Interleaved with LP-Nesterov for final refinement
3. Multi-length exploration
4. Peak shaving for final polish

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def hankel_matvec(y, b):
    """Compute Q_y @ b where Q_y[j,l] = y[j+l]. Uses FFT-based correlation."""
    # (Q_y @ b)[j] = sum_l y[j+l] * b[l] = correlate(y, b, mode='valid')[j]
    return correlate(y, b, mode='valid', method='fft')

def minimize_quadratic_simplex(y, n, max_iters=50, verbose=False):
    """Minimize b^T Q_y b over the simplex using FISTA with backtracking."""
    # Initialize
    b = np.ones(n) / n
    b_old = b.copy()
    t = 1.0
    
    # Estimate L (Lipschitz constant of grad f = 2*Q*y*b)

    rng = np.random.default_rng(12345)

        norm = np.sqrt(np.dot(v, Qv) * 2 + 1e-30)
        # Actually, L = 2 * lambda_max(Q_y)

        v = Qv / Qv_norm
        L = 2.0 * np.dot(v, hankel_matvec(y, v))
        # Actually this gives 2 * v^T Q v / v^T v = 2 * Rayleigh quotient = 2*lambda_max
        # Wait, power iteration converges to the top eigenvector, and v^T Q v -> lambda_max
        # So L = 2 * lambda_max
    L = max(L, 1e-10)
    
    prev_obj = np.inf
    for i in range(max_iters):
        # FISTA momentum
        z = b + ((t - 1) / (t + 1)) * (b - b_old)

    b = np.maximum(b_init, 0).astype(np.float64)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n

    autoc = fast_autocorr(b)

    y = y / y.sum()

    best_score = autoc.max()
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP-Nesterov (Fessler-Parks) with HiGHS solver and top-K constraints
2. Von Neumann minimax iteration with FISTA inner solver (different optimization landscape)
3. Multi-length exploration with interpolation
4. Adaptive time allocation based on progress
5. Peak shaving with simulated annealing for final polish

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

# Global state
_best_seq_global = None
_best_score_global = float('inf')


def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution of a with itself reversed... no, a*a)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(y, b):
    """Compute Q_y @ b where Q_y[j,l] = y[j+l]. Uses scipy correlate."""
    from scipy.signal import correlate
    return correlate(y, b, mode='valid', method='fft')


def solve_lp_topk(a, M, K, use_highs=True):
    """
    Solve: max 1^T b s.t. (a*b)[i] <= M for i in active_set, b >= 0
    where (a*b) is the linear convolution of a and b.
    Uses only top-K constraints.
    Returns b (the best response) or None.
    """
    n = len(a)
    size = 2 * n - 1
    K = min(K, size)
    
    # Find top-K autocorrelation peaks of a (these are the most likely binding constraints)
    c = fast_autocorr(a)
    active_idx = np.argsort(c)[::-1][:K]
    
    # Build T matrix: T[k, j] = a[active_idx[k] - j] for valid j
    T = np.zeros((K, n))
    for k in range(K):
        i = active_idx[k]
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            T[k, js] = a[i - js]
    
    # Solve LP: max 1^T b s.t. T @ b <= M, b >= 0
    if use_highs:
        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status == 0 and res.x is not None:
                b_val = np.maximum(res.x, 0.0)
                if b_val.sum() > 1e-15:
                    return b_val
        except Exception:
            pass
    
    # Fallback: use cvxpy
    try:
        import cvxpy as cp
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
        prob.solve(solver=cp.PDLP, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            if b_val.sum() > 1e-15:
                return b_val
    except Exception:
        pass
    
    return None

def minimax_fista(a0, time_limit, n_inner=40, verbose=True):
    """
    Von Neumann minimax iteration with FISTA inner solver.
    min_b max_k (b*b)_k  <=>  max_y min_b b^T Q_y b
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    b = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s
    
    # Initialize y from current autocorrelation
    autoc = fast_autocorr(b)
    T_init = max(autoc.max() * 0.1, 1e-10)
    y = np.exp(autoc / T_init)
    y = y / y.sum()
    
    best_b = b.copy()
    best_score = autoc.max()
    t0 = time.monotonic()
    
    # FISTA parameters
    rng = np.random.default_rng(42)
    
    for outer in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Inner: minimize b^T Q_y b over simplex using FISTA
        b_fista = b.copy()
        b_prev_fista = b.copy()
        t_fista = 1.0
        
        # Estimate Lipschitz constant via power iteration (do this every outer iter)
        v_pi = rng.dirichlet(np.ones(n))
        L_est = 2.0
        for _ in range(15):
            Qv = hankel_matvec(y, v_pi)
            nv = np.linalg.norm(Qv)
            if nv < 1e-15:
                break
            v_pi = Qv / nv
            L_est = 2.0 * np.dot(v_pi, hankel_matvec(y, v_pi))
        L = max(L_est * 1.5, 1e-8)
        eta = 1.0 / L
        
        for inner in range(n_inner):
            z = b_fista + ((t_fista - 1.0) / (t_fista + 1.0)) * (b_fista - b_prev_fista)
            Qz = hankel_matvec(y, z)
            grad = 2.0 * Qz
            b_new = project_simplex(z - eta * grad)
            b_prev_fista = b_fista
            b_fista = b_new
            t_fista = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        
        b = b_fista
        
        # Evaluate
        autoc = fast_autocorr(b)
        max_autoc = autoc.max()
        
        if max_autoc < best_score:
            best_score = max_autoc
            best_b = b.copy()
        
        # Update y: concentrate on top peaks
        topK = min(30, 2 * n - 1)
        top_idx = np.argsort(autoc)[::-1][:topK]
        T_sm = max(autoc.max() * 0.03, 1e-10)
        y_new = np.zeros(2 * n - 1)
        y_new[top_idx] = np.exp(autoc[top_idx] / T_sm)
        ys = y_new.sum()
        if ys > 1e-15:
            y = y_new / ys
        else:
            y = autoc / autoc.sum()
        
        if verbose and outer % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Minimax outer={outer} ({elapsed:.0f}s): best={best_score:.8f}, cur={max_autoc:.8f}")
    
    return best_b * norm, best_score * 2 * n


def hankel_matvec_fast(y, b):
    """Fast Hankel matrix-vector product using FFT."""
    n = len(b)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    Y = rfft(y, fft_size)
    B = rfft(b, fft_size)
    # correlate(y, b, 'valid') = irfft(Y * conj(B)) [shifted]
    # Actually let's use the convolution formulation:
    # (Q_y @ b)[j] = sum_l y[j+l] * b[l]
    # = convolve(reversed_y_padded, b) at appropriate index
    # Simpler: use scipy
    from scipy.signal import correlate
    return correlate(y, b, mode='valid', method='fft')


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find contributing pairs to the peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-20:
            stall += 1
            if stall > 200:
                break
            continue
        
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        # Find good destination (non-contributing, low value)
        non_contrib = np.zeros(n, dtype=bool)
        if js > 0:
            non_contrib[:js] = True
        if je < n - 1:
            non_contrib[je + 1:] = True
        non_contrib[src] = True
        nc_idx = np.where(non_contrib)[0]
        
        if len(nc_idx) == 0:
            nc_idx = np.array([i for i in range(n) if i != src])

                continue
