```python
def build_T(a, active_set=None):

    n = len(a)

        hi = min(n, k + 1)
        for j in range(lo, hi):

    return T


def lp_solve(a, M, solver, active_set=None):

        return None, active_set, None

        if dual_val is not None:

    except Exception:
        pass
    
    return b_val, active_set, duals

        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-14:
            break
        
        # Adaptive active set
        if active_set is None or t % 200 == 0:
            tol = 0.01 + 0.02 * np.exp(-t / 300.0)
            active_set = np.where(conv_y > M * (1.0 - tol))[0]
            if len(active_set) < 3:
                active_set = np.where(conv_y > M * 0.9)[0]
        
        b, active_set, duals = lp_solve(y, M, solver, active_set)
        
        if b is None:
            active_set = None
            b, active_set, duals = lp_solve(y, M, solver, None)
            if b is None:
                continue
        
        # Verify constraints; expand active set if violated
        full_size = 2 * n - 1
        if len(active_set) < full_size:
            remaining = np.setdiff1d(np.arange(full_size), active_set)
            if len(remaining) > 0:
                T_rem = build_T(y, remaining)
                viol = T_rem @ b
                viol_idx = np.where(viol > M * (1 + 1e-6))[0]
                if len(viol_idx) > 0:
                    active_set = np.concatenate([active_set, remaining[viol_idx]])
                    if len(active_set) > 0.9 * full_size:
                        active_set = None
                    b, active_set, duals = lp_solve(y, M, solver, active_set)
                    if b is None:
                        continue
        
        b_sum = b.sum()
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        y_new = y_new * norm / s_y if s_y > 1e-15 else y.copy()
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.95)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            a_new = a_new * norm / s_a if s_a > 1e-15 else y_new.copy()
        
        y = y_new
        a = a_new
        
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_autocorr(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()
                    stall = 0
        
        if stall > 100:
            y = best_a.copy()
            a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall = 0
        
        if verbose and t % 200 == 0:
            print(f"    NLP t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score
```

```python
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue

            a_i = a[i]
            d_max = a_i

            L = np.zeros(size)
            lo_j = j
            hi_j = min(size, j + n)
            L[lo_j:hi_j] += a[:hi_j - lo_j]
            lo_i = i
            hi_i = min(size, i + n)
            L[lo_i:hi_i] -= a[:hi_i - lo_i]

            Q = np.zeros(size)
            if 2 * i < size:
                Q[2 * i] = 1.0
            if 2 * j < size:
                Q[2 * j] = 1.0
            if i + j < size:
                Q[i + j] -= 2.0

            two_L = 2.0 * L

            best_d = 0.0
            best_f = conv.max()

            candidates = [d_max * x for x in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 0.95]]
            for d in candidates:
                val = np.max(conv + d * two_L + d * d * Q)
                if val < best_f - 1e-13:
                    best_f = val
                    best_d = d

            if best_d > 0:
                lo_d = max(0.0, best_d * 0.5)
                hi_d = min(d_max, best_d * 1.5)
                if best_d < d_max * 0.1:
                    hi_d = d_max * 0.15
                if best_d > d_max * 0.9:
                    lo_d = d_max * 0.85

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(20):
                    m1 = hi_d - gr * (hi_d - lo_d)
                    m2 = lo_d + gr * (hi_d - lo_d)
                    f1 = np.max(conv + m1 * two_L + m1 * m1 * Q)
                    f2 = np.max(conv + m2 * two_L + m2 * m2 * Q)
                    if f1 < f2:
                        hi_d = m2
                    else:
                        lo_d = m1

                d_star = (lo_d + hi_d) / 2.0
                f_star = np.max(conv + d_star * two_L + d_star * d_star * Q)
                smoother = f_star + 0.5 * d_star * d_star * (Q[2 * j] if 2 * j < size else 0)
                if f_star < best_f - 1e-13:
                    best_d = d_star
                    best_f = f_star

            if best_d > 1e-14:
                a[i] -= best_d
                a[j] += best_d
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                else:
                    a[i] += best_d
                    a[j] -= best_d
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 6:
                break

        if verbose and n_passes % 30 == 0:
            print(f"    PCD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
```
