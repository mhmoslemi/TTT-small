import numpy as np

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]

    except Exception:

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_palindromic(a):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(a)
    a_rev = a[::-1].copy()
    return 0.5 * (a + a_rev)

def is_palindromic(a, tol=1e-6):

def lp_oracle(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Returns normalized b (sum=1) or None."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix T: T[k,j] = a[k-j] if 0<=k-j<n
    # T @ b gives (a*b)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    
    return b_val / b_sum

    return best_a, best_m, iter_count

def soft_subgradient(a, ac, top_k=30, beta=5.0):
    """Compute soft subgradient using top-k active constraints."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_avg = 0
    gamma = 0.8  # Step size parameter

        if time.monotonic() - t0 > time_limit:
            break
        
        step = gamma / (np.sqrt(t) + 1.0)

        g = soft_subgradient(y, ac, top_k=top_k, beta=beta)
        
        log_y = np.log(y + 1e-30)
        log_y_new = log_y - step * g
        max_log = log_y_new.max()
        y_new = np.exp(log_y_new - max_log)
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        y = y_new
        
        avg_a += y
        n_avg += 1
        
        # Check average (Nesterov averaging)
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        M_y = fast_autocorr(y).max()

            best_m = M_y
            best_a = y.copy()

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    iter_count = 0
    stalled = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        g = lp_oracle(y, M_curr, solver)
        if g is None:
            stalled += 1
            if stalled > 50:
                break
            continue
        stalled = 0
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={iter_count} t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def soft_subgradient(a, ac, top_k=30, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad
