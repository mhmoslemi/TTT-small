def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_best_response(y, M, solver):
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build the constraint matrix T where (T @ b)_k = conv(y, b)_k
    T = np.zeros((size, n))
    for i in range(n):
        end = min(i + n, size)
        T[i:end, i] = y[:end - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:

    return b_val

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    n_lp_solves = 0

    prev_score = fast_autocorr(a).max()

        if M_curr < 1e-12:

        n_lp_solves += 1

        if b_sum < 1e-15:

        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:

        else:

            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # FISTA restart: if the score increased, reset momentum
        curr_score = min(score_y, score_a)
        if curr_score > prev_score * 1.01:
            y = best_a.copy()
            a = best_a.copy()
        
        prev_score = curr_score
        
        if verbose and n_lp_solves % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_lp_solves} (t={t}, {elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lse_objective(x, n, T):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    shifted = T * (b - max_b)
    max_shifted = shifted.max()
    exp_vals = np.exp(shifted - max_shifted)
    lse = max_b + (max_shifted + np.log(exp_vals.sum())) / T
    obj = 2.0 * n * lse / (s * s)
    return obj

def lse_gradient(x, n, T):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shifted = T * (b - max_b)
    max_shifted = shifted.max()
    exp_vals = np.exp(shifted - max_shifted)
    w = exp_vals / exp_vals.sum()
    lse = max_b + (max_shifted + np.log(exp_vals.sum())) / T
    
    # Compute g_j = sum_{i=0}^{n-1} a_i * w_{j+i}
    # Using: g = conv(rev(a), w)[n-1 : 2n-1]
    fft_len = 1 << (3 * n - 1).bit_length()
    a_rev = a[::-1]
    A_rev = rfft(a_rev, fft_len)
    W = rfft(w, fft_len)
    conv_result = irfft(A_rev * W, fft_len)
    g = conv_result[n-1 : 2*n-1]
    
    conv_grad = 2.0 * g
    obj_val = 2.0 * n * lse / (s * s)
    grad = (2.0 * n / s**2) * conv_grad - (4.0 * n * lse / s**3) * a
    return grad

def lse_scipy_optimize(a0, time_limit, T_start=50.0, T_end=1.5, n_stages=10):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / max(1, n_stages - si), remaining * 0.3)

        x0 = a.copy()

        res = scipy_minimize(lse_objective, x0, args=(n, T),
                              method='L-BFGS-B',
                              jac=lambda x, n_n, T_T: lse_gradient(x, n_n, T_T),
                              bounds=[(0, 1000.0)] * n,
                              options={'maxiter': 200, 'maxfun': 500,
                                       'ftol': 1e-12, 'gtol': 1e-10})
        if res.x is not None:
            a_candidate = np.maximum(res.x, 0.0)
            s_c = a_candidate.sum()
            if s_c > 1e-10:
                a_candidate = a_candidate * norm / s_c
                score_c = fast_autocorr(a_candidate).max()
                if score_c < best_score:
                    best_score = score_c
                    best_a = a_candidate.copy()
                    a = a_candidate
                    print(f"    LSE T={T:.1f}: score={best_score:.8f}")

    return best_a, best_score
