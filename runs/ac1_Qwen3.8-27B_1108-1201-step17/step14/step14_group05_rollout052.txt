import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def lse_gradient(a, T):
    """Compute gradient of LSE_T(autocorr(a)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    
    # grad_j = 2 * sum_k w_k * a_{k-j} for valid k-j
    # This is 2 * (w * a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad

def lse_value(a, T):
    """Compute LSE_T(autocorr(a))."""
    b = fast_autocorr(a)
    max_b = b.max()
    return max_b + T * np.log(np.sum(np.exp((b - max_b) / T)))

def adam_lse_optimize(a0, time_limit, verbose=True):
    """LSE-smoothed optimization with Adam."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 1.0 / n  # initial learning rate
    
    t0 = time.monotonic()
    step = 0
    
    # Temperature schedule
    b_init = fast_autocorr(a).max()
    T_start = max(0.1, b_init * 0.1)
    T_end = max(0.001, b_init * 0.003)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        if max_b < best_max:
            best_max = max_b
            best_a = a.copy()
        
        # Compute gradient
        grad = lse_gradient(a, T)
        
        # Project gradient onto tangent space (sum constraint)
        grad -= np.mean(grad)
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        
        # Adaptive learning rate
        if step % 100 == 0:
            score_now = fast_autocorr(a).max()
            if score_now > best_max * 1.01:
                lr *= 0.5
            elif score_now < best_max * 0.99:
                lr = min(lr * 1.2, 10.0 / n)
        
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam step={step} ({elapsed:.0f}s): T={T:.4f}, lr={lr:.4f}, best={best_max:.8f}, curr={max_b:.6f}")
    
    return best_a, best_max

        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if stall > 50:
            gamma = max(1.3, gamma - 0.05)
        if stall > 150:
            gamma = min(2.0, gamma + 0.1)

        # Adaptive constraint count: reduce if stalling (faster LP), increase if progressing
        if stall > 100 and constraint_count > 100:
            constraint_count = max(100, constraint_count - 20)
        elif stall == 0 and constraint_count < max_constraints:
            constraint_count = min(max_constraints, constraint_count + 10)

        # Restart if very stuck
        if stall > 400:
            rng = np.random.default_rng(n_iters * 7 + 3)
            pert = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            s = y.sum()
            if s > 1e-15:
                y = y * norm / s
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            prev_y = y.copy()
            warm_g = None
            stall = 0
            gamma = 1.7
            constraint_count = max_constraints
            if verbose:
                print(f"    [Restart at iter {n_iters}]")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, "
                  f"gam={gamma:.2f}, constraints={constraint_count}, stall={stall}")

    return best_a, best_score


def _adam_lse_optimize(a0, time_limit, verbose=True):
    """
    LSE-smoothed gradient descent with Adam optimizer.
    Fast iterations using FFT-based gradients.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_max = _fast_autocorr(a).max()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = 2.0 / n

    t0 = time.monotonic()
    step = 0

    b_init = _fast_autocorr(a).max()
    T_start = max(0.05, b_init * 0.08)
    T_end = max(0.0005, b_init * 0.002)

    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)

        b = _fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        if max_b < best_max:
            best_max = max_b
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient
        grad = _lse_gradient(a, T)

        # Project onto tangent space of simplex constraint
        grad -= np.mean(grad)

        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new

        # Adaptive learning rate
        if step % 200 == 0:
            score_now = _fast_autocorr(a).max()
            if score_now > best_max * 1.005:
                lr *= 0.6
                no_improve = 1000
            elif score_now < best_max * 0.999:
                lr = min(lr * 1.3, 20.0 / n)

        # Escape stagnation
        if no_improve > 5000:
            rng = np.random.default_rng(step)
            a = np.maximum(a + rng.normal(0, 0.005 * a.mean(), n), 0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            m = np.zeros(n)
            v = np.zeros(n)
            step = 0
            no_improve = 0
            lr = 2.0 / n

        if verbose and step % 2000 == 0:
            print(f"    LSE-Adam step={step} ({elapsed:.0f}s): T={T:.5f}, lr={lr:.4f}, "
                  f"best={best_max:.8f}, curr={max_b:.6f}")

    return best_a, best_max


def _peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = _fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        # Find top contributing pair
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 200:
                break
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]
        if src_val < 1e-15:
            n_stall += 1
            continue

        # Find non-contributing positions
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[idx] = False
        non_contrib[src] = True
        nc_idx = np.where(non_contrib)[0]

        if len(nc_idx) == 0:
            nc_idx = np.array([i for i in range(n) if i != src])

        # Pick destination from low-value non-contributing positions
        n_sel = min(20, len(nc_idx))
        sorted_dst = nc_idx[np.argsort(a[nc_idx])][:n_sel]
        dst = sorted_dst[rng.integers(len(sorted_dst))]

        # Try several transfer fractions
        best_trial = max_b
        best_frac = 0.0
        for frac in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            amt = src_val * frac
            if amt < 1e-18:
                break
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t < 1e-15:
                break
            a_t = a_t * norm / s_t
            sc = _fast_autocorr(a_t).max()
            if sc < best_trial:
                best_trial = sc
                best_frac = frac

        if best_frac > 0:
            a_t = a.copy()
            amt = src_val * best_frac
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            a_t = a_t * norm / s_t
            sc = _fast_autocorr(a_t).max()

            if sc < best_score:
                a = a_t
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                T_ann = best_score * 0.002 * (1.0 - elapsed_frac)
                if rng.random() < np.exp(-(sc - best_score) / max(T_ann, 1e-10)):
                    a = a_t
                n_stall += 1
        else:
            n_stall += 1

        # Random exploration when stuck
        if n_stall > 30:
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 0:
                    amt = a[i] * rng.uniform(0.05, 0.25)
                    a_t = a.copy()
                    a_t[i] -= amt
                    a_t[j] += amt
                    a_t = np.maximum(a_t, 0)
                    s_t = a_t.sum()
                    if s_t > 1e-15:
                        a_t = a_t * norm / s_t
                        sc = _fast_autocorr(a_t).max()
                        if sc < best_score:
                            a = a_t
                            best_score = sc
                            best_a = a.copy()
                            n_improve += 1
                            n_stall = 0
                            break

        if n_stall > 200:
            # Larger perturbation
            n_pert = max(10, n // 40)
            idx_p = rng.choice(n, size=n_pert, replace=False)
            a_t = a.copy()
            for ip in idx_p:
                amt = a[ip] * rng.uniform(0.1, 0.4)
                j = rng.integers(n)
                a_t[ip] -= amt
                a_t[j] += amt
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t > 1e-15:
                a_t = a_t * norm / s_t
                sc = _fast_autocorr(a_t).max()
                if sc < best_score * 1.01:
                    a = a_t
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave imp={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: hybrid LP Nesterov + LSE-Adam + peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = _get_cvxpy_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _eval_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        n_base = 1600

    def _update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.8f}, n={len(a)}, t={elapsed:.0f}s")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov (main workhorse, ~35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, t1, solver, verbose=True, max_constraints=250)
        _update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: LSE-Adam (fast gradient, ~20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LSE-Adam n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _adam_lse_optimize(a_norm, t2, verbose=True)
        _update(a_out, f"lse1_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 (~25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, t3, solver, verbose=True, max_constraints=200)
        _update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: LSE-Adam round 2 (~10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LSE-Adam r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _adam_lse_optimize(a_norm, t4, verbose=True)
        _update(a_out, f"lse2_n{n_cur}")

    # === Phase 5: Try longer length with LP Nesterov (~10% time) ===
    n_long = n_base + 80
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n_long} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 180.0)
    if t5 > 60:
        init_long = _interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, t5, solver, verbose=True, max_constraints=200)
        _update(a_out, f"lp_long_n{n_long}")

    # === Phase 6: Peak shaving polish (~5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.05, 80.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t6, seed=777, verbose=True)
        _update(a_out, f"peak1_n{n_cur}")

    # === Phase 7: Final LP polish (~5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.05, 100.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _nesterov_lp_optimize(a_norm, t7, solver, verbose=True, max_constraints=150)
        _update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.05, 60.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t8, seed=31415, verbose=True)
        _update(a_out, f"peak_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
