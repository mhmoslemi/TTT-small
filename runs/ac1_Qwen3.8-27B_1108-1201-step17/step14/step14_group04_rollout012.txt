def fast_autocorr(a):
    """FFT-based full convolution (autocorrelation) of a with itself."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_T_structure(n):
    """Build the fixed row/col structure for the convolution matrix T."""
    global _global_T_structure, _global_T_rows, _global_T_cols
    if _global_T_structure is not None and _global_T_structure.shape == (2 * n - 1, n):
        return _global_T_structure
    rows = []
    cols = []
    for j in range(n):
        for offset in range(n):
            rows.append(j + offset)
            cols.append(j)
    _global_T_rows = np.array(rows, dtype=np.int32)
    _global_T_cols = np.array(cols, dtype=np.int32)
    _global_T_structure = (2 * n - 1, n)
    print(f"  Built T structure: {len(rows)} nonzeros")
    return _global_T_structure


def build_T_matrix(a):
    """Build sparse T matrix with coefficients from a."""
    n = len(a)
    build_T_structure(n)
    # values: T[k, j] = a[k-j] where k-j is in [0, n-1]
    # For column j, rows j to j+n-1 have values a[0], a[1], ..., a[n-1]
    vals = np.tile(a, n)
    T = sp.csr_matrix((vals, (_global_T_rows, _global_T_cols)), shape=(2 * n - 1, n))
    return T


def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def compute_subgradient(a, c, n_active=5):
    """Compute subgradient of max_k c_k at a, using top n_active peaks."""
    n = len(a)
    max_c = c.max()
    if max_c < 1e-15:
        return np.zeros(n)
    # Find lags within 1% of max
    threshold = max_c * 0.99
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Sort by value, take top n_active
    if len(active) > n_active:
        active_vals = c[active]
        top_idx = np.argsort(active_vals)[::-1][:n_active]
        active = active[top_idx]
    
    g = np.zeros(n)
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        g[js] += 2.0 * a[k - js]
    g /= len(active)
    return g


def project_simplex_sum(v, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def subgradient_descent(a0, time_limit, verbose=False):
    """Subgradient descent with diminishing step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    S = np.sqrt(2.0 * n)
    a = a * S
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    base_L = n * 4.0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_autocorr(a)
        max_c = c.max()
        g = compute_subgradient(a, c, n_active=3)
        
        # Also add gradient from the constraint sum(a) = S (just uniform direction)
        # Step size: diminishing
        alpha = 1.0 / (base_L * (1.0 + t * 0.01))
        
        a_new = a - alpha * g
        a_new = project_simplex_sum(a_new, S)
        
        c_new = fast_autocorr(a_new)
        score_new = c_new.max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and t % 1000 == 0:
            print(f"    Subgrad iter={t}: score={best_score:.8f}")
    
    return best_a, best_score


def lp_simons_s(a0, time_limit, solver, verbose=True):

    LP-based Simons-S method with tail averaging and adaptive restarts.

    best_score = fast_autocorr(a).max()

    n_full = 0
    n_tail = 0
    tail_start_detected = False

    bail_threshold = best_score * 1.005  # 0.5% above best
    
    # Stall detection
    stall_counter = 0
    stall_best = best_score
    rng = np.random.default_rng(12345)
    
    n_iters = 0
    last_print = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute M from current iterate

        M_curr = c_y.max()
        if M_curr < 1e-12:
            break
        
        # Solve LP
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * S / b_sum
        n_iters += 1

        y_new = (1.0 - alpha_n) * y + alpha_n * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:

        else:
            y_new = y_new * S / s_y

            a_new = g.copy()
        else:

            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()

                a_new = a_new * S / s_a

        a = a_new

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        n_full += 1

            if n_full > 50 and score_a < bail_threshold:
                tail_start_detected = True
                tail_start_score = best_score
                tail_avg = np.zeros(n)
                n_tail = 0
        if tail_start_detected:
            tail_avg += a
            n_tail += 1
        
        # Periodic checks
        if n_iters % 10 == 0:
            # Full average score
            if n_full >= 20:
                fa = full_avg / n_full
                fa = fa * S / fa.sum()
                sc_fa = fast_autocorr(fa).max()
                if sc_fa < best_score:
                    best_score = sc_fa
                    best_a = fa.copy()

                ta = ta * S / ta.sum()
                sc_ta = fast_autocorr(ta).max()
                if sc_ta < best_score:
                    best_score = sc_ta
                    best_a = ta.copy()

            if best_score < stall_best * (1 - 1e-5):

                y = y_pert
                a = y_pert.copy()
                stall_counter = 0
                # Reset tail average
                tail_start_detected = False
                tail_avg = np.zeros(n)
                n_tail = 0
        
        if verbose and n_iters - last_print >= 200:
            elapsed = time.monotonic() - t0
            print(f"    LP-SimonsS iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    stall = 0
    best_c = fast_autocorr(a)
    
    while time.monotonic() - t0 < time_limit:
        c = best_c
        max_c = c.max()
        # Find peak lags
        peak_threshold = max_c * 0.995
        peak_lags = np.where(c >= peak_threshold)[0]
        
        # Compute subgradient at peak
        g = compute_subgradient(a, c, n_active=min(len(peak_lags), 5))
        
        # Try to reduce peak: move mass from high-subgradient positions to low-subgradient positions
        # High subgradient = "contributing to peak" => move mass away
        # Low subgradient = "not contributing" => move mass here
        
        src_candidates = np.where(g >= np.percentile(g, 85))[0]
        dst_candidates = np.where(g <= np.percentile(g, 15))[0]
        
        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            src_candidates = np.arange(n)
            dst_candidates = np.arange(n)
        
        improved = False
        # Try several candidate moves
        forattempt in range(30):
            src = src_candidates[rng.integers(len(src_candidates))]
            dst = dst_candidates[rng.integers(len(dst_candidates))]
            if src == dst or a[src] < 1e-10:
                continue
            
            for frac in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01]:
                amt = a[src] * frac
                if amt < 1e-12:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    break
                a_trial = a_trial * S / a_trial.sum()
                sc = fast_autocorr(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    best_c = fast_autocorr(a)
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > S * 1e-8:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * S / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        best_c = fast_autocorr(a)
                        stall = 0
            stall += 1
            if stall > 500:
                break
    
    return best_a, best_score


def multi_start_lp(a0, time_limit, solver, verbose=True):
    """LP-SimonsS with multiple random restarts from perturbed versions of best found."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    # Run for a while, then restart with perturbation
    phase_time = min(300.0, time_limit * 0.4)
    
    current = a.copy()
    n_phases = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        t_phase = min(phase_time, remaining)
        if t_phase < 10:
            t_phase = remaining
        
        print(f"    Multi-LP phase {n_phases} ({t_phase:.0f}s): current best={best_score:.8f}")
        a_out, sc_out = lp_simons_s(current, t_phase, solver, verbose=False)
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
        
        # Perturb for next phase
        if time.monotonic() - t0 < time_limit - 30:
            pert_scale = max(0.01, 0.05 / (n_phases + 1))
            noise = rng.standard_normal(n) * pert_scale * S
            current = best_a + noise
            current = project_simplex_sum(current, S)
            n_phases += 1
    
    return best_a, best_score


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        # Use a raised-cosine shape as initial
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x))
        best_seq = np.maximum(best_seq, 1e-6)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init (raised cosine): n={n_base}, score={best_score:.6f}")
