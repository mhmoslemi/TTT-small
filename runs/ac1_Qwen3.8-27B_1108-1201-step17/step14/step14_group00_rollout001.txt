    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Check for violations
    if len(active_set) < 2 * n - 1:
        full_T = np.zeros((2*n-1, n))
        for j in range(n):
            full_T[j:j+n, j] = a
        full_conv = full_T @ b_val
        violations = np.where(full_conv > M * (1 + 1e-6))[0]
        if len(violations) > 0:
            # Need to expand active set
            return None, violations
    
    duals = None
    try:
        duals = np.asarray(constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    
    return b_val, duals


def line_search(a, b, norm_target, max_iters=50):
    """Golden section line search to minimize max(((1-t)*a + t*b)*((1-t)*a + t*b))."""
    # Normalize path
    def obj(t):
        x = (1 - t) * a + t * b
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x * norm_target / s
        return fast_autocorr(x).max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(max_iters):
        if hi - lo < 1e-6:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)
    
    t_opt = (lo + hi) / 2
    return t_opt


def nesterov_lp_optimize_v2(a0, time_limit, solver, verbose=True, seed=0):
    """Improved Nesterov LP with exact line search and dual-based restart."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()  # Nesterov "lookahead" point
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    use_full = True  # Start with full LP
    
    # Adaptive: switch between full and screened LP
    screen_interval = 200  # Do full LP every 200 iterations
    active_set = None
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        # Decide whether to use full or screened LP
        if t % screen_interval == 1 or active_set is None:
            b, duals = lp_best_response_full(y, M_curr, solver)
            if b is not None and b.sum() > 1e-15:
                # Update active set based on duals
                if duals is not None:
                    # Constraints with significant dual values are "active"
                    active_set = np.where(duals > duals.max() * 0.01)[0] if duals.max() > 0 else np.arange(2*n-1)
                    if len(active_set) < 10:
                        active_set = np.where(conv_y > M_curr * 0.9)[0]
            use_full = True
        else:
            b, violations = lp_best_response_screened(y, M_curr, solver, active_set)
            if b is None and violations is not None:
                # Expand active set
                active_set = np.union1d(active_set, violations)
                if len(active_set) > 0.9 * (2*n-1):
                    active_set = None
                    b, duals = lp_best_response_full(y, M_curr, solver)
                else:
                    b, _ = lp_best_response_screened(y, M_curr, solver, active_set)
            use_full = False
        
        if b is None or b.sum() < 1e-15:
            continue
        
        b_sum = b.sum()
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y
        y = y_new
        a = a_new
        
        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Restart if stalled
        if stall_count > 80:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                print(f"    Restart at iter {t} ({time.monotonic()-t0:.0f}s)")
        
        if verbose and n_iters % 500 == 0:
            print(f"    NLP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def pair_cd_v2(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with golden section line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()

    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                # Compute direction: move mass from i to j
                # a_new = a + delta * (e_j - e_i)
                # Effect on convolution: delta * (a shifted right by j - a shifted right by i)

                    # If we change a[i] by -delta:
                    # Contribution to (a*a)_k changes by:
                    # -delta * a[k-i] (for the term where m=i)
                    # -delta * a[k-i] (for the term where k-m=i, i.e., m=k-i)
                    # So delta_k = -2*delta*a[k-i] + delta^2 (if k=2i)
                    
                # More efficient: precompute the "sensitivity" vector
                # s_k = sum of a[m] where k-m = i OR m = i
                # s_k = a[k-i] + a[k-i] = 2*a[k-i] (if both valid)
                # But we need to be careful about the k=2i case

                #           + delta*(col_j_forward + col_j_backward) - delta^2*e_{2j}

                # And col_i_backward[k] = a[k-i] if 0<=k-i<n else 0 (same thing for symmetric conv)

                #   (a*a)_k decreases by delta*a[k-i] (from m=i term) + delta*a[k-i] (from k-m=i term)
                #   = 2*delta*a[k-i], unless k=2i where it's delta*a[i] + delta*a[i] = 2*delta*a[i] (same)
                #   But wait, if k=2i, the m=i and k-m=i are the SAME term, so it's only delta*a[i]*2 - delta^2
                #   Actually no: in the sum sum_m a[m]*a[k-m], if m=i and k-m=i (i.e., k=2i),
                #   this is a single term a[i]*a[i] = a[i]^2, which changes to (a[i]-delta)^2 = a[i]^2 - 2a[i]*delta + delta^2
                #   So the change is -2a[i]*delta + delta^2 (not -2*delta*a[i])
                #   And for k != 2i, the change is -2*delta*a[k-i] (two separate terms)
                #   Wait, that's not right either. Let me be more careful.
                #
                # (a*a)_k = sum_{m: 0<=m<n, 0<=k-m<n} a[m] * a[k-m]
                # If we change a[i] -> a[i] - delta:
                # The terms involving a[i] are:
                #   m=i: a[i] * a[k-i] → (a[i]-delta) * a[k-i], change = -delta * a[k-i]
                #   k-m=i, i.e., m=k-i: a[k-i] * a[i] → a[k-i] * (a[i]-delta), change = -delta * a[k-i]

                #   If i != k-i, they are separate: total change = -2*delta*a[k-i]

                #              (using a[k-i]=0 if k-i not in [0,n))
                
                # Similarly for adding to position j:
                # change_k = +2*delta*a[k-j] + delta^2 * [k==2j]
                
                # Combined (move delta from i to j):
                # change_k = 2*delta*(a[k-j] - a[k-i]) + delta^2*([k==2j] - [k==2i])
                
                # We want to minimize max_k (conv_k + change_k) over delta in [0, ai]
                
                # Golden section search
                lo_d, hi_d = 0.0, ai
                gr = (np.sqrt(5) - 1) / 2
                
                def obj_delta(d):
                    new_conv = conv + 2*d*(np.roll(a, -j) - np.roll(a, -i))  # This is wrong
                    # Let me just compute it directly
                    ...
                
                # Actually, let me just do a simpler approach: evaluate at a few points
                candidates = [ai*0.1, ai*0.25, ai*0.5, ai*0.75, ai]
                best_d = 0
                best_val = conv.max()
                
                for d in candidates:
                    a[i] -= d
                    a[j] += d
                    s = a.sum()
                    a = a * norm / s
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_val:
                        best_val = new_max
                        best_d = d
                    a[i] += d
                    a[j] -= d
                    a = a * norm / s
                
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequence."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0, 2*np.pi, n)
        a = 0.5 + 0.3*np.cos(2*x) + 0.15*np.cos(5*x) + 0.05*np.cos(9*x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6*(x-0.3)**2) + 0.7*np.exp(-6*(x-0.7)**2)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = (1 - np.abs(2*x - 1)) ** 3
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7*np.sin(np.pi*x)**0.7
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 1 + 0.3*np.cos(4*np.pi*x) + 0.1*np.cos(8*np.pi*x)
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi*x)**0.5 * (1 + 0.2*np.cos(6*np.pi*x))
    elif choice == 8:
        # Multi-peak
        x = np.linspace(0, 1, n)
        a = sum(np.exp(-20*(x-c)**2) for c in [0.2, 0.5, 0.8])
    else:
        # Sidelobe-like
        x = np.linspace(0, 1, n)
        sinc_arg = np.pi * x * 4
        a = np.sinc(sinc_arg / np.pi) ** 2 * np.sin(np.pi * x)**0.3
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
