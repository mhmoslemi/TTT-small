def project_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho_val = rho[-1]
    theta = (cssv[rho_val] - S) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve(a, M, solver, max_active=200, warm_b=None):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None

    for iteration in range(8):
        circ = T @ b_val
        tol = 1e-6 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active < 10 or n_active >= min(size, max_active + 100):
            break
        if n_active > max_active:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
            active = np.zeros(size, dtype=bool)
            active[top_idx] = True
        else:
            active_idx = np.where(active)[0]
        sub_T = T[active_idx, :]
        sub_M = M[active_idx]
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [sub_T @ b_var <= sub_M])
        try:
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-4):
            b_val = b_new
        else:
            break
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lse_adam_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    T_start = max(0.1, best_score * 0.3)
    T_end = max(0.005, best_score * 0.002)
    lr = 0.01 / n
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)
    steps = 0
    stall = 0

    fft_size = 1 << (3 * n - 3).bit_length()

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        b_shifted = b - max_b
        w = np.exp(b_shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        grad_fft = irfft(np.conj(W) * A, fft_size)
        grad = 2.0 * grad_fft[:n]
        grad -= np.mean(grad)

        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
            lr = min(lr * 1.01, 0.1 / n)
            a = a_new
        else:
            stall += 1
            if stall > 200:
                rng = np.random.default_rng(steps + 7)
                a_pert = np.maximum(a + rng.normal(0, 0.02 * a.mean(), n), 0)
                a = a_pert * norm / a_pert.sum()
                m = np.zeros(n)
                v = np.zeros(n)
                steps = 0
                stall = 0
                lr = 0.005 / n
            else:
                a = a_new

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE-Adam step={steps} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.4f}")

    return best_a, best_score

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum

        if alpha > 0.5:
            alpha = 0.5

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:

        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)

            if momentum < 0.0:
                momentum = 0.0
            a_new = y_new + momentum * (y_new - prev_y)

            break

    return best_a, best_score


def peak_shave_local(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_stall = 0
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))

        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            n_stall += 1
            if n_stall > 200:
                break
            continue
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-20:
            n_stall += 1
            if n_stall > 200:
                break
            continue

        src = idx[np.argmax(contrib)]

        non_contrib = np.zeros(n, dtype=bool)
        if js > 0:
            non_contrib[:js] = True
        if je < n - 1:
            non_contrib[je + 1:] = True
        non_contrib[src] = True
        non_contrib_idx = np.where(non_contrib)[0]

        if len(non_contrib_idx) == 0:
            non_contrib_idx = np.array([i for i in range(n) if i != src])

        n_select = min(20, len(non_contrib_idx))
        sorted_dst = non_contrib_idx[np.argsort(a[non_contrib_idx])][:n_select]
        dst = sorted_dst[rng.integers(len(sorted_dst))]

        improved = False
        for frac in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2]:
            amt = a[src] * frac
            if amt < 1e-18:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                n_stall = 0
                improved = True
                break

        if not improved:
            T_anneal = best_score * 0.002 * (1.0 - elapsed_frac)
            amt = a[src] * rng.uniform(0.02, 0.15)
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if T_anneal > 1e-10 and rng.random() < np.exp(-(sc - best_score) / T_anneal):
                    a = a_trial
            n_stall += 1

        if n_stall > 50:
            n_pert = max(3, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                if a_trial[idx] <= 0:
                    continue
                j = rng.integers(n)
                if j == idx:
                    j = (j + 1) % n
                amt = a_trial[idx] * rng.uniform(0.05, 0.3)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.01:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        iterations += 1
        if verbose and iterations % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PeakShave iter={iterations} ({elapsed:.0f}s): best={best_score:.8f}, stall={n_stall}")

    return best_a, best_score


def multi_peak_subgrad(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step_size = 0.5 / n
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()

        K = min(10, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:K]
        peak_vals = b[top_peaks]

        T_speed = max(0.01, (max_b - peak_vals[-1] + 1e-10) * 0.1 * (1.0 - elapsed_frac))
        w = np.exp((peak_vals - max_b) / T_speed)
        w = w / w.sum()

        grad = np.zeros(n)
        for m, p in enumerate(top_peaks):
            if w[m] < 1e-7:
                continue
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            grad[js:je + 1] += 2.0 * w[m] * a[p - np.arange(js, je + 1)]

        grad -= np.mean(grad)
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break

        polyak_step = (max_b - best_score) / grad_norm_sq if max_b > best_score else step_size
        step = min(step_size, polyak_step) if polyak_step > 0 else step_size
        step = max(step, 1e-8 / n)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            step_size = min(step_size * 1.02, 1.0 / n)
        else:
            T_anneal = best_score * 0.001 * (1.0 - elapsed_frac)
            if T_anneal > 1e-12 and rng.random() < np.exp(-(score_new - best_score) / T_anneal):
                a = a_new
            step_size *= 0.85
            if step_size < 1e-12 / n:
                a_pert = np.maximum(a + rng.normal(0, 0.01 * a.mean(), n), 0)
                a = a_pert * norm / a_pert.sum()
                step_size = 0.3 / n

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    new_x = np.linspace(0.0, 1.0, new_n + 1)
    f_interp = np.interp(new_x, old_x, np.concatenate([f_norm, [0]]))
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        candidates = [
            np.sin(np.pi * x) ** 1.2,
            np.sin(np.pi * x) ** 1.5,
            (1.0 + np.cos(2 * np.pi * x)) / 2.0,
            np.exp(-4.0 * (x - 0.5) ** 2),
            np.ones(n_base) * 0.5 + 0.5 * np.sin(np.pi * x),
        ]
        for c in candidates:
            c = np.maximum(c, 0.0)
            s = c.sum()
            if s > 1e-10:
                c = c / s
                sc = evaluate_safe(c)
                if sc < best_score:
                    best_score = sc
                    best_seq = c.copy()
        print(f"Best init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting optimization from n={n_base}, score={best_score:.6f}")

    # === Phase 1: Log-sum-exp Adam (fast initial improvement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_adom_optimize(a_norm, t1, verbose=True)
        do_try_update(a_out, f"lse1_n{n_cur}")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, max_active=150, gamma_initial=1.8)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 3: Multi-peak subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 3: MultiPeak SubGrad n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = multi_peak_subgrad(a_norm, t3, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: LP Nesterov r2 n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, max_active=100, gamma_initial=1.6)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: PeakShave n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_local(a_norm, t5, seed=777, verbose=True)
        do_try_update(a_out, f"peak1_n{n_cur}")

    # === Phase 6: Try longer length ===
    n2 = n_base + 150
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 6: LP Nesterov n={n2} (budget={t6:.0f}s) ===")
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, max_active=120, gamma_initial=1.8)
        do_try_update(a_out, f"lp_n{n2}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True, max_active=80, gamma_initial=1.5)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.12, 100.0)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_local(a_norm, t8, seed=31415, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
