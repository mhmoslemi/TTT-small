def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=2.0, n_stages=8, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        stage_iters = 0
        stage_best = best_score
        stage_stall = 0
        
        while time.monotonic() - t0 < time_limit:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 2:
                break
            # Max iterations per stage based on time
            if stage_iters > 5000:
                break
            
            obj_val, grad = lse_objective_and_grad(a.copy().ravel(), n, T)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new * norm_val / s_new
            
            score = fast_autocorr(a_new).max()
            if score < stage_best - 1e-10:
                stage_best = score
                stage_stall = 0
            else:
                stage_stall += 1
            a = a_new
            stage_iters += 1
            
            if stage_stall > 500:
                break
        
        if best_score - stage_best > 1e-8:
            best_score = stage_best
            best_a = a.copy()
            print(f"    Adam-LSE T={T:.1f}: score={best_score:.8f}")

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_tight_M=True, eps_base=0.03, eps_growth=0.002, eps_max=0.08):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 500000):

            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

            eps = min(eps_max, eps_base + eps_growth * np.sqrt(t + 1))
            M_used = M_curr * (1.0 - eps)
        else:
            M_used = M_curr

        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.05), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:

            else:
                a_new = a_new * norm_val / s_a

        a = a_new

        score_y = fast_autocorr(y).max()

            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):

        T = build_constraint_matrix(a, n)

def lse_obj_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

            v = 0.999 * v + 0.001 * grad ** 2
            m_hat = m / (1 - 0.9 ** t_adam)
            v_hat = v / (1 - 0.999 ** t_adam)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new * norm_val / s_new

            score = fast_autocorr(a_new).max()
            if score < stage_best - 1e-10:
                stage_best = score
                stage_stall = 0
            else:
                stage_stall += 1
            a = a_new
            stage_iters += 1

            if stage_stall > 800:
                break

        if best_score - stage_best > 1e-8:
            best_score = stage_best
            best_a = a.copy()
            print(f"    Adam-LSE T={T:.1f}: score={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, eps_base=0.04, eps_growth=0.003, eps_max=0.10):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        eps = min(eps_max, eps_base + eps_growth * np.sqrt(t + 1))
        M_used = M_curr * (1.0 - eps)

        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        M_new = fast_autocorr(g).max()
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.0)
            alpha = min(alpha / max(ratio, 0.03), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 120.0:
            stall_count += 1
            if stall_count > 6:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gradient_informed_cd(a0, time_limit, seed=42):
    """Coordinate descent with gradient-informed move selection and line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    T_smooth = 50.0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Compute smoothing weights (peak-focused)
        shifted = T_smooth * (b - max_b)
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        top_peaks = np.argsort(b)[::-1][:20]

        # Compute gradient of smoothed objective w.r.t. a
        fft_size = 1 << (2 * n - 1).bit_length()
        w_full = np.zeros(fft_size)
        w_full[:2 * n - 1] = w
        a_full = np.zeros(fft_size)
        a_full[:n] = a
        W_fft = rfft(w_full, fft_size)
        A_fft = rfft(a_full, fft_size)
        cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

        # Gradient suggests which coordinates to decrease (large positive cross)
        # and which to increase (large negative cross)
        # Focus on coordinates with |cross| above median
        med_cross = np.median(np.abs(cross))
        high_grad = np.where(cross > med_cross)[0]
        low_grad = np.where(cross < -med_cross * 0.5)[0]

        improved = False

        # Try moves guided by gradient: decrease high-grad, increase low-grad
        if len(high_grad) > 0 and len(low_grad) > 0:
            src_candidates = high_grad[np.argsort(a[high_grad])[::-1]][:20]
            dst_candidates = low_grad[np.argsort(a[low_grad])][:20]
            for src in src_candidates:
                if a[src] < 1e-10:
                    continue
                for dst in dst_candidates:
                    if dst == src:
                        continue
                    # Bisection line search
                    lo = 0.0
                    hi = a[src] * 0.4
                    best_frac = 0.0
                    best_s = best_score
                    for _ in range(7):
                        mid = (lo + hi) / 2.0
                        amt = a[src] * mid
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            hi = mid
                            continue
                        a_trial = a_trial * norm_val / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_s:
                            best_s = score
                            best_frac = mid
                            lo = mid
                        else:
                            hi = mid
                    if best_frac > 0 and best_s < best_score - 1e-13:
                        amt = a[src] * best_frac
                        a[src] -= amt
                        a[dst] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        a = a * norm_val / s_t
                        best_score = best_s
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            # Random pair moves with line search
            for _ in range(60):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                lo = 0.0
                hi = a[i] * 0.3
                best_frac = 0.0
                best_s = best_score
                for _ in range(6):
                    mid = (lo + hi) / 2.0
                    amt = a[i] * mid
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        hi = mid
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_s:
                        best_s = score
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid
                if best_frac > 0 and best_s < best_score - 1e-13:
                    amt = a[i] * best_frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    a = a * norm_val / s_t
                    best_score = best_s
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Multi-coordinate simultaneous perturbation
            for _ in range(30):
                k = int(rng.integers(2, 5))
                idx = rng.choice(n, size=k, replace=False)
                scale = float(rng.uniform(0.005, 0.03))
                delta = rng.standard_normal(k) * scale
                a_trial = a.copy()
                a_trial[idx] = a_trial[idx] * (1.0 + delta)
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 300:
                break

    return best_a, best_score


def sa_micro_search(a0, time_limit, seed=42):
    """Simulated annealing with very small moves to escape local minima."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    cur_score = best_score
    t0 = time.monotonic()

    T_init = (cur_score - best_score + 1e-6) * 0.1
    T_min = 1e-10
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        progress = (time.monotonic() - t0) / time_limit
        T = T_init * np.exp(-3.0 * progress)
        if T < T_min:
            T = T_min

        # Choose a small random perturbation
        k = int(rng.integers(1, 6))
        idx = rng.choice(n, size=k, replace=False)
        scale = float(rng.uniform(0.002, 0.02))
        delta = rng.standard_normal(k) * scale
        a_trial = a.copy()
        a_trial[idx] = a_trial[idx] * (1.0 + delta)
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            continue
        a_trial = a_trial * norm_val / s_t
        score = fast_autocorr(a_trial).max()

        if score < cur_score or rng.random() < np.exp(-(score - cur_score) / max(T, 1e-15)):
            a = a_trial
            cur_score = score
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_state, candidate_a, label, t_global):
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-Nesterov with adaptive M targets (primary)
    2. Adam-LSE for continuous refinement
    3. Gradient-informed coordinate descent (fine polish)
    4. SA micro-search for local minimum escape
    5. LP-Nesterov round 2
    6. Multi-length exploration
    7. Final combined polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov (main workhorse) ~38%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.38
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True,
                                     eps_base=0.05, eps_growth=0.004, eps_max=0.12)
    best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp1_n{n_best}", t_global)

    # Phase 2: Adam-LSE ~10%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Adam-LSE n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = adam_lse_optimize(a_norm, t2, T_start=100.0, T_end=3.0,
                                       n_stages=7, lr=0.006)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"adam_n{n_best}", t_global)

    # Phase 3: Gradient-informed CD ~20%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Grad-CD n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_informed_cd(a_norm, t3, seed=42)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"gcd1_n{n_best}", t_global)

    # Phase 4: SA micro-search ~8%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SA micro n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = sa_micro_search(a_norm, t4, seed=777)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"sa_n{n_best}", t_global)

    # Phase 5: LP Nesterov round 2 ~12%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-Nest r2 n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False,
                                         eps_base=0.06, eps_growth=0.003, eps_max=0.15)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp2_n{n_best}", t_global)

    # Phase 6: Multi-length exploration ~12%
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length (budget={remaining:.0f}s) ===")
    n_candidates = [n_best + 200, n_best + 400]
    n_candidates = [n for n in n_candidates if 800 <= n <= 3000]
    for n_try in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        t6 = min(remaining * 0.45, 180.0)
        if t6 < 50:
            break
        print(f"  Trying n={n_try} (budget={t6:.0f}s)")
        init = interpolate_sequence(best_seq, n_try)
        a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False,
                                         eps_base=0.05, eps_growth=0.003, eps_max=0.12)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_long_n{n_try}", t_global)

    # Phase 7: Final polish - CD then SA
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        half = remaining * 0.6
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_informed_cd(a_norm, half, seed=99999)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "cd_final", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            n_cur = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = sa_micro_search(a_norm2, remaining2, seed=31337)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "sa_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
