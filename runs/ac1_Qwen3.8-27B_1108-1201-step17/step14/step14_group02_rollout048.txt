"""

def get_solver():

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_T_sparse(y, n):
    """Build the sparse constraint matrix T for the LP.
    T[k,j] = y[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1) x n
    """
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if y[j] == 0:
            continue
        # T[k,j] = y[k-j] for k = j, j+1, ..., j+n-1
        # So k-j = 0, 1, ..., n-1
        # i.e., T[j+i, j] = y[i] for i = 0, ..., n-1
        i = np.arange(n)
        k = j + i
        rows.extend(k)
        cols.extend([j] * n)
        vals.extend(y)
    T = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T

def solve_lp_scipy(y, n, M, time_per_lp=5.0):
    """Solve the LP: max sum(b) s.t. T@b <= M, b >= 0 using scipy HiGHS.
    Returns the optimal b (normalized to sum=1) or None.
    """
    T = build_T_sparse(y, n)
    
    c = np.ones(n)  # minimize -sum(b) => maximize sum(b)
    A_ub = T
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(-c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s < 1e-15:
                return None
            return b / s
    except Exception as e:
        print(f"    LP error: {e}")
    return None

def solve_lp_cvxpy(y, n, M, solver):
    """Solve the LP using cvxpy (fallback)."""
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_sparse(y, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

    where C = conv(a,a)

    where p[k] = exp(beta*C[k]) / sum exp(beta*C[k])
    """

    best_m = fast_autocorr(a).max()

    # Continuation: gradually increase beta
    beta_start = 1.0
    beta_end = 100.0

    # Paired with step sizes
    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta = beta_start * (beta_end / beta_start) ** (phase / max(n_phases - 1, 1))
        phase_time = (time_limit - (time.monotonic() - t0)) / (n_phases - phase)
        
        # Within each phase, do gradient steps
        T = 0
        gamma = 0.5  # step size parameter

def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate push: move mass from high to low autocorrelation contribution positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top contributing lags
        top_k = np.argsort(ac)[::-1][:min(8, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find destination: position that minimizes impact on top peaks
            peak_indices = top_k[:5]  # Only consider top 5 peaks
            best_dst = -1
            best_score = float('inf')
            
            # Sample candidates
            n_candidates = min(n, 100)
            candidates = rng.integers(0, n, size=min(n, 500))
            unique = np.unique(candidates)[:n_candidates]
            
            for dst in unique:
                dst = int(dst)
                if dst == int(src):
                    continue
                # Score: how much does adding to dst increase the top peaks?
                sens = 0.0
                for k2 in peak_indices:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    sens += add
                
                # Reduction from removing from src
                rem_sens = 0.0
                for k2 in peak_indices:
                    k2i = int(k2)
                    d1 = k2i - int(src)
                    d2 = int(src) - k2i
                    rem = 0.0
                    if 0 <= d1 < n:
                        rem += a[d1]
                    if 0 <= d2 < n:
                        rem += a[d2]
                    rem_sens += rem
                
                net = sens - rem_sens
                if net < best_score:
                    best_score = net
                    best_dst = dst
            
            if best_dst < 0:
                continue
            dst = best_dst
            
            # Try step sizes
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[int(src)] -= amt
                a_trial[dst] += amt
                if a_trial[int(src)] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
