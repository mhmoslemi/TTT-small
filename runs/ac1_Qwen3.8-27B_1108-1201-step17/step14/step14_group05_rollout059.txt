def sliding_min(arr, window):
    from collections import deque
    result = np.full(len(arr) - window + 1, np.inf)
    dq = deque()
    for i in range(len(arr)):
        while dq and arr[dq[-1]] >= arr[i]:
            dq.pop()
        dq.append(i)
        if dq[0] <= i - window:
            dq.popleft()
        if i >= window - 1:
            result[i - window + 1] = arr[dq[0]]
    return result
```

    """Nesterov acceleration with Gauss-Seidel LP."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    b_prev = None  # warm start for LP
    
    gamma = 1.8
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()
        
        # Solve LP with Gauss-Seidel
        b = fast_lp_gauss_seidel(y, M_curr, b_init=b_prev, n_sweeps=n_sweeps)
        b_prev = b.copy()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()

3. **Spectral domain refinement**: After getting a good time-domain solution, work in the frequency domain:

   - The autocorrelation is IFFT(|A|^2)

Let me think about the spectral approach more carefully. We want to minimize max_k (a*a)[k]. In the frequency domain:
(a*a)[k] = IFFT(|A|^2)[k]

We want to minimize the max of IFFT(|A|^2). This is equivalent to minimizing the L_infinity norm of IFFT(|A|^2).

If |A|^2 were perfectly flat, then IFFT(|A|^2) would be a delta at lag 0, and the max would be at lag 0 with value |A|^2 / N (where N is the FFT length). All other lags would be 0. This is the ideal case.

But we have the constraint that a >= 0, which constrains what |A|^2 can be (it's the FFT of a non-negative sequence).

A practical approach:
1. Start with a good a
2. Compute A = FFT(a), S = |A|^2
3. Compute c = IFFT(S) = (a*a) / N (approximately, with zero-padding)
4. Find peaks in c
5. Modify S to reduce peaks: S_new = S - delta * (c - mean(c)) (clipped to be non-negative)
   - This is like a "peak reduction" in the spectral domain
6. Set A_new = sqrt(S_new) * exp(i * angle(A))
7. Set a_new = IFFT(A_new) (take real part, clip to non-negative)
8. Iterate

This is similar to the " iteratively reweighted" approach or the "spectral algorithm" for PAPR reduction.

Actually, I realize there's a subtlety: modifying |A|^2 changes the phase relationship, and the resulting a might not be non-negative. We need to clip to non-negative and renormalize, which might undo some of the progress.

A better approach: use the fact that we're minimizing max_k c_k where c = IFFT(|A|^2). The gradient with respect to |A|^2 is straightforward: d c_k / d S_m = exp(2πi mk/N) / N. So the subgradient of max_k c_k with respect to S is exp(2πi k* m/N) / N where k* is the argmax.

This gives a subgradient descent in the spectral domain:
S_new = S - step * exp(2πi k* m/N) / N (for real sequences, this has a specific form)

def spectral_refine(a, n_iters=100):

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Gauss-Seidel LP solver (O(n^2) per sweep vs. cvxpy's general LP solver)
2. Nesterov acceleration with adaptive parameters
3. Multi-length exploration (n = 1600, 2000, 3000)
4. LSE-smoothed gradient descent for fine-tuning
5. Targeted peak shaving for final polish
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def fast_lp_gs(a, M, b_init=None, n_sweeps=50, verbose=False):
    """Gauss-Seidel LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    if b_init is not None:
        b = np.maximum(b_init.copy(), 0)
        interf = np.convolve(a, b, 'full')
        if interf.max() > M * 1.001:
            b = b * (0.999 * M / interf.max())
    else:
        b = np.zeros(n)
    
    interf = np.convolve(a, b, 'full')
    slack = M - interf
    
    prev_sum = b.sum()
    
    for sweep in range(n_sweeps):
        for j in range(n):
            sl_window = slack[j:j+n]
            av = a
            valid = av > 1e-15
            if valid.any():
                increments = sl_window[valid] / av[valid]
                max_incr = float(np.min(increments))
                if max_incr > 1e-15:
                    b[j] += max_incr
                    slack[j:j+n] -= av * max_incr
        
        curr_sum = b.sum()
        if curr_sum - prev_sum < 1e-12 * max(curr_sum, 1.0):
            break
        prev_sum = curr_sum
    
    return b

def nesterov_lp_gs(a0, time_limit, n_sweeps=30, verbose=True):
    """Nesterov LP with Gauss-Seidel solver."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    b_prev = None
    gamma = 1.8
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = fast_lp_gs(y, M_curr, b_init=b_prev, n_sweeps=n_sweeps)
        b_prev = b.copy()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        if stall > 150:
            rng = np.random.default_rng(n_iters * 31 + 17)
            perturb = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb, n), 0)
            y = y * norm / y.sum()
            a = y.copy()
            prev_y = y.copy()
            b_prev = None
            stall = 0
            gamma = 1.8
        
        # Adaptive gamma
        if n_iters > 100 and n_iters % 100 == 0:
            if stall > 50:
                gamma = max(1.0, gamma - 0.1)
            elif stall < 10:
                gamma = min(2.0, gamma + 0.05)
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS-NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, gamma={gamma:.2f}")
    
    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = max(0.5, best_score * 0.05)
    T_end = max(0.001, best_score * 0.001)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Compute softmax weights
        shifted = (b - max_b) / T
        # Clip to avoid underflow
        shifted = np.maximum(shifted, -50)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient via FFT
        # grad_j = 2 * sum_k w_k * a[k-j] (for valid k-j)
        # This is 2 * (w * a_rev) restricted to valid range
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a[::-1], fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n-1:2*n-1]
        
        # Project onto sum=norm hyperplane
        grad -= np.mean(grad)
        
        step = 1.0 / (n * 0.5)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score

def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find top contributor to the peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 500:
                break
            continue
        
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        # Choose destination: random position far from peak
        dst = rng.integers(n)
        
        # Try different transfer fractions
        best_trial = max_b
        best_frac = 0
        for frac in [0.03, 0.05, 0.08, 0.12, 0.18, 0.25]:
            amt = src_val * frac
            if amt < 1e-18:
                break
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t < 1e-15:
                break
            a_t = a_t * norm / s_t
            sc = fast_autocorr(a_t).max()
            if sc < best_trial:
                best_trial = sc
                best_frac = frac
        
        if best_frac > 0:
            a_t = a.copy()
            a_t[src] -= src_val * best_frac
            a_t[dst] += src_val * best_frac
            a_t = np.maximum(a_t, 0)
            a_t = a_t * norm / a_t.sum()
            sc = fast_autocorr(a_t).max()
            
            if sc < best_score:
                a = a_t
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                T = best_score * 0.002 * (1.0 - elapsed_frac)
                if T > 1e-10 and rng.random() < np.exp(-(sc - best_score) / T):
                    a = a_t
                n_stall += 1
        else:
            n_stall += 1
        
        if n_stall > 200:
            # Random perturbation
            n_pert = max(3, n // 100)
            srcs = rng.choice(n, size=n_pert, replace=False)
            dsts = rng.integers(n, size=n_pert)
            a_t = a.copy()
            for i, j in zip(srcs, dsts):
                if a_t[i] > 0:
                    amt = a_t[i] * rng.uniform(0.1, 0.3)
                    a_t[i] -= amt
                    a_t[j] += amt
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t > 1e-15:
                a_t = a_t * norm / s_t
                sc = fast_autocorr(a_t).max()
                if sc < best_score * 1.01:
                    a = a_t
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0
        
        if verbose and n_improve > 0 and n_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_n = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-15:
        f_interp = np.ones(new_n) / new_n
    else:
        f_interp = f_interp / s
    return f_interp
