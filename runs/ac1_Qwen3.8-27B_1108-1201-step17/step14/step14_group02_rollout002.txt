"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov mirror descent with exact steepest-descent subgradient (primary)
2. Adaptive step size with progress tracking
3. LP-based Frank-Wolfe for polishing (secondary)
4. Coordinate descent for final refinement
5. Multi-length exploration and multi-initialization

import numpy as np
import time
from scipy.fft import rfft, irfft
import scipy.optimize as sco

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]

def compute_lag_gradient(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a. Result: 2*a[k-i] for valid i."""
    n = len(a)
    k = int(k)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def steepest_descent_subgrad(a, ac, tol=1e-10, max_active=60):
    """Compute the steepest-descent subgradient of max_k ac[k].
    
    If |active set| is small, solve the QP exactly.
    Otherwise, use soft weighting.
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    # Find active set
    if tol is None:
        tol = M * 1e-8
    active_mask = ac >= (M - tol)
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(ac)])
    
    if len(active_idx) == 1:
        return compute_lag_gradient(a, active_idx[0])
    
    if len(active_idx) <= max_active:
        # Solve QP exactly
        m = len(active_idx)
        grads = np.zeros((m, n))
        for i, k in enumerate(active_idx):
            grads[i] = compute_lag_gradient(a, k)
        
        # G = grads @ grads.T
        G = grads @ grads.T
        
        # Solve: min lambda^T G lambda s.t. lambda >= 0, sum lambda = 1
        # Using scipy
        def qf(lam):
            return lam @ G @ lam
        def qf_grad(lam):
            return 2 * G @ lam
        
        cons = [{'type': 'eq', 'fun': lambda lam: lam.sum() - 1}]
        bounds = [(0, 1)] * m
        lam0 = np.ones(m) / m
        res = sco.minimize(qf, lam0, jac=qf_grad, method='SLSQP',
                          bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
        if res.success and res.fun < 1e30:
            lam = np.maximum(res.x, 0)
            lam_sum = lam.sum()
            if lam_sum > 1e-15:
                lam = lam / lam_sum
                return lam @ grads
        # Fallback to uniform
        return grads.mean(axis=0)
    else:
        # Soft weighting
        active_vals = np.maximum(ac[active_idx], 0)
        beta = 8.0
        w = active_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(active_idx)) / len(active_idx)
        else:
            w = w / w_sum
        grad = np.zeros(n)
        for wi, k in zip(w, active_idx):
            grad += wi * compute_lag_gradient(a, k)
        return grad

def nesterov_mirror_descent(a0, time_limit, verbose=True, eta0=0.5, tol=1e-10):
    """Nesterov mirror descent on the simplex with exact subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov point
    x_prev = a.copy()
    avg_a = a.copy()
    n_avg = 1
    
    eta = eta0
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        if t == 1:
            y = a.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y = a + beta * (a - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        
        x_prev = a.copy()
        
        # Compute subgradient at y
        ac = fast_autocorr(y)
        g = steepest_descent_subgrad(y, ac, tol=tol)
        
        # Mirror descent step (exponential / multiplicative)
        # a_new[i] = y[i] * exp(-eta * g[i]) / Z
        log_step = -eta * g
        # Clip to avoid overflow
        log_step = np.clip(log_step, -50, 50)
        a_new = y * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        # Update running average
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            avg_check = norm_sum1(avg_check)
            M_avg = fast_autocorr(avg_check).max()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg_check.copy()
        
        # Adaptive step size
        if stall_count > 200:
            eta *= 0.7
            stall_count = 0
        elif stall_count > 0 and t % 5000 == 0:
            eta *= 1.05  # slowly increase if making progress
        
        # Periodic restart perturbation
        if t > 1000 and t % 10000 == 0 and stall_count > 500:
            # Small random perturbation
            idx = rng.integers(n)
            amt = a[idx] * 0.01
            a[idx] -= amt
            jdx = rng.integers(n)
            a[jdx] += amt
            a = norm_sum1(a)
            y = a.copy()
            x_prev = a.copy()
            stall_count = 0
            eta = max(eta, 0.1)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}, stall={stall_count}")
    
    return best_a, best_m

            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Shape: (2n-1) x n
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j+n, j] = y

            prob.solve(solver=cp.CLARABEL, verbose=False)
