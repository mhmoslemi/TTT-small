def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_sparse_T(a, n):
    """Build sparse constraint matrix T where T[k,i] = a[k-i] if 0<=k-i<n else 0."""
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # T[k, i] = a[k-i] for k where 0 <= k-i < n, i.e., i <= k < i+n
        # and k must be in [0, 2n-2]
        k_start = i
        k_end = min(i + n, size)
        ks = np.arange(k_start, k_end)
        js = ks - i  # = 0, 1, ..., k_end-1-i
        rows.extend(ks.tolist())
        cols.extend(i * [i])
        vals.extend(a[js].tolist())
    T = sp.csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def lp_best_response_sparse(a, M, solver, T_sparse):
    """Solve LP best response using sparse constraint matrix."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # T @ b <= M * ones
    constraints = [T_sparse @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, m_reduction=0.02):

    n_iters = 0

        b_sum = b.sum()

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:

        else:

            a_new = g.copy()
        else:

            if s_a < 1e-15:

        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:

            last_improve = time.monotonic()

    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, m_factor_init=0.98, m_factor_end=0.90):
    """LP best-response with Nesterov acceleration and adaptive M reduction."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve_t = time.monotonic()
    stall_count = 0

    max_iters = 500000
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-14:
            break

        # Adaptive M factor: start gentle, get more aggressive
        m_factor = m_factor_init + (m_factor_end - m_factor_init) * progress
        M_used = M_curr * m_factor

        T_sparse = build_sparse_T(y, n)
        b = lp_best_response(y, M_used, solver, T_sparse)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T_sparse)
            if b is None:
                b = lp_best_response(y, M_curr * 1.001, solver, T_sparse)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve_t = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve_t = time.monotonic()

        # Polyak averaging
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve_t = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve_t > 40.0:
            stall_count += 1
            if stall_count > 10:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}, t={elapsed:.0f}s")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 500 == 0:
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gradient_flow_optimize(a0, time_limit, verbose=True):
    """Gradient flow on LSE-smoothed max autocorrelation with adaptive step size."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = 1.0 / n
    no_improve = 0
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()
        if M_curr < 1e-14:
            break

        # LSE temperature: start high, decrease
        progress = min((time.monotonic() - t0) / time_limit, 1.0)
        T_temp = 50.0 * (1.0 - progress) + 2.0

        shifted = (b - M_curr) / T_temp
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()

        # Gradient: grad[i] = sum_k w[k] * (a[k-i]*1[0<=k-i<n] + a[i-k]*1[0<=i-k<n])
        # Use only significant weights for speed
        threshold = 1e-5
        peak_positions = np.where(w > threshold)[0]

        grad = np.zeros(n)
        for k in peak_positions:
            wk = w[k]
            # Term 1: a[i+k] for i in [0, n-1-k]
            if k < n:
                grad[:n - k] += wk * a[k:]
            # Term 2: a[i-k] for i in [k, n-1]
            if k < n:
                grad[k:] += wk * a[:n - k]

        # Normalize gradient and take step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-10:
            a_trial = a.copy() * 0.99 + 0.01 * np.ones(n) / n
            s_trial = a_trial.sum()
        a_trial = a_trial * norm_val / s_trial

        score = fast_autocorr(a_trial).max()
        n_iters += 1

        if score < best_score * (1.0 + 1e-8):
            a = a_trial.copy()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                # Increase learning rate slightly on improvement
                lr = min(lr * 1.01, 5.0 / n)
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Decrease learning rate on failure
            lr *= 0.95

        if no_improve > 200:
            # Reset to best and try different lr
            a = best_a.copy()
            lr = 2.0 / n
            no_improve = 0

        if time.monotonic() - t0 > time_limit:
            break

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradFlow iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    """Targeted coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        # Find top contributing pairs for the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:20]

        # Find peak positions
        top_peaks = np.argsort(b)[::-1][:20]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:20]

        improved = False

        # Try moving mass from peak contributors to low-value positions
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        # If no targeted improvement, try random moves
        if not improved:
            for _ in range(100):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = float(rng.uniform(0.01, 0.15))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def lse_scipy_optimize(a0, time_limit, T_start=60.0, T_end=5.0, n_stages=3):
    """LSE smoothed optimization with L-BFGS-B."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def lse_obj(x, T_val=T):
            a_x = np.maximum(x, 0.0)
            s = a_x.sum()
            if s < 1e-10:
                return 1e10
            b = fast_autocorr(a_x)
            max_b = b.max()
            if max_b < 1e-15:
                return 1e10
            val = max_b + np.log(np.sum(np.exp((b - max_b) / T_val))) / T_val
            return 2.0 * n * val / (s * s)

        def lse_grad(x, T_val=T):
            a_x = np.maximum(x, 0.0)
            s = a_x.sum()
            if s < 1e-10:
                return np.zeros(n)
            b = fast_autocorr(a_x)
            max_b = b.max()
            if max_b < 1e-15:
                return np.zeros(n)
            shifted = (b - max_b) / T_val
            exp_vals = np.exp(shifted)
            log_sum = max_b + np.log(exp_vals.sum()) / T_val
            w = exp_vals / exp_vals.sum()

            # Compute gradient efficiently
            threshold = 1e-5
            peak_positions = np.where(w > threshold)[0]
            grad_direct = np.zeros(n)
            for k in peak_positions:
                wk = w[k]
                if k < n:
                    grad_direct[:n - k] += wk * a_x[k:]
                    grad_direct[k:] += wk * a_x[:n - k]

            obj_val = 2.0 * n * log_sum / (s * s)
            grad_final = 2.0 * n * grad_direct / (s * s) - 2.0 * obj_val * a_x / s
            return grad_final

        try:
            res = scipy_minimize(lse_obj, a.copy(), method='L-BFGS-B',
                                  jac=lse_grad,
                                  options={'maxiter': 150, 'maxfun': 400,
                                           'gtol': 1e-7, 'ftol': 1e-10,
                                           'maxls': 30})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x) + 0.1 * np.sin(10 * np.pi * x)
    else:
        a = np.ones(n) * (1.0 + 0.2 * np.sin(2 * np.pi * np.arange(n) / n * 3))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    """Try to update best state with candidate."""
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

        n_cur = len(best_seq)
        print(f"\n=== Phase 8: Final coord descent n={n_cur} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, remaining - 10, seed=98765)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out,
                                               "coord_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
