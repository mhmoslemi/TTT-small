    risk_analysis = risk_assessment(n_vals, improvement_threshold, tpar_range)
    result = search_hierarchy(init_seq, risk_analysis)
    
    # Analyze final form
    try:
        final_score = evaluate_sequence(result)
        compare_heading = ""
        benchmark = 1.5027
        if final_score < benchmark:
            compare_heading += "NEW RECORD!"
        return f"Improved to: {final_score:.6f}"
    except:
        return "Boundary hit during evaluation"
