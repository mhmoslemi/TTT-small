        return None
    b = np.maximum(res.x, 0)
    if b.sum() < 1e-15:
        return None

def nesterov_lp_search(a0, time_limit, rows, cols, a_idx, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s  # normalize so sum = sqrt(2n)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    
    t0 = time.monotonic()
    iteration = 0
    last_improve = t0
    
    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-10:
            break
        
        b = lp_best_response_fast(y, M_curr, rows, cols, a_idx)
        if b is None:
            break
        
        g = b * norm / b.sum()
        iteration += 1
        
        # Nesterov step with line search
        alpha = 2.0 / (iteration + 2.0)
        # Try a few step sizes
        best_alpha = alpha
        best_obj = fast_autocorr(np.maximum((1 - alpha) * y + alpha * g, 0)).max()
        for trial in [alpha * 0.5, alpha * 1.5, alpha * 2.0]:
            trial = min(trial, 1.0)
            cand = np.maximum((1 - trial) * y + trial * g, 0)
            sc = fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial
        
        y_new = np.maximum((1 - best_alpha) * y + best_alpha * g, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_new
        
        # Nesterov extrapolation
        if iteration > 1:
            momentum = (iteration - 1.0) / (iteration + 2.0)
            a_next = y_new + momentum * (y_new - y)
            a_next = np.maximum(a_next, 0)
            s_a = a_next.sum()
            if s_a > 1e-15:
                a_next = a_next * norm / s_a
        else:
            a_next = g.copy()
        
        y = y_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        score_a = fast_autocorr(a_next).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_next.copy()
            last_improve = time.monotonic()
        
        # Running average
        avg_a += a_next
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()
        
        # Restart if stalled
        if time.monotonic() - last_improve > 120:
            y = best_a.copy()
            a_next = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            last_improve = time.monotonic()
            if verbose:
                print(f"    Restart at iter={iteration}")
        
        if verbose and iteration % 50 == 0:
            print(f"    Iter {iteration}: best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score

_structure_cache = {}

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_conv(a, b):
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def get_structure(n):
    if n in _structure_cache:
        return _structure_cache[n]
    size = 2 * n - 1
    total_nz = n * n
    rows = np.empty(total_nz, dtype=np.int64)
    cols = np.empty(total_nz, dtype=np.int64)
    a_idx = np.empty(total_nz, dtype=np.int64)
    idx = 0
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(j_start, j_end + 1)
        a_idx[idx:idx+count] = k - np.arange(j_start, j_end + 1)
        idx += count
    _structure_cache[n] = (rows, cols, a_idx)

    size = 2 * n - 1
    return csc_matrix((vals, (rows, cols)), shape=(size, n))

def lp_best_response(a, M, rows, cols, a_idx):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a, rows, cols, a_idx)
    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    if res.status != 0:
        return None
    b = np.maximum(res.x, 0)
    if b.sum() < 1e-15:
        return None
    return b

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def nesterov_lp_search(a0, time_limit, verbose=True):
    n = len(a0)
    rows, cols, a_idx = get_structure(n)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    iteration = 0
    last_improve = t0
    consecutive_fail = 0
    stall_counter = 0
    
    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-10:
            break
        
        t_lp_start = time.monotonic()
        b = lp_best_response(y, M_curr, rows, cols, a_idx)
        t_lp_end = time.monotonic()
        
        if b is None:
            consecutive_fail += 1
            if consecutive_fail > 10:
                break
            continue
        consecutive_fail = 0
        
        g = b * norm / b.sum()
        iteration += 1
        
        # Nesterov step with adaptive line search
        base_alpha = 2.0 / (iteration + 2.0)
        best_alpha = base_alpha
        best_obj = fast_autocorr(np.maximum((1 - base_alpha) * y + base_alpha * g, 0)).max()
        
        for mult in [0.5, 1.5, 2.0, 3.0]:
            trial = min(base_alpha * mult, 1.0)
            if abs(trial - best_alpha) < 1e-6:
                continue
            cand = np.maximum((1 - trial) * y + trial * g, 0)
            sc = fast_autocorr(cand).max()
            if sc < best_obj:
                best_obj = sc
                best_alpha = trial
        
        y_new = np.maximum((1 - best_alpha) * y + best_alpha * g, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_new
        
        # Nesterov extrapolation
        if iteration > 1:
            momentum = (iteration - 1.0) / (iteration + 2.0)
            a_next = y_new + momentum * (y_new - y)
            a_next = np.maximum(a_next, 0.0)
            s_a = a_next.sum()
            if s_a > 1e-15:
                a_next = a_next * norm / s_a
        else:
            a_next = g.copy()
        
        y = y_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
            stall_counter = 0
        else:
            stall_counter += 1
        
        score_a = fast_autocorr(a_next).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_next.copy()
            last_improve = time.monotonic()
            stall_counter = 0
        
        # Running average (Feynman average)
        avg_a += a_next
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()
                    stall_counter = 0
        
        # Restart if stalled for too long
        if time.monotonic() - last_improve > 180:
            y = best_a.copy()
            a_next = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            last_improve = time.monotonic()
            if verbose:
                print(f"    [NLP] Restart at iter={iteration}, t={time.monotonic()-t0:.0f}s")
        
        if stall_counter > 5000:
            break
        
        if verbose and iteration % 100 == 0:
            lp_time = (time.monotonic() - t0) / max(1, iteration)
            print(f"    [NLP] iter={iteration}: best={best_score:.8f}, "
                  f"LP_time={lp_time*1000:.1f}ms/iter, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score

def lse_objective_and_grad(x, n, T_temp):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    
    # Cross-correlation: c_i = sum_k w_k * a_{k-i}
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)[:n]
    
    conv_grad = 2.0 * cross  # d_lse/da
    
    obj_val = 2.0 * n * lse / (s * s)
    grad = 2.0 * n * conv_grad / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad
