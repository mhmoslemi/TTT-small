import time

# Global solver
_solver = None

def get_solver():

    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:

            p.solve(solver=s)

        except Exception:

    _solver = cp.SCS

def fft_conv_self(a):
    n = len(a)
    size = 2*n - 1
    fs = 1 << (size-1).bit_length()
    A = rfft(a, fs)
    return irfft(A*A, fs)[:size]

def fft_conv_cross(a, b):
    n = len(a)
    size = 2*n - 1
    fs = 1 << (size-1).bit_length()
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A*B, fs)[:size]

def build_active_matrix(a, active):
    """Build T[active, :] where T[k,i] = a[k-i] if valid."""
    n = len(a)
    mat = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(k + 1, n)
        mat[idx, i_lo:i_hi] = a[k-i_hi+1 : k-i_lo+1][::-1]
    return mat

def solve_lp(a, M, active, solver):
    """max sum(b) s.t. T_active @ b <= M, b >= 0"""
    n = len(a)
    T = build_active_matrix(a, active)
    b = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    prob.solve(solver=solver, verbose=False, **{'tol_abs': 1e-9, 'tol_rel': 1e-9})
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b.value, dtype=np.float64)
    val[np.isnan(val)] = 0.0
    return np.maximum(val, 0.0)

                        best_delta = d
                        best_j = j
            
            if best_j >= 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = a * norm / a.sum()
                v = fft_conv_self(a).max()
                if v < best_mc:
                    best_mc = v
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            # Perturb
            if no_improve <= 20:
                npert = max(3, n // 100)
                idx = rng.choice(n, npert, replace=False)
                a[idx] *= (1 + 0.2 * rng.standard_normal(npert))
                a = np.maximum(a, 0)
                a = a * norm / a.sum()
        else:
            no_improve = 0
    
    return best_a
