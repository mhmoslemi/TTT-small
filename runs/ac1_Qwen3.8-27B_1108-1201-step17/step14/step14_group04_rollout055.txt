def fast_autocorr(a):
    """Compute full autocorrelation a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_triple_conv(a):
    """Compute a*a*a (triple convolution) via FFT, return first n elements."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A * A, fft_size)[:n]


def pnorm_autocorr_gradient(a, p):
    """
    Gradient of ||a*a||_p^p = sum_k (a*a)_k^p with respect to a.
    Result: 2 * (c^{[p-1]} * a) sliced to length n, where c = a*a.
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    c = fast_autocorr(a)
    if p == 2:
        # grad = 4 * (a*a*a)[0:n]
        grad = 4.0 * fast_triple_conv(a)
    else:
        c_pow = c ** (p - 1)
        # convolve c_pow with a, take appropriate slice
        size_c = len(c_pow)
        size = size_c + n - 1
        fft_size = 1 << (size - 1).bit_length()
        Cp = rfft(c_pow, fft_size)
        A = rfft(a, fft_size)
        result = irfft(Cp * A, fft_size)[:size]
        # We need (c_pow * a)[i] for i = 0..n-1
        # (c_pow * a)[i] = sum_j c_pow[j] * a[i-j]
        grad = 2.0 * result[:n]
    return grad


def project_nonneg_sum(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def pnorm_fista(a0, time_limit, p_schedule=None, verbose=True):
    """
    FISTA optimization of ||a*a||_p subject to a >= 0, sum(a) = 1.
    p_schedule: list of p values to anneal through.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128]

    best_a = a.copy()
    best_pnorm = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Estimate Lipschitz constant: L ~ n * p * (max a)^(p-1) * max((a*a))^(p-1)
    # Use adaptive step size instead
    for p in p_schedule:
        # Determine time for this p
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining <= 5:
            break

        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < 1e-15:
            break
        # Lipschitz constant estimate for ||a*a||_p^p: L ~ 2*p*n*c_max^(p-1)
        L = 2.0 * p * n * max(c_max, 1e-10) ** (p - 1)
        step = 1.0 / L
        # Cap step to avoid overshooting
        step = min(step, 1.0 / (n * 0.1))

        y = a.copy()
        a_prev = a.copy()
        t_fista = 1

        while time.monotonic() - t0 < time_limit:
            grad = pnorm_autocorr_gradient(y, p)
            z = y - step * grad
            a_new = project_nonneg_sum(z, 1.0)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - a_prev)
                y_new = project_nonneg_sum(y_new, 1.0)

            a_prev = a_new
            a = a_new
            y = y_new
            t_fista += 1

            if t_fista % 200 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_pnorm:
                    best_pnorm = sc
                    best_a = a.copy()
                    if verbose:
                        print(f"    p-norm FISTA p={p} iter={t_fista}: max_acorr={sc:.8f}")

        # Final check
        sc = fast_autocorr(a).max()
        if sc < best_pnorm:
            best_pnorm = sc
            best_a = a.copy()

    return best_a, best_pnorm


def lp_best_response_active(a, M, K_max, solver):
    """
    LP: maximize sum(b) s.t. (a*b)_k <= M for top K constraints, b >= 0.
    Uses active-set reduction: only keep top K values of (a*a) as constraints.
    """

    aa = fast_autocorr(a)
    # Select top K constraints
    K = min(K_max, 2 * n - 1)
    top_indices = np.argsort(aa)[::-1][:K]

    # Build constraint matrix for selected constraints only
    # T[k, j] = a[j] if 0 <= k-j < n, else 0
    T = np.zeros((K, n))
    for idx, k in enumerate(top_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        T[idx, js:je + 1] = a[js:je + 1]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_minimax_active(a0, time_limit, solver, K_max=500, verbose=True):
    """
    LP minimax iteration with active-set constraint reduction.
    Uses ergodic averaging and adaptive K.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    # Ergodic average
    ergodic_a = a.copy()
    n_ergodic = 1

    # Nesterov momentum
    y = a.copy()
    t = 1

    for _ in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Adaptive K: start small, increase over time
        progress = min(1.0, n_iters / max(1, int(time_limit / 0.1)))
        K = int(max(50, K_max * (0.3 + 0.7 * progress)))

        b = lp_best_response_active(y, M_curr, K, solver)
        if b is None:
            # Try with fewer constraints
            b = lp_best_response_active(y, M_curr, 50, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b / b_sum  # normalize to sum=1
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new
        t += 1

        # Ergodic average
        ergodic_a += a
        n_ergodic += 1

        # Check scores periodically
        if n_iters % 5 == 0:
            sc_y = fast_autocorr(y).max()
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()

            sc_a = fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()

        if n_iters % 20 == 0:
            ergodic_a_norm = ergodic_a / n_ergodic
            sc_e = fast_autocorr(ergodic_a_norm).max()
            if sc_e < best_score:
                best_score = sc_e
                best_a = ergodic_a_norm.copy()

            if verbose and time.monotonic() - t0 > 2:
                print(f"    LP-active iter={n_iters} K={K} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    # Final check of ergodic average
    ergodic_a_norm = ergodic_a / n_ergodic
    sc_e = fast_autocorr(ergodic_a_norm).max()
    if sc_e < best_score:
        best_score = sc_e
        best_a = ergodic_a_norm.copy()

    return best_a, best_score


def targeted_local_search(a0, time_limit, seed=42, verbose=True):
    """
    Gradient-guided local search: use p-norm gradient to identify
    where to move mass to reduce peaks.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    # Use p=8 gradient for guidance
    p = 8
    c = fast_autocorr(a)
    c_max = c.max()

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < 1e-12:
            break

        # Compute gradient of ||c||_p^p
        grad = pnorm_autocorr_gradient(a, p)

        # Find peak positions in autocorrelation
        top_k = np.argsort(c)[::-1][:8]

        improved = False
        for k in top_k:
            # The peak at position k in a*a is contributed by pairs (j, k-j)
            # We want to reduce this peak by moving mass
            # Gradient tells us: to reduce c_k, we should decrease a where grad is high
            # and increase a where grad is low

            # Find the "worst" contribution to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            # Contribution of a[j] to c_k is a[j] * a[k-j]
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            # Destination: where gradient is low (removing mass there helps least)
            # But we also need non-negativity
            grad_sorted = np.argsort(grad)
            candidates = grad_sorted[grad_sorted != src][:max(1, n // 4)]
            dst = candidates[rng.integers(len(candidates))]

            # Try several step sizes
            for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape local minimum
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            else:
                stall += 1
                if stall > 500:
                    break

        if verbose and time.monotonic() - t0 > 5 and int(time.monotonic() - t0) % 5 == 0:
            print(f"    Local search ({time.monotonic()-t0:.0f}s): best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def spectral_flatness_opt(a0, time_limit, seed=42, verbose=True):
    """
    Optimize for flat power spectrum |A|^2, which corresponds to
    flat autocorrelation. Use projected gradient in frequency domain.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    pad_len = 1 << (2 * n - 1).bit_length()
    A_full = rfft(a, pad_len)
    power = np.abs(A_full) ** 2

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < 1e-12:
            break

        # Compute power spectrum
        A_full = rfft(a, pad_len)
        power = np.abs(A_full) ** 2
        power_mean = power.mean()

        # We want to flatten the power spectrum
        # Gradient of variance of power w.r.t. a:
        # var(power) = mean(|A|^4) - mean(|A|^2)^2
        # d(mean |A|^4)/da = 4 * Re(INVDFT(|A|^2 * A))
        # d(mean |A|^2)/da = 2 * a (by Parseval)

        # Simplified: use gradient of (max power - mean power)
        # Find peak frequency bins
        peak_idx = np.argmax(power)
        peak_val = power[peak_idx]

        if peak_val < power_mean * 1.5:
            # Spectrum is already fairly flat
            stall += 1
            if stall > 300:
                break
            # Try small random perturbations
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.01)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            continue

        # Reduce peak bins by adjusting phases in frequency domain
        top_k = np.argsort(power)[::-1][:4]
        A_new = A_full.copy()
        for ki in top_k:
            if ki == 0:
                continue
            if ki > pad_len // 2:
                continue
            mag = np.abs(A_new[ki])
            if mag < 1e-15:
                continue
            # Reduce magnitude towards mean
            target_mag = np.sqrt(power_mean)
            new_mag = mag + 0.05 * (target_mag - mag)
            new_mag = max(new_mag, 0.0)
            phase = np.angle(A_new[ki])
            A_new[ki] = new_mag * np.exp(1j * phase)

        a_new = irfft(A_new, pad_len)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            stall += 1
            if stall > 500:
                break
            continue
        a_new = a_new / s_new

        sc = fast_autocorr(a_new).max()
        if sc < best_score - 1e-14:
            a = a_new
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

        if verbose and time.monotonic() - t0 > 5 and int(time.monotonic() - t0) % 5 == 0:
            print(f"    Spectral ({time.monotonic()-t0:.0f}s): best={best_score:.8f}, peak/mean={peak_val/max(power_mean,1e-15):.3f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines p-norm FISTA, active-set LP,
    targeted local search, and spectral optimization.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        # Use a non-trivial initialisation
        x = np.linspace(0, 1, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + 0.5)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # === Phase 1: p-norm FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 180.0)
    if t1 > 20:
        print(f"\n=== Phase 1: p-norm FISTA n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = pnorm_fista(best_seq, t1, p_schedule=[4, 8, 16, 32, 64], verbose=True)
        try_update(a_out, f"pnorm_n{n_base}")

    # === Phase 2: LP minimax with active sets (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-active n={n_best} (budget={t2:.0f}s) ===")
        a_out, _ = lp_minimax_active(best_seq, t2, solver, K_max=400, verbose=True)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Targeted local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Local search n={n_best} (budget={t3:.0f}s) ===")
        a_out, _ = targeted_local_search(best_seq, t3, seed=12345, verbose=True)
        try_update(a_out, f"local_n{n_best}")

    # === Phase 4: Spectral flatness ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 100.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Spectral n={n_best} (budget={t4:.0f}s) ===")
        a_out, _ = spectral_flatness_opt(best_seq, t4, seed=42, verbose=True)
        try_update(a_out, f"spectral_n{n_best}")

    # === Phase 5: LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP polish n={n_best} (budget={t5:.0f}s) ===")
        a_out, _ = lp_minimax_active(best_seq, t5, solver, K_max=600, verbose=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP at n={n2} (budget={t6:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_out, _ = lp_minimax_active(init_long, t6, solver, K_max=400, verbose=True)
        try_update(a_out, f"lp_n{n2}")

    # === Phase 7: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.05, 60.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Final local search n={n_best} (budget={t7:.0f}s) ===")
        a_out, _ = targeted_local_search(best_seq, t7, seed=99999, verbose=False)
        try_update(a_out, "final_local")

    # === Phase 8: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 20:
            print(f"\n=== Phase 8: Final LP n={n_best} (budget={t8:.0f}s) ===")
            a_out, _ = lp_minimax_active(best_seq, t8, solver, K_max=800, verbose=False)
            try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
