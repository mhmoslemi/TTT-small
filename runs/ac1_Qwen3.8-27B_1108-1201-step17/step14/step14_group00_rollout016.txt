def coord_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue
            base = conv.copy()
            a[i] = 0.0
            base = fast_autocorr(a)
            coeff = np.zeros(size)
            if i < n:
                lo = i
                hi = min(size, i + n)
                if hi > lo:
                    src = np.arange(lo, hi) - i
                    valid = (src >= 0) & (src < n)
                    if np.any(valid):
                        coeff[lo:hi][valid] = 2.0 * a[src[valid]]
            a[i] = ai
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0]:
                t_val = ai * frac
                t_val = max(0.0, t_val)
                new_conv = base + coeff * t_val
                diag = 2 * i
                if diag < size:
                    new_conv[diag] += t_val * t_val
                    new_conv[diag] -= ai * ai
                new_max = new_conv.max()
                if new_max < best_score - 1e-12:
                    a[i] = t_val
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = ai
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    break
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
        if verbose and n_passes % 5 == 0:
            print(f"    CD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True, n_pairs=6):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue
            for pa in range(n_pairs):
                j = int(rng.integers(n))
                if j == i:
                    continue
                g = np.zeros(size)
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    src1 = np.arange(lo1, hi1) - i
                    valid1 = (src1 >= 0) & (src1 < n)
                    if np.any(valid1):
                        g[lo1:hi1][valid1] -= a[src1[valid1]]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n + i)
                if hi2 > lo2:
                    src2 = n - 1 - np.arange(lo2, hi2)
                    valid2 = (src2 >= 0) & (src2 < n) & (src2 != i)
                    if np.any(valid2):
                        g[lo2:hi2][valid2] -= a[src2[valid2]]
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    src3 = np.arange(lo3, hi3) - j
                    valid3 = (src3 >= 0) & (src3 < n)
                    if np.any(valid3):
                        g[lo3:hi3][valid3] += a[src3[valid3]]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n + j)
                if hi4 > lo4:
                    src4 = n - 1 - np.arange(lo4, hi4)
                    valid4 = (src4 >= 0) & (src4 < n) & (src4 != i)
                    if np.any(valid4):
                        g[lo4:hi4][valid4] += a[src4[valid4]]
                best_d = 0.0
                best_val = conv.max()
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    d = ai * frac
                    new_max = (conv + d * g).max()
                    di = 2 * i
                    if di < size:
                        new_max = max(new_max, conv[di] - 2 * ai * d + d * d)
                    dj = 2 * j
                    if dj < size:
                        new_max = max(new_max, conv[dj] + 2 * a[j] * d + d * d)
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                    break
        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
        if verbose and n_passes % 5 == 0:
            print(f"    PCD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 4
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.2 + 0.8 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.4 * np.cos(6 * np.pi * x) + 0.15 * np.cos(12 * np.pi * x)
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.3 * (1 + 0.3 * np.cos(8 * np.pi * x))
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 / (1.0 + 5.0 * np.sin(np.pi * x) ** 2)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2) * (1.0 + 0.5 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_seq(a, rng, scale=0.05, mode="multiplicative"):
    n = len(a)
    if mode == "multiplicative":
        noise = 1.0 + scale * rng.standard_normal(n)
        noise = np.maximum(noise, 0.05)
        a_pert = a * noise
    elif mode == "additive":
        noise = scale / np.sqrt(n) * rng.standard_normal(n)
        a_pert = a + noise
        a_pert = np.maximum(a_pert, 0.0)
    else:
        # Frequency-domain perturbation
        fft_vals = rfft(a)
        mag = np.abs(fft_vals)
        phase = np.angle(fft_vals)
        mag_pert = mag * (1.0 + 0.1 * scale * rng.standard_normal(len(mag)))
        mag_pert = np.maximum(mag_pert, 0.0)
        fft_pert = mag_pert * np.exp(1j * phase)
        a_pert = irfft(fft_pert, n)
        a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def subgrad_fw(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))
        grad = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                v += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n and j2 != j1:
                v += y[j2]
            grad[i] = 2.0 * v
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new
        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
        if verbose and t % 100000 == 0:
            print(f"    FW t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    return best_a, best_score


def propose_candidate():

    t_global = time.monotonic()
    time_budget = 1750.0

    best_seq = None
    best_score = float('inf')
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 1: LP Nesterov n={n_base} (main) ===")
    t1 = min(remaining * 0.25, 420.0)

        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()

    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)
