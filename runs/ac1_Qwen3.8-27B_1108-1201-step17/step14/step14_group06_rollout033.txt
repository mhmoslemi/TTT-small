def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def normalize_sequence(a, n):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * np.sqrt(2.0 * n) / n
    return a * np.sqrt(2.0 * n) / s


def get_score(a):
    b = fast_autocorr(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    return 2.0 * len(a) * b.max() / (s * s)


def build_T_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def compute_lse_grad(a, T_temp):
    """Compute gradient of 2n*LSE_T(conv(a,a))/(sum(a))^2 using FFT."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n), 1e10
    
    b = fast_autocorr(a)  # length 2n-1
    Tb = T_temp * b
    Tb_shifted = Tb - Tb.max()
    exp_vals = np.exp(np.clip(Tb_shifted, -700, 0))
    p_sum = exp_vals.sum()
    if p_sum < 1e-300:
        return np.zeros(n), 1e10
    p = exp_vals / p_sum
    lse_val = b.max() + np.log(p_sum) / T_temp
    
    # cross[i] = sum_k p[k] * a[k-i] for i in [0, n-1]
    fft_size = 1 << (2 * n - 1).bit_length()
    p_pad = np.zeros(fft_size)
    p_pad[:2 * n - 1] = p
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    
    P_fft = rfft(p_pad, fft_size)
    A_fft = rfft(a_pad, fft_size)
    cross = irfft(P_fft * np.conj(A_fft), fft_size)[:n]
    
    obj_val = 2.0 * n * lse_val / (s * s)
    grad = 4.0 * n * cross / (T_temp * s * s) - 4.0 * n * lse_val * a / (s * s * s)
    return grad, obj_val


def lse_gradient_descent(a0, time_limit, T_start=20.0, T_end=200.0, n_stages=6):
    """LSE gradient descent with continuation on temperature."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_sequence(a0, n)
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = remaining / (n_stages - si)
        
        # Adaptive step size: start large, decrease
        alpha = min(1.0, 5.0 / (T * T))
        
        for it in range(10000):
            if time.monotonic() - t0 > time_limit:
                break
            grad, obj_val = compute_lse_grad(a, T)
            if np.all(np.abs(grad) < 1e-12):
                break
            
            # Gradient step with projection
            a_new = a - alpha * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                alpha *= 0.5
                continue
            a_new = a_new * norm_val / s_new
            
            score = get_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            # Adaptive step size
            if score < get_score(a):
                a = a_new
                alpha = min(alpha * 1.05, 2.0)
            else:
                alpha *= 0.8
                if alpha < 1e-6:
                    break
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_score


def lp_best_response(a, M_vec, solver, T_mat=None):
    """LP best response with per-lag M vector."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T_mat is None:
        T_mat = build_T_matrix(a, n)
    
    constraints = [T_mat @ b_var <= M_vec]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def coord_descent_linesearch(a0, time_limit, seed=42):
    """Coordinate descent with line search targeting peak contributions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_sequence(a0, n)
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        
        # Find contributing elements to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        top_contrib = idx[np.argsort(contrib)[::-1]][:10]
        
        # Find low-value elements to receive mass
        non_peak_mask = np.ones(n, dtype=bool)
        for k2 in np.argsort(b)[::-1][:20]:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            non_peak_mask[js2:je2+1] = False
        low_candidates = np.where(non_peak_mask)[0]
        if len(low_candidates) == 0:
            low_candidates = np.arange(n)
        low_sorted = low_candidates[np.argsort(a[low_candidates])][:10]
        
        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_sorted:
                if dst == src:
                    continue
                # Line search: try several fractions
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = get_score(a_trial)
                    if score < best_score - 1e-10:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbations
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = get_score(a_trial)
                if score < best_score - 1e-10:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an integrated approach:
1. FFT-based LSE gradient descent with continuation on temperature
2. LP best-response with per-lag weighted peak targeting + adaptive Nesterov
3. Line-search coordinate descent targeting peak-contributing elements
4. Multi-length exploration, perturbation restarts, and final polish
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_sequence(a, n):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * np.sqrt(2.0 * n) / n
    return a * np.sqrt(2.0 * n) / s


def get_score(a):
    b = fast_autocorr(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    return 2.0 * len(a) * b.max() / (s * s)


def build_T_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def compute_lse_grad(a, T_temp):
    """Compute gradient of 2n*LSE_T(conv(a,a))/(sum(a))^2 using FFT."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n), 1e10

    b = fast_autocorr(a)
    Tb = T_temp * b
    Tb_shifted = Tb - Tb.max()
    exp_vals = np.exp(np.clip(Tb_shifted, -700, 0))
    p_sum = exp_vals.sum()
    if p_sum < 1e-300:
        return np.zeros(n), 1e10
    p = exp_vals / p_sum
    lse_val = b.max() + np.log(p_sum) / T_temp

    fft_size = 1 << (2 * n - 1).bit_length()
    p_pad = np.zeros(fft_size)
    p_pad[:2 * n - 1] = p
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a

    P_fft = rfft(p_pad, fft_size)
    A_fft = rfft(a_pad, fft_size)
    cross = irfft(P_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * lse_val / (s * s)
    grad = 4.0 * n * cross / (T_temp * s * s) - 4.0 * n * lse_val * a / (s * s * s)
    return grad, obj_val


def lse_gradient_descent(a0, time_limit, T_start=15.0, T_end=200.0, n_stages=5):
    """LSE gradient descent with continuation on temperature."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_sequence(a0, n)
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = remaining / (n_stages - si)

        alpha = min(0.5, 3.0 / (T * T) * 10.0)
        stagnant = 0

        for it in range(20000):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - t0 > t0 + stage_time:
                break

            grad, obj_val = compute_lse_grad(a, T)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break

            a_new = a - alpha * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                alpha *= 0.5
                continue
            a_new = a_new * norm_val / s_new

            score = get_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                stagnant = 0
                a = a_new.copy()
                alpha = min(alpha * 1.1, 2.0)
            else:
                alpha *= 0.7
                stagnant += 1
                if stagnant > 30 or alpha < 1e-8:
                    break

    return best_a, best_score


def lp_best_response(a, M_vec, solver, T_mat=None):
    """LP best response with per-lag M vector."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T_mat is None:
        T_mat = build_T_matrix(a, n)

    constraints = [T_mat @ b_var <= M_vec]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, n_peaks=5, eps_base=0.02):
    """LP best-response with per-lag weighted constraints and adaptive Nesterov."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_sequence(a0, n)
    best_a = a.copy()
    best_score = get_score(a)

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        top_peaks = np.argsort(b_y)[::-1][:n_peaks]

        eps = min(0.06, eps_base * np.sqrt(t + 1) / 8.0)
        M_vec = np.full(2 * n - 1, M_curr)
        for k in top_peaks:
            M_vec[k] = M_curr * (1.0 - eps)

        T_mat = build_T_matrix(y, n)
        b = lp_best_response(y, M_vec, solver, T_mat)
        if b is None:
            M_vec_uniform = np.full(2 * n - 1, M_curr)
            b = lp_best_response(y, M_vec_uniform, solver, T_mat)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            x_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm_val / s_x

        y = y_new

        score_y = get_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_x = get_score(x_new)
        if score_x < best_score:
            best_score = score_x
            best_a = x_new.copy()
            last_improve = time.monotonic()

        if n_iters % 30 == 0:
            avg_check = (y + x_new) / 2.0
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = get_score(avg_check)
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 40.0:
            stall_count += 1
            if stall_count > 10:
                if verbose:
                    print(f"    LP stall at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_linesearch(a0, time_limit, seed=42):
    """Coordinate descent with line search targeting peak contributions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_sequence(a0, n)
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        top_contrib = idx[np.argsort(contrib)[::-1]][:12]

        non_peak_mask = np.ones(n, dtype=bool)
        top_peaks_arr = np.argsort(b)[::-1][:20]
        for k2 in top_peaks_arr:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            non_peak_mask[js2:je2 + 1] = False
        low_candidates = np.where(non_peak_mask)[0]
        if len(low_candidates) == 0:
            low_candidates = np.arange(n)
        low_sorted = low_candidates[np.argsort(a[low_candidates])][:12]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_sorted:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.04, 0.06, 0.1, 0.15, 0.25]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = get_score(a_trial)
                    if score < best_score - 1e-10:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
