The key constraint is: for non-negative sequences with sum = sqrt(2n), we want to minimize max_k conv(a,a)[k].

The "energy" constraint: by Cauchy-Schwarz, sum a[i]^2 >= (sum a)^2/n = 2. And conv[0] = a[0]^2, conv[2n-2] = a[n-1]^2.

The center: conv[n-1] = sum a[i]*a[n-1-i]. For non-negative a, this is >= 0 but can be made small if a[i] is large where a[n-1-i] is small.

The "off-center" peaks: conv[k] for k near n. These are the dangerous ones.

For a symmetric sequence (a[i] = a[n-1-i]), conv[n-1] = sum a[i]^2 >= 2. And by the uncertainty principle, there's a tradeoff between the width of the sequence and the width of its autocorrelation.

For a non-symmetric sequence, we can potentially reduce conv[n-1] below 2 (since a[i]*a[n-1-i] can be small if they're "misaligned"), but then other lags might have large values.

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def lse_smooth_max(conv, T):
    """Compute LSE smoothing of max and its weights."""
    max_c = conv.max()
    if max_c < 1e-15:
        return 0.0, np.ones(len(conv)) / len(conv)
    shifted = (conv - max_c) / T
    shifted = np.maximum(shifted, -50.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    return max_c + np.log(exp_c.sum()) / len(conv) * 0, w  # just return weights

def compute_gradient_lse(a, w):
    """Compute gradient of LSE-smoothed max autocorrelation."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # grad_i = 2 * sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # First term: sum_k w[k] * a[k-i] = cross-correlation
    g1 = irfft(np.conj(A) * W, fft_size)[:n]
    # Second term: sum_k w[k] * a[i-k] = convolution
    g2 = irfft(A * W, fft_size)[:n]
    return 2.0 * (g1 + g2)

def build_T_rows(active, a):
    """Build rows of the Toeplitz constraint matrix for active constraints only."""
    n = len(a)
    m = len(active)
    T = np.zeros((m, n))
    for idx, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        cols = np.arange(lo, hi + 1)
        T[idx, cols] = a[k - cols]
    return T

def lp_solve_fast(a, M, solver, active_indices):
    """Solve LP: max sum(b) s.t. T_active @ b <= M, b >= 0."""
    n = len(a)
    T = build_T_rows(active_indices, a)
    if T.shape[0] == 0:
        return None
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)
