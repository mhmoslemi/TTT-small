def build_conv_matrix_full(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_rows(a, k_indices):
    n = len(a)
    size = 2 * n - 1
    rows = []
    for k in k_indices:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a[idx]
        rows.append(row)
    return np.array(rows)


def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0 and len(active_set) < size:
        T = build_conv_matrix_rows(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        T = build_conv_matrix_full(a)
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_constraint_generation(a, M, solver, max_rounds=50):
    """LP with constraint generation: start with few constraints, add violated ones."""
    n = len(a)
    size = 2 * n - 1
    conv_a = fast_autocorr(a)

    # Initial active set: top constraints
    tol = 0.05
    active = np.where(conv_a > M * (1.0 - tol))[0]
    if len(active) < 3:
        active = np.argsort(conv_a)[-10:]

    b = None
    for round_idx in range(max_rounds):
        b = lp_best_response(a, M, solver, active)
        if b is None:
            # Fallback to full
            b = lp_best_response(a, M, solver, None)
            if b is None:
                return None
            return b

        # Check for violations in inactive constraints
        if len(active) < size:
            inactive = np.setdiff1d(np.arange(size), active)
            if len(inactive) > 0:
                T_check = build_conv_matrix_rows(a, inactive)
                violations = T_check @ b
                viol_mask = violations > M * (1 + 1e-6)
                if np.any(viol_mask):
                    # Add up to 20 most violated constraints
                    viol_indices = inactive[viol_mask]
                    viol_values = violations[viol_mask]
                    order = np.argsort(viol_values)[::-1]
                    new_constr = viol_indices[order[:20]]
                    active = np.union1d(active, new_constr)
                    if len(active) > 0.9 * size:
                        active = None  # Use full
                    continue
        # No violations, we have a valid solution
        return b
    return b


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=0):

    rng = np.random.default_rng(seed)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    else:
        a = a * norm / s0

    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    stall_count = 0
    stall_threshold = 80
    use_cg = True  # Use constraint generation

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:

        else:

            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:

def compute_change_vector(a, src, dst):
    """First-order change in conv when moving mass from src to dst."""
    n = len(a)
    size = 2 * n - 1
    g = np.zeros(size)
    # +a[k-dst] contribution
    lo = dst
    hi = min(dst + n, size)
    if lo < size:
        g[lo:hi] += a[:hi - lo]
    # -a[k-src] contribution
    lo = src
    hi = min(src + n, size)
    if lo < size:
        g[lo:hi] -= a[:hi - lo]
    return g


def find_optimal_delta(conv, g, max_delta):
    """Find delta in [0, max_delta] minimizing max(conv + delta*g)."""
    if max_delta <= 1e-15:
        return 0.0
    if np.all(g >= -1e-14):
        return 0.0
    if np.all(g <= 1e-14):
        return max_delta

    lo = 0.0
    hi = max_delta
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = (conv + m1 * g).max()
        f2 = (conv + m2 * g).max()
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-14:
            break
    delta = (lo + hi) / 2.0
    f0 = conv.max()
    f1 = (conv + delta * g).max()
    if f1 >= f0 - 1e-14:
        return 0.0
    return delta


def peak_shaving_cd(a0, time_limit, rng, verbose=True):
    """Targeted coordinate descent peaking at the autocorrelation maximum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # Normalize to sum=1

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    n_moves = 0

    while time.monotonic() - t0 < time_limit:
        k_star = int(np.argmax(conv))

        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)

        contributors = []
        for i in range(lo, hi):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        pass_improved = False

        for contrib, i, j in contributors[:40]:
            if time.monotonic() - t0 > time_limit:
                break
            if contrib < best_score * 0.005:
                break

            if a[i] >= a[j]:
                source = i
            else:
                source = j

            # Try several destinations
            for attempt in range(8):
                dest = int(rng.integers(n))
                if dest == source:
                    continue
                # Skip if dest also contributes to the peak
                if lo <= dest < hi:
                    dj = k_star - dest
                    if 0 <= dj < n and a[dest] * a[dj] > best_score * 0.05:
                        continue

                g = compute_change_vector(a, source, dest)
                max_delta = a[source] * 0.7
                delta = find_optimal_delta(conv, g, max_delta)

                if delta > 1e-12:
                    a[source] -= delta
                    a[dest] += delta
                    a = np.maximum(a, 0.0)
                    # Sum is preserved (moved, not added)
                    conv = fast_autocorr(a)
                    new_max = conv.max()

                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        n_moves += 1
                        break
                    else:
                        a[dest] -= delta
                        a[source] += delta
                        conv = fast_autocorr(a)

            if pass_improved:
                continue

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0

        if verbose and n_moves > 0 and n_moves % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PSCD moves={n_moves} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

                    a[i] -= delta
                    a[j] += delta
                    conv_new = fast_autocorr(a)
                    new_max = conv_new.max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        conv = conv_new
                        pass_improved = True
                        n_moves += 1
                        break
                    else:
                        a[j] -= delta
                        a[i] += delta
                if pass_improved:
                    break
            ai = a[i]  # Refresh

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_moves % 500 == 0 and n_moves > 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD moves={n_moves} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, choice=None):
    """Generate a structured initial sequence."""
    if choice is None:
        choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    """Add multiplicative noise and renormalize."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def swap_perturb(a, rng, n_swaps=10):
    """Perturb by swapping random pairs of values."""
    a = a.copy()
    n = len(a)
    for _ in range(n_swaps):
        i, j = rng.integers(0, n, 2)
        if i != j:
            a[i], a[j] = a[j], a[i]
    return a


def blend_perturb(a, rng, structured_fn, beta=0.3):
    """Blend with a structured sequence."""
    n = len(a)
    s_struct = structured_fn(n, rng)
    a_blend = (1.0 - beta) * a + beta * s_struct
    s = a_blend.sum()
    if s < 1e-15:
        return a.copy()
    return a_blend / s

        pass

        n_base = 1600
