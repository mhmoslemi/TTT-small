def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_active_matrix(a, active_lags):
    """Build the constraint matrix for active lags only.
    
    Returns A matrix where A[k, j] = a[j] if the (j, k-j) pair contributes to lag k.
    Specifically, (A @ b)[k] = conv(a, b)[k] for k in active_lags.
    """
    n = len(a)
    m = len(active_lags)
    # Use sparse matrix
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        for j in range(js, je + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A


def solve_lp(a, M, active_lags, method='highs'):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for k in active_lags, b >= 0.
    
    Returns (b_optimal, dual_variables) or (None, None) on failure.
    """
    n = len(a)
    A = build_active_matrix(a, active_lags)
    m = A.shape[0]
    if m == 0:
        return np.ones(n) / n, np.ones(2*n-1) / (2*n-1)
    
    b_ub = M * np.ones(m)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method=method)
        if not res.success:
            return None, None
        b = np.maximum(res.x, 0)
        # Dual variables (shadow prices) for the inequality constraints
        y = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else np.zeros(m)
        # Map back to full lag space
        y_full = np.zeros(2 * n - 1)
        for i, k in enumerate(active_lags):
            y_full[k] = y[i] if i < len(y) else 0
        return b, y_full
    except Exception:
        return None, None

def peak_shave(a0, time_limit, seed=42, verbal=True):
    """Peak shaving with simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()

    t0 = time.monotonic()
    n_iter = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find the main contributor to the peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-18:
            stall += 1
            if stall > 200:
                break
            continue
        
        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        # Find good destination (low value, far from peak)
        non_contrib = np.zeros(n, dtype=bool)
        if js > 0:
            non_contrib[:js] = True
        if je < n - 1:
            non_contrib[je + 1:] = True
        non_contrib[src] = True
        
        if non_contrib.sum() > 0:
            dst_candidates = np.where(non_contrib)[0]
        else:
            dst_candidates = np.array([i for i in range(n) if i != src])
        
        # Multiple random destinations
        n_tries = min(5, len(dst_candidates))
        dst_choice = rng.choice(dst_candidates, size=n_tries, replace=False)

        best_trial_M = max_b

        if best_trial is not None and best_trial_M < best_M:

        elif best_trial is not None:

            if rng.random() < np.exp(-(best_trial_M - best_M) / T * (1 - elapsed_frac)):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def spectral_init(n, seed=0):
    """Generate initial sequence using spectral method.
    
    Create a sequence with approximately flat autocorrelation by
    starting from a flat power spectrum with random phases.
    """
    rng = np.random.default_rng(seed)
    # Random phases
    phases = rng.uniform(0, 2*np.pi, n)
    # Flat magnitude with some structure
    mags = np.ones(n)
    # Make the DC component slightly higher for non-negativity
    mags[0] = 2.0
    spectrum = mags * np.exp(1j * phases)
    # Inverse FFT
    a = np.real(irfft(spectrum, n))
    # Make non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s
