```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1 and (v >= 0).all():
        # Already in simplex (or outside but projection is simple)
        pass
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

"""

1. Smooth FISTA optimization on the probability simplex with adaptive log-sum-exp temperature
2. LP best-response polish via scipy HiGHS solver

import numpy as np
from scipy.fft import rfft, irfft
import time
from scipy.optimize import linear_program

_global_best = None


def fast_autocorr(a):
    """Compute full autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def smooth_obj_grad(a, tau):
    """Compute smooth approximation F_tau(a) and its gradient.
    
    F_tau = (1/tau) * log(sum_k exp(tau * c_k)) where c = a*a
    grad_i = 2 * sum_m a_m * p_{m+i} (cross-correlation of a and p)
    """
    n = len(a)
    c = fast_autocorr(a)
    size = len(c)  # 2n-1
    
    # Numerically stable log-sum-exp
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    F = c_max + np.log(sum_exp) / tau
    
    # Gradient via cross-correlation: grad_i = 2 * sum_m a_m * p_{m+i}
    a_padded = np.zeros(size)
    a_padded[:n] = a
    
    fft_size = 1 << (2 * size - 1).bit_length()
    A = rfft(a_padded, fft_size)
    P = rfft(p, fft_size)
    cross_corr = irfft(P * np.conj(A), fft_size)[:size]
    
    grad = 2.0 * cross_corr[:n]
    return F, grad


def fista_smooth(a0, time_limit, tau_start=50.0, tau_end=500.0, verbose=True, seed=0):
    """FISTA optimization of smooth objective on the simplex with adaptive tau."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()  # This is max_c when sum=1
    t0 = time.monotonic()
    
    y = a.copy()
    a_prev = a.copy()
    t_fista = 1.0
    step = 1.0
    n_iters = 0
    tau = tau_start
    
    # Annealing schedule: roughly linear in log space
    n_anneal_points = 50
    baseline_score = best_score
    
    for it in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive tau: increase over time, with occasional decreases for exploration
        if it < n_anneal_points:
            frac = it / n_anneal_points
            tau = tau_start * (tau_end / tau_start) ** frac
        else:
            # After initial annealing, keep high tau but occasionally drop
            if it % 500 == 0 and rng.random() < 0.1:
                tau = tau_start * 2  # Explore
            else:
                tau = tau_end
        
        # Compute gradient at y
        F_y, grad_y = smooth_obj_grad(y, tau)
        
        # Backtracking line search
        step = min(step * 1.5, 10.0)  # Increase step cautiously
        max_bt = 20
        for bt in range(max_bt):
            x_new = project_simplex(y - step * grad_y)
            F_x, _ = smooth_obj_grad(x_new, tau)
            # Descent condition (relaxed)
            if F_x <= F_y - 0.01 * step * np.dot(grad_y, x_new - y) + 1e-10:
                break
            step *= 0.5
        else:
            x_new = a.copy()  # Fallback
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2)) / 2.0
        y_new = x_new + ((t_fista - 1.0) / t_new) * (x_new - a_prev)
        y_new = project_simplex(y_new)
        
        a_prev = a.copy()
        a = x_new
        y = y_new
        t_fista = t_new
        n_iters += 1
        
        # Check improvement using exact max
        if n_iters % 5 == 0:
            exact_score = fast_autocorr(a).max()
            if exact_score < best_score - 1e-12:
                best_score = exact_score
                best_a = a.copy()
                t_fista = 1.0  # Restart momentum on improvement
        
        # Periodic restart if stuck
        if n_iters % 200 == 0:
            exact_score = fast_autocorr(a).max()
            if exact_score >= best_score - 1e-14:
                # No improvement in 200 iters, restart
                t_fista = 1.0
                step = min(step, 1.0)
                # Small perturbation
                if rng.random() < 0.3:
                    noise = 0.01 * rng.standard_normal(n)
                    a = project_simplex(a + noise)
                    y = a.copy()
                    a_prev = a.copy()
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA iter={n_iters} tau={tau:.0f} ({elapsed:.0f}s): best_max_c={best_score:.10f}")
    
    return best_a, best_score * 2 * n  # Convert back to score (2n * max_c when sum=1)


def lp_best_response_sci(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0.
    Returns optimal b vector or None."""
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix A_ub: (size x n) where A_ub @ b <= M
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((size, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    
    c_obj = -np.ones(n)  # Minimize -sum(b) = Maximize sum(b)
    
    res = linear_program(c_obj, A_ub=A_ub, b_ub=np.full(size, M),
                         bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None


def lp_nesterov_polish(a0, time_limit, n_iters_max=200, verbose=True, seed=0):
    """NP polish using scipy LP solver."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # Normalize to sum=1
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    a_mom = a.copy()
    t_nest = 1.0
    stall = 0
    
    for it in range(1, n_iters_max + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        
        b = lp_best_response_sci(y, M_curr)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b / b_sum  # Best response on simplex
        
        # Nesterov update
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = project_simplex(y_new)
        
        if it == 1:
            a_new = g.copy()
        else:
            momentum = min((it - 1.0) / (it + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)
        
        prev_y = y
        y = y_new
        a_mom = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            t_nest = 1.0
        else:
            stall += 1
        
        score_a = fast_autocorr(a_mom).max()
        if score_a < best_score - 1e-14:
            best_score = score_a
            best_a = a_mom.copy()
            stall = 0
        
        # Restart if stalled
        if stall > 30:
            y = best_a.copy()
            a_mom = best_a.copy()
            t_nest = 1.0
            stall = 0
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it} ({elapsed:.0f}s): best_max_c={best_score:.10f}")
    
    return best_a, best_score * 2 * n


def pair_cd_polish(a0, time_limit, n_passes_max=50, verbose=True, seed=0):
    """Pair coordinate descent with exact line search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0
    
    for p in range(n_passes_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(perm[idx])
            if a[i] < 1e-10:
                continue
            
            # Try moving mass from i to j
            j = int(rng.integers(n))
            if j == i:
                continue
            if a[j] >= 1.0:  # j already has too much
                continue
            
            # First-order gradient for moving mass from i to j
            # delta = amount to move from i to j
            # Effect on conv: -delta * (a shifted by i from left + a shifted by i from right)
            #                + delta * (a shifted by j from left + a shifted by j from right)
            # + delta^2 terms at positions 2i and 2j
            
            # Compute the gain vector (effect per unit delta)
            gain = np.zeros(size)
            # Remove delta from i: subtract a from positions [i, i+n) and [n-1-i, n+i)
            lo1, hi1 = i, min(size, i + n)
            gain[lo1:hi1] -= a[:hi1 - lo1]
            lo2, hi2 = n - 1 - i, min(size, n + i)
            if hi2 > lo2:
                src = n - 1 - np.arange(lo2, hi2)
                gain[lo2:hi2] -= a[src]
            
            # Add delta to j: add a to positions [j, j+n) and [n-1-j, n+j)
            lo3, hi3 = j, min(size, j + n)
            gain[lo3:hi3] += a[:hi3 - lo3]
            lo4, hi4 = n - 1 - j, min(size, n + j)
            if hi4 > lo4:
                src = n - 1 - np.arange(lo4, hi4)
                gain[lo4:hi4] += a[src]
            
            # Binary search for optimal delta in [0, a[i]]
            # First-order: minimize max_k(conv_k + delta * gain_k)
            # Use golden section or just try a few values
            lo_d, hi_d = 0.0, a[i]
            best_d = 0.0
            best_v = conv.max()
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_d
            b_g = hi_d
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            
            f_c = (conv + c_g * gain).max()
            f_d = (conv + d_g * gain).max()
            
            for _ in range(20):
                if f_c < f_d:
                    b_g = d_g
                    d_g = c_g
                    f_d = f_c
                    c_g = b_g - gr * (b_g - a_g)
                    f_c = (conv + c_g * gain).max()
                else:
                    a_g = c_g
                    c_g = d_g
                    f_c = f_d
                    d_g = a_g + gr * (b_g - a_g)
                    f_d = (conv + d_g * gain).max()
                
                if abs(b_g - a_g) < 1e-10 * a[i]:
                    break
            
            best_d = (a_g + b_g) / 2.0
            if best_d > 1e-12:
                # Verify with exact computation
                test_a = a.copy()
                test_a[i] -= best_d
                test_a[j] += best_d
                test_a = np.maximum(test_a, 0)
                ts = test_a.sum()
                if ts > 1e-15:
                    test_a = test_a / ts
                    test_conv = fast_autocorr(test_a)
                    if test_conv.max() < best_score - 1e-14:
                        a = test_a
                        conv = test_conv
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
        
        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0
        
        if verbose and (p + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={p+1} ({elapsed:.0f}s): best_max_c={best_score:.10f}")
    
    return best_a, best_score * 2 * n


def structured_init(n, rng):
    """Generate diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (2 * x - 1)) + 0.1 * np.cos(4 * np.pi * (3 * x - 1))
    elif choice == 3:
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.8 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = np.sin(np.pi * x) ** 0.7 * (1 + 0.3 * np.cos(8 * np.pi * x))
    elif choice == 6:
        a = 1.0 + 0.2 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif choice == 7:
        # Symmetric with dip in center
        a = 0.5 + 0.5 * np.abs(np.cos(3 * np.pi * (x - 0.5)))
    elif choice == 8:
        # Multi-peak
        a = np.exp(-20*(x-0.2)**2) + np.exp(-20*(x-0.5)**2) + np.exp(-20*(x-0.8)**2)
    else:
        # Random smooth
        k = 5
        coeffs = rng.standard_normal(k)
        a = np.zeros(n)
        for m in range(k):
            a += coeffs[m] * np.cos(m * np.pi * x)
        a = np.abs(a) + 0.1
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum=1 normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def perturb(a, rng, scale=0.05):
    """Multiplicative perturbation preserving sum=1."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.1)
    a_p = a * noise
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s

            a_norm = best_seq / best_seq.sum() if len(best_seq) == n2 else interpolate_seq(best_seq, n2)
            a_out, sc = lp_nesterov_polish(a_norm, min(remaining * 0.4, 80.0),
                                           n_iters_max=50, verbose=False, seed=11)
            try_update(a_out, f"lp_long_n{n2}")
    
    # === Phase 6: Final LP polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final LP polish n={n_cur} ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 10:
        a_norm = best_seq / best_seq.sum()
        a_out, sc = lp_nesterov_polish(a_norm, t6, n_iters_max=60, verbose=True, seed=42)
        try_update(a_out, f"lp_final_n{n_cur}")
    
    # === Phase 7: Final pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final pair CD n={n_cur} ===")
    t7 = min(remaining * 0.20, 150.0)
    if t7 > 10:
        a_norm = best_seq / best_seq.sum()
        a_out, sc = pair_cd_polish(a_norm, t7, n_passes_max=40, verbose=True, seed=77)
        try_update(a_out, f"pcd_final_n{n_cur}")
    
    # === Phase 8: Final smooth refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final smooth refinement n={n_cur} ===")
    t8 = min(remaining * 0.25, 150.0)
    if t8 > 10:
        a_norm = best_seq / best_seq.sum()
        a_out, sc = fista_smooth(a_norm, t8, tau_start=100.0, tau_end=2000.0,
                                 verbose=True, seed=88)
        try_update(a_out, f"fista_final_n{n_cur}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
