The LP dual is:
$$\min M \sum_k y_k \quad \text{s.t.} \quad (a * y)_i \geq 1 \text{ for all } i, \quad y_k \geq 0$$

"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T

def lp_solve(T, n, M, solver, active=None):
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 8
    temps = np.geomspace(40.0, 0.15, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(30, min(300, int(remaining * 10)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.05 * it)
                for trial in range(12):
                    a_new = a - step * grad
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        step *= 0.5
                        continue
                    a_new = a_new * norm / s_new
                    sc = fast_conv_self(a_new).max()
                    if sc <= conv.max() + 0.05 * T_val:
                        break
                    step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    initial_score = best_score
    t0 = time.monotonic()
    y = a.copy()
    size = 2 * n - 1

    screen_frac = 0.88
    n_it = 0
    stagnant_counter = 0
    stagnant_threshold = best_score - 1e-6

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        n_it += 1

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        T = build_T(y)

        # Adaptive constraint screening
        threshold = M * screen_frac
        active = np.where(conv_y > threshold)[0]
        if len(active) < 10:
            active = np.where(conv_y > M * 0.5)[0]

        b = lp_solve(T, n, M, solver, active)
        if b is None:
            b = lp_solve(T, n, M, solver, None)
        if b is None:
            continue

        # Check dropped constraints
        if 0 < len(active) < size:
            check = fast_conv_cross(y, b)
            violated = np.where(check > M * (1 + 5e-4))[0]
            if len(violated) > 0:
                active2 = np.unique(np.concatenate([active, violated]))
                b2 = lp_solve(T, n, M, solver, active2)
                if b2 is not None:
                    b = b2
                screen_frac = max(0.5, screen_frac - 0.04)

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        sc_y = fast_conv_self(y_new).max()
        sc_a = fast_conv_self(a_new).max()

        improved = False
        if sc_y < best_score:
            best_score = sc_y
            best_a = y_new.copy()
            improved = True
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_new.copy()
            improved = True

        # Stagnation detection and restart
        if not improved:
            stagnant_counter += 1
        else:
            stagnant_counter = 0

        if stagnant_counter > 500:
            # Perturb to escape
            n_pert = max(10, n // 40)
            idx_p = np.random.default_rng(t).choice(n, size=n_pert, replace=False)
            y_new[idx_p] *= (1.0 + 0.15 * np.random.default_rng(t + 1).standard_normal(n_pert))
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            stagnant_counter = 0

        # Direct jump every 1000 iterations
        if t % 1000 == 0:
            sc_g = fast_conv_self(g).max()
            if sc_g < best_score:
                best_score = sc_g
                best_a = g.copy()
                if verbose:
                    print(f"    Direct jump at t={t}: score={best_score:.8f}")

        # Running average
        avg_a += a_new
        n_avg += 1
        if t % 200 == 0 and n_avg >= 100:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc_avg = fast_conv_self(avg_check).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_check.copy()

        y = y_new
        a = a_new

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.8f} (init={initial_score:.8f})")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    n_pass = 0

    while time.monotonic() - t0 < time_limit:
        n_pass += 1
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Candidate j's: mix of random and structured
            peak_lags = np.where(conv >= cur_max * 0.92)[0]
            j_candidates = set()
            for _ in range(4):
                j_candidates.add(int(rng.integers(n)))
            for kl in peak_lags[:4]:
                j1 = int((kl - i) % n)
                j2 = int((kl + i) % n)
                j_candidates.add(j1)
                j_candidates.add(j2)
            j_candidates.discard(i)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute gain vector g_k = a[k-j]+a[j-k]-a[k-i]-a[i-k]
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]

                # Vectorized delta search
                n_samples = 12
                deltas = np.linspace(0.0, ai, n_samples)
                best_d = 0.0
                best_v = cur_max
                for d in deltas[1:]:
                    new_conv = conv + d * g
                    # Second-order corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d

                if best_d > 0 and best_v < best_val - 1e-12:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 3:
                n_perturb = max(10, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
        else:
            no_improve = 0

        if verbose and n_pass % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_pass} imp={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    else:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def normalize_seq(a, n):
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_fn(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast initialization)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 80.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_work = best_seq.copy()
        a_out, _ = lse_gradient_descent(a_work, t1, verbose=True)
        check_and_update(a_out, f"lse1_n{n_base}")

    # Phase 2: LP Nesterov (main optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.38, 600.0)
    print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        check_and_update(a_out, f"lp1_n{n2}")

    # Phase 3: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 3: Pair CD n={n3} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        check_and_update(a_out, f"pcd1_n{n3}")

    # Phase 4: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.22, 350.0)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        check_and_update(a_out, f"lp2_n{n4}")

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (budget={remaining:.0f}s) ===")
    if remaining > 200:
        # Try longer sequence
        n_long = n_base + 400
        t_long = min(remaining * 0.35, 200.0)
        print(f"  Trying n={n_long} (budget={t_long:.0f}s)")
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t_long, solver, verbose=False)
        check_and_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            n_long2 = n_base + 800
            t_long2 = min(remaining * 0.35, 150.0)
            print(f"  Trying n={n_long2} (budget={t_long2:.0f}s)")
            init_long2 = interpolate_seq(best_seq, n_long2)
            a_out, _ = nesterov_lp_optimize(init_long2, t_long2, solver, verbose=False)
            check_and_update(a_out, f"lp_long2_{n_long2}")

        # Try shorter sequence (sometimes shorter is better due to less noise)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and n_base > 1000:
            n_short = n_base - 300
            t_short = min(remaining * 0.3, 100.0)
            print(f"  Trying n={n_short} (budget={t_short:.0f}s)")
            init_short = interpolate_seq(best_seq, n_short)
            a_out, _ = nesterov_lp_optimize(init_short, t_short, solver, verbose=False)
            check_and_update(a_out, f"lp_short_{n_short}")

    # Phase 6: Final CD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    nf = len(best_seq)
    t6 = min(remaining * 0.25, 120.0)
    print(f"\n=== Phase 6: Final CD n={nf} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = pair_cd_optimize(best_seq, t6, rng, verbose=True)
        check_and_update(a_out, f"pcd_final_n{nf}")

    # Phase 7: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    nf2 = len(best_seq)
    t7 = min(remaining * 0.8, 200.0)
    print(f"\n=== Phase 7: Final LP n={nf2} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t7, solver, verbose=True)
        check_and_update(a_out, f"lp_final_n{nf2}")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Safety: ensure we return within time
    elapsed = time.monotonic() - t_global
    if elapsed > float(budget_s) - 30:
        print(f"WARNING: Close to timeout ({elapsed:.0f}s)")

    return [float(x) for x in best_seq]
