import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv(a, b):
    """Compute linear convolution a*b using FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def softmax(x, temperature=1.0):
    """Stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    x_shifted = x - x.max()
    ex = np.exp(x_shifted / temperature)
    s = ex.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return ex / s

def evaluate_safe(a):

        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)

def softmax_mirror_descent(a0, time_limit, verbose=True, seed=42):
    """
    Softmax Mirror Descent: 
    - Compute autocorrelation R = conv(a,a)
    - Compute peak weights w = softmax(R / tau)
    - Compute gradient direction g = conv(w, a) / sum(w)
    - Update: b = exp(-eta * g) / Z (additive, like LP best response)
    - Mix: a = (1-t)*a + t*b
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0

    # Adaptive parameters
    tau = 5.0  # Temperature for softmax weights
    eta = 1.0  # Learning rate for exponential
    alpha = 0.1  # Mixing parameter
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        R = fast_autocorr(a)
        R_max = R.max()
        
        if R_max < 1e-15:
            break
        
        # Compute softmax weights (peak-targeted)
        # Scale R so that the peak is at 0
        w = softmax(R, temperature=tau)
        
        # Compute gradient direction: g[i] = sum_k w[k] * a[k-i]
        # This is conv(w, a) but w has length 2n-1 and a has length n
        # Result has length 2n-1+n-1 = 3n-2, but we only need the first n
        
        # Actually, g[i] = sum_k w[k] * a[k-i] where a[j]=0 for j outside [0,n-1]
        # This is the cross-correlation of w and a
        # Using FFT:
        fft_size = 1 << (2 * (2*n - 1) - 1).bit_length()  # enough for conv(w, a)
        # Actually conv(w, a) has length (2n-1) + n - 1 = 3n - 2
        fft_size = 1 << (3 * n - 2 + 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g_full = irfft(W * A, fft_size)[:n]
        
        # Normalize g
        g_mean = g_full.mean()
        g_std = g_full.std()
        if g_std < 1e-15:
            break
        
        # Compute b = exp(-eta * (g - g_mean) / g_std) / Z
        # Subtracting mean and dividing by std makes eta easier to interpret
        log_b = -eta * (g_full - g_mean) / g_std
        log_b -= log_b.max()  # for numerical stability
        b = np.exp(log_b)
        b_sum = b.sum()
        if b_sum < 1e-300:
            b = np.ones(n) / n
        else:
            b = b / b_sum
        b = b * norm_val  # normalize to correct sum
        
        # Mix
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm_val / s_new
        
        a = a_new
        n_iters += 1
        
        # Check improvement
        score = fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            last_improve = time.monotonic()
            stall = 0
        
        # Adaptive parameters
        # Decrease tau over time (focus more on peaks)
        tau = max(1.0, 5.0 * np.exp(-t / 5000))
        # Increase alpha over time (more aggressive mixing)
        alpha = min(0.5, 0.05 + 0.05 * np.log1p(t / 100))
        
        # Stall detection
        if time.monotonic() - last_improve > 30.0:
            stall += 1
            if stall > 20:
                break
        
        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, tau={tau:.2f}")

    return best_a, best_score


def projected_gradient_nesterov(a0, time_limit, verbose=True):
    """
    Nesterov Accelerated Projected Gradient Descent on the objective.
    Uses smoothed subgradient via softmax weights.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0

    y = a.copy()  # Nesterov lookahead
    tau = 3.0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute smoothed gradient at y
        R = fast_autocorr(y)
        R_max = R.max()
        if R_max < 1e-15:
            break
        
        w = softmax(R, temperature=tau)
        
        # Gradient: g[i] = sum_k w[k] * y[k-i]
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        Y = rfft(y, fft_size)
        g = irfft(W * Y, fft_size)[:n]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        g = g / g_norm  # unit norm gradient
        
        # Step size: eta = C / sqrt(t) (standard for subgradient)
        # or eta = C / t (for Nesterov with smooth approximation)
        eta = 2.0 / (np.sqrt(t) + 1.0)
        
        # Projected gradient step
        y_new = y - eta * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        # Nesterov update
        if t == 1:
            a_new = y_new
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        y = y_new
        a = a_new
        n_iters += 1
        
        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
            stall = 0
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
            stall = 0
        
        if time.monotonic() - last_improve > 30.0:
            stall += 1
            if stall > 15:
                break
        
        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGN iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    """Targeted coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        R = fast_autocorr(a)
        k_peak = int(np.argmax(R))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:10]

        top_peaks = np.argsort(R)[::-1][:8]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:10]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 100:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Generate various structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:

        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)

    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)

    elif choice == 5:

        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # "W" shape
        x = np.linspace(0.0, 1.0, n)
        a = (1 - np.cos(4 * np.pi * x)) ** 2 + 0.1

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n

def fast_conv(a, b):

def build_constraint_matrix(a, n):

    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0

    tau_start = 10.0
    tau_end = 1.0
    eta_base = 2.0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        R = fast_autocorr(a)
        R_max = R.max()
        if R_max < 1e-15:
            break

        # Adaptive temperature
        frac = min(1.0, t / 20000.0)
        tau = tau_start + (tau_end - tau_start) * frac
        eta = eta_base * (1.0 + 0.5 * frac)

        # Peak weights
        w = softmax_fn(R, temperature=tau)

        # Gradient direction: g[i] = conv(w, a)[i] (first n elements)
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g = irfft(W * A, fft_size)[:n]

        g_mean = g.mean()
        g_std = g.std()
        if g_std < 1e-15:
            break

        # Exponential update (additive, like LP best response)
        log_b = -eta * (g - g_mean) / g_std
        log_b -= log_b.max()
        b = np.exp(log_b)
        b_sum = b.sum()
        if b_sum < 1e-300:
            b = np.ones(n) / n
        else:
            b = b / b_sum
        b = b * norm_val

        # Mix
        alpha = 0.15 + 0.2 * min(1.0, t / 10000.0)
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm_val / s_new
        a = a_new
        n_iters += 1

        score = fast_autocorr(a).max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            last_improve = time.monotonic()
            stall = 0

        if time.monotonic() - last_improve > 40.0:
            stall += 1
            if stall > 15:
                break

        if verbose and n_iters % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, tau={tau:.2f}")

    return best_a, best_score


def targeted_coord_descent(a0, time_limit, seed=42):
    """Targeted coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        R = fast_autocorr(a)
        k_peak = int(np.argmax(R))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_contrib = sorted_idx[:12]

        top_peaks = np.argsort(R)[::-1][:10]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:12]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.03, 0.07, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(60):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.12)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 120:
                break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.cos(4.0 * np.pi * x)) ** 2 + 0.05
    elif choice == 7:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 + np.cos(6.0 * np.pi * x)) ** 2 + 0.05
    elif choice == 8:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.4 * np.sin(8.0 * np.pi * x) + 0.2 * np.sin(16.0 * np.pi * x)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.abs(np.sin(3.0 * np.pi * x))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def attempt_update(best_state, candidate_a, label, t_global):
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score
