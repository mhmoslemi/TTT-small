def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_conv_matrix(a):
    """Build full convolution matrix T where (T@b)_k = (a*b)_k."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns (b, duals)."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

2. Geometric (multiplicative) and linear interpolation strategies
3. Multi-scale perturbation restarts with annealing schedule
4. Fast coordinate descent with multi-point line search
5. Adaptive length exploration
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.2])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0, use_geometric=False):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    stall = 0
    stall_thresh = 40
    n_it = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M = float(conv_y.max())
        if M < 1e-12:
            break

        b = lp_best_response(y, M, solver)
        if b is None:
            b = lp_best_response(y, M * 1.001, solver)
            if b is None:
                continue

        n_it += 1
        g = b * norm / b.sum()

        alpha = 2.0 / (t + 2.0)

        if use_geometric:
            # Geometric (multiplicative) update
            eps = 1e-10
            y_safe = np.maximum(y, eps)
            g_safe = np.maximum(g, eps)
            y_new = y_safe * (g_safe / y_safe) ** alpha
            y_new = np.maximum(y_new, 0.0)
        else:
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)

        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        sc_a = fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0

        if stall > stall_thresh:
            y = best_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                print(f"    [NLP] restart iter={t}, score={best_score:.10f}")

        if verbose and n_it % 300 == 0:
            print(f"    [NLP] iter={n_it} t={time.monotonic()-t0:.0f}s score={best_score:.10f}")

    return best_a, best_score


def coord_descent(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent with multi-point line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_imp = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # First-order gain vector
                gain = np.zeros(size)
                lo_j = j
                hi_j = min(j + n, size)
                gain[lo_j:hi_j] += 2.0 * a[:hi_j - lo_j]
                lo_i = i
                hi_i = min(i + n, size)
                gain[lo_i:hi_i] -= 2.0 * a[:hi_i - lo_i]

                # Binary search for optimal delta
                lo_d, hi_d = 0.0, ai
                cur_max = float(conv.max())
                best_d = 0.0
                best_val = cur_max

                # Evaluate at geometrically spaced points
                n_eval = 8
                for k in range(n_eval + 1):
                    frac = k / n_eval
                    d = frac * ai
                    if d <= 1e-15:
                        continue
                    new_max = float(np.max(conv + d * gain))
                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d

                # Refine around best_d
                if best_d > 0:
                    step = ai / (n_eval * 4)
                    for k in range(-4, 5):
                        d = best_d + k * step
                        if d <= 0 or d > ai:
                            continue
                        new_max = float(np.max(conv + d * gain))
                        if new_max < best_val - 1e-14:
                            best_val = new_max
                            best_d = d

                if best_d > 1e-15:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    cm = float(conv.max())
                    if cm < best_score - 1e-13:
                        best_score = cm
                        best_a = a.copy()
                        improved = True
                        no_imp = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break

        if verbose and n_passes % 5 == 0:
            print(f"    [CD] pass={n_passes} t={time.monotonic()-t0:.0f}s score={best_score:.10f}")

    return best_a, best_score


def single_coord_descent(a0, time_limit, rng, verbose=False):
    """Single-coordinate scaling descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_imp = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Try scaling a[i] by different factors
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.7]:
                a_new = a.copy()
                a_new[i] = ai * factor
                a_new = a_new * norm / a_new.sum()
                conv_new = fast_autocorr(a_new)
                sc = float(conv_new.max())
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    conv = conv_new
                    improved = True
                    no_imp = 0

        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break

        if verbose and n_passes % 5 == 0:
            print(f"    [SCD] pass={n_passes} t={time.monotonic()-t0:.0f}s score={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.15 * np.cos(8.0 * np.pi * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = 1.0 + 0.3 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif choice == 7:
        a = x ** 0.5 * (1.0 - x) ** 0.3
    elif choice == 8:
        a = np.exp(-2.0 * x) * (1.0 + 0.5 * np.sin(10 * np.pi * x))
    else:
        a = (1.0 + x) ** 0.8 * (1.0 - 0.5 * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    sn = f.sum()
    if sn < 1e-15:
        return np.ones(new_n)
    fn = f / sn
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return fi / ss


def perturb(a, rng, scale):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def propose_candidate():
    rng = np.random.default_rng(2024)
    t_start = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")

    if best_seq is None:
        n0 = 1600
        best_seq = structured_init(n0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n0}, score={best_score:.8f}")

    # Phase 1: LP Nesterov from parent
    n = len(best_seq)
    print(f"\n=== Phase 1: LP-Nesterov n={n} ===")
    t1 = 350.0
    a_norm = best_seq * np.sqrt(2.0 * n) / best_seq.sum()
    a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, seed=0, use_geometric=False)
    try_update(a_out, "lp1")

    # Phase 2: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_start)
    n = len(best_seq)
    print(f"\n=== Phase 2: CD n={n} ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t2, rng, verbose=True)
        try_update(a_out, "cd1")

    # Phase 3: Multiple perturbation restarts
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: Perturbation restarts ===")
    n_restarts = 5
    t3_each = min(60.0, remaining / (n_restarts + 2))
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 80:
            break
        t3 = min(t3_each, remaining - 80)
        if t3 < 20:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.01 * r
        pert = perturb(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        use_geo = (r % 2 == 1)
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=False, seed=100 + r, use_geometric=use_geo)
        try_update(a_out, f"pert{r}")

    # Phase 4: CD on best
    remaining = time_budget - (time.monotonic() - t_start)
    n = len(best_seq)
    print(f"\n=== Phase 4: CD + SCD n={n} ===")
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t4 * 0.6, rng, verbose=False)
        try_update(a_out, "cd2")
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 30:
            a_norm = best_seq * np.sqrt(2.0 * n) / best_seq.sum()
            a_out, _ = single_coord_descent(a_norm, min(remaining * 0.15, 60.0), rng, verbose=False)
            try_update(a_out, "scd1")

    # Phase 5: More restarts with smaller perturbation
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: Fine restarts ===")
    for r in range(3):
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 60:
            break
        t5 = min(40.0, remaining / 3 - 10)
        if t5 < 15:
            break
        n_cur = len(best_seq)
        scale = 0.005 + 0.003 * r
        pert = perturb(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=False, seed=500 + r, use_geometric=False)
        try_update(a_out, f"fine{r}")

    # Phase 6: Try different lengths
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 6: Length exploration ===")
    for n_trial in [len(best_seq) + 200, len(best_seq) - 200, len(best_seq) + 400]:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 80:
            break
        if n_trial < 500:
            continue
        t6 = min(60.0, remaining / 3)

        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=False, seed=900, use_geometric=False)
        try_update(a_out, f"len{n_trial}")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_start)
    n = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n} ===")
    if remaining > 30:
        t7 = min(100.0, remaining * 0.5)
        a_norm = best_seq * np.sqrt(2.0 * n) / best_seq.sum()
        a_out, _ = coord_descent(a_norm, t7, rng, verbose=True)

    remaining = time_budget - (time.monotonic() - t_start)
    n = len(best_seq)
    if remaining > 20:
        t8 = min(50.0, remaining * 0.5)
        a_norm = best_seq * np.sqrt(2.0 * n) / best_seq.sum()
        a_out, _ = single_coord_descent(a_norm, t8, rng, verbose=True)
        try_update(a_out, "scd_final")

    # Final LP polish
    remaining = time_budget - (time.monotonic() - t_start)
    n = len(best_seq)
    if remaining > 30:
        t9 = min(60.0, remaining - 5)
        a_norm = best_seq * np.sqrt(2.0 * n) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t9, solver, verbose=False, seed=777, use_geometric=True)

        # CD after

        if remaining > 15:
            n = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n) / best_seq.sum()

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):

    else:

        return [1.0]

    return [float(x) for x in best_seq]
