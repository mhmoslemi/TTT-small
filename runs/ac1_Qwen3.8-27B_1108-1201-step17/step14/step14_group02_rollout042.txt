    i_end = min(n - 1, k_star)

        idx = k_star - np.arange(i_start, i_end + 1)
        g[i_start:i_end + 1] = 2.0 * a[idx]
    return g


def compute_multilag_subgrad(a, top_k, beta):
    ac = fast_autocorr(a)
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = k_int - np.arange(i_start, i_end + 1)
            g[i_start:i_end + 1] += 2.0 * wi * a[idx]
    return g, ac


def md_step(a, g, eta):
    log_a = np.log(a + 1e-30)
    log_a_new = log_a - eta * g
    max_log = log_a_new.max()
    a_new = np.exp(log_a_new - max_log)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def mirror_descent_primary(a0, time_limit, verbose=True, top_k_mode=1):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 1.0
    t = 0
    stall = 0
    best_stall_m = best_m

    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            # Restart with perturbation
            rng = np.random.default_rng(t)
            a = norm_sum1(a + rng.random(n) * 0.01 * a)
            stall = 0

        k_star = int(ac.argmax())
        g = compute_true_subgrad(a, k_star)

        if top_k_mode > 1 and t % top_k_mode == 0:
            g_multi, _ = compute_multilag_subgrad(a, top_k=8, beta=4.0)
            g = 0.5 * g + 0.5 * g_multi

        max_g = g.max()
        if max_g > 1e-15:
            eta_target = 0.7 / max_g
        else:
            eta_target = eta
        eta = 0.9 * eta + 0.1 * eta_target

        a_new = md_step(a, g, eta)
        a = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): score={score:.6f}, M={best_m:.8e}, eta={eta:.4f}")

    return best_a, best_m


def nesterov_mirror(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Nesterov point
    x = a.copy()  # current point
    eta = 1.0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        alpha = 2.0 / (t + 2.0)

        ac_y = fast_autocorr(y)
        M_y = ac_y.max()
        k_star = int(ac_y.argmax())
        g = compute_true_subgrad(y, k_star)

        max_g = g.max()
        if max_g > 1e-15:
            eta = 0.7 / max_g

        x_new = md_step(y, g, eta)

        # Nesterov update
        beta = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        log_y_new = (1.0 - alpha) * np.log(y + 1e-30) + alpha * np.log(x_new + 1e-30)
        if t > 1:
            log_y_new += beta * (np.log(x_new + 1e-30) - np.log(y + 1e-30))
        y_new = np.exp(log_y_new - log_y_new.max())
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        else:
            y_new = x_new

        y = y_new
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()

        M_y_new = fast_autocorr(y).max()
        if M_y_new < best_m:
            best_m = M_y_new
            best_a = y.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): score={score:.6f}, M={best_m:.8e}")

    return best_a, best_m


def spectral_shaping(a, strength=0.1, verbose=False):
    """Apply spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    P = np.abs(A) ** 2
    P_mean = P.mean()

    # Soft cap: reduce high-power frequencies
    P_capped = P * np.exp(-strength * np.maximum(P - P_mean, 0) / (P_mean + 1e-30))

    # Reconstruct with modified magnitudes, same phases
    phase = np.angle(A)
    A_new = np.sqrt(P_capped / (np.abs(A) ** 2 + 1e-60)) * A
    A_new = np.where(np.abs(A) > 1e-30, A_new, A)
    a_new = irfft(A_new, fft_size)[:n].real
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def coordinate_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        k_star = int(ac.argmax())
        improved = False

        # Find the position contributing most to the peak
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_star - idx]

        if contrib.max() < 1e-25:
            stall += 1
            continue

        src = idx[np.argmax(contrib)]

        # Find best destination: minimize increase in autocorrelation
        # Score each candidate by how much it would increase the top lags
        top_lags = np.argsort(ac)[::-1][:6]
        n_cand = min(n, 300)
        if n_cand < n:
            # Prefer positions far from peak-contributing regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_lags:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) < n_cand:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]
        else:
            chosen = np.arange(n)

        best_dst = -1
        best_score = float('inf')

        for dst in chosen:
            if dst == src:
                continue
            score = 0.0
            for k2 in top_lags:
                k2i = int(k2)
                d = k2i - dst
                add = 0.0
                if 0 <= d < n:
                    add += a[d]
                d2 = dst - k2i
                if 0 <= d2 < n:
                    add += a[d2]
                score += add
            if score < best_score:
                best_score = score
                best_dst = dst

        if best_dst < 0:
            stall += 1
            continue

        dst = best_dst

        # Binary search for optimal transfer amount
        lo, hi = 0.0, a[src]
        found = False
        for _ in range(12):
            mid = 0.5 * (lo + hi)
            a_trial = a.copy()
            a_trial[src] -= mid
            a_trial[dst] += mid
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                hi = mid
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m:
                best_m = M_trial
                best_a = a_trial.copy()
                a = a_trial
                lo = mid
                improved = True
                found = True
            else:
                hi = mid

        if not improved:
            stall += 1
            # Random kick
            ii = rng.integers(n)
            jj = rng.integers(n)
            if ii != jj and a[ii] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[ii] * frac
                a[ii] -= amt
                a[jj] += amt
                a = norm_sum1(a)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[ii] += amt
                    a[jj] -= amt
                    a = norm_sum1(a)

    if verbose:
        elapsed = time.monotonic() - t0
        score = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): score={score:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def shift_blend(a, shift_frac, blend=0.3):
    """Shift the sequence and blend with original to exploit asymmetry."""
    n = len(a)
    shift = int(n * shift_frac)
    shifted = np.roll(a, shift)
    blended = (1 - blend) * a + blend * shifted
    return norm_sum1(blended)
