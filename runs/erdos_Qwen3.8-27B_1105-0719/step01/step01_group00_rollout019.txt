    for _ in range(80):

        h = np.clip(x - mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid

def lag_info(n, r):
    """Return (base_indices, shifted_indices) for lag r in np.correlate output."""
    offset = r - (n - 1)
    b_min = max(0, -offset)
    b_max = min(n - 1, n - 1 - offset)
    if b_min > b_max:
        return None
    base = np.arange(b_min, b_max + 1)
    shifted = base + offset
    return base, shifted

def lag_grad(h, n, r):
    """Gradient of raw correlation at lag r."""
    info = lag_info(n, r)
    if info is None:
        return None
    base, shifted = info
    grad = np.zeros(n)
    grad[shifted] += (1.0 - h[base])
    grad[base] += (-h[shifted])
    return grad

    info = lag_info(n, r)

        return None
    base, shifted = info
    diag = np.zeros(n)
    for b, j in zip(base, shifted):
        diag[j] += 1.0
        diag[b] += 1.0
    return diag

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (as above)

def solve_trust_subproblem(h_c, corrs_c, n, dx, target_sum, active_indices, alpha, rho, maxiter):
    """Solve one trust-region subproblem."""
    m = len(active_indices)
    # Precompute model data
    R0_list = []
    G_list = []
    D_list = []
    for r in active_indices:
        R0_list.append(corrs_c[r])
        G_list.append(lag_grad(h_c, n, r))
        D_list.append(lag_diag_counts(n, r))
    
    # Variable: z = [h(0..n-1), t], length n+1
    n_vars = n + 1
    
    def obj(z):
        h = z[:n]
        t = z[n]
        prox = alpha * np.sum((h - h_c)**2)
        return t + prox
    
    def obj_grad(z):
        h = z[:n]
        grad = np.empty(n_vars)
        grad[:n] = 2 * alpha * (h - h_c)
        grad[n] = 1.0
        return grad
    
    # Inequality constraints: t - dx*(R0 + G^T(h-h_c) + 0.5*D^T(h-h_c)^2) >= 0
    cons = []
    # Equality: sum(h) = target_sum
    cons.append({
        'type': 'eq',
        'fun': lambda z: np.sum(z[:n]) - target_sum,
        'jac': lambda z: np.concatenate([np.ones(n), [0.0]])
    })
    
    for i, r in enumerate(active_indices):
        R0 = R0_list[i]
        G = G_list[i]
        D = D_list[i]
        
        def constr_fun(z, R0=R0, G=G, D=D):
            h = z[:n]
            t = z[n]
            delta = h - h_c
            val = t - dx * (R0 + np.dot(G, delta) + 0.5 * np.dot(D, delta**2))
            return val
        
        def constr_jac(z, R0=R0, G=G, D=D):
            h = z[:n]
            delta = h - h_c
            jac = np.empty(n_vars)
            jac[:n] = -dx * G + dx * D * delta
            jac[n] = 1.0
            return jac
        
        cons.append({
            'type': 'ineq',
            'fun': constr_fun,
            'jac': constr_jac
        })
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(0.0, h_c[i] - rho)
        hi = min(1.0, h_c[i] + rho)
        bounds.append((lo, hi))
    t_max = corrs_c.max() * dx + 1.0
    bounds.append((0.0, t_max))
    
    # Initial point
    z0 = np.concatenate([h_c, [corrs_c.max() * dx + 1e-6]])
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_out = res.x[:n]
            return h_out
    except Exception:
        pass
    return None
