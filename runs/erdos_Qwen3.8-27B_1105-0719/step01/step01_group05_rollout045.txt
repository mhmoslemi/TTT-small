import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] + step, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def lag_gradient(h, lag, dx):
    """Compute gradient of q_lag with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            # q = dx * sum_{i=0}^{s-1} h[i+lag]*(1-h[i])
            # d/dh[j]: from h[i+lag] term: dx*(1-h[i]) where i=j-lag, valid j in [lag, n-1]
            # from (1-h[i]) term: -dx*h[i+lag] where i=j, valid j in [0, s-1]
            grad[lag:n] += dx * (1.0 - h[:s])
            grad[:s] -= dx * h[lag:n]
    else:
        p = -lag
        s = n - p
        if s > 0:
            # q = dx * sum_{i=p}^{n-1} h[i-p]*(1-h[i])
            # d/dh[j]: from h[i-p] term: dx*(1-h[i]) where i=j+p, valid j in [0, s-1]
            # from (1-h[i]) term: -dx*h[i-p] where i=j, valid j in [p, n-1]
            grad[:s] += dx * (1.0 - h[p:n])
            grad[p:n] -= dx * h[:s]
    return grad

def active_set_gradient(h, k_active=5):
    """Compute weighted gradient focusing on top-k active lags."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    pmax = np.max(profile)
    if pmax < 1e-15:
        return np.zeros(n), pmax
    # Use softmax weighting on top lags
    shifted = (profile - pmax) / max(pmax * 0.01, 1e-6)
    weights = np.exp(np.clip(shifted, -50, 0))
    wsum = weights.sum()
    if wsum < 1e-15:
        # Only the max
        idx = np.argmax(profile)
        lag = idx - (n - 1)
        return lag_gradient(h, lag, dx), pmax
    weights = weights / wsum
    grad = np.zeros(n)
    for idx in range(len(profile)):
        if weights[idx] > 1e-8:
            lag = idx - (n - 1)
            grad += weights[idx] * lag_gradient(h, lag, dx)
    return grad, pmax

def projected_gradient_descent(h0, max_iter=500, deadline=None):
    """Active-set projected gradient descent."""
    n = len(h0)
    target = 0.5 * n
    dx = 2.0 / n
    h = h0.copy()
    c_best = exact_objective(h)
    h_best = h.copy()
    
    alpha = 0.1
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline - 2:
            break
        grad, c_cur = active_set_gradient(h)
        # Project gradient to maintain sum constraint
        grad_proj = grad - np.mean(grad)
        # Line search
        improved = False
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_new = h - alpha * step_mult * grad_proj
            h_new = project_box_sum(h_new, target)
            c_new = exact_objective(h_new)
            if c_new < c_cur - 1e-13:
                h = h_new
                c_cur = c_new
                if c_new < c_best:
                    c_best = c_new
                    h_best = h.copy()
                improved = True
                break
        if not improved:
            alpha *= 0.5
            if alpha < 1e-8:
                break
        else:
            alpha = min(alpha * 1.1, 0.5)
    return h_best, c_best

def slsqp_direct(h0, deadline, max_iter=100):
    """SLSQP with direct constraint formulation."""
    n = len(h0)
    target = 0.5 * n
    dx = 2.0 / n
    c0 = exact_objective(h0)
    
    # Variable: z = [h_0, ..., h_{n-1}, c]
    z0 = np.concatenate([h0, [c0 + 1e-8]])
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(z):
        return np.sum(z[:n]) - target
    
    def eq_jac(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    def ineq_fun(z):
        h = z[:n]
        c = z[-1]
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        return c - profile  # need >= 0
    
    def ineq_jac(z):
        h = z[:n]
        jac = np.zeros((2 * n - 1, n + 1))
        for r in range(2 * n - 1):
            lag = r - (n - 1)
            g = lag_gradient(h, lag, dx)
            jac[r, :n] = -g  # d(c - q_r)/dh = -dq_r/dh
            jac[r, -1] = 1.0   # d(c - q_r)/dc = 1
        return jac
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], target)
            c_new = exact_objective(h_new)
            if c_new < c0 - 1e-12:
                return h_new, c_new
    except Exception:
        pass
    return h0, c0

def binary_local_search(b, max_rounds=100, deadline=None):

    n = len(b)

        return b, exact_objective(b)

            for i in ones:

            candidates = [(ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))])

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)
