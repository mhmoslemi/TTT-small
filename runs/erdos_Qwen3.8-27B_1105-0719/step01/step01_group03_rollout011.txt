def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def compute_profile(h):

    dx = 2.0 / n

def build_gradient_matrix(h, n, dx):
    """Build G[m,j] = dr_m/dh_j for the full correlation profile."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    mid = n - 1
    for m in range(M):
        d = m - mid
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # p_d = dx * sum_{i=0}^{n-d-1} h_{i+d}*(1-h_i)
            # d(p_d)/d(h_j): 
            #   if j >= d: term from h_j in h_{i+d} with i=j-d: +(1-h_{j-d})*dx
            #   if j < n-d: term from h_j in (1-h_i) with i=j: -h_{j+d}*dx
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            s = -d
            # p_{-s} = dx * sum_{i=0}^{n-s-1} h_i*(1-h_{i+s})

            G[m, 0:n-s] += dx * (1.0 - h[s:n])
            G[m, s:n] += -dx * h[0:n-s]
    return G

def soft_gradient(h, tau):
    """Compute gradient of the log-sum-exp surrogate."""
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    max_p = np.max(profile)
    weights = np.exp((profile - max_p) / tau)
    weights /= weights.sum()
    G = build_gradient_matrix(h, n, dx)
    grad = G.T @ weights
    grad -= grad.mean()  # project onto sum-preserving subspace
    return grad, profile

import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    mid = n - 1
    for m in range(M):
        d = m - mid
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            s = -d
            G[m, 0:n-s] += dx * (1.0 - h[s:n])
            G[m, s:n] += -dx * h[0:n-s]
    return G


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def epigraph_obj(x, n):
    return x[-1]


def epigraph_obj_jac(x, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g


def epigraph_ineq_fun(x, n):
    h = np.clip(x[:-1], 0.0, 1.0)
    h = repair_feasible(h)
    profile = compute_profile(h)
    return x[-1] - profile


def epigraph_ineq_jac(x, n):
    h = np.clip(x[:-1], 0.0, 1.0)
    h = repair_feasible(h)
    dx = 2.0 / n
    G = build_gradient_matrix(h, n, dx)
    M = 2 * n - 1
    J = np.zeros((M, n + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J


def solve_epigraph_tc(h_start, maxiter=300, tol=1e-9):
    n = len(h_start)
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    x0 = np.concatenate([h_start, [t0]])

    lb = np.zeros(n + 1)
    ub = np.ones(n + 1)
    ub[-1] = c0 + 1.0
    bounds = list(zip(lb, ub))

    sum_row = np.zeros(n + 1)
    sum_row[:n] = 1.0
    sum_con = LinearConstraint(sum_row.reshape(1, -1), n / 2.0 - 1e-12, n / 2.0 + 1e-12)

    ineq_con = NonlinearConstraint(epigraph_ineq_fun, 0.0, np.inf,
                                   jac=epigraph_ineq_jac, args=(n,))

    try:
        res = minimize(epigraph_obj, x0, method='trust-constr',
                       jac=lambda x: epigraph_obj_jac(x, n),
                       constraints=[sum_con, ineq_con],
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': tol, 'xtol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c = exact_score(h_new)
            return h_new, c
    except Exception:
        pass
    return None, None


def solve_smooth_tc(h_start, tau, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n

    def smooth_obj(x):
        h = repair_feasible(np.clip(x, 0.0, 1.0))
        profile = compute_profile(h)
        mp = np.max(profile)
        vals = (profile - mp) / tau
        return tau * np.log(np.sum(np.exp(vals))) + mp

    def smooth_jac(x):
        h = repair_feasible(np.clip(x, 0.0, 1.0))
        profile = compute_profile(h)
        mp = np.max(profile)
        w = np.exp((profile - mp) / tau)
        w /= w.sum()
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ w
        grad -= grad.mean()
        return grad

    sum_con = LinearConstraint(np.ones((1, n)), n / 2.0 - 1e-12, n / 2.0 + 1e-12)
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(smooth_obj, h_start, method='trust-constr',
                       jac=smooth_jac, constraints=sum_con, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-8, 'xtol': 1e-10})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x)
            c = exact_score(h_new)
            return h_new, c
    except Exception:
        pass
    return None, None


def active_set_descent(h, max_iters=200):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    stalled = 0

    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 20:
                break

        thresh = c - 2e-4
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])

        w = profile[active].copy()
        w = w - w.min() + 1e-10
        w /= w.sum()

        G = build_gradient_matrix(h, n, dx)
        g = G[active, :].T @ w
        g -= g.mean()

        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break
        g /= norm

        improved = False
        step_sizes = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 1e-4]
        for alpha in step_sizes:
            h_try = h - alpha * g
            h_try = repair_feasible(h_try)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                improved = True
                break

    return best_h, best_c


def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = repair_feasible(x[:-1])
        profile = compute_profile(h)
        return x[-1] - profile

    def ineq_jac(x):
        h = repair_feasible(x[:-1])
        G = build_gradient_matrix(h, n, 2.0 / n)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c = exact_score(h_new)
            return h_new, c
    except Exception:
        pass
    return None, None
