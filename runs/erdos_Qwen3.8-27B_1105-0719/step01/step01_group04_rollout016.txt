def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Fix any remaining tiny drift
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h

def compute_lag_values_and_grads(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    for delta in range(0, n):
        if delta == 0:
            # Q_0 = dx * sum h*(1-h)
            values[n - 1] = dx * np.sum(h * (1.0 - h))
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - delta)
            # Positive lag: sum_{j=0}^{n-delta-1} h[j] * (1-h[j+delta])
            val_pos = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
            # Negative lag: sum_{j=delta}^{n-1} h[j] * (1-h[j-delta])
            val_neg = dx * np.sum(h[idx + delta] * (1.0 - h[idx]))
            
            values[n - 1 + delta] = val_pos
            values[n - 1 - delta] = val_neg

            grads[n - 1 + delta, idx + delta] += -dx * h[idx]
            
            # Gradients for negative lag
            grads[n - 1 - delta, idx + delta] += dx * (1.0 - h[idx])
            grads[n - 1 - delta, idx] += -dx * h[idx + delta]

import numpy as np
from scipy.optimize import minimize
import time
import math

try:
    import cvxpy as cp
    HAS_CP = True
except ImportError:
    HAS_CP = False


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-14:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-14:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h


def compute_lag_data(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))

    # Lag 0: index n-1
    values[n - 1] = dx * np.sum(h * (1.0 - h))
    grads[n - 1, :] = dx * (1.0 - 2.0 * h)

    for d in range(1, n):
        # Lag index n-1-d (positive shift d): Q = dx*sum_{m=0}^{n-1-d} h[m]*(1-h[m+d])
        idx = np.arange(n - d)
        values[n - 1 - d] = dx * np.sum(h[idx] * (1.0 - h[idx + d]))
        # Gradient
        grads[n - 1 - d, idx] += dx * (1.0 - h[idx + d])  # first factor
        if d < n:
            idx2 = np.arange(d, n)
            grads[n - 1 - d, idx2] += -dx * h[idx2 - d]   # second factor

        # Lag index n-1+d (negative shift): Q = dx*sum_{m=d}^{n-1} h[m]*(1-h[m-d])
        idx2 = np.arange(d, n)
        values[n - 1 + d] = dx * np.sum(h[idx2] * (1.0 - h[idx2 - d]))
        # Gradient
        grads[n - 1 + d, idx2] += dx * (1.0 - h[idx2 - d])  # first factor
        idx3 = np.arange(0, n - d)
        grads[n - 1 + d, idx3] += -dx * h[idx3 + d]         # second factor

    return values, grads


def interpolate_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_src = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_tgt = (np.arange(n_new) + 0.5) * 2.0 / n_new
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_tgt, xs_ext, ys_ext)
    return repair_feasible(h_new)


def solve_qp(h_cur, values, grads, lam):
    """Solve the trust-region QP: min t + lam*||h - h_cur||^2 s.t. linearized constraints."""
    n = len(h_cur)
    n_lags = 2 * n - 1

    # Select top lags for active set (all if small)
    if n_lags <= 300:
        active_idx = np.arange(n_lags)
    else:
        c_max = np.max(values)
        threshold = c_max - 0.02 * c_max
        active_idx = np.where(values >= threshold)[0]
        if len(active_idx) < 20:
            active_idx = np.argsort(values)[::-1][:40]
        active_idx = np.sort(active_idx)

    G = grads[active_idx, :]  # (K, n)
    Q_act = values[active_idx]  # (K,)

    # Linearization: Q_k(h) ≈ Q_k(h_cur) + G_k . (h - h_cur)
    # Constraint: G_k . h <= t - Q_k(h_cur) + G_k . h_cur
    rhs_offset = -Q_act + G @ h_cur  # (K,)

    h_var = cp.Variable(n)
    t_var = cp.Variable()

    objective = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h_cur))
    constraints = [
        G @ h_var - t_var <= rhs_offset,
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
    ]

    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None

    if h_var.value is None:
        return None
    h_new = np.array(h_var.value, dtype=np.float64)
    if not np.all(np.isfinite(h_new)):
        return None
    return h_new


def trust_region_loop(h_init, max_iter=200, time_budget=300, start_lambda=0.05):
    """Main trust-region optimization loop."""
    t_start = time.perf_counter()
    h = repair_feasible(h_init.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    lam = start_lambda
    prev_h = h.copy()
    stall_count = 0

    for it in range(max_iter):
        remaining = time_budget - (time.perf_counter() - t_start)
        if remaining < 10:
            break

        values, grads = compute_lag_data(h)
        c_cur = np.max(values)

        # Solve QP
        h_sol = solve_qp(h, values, grads, lam)
        if h_sol is None:
            lam *= 2.0
            continue

        h_cand = repair_feasible(h_sol)
        score_cand = exact_score(h_cand)

        if score_cand < best_score - 1e-14:
            # Accept with possible backtracking
            h_new = h_cand
            score_new = score_cand

            # Check if we should backtrack from h (the current point, not best)
            if score_new >= exact_score(h) - 1e-14:
                # Step from h didn't help; try interpolation
                improved = False
                for alpha in [0.75, 0.5, 0.25, 0.125]:
                    h_try = repair_feasible(h + alpha * (h_sol - h))
                    s_try = exact_score(h_try)
                    if s_try < exact_score(h) - 1e-14:
                        h_new = h_try
                        score_new = s_try
                        improved = True
                        break
                if not improved:
                    # Shrink trust region
                    lam *= 2.0
                    stall_count += 1
                    if stall_count > 8:
                        break
                    continue

            h = h_new
            best_h = h.copy()
            best_score = score_new
            stall_count = 0

            # Adapt lambda: if step was good, can increase (larger steps OK)
            step_size = np.linalg.norm(h - prev_h)
            if step_size > 0.01:
                lam = max(0.001, lam * 0.7)
            prev_h = h.copy()
        else:
            # Try stepping from h toward h_sol
            improved = False
            for alpha in [0.5, 0.25, 0.125, 0.0625]:
                h_try = repair_feasible(h + alpha * (h_sol - h))
                s_try = exact_score(h_try)
                if s_try < exact_score(h) - 1e-14:
                    h = h_try
                    if s_try < best_score:
                        best_h = h.copy()
                        best_score = s_try
                    stall_count = 0
                    improved = True
                    break
            if not improved:
                lam *= 2.0
                stall_count += 1
                if stall_count > 10:
                    break

    return best_h, best_score


def active_descent(h, max_steps=80, time_budget=60):
    """Active-set gradient descent for polishing."""
    t_start = time.perf_counter()
    n = len(h)
    h = repair_feasible(h.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    prev_score = best_score

    for step in range(max_steps):
        if time.perf_counter() - t_start > time_budget:
            break
        values, grads = compute_lag_data(h)
        c_max = np.max(values)
        n_lags = len(values)

        # Select active lags
        n_active = min(30, n_lags)
        active_idx = np.argsort(values)[::-1][:n_active]

        # Compute weights
        gaps = np.maximum(c_max - values[active_idx], 0.0)
        sigma = 0.003 * c_max + 1e-10
        weights = np.exp(-gaps / sigma)
        weights /= weights.sum() + 1e-15

        # Direction: weighted sum of negative gradients
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        direction -= np.mean(direction)  # preserve sum

        # Try multiple step sizes
        improved = False
        for alpha in [0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.001]:
            h_try = repair_feasible(h + alpha * direction)
            s_try = exact_score(h_try)
            if s_try < best_score - 1e-14:
                h = h_try
                best_h = h.copy()
                best_score = s_try
                improved = True
                break

        if not improved:
            # Try smaller active set (only top few)
            for n_act_small in [10, 5, 3]:
                active_small = np.argsort(values)[::-1][:n_act_small]
                gaps_s = np.maximum(c_max - values[active_small], 0.0)
                w_s = np.exp(-gaps_s / sigma)
                w_s /= w_s.sum() + 1e-15
                dir_s = -np.sum(w_s[:, None] * grads[active_small, :], axis=0)
                dir_s -= np.mean(dir_s)
                for alpha in [0.02, 0.01, 0.005, 0.002]:
                    h_try = repair_feasible(h + alpha * dir_s)
                    s_try = exact_score(h_try)
                    if s_try < best_score - 1e-14:
                        h = h_try
                        best_h = h.copy()
                        best_score = s_try
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

        # Convergence check
        if prev_score - best_score < 1e-15:
            break
        prev_score = best_score

    return best_h, best_score


def slsqp_polish(h_init, maxiter=100, time_budget=60):
    """SLSQP-based polishing as fallback/supplement."""
    t_start = time.perf_counter()
    n = len(h_init)
    h = repair_feasible(h_init.copy())
    best_h = h.copy()
    best_score = exact_score(h)

    def obj(x):
        h_local = repair_feasible(x)
        return exact_score(h_local)

    def jac(x):
        h_local = np.clip(x, 0, 1)
        values, grads = compute_lag_data(h_local)
        c_max = np.max(values)
        # Use top active lags
        n_act = min(20, len(values))
        act = np.argsort(values)[::-1][:n_act]
        gaps = np.maximum(c_max - values[act], 0.0)
        w = np.exp(-gaps / (0.005 * c_max + 1e-10))
        w /= w.sum() + 1e-15
        g = -np.sum(w[:, None] * grads[act, :], axis=0)
        # Project to maintain sum constraint
        g -= np.mean(g)
        return g

    try:
        result = minimize(
            obj, h, jac=jac,
            method='SLSQP',
            bounds=[(0, 1)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if result.x is not None:
            h_sol = repair_feasible(result.x)
            s_sol = exact_score(h_sol)
            if s_sol < best_score - 1e-14:
                best_h = h_sol
                best_score = s_sol
    except Exception:
        pass

    return best_h, best_score
