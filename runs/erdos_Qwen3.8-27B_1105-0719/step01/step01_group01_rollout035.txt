def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            eligible = np.where(h < 1.0 - 1e-10)[0]
        else:
            eligible = np.where(h > 1e-10)[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-10:
            continue
        p_lo1 = max(0, -k)
        p_hi1 = min(n - 1, n - 1 - k)
        if p_lo1 <= p_hi1:
            grad[p_lo1:p_hi1 + 1] += w * (1.0 - h[p_lo1 + k:p_hi1 + 1 + k])
        p_lo2 = max(0, k)

def line_search_pair(h, i, j, direction, n_iter=18):
    """Line search for transfer: if direction=1, h[i]+=tau, h[j]-=tau"""
    if direction > 0:
        max_tau = min(1.0 - h[i], h[j])
    else:
        max_tau = min(1.0 - h[j], h[i])
    
    if max_tau < 1e-12:
        return 0.0, h, compute_c5(h)

    best_tau = 0.0
    best_c5 = compute_c5(h)

        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h1 = h.copy()
        h2 = h.copy()
        if direction > 0:
            h1[i] += m1; h1[j] -= m1
            h2[i] += m2; h2[j] -= m2
        else:
            h1[j] += m1; h1[i] -= m1
            h2[j] += m2; h2[i] -= m2
        c1 = compute_c5(h1)
        c2 = compute_c5(h2)
        if c1 < best_c5:
            best_c5 = c1; best_tau = m1
        if c2 < best_c5:
            best_c5 = c2; best_tau = m2

            hi = m2
        else:
            lo = m1

        else:
            h_best[j] += best_tau; h_best[i] -= best_tau

import numpy as np
import time
import math
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            eligible = np.where(h < 1.0 - 1e-10)[0]
        else:
            eligible = np.where(h > 1e-10)[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_grad(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-10:
            continue
        p_lo1 = max(0, -k)
        p_hi1 = min(n - 1, n - 1 - k)
        if p_lo1 <= p_hi1:
            grad[p_lo1:p_hi1 + 1] += w * (1.0 - h[p_lo1 + k:p_hi1 + 1 + k])
        p_lo2 = max(0, k)
        p_hi2 = min(n - 1, n - 1 + k)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2 + 1] -= w * h[p_lo2 - k:p_hi2 + 1 - k]
    return grad * dx


def line_search_pair(h, i, j, direction, n_iter=16):
    if direction > 0:
        max_tau = min(1.0 - h[i], h[j])
    else:
        max_tau = min(1.0 - h[j], h[i])
    if max_tau < 1e-12:
        return h, compute_c5(h)
    lo, hi = 0.0, max_tau
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h1 = h.copy()
        h2 = h.copy()
        if direction > 0:
            h1[i] += m1; h1[j] -= m1
            h2[i] += m2; h2[j] -= m2
        else:
            h1[j] += m1; h1[i] -= m1
            h2[j] += m2; h2[i] -= m2
        c1 = compute_c5(h1)
        c2 = compute_c5(h2)
        if c1 < best_c5:
            best_c5 = c1; best_h = h1
        if c2 < best_c5:
            best_c5 = c2; best_h = h2
        if c1 < c2:
            hi = m2
        else:
            lo = m1
    return best_h, best_c5


def local_opt_pairs(h, time_limit, rng, n_iter_ls=14, n_top=150):
    n = len(h)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    t_start = time.time()
    stall = 0
    while time.time() - t_start < time_limit:
        grad = smooth_grad(h, beta=300.0)
        scores = np.zeros(n)
        for i in range(n):
            scores[i] = -grad[i]
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                if h[i] < 1.0 - 1e-8 and h[j] > 1e-8:
                    diff = scores[i] - scores[j]
                    if diff > 0.02:
                        pairs.append((diff, i, j, 1))
                if h[j] < 1.0 - 1e-8 and h[i] > 1e-8:
                    diff2 = scores[j] - scores[i]
                    if diff2 > 0.02:
                        pairs.append((diff2, j, i, 1))
        for _ in range(n // 3):
            a = rng.randint(n)
            b = rng.randint(n)
            if a != b and h[a] < 1.0 - 1e-8 and h[b] > 1e-8:
                pairs.append((rng.random(), a, b, 1))
            elif a != b and h[b] < 1.0 - 1e-8 and h[a] > 1e-8:
                pairs.append((rng.random(), b, a, 1))
        pairs.sort(key=lambda x: -x[0])
        improved = False
        for _, i, j, d in pairs[:n_top]:

                current_c5 = c5_new
                improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
        if not improved:
            stall += 1
            if stall >= 4:
                break
        else:
            stall = 0
    return best_h, best_c5


def anneal_pairs(h, time_limit, rng, T_init=0.04, T_final=0.0005):
    n = len(h)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    t_start = time.time()
    n_acc = 0
    last_improve = 0
    while time.time() - t_start < time_limit:
        progress = min(1.0, (time.time() - t_start) / max(1.0, time_limit))
        T = T_init * math.exp(math.log(max(T_final, 1e-8) / T_init) * progress)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        if direction > 0:
            max_tau = min(1.0 - h[i], h[j])
        else:
            max_tau = min(1.0 - h[j], h[i])
        if max_tau < 1e-12:
            continue
        frac = rng.random() ** (1.0 + 2.0 * progress)
        tau = max_tau * frac
        if tau < 1e-10:
            continue
        h_new = h.copy()
        if direction > 0:
            h_new[i] += tau; h_new[j] -= tau
        else:
            h_new[j] += tau; h_new[i] -= tau
        if h_new.min() < -1e-9 or h_new.max() > 1.0 + 1e-9:
            continue
        new_c5 = compute_c5(h_new)
        delta = new_c5 - current_c5
        if delta < 0 or (T > 1e-10 and math.log(rng.random() + 1e-15) > -delta / T):
            h = h_new
            current_c5 = new_c5
            n_acc += 1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                last_improve = n_acc
        if n_acc - last_improve > 80000 and rng.random() < 0.005:
            h = best_h.copy()
            current_c5 = best_c5
    return best_h, best_c5


def interp_res(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_interp = np.interp(x_new, x_old, h_old, period=0)
    return project_feasible(h_interp, n_new)


def gradient_descent_step(h, lr=0.05, beta=200.0):
    grad = smooth_grad(h, beta=beta)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, len(h))
    return h_new
