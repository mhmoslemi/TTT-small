            G[m, dp:n] += dx * (1.0 - h[0:n-dp])

def binary_round(h):
    """Try rounding to binary and fixing sum constraint."""
    n = len(h)
    target = n / 2
    h_b = (h > 0.5).astype(np.float64)
    # Count ones
    n_ones = h_b.sum()
    diff = target - n_ones
    if diff > 0:
        # Need more ones: flip zeros closest to 0.5
        zeros = np.where(h_b == 0)[0]
        if len(zeros) >= int(diff):
            order = np.argsort(-h[zeros])  # closest to 0.5 first
            for i in range(int(diff)):
                h_b[zeros[order[i]]] = 1.0
    elif diff < 0:
        ones = np.where(h_b == 1)[0]
        order = np.argsort(h[ones])  # closest to 0.5 first
        for i in range(int(-diff)):
            h_b[ones[order[i]]] = 0.0
    # Verify sum
    if abs(h_b.sum() - target) > 1e-10:

def interpolate_to_n(h, new_n):
    n = len(h)

    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)

        return x[-1]

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

        h = x[:-1]
        G = build_grad_matrix(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,

def exact_score(h):

def compute_profile(h):

def project_box_hyperplane(y, target_sum):
    y = np.asarray(y, dtype=np.float64)
    lo = -max(float(y.max()), 0.0) - 1.0
    hi = -min(float(y.min()), 0.0) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return np.clip(h, 0.0, 1.0)

def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, 0:n-d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n-d]
        else:
            dp = -d
            G[m, dp:n] += dx * (1.0 - h[0:n-dp])
            G[m, 0:n-dp] += -dx * h[dp:n]
    return G

def binary_round(h):
    n = len(h)
    target = n // 2
    h_b = (h > 0.5).astype(np.float64)
    n_ones = int(h_b.sum())
    diff = target - n_ones
    if diff > 0:
        zeros = np.where(h_b == 0)[0]
        if len(zeros) >= diff:
            order = np.argsort(-h[zeros])
            for i in range(diff):
                h_b[zeros[order[i]]] = 1.0
    elif diff < 0:
        ones = np.where(h_b == 1)[0]
        order = np.argsort(h[ones])
        for i in range(-diff):
            h_b[ones[order[i]]] = 0.0
    if abs(int(h_b.sum()) - target) > 0:
        return None
    return h_b

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_box_hyperplane(h_new, new_n / 2.0)

def solve_slsqp(h0, maxiter=300):
    n = len(h0)
    c0 = exact_score(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    t0 = c0 + 1e-3

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        G = build_grad_matrix(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}

        h_s2 = solve_slsqp(best_h, maxiter=400)
        if h_s2 is not None:
            c_s2 = exact_score(h_s2)
            if c_s2 < best_c - 1e-10:
                best_h = h_s2
                best_c = c_s2

    # Phase 8: Try binary rounding
    h_bin = binary_round(best_h)
    if h_bin is not None:
        c_bin = exact_score(h_bin)
        if c_bin < best_c - 1e-10:
            best_h = h_bin
            best_c = c_bin

    # Phase 9: Final top-k + SLSQP
    if time_left() > 30:
        h_tk3, c_tk3 = topk_descent(best_h, max_iters=150, k=2, thresh_frac=0.0005)
        if c_tk3 < best_c - 1e-10:
            best_h = h_tk3
            best_c = c_tk3
    if time_left() > 30:
        h_s3 = solve_slsqp(best_h, maxiter=200)
        if h_s3 is not None:
            c_s3 = exact_score(h_s3)
            if c_s3 < best_c - 1e-10:
                best_h = h_s3
                best_c = c_s3

    # Final validation
    best_h = project_box_hyperplane(best_h, len(best_h) / 2.0)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
