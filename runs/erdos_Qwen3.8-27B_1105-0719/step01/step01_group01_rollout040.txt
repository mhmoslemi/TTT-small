For k = -3: n+k = n-3, need 0≤n-3<4 and 0≤n<4, so n=3. c = a[0]*v[3] = 1*1 = 1.
For k = -2: n-2 in [0,3], n in [0,3], so n=2,3. c = a[0]*v[2] + a[1]*v[3] = 1*0 + 0*1 = 0.
For k = -1: n-1 in [0,3], n in [0,3], so n=1,2,3. c = a[0]*v[1] + a[1]*v[2] + a[2]*v[3] = 1*1 + 0*0 + 1*1 = 2.
For k = 0: n in [0,3]. c = a[0]*v[0] + a[1]*v[1] + a[2]*v[2] + a[3]*v[3] = 0 + 0 + 0 + 0 = 0.
For k = 1: n+1 in [0,3], n in [0,3], so n=0,1,2. c = a[1]*v[0] + a[2]*v[1] + a[3]*v[2] = 0 + 1*1 + 0 = 1.
For k = 2: n+2 in [0,3], n in [0,3], so n=0,1. c = a[2]*v[0] + a[3]*v[1] = 1*0 + 0*1 = 0.
