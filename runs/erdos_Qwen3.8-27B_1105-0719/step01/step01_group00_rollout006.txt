def solve_epi_sym(q0, n, active_lags, t0, maxiter=300):

    m = n // 2
    dx = 2.0 / n
    h0 = make_sym_h(q0, n)
    corrs0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
    
    x0 = np.concatenate([[t0], q0])
    
    epi = EpiSym(n, dx, active_lags)
    
    cons = []
    for k in range(len(active_lags)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, k=k: epi.constraint(x, k),
            'jac': lambda x, k=k: epi.constraint_jac(x, k)
        })
    
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    
    try:
        res = minimize(epi.fun, x0, jac=epi.jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})

    except Exception:
        pass
    return None, None

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):

            lo = mid
        else:

    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def weighted_grad_full(h, n, dx, weights):
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        wk = weights[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    return grad * dx

    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def smooth_sym_grad(x, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    G = weighted_grad_full(h, n, dx, beta * w)

class EpiSymSolver:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.active_lags = np.array(active_lags, dtype=int)

    def fun(self, x):
        return x[0]

    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, self.n)
        corrs = corr_vals(h)
        return t - corrs[self.active_lags]

    def con_jac(self, x):
        q = x[1:]
        h = make_sym_h(q, self.n)
        n = self.n
        m = self.m
        dx = self.dx
        L = len(self.active_lags)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.active_lags):
            d = lag - (n - 1)

                G[idx + d] += (-h[idx])
            G *= dx

        q_new, c5_new = solve_epi_sym(q_best, n, active,
                                       t_init=corrs.max() * 1.0001, maxiter=600)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  final epi: C5={c5_best:.7f}")

    # Phase 6: Try nearby n values
    if remaining() > 40:
        for delta in [-2, 2, -4, 4, -6, 6]:
            n_try = n + delta
            if n_try < 40 or n_try % 2 != 0 or remaining() < 20:
                continue
            dx_try = 2.0 / n_try
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)

            q_new, c5_new = smooth_sym_optimize(q_try, n_try, 200, maxiter=300)
            if q_new is not None:
                q_new, c5_new = smooth_sym_optimize(q_new, n_try, 600, maxiter=300)
            if q_new is not None and c5_new < c5_best - 1e-9:
                c5_best = c5_new
                n = n_try
                m = m_try
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  n={n}: C5={c5_best:.7f}")

    # Phase 7: Final full-space polish
    if remaining() > 20:
        h_new, c5_new = full_smooth_optimize(h_best, len(h_best), 800, maxiter=500)
        if c5_new < c5_best - 1e-10:
            c5_best = c5_new
            h_best = h_new
            print(f"  final full: C5={c5_best:.7f}")

    # Final validation and cleanup
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)

    return h_best, c5_best, len(h_best)
