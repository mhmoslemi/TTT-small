import numpy as np
from scipy.optimize import linprog
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(prof)), prof

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.clip(np.nan_to_num(h), 0.0, 1.0).astype(np.float64).copy()
    lo = h.min() - 1.5
    hi = h.max() + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-14:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if np.any(free):
            proj[free] += diff / np.sum(free)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def lag_grad(h, lag, dx):
    n = len(h)
    g = np.zeros(n)
    if 0 <= lag < n:
        s = n - lag
        g[:s] -= h[lag:]
        g[lag:] += 1.0 - h[:s]
    elif -n < lag < 0:
        a = -lag
        s = n - a
        g[:s] += 1.0 - h[a:]
        g[a:] -= h[:s]
    return g * dx

def line_search(h, d, c_init, dx, max_alpha=1.0):
    n = len(h)
    ma = max_alpha
    neg = d < -1e-12
    pos = d > 1e-12
    if np.any(neg):
        ma = min(ma, np.min(h[neg] / (-d[neg])))
    if np.any(pos):
        ma = min(ma, np.min((1.0 - h[pos]) / d[pos]))
    ma = min(ma, max_alpha)
    if ma < 1e-10:
        return h.copy(), c_init
    best_h = h.copy()
    best_c = c_init
    alphas = np.concatenate([np.linspace(0, ma, 12), np.geomspace(ma * 0.01, ma, 15)])
    alphas = np.unique(np.sort(np.clip(alphas, 1e-8, ma)))
    for a in alphas:
        ht = h + a * d
        ht = np.clip(ht, 0.0, 1.0)
        ct = float(np.max(np.correlate(ht, 1.0 - ht, mode='full') * dx))
        if ct < best_c - 1e-14:
            best_c = ct
            best_h = ht.copy()
    return best_h, best_c

def active_set_step(h, profile, c, dx, trust):
    n = len(h)
    atol = max(1e-6, 0.04 * c)
    active = np.where(profile >= c - atol)[0]
    if len(active) > 30:
        ap = profile[active]
        top = np.argsort(-ap)[:30]
        active = active[top]
    lags = active - (n - 1)
    A_rows = []
    for lag in lags:
        g = lag_grad(h, lag, dx)
        A_rows.append(np.append(g, -1.0))
    if not A_rows:
        return h, c, False
    A_ub = np.array(A_rows)
    b_ub = np.zeros(len(lags))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    lower = np.full(n, -trust)
    upper = np.full(n, trust)
    for i in range(n):
        if h[i] <= 1e-10:
            lower[i] = 0.0
        if h[i] >= 1.0 - 1e-10:
            upper[i] = 0.0
    bounds = list(zip(lower, upper))
    bounds.append((-1e6, 1e6))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h, c, False
    if not res.success or res.x is None:
        return h, c, False
    t_val = res.x[n]
    if t_val >= -1e-10:
        return h, c, False
    d = res.x[:n]
    new_h, new_c = line_search(h, d, c, dx, max_alpha=trust)
    if new_c < c - 1e-13:
        return new_h, new_c, True
    return h, c, False

def neg_grad_step(h, profile, c, dx, trust):
    n = len(h)
    max_idx = int(np.argmax(profile))
    lag = max_idx - (n - 1)
    g = lag_grad(h, lag, dx)
    d = -g
    d -= d.mean()
    for i in range(n):
        if h[i] <= 1e-10 and d[i] < 0:
            d[i] = 0.0
        if h[i] >= 1.0 - 1e-10 and d[i] > 0:
            d[i] = 0.0
    d -= d.mean()
    mn = np.max(np.abs(d))
    if mn < 1e-12:
        return h, c, False
    d = d / mn * trust
    d -= d.mean()
    new_h, new_c = line_search(h, d, c, dx, max_alpha=trust)
    if new_c < c - 1e-13:
        return new_h, new_c, True
    return h, c, False

def active_set_optimize(h, deadline, max_iter=300, init_trust=0.15):
    n = len(h)
    dx = 2.0 / n
    h = project_box_sum(h, 0.5 * n)
    c, profile = exact_objective(h)
    trust = init_trust
    stall = 0
    max_stall = 80
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        new_h, new_c, ok = active_set_step(h, profile, c, dx, trust)
        if ok:
            h = new_h
            c = new_c
            profile = exact_objective(h)[1]
            stall = 0
            trust = min(0.4, trust * 1.08)
        else:
            new_h, new_c, ok2 = neg_grad_step(h, profile, c, dx, trust * 0.5)
            if ok2:
                h = new_h
                c = new_c
                profile = exact_objective(h)[1]
                stall = 0
                trust = min(0.4, trust * 1.02)
            else:
                stall += 1
                trust *= 0.72
        if stall > max_stall or trust < 5e-7:
            break
    return h, c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=100, rng=None):
    n = len(b)
    if b is None or n % 2 != 0:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    c5, _ = exact_objective(b)
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 2000:
            for i in ones:
                for j in zeros:
                    b[i], b[j] = 0.0, 1.0
                    nc, _ = exact_objective(b)
                    if nc < c5 - 1e-13:
                        c5 = nc
                        improved = True
                        ones = np.where(b > 0.5)[0]
                        zeros = np.where(b < 0.5)[0]
                        break
                    b[i], b[j] = 1.0, 0.0
                if improved:
                    break
        else:
            n_tries = 800
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc, _ = exact_objective(b)
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = np.where(b > 0.5)[0]
                    zeros = np.where(b < 0.5)[0]
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        cnt = int(round(target))
        idx = rng.choice(n, size=min(cnt, n), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(2, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.12
    else:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.2
    return project_box_sum(h, target)

def smooth_and_refine(h, deadline, max_iter=80):
    n = len(h)
    dx = 2.0 / n
    target = 0.5 * n
    c, prof = exact_objective(h)

    def smooeval_tau(x, tau):
        p = np.correlate(x, 1.0 - x, mode='full') * dx
        pmax = np.max(p)
        sh = np.clip((p - pmax) / tau, -80, 80)
        w = np.exp(sh)
        ws = np.sum(w)
        return pmax + tau * np.log(ws)

    from scipy.optimize import minimize
    best_h = h.copy()
    best_c = c
    taus = [0.003, 0.001, 0.0004, 0.00015, 0.00006]
    for tau in taus:
        if time.time() > deadline:
            break
        def obj_grad(x):
            p = np.correlate(x, 1.0 - x, mode='full') * dx
            pmax = np.max(p)
            sh = np.clip((p - pmax) / tau, -80, 80)
            w = np.exp(sh)
            ws = np.sum(w)
            val = pmax + tau * np.log(ws)
            wnorm = w / ws
            grad = np.zeros(n)
            active_idx = np.where(wnorm > 1e-6)[0]
            for m in active_idx:
                lag = m - (n - 1)
                g = lag_grad(x, lag, dx)
                grad += wnorm[m] * g
            se = np.sum(x) - target
            val += 1e4 * se * se
            grad = grad + 2e4 * se
            below = np.maximum(0.0, -x)
            above = np.maximum(0.0, x - 1.0)
            val += 1e4 * (np.sum(below * below) + np.sum(above * above))
            grad = grad - 2e4 * below + 2e4 * above
            return val, grad
        try:
            res = minimize(obj_grad, best_h, method='L-BFGS-B',
                           jac=True, options={'maxiter': max(30, min(max_iter, n + 20))})
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                cc, _ = exact_objective(cand)
                if cc < best_c - 1e-13:
                    best_c = cc
                    best_h = cand
        except Exception:
            pass
    return best_h, best_c
