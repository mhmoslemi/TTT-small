def project_box_sum(h, target=None):

    n = len(h)

            h[mask] -= step
        else:

            h[mask] += step

def compute_subgradient(h, tol=1e-8):
    """Compute a subgradient of C(h) = max_t p[t]."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.max(profile)
    active = profile >= c - tol
    # Subgradient is average gradient over active profiles
    grad = np.zeros(n)
    for t in range(2*n-1):
        if not active[t]:
            continue
        s = t - (n-1)  # signed offset
        # Gradient of p[t] with respect to h
        # p[t] = dx * sum_{valid k} h[k] * (1-h[k-s])
        # ∂p[t]/∂h[j] = dx * [(1-h[j-s]) if j-s valid (j is first index)
        #                  - h[j+s] if j+s valid (j is second index)]
        if s == 0:
            grad += dx * (1.0 - 2.0 * h)
        elif s > 0:
            # j is first index: 0 ≤ j < n, 0 ≤ j-s < n → s ≤ j ≤ n-1
            # j is second index: 0 ≤ j+s < n → j ≤ n-1-s
            # First index contribution: (1-h[j-s]) for j in [s, n-1]
            grad[s:] += dx * (1.0 - h[:n-s])
            # Second index contribution: -h[j+s] for j in [0, n-1-s]
            grad[:n-s] -= dx * h[s:]
        else:
            s2 = -s
            # j is first index: 0 ≤ j < n, 0 ≤ j+s2 < n → j ≤ n-1-s2
            # j is second index: 0 ≤ j-s2 < n → j ≥ s2
            grad[:n-s2] += dx * (1.0 - h[s2:])
            grad[s2:] -= dx * h[:n-s2]
    # Average over active set
    n_active = np.sum(active)
    grad /= n_active
    return c, grad, active

    # First, find the range of valid alpha
    # h + alpha * direction must be in [0,1]

    neg_mask = direction < 0

    if np.any(pos_mask):

    if np.any(neg_mask):
        alpha_max = min(alpha_max, np.min(-h[neg_mask] / direction[neg_mask]))
    alpha_max = max(alpha_max, 0.0)
    
    # Sample alpha at many points
    best_alpha = 0.0
    best_c = c_current
    alphas = np.linspace(0, alpha_max, max_iter)
    for a in alphas:
        if a <= 0:
            continue
        h_new = h + a * direction
        c_new = exact_objective(h_new)
        if c_new < best_c:
            best_c = c_new
            best_alpha = a
    return best_alpha, best_c

def subgradient_descent(h, max_iter=500, tol=1e-10):
    """Subgradient descent with exact line search."""
    h = h.copy()
    c = exact_objective(h)
    for it in range(max_iter):
        c, grad, active = compute_subgradient(h)
        # Direction: -grad projected onto feasible set
        # Project -grad to have sum 0 (to maintain sum constraint)
        direction = -grad - np.mean(-grad)
        # Also need to respect box constraints - for now, just use line search
        alpha, c_new = line_search(h, direction, c)
        if alpha <= 1e-12:
            break
        h = h + alpha * direction
        h = project_box_sum(h)  # Re-project in case of numerical issues
        c = exact_objective(h)
        if c_new < c - tol:
            c = c_new
    return h, c
