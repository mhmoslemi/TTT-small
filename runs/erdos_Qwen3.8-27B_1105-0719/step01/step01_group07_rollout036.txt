def exact_objective(h):

    if n == 0:

    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c = float(np.max(profile))
    return c if np.isfinite(c) else float('inf')

    n = len(h)
    if n == 0:

        s = np.sum(h)

        if abs(diff) < 1e-13:

            cnt = int(np.sum(mask))
            if cnt == 0:

            step = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:

            if cnt == 0:

    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])

                        c = exact_objective(b)
                        b[oi] = b[oj] = 1.0
                        b[ki] = b[kj] = 0.0
                        imp = curr_c - c
                        if imp > best_improve:
                            best_improve = imp
                            best_move = (oi, oj, ki, kj)
        if best_move is None:
            break
        oi, oj, ki, kj = best_move
        b[oi] = b[oj] = 0.0
        b[ki] = b[kj] = 1.0
        curr_c = exact_objective(b)
    
    return b, curr_c

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    prof = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * dx
    c = float(np.max(prof))
    return c if np.isfinite(c) else float('inf')

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if n == 0:
        return h
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)
