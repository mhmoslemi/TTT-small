def active_set_descent(h, max_iters=200, base_step=0.01):
    """Active-set subgradient minimization with adaptive step size."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = profile.max()
        
        # Select active constraints (within tolerance of max)
        threshold = c - max(1e-4, 0.01 * c)
        active = np.where(profile >= threshold)[0]
        
        # If too many active, use softmax weighting
        if len(active) > 20:
            w = np.exp((profile[active] - c) / 0.001)
            w /= w.sum()
        elif len(active) > 1:
            # Weight by how close to max
            spans = profile[active] - threshold
            w = spans / spans.sum()
        else:
            w = np.array([1.0])
        
        # Compute gradient of weighted sum of active constraints
        G = build_gradient(h, n, dx)
        grad = G[active, :].T @ w
        # Remove mean (sum constraint)
        grad -= grad.mean()
        
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break
        grad /= norm
        
        # Backtracking line search
        step = base_step
        found = False
        for _ in range(10):
            h_try = repair_feasible(h - step * grad)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-12:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                found = True
                break
            step *= 0.5
        if not found:
            base_step *= 0.7
            if base_step < 1e-6:
                break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.asarray(vec, dtype=np.float64).copy()
    target = n / 2.0
    a_lo = -abs(h).max() - 1.0
    a_hi = abs(h).max() + 1.0
    for _ in range(60):
        a_mid = 0.5 * (a_lo + a_hi)
        s = np.clip(h + a_mid, 0.0, 1.0).sum()
        if s < target:
            a_lo = a_mid
        else:
            a_hi = a_mid
        if a_hi - a_lo < 1e-16:
            break
    h = np.clip(h + 0.5 * (a_lo + a_hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if abs(diff) < 1e-15:
                    break
    return h

def build_gradient(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    mid = n - 1
    for m in range(M):
        d = m - mid
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            j_arr = np.arange(n)
            g = np.zeros(n)
            mask1 = (j_arr >= d) & (j_arr < n)
            g[mask1] += dx
            mask2 = j_arr >= d
            g[mask2] -= dx * h[j_arr[mask2] - d]
            mask3 = j_arr <= n - 1 - d
            g[mask3] -= dx * h[j_arr[mask3] + d]
            G[m, :] = g
        else:
            s = -d
            j_arr = np.arange(n)
            g = np.zeros(n)
            mask1 = j_arr <= n - 1 - s
            g[mask1] += dx
            mask2 = j_arr <= n - 1 - s
            g[mask2] -= dx * h[j_arr[mask2] + s]
            mask3 = j_arr >= s
            g[mask3] -= dx * h[j_arr[mask3] - s]
            G[m, :] = g
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

    return x[-1]

    g = np.zeros(n + 1)

    return g

    return x[:-1].sum() - n / 2.0

                    improved = True
                    break
        if not improved:
            eps *= 0.6
            if eps < 1e-6:
                break
    return h, best_c

def pair_descent_polish(h, max_rounds=40, eps=2e-3, rng=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    if rng is None:
        rng = np.random.RandomState(0)
    for _ in range(max_rounds):
        improved = False
        for _ in range(n * 3):
            i, j = rng.choice(n, size=2, replace=False)
            for di in [eps, -eps]:
                for dj in [eps, -eps]:
                    h_try = h.copy()
                    h_try[i] += di
                    h_try[j] += dj
                    h_try = repair_feasible(h_try)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-14:
                        h = h_try
                        best_c = c_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c

def random_subset_search(h, rng, max_iters=5000, amp=0.04):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    for it in range(max_iters):
        k = rng.randint(2, 7)
        idx = rng.choice(n, size=k, replace=False)
        delta = rng.randn(k) * amp
        h_try = h.copy()
        h_try[idx] += delta
        h_try = repair_feasible(h_try)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-13:
            h = h_try
            best_h = h.copy()
            best_c = c_try
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(66, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    h = parent.copy()
    best_c = exact_score(h)
    best_h = h.copy()

    # Symmetrize
    h_sym = repair_feasible((h + h[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c:
        h, best_c, best_h = h_sym, c_sym, h_sym.copy()

    # Stage 1: Trust-constr from symmetrized parent
    if time_left() > 200:
        h_tc, c_tc = trust_constr_solve(h, maxiter=400, tol=1e-10)
        if h_tc is not None and c_tc < best_c:
            h, best_c, best_h = h_tc, c_tc, h_tc.copy()

    # Stage 2: SLSQP from best
    if time_left() > 150:
        h_sq, c_sq = slsqp_solve(h, maxiter=400, tol=1e-11)
        if h_sq is not None and c_sq < best_c:
            h, best_c, best_h = h_sq, c_sq, h_sq.copy()

    # Stage 3: Subgradient descent
    if time_left() > 80:
        h_sg, c_sg = active_set_descent(h, max_iters=200, base_step=0.006)
        if c_sg < best_c:
            h, best_c, best_h = h_sg, c_sg, h_sg.copy()

    # Stage 4: Coordinate descent polish
    if time_left() > 60:
        h_cd, c_cd = coordinate_descent_polish(h, max_rounds=60, init_eps=3e-3)
        if c_cd < best_c:
            h, best_c, best_h = h_cd, c_cd, h_cd.copy()

    # Stage 5: Random subset search
    if time_left() > 80:
        h_rs, c_rs = random_subset_search(h, rng, max_iters=4000, amp=0.03)
        if c_rs < best_c:
            h, best_c, best_h = h_rs, c_rs, h_rs.copy()

    # Stage 6: Trust-constr from improved point
    if time_left() > 150:
        h_tc2, c_tc2 = trust_constr_solve(h, maxiter=400, tol=1e-10)
        if h_tc2 is not None and c_tc2 < best_c:
            h, best_c, best_h = h_tc2, c_tc2, h_tc2.copy()

    # Stage 7: Subgradient + CD
    if time_left() > 60:
        h_sg2, c_sg2 = active_set_descent(h, max_iters=150, base_step=0.004)
        if c_sg2 < best_c:
            h, best_c, best_h = h_sg2, c_sg2, h_sg2.copy()
        h_cd2, c_cd2 = coordinate_descent_polish(h, max_rounds=50, init_eps=2e-3)
        if c_cd2 < best_c:
            h, best_c, best_h = h_cd2, c_cd2, h_cd2.copy()

    # Stage 8: Pair descent
    if time_left() > 40:
        h_pd, c_pd = pair_descent_polish(h, max_rounds=50, eps=1.5e-3, rng=rng)
        if c_pd < best_c:
            h, best_c, best_h = h_pd, c_pd, h_pd.copy()

    # Stage 9: Try perturbed starts with SLSQP
    if time_left() > 100:
        for eps_val in [0.005, 0.01, 0.02, 0.03]:
            if time_left() < 40:
                break
            z = rng.randn(len(h))
            z -= z.mean()
            h_pert = repair_feasible(h + eps_val * z)
            h_sq3, c_sq3 = slsqp_solve(h_pert, maxiter=200, tol=1e-10)
            if h_sq3 is not None and c_sq3 < best_c:
                h, best_c, best_h = h_sq3, c_sq3, h_sq3.copy()

    # Stage 10: Try different dimensions
    if time_left() > 100:
        n_cur = len(h)
        for new_n in [n_cur - 2, n_cur + 2, n_cur - 4, n_cur + 4]:
            if time_left() < 40:
                break
            if new_n < 20 or new_n > 200:
                continue
            h_interp = interpolate_to_n(h, new_n)
            h_tc3, c_tc3 = trust_constr_solve(h_interp, maxiter=200, tol=1e-10)
            if h_tc3 is not None and c_tc3 < best_c:
                h, best_c, best_h = h_tc3, c_tc3, h_tc3.copy()
            else:
                h_sq4, c_sq4 = slsqp_solve(h_interp, maxiter=200, tol=1e-10)
                if h_sq4 is not None and c_sq4 < best_c:
                    h, best_c, best_h = h_sq4, c_sq4, h_sq4.copy()

    # Stage 11: Final subgradient + CD polish
    if time_left() > 40:
        h_sg3, c_sg3 = active_set_descent(h, max_iters=100, base_step=0.003)
        if c_sg3 < best_c:
            h, best_c, best_h = h_sg3, c_sg3, h_sg3.copy()
        h_cd3, c_cd3 = coordinate_descent_polish(h, max_rounds=80, init_eps=1e-3)
        if c_cd3 < best_c:
            h, best_c, best_h = h_cd3, c_cd3, h_cd3.copy()

    # Stage 12: Pair descent final
    if time_left() > 20:
        h_pd2, c_pd2 = pair_descent_polish(h, max_rounds=30, eps=8e-4, rng=rng)
        if c_pd2 < best_c:
            h, best_c, best_h = h_pd2, c_pd2, h_pd2.copy()

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
