import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def profile_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_polytope(x, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(free) > 0:
            idx = free[np.argmin(np.abs(h[free] - 0.5))]
            h[idx] += diff
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def fw_vertex(g, n):
    target = n // 2
    v = np.zeros(n)
    v[np.argsort(g)[:target]] = 1.0
    return v

def smooth_value_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    mu = np.max(profile)
    shifted = np.clip((profile - mu) / tau, -500.0, 500.0)
    exps = np.exp(shifted)
    logsum = np.log(exps.sum())
    val = mu + tau * logsum
    w = exps / exps.sum()
    grad = np.zeros(n)
    for m in range(len(profile)):
        d = m - (n - 1)
        wm = w[m]
        if wm < 1e-12:
            continue
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += wm * dx * (1.0 - h[:n - d])
            grad[:n - d] -= wm * dx * h[d:]
        else:
            s = -d
            grad[:n - s] += wm * dx * (1.0 - h[s:])
            grad[s:] -= wm * dx * h[:n - s]
    return val, grad

def fw_steps(h, tau, max_iter, deadline, verbose=False):
    n = len(h)
    h = h.copy()
    f_prev, _ = smooth_value_and_grad(h, tau)
    best_exact = exact_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        if time.perf_counter() > deadline:
            break
        f, g = smooth_value_and_grad(h, tau)
        v = fw_vertex(g, n)
        # Check if v == h (stationary in FW sense)
        if np.allclose(h, v, atol=1e-8):
            break
        # Shrinkage step: alpha = 2/(t+2)
        alpha = 2.0 / (it + 2.0)
        h_new = (1.0 - alpha) * h + alpha * v
        h_new = project_polytope(h_new, n)
        f_new, _ = smooth_value_and_grad(h_new, tau)
        if f_new < f_prev + 1e-12:
            h = h_new
            f_prev = f_new
            c = exact_c5(h)
            if c < best_exact:
                best_exact = c
                best_h = h.copy()
        else:
            # Try smaller step
            alpha2 = alpha * 0.5
            h_new2 = (1.0 - alpha2) * h + alpha2 * v
            h_new2 = project_polytope(h_new2, n)
            f_new2, _ = smooth_value_and_grad(h_new2, tau)
            if f_new2 < f_prev + 1e-12:
                h = h_new2
                f_prev = f_new2
                c = exact_c5(h)
                if c < best_exact:
                    best_exact = c
                    best_h = h.copy()
            else:
                break
    return best_h, best_exact

def proj_grad_desc(h, tau, max_iter, deadline, lr_init=0.01):
    n = len(h)
    h = project_polytope(h.copy(), n)
    f_prev, _ = smooth_value_and_grad(h, tau)
    best_exact = exact_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        if time.perf_counter() > deadline:
            break
        f, g = smooth_value_and_grad(h, tau)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        # Backtracking line search
        lr_try = lr
        improved = False
        for bt in range(20):
            y = h - lr_try * g
            h_new = project_polytope(y, n)
            f_new, _ = smooth_value_and_grad(h_new, tau)
            if f_new < f - 1e-4 * lr_try * (g @ (h_new - h)):
                h = h_new
                f_prev = f_new
                lr = lr_try * 1.5
                improved = True
                c = exact_c5(h)
                if c < best_exact:
                    best_exact = c
                    best_h = h.copy()
                break
            lr_try *= 0.5
        if not improved:
            lr = lr_try
            if lr < 1e-10:
                break
            # Try anyway with a very small step
            h_new = project_polytope(h - lr * g, n)
            f_new, _ = smooth_value_and_grad(h_new, tau)
            if f_new < f_prev:
                h = h_new
                f_prev = f_new
                c = exact_c5(h)
                if c < best_exact:
                    best_exact = c
                    best_h = h.copy()
    return best_h, best_exact

def slsqp_epigraph(h0, maxiter=200, tol=1e-10):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_c5(h0)
    t0 = c0 + 5e-5

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - prof

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                J[m, :-1] = -dx * (1.0 - 2.0 * h)
            elif d > 0:
                J[m, d:n] += -dx * (1.0 - h[:n - d])
                J[m, 0:n - d] += dx * h[d:]
            else:
                s = -d
                J[m, 0:n - s] += -dx * (1.0 - h[s:])
                J[m, s:n] += dx * h[:n - s]
            J[m, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = project_polytope(res.x[:-1], n)
            return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return project_polytope(h.copy(), new_n)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_polytope(h_new, new_n)

def active_set_refine(h, max_iter=100, deadline=None):
    """Iteratively equalize the top profile entries."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_c5(h)
    for _ in range(max_iter):
        if deadline and time.perf_barrier() > deadline:
            break
        if time.perf_counter() > (deadline or 1e18):
            break
        prof = profile_of(h)
        c = prof.max()
        if c < best_c - 1e-12:
            best_c = c
        # Find top K active entries
        K = min(10, M)
        active = np.argsort(prof)[-K:]
        thresh = prof[active[0]] - 1e-6
        active = np.where(prof >= thresh)[0]
        if len(active) < 2:
            break
        # Compute weighted gradient on active set
        w = prof[active] - thresh
        w = w / (w.sum() + 1e-15)
        grad = np.zeros(n)
        for idx in active:
            d = idx - (n - 1)
            wi = w[list(active).index(idx)]
            if d == 0:
                grad += wi * dx * (1.0 - 2.0 * h)
            elif d > 0:
                grad[d:] += wi * dx * (1.0 - h[:n - d])
                grad[:n - d] -= wi * dx * h[d:]
            else:
                s = -d
                grad[:n - s] += wi * dx * (1.0 - h[s:])
                grad[s:] -= wi * dx * h[:n - s]
        # Subtract mean to respect sum constraint
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        # Backtracking
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = project_polytope(h - alpha * grad, n)
            c_try = exact_c5(h_try)
            if c_try < best_c - 1e-11:
                h = h_try
                best_c = c_try
                break
    return h, best_c
