```python
import numpy as np
import time

def eval_profile(h):
    """Return raw profile (without dx scaling)."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    p = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(p) * dx)

def project_box_sum(h, target):
    n = len(h)
    h = h.astype(np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff/cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff/cnt, 0.0)
    return h

def to_binary(h):
    """Convert to binary: top n/2 values become 1."""
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b
```

```python
    tried = set()
    n_improvements = 0
    
    # Try pairs: low index (donor) to high index (receiver)
    for a in range(n):
        for b in range(n):
            if a == b or time.time() > deadline - 5:
                continue
            key = (min(a,b), max(a,b))
            if key in tried:
                continue
            tried.add(key)
            
            i = a  # receiver
            j = b  # donor
            L = max(-h[i], h[j] - 1.0)
            U = min(1.0 - h[i], h[j])
            if U - L < 1e-10:
                continue
            
            h_new, c_new = fractional_1d(h.copy(), i, j, n_gs=35)
            if c_new < c5 - 1e-12:
                c5 = c_new
                h = h_new
                n_improvements += 1
    
    return h, c5

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def make_starts(n, rng):
    """Generate diverse starting points."""
    target = 0.5 * n
    starts = []
    
    # Alternating
    h1 = np.zeros(n)
    h1[::2] = 1.0
    h1[n-1] = 0.0
    h1 = project_box_sum(h1, target)
    starts.append(h1)
    
    # Block
    h2 = np.zeros(n)
    h2[:target] = 1.0
    starts.append(h2)
    
    # Random
    h3 = np.zeros(n)
    idx = rng.choice(n, size=int(round(target)), replace=False)
    h3[idx] = 1.0
    starts.append(h3)
    
    # Uniform + noise
    h4 = np.full(n, 0.5) + rng.randn(n) * 0.1
    h4 = project_box_sum(h4, target)
    starts.append(h4)
    
    # Two blocks
    q = n // 4
    h5 = np.zeros(n)
    h5[:q] = 1.0
    pos2 = n // 2 + rng.randint(0, n // 4)
    pos2 = min(pos2, n - q)
    h5[pos2:pos2+q] = 1.0
    h5 = project_box_sum(h5, target)
    starts.append(h5)
    
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)
    
    best_c = float('inf')
    best_h = None
    best_n = 0
    
    # Try to use initial_h_values
    n_use = 70
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n_use = len(init)
            init = project_box_sum(init, 0.5 * n_use)
            c_init = eval_c5(init)
            best_h = init.copy()
            best_c = c_init
    except (NameError, Exception):
        pass
    
    if best_h is None:
        best_h = make_starts(n_use, rng)[0]
        best_c = eval_c5(best_h)
    
    # Phase 1: Binary ILI on the parent resolution
    if n_use % 2 == 0:
        b0 = to_binary(best_h)
        if b0 is not None:
            b1, c1 = binary_ils(b0, rng, deadline - 60)
            if c1 < best_c:
                best_c = c1
                best_h = b1
                best_n = n_use
    
    # Phase 2: Fractional refinement on best binary
    n_cur = len(best_h)
    if time.time() < deadline - 80:
        h_frac = best_h.astype(np.float64).copy()
        h_frac = project_box_sum(h_frac, 0.5 * n_cur)
        # Try selective fractional pairs (only ones near 0 or 1 that might benefit)
        for _ in range(3):
            if time.time() > deadline - 60:
                break
            h_frac, c_frac = fractional_pass(h_frac, deadline - 40)
            if c_frac < best_c:
                best_c = c_frac
                best_h = h_frac.copy()
                best_n = n_cur
            else:
                break
    
    # Phase 3: Try other resolutions
    sizes_to_try = []
    base = n_cur
    for s in [base - 2, base - 4, base + 2, base + 4, base - 6, base + 6,
              56, 64, 72, 80, 84, 60, 68, 76]:
        if 20 <= s <= 120 and s % 2 == 0 and s != base:
            sizes_to_try.append(s)
    sizes_to_try = sorted(set(sizes_to_try), key=lambda x: abs(x - base))
    
    for n_try in sizes_to_try:
        if time.time() > deadline - 30:
            break
        
        # Create candidate: downsample or upsample best solution
        h_cand = downsample(best_h, n_try)
        b_cand = to_binary(h_cand)
        
        if b_cand is not None:
            c_cand = eval_c5(b_cand)
            if c_cand < eval_c5(h_cand):
                h_cand = b_cand
            
            # Binary ILI
            b_opt, c_opt = binary_ils(h_cand, rng, deadline - 20)
            if c_opt < best_c:
                best_c = c_opt
                best_h = b_opt
                best_n = n_try
            
            # Fractional
            if time.time() < deadline - 15:
                h_f = b_opt.astype(np.float64).copy()
                h_f = project_box_sum(h_f, 0.5 * n_try)
                h_f, c_f = fractional_pass(h_f, deadline - 10)
                if c_f < best_c:
                    best_c = c_f
                    best_h = h_f
                    best_n = n_try
    
    # Phase 4: More aggressive multi-start at best resolution
    n_best = best_n
    if time.time() < deadline - 40:
        for _ in range(5):
            if time.time() > deadline - 20:
                break
            starts = make_starts(n_best, rng)
            for s in starts:
                b_s = to_binary(s)
                if b_s is None:
                    continue
                b_opt, c_opt = binary_ils(b_s, rng, deadline - 10)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = b_opt
                    best_n = n_best
                    # Fractional from new best
                    if time.time() < deadline - 10:
                        h_f = best_h.copy()
                        h_f, c_f = fractional_pass(h_f, deadline - 5)
                        if c_f < best_c:
                            best_c = c_f
                            best_h = h_f
                            best_n = n_best
    
    # Phase 5: Final tight refinement
    if time.time() < deadline - 10:
        n_f = len(best_h)
        if n_f % 2 == 0:
            b_f = to_binary(best_h)
            if b_f is not None:
                b_f, c_f = binary_ils(b_f, rng, deadline)
                if c_f < best_c:
                    best_c = c_f
                    best_h = b_f
                    best_n = n_f
        
        # Final fractional pass
        if time.time() < deadline - 5:
            h_f = best_h.astype(np.float64).copy()
            h_f = project_box_sum(h_f, 0.5 * len(h_f))
            h_f, c_f = fractional_pass(h_f, deadline)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
                best_n = len(h_f)
    
    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c = eval_c5(best_h)
    best_c = final_c
    best_n = len(best_h)
    
    return (best_h, best_c, best_n)
```
