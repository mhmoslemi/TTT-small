def coord_line_min(h, j, corr_cache=None):
    """Find optimal h[j] given all other coordinates fixed."""
    n = len(h)
    t_old = h[j]
    
    # For each lag k (mapped to correlation index i = k + n - 1),
    # compute alpha_k and beta_k such that corr[i] = alpha_k * t + beta_k
    alphas = np.zeros(2*n-1)
    betas = np.zeros(2*n-1)
    
    for i in range(2*n-1):
        k = i - (n-1)
        # Valid j indices: 0 <= j' < n and 0 <= j'+k < n
        j_lo = max(0, -k)
        j_hi = min(n-1, n-1-k)
        
        alpha = 0.0
        beta = 0.0
        for j_prime in range(j_lo, j_hi+1):
            if j_prime == j:
                # This term is t * (1 - h[j+k])
                alpha += (1.0 - h[j+k])
            elif j_prime + k == j:
                # This term is h[j_prime] * (1 - t)
                alpha += -h[j_prime]
                beta += h[j_prime]
            else:
                beta += h[j_prime] * (1.0 - h[j_prime+k])
        
        alphas[i] = alpha
        betas[i] = beta
    
    # Now find min of max_i (alphas[i]*t + betas[i]) over t in [0,1]
    # This is a 1D LP in disguise (min of max of linear functions)
    # Critical points: intersections of pairs of lines, plus t=0 and t=1
    
    candidates = [0.0, 1.0]
    for i in range(len(alphas)):
        for m in range(i+1, len(alphas)):
            da = alphas[i] - alphas[m]
            db = betas[i] - betas[m]
            if abs(da) > 1e-14:
                t_cross = -db / da
                if 0.0 <= t_cross <= 1.0:
                    candidates.append(t_cross)
    
    best_t = t_old
    best_val = np.inf
    for t in candidates:
        vals = alphas * t + betas
        obj = np.max(vals)
        if obj < best_val:
            best_val = obj
            best_t = t
    
    return best_t, best_val

def coord_desc_step(h, j):
    """Minimize C5 over h[j] with all other coordinates fixed. Returns new h[j]."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # For each lag k (index i = k + n - 1), compute alpha and beta
    # such that C[k](t) = alpha * t + beta where t = h[j]
    alphas = np.zeros(2*n-1)
    betas = np.zeros(2*n-1)
    
    for i in range(2*n-1):
        k = i - (n-1)
        # valid1: j is a valid first index, i.e., 0 <= j+k < n
        valid1 = (0 <= j + k < n)
        # valid2: j-k is a valid first index, i.e., 0 <= j-k < n
        valid2 = (0 <= j - k < n)
        
        if valid1 and valid2:
            alphas[i] = (1.0 - h[j+k]) - h[j-k]
            betas[i] = corr[i] - h[j]*(1.0 - h[j+k]) + h[j]*h[j-k]
        elif valid1:
            alphas[i] = (1.0 - h[j+k])
            betas[i] = corr[i] - h[j]*(1.0 - h[j+k])
        elif valid2:
            alphas[i] = -h[j-k]
            betas[i] = corr[i] + h[j]*h[j-k]
        else:
            alphas[i] = 0.0
            betas[i] = corr[i]
    
    # Find min of max_i (alpha_i * t + beta_i) over t in [0, 1]
    # All candidate points: t=0, t=1, and all pairwise intersections
    candidates = [0.0, 1.0]
    
    # Find intersections
    for i in range(2*n-1):
        for m in range(i+1, 2*n-1):
            da = alphas[i] - alphas[m]
            if abs(da) < 1e-15:
                continue
            t_cross = (betas[m] - betas[i]) / (alphas[i] - alphas[m])
            if -1e-9 < t_cross < 1.0 + 1e-9:
                candidates.append(np.clip(t_cross, 0.0, 1.0))
    
    # Evaluate at all candidates
    candidates = np.array(candidates)
    # Vectorized evaluation: for each candidate t, compute max of all lines
    vals = alphas[np.newaxis, :] * candidates[:, np.newaxis] + betas[np.newaxis, :]
    max_vals = np.max(vals, axis=1)
    
    best_idx = np.argmin(max_vals)
    best_t = candidates[best_idx]
    
    return best_t

    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        v, _ = smooth_obj_grad(x, betas[-1])
        return v

        return smooth_obj_grad(x, b)[0]

                # Update h[j] and corr incrementally
                delta = best_t - old_t
                h[j] = best_t
                for i in range(2*n-1):
                    k = i - (n-1)
                    d = 0.0
                    if 0 <= j-k < n:
                        d += (1.0 - h[j-k])  # NOTE: h[j-k] hasn't changed, only h[j] did
                    # Wait, I need to be careful. The change in corr[i] due to h[j] changing:
                    # corr[i] = sum_j' h[j']*(1-h[j'-k])
                    # h[j] appears as: h[j]*(1-h[j-k]) if valid as first index
                    #                 h[j-k]*(1-h[j]) if j-k valid as first index
                    # Change: delta*(1-h[j-k])*[valid1] + h[j-k]*(-delta)*[valid2]
                    #        = delta * [(1-h[j-k])*[valid1] - h[j-k]*[valid2]]
                    # But wait, "valid1" and "valid2" here are about j relative to the sum range
                    # valid1: j is in the sum range AND j+k is in [0,n-1]
                    # valid2: j-k is in the sum range (i.e., 0<=j-k<n AND 0<=(j-k)+k<n=0<=j<n, always true)
                    
                    # Actually from the alpha computation:
                    # alpha[i] = [valid1]*(1-h[j+k]) - [valid2]*h[j-k]
                    # where valid1 = (0 <= j+k < n) and valid2 = (0 <= j-k < n)
                    # But we also need j to be in the valid range of the sum!
                    
                    # The valid range for the sum at lag k is j' from max(0,k) to min(n-1, n-1+k)
                    # j is in this range iff max(0,k) <= j <= min(n-1, n-1+k)
                    # AND j+k must be in [0,n-1] for the first factor to exist
                    # These are actually the same condition: max(0,k) <= j <= min(n-1, n-1+k)
                    # iff 0<=j<n and 0<=j+k<n
                    # (since max(0,k) <= j means j>=0 and j>=k, and j<=min(n-1,n-1+k) means j<n and j<=n-1+k i.e. j-k<=n-1)
                    
                    # Hmm, I think I'm overcomplicating this. Let me just use the alpha formula.
                    # The change in corr[i] when h[j] changes by delta is: delta * alpha[i]
                    # (since corr[i](t) = alpha[i]*t + beta[i])
                    corr[i] += delta * alphas[i]
                
                improved = True
        
        if not improved:
            break
    
    return h

Wait, I realize the incremental update is simply `corr[i] += delta * alphas[i]` since corr[i](t) = alpha[i]*t + beta[i]. This is very clean!

But I need to be careful: the alphas depend on the current h (specifically h[j+k] and h[j-k]), which don't change when we update h[j]. So the alphas are fixed for this coordinate update. Good.

Now let me also think about whether the coordinate descent might get stuck. Since we're doing exact 1D minimization at each step, we'll reach a point where no single coordinate change helps. This is a "1-coord local optimum." To escape, we need:
1. SLSQP/gradient methods (which consider multiple coordinates simultaneously)
2. Random perturbations
3. Pair moves

I think the combination of coordinate descent + SLSQP should be quite powerful.

Let me also think about the "complement" trick. If h is feasible (sum = n/2, 0<=h<=1), then 1-h is also feasible. The C5 values might be different:
C5(h) = max_k sum_i h[i]*(1-h[i+k]) * dx
C5(1-h) = max_k sum_i (1-h[i])*h[i+k] * dx = max_k (sum_i h[i+k] - sum_i h[i]*h[i+k]) * dx

These are generally different. So we should try both h and 1-h as starting points.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = (m + np.log(s) / beta) * dx
    weights = exps / s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-10:
            continue
        m_lo = max(0, k)
        m_hi = min(n - 1, n - 1 + k)
        if m_lo <= m_hi:
            grad[m_lo:m_hi+1] += w * (1.0 - h[m_lo-k:m_hi+1-k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, n - 1 - k)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2+1] -= w * h[m_lo2+k:m_hi2+1+k]
    return val, grad * dx

def slsqp_refine(h, betas, max_iter=200, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if deadline is not None and time.time() > deadline:
            break
        def _obj(x):
            return smooth_obj_grad(x, beta)[0]
        def _grad(x):
            return smooth_obj_grad(x, beta)[1]
        def _eq_fun(x):
            return np.sum(x) - n / 2.0
        def _eq_jac(x):
            return np.ones(n)
        
        res = minimize(
            _obj, best_h, jac=_grad,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': max_iter, 'ftol': 1e-15}
        )
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    return best_h, best_c5

def coord_desc(h, n_sweeps=1, deadline=None):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = 2 * n - 1
    for _ in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old_t = h[j]
            alphas = np.zeros(n_corr)
            betas = np.zeros(n_corr)
            for i in range(n_corr):
                k = i - (n - 1)
                v1 = (0 <= j + k < n)
                v2 = (0 <= j - k < n)
                if v1 and v2:
                    alphas[i] = (1.0 - h[j + k]) - h[j - k]
                    betas[i] = corr[i] - h[j] * (1.0 - h[j + k]) + h[j] * h[j - k]
                elif v1:
                    alphas[i] = (1.0 - h[j + k])
                    betas[i] = corr[i] - h[j] * (1.0 - h[j + k])
                elif v2:
                    alphas[i] = -h[j - k]
                    betas[i] = corr[i] + h[j] * h[j - k]
                else:
                    betas[i] = corr[i]
            
            # Find min of max_i(alphas[i]*t + betas[i]) on [0,1]
            cands = [0.0, 1.0]
            for i in range(n_corr):
                for m in range(i + 1, n_corr):
                    da = alphas[i] - alphas[m]
                    if abs(da) < 1e-15:
                        continue
                    tc = (betas[m] - betas[i]) / da
                    if -0.1 < tc < 1.1:
                        cands.append(min(1.0, max(0.0, tc)))
            
            if len(cands) > 2:
                cands_arr = np.array(cands)
                vals = alphas[None, :] * cands_arr[:, None] + betas[None, :]
                max_vals = np.max(vals, axis=1)
                best_t = cands_arr[np.argmin(max_vals)]
            else:
                # Only 0 and 1
                v0 = np.max(betas)
                v1 = np.max(alphas + betas)
                best_t = 0.0 if v0 <= v1 else 1.0
            
            if abs(best_t - old_t) > 1e-12:
                delta = best_t - old_t
                h[j] = best_t
                corr += delta * alphas
                improved = True
        if not improved:
            break
    return h

def swap_local_search(h, n_iters, rng, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iters):

        h_new[i], h_new[j] = best_h[j], best_h[i]
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5
