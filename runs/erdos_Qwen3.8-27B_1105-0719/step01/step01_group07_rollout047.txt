    else:
        a = -s
        return dx * np.sum(h[a:n] * (1.0 - h[:n - a]))


def _C_grad(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += dx * (1.0 - h[s:n])
        grad[s:] -= dx * h[:n - s]
    else:
        a = -s
        grad[a:] += dx * (1.0 - h[:n - a])
        grad[:n - a] -= dx * h[a:n]
    return grad


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def binary_greedy(b, deadline, max_rounds=300):
    n = len(b)
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b, curr_c


def binary_sa(b0, deadline, rng, max_iters=80000, T_start=0.04, T_end=0.0005):
    """Simulated annealing with restarts for binary swap search."""
    n = len(b0)
    best_b = b0.copy()
    best_c = exact_objective(b0)

    total_budget = max(1, int(deadline - time.time() - 3))
    n_phases = 3
    iters_per_phase = max(1000, max_iters // n_phases)

    for phase in range(n_phases):
        if time.time() > deadline - 3:
            break
        b = best_b.copy() if phase > 0 else b0.copy()
        if phase > 0:
            # Perturb best solution for restart
            flips = rng.randint(1, max(2, n // 8) + 1)
            ones_idx = list(np.where(b == 1.0)[0])
            zeros_idx = list(np.where(b == 0.0)[0])
            for _ in range(flips):
                if ones_idx and zeros_idx:
                    i = ones_idx[rng.randint(len(ones_idx))]
                    j = zeros_idx[rng.randint(len(zeros_idx))]
                    b[i] = 0.0
                    b[j] = 1.0
                    ones_idx.remove(i)
                    zeros_idx.remove(j)
                    ones_idx.append(j)
                    zeros_idx.append(i)
            b = project_box_sum(b)
            # Re-binarize
            m = n // 2
            idx = np.argsort(b)[::-1][:m]
            b = np.zeros(n, dtype=np.float64)
            b[idx] = 1.0

        curr_c = exact_objective(b)
        ones = list(np.where(b == 1.0)[0])
        zeros = list(np.where(b == 0.0)[0])

        T_phase_start = T_start * (0.5 ** phase)
        T_phase_end = T_end * (0.5 ** phase)

        for it in range(iters_per_phase):
            if time.time() > deadline - 3:
                break
            progress = it / iters_per_phase
            T = T_phase_start * (T_phase_end / T_phase_start) ** progress

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
            new_c = exact_objective(b)
            delta = new_c - curr_c

            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
                curr_c = new_c
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                if curr_c < best_c - 1e-14:
                    best_c = curr_c
                    best_b = b.copy()
            else:
                b[i] = 1.0
                b[j] = 0.0

        # After SA phase, run greedy to polish
        b, curr_c = binary_greedy(best_b.copy(), deadline, max_rounds=100)
        if curr_c < best_c - 1e-14:
            best_c = curr_c
            best_b = b.copy()

    return best_b, best_c


def binary_greedy_sa(b0, deadline, rng):
    """Combined greedy + SA: greedy first, then SA to escape, then greedy again."""
    n = len(b0)
    b, c = binary_greedy(b0.copy(), deadline, max_rounds=200)
    if time.time() > deadline - 10:
        return b, c
    b2, c2 = binary_sa(b.copy(), deadline, rng, max_iters=60000)
    if c2 < c:
        b, c = b2, c2
    if time.time() < deadline - 5:
        b, c = binary_greedy(b.copy(), deadline, max_rounds=200)
    return b, c


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 6)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[3 * q:] = 1.0
    elif mode == 6:
        # Golden ratio placement
        h = np.zeros(n)
        golden = 0.618
        positions = set()
        for i in range(n // 2):
            pos = int((i * golden) % 1.0 * n)
            positions.add(pos)
        for p in positions:
            h[p] = 1.0
    elif mode == 7:
        # Fractal-like pattern
        h = np.zeros(n)
        level = 0
        while (1 << (level + 1)) <= n:
            level += 1
        for i in range(n):
            bits = bin(i)[2:].zfill(level)
            if sum(int(b) for b in bits) % 2 == 0:
                h[i] = 1.0
    elif mode == 8:
        # Clustered with gaps
        h = np.zeros(n)
        cluster = max(3, n // 6)
        gap = max(2, n // 12)
        pos = 0
        count = 0
        while count < n // 2 and pos < n:
            h[pos:pos + cluster] = 1.0
            count += cluster
            pos += cluster + gap
        h = h[:n]
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new)
