    dx = 2.0 / n

def project_h(v, n):
    """Fast projection onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    v = np.asarray(v, dtype=np.float64)
    lo = v.min() - 1.5
    hi = v.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_profile_at(h, m, n, dx):
    """Gradient of profile[m] with respect to h."""
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] += -dx * h[d:]
    else:
        dp = -d
        g[:n-dp] += dx * (1.0 - h[dp:])
        g[dp:] += -dx * h[:n-dp]
    return g

def subgrad_minimax(h, n, dx, top_k=5):
    """Compute a subgradient of max(profile)."""
    prof = compute_profile(h)
    c = np.max(prof)
    # Top-k active constraints
    k = min(top_k, 2*n-1)
    # Get indices of top-k profile values
    idx = np.argpartition(prof, -k)[-k:]
    # Weights: proportional to how close to max
    w = prof[idx] - c + 1e-4
    w = w / (w.sum() + 1e-15)
    g = np.zeros(n)
    for j, m in enumerate(idx):
        g += w[j] * grad_profile_at(h, m, n, dx)
    return g, c

    dx = 2.0 / n
    alpha = 0.005

        g, c = subgrad_minimax(h, n, dx, top_k)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()

        a = alpha

        return x[-1]

                        best_h, best_c, best_n = h_sol, c_sol, new_n

    # Stage 5: Random restarts at best dimension
    if time_left() > 120:
        for trial in range(15):
            if time_left() < 30:
                break
            eps = 0.01 + 0.04 * rng.rand()
            z = rng.randn(best_n)
            z -= z.mean()
            h_pert = project_h(best_h + eps * z, best_n)
            h_pg, c_pg = projected_gradient(h_pert, best_n, max_iters=2000, top_k=8, seed=seed + 100 + trial)
            if c_pg < best_c:
                best_h, best_c, best_n = h_pg, c_pg, best_n

    # Stage 6: Final SLSQP
    if time_left() > 80:
        h_sol = slsqp_solve(best_h, maxiter=600, ftol=1e-14)
        if h_sol is not None:
            c_sol = float(exact_score(h_sol))
            if c_sol < best_c:
                best_h, best_c, best_n = h_sol, c_sol, len(h_sol)

    # Stage 7: Final PG polish
    if time_left() > 40:
        h_pg, c_pg = projected_gradient(best_h, best_n, max_iters=3000, top_k=12, alpha0=0.005, seed=seed + 999)
        if c_pg < best_c:
            best_h, best_c = h_pg, c_pg

    # Final
    best_h = project_h(best_h, best_n)
    final_c = float(exact_score(best_h))

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
