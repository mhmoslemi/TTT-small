    #   Contribution as (1-h[n+m]) where n+m=i → n=i-m: -h[i-m] * (delta of (1-h[i])) = -h[i-m]*(+1) for valid n=i-m
    #     Valid: 0 <= i-m < n → i-n+1 <= m <= i
    #
    # When h[j] changes by delta_j = +1 (0→1):
    #   Contribution as h[n] (n=j): +1 * (1-h[j+m]) for valid m: -j <= m <= n-1-j
    #   Contribution as (1-h[n+m]) where n+m=j → n=j-m: -h[j-m] * (-1) = +h[j-m] for valid: j-n+1 <= m <= j
    
    # Total Δ(m):
    # delta_i contributes: -(1-h[i+m]) [if -i<=m<=n-1-i] + h[i-m] [if i-n+1<=m<=i]
    # delta_j contributes: +(1-h[j+m]) [if -j<=m<=n-1-j] - h[j-m] [if j-n+1<=m<=j]
    
    # For binary h, (1-h[x]) = 1-h[x] is just the complement.
    
    nm1 = n - 1
    # m ranges from -(n-1) to (n-1)
    # We only need m where at least one of the above conditions holds
    
    m_min = max(-i, i-n+1, -j, j-n+1)
    m_max = min(n-1-i, i, n-1-j, j)
    
    deltas = {}
    for m in range(m_min, m_max + 1):
        d = 0.0
        # i first: -i <= m <= n-1-i
        if -i <= m <= n-1-i:
            d -= (1.0 - h[i + m])
        # i second: i-n+1 <= m <= i
        if i - n + 1 <= m <= i:
            d += h[i - m]
        # j first: -j <= m <= n-1-j
        if -j <= m <= n-1 - j:
            d += (1.0 - h[j + m])
        # j second: j-n+1 <= m <= j
        if j - n + 1 <= m <= j:
            d -= h[j - m]
        if abs(d) > 1e-15:
            deltas[m] = d
    return deltas  # m -> delta

Actually, let me just use a different approach. Since n is small (140), I'll precompute the affected indices for each pair and use vectorized numpy operations.

Better idea: for the SA, use a "fast evaluation" approach:

For a swap (i, j), the affected correlate indices k range over a bounded set. I can compute the new values at those indices in O(1) each.

Let me define it more carefully using the correlate index k (0 to 2n-2):
correlate(h, 1-h, 'full')[k] = sum_{n valid} h[n] * (1-h[n + k - (n-1)])

Let m = k - (n-1). Then:
P(m) = sum_{n: 0<=n<n, 0<=n+m<n} h[n] * (1-h[n+m])

When we swap h[i]: 1→0 and h[j]: 0→1:
The new value P'(m) = P(m) + Δ_i(m) + Δ_j(m) where:
- Δ_i(m) = [i valid as first] * (-1)*(1-h[i+m]) + [i valid as second] * h[i-m]*(+1)
  (using OLD values of h[i+m] and h[i-m], before the swap)
  BUT if i+m == j, we need to use the new value h[j]=1, not old h[j]=0.
  Similarly if i-m == j.

This is getting complicated. Let me just handle it by computing P'(m) directly for affected m values:

P'(m) = P(m) - [contribution of old h[i]] + [contribution of new h[i]] + [contribution change at h[j]]

Actually, the cleanest approach for small n:

```python
def fast_swap_eval(prof, cur, i, j, n):
    """
    Given current profile prof (raw, unscaled), current h=cur, and swap i→j,
    compute the new max of the profile.
    Returns (new_max_raw, list_of_affected_k, new_values_at_affected_k)
    """

    # Affected k values (correlate indices):
    # i appears in P(m) for m where i is first (0<=i+m<n) or i is second (0<=i-m<n)
    # As first: m in [max(0,-i), min(nm1, nm1-i)] → k = nm1+m in [nm1+max(0,-i), nm1+min(nm1, nm1-i)]
    #   Since -i <= 0 always (i>=0): m in [0, nm1-i] → k in [nm1, nm1+nm1-i]
    #   Wait, m can be negative too. If m=-i, then i+m=0, valid.
    #   m range for i as first: -i <= m <= n-1-i, i.e., 0 <= i+m <= n-1
    #     m in [-i, n-1-i]
    #   k = nm1 + m in [nm1-i, nm1+n-1-i] = [nm1-i, 2*n-2-i]
    # 
    # i as second: i-m is the first index, so 0 <= i-m <= n-1 → m in [i-n+1, i]
    #   k in [nm1+i-n+1, nm1+i] = [i, nm1+i]
    #
    # Similarly for j.
    #
    # Union of affected k:
    # From i: [max(nm1-i, i), min(2*n-2-i, nm1+i)]
    # From j: [max(nm1-j, j), min(2*n-2-j, nm1+j)]
    
    k_lo_i = max(nm1 - i, i)
    k_hi_i = min(2*n - 2 - i, nm1 + i)
    k_lo_j = max(nm1 - j, j)
    k_hi_j = min(2*n - 2 - j, nm1 + j)
    
    k_lo = max(0, min(k_lo_i, k_lo_j))
    k_hi = min(2*n - 2, max(k_hi_i, k_hi_j))

    # = max(prof.max() [but excluding affected], new values at affected)

    # Unaffected max: max of prof outside [k_lo, k_hi]
    if k_lo > 0:
        unaff_left = prof[:k_lo].max()
    else:
        unaff_left = 0  # or -inf
    if k_hi < 2*n - 2:
        unaff_right = prof[k_hi+1:].max()
    else:
        unaff_right = 0
    
    unaffected_max = max(unaff_left, unaff_right) if (k_lo > 0 or k_hi < 2*n-2) else 0
    
    # Affected new values
    affected_max = 0
    for k in range(k_lo, k_hi + 1):
        m = k - nm1
        # Compute new P(m) = sum_{valid n} cur'[n] * (1 - cur'[n+m])
        # Where cur' = cur with i: 1→0, j: 0→1
        # New P(m) = P(m) + delta from i change + delta from j change
        # But careful: if the "other" index in a term is i or j, we need new values.
        
        # Old P(m) contribution involving i:
        old_i_first = (0 <= i + m < n) and (cur[i] == 1)  # i as first, old h[i]=1
        old_i_second = (0 <= i - m < n) and (cur[i] == 1)  # i as second, old 1-h[i]=0... 
        # Wait, I need to think about this differently.
        
        # Old term involving h[i]:
        # If i is first: term = h[i] * (1-h[i+m]), valid if 0<=i+m<n
        # If i is second (i-m is first): term = h[i-m] * (1-h[i]), valid if 0<=i-m<n
        # New term involving h'[i]=0:
        # If i is first: term = 0 * (1-h'[i+m]) = 0
        # If i is second: term = h'[i-m] * (1-0) = h'[i-m]

        # If j is first: term = h[j] * (1-h[j+m]) = 0 (old h[j]=0)

        if 0 <= i + m < n:

            old_other = cur[other] if other != j else 0.0  # if other is j, old h[j]=0
            # Wait, cur still has old values (we haven't modified it yet in this function)
            # Actually, let me compute the full new P(m) directly.
            pass
        
        # This is getting too complex. Let me just compute new P(m) directly.
        # P'(m) = sum_{valid n} h'[n] * (1-h'[n+m])
        # where h' = cur with h'[i]=0, h'[j]=1
        
        s = 0.0
        if m >= 0:
            n_start, n_end = 0, n - m - 1
        else:
            n_start, n_end = -m, n - 1

            second_idx = idx + m
            second = cur[second_idx]
            if second_idx == j:
                second = 1.0
            elif second_idx == i:
                second = 0.0
            s += first * (1.0 - second)
        
        affected_max = max(affected_max, s)
    
    new_max = max(unaffected_max, affected_max)
    return new_max

This inner loop for computing P'(m) is O(n) for each affected m, and there are O(n) affected m's, so O(n²) per swap. That's the same as recomputing the full profile. Not better.

Let me go back to the incremental update approach, being careful about the i+m==j case:

```python
def swap_delta_profile(prof, cur, i, j):
    """
    Compute new profile values at affected indices after swapping h[i] 1→0, h[j] 0→1.
    prof: current raw profile (length 2n-1)
    cur: current h (length n)
    Returns: (new_prof_at_affected, affected_indices)
    """
    n = len(cur)
    nm1 = n - 1
    
    # For each affected k, compute new P(m) where m = k - nm1
    # P'(m) = P(m) + change_i(m) + change_j(m)
    # where change_i(m) accounts for h[i] going 1→0
    # and change_j(m) accounts for h[j] going 0→1
    
    # change_i(m):
    #   If i is a valid "first" index (0 <= i+m < n):
    #     Old contribution: h[i] * (1 - h[i+m]) = 1 - h[i+m]
    #     New contribution: 0 * (1 - h'[i+m]) = 0
    #     BUT if i+m == j, then h'[i+m] = h'[j] = 1, so new = 0*(1-1) = 0. Same.
    #     Actually the new contribution is 0 regardless.
    #     Change = -(1 - h[i+m]) + [if i+m==j: correction]
    #     If i+m == j: old h[i+m] = old h[j] = 0, so old contribution = 1-0 = 1
    #     If i+m != j: old contribution = 1 - h[i+m]
    #     In both cases, new contribution = 0, so change = -(1 - h_old[i+m])
    #     where h_old[i+m] is the value BEFORE the swap.
    #     Since cur still has old values: change = -(1 - cur[i+m]) if i+m != j, else -(1-0) = -1
    #     Wait, cur[i+m] is the old value. If i+m == j, cur[j] = 0 (old), so -(1-0) = -1. Correct.
    #
    #   If i is a valid "second" index (0 <= i-m < n, i.e., i-m is a valid first index):
    #     Old contribution: h[i-m] * (1 - h[i]) = h[i-m] * 0 = 0
    #     New contribution: h'[i-m] * (1 - 0) = h'[i-m]
    #     Change = h'[i-m] - 0 = h'[i-m]
    #     If i-m == j: h'[j] = 1, so change = 1
    #     If i-m != j: h'[i-m] = cur[i-m] (unchanged), so change = cur[i-m]
    #
    # change_j(m):
    #   If j is a valid "first" index (0 <= j+m < n):
    #     Old contribution: h[j] * (1-h[j+m]) = 0
    #     New contribution: 1 * (1-h'[j+m]) = 1 - h'[j+m]
    #     If j+m == i: h'[i] = 0, so new = 1-0 = 1, change = 1
    #     If j+m != i: h'[j+m] = cur[j+m], new = 1-cur[j+m], change = 1-cur[j+m]
    #
    #   If j is a valid "second" index (0 <= j-m < n):
    #     Old contribution: h[j-m] * (1-h[j]) = h[j-m] *
