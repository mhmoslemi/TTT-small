def exact_objective(h):

    return float(np.max(profile)), profile

def project_box_sum(h, target):
    # Project h to {0≤h≤1, sum(h)=target}
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0: break
            step = min(diff / cnt, 1.0 - h[mask].max())
            h[mask] += step
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0: break
            step = max(diff / cnt, -h[mask].min())
            h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def solve_1d_minimax(a, b, n_bisect=50):
    """Solve min_{t in [0,1]} max_k (a[k]*t + b[k])"""
    # Bounds
    vals_0 = b
    vals_1 = a + b
    V_low = np.max(np.minimum(vals_0, vals_1))
    V_high = max(np.max(vals_0), np.max(vals_1))
    
    if V_high - V_low < 1e-14:
        return 0.5
    
    for _ in range(n_bisect):
        V_mid = 0.5 * (V_low + V_high)
        # Check feasibility: exists t in [0,1] with a_k*t + b_k <= V_mid for all k
        pos = a > 1e-14
        neg = a < -1e-14
        zero = ~(pos | neg)
        
        if np.any(zero):
            if np.max(b[zero]) > V_mid + 1e-14:
                V_low = V_mid
                continue
        
        t_lo = 0.0
        t_hi = 1.0
        if np.any(pos):
            t_hi = min(t_hi, np.min((V_mid - b[pos]) / a[pos]))
        if np.any(neg):
            t_lo = max(t_lo, np.max((V_mid - b[neg]) / a[neg]))
        
        if t_lo <= t_hi + 1e-12:
            V_high = V_mid
        else:
            V_low = V_mid

    neg = a < -1e-14

    if t_lo > t_hi:

    return max(0.0, min(1.0, 0.5 * (t_lo + t_hi)))

def compute_effect(i, h, dx):

    n = len(h)
    k_vals = np.arange(-(n-1), n)

    a = np.zeros(2*n-1)

    valid2 = (idx2 >= 0) & (idx2 < n)
    a[valid1] += (1.0 - h[idx1[valid1]]) * dx
    a[valid2] -= h[idx2[valid2]] * dx
    return a

    target = 0.5 * n
    h = project_box_sum(h_init.copy(), target)
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c = float(np.max(profile))
    best_h = h.copy()
    
    for pass_num in range(max_passes):
        changed = False
        for i in range(n):
            old_hi = h[i]
            a = compute_effect(i, h, dx)
            b = profile - old_hi * a
            t_opt = solve_1d_minimax(a, b, n_bisect)
            if abs(t_opt - old_hi) > 1e-10:
                delta = t_opt - old_hi
                h[i] = t_opt
                profile += delta * a
                changed = True
        
        # Project

        if not np.allclose(h, h_proj, atol=1e-12):
            h = h_proj
            profile = np.correlate(h, 1.0 - h, mode='full') * dx

            best_c = c_now
            best_h = h.copy()
        elif c_now >= best_c and pass_num > 5:

            if not changed:
                break
    
    return best_h, best_c
