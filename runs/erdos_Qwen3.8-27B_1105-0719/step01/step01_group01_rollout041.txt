def pair_descent(h, max_passes=2000, deadline=None):
    N = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    n_passes_done = 0
    
    for p in range(max_passes):
        n_passes_done = p + 1
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(N):
            for j2 in range(j + 1, N):
                s = h[j] + h[j2]
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-12:
                    continue
                t_cur = h[j]
                # Generate candidates
                mid = (t_lo + t_hi) * 0.5
                candidates = [t_lo, t_hi, t_cur, mid]
                # Add quarter points
                q1 = t_lo + 0.25 * (t_hi - t_lo)
                q3 = t_lo + 0.75 * (t_hi - t_lo)
                candidates.extend([q1, q3])
                
                best_t = t_cur
                best_val = best_c5
                for t in candidates:
                    if t < t_lo or t > t_hi:
                        continue
                    h[j] = t
                    h[j2] = s - t
                    val = compute_c5(h)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                h[j] = best_t
                h[j2] = s - best_t
                if best_val < best_c5 - 1e-14:
                    best_c5 = best_val
                    improved = True
        if not improved:
            break
    
    return h, best_c5, n_passes_done
