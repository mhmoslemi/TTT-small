```python
import numpy as np
from scipy.optimize import minimize, linprog
import time
import math

def exact_score(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    p = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(p))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_profile_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return profile, G

def sca_lp_solve(h0, radius):
    """Solve the SCA linear program."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    profile, G = build_profile_and_grad(h0)
    
    # Variables: z[0..n-1], tau
    # Minimize tau
    # s.t. G[m,:] @ z - tau <= G[m,:] @ h0 - profile[m] for all m
    #      sum(z) = n/2
    #      max(0, h0[i]-r) <= z[i] <= min(1, h0[i]+r)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize tau
    
    # Inequality constraints: A_ub @ x <= b_ub
    # G[m,:] @ z - tau <= G[m,:] @ h0 - profile[m]
    A_ub = np.zeros((M, n + 1))
    A_ub[:, 0:n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - profile
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lo = np.maximum(0.0, h0 - radius)
    hi = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, np.inf))  # tau >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None and np.isfinite(res.x).all():
        return res.x[0:n]
    return None

def line_search(h0, z_target, max_its=20):
    """Exact line search from h0 toward z_target."""
    n = len(h0)
    c0 = exact_score(h0)
    best_alpha = 0.0
    best_c = c0
    best_h = h0.copy()
    
    alphas = np.linspace(0.05, 1.0, max_its)
    for alpha in alphas:
        h_try = h0 + alpha * (z_target - h0)
        h_try = repair_feasible(h_try)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-12:
            best_c = c_try
            best_alpha = alpha
            best_h = h_try.copy()
    
    if best_alpha == 0.0:
        # Try finer search near 0
        for alpha in np.linspace(0.01, 0.05, 10):
            h_try = h0 + alpha * (z_target - h0)
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-12:
                best_c = c_try
                best_alpha = alpha
                best_h = h_try.copy()
    
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def sca_refine(h, max_iters=100, initial_radius=0.3, deadline=None, time_start=None):
    """SCA refinement loop."""
    n = len(h)
    h = h.copy()
    c = exact_score(h)
    radius = initial_radius
    stale = 0
    
    for it in range(max_iters):
        if deadline is not None and time_start is not None:
            if time.perf_counter() - time_start > deadline:
                break
        
        z = sca_lp_solve(h, radius)
        if z is None:
            radius *= 0.5
            stale += 1
            if stale > 5:
                break
            continue
        
        z = repair_feasible(z)
        
        # Line search
        h_new, c_new = line_search(h, z, max_its=15)
        
        if c_new < c - 1e-11:
            improvement = c - c_new
            h = h_new
            c = c_new
            stale = 0
            # Adapt radius
            if improvement > 1e-5:
                radius = min(radius * 1.5, 0.5)
            else:
                radius = min(radius * 1.1, 0.3)
        else:
            radius *= 0.6
            stale += 1
            if stale > 8:
                break
    
    return h, c

def binary_local_search(h, max_iters=2000, deadline=None, time_start=None):
    """Local search on binary solutions."""
    n = len(h)
    h = h.copy()
    c = exact_score(h)
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    p = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(p))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_profile_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return profile, G

def sca_lp_solve(h0, radius):
    n = len(h0)
    M = 2 * n - 1
    profile, G = build_profile_and_grad(h0)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, 0:n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - profile + 1e-12
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h0 - radius)
    hi = np.minimum(1.0, h0 + radius)
    # Ensure feasibility of bounds
    for i in range(n):
        if lo[i] > hi[i]:
            mid = (lo[i] + hi[i]) / 2.0
            lo[i] = hi[i] = mid
    bounds = list(zip(lo, hi))
    bounds.append((0.0, float('inf')))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None and np.isfinite(res.x).all():
            return res.x[0:n]
    except Exception:
        pass
    return None

    c0 = exact_score(h0)
    best_alpha = 0.0
    best_c = c0
    best_h = h0.copy()
    alphas = np.geomspace(0.01, 1.0, n_alphas)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lse_smooth_objective(h, lam):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = profile - profile.max()
    logsum = profile.max() + np.log(np.sum(np.exp(lam * shifted)))
    return (1.0 / lam) * logsum

def lse_gradient(h, lam):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    shifted = profile - profile.max()
    weights = np.exp(lam * shifted)
    weights /= weights.sum()
    grad = G.T @ weights
    return grad, profile

def lse_descent(h, lam_start, lam_end, max_iters, deadline, t_start):
    h = h.copy()
    c = exact_score(h)
    lam = lam_start
    n = len(h)
    alpha = 0.01
    for it in range(max_iters):
        if time.perf_counter() - t_start > deadline:
            break
        grad, profile = lse_gradient(h, lam)
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_normed = grad / gnorm
        for _ in range(20):
            h_try = repair_feasible(h - alpha * grad_normed)
            c_try = exact_score(h_try)
            if c_try < c - 1e-14:
                h = h_try
                c = c_try
                alpha = min(alpha * 1.3, 0.1)
                break
            else:
                alpha *= 0.6
        lam = lam * (lam_end / lam_start) ** (1.0 / max(max_iters, 1))
    return h, c

def sca_refine(h, max_iters, initial_radius, deadline, t_start):
    n = len(h)
    h = h.copy()
    c = exact_score(h)
    radius = initial_radius
    stale = 0
    for it in range(max_iters):
        if time.perf_counter() - t_start > deadline:
            break
        z = sca_lp_solve(h, radius)
        if z is None:
            radius *= 0.5
            stale += 1
            if stale > 6:
                break
            continue
        z = repair_feasible(z)
        h_new, c_new = line_search(h, z, n_alphas=15)
        if c_new < c - 1e-12:
            improvement = c - c_new
            h = h_new
            c = c_new
            stale = 0
            if improvement > 1e-5:
                radius = min(radius * 1.4, 0.5)
            else:
                radius = min(radius * 1.05, 0.3)
        else:
            radius *= 0.55
            stale += 1
            if stale > 10:
                break
    return h, c

def subgradient_descent(h, max_iters, deadline, t_start):
    n = len(h)
    h = h.copy()
    c = exact_score(h)
    alpha = 0.02
    for it in range(max_iters):
        if time.perf_counter() - t_start > deadline:
            break
        profile, G = build_profile_and_grad(h)
        c_val = np.max(profile)
        thresh = c_val - 1e-5
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        w = profile[active] - thresh
        w = w / (w.sum() + 1e-15)
        grad = G[active, :].T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            c_try = exact_score(h_try)
            if c_try < c - 1e-13:
                h = h_try
                c = c_try
                improved = True
                break
        if improved:
            alpha = min(alpha * 1.2, 0.1)
        else:
            alpha *= 0.6
            if alpha < 1e-6:
                break
    return h, c

def binary_sa(h_init, max_iters, T_start, T_end, rng, deadline, t_start):
    n = len(h_init)
    h = h_init.copy()
    # Round to binary
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = repair_feasible(h_bin)
    c = exact_score(h_bin)
    h = h_bin
    T = T_start
    cooling = (T_end / T_start) ** (1.0 / max(max_iters, 1))
    for it in range(max_iters):
        if time.perf_counter() - t_start > deadline:
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c_new = exact_score(h_new)
        delta = c_new - c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            c = c_new
        T *= cooling
    return h, c

def generate_wave_start(n, period, phase, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * period * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    h = repair_feasible(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(66, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrization
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-13:
        best_h = h_sym.copy()
        best_c = c_sym

    n_best = len(best_h)

    # Stage 1: SCA from best
    if time_left() > 100:
        h_r, c_r = sca_refine(best_h, max_iters=80, initial_radius=0.25,
                              deadline=budget_s * 0.25, t_start=t_start)
        if c_r < best_c - 1e-12:
            best_h, best_c = h_r, c_r

    # Stage 2: Subgradient refinement
    if time_left() > 80:
        h_r, c_r = subgradient_descent(best_h, max_iters=200,
                                       deadline=min(time_left() - 20, 60), t_start=t_start)
        if c_r < best_c - 1e-12:
            best_h, best_c = h_r, c_r

    # Stage 3: SCA from perturbed starts
    if time_left() > 120:
        for eps in [0.03, 0.07, 0.12, 0.05]:
            if time_left() < 50:
                break
            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_r, c_r = sca_refine(h_pert, max_iters=40, initial_radius=0.2,
                                  deadline=min(time_left() - 30, 40), t_start=t_start)
            if c_r < best_c - 1e-12:
                best_h, best_c, n_best = h_r, c_r, len(h_r)

    # Stage 4: LSE descent from best
    if time_left() > 80:
        h_r, c_r = lse_descent(best_h, lam_start=10.0, lam_end=100.0,
                               max_iters=150, deadline=min(time_left() - 20, 60),
                               t_start=t_start)
        if c_r < best_c - 1e-12:
            best_h, best_c = h_r, c_r
        # Follow up with SCA
        if time_left() > 40:
            h_r2, c_r2 = sca_refine(best_h, max_iters=40, initial_radius=0.1,
                                    deadline=min(time_left() - 10, 30), t_start=t_start)
            if c_r2 < best_c - 1e-12:
                best_h, best_c = h_r2, c_r2

    # Stage 5: Binary SA
    if time_left() > 60:
        h_bin, c_bin = binary_sa(best_h, max_iters=3000, T_start=0.05, T_end=0.001,
                                 rng=rng, deadline=min(time_left() - 10, 30),
                                 t_start=t_start)
        if c_bin < best_c - 1e-12:
            best_h, best_c, n_best = h_bin, c_bin, len(h_bin)
        # SCA from binary
        if time_left() > 40:
            h_r, c_r = sca_refine(best_h, max_iters=40, initial_radius=0.15,
                                  deadline=min(time_left() - 10, 30), t_start=t_start)
            if c_r < best_c - 1e-12:
                best_h, best_c = h_r, c_r

    # Stage 6: Wave starts + SCA
    if time_left() > 100:
        wave_starts = []
        for period in [2, 3, 4, 5, 6, 8]:
            for phase in [0.0, np.pi / 2, np.pi]:
                wave_starts.append(generate_wave_start(n_best, period, phase, rng))
        # Also random binary starts
        for _ in range(3):
            mask = rng.rand(n_best)
            h_rand = np.where(mask < 0.5, 0.0, 1.0)
            h_rand = repair_feasible(h_rand)
            wave_starts.append(h_rand)
        # Evaluate and pick top 5
        scored = [(exact_score(ws), ws) for ws in wave_starts]
        scored.sort(key=lambda x: x[0])
        for sc, h_s in scored[:5]:
            if time_left() < 40:
                break
            if sc >= best_c:
                continue
            h_r, c_r = sca_refine(h_s, max_iters=50, initial_radius=0.2,
                                  deadline=min(time_left() - 20, 30), t_start=t_start)
            if c_r < best_c - 1e-12:
                best_h, best_c = h_r, c_r

    # Stage 7: Dimension sweeping
    if time_left() > 80:
        n_curr = len(best_h)
        dims = sorted(set([n_curr - 6, n_curr - 3, n_curr + 3, n_curr + 6,
                           n_curr + 12, max(40, n_curr // 2), min(200, n_curr * 2)]))
        for new_n in dims:
            if time_left() < 35:
                break
            if new_n < 20 or new_n > 200 or new_n == n_curr:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_r, c_r = sca_refine(h_interp, max_iters=40, initial_radius=0.2,
                                  deadline=min(time_left() - 15, 25), t_start=t_start)
            if c_r < best_c - 1e-12:
                best_h, best_c, n_best = h_r, c_r, new_n

    # Stage 8: Final SCA polish
    if time_left() > 30:
        h_r, c_r = sca_refine(best_h, max_iters=80, initial_radius=0.08,
                              deadline=min(time_left() - 5, 25), t_start=t_start)
        if c_r < best_c - 1e-12:
            best_h, best_c = h_r, c_r

    # Stage 9: Very fine SCA
    if time_left() > 20:
        h_r, c_r = sca_refine(best_h, max_iters=60, initial_radius=0.02,
                              deadline=min(time_left() - 5, 15), t_start=t_start)
        if c_r < best_c - 1e-12:
            best_h, best_c = h_r, c_r

    # Stage 10: Final subgradient
    if time_left() > 15:
        h_r, c_r = subgradient_descent(best_h, max_iters=200,
                                       deadline=min(time_left() - 5, 12), t_start=t_start)
        if c_r < best_c - 1e-12:
            best_h, best_c = h_r, c_r

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
```
