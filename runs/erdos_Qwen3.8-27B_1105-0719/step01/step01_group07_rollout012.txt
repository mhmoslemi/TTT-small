  - h[i] in (1-h[i'+a]): i'+a=i, i'=i-a. Valid if 0 ≤ i-a ≤ n-1-a, i.e., a ≤ i ≤ n-1.
    1-h[i] goes 0→1. Term was h[i-a]*0=0, now h[i-a]*1. Change: +h[i-a]
  - h[j] in (1-h[j'+a]): j'+a=j, j'=j-a. Valid if 0 ≤ j-a ≤ n-1-a, i.e., a ≤ j ≤ n-1.
    1-h[j] goes 1→0. Term was h[j-a]*1=h[j-a], now h[j-a]*0=0. Change: -h[j-a]
  
  Total for s = -a < 0:
  ΔC_{-a} = -(1-h[i+a])·ᵢ₌₀ⁿ⁻¹⁻ᵃ + (1-h[j+a])·ⱼ₌₀ⁿ⁻¹⁻ᵃ + h[i-a]·ᵢ₌ₐⁿ⁻¹ - h[j-a]·ⱼ₌ₐⁿ⁻¹

This is O(1) per lag per swap! So for a given swap (i,j), I can compute the change in the max by checking all 2n-1 lags in O(n) time. This is much faster than recomputing the full profile each time.

import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad[i + s] += w * dx * (1.0 - h[i])
                grad[i] -= w * dx * h[i + s]
        else:
            a = -s
            for i in range(n - a):
                grad[i] += w * dx * (1.0 - h[i + a])
                grad[i + a] -= w * dx * h[i]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))


def slsqp_lse(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_box_sum(np.array(res.x, dtype=np.float64))
    except Exception:
        return h0.copy()


def refine_continuous(h0, deadline, max_taus=5):
    h = project_box_sum(h0.copy())
    curr_c = exact_objective(h)
    n = len(h)
    taus = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.00002]
    taus = taus[:max_taus]
    for tau in taus:
        if time.time() > deadline - 8:
            break
        h_new = slsqp_lse(h, tau, maxiter=min(300, n * 4 + 20))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-14:
            h = h_new
            curr_c = c_new
    return h, curr_c


class EpigraphSolver:
    def __init__(self, h0, margin=0.005):
        self.n = len(h0)
        self.margin = margin
        self.dx = 2.0 / self.n
        self.target = self.n / 2.0
        self.h0 = h0.copy()
        self.t0 = exact_objective(h0) + 0.001

    def unpack(self, x):
        return x[:-1], x[-1]

    def fun(self, x):
        return x[-1]

    def jac(self, x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def all_constraints(self, x):
        h = np.clip(x[:-1], 0.0, 1.0)
        t = x[-1]
        profile = get_profile(h)
        active = profile > t - self.margin
        cons = []
        n = self.n
        for idx in np.where(active)[0]:
            s = idx - (n - 1)
            prof_idx = idx
            fval = profile[idx] - t
            grads = _lag_gradient(h, s, self.n, self.dx)
            g = np.zeros(n + 1)
            g[:n] = grads
            g[n] = -1.0
            cons.append((fval, g))
        # equality constraint
        eq_f = np.sum(x[:-1]) - self.target
        eq_g = np.zeros(n + 1)
        eq_g[:n] = 1.0
        cons.append((eq_f, eq_g))
        return cons

    def ineq_fun(self, x):
        h = np.clip(x[:-1], 0.0, 1.0)
        t = x[-1]
        profile = get_profile(h)
        active = profile > t - self.margin
        return np.maximum(profile[active] - t, 0.0)

    def ineq_jac(self, x):
        n = self.n
        h = np.clip(x[:-1], 0.0, 1.0)
        t = x[-1]
        profile = get_profile(h)
        active_idx = np.where(profile > t - self.margin)[0]
        m = len(active_idx)
        J = np.zeros((m + 1, n + 1))
        for row, idx in enumerate(active_idx):
            s = idx - (n - 1)
            J[row, :n] = _lag_gradient(h, s, n, self.dx)
            J[row, n] = -1.0
        J[m, :n] = 1.0
        return J

    def solve(self, maxiter=150):
        n = self.n
        x0 = np.zeros(n + 1)
        x0[:n] = self.h0
        x0[n] = self.t0
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        res = minimize(self.fun, x0, method='SLSQP', jac=self.jac,
                       bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': self._ineq_fun2,
                                     'jac': self._ineq_jac2},
                                    {'type': 'eq', 'fun': self._eq_fun,
                                     'jac': self._eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            h_out = project_box_sum(np.clip(res.x[:n], 0, 1))
            return h_out
        return self.h0.copy()

    def _ineq_fun2(self, x):
        h = np.clip(x[:-1], 0.0, 1.0)
        t = x[-1]
        profile = get_profile(h)
        active = profile > t - self.margin
        return profile[active] - t

    def _ineq_jac2(self, x):
        n = self.n
        h = np.clip(x[:-1], 0.0, 1.0)
        t = x[-1]
        profile = get_profile(h)
        active_idx = np.where(profile > t - self.margin)[0]
        m = len(active_idx)
        if m == 0:
            return np.zeros((1, n + 1))
        J = np.zeros((m, n + 1))
        for row, idx in enumerate(active_idx):
            s = idx - (n - 1)
            J[row, :n] = _lag_gradient(h, s, n, self.dx)
            J[row, n] = -1.0
        return J

    def _eq_fun(self, x):
        return np.sum(x[:-1]) - self.target

    def _eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


def _lag_gradient(h, s, n, dx):
    grad = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            grad[i + s] += dx * (1.0 - h[i])
            grad[i] -= dx * h[i + s]
    else:
        a = -s
        for i in range(n - a):
            grad[i] += dx * (1.0 - h[i + a])
            grad[i + a] -= dx * h[i]
    return grad


def binary_threshold(h):

    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6, 7, 8])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        # Golden ratio-based distribution
        h = np.zeros(n)
        phi = (1 + np.sqrt(5)) / 2
        pos = np.floor((np.arange(n) / n + 0.5) * phi).astype(int) % n
        h[np.argsort(pos)[:n // 2]] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new)


def cyclic_shift(h, shift):
    return np.roll(h, shift)


def complement_transform(h):
    """Complement: 1-h also satisfies sum= n/2 and has same C5"""
    return 1.0 - h

            for c in cands:
                c_val = exact_objective(c)
                if c_val < best_c - 1e-13:
                    best_c = c_val
                    best_h = c.copy()
                # Quick swap polish on promising candidates
                if c_val < best_c + 0.01 and time.time() < deadline - 8:
                    b_sp, c_sp = binary_swap_fast(c, deadline, max_rounds=50)
                    if c_sp < best_c - 1e-13:
                        best_c = c_sp
                        best_h = b_sp.copy()

    # Phase 5: Differential evolution on continuous
    if time.time() < deadline - 30:
        n_de = parent_n
        if n_de % 2 != 0:
            n_de += 1
        pop_size = 12
        pop = []
        if parent_h is not None:
            pop.append(transfer_to_size(parent_h, n_de))
            pop.append(complement_transform(transfer_to_size(parent_h, n_de)))
        for mode in range(2, 6):
            pop.append(make_start(n_de, rng, mode))
        while len(pop) < pop_size:
            base = pop[rng.randint(len(pop))]
            pop.append(random_perturb(base, 0.04, rng))

        de_generations = 30
        for gen in range(de_generations):
            if time.time() > deadline - 15:
                break
            # Evaluate population
            pop_scores = [(exact_objective(p), p) for p in pop]
            pop_scores.sort(key=lambda x: x[0])
            for s, p in pop_scores:
                if s < best_c - 1e-13:
                    best_c = s
                    best_h = p.copy()

            # Create trial vectors
            new_pop = []
            for _ in range(pop_size):
                if time.time() > deadline - 12:
                    break
                a = pop_scores[rng.randint(len(pop_scores))][1]
                b_ = pop_scores[rng.randint(len(pop_scores))][1]
                c_ = pop_scores[rng.randint(len(pop_scores))][1]
                F = rng.uniform(0.5, 1.2)
                trial = a + F * (b_ - c_)
                trial = project_box_sum(trial)
                # LSE refine trial slightly
                trial = slsqp_lse(trial, 0.005, maxiter=50)
                new_pop.append(trial)

            # Selection
            if new_pop:
                all_pop = pop + new_pop
                all_scores = [(exact_objective(p), p) for p in all_pop]
                all_scores.sort(key=lambda x: x[0])
                pop = [p for _, p in all_scores[:pop_size]]

    # Phase 6: Final binary polish on best solution
    if best_h is not None and time.time() < deadline - 10:
        n_final = len(best_h)
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            b_final, c_final = binary_swap_fast(best_h.copy(), deadline, max_rounds=100)
            if c_final < best_c - 1e-13:
                best_c = c_final
                best_h = b_final.copy()
            if n_final <= 56 and time.time() < deadline - 8:
                b_final2, c_final2 = binary_2swap_search(b_final, deadline, max_rounds=30)
                if c_final2 < best_c - 1e-13:
                    best_c = c_final2
                    best_h = b_final2.copy()
        else:
            # Final LSE + threshold
            h_ref, c_ref = refine_continuous(best_h.copy(), deadline, max_taus=3)
            if c_ref < best_c - 1e-13:
                best_c = c_ref
                best_h = h_ref.copy()
            if n_final % 2 == 0:
                b = binary_threshold(best_h)
                if b is not None:
                    b, c_b = binary_swap_fast(b, deadline, max_rounds=80)
                    if c_b < best_c - 1e-13:
                        best_c = c_b
                        best_h = b.copy()

    # Phase 7: One more epigraph attempt on the very best
    if best_h is not None and time.time() < deadline - 8:
        try:
            ep = EpigraphSolver(best_h, margin=0.003)
            h_epi = ep.solve(maxiter=80)
            c_epi = exact_objective(h_epi)
            if c_epi < best_c - 1e-13:
                best_c = c_epi
                best_h = h_epi.copy()
            # Threshold and swap
            n_e = len(best_h)
            if n_e % 2 == 0 and time.time() < deadline - 4:
                b = binary_threshold(best_h)
                if b is not None:
                    b, c_b = binary_swap_fast(b, deadline, max_rounds=50)
                    if c_b < best_c - 1e-13:
                        best_c = c_b
                        best_h = b.copy()
        except Exception:
            pass

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)
    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
