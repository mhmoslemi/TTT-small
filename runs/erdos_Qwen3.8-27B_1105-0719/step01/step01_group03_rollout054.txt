import numpy as np
from scipy.optimize import minimize, minimize_n johansson
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Redistribute
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0, 1)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def softmax_grad(h, beta, lam):
    """Gradient of softmax objective + sum penalty."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.correlate(h, 1 - h, mode="full") * dx
    
    # Softmax weights
    max_p = np.max(profile)
    exp_p = np.exp(beta * (profile - max_p))
    w = exp_p / exp_p.sum()
    
    # Gradient: for each i, sum_d w_d * d(p_d)/d(h_i)
    # d(p_d)/d(h_i) = dx * [(1-h[i+d])*1{valid} - h[i-d]*1{valid}] for d≠0
    #                = dx * (1-2*h[i]) for d=0
    
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for d in range(-(n-1), n):
            wd = w[n-1+d]
            if wd < 1e-18:
                continue
            if d == 0:
                s += wd * dx * (1.0 - 2.0 * h[i])
            else:
                if 0 <= i + d < n:
                    s += wd * dx * (1.0 - h[i + d])
                if 0 <= i - d < n:
                    s += wd * dx * (-h[i - d])
        grad[i] = s + 2.0 * lam * (h.sum() - n / 2.0)
    
    # Zero out gradient where h is at bound (L-BFGS-B handles this, but helps)
    return grad

def softmax_obj(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    max_p = np.max(profile)
    exp_p = np.exp(beta * (profile - max_p))
    f = (max_p + np.log(exp_p.sum())) / beta
    f += lam * (h.sum() - n / 2.0) ** 2
    return f

def softmax_optimize(h_start, beta_list=None, lam_list=None, maxiter=200):
    """Optimize using L-BFGS-B with increasing beta and lambda."""
    if beta_list is None:
        beta_list = [10, 20, 50, 100, 200, 500, 1000]
    if lam_list is None:
        lam_list = [1.0, 10.0, 100.0]
    
    h = h_start.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    
    for beta in beta_list:
        for lam in lam_list:
            def obj(x):
                return softmax_obj(x, beta, lam)
            def grad(x):
                return softmax_grad(x, beta, lam)
            
            try:

                              bounds=[(0, 1)] * len(h),
                              options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
                if res.x is not None and np.isfinite(res.x).all():
                    h_new = repair_feasible(res.x)
                    c_new = exact_score(h_new)
                    if c_new < best_c:
                        best_h = h_new
                        best_c = c_new
                    h = h_new
            except Exception:
                h = repair_feasible(h)
    
    return best_h, best_c

def coordinate_descent_1d(h, i, n, dx):
    """Find optimal h[i] given all others fixed, using golden section search."""
    full_profile = np.correlate(h, 1 - h, mode="full") * dx
    old_t = h[i]
    M = 2 * n - 1
    
    # Compute slopes and intercepts for p_new[m] as function of t
    slopes = np.zeros(M)
    intercepts = np.zeros(M)
    quad_coef = np.zeros(M)
    
    for d in range(-(n-1), n):
        m = n - 1 + d
        if d == 0:
            quad_coef[m] = -dx
            slopes[m] = dx
            intercepts[m] = full_profile[m] - dx * old_t + dx * old_t * old_t
        else:
            A = 0.0
            if 0 <= i + d < n:
                A += -dx * (1.0 - h[i + d])
            if 0 <= i - d < n:
                A += dx * h[i - d]
            slopes[m] = A
            intercepts[m] = full_profile[m] - old_t * A
    
    def eval_t(t):
        vals = intercepts + slopes * t + quad_coef * t * t
        return np.max(vals)
    
    # Coarse grid
    t_grid = np.linspace(0, 1, 100)
    obj_vals = np.array([eval_t(t) for t in t_grid])
    best_idx = np.argmin(obj_vals)
    
    # Refine with golden section
    lo = t_grid[max(0, best_idx - 1)]
    hi = t_grid[min(99, best_idx + 1)]
    
    if hi - lo < 1e-12:
        return float(t_grid[best_idx])
    
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc, fd = eval_t(c), eval_t(d_pt)
    
    for _ in range(60):
        if fc < fd:
            b, d_pt, fd = d_pt, c, fc
            c = b - gr * (b - a)
            fc = eval_t(c)
        else:
            a, c, fc = c, d_pt, fd
            d_pt = a + gr * (b - a)
            fd = eval_t(d_pt)
        if b - a < 1e-12:
            break
    
    return float((a + b) / 2)

def cyclic_cd(h, max_sweeps=200, tol=1e-13, random_order=False, rng=None):
    """Cyclic coordinate descent."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = exact_score(h)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for i in order:
            t_opt = coordinate_descent_1d(h, i, n, dx)
            change = abs(t_opt - h[i])
            max_change = max(max_change, change)
            h[i] = t_opt
        h = repair_feasible(h)
        cur = exact_score(h)
        if cur < best:
            best = cur
        if max_change < tol:
            break
    
    return h, best

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def slsqp_epigraph(h_start, maxiter=300, tol=1e-10):
    """SLSQP epigraph optimization."""
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():

def local_descent(h, max_iters=100):
    """Active-set subgradient descent."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = exact_score(h)
    
    for _ in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best - 1e-12:
            best = c
        active_thresh = c - 1e-6
        active = np.where(profile >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        
        G = build_gradient_matrix(h, n, dx)
        G_active = G[active, :]
        w = profile[active] - active_thresh
        w = w / (w.sum() + 1e-15)
        grad = G_active.T @ w
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_try = h - alpha * grad
            h_try = repair_feasible(h_try)
            s = exact_score(h_try)
            if s < best - 1e-12:
                h = h_try
                best = s
                break
    
    return h, best

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n - d])
            G[m, 0:n - d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n - dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n - dp]
    return G


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

                if d == 0:
                    slope[m] = dx
                else:
                    s = 0.0
                    if 0 <= i + d < n:
                        s += -dx * (1.0 - h[i + d])
                    if 0 <= i - d < n:
                        s += dx * h[i - d]
                    slope[m] = s

            intercept = profile - old_t * slope
            intercept[m0] = profile[m0] - dx * old_t + dx * old_t * old_t
            quad = np.zeros(M)
            quad[m0] = -dx

            def _eval_t(t):
                vals = intercept + slope * t + quad * t * t
                return np.max(vals)

            t_grid = np.linspace(0.0, 1.0, 100)
            obj_vals = np.array([_eval_t(tt) for tt in t_grid])
            best_idx = int(np.argmin(obj_vals))
            lo = t_grid[max(0, best_idx - 2)]
            hi = t_grid[min(99, best_idx + 2)]

            if hi - lo > 1e-10:
                gr_const = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c = b - gr_const * (b - a)
                d_pt = a + gr_const * (b - a)
                fc = _eval_t(c)
                fd = _eval_t(d_pt)
                for _ in range(60):
                    if fc < fd:
                        b, d_pt, fd = d_pt, c, fc
                        c = b - gr_const * (b - a)
                        fc = _eval_t(c)
                    else:
                        a, c, fc = c, d_pt, fd
                        d_pt = a + gr_const * (b - a)
                        fd = _eval_t(d_pt)
                    if b - a < 1e-13:
                        break
                t_opt = (a + b) / 2.0
            else:
                t_opt = float(t_grid[best_idx])

            delta_t = t_opt - old_t
            max_change = max(max_change, abs(delta_t))
            h[i] = t_opt

            if abs(delta_t) > 1e-15:
                profile += delta_t * slope
                profile[m0] += -dx * (t_opt * t_opt - old_t * old_t)

        h = repair_feasible(h)
        if abs(h.sum() - n / 2.0) > 1e-10:
            profile = np.correlate(h, 1 - h, mode="full") * dx
        cur = float(np.max(profile))
        if cur < best:
            best = cur
        if max_change < tol:
            break

    return h, best


def softmax_obj(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    max_p = np.max(profile)
    exp_p = np.exp(beta * (profile - max_p))
    f = (max_p + np.log(exp_p.sum())) / beta
    f += lam * (h.sum() - n / 2.0) ** 2
    return f


def softmax_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.correlate(h, 1 - h, mode="full") * dx
    max_p = np.max(profile)
    exp_p = np.exp(beta * (profile - max_p))
    w = exp_p / exp_p.sum()

    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for d in range(-(n - 1), n):
            wd = w[n - 1 + d]
            if wd < 1e-18:
                continue
            if d == 0:
                s += wd * dx * (1.0 - 2.0 * h[i])
            else:
                if 0 <= i + d < n:
                    s += wd * dx * (1.0 - h[i + d])
                if 0 <= i - d < n:
                    s += wd * dx * (-h[i - d])
        grad[i] = s + 2.0 * lam * (h.sum() - n / 2.0)
    return grad


def softmax_optimize(h_start, beta_list=None, lam_list=None, maxiter=100):
    if beta_list is None:
        beta_list = [20, 50, 100, 200, 500, 1000]
    if lam_list is None:
        lam_list = [1.0, 10.0, 100.0]

    h = h_start.copy()
    best_h = h.copy()
    best_c = exact_score(h)

    for beta in beta_list:
        for lam in lam_list:
            def obj(x):
                return softmax_obj(x, beta, lam)

            def gr(x):
                return softmax_grad(x, beta, lam)

            try:
                res = minimize(obj, h, jac=gr, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * len(h),
                               options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
                if res.x is not None and np.isfinite(res.x).all():
                    h_new = repair_feasible(res.x)
                    c_new = exact_score(h_new)
                    if c_new < best_c:
                        best_h = h_new
                        best_c = c_new
                    h = h_new
            except Exception:
                h = repair_feasible(h)

    return best_h, best_c


def slsqp_epigraph(h_start, maxiter=300, tol=1e-11):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            return h_new
    except Exception:
        pass
    return None


def local_descent(h_start, max_iters=80):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    best = exact_score(h)

    for _ in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c < best - 1e-12:
            best = c
        active_thresh = c - 1e-6
        active = np.where(profile >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])

        G = build_gradient_matrix(h, n, dx)
        G_active = G[active, :]
        w = (profile[active] - active_thresh)
        w = w / (w.sum() + 1e-15)
        grad = G_active.T @ w
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm

        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_try = h - alpha * grad
            h_try = repair_feasible(h_try)
            s = exact_score(h_try)
            if s < best - 1e-13:
                h = h_try
                best = s
                improved = True
                break
        if not improved:
            break

    return h, best


def symmetrize(h):
    h_sym = (h + h[::-1]) / 2.0
    return repair_feasible(h_sym)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    time_budget = budget_s
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(66, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent

    def time_left():
        return time_budget + t_start - time.perf_counter()

    # Stage 0: Symmetrization
    h_sym = symmetrize(best_h)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Stage 1: Cyclic coordinate descent (main workhorse)
    if time_left() > 120:
        h_cd, c_cd = cyclic_cd_fast(best_h, max_sweeps=150, tol=1e-14)
        if c_cd < best_c - 1e-12:
            best_h = h_cd
            best_c = c_cd

    # Stage 2: Softmax L-BFGS-B
    if time_left() > 100:
        h_sm, c_sm = softmax_optimize(best_h,
                                      beta_list=[20, 50, 100, 300],
                                      lam_list=[1.0, 10.0],
                                      maxiter=100)
        if c_sm < best_c - 1e-12:
            best_h = h_sm
            best_c = c_sm

    # Stage 3: Coordinate descent again after softmax
    if time_left() > 80:
        h_cd, c_cd = cyclic_cd_fast(best_h, max_sweeps=100, tol=1e-14)
        if c_cd < best_c - 1e-12:
            best_h = h_cd
            best_c = c_cd

    # Stage 4: SLSQP epigraph
    if time_left() > 60:
        h_sol = slsqp_epigraph(best_h, maxiter=400, tol=1e-12)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h = h_sol
                best_c = c_sol

    # Stage 5: Random-restart coordinate descent
    if time_left() > 80:
        for trial in range(8):
            if time_left() < 30:
                break
            eps = 0.005 + 0.01 * trial
            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_cd, c_cd = cyclic_cd_fast(h_pert, max_sweeps=60, tol=1e-14,
                                        random_order=True, rng=rng)
            if c_cd < best_c - 1e-12:
                best_h = h_cd
                best_c = c_cd

    # Stage 6: Local descent from best
    if time_left() > 40:
        h_ld, c_ld = local_descent(best_h, max_iters=60)
        if c_ld < best_c - 1e-12:
            best_h = h_ld
            best_c = c_ld

    # Stage 7: Multi-resolution exploration
    if time_left() > 80:
        n_cur = len(best_h)
        candidates = []
        for delta in [-6, -4, -2, 2, 4, 6, 8, 12, 16]:
            nn = n_cur + delta
            if 20 <= nn <= 200:
                candidates.append(nn)
        candidates = sorted(set(candidates), key=lambda x: abs(x - n_cur))
        for new_n in candidates[:5]:
            if time_left() < 30:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_cd, c_cd = cyclic_cd_fast(h_interp, max_sweeps=60, tol=1e-14)
            if c_cd < best_c - 1e-12:
                h_sm, c_sm = softmax_optimize(h_cd,
                                              beta_list=[50, 200, 500],
                                              lam_list=[10.0, 100.0],
                                              maxiter=100)
                if c_sm < c_cd:
                    h_cd, c_cd = h_sm, c_sm
                if c_cd < best_c - 1e-12:
                    best_h = h_cd
                    best_c = c_cd

    # Stage 8: Final coordinate descent polish
    if time_left() > 30:
        h_cd, c_cd = cyclic_cd_fast(best_h, max_sweeps=200, tol=1e-15)
        if c_cd < best_c:
            best_h = h_cd
            best_c = c_cd

    # Stage 9: High-beta softmax for final squeeze
    if time_left() > 30:
        h_sm, c_sm = softmax_optimize(best_h,
                                      beta_list=[500, 1000, 2000],
                                      lam_list=[100.0, 1000.0],
                                      maxiter=200)
        if c_sm < best_c - 1e-13:
            best_h = h_sm
            best_c = c_sm

    # Stage 10: One more CD pass
    if time_left() > 20:
        h_cd, c_cd = cyclic_cd_fast(best_h, max_sweeps=100, tol=1e-15)
        if c_cd < best_c:
            best_h = h_cd
            best_c = c_cd

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
