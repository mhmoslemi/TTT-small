import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def get_feasible_interval(h, d):
    t_lo = 0.0
    t_hi = 0.0
    pos_mask = d > 1e-14
    if np.any(pos_mask):
        t_hi = min(t_hi, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
        t_lo = max(t_lo, np.min(-h[pos_mask] / d[pos_mask]))
    neg_mask = d < -1e-14
    if np.any(neg_mask):
        t_hi = min(t_hi, np.min(-h[neg_mask] / d[neg_mask]))
        t_lo = max(t_lo, np.min((1.0 - h[neg_mask]) / d[neg_mask]))
    return t_lo, t_hi

def line_search_1d(h, d, n_evals=40):
    t_lo, t_hi = get_feasible_interval(h, d)
    if t_hi - t_lo < 1e-10:
        return h, exact_score(h)
    best_h = h.copy()
    best_score = exact_score(h)
    t_grid = np.linspace(t_lo, t_hi, n_evals)
    for t in t_grid:
        if t == 0.0:
            continue
        h_trial = h + t * d
        if np.any(h_trial < -1e-9) or np.any(h_trial > 1.0 + 1e-9):
            continue
        h_trial = repair_feasible(h_trial)
        s = exact_score(h_trial)
        if s < best_score - 1e-12:
            best_score = s
            best_h = h_trial.copy()
    if best_score > exact_score(h) - 1e-12:
        return h, best_score
    # Golden section refinement
    for _ in range(20):
        t_lo2 = max(t_lo, t_grid[np.argmin([exact_score(repair_feasible(h + t*d)) for t in t_grid])])
        break
    return best_h, best_score

def make_zero_sum(d):
    d = d - d.mean()
    nd = np.linalg.norm(d)
    if nd < 1e-14:
        return None
    return d / nd

def subgradient_direction(h, profile, top_k=5):
    n = len(h)
    M = 2 * n - 1
    active = np.argsort(profile)[::-1][:top_k]
    w = np.zeros(M)
    max_p = profile.max()
    for k in active:
        w[k] = max(0, profile[k] - (max_p - 1e-4))
    if w.sum() < 1e-10:
        w[active[0]] = 1.0
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active:
        if w[k] < 1e-10:
            continue
        delta = k - (n - 1)
        if delta == 0:
            grad += w[k] * dx * (1.0 - 2.0 * h)
        elif delta > 0:
            grad[delta:] += w[k] * dx * (1.0 - h[:-delta] if delta < n else 0)
            grad[:-delta] -= w[k] * dx * h[delta:]
        else:
            sep = -delta
            grad[:-sep] += w[k] * dx * (1.0 - h[sep:])
            grad[sep:] -= w[k] * dx * h[:-sep]
    d = -grad
    d -= d.mean()
    nd = np.linalg.norm(d)
    if nd < 1e-14:
        return None
    return d / nd

def random_zero_sum(n, rng):
    d = rng.standard_normal(n)
    d -= d.mean()
    nd = np.linalg.norm(d)
    if nd < 1e-14:
        return None
    return d / nd

def block_direction(h, n_blocks, rng):
    n = len(h)
    block_size = n // n_blocks
    d = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        val = rng.standard_normal(1)[0]
        d[start:end] = val
    d -= d.mean()
    nd = np.linalg.norm(d)
    if nd < 1e-14:
        return None
    return d / nd

def pair_exchange_direction(h, i, j, rng):
    d = np.zeros(len(h))
    amount = rng.uniform(0.01, 0.05)
    if rng.random() < 0.5:
        d[i] = amount
        d[j] = -amount
    else:
        d[i] = -amount
        d[j] = amount
    return d

def freq_direction(n, freq, rng):
    x = np.arange(n)
    if rng.random() < 0.5:
        d = np.cos(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
    else:
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
    d -= d.mean()
    nd = np.linalg.norm(d)
    if nd < 1e-14:
        return None
    return d / nd

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_slsqp_reduced(h_start, top_nc, maxiter=60):
    n = len(h_start)
    profile = get_profile(h_start)
    t0 = profile.max()
    active_lags = np.argsort(profile)[::-1][:top_nc]

    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def cons_fun(x):
        h = x[:-1]
        t = x[-1]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:-1]
        n_h = len(h)
        dx = 2.0 / n_h
        M = 2 * n_h - 1
        J = np.zeros((1 + len(active_lags), n_h + 1))
        J[0, :n_h] = 1.0
        for ii, k in enumerate(active_lags):
            delta = k - (n_h - 1)
            if delta == 0:
                J[1+ii, :n_h] = -dx * (1.0 - 2.0 * h)
            elif delta > 0 and delta < n_h:
                J[1+ii, delta:n_h] -= dx * (1.0 - h[:n_h-delta])
                J[1+ii, :n_h-delta] += dx * h[delta:]
            else:
                sep = -delta
                if sep < n_h:
                    J[1+ii, :n_h-sep] -= dx * (1.0 - h[sep:])
                    J[1+ii, sep:] += dx * h[:n_h-sep]
            J[1+ii, n_h] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    x0 = np.concatenate([h_start, [t0 + 1e-6]])
    try:
        result = minimize(obj, x0, jac=obj_jac, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:-1])
    except Exception:
        pass
    return None

def make_binary_start(n, rng, pattern='alternating'):
    h = np.zeros(n)
    n_ones = n // 2
    if pattern == 'alternating':
        indices = list(range(0, n, 2))[:n_ones]
        if len(indices) < n_ones:
            indices = indices + list(range(1, n, 2))[:n_ones - len(indices)]
    elif pattern == 'block_half':
        indices = list(range(n_ones))
    elif pattern == 'three_band':
        third = n // 3
        indices = list(range(0, third)) + list(range(2 * third, 2 * third + (n_ones - third)))
        indices = indices[:n_ones]
    elif pattern == 'random':
        indices = list(rng.choice(n, size=n_ones, replace=False))
    else:
        indices = list(range(0, n, 2))[:n_ones]
    h[indices] = 1.0
    h_sum = h.sum()
    if h_sum < n_ones:
        zeros = [i for i in range(n) if h[i] == 0]
        for i in zeros[:n_ones - h_sum]:
            h[i] = 1.0
    elif h_sum > n_ones:
        ones = [i for i in range(n) if h[i] == 1]
        for i in ones[h_sum - n_ones:]:
            h[i] = 0.0
    return h

def smooth_binary(h, n_iters=50, alpha=0.3):
    h = h.astype(float).copy()
    for _ in range(n_iters):
        h_smooth = np.convolve(h, np.ones(5)/5.0, mode='same')
        h = repair_feasible(alpha * h + (1 - alpha) * h_smooth)
    return h
