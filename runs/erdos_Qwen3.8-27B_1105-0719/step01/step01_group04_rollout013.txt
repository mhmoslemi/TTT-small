def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:

    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv

    if abs(diff2) > 1e-10:

                break

import numpy as np
import time
import cvxpy as cp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-10:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h

def compute_lag_data(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    values[n-1] = dx * np.sum(h * (1.0 - h))
    grads[n-1, :] = dx * (1.0 - 2.0 * h)
    
    for delta in range(1, n):
        # c[n-1+delta] = sum_{j=0}^{n-1-delta} h[j+delta] * (1-h[j])
        values[n-1+delta] = dx * np.sum(h[delta:] * (1.0 - h[:-delta]))
        idx1 = np.arange(delta, n)
        grads[n-1+delta, idx1] += dx * (1.0 - h[idx1 - delta])
        idx2 = np.arange(0, n - delta)
        grads[n-1+delta, idx2] += -dx * h[idx2 + delta]
        
        # c[n-1-delta] = sum_{j=0}^{n-1-delta} h[j] * (1-h[j+delta])
        values[n-1-delta] = dx * np.sum(h[:-delta] * (1.0 - h[delta:]))
        idx1 = np.arange(0, n - delta)
        grads[n-1-delta, idx1] += dx * (1.0 - h[idx1 + delta])
        idx2 = np.arange(delta, n)
        grads[n-1-delta, idx2] += -dx * h[idx2 - delta]
    
    return values, grads

def trust_region_step(h_cur, values, grads, lam=0.1, top_k=None):
    """Solve trust-region QP and return proposed step."""
    n = len(h_cur)
    n_lags = len(values)
    
    # Select active constraints
    if top_k is not None and top_k < n_lags:
        top_idx = np.argsort(values)[::-1][:top_k]
    else:
        top_idx = np.arange(n_lags)
    
    active_vals = values[top_idx]
    active_grads = grads[top_idx, :]  # shape (k, n)
    
    # QP: min t + lam * ||h - h_cur||^2
    # s.t. active_vals[j] + active_grads[j] @ (h - h_cur) <= t  for all j
    #      sum(h) = n/2
    #      0 <= h <= 1
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    objective = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h_cur))
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
    ]
    
    # Linearized constraints
    # active_vals[j] + active_grads[j] @ (h_var - h_cur) <= t_var
    # => active_grads @ h_var <= t_var - active_vals + active_grads @ h_cur
    rhs = -active_vals + active_grads @ h_cur
    constraints.append(active_grads @ h_var - t_var <= rhs)
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None
    
    if h_var.value is None:
        return None
    
    h_proposed = np.asarray(h_var.value, dtype=np.float64)
    if not np.all(np.isfinite(h_proposed)):
        return None
    
    return h_proposed
