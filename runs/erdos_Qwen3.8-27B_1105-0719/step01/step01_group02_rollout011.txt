def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient(h, t):
    """Compute gradient of r(t) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if t >= 0:
        # r(t) = sum_{n=0}^{n-1-t} (1-h[n]) * h[n+t]
        # dr/dh[j] = (1-h[j-t]) * [j>=t] - h[j+t] * [j<=n-1-t]
        if t == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            # j >= t: contributes (1-h[j-t])
            grad[t:] += (1.0 - h[:n-t])
            # j <= n-1-t: contributes -h[j+t]
            grad[:n-t] -= h[t:]
    else:
        a = -t
        # r(-a) = sum_{m=0}^{n-1-a} (1-h[m+a]) * h[m]
        # dr/dh[j] = (1-h[j+a]) * [j<=n-1-a] - h[j-a] * [j>=a]
        grad[:n-a] += (1.0 - h[a:])
        grad[a:] -= h[:n-a]
    return grad

def project_box_sum(h, target=None):

    h = np.array(h, dtype=np.float64).copy()

            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]) - 1e-15)
            if step <= 0:
                step = diff / cnt
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15

                break

        else:

def active_set_direction(h, tolerance=0.005, max_active=None):

    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')  # raw, without dx

            order = np.argsort(-r[active_indices])
            active_indices = active_indices[order[:max_active]]
            active_mask = np.zeros(len(r), dtype=bool)

    #   For each active lag t (mapped to index in full correlate output):

        A_ub_rows.append(row2)

    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, total_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    c = np.zeros(total_vars)
    c[n] = -1.0
    
    try:

        return None, None
    
    d = res.x[:n]
    delta = res.x[n]
    
    if delta <= 1e-10:

def line_search_step(h, d, max_alpha=1.0):
    """Find best alpha in [0, max_alpha] such that h + alpha*d is feasible and minimizes objective."""
    n = len(h)
    c0 = exact_objective(h)
    
    # Find max feasible alpha
    if np.max(d) > 0:
        alpha_max_box = min((1.0 - h[d > 0]) / d[d > 0])
    else:
        alpha_max_box = max_alpha
    if np.min(d) < 0:
        alpha_min_box = min(-h[d < 0] / d[d < 0])
    else:
        alpha_min_box = max_alpha
    alpha_max = min(max_alpha, alpha_max_box, alpha_min_box)
    alpha_max = max(alpha_max, 0.0)
    
    if alpha_max <= 0:
        return h, c0
    
    # Try a few alphas
    best_alpha = 0.0
    best_c = c0
    
    for alpha in np.linspace(0.1, alpha_max, 10):
        h_new = h + alpha * d
        h_new = np.clip(h_new, 0.0, 1.0)
        # Re-project to maintain sum constraint
        h_new = project_box_sum(h_new, n/2.0)
        c_new = exact_objective(h_new)
        if c_new < best_c:
            best_c = c_new
            best_alpha = alpha
    
    if best_alpha == 0.0:
        return h, c0
    
    h_new = project_box_sum(h + best_alpha * d, n/2.0)
    return h_new, best_c

def binary_threshold(h):

        if s not in sizes and 20 <= s <= 300:
            sizes.append(s)

    for n in sizes:
        if time.time() > deadline - 50:
            break

        h_work = transfer(best_h, n)
        c_work = exact_objective(h_work)
        if c_work < best_c - 1e-13:
            best_c = c_work
            best_h = h_work.copy()
            best_n = n

        # Try a few diverse starts
        candidates = [h_work]
        if n % 2 == 0:
            alt = np.zeros(n)
            alt[::2] = 1.0
            candidates.append(project_box_sum(alt))
        for freq in [2, 3]:
            x = np.linspace(0, freq * np.pi, n, endpoint=False)
            candidates.append(project_box_sum(0.5 + 0.35 * np.sin(x)))
        
        for cand in candidates:
            cc = exact_objective(cand)
            if cc < c_work - 1e-13:
                c_work = cc
                h_work = cand.copy()

        # Phase 1: Smoothed SLSQP
        if time.time() < deadline - 40:
            h_work, c_work = smooth_minimize(h_work)
            if c_work < best_c - 1e-13:
                best_c = c_work
                best_h = h_work.copy()
                best_n = n

        # Phase 2: Active-set LP polish (main method)
        if time.time() < deadline - 35:
            h_work, c_work = active_set_polish(h_work, tol=0.005, max_iter=50)
            if c_work < best_c - 1e-13:
                best_c = c_work
                best_h = h_work.copy()
                best_n = n

        # Phase 3: Subgradient polish
        if time.time() < deadline - 30:
            h_work, c_work = subgradient_polish(h_work, max_iter=60, tol=0.004)
            if c_work < best_c - 1e-13:
                best_c = c_work
                best_h = h_work.copy()
                best_n = n

        # Phase 4: Second active-set pass with tighter tol
        if time.time() < deadline - 25:
            h_work, c_work = active_set_polish(h_work, tol=0.002, max_iter=40)
            if c_work < best_c - 1e-13:
                best_c = c_work
                best_h = h_work.copy()
                best_n = n

        # Phase 5: Binary threshold + swap
        if n % 2 == 0 and time.time() < deadline - 20:
            b = binary_threshold(h_work)
            if b is not None:
                b_c = exact_objective(b)
                if b_c < best_c - 1e-13:
                    best_c = b_c
                    best_h = b.copy()
                    best_n = n
                b2, b2_c = binary_swap_improve(b, max_rounds=80, rng_seed=seed + n)
                if b2_c < best_c - 1e-13:
                    best_c = b2_c
                    best_h = b2.copy()
                    best_n = n

        # Phase 6: Final active-set on best
        if time.time() < deadline - 20:
            h_work2, c_work2 = active_set_polish(best_h, tol=0.001, max_iter=30)
            if c_work2 < best_c - 1e-13:
                best_c = c_work2
                best_h = h_work2.copy()
                best_n = n

    # Final polish at best size
    if time.time() < deadline - 10:
        h_work, c_work = active_set_polish(best_h, tol=0.001, max_iter=30)
        if c_work < best_c - 1e-13:
            best_c = c_work
            best_h = h_work.copy()
            best_n = n

    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
