```python
    n = len(h)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_objective(h):
    return float(np.max(exact_profile(h)))

def compute_grad_matrix(h):
    """Compute gradient matrix G[j, p] = d(c[j])/d(h[p]) for all j, p.
    c[j] = dx * sum_valid_i h[i] * (1-h[i+k]) where k = j-(n-1)"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        j = k + (n-1)
        for i in range(max(0, -k), min(n, n-k)):
            q = i + k
            # ∂h[i]* (1-h[q]) / ∂h[p]
            # for p = i: contributes (1-h[q])
            # for p = q: contributes -h[i]
            G[j, i] += dx * (1.0 - h[q])
            if q != i:
                G[j, q] += -dx * h[i]
            else:
                # h[i]*(1-h[i]), derivative = 1-2h[i]
                # We already added (1-h[i]), need to also add -h[i], total = 1-2h[i] ✓
                pass  # Already handled: G[j,i] += dx*(1-h[i]) and G[j,i] += -dx*h[i] = dx*(1-2h[i])
    return G

def project(h):

    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.0
    # Bisection on lambda: find lam such that sum(clip(h+lam, 0, 1)) = target
    lo, hi = -1.0 - np.max(h), 1.0 - np.min(h) + 1.0
    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo+hi)/2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = result < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt > 0:
                result[mask] = np.minimum(1.0, result[mask] + residual/cnt)
        else:
            mask = result > 1e-14
            cnt = np.sum(mask)
            if cnt > 0:
                result[mask] = np.maximum(0.0, result[mask] + residual/cnt)
    return np.clip(result, 0.0, 1.0)

def active_subgrad_step(h, G, P, alpha, top_k=10):
    """Take a projected gradient step using top-k active profile entries."""
    C = np.max(P)
    n = len(h)

    idx = np.argsort(P)[::-1][:top_k]

    weights = np.exp(-slacks / 0.01)

    for i, j in enumerate(idx):
        g += weights[i] * G[j]
    # Projected step
    h_new = project(h - alpha * g)
    return h_new
```

**Alternative: Use the fact that the problem is a DCP (disciplined convex programming) problem in a local sense.**

At a fixed point h₀, the function f(h) = max_j [P_j(h₀) + ∇P_j(h₀)·(h-h₀)] is convex (max of affine functions). We can minimize this with box and sum constraints using CVXPY. This gives a candidate h_new. Then we check the exact objective at h_new.

This is essentially a cutting-plane / Kelley's cutting plane method. Let me implement this:

```python
import cvxpy as cp

def cutting_plane_step(h0, P0, G0, top_k=20):
    """Solve min t s.t. P_j(h0) + G_j·(h-h0) <= t, 0<=h<=1, sum(h)=n/2"""
    n = len(h0)
    target = n / 2.0
    # Select top-k constraints
    idx = np.argsort(P0)[::-1][:top_k]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, np.sum(h_var) == target]
    for j in idx:
        linearized = P0[j] + G0[j] @ (h_var - h0)
        constraints.append(linearized <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=5000)
        if h_var.value is not None:
            return project(np.asarray(h_var.value, dtype=np.float64))
    except:
        pass
    return h0.copy()
```

```python
    s = e.sum()
    return m + tau * np.log(s)
```

**Another idea: Use the symmetry of the problem**

Actually, correlate(a, b) ≠ correlate(b, a) in general. Let me check:
correlate(a, b)[k] = sum_n a[n+k] * b[n]
correlate(b, a)[k] = sum_n b[n+k] * a[n]

These are different unless a and b have special structure.

```python
def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        j = k + (n-1)
        i_lo = max(0, -k)
        i_hi = min(n, n-k)
        i = np.arange(i_lo, i_hi)
        q = i + k
        G[j, i] += dx * (1.0 - h[q])
        if k != 0:
            G[j, q] += -dx * h[i]
        else:
            # For k=0, q=i, so h[i]*(1-h[i]), derivative = 1-2h[i]
            # We added (1-h[i]), need to also add -h[i]
            # Total: 1-h[i] - h[i] = 1-2*h[i] ✓
            # But we already added (1-h[q]) = (1-h[i]) to G[j, i]
            # And we need to add -h[i] to G[j, i] as well
            # So we just need to add -h[i] to G[j, i]
            G[j, i] += -dx * h[i]
    return G
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_objective(h):
    return float(np.max(exact_profile(h)))

def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        j = k + (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        i = np.arange(i_lo, i_hi)
        q = i + k
        G[j, i] += dx * (1.0 - h[q])
        G[j, q] += -dx * h[i]
    return G

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - result.sum()
    if abs(res) > 1e-12:
        if res > 0:
            m = result < 1.0 - 1e-14
            if m.any():
                result[m] = np.minimum(1.0, result[m] + res / m.sum())
        else:
            m = result > 1e-14
            if m.any():
                result[m] = np.maximum(0.0, result[m] + res / m.sum())
    return np.clip(result, 0.0, 1.0)

def transfer(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return project(h_src.copy())
    xs = (np.arange(n_src) + 0.5) / n_src
    xt = (np.arange(n_tgt) + 0.5) / n_tgt
    return project(np.interp(xt, xs, h_src))

def lse_value(h, tau):
    P = exact_profile(h)
    m = float(np.max(P))
    shifted = np.minimum((P - m) / tau, 0.0)
    e = np.exp(shifted)
    return m + tau * np.log(float(np.sum(e)))

def lse_gradient(h, tau):
    P = exact_profile(h)
    m = float(np.max(P))
    shifted = np.minimum((P - m) / tau, 0.0)
    e = np.exp(shifted)
    w = e / float(np.sum(e))
    G = compute_grad_matrix(h)
    return G.T @ w

def slsqp_lse(h0, tau, maxiter=200):
    n = len(h0)
    h_cur = h0.copy()

    def obj(x):
        return lse_value(project(x), tau)

    def grad(x):
        return lse_gradient(project(x), tau)

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project(np.asarray(res.x, dtype=np.float64))
    except Exception:
        return h_cur.copy()

def cvx_bundle_step(h0, P0, G0, top_k, radius):
    try:
        import cvxpy as cp
    except ImportError:
        return h0.copy()
    n = len(h0)
    target = n / 2.0
    idx = np.argsort(P0)[::-1][:top_k]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    con = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target,
           cp.sum_squares(h_var - h0) <= radius]
    for j in idx:
        con.append(P0[j] + G0[j] @ (h_var - h0) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), con)
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-8)
        if h_var.value is not None and np.all(np.isfinite(h_var.value)):
            return project(np.asarray(h_var.value, dtype=np.float64))
    except Exception:
        pass
    return h0.copy()

def subgrad_polish(h0, max_iters=80, top_k=6):
    h = project(h0.copy())
    n = len(h)
    curr_C = exact_objective(h)
    for it in range(max_iters):
        P = exact_profile(h)
        C = float(np.max(P))
        idx = np.argsort(P)[::-1][:top_k]
        slacks = C - P[idx]
        w = np.exp(-slacks / 0.003)
        w /= w.sum()
        G = compute_grad_matrix(h)
        g = np.zeros(n)
        for wi, j in zip(w, idx):
            g += wi * G[j]
        improved = False
        for alpha in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            h_try = project(h - alpha * g)
            c_try = exact_objective(h_try)
            if c_try < C - 1e-13:
                h, curr_C = h_try, c_try
                improved = True
                break
        if not improved:
            break
    return h, curr_C

def optimize_multi(h0, deadline):
    n = len(h0)
    h = project(h0.copy())
    curr_C = exact_objective(h)
    best_h = h.copy()
    best_C = curr_C

    # Phase A: LSE + SLSQP with decreasing tau
    taus = [0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.0015, 0.0008, 0.0004]
    for tau in taus:
        if time.time() > deadline - 12:
            break
        h_new = slsqp_lse(h, tau, maxiter=min(200, n * 3 + 50))
        c_new = exact_objective(h_new)
        if c_new < curr_C - 1e-13:
            h = h_new
            curr_C = c_new
            if curr_C < best_C:
                best_C = curr_C
                best_h = h.copy()

    # Phase B: CVXPY bundle iterations
    for it in range(40):
        if time.time() > deadline - 10:
            break
        P = exact_profile(h)
        G = compute_grad_matrix(h)
        C = float(np.max(P))
        top_k = min(2 * n - 1, max(10, n // 3))
        for radius in [0.08, 0.04, 0.02]:
            h_new = cvx_bundle_step(h, P, G, top_k, radius)
            c_new = exact_objective(h_new)
            if c_new < C - 1e-13:
                h = h_new
                curr_C = c_new
                if curr_C < best_C:
                    best_C = curr_C
                    best_h = h.copy()
                break

    # Phase C: Subgradient polish
    if time.time() < deadline - 5:
        h_pol, c_pol = subgrad_polish(h, max_iters=100, top_k=max(4, n // 8))
        if c_pol < best_C:
            best_C = c_pol
            best_h = h_pol.copy()

    return best_h, best_C

def make_starts(h_parent, n, rng, count=8):
    starts = []
    h_tr = transfer(h_parent, n)
    starts.append(h_tr)
    for scale in [0.008, 0.02, 0.04, 0.07]:
        starts.append(project(h_tr + scale * rng.randn(n)))
    # Mirror complement
    h_mirror = 1.0 - h_tr[::-1]
    starts.append(project(h_mirror))
    # Frequency perturbation
    x = (np.arange(n) + 0.5) / n
    freq = rng.choice([2, 3, 4, 5, 6])
    phase = rng.uniform(0, 2 * np.pi)
    h_freq = h_tr + 0.03 * np.sin(2 * np.pi * freq * x + phase)
    starts.append(project(h_freq))
    starts = starts[:count]
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_C = float('inf')
    best_n = 0

    # Load parent
    try:
        p = np.array(globals()['initial_h_values'], dtype=np.float64)
        p = project(p)
        pc = exact_objective(p)
        best_C = pc
        best_h = p.copy()
        best_n = len(p)
    except Exception:
        best_n = 48
        x = (np.arange(best_n) + 0.5) / best_n
        best_h = project(0.5 + 0.35 * np.sin(2 * np.pi * 4 * x + 0.7))
        best_C = exact_objective(best_h)

    # Build size list: parent size and neighbors
    pn = best_n
    sizes = [pn]
    for s in [pn + 16, pn + 8, pn - 8, pn - 16, 64, 80, 96, 112]:
        if 16 <= s <= 128 and s not in sizes:
            sizes.append(s)
    sizes = list(dict.fromkeys(sizes))

    # Sort by expected value: try parent size first, then nearby
    sizes.sort(key=lambda s: abs(s - pn))

    for n in sizes:
        if time.time() > deadline - 35:
            break

        starts = make_starts(best_h, n, rng, count=6)
        scored = []
        for s in starts:
            cs = exact_objective(s)
            scored.append((cs, s))
        scored.sort(key=lambda x: x[0])

        n_top = min(3, len(scored))
        for i in range(n_top):
            if time.time() > deadline - 25:
                break
            h_opt, c_opt = optimize_multi(scored[i][1], deadline)
            if c_opt < best_C - 1e-12:
                best_C = c_opt
                best_h = h_opt.copy()
                best_n = n

    # Final polish on the best solution found
    if time.time() < deadline - 5 and best_h is not None:
        h_pol, c_pol = subgrad_polish(best_h, max_iters=200, top_k=max(4, len(best_h) // 8))
        if c_pol < best_C:
            best_C = c_pol
            best_h = h_pol.copy()

    # Ensure feasibility and final evaluation
    best_h = project(best_h)
    best_C = exact_objective(best_h)
    best_n = len(best_h)

    return best_h, best_C, best_n
```
