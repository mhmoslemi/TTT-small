- For numerical stability: F_tau(h) = tau * (mu + log(sum(exp((p_m - mu)/tau)))) where mu = max(p_m)
- Gradient: weighted sum of row gradients where w_m = exp((p_m - mu)/tau) / sum(exp((p_m - mu)/tau))

For the conditional gradient direction:
- min g^T v subject to 0 <= v_i <= 1, sum(v_i) = n/2
- Solution: sort g ascending, fill v = 1 for first floor(n/2) entries, v = fractional for next, v = 0 for rest
- Actually more carefully: sort by g value ascending. Set v_i = 1 for the smallest g values until sum reaches n/2, last one is fractional.

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: add to entries with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass: remove from entries with most value
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final precision fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    """Build the gradient matrix for all profile entries."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # p_d = dx * sum_{i=0}^{n-1-d} h_{i+d} * (1 - h_i)
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            s = -d
            # p_{-s} = dx * sum_{i=0}^{n-1-s} h_i * (1 - h_{i+s})
            G[m, 0:n-s] += dx * (1.0 - h[s:n])
            G[m, s:n] += -dx * h[0:n-s]
    return G

def weighted_gradient(h, weights, n, dx):
    """Compute weighted sum of profile row gradients."""
    M = 2 * n - 1
    grad = np.zeros(n)
    for m in range(M):
        w = weights[m]
        if abs(w) < 1e-15:
            continue
        d = m - (n - 1)
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:n] += w * dx * (1.0 - h[0:n-d])
            grad[0:n-d] -= w * dx * h[d:n]
        else:
            s = -d
            grad[0:n-s] += w * dx * (1.0 - h[s:n])
            grad[s:n] -= w * dx * h[0:n-s]
    return grad

def smooth_objective(h, tau):
    """Log-sum-exp smoothed objective."""
    profile = compute_profile(h)
    mu = np.max(profile)
    shifted = (profile - mu) / tau
    # Clip for numerical stability
    shifted = np.maximum(shifted, -100)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return tau * (mu / tau + log_sum)  # = mu + tau * log_sum

def smooth_gradient(h, tau):
    """Gradient of the smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    mu = np.max(profile)
    shifted = (profile - mu) / tau
    shifted = np.maximum(shifted, -100)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return weighted_gradient(h, weights, n, dx)

def fw_vertex(h, grad, n):
    """Solve min grad^T v s.t. 0<=v<=1, sum(v)=n/2."""
    target = n / 2.0
    # Sort by gradient ascending
    order = np.argsort(grad)
    v = np.zeros(n)
    remaining = target
    for i in range(n):
        idx = order[i]
        if remaining >= 1.0:
            v[idx] = 1.0
            remaining -= 1.0
        else:
            v[idx] = remaining
            remaining = 0.0
            break
    return v

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def fw_anneal(h_start, tau_start, tau_end, max_iters, deadline):
    """Frank-Wolfe with annealing."""
    h = h_start.copy()
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    tau = tau_start
    
    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        
        # Compute smooth gradient
        grad = smooth_gradient(h, tau)
        
        # Find FW vertex
        v = fw_vertex(h, grad, n)
        
        # Direction
        direction = v - h
        dir_norm = np.linalg.norm(direction)
        
        if dir_norm < 1e-10:
            break
        
        # Line search: try different alpha values
        best_alpha = 0.0
        best_smooth = smooth_objective(h, tau)
        
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05, 0.7, 0.4, 0.15]:
            h_try = h + alpha * direction
            h_try = repair_feasible(h_try)
            s_try = smooth_objective(h_try, tau)

                    best_alpha = alpha

            h = repair_feasible(h + best_alpha * direction)
            c = exact_score(h)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
        
        # Anneal tau
        if it < max_iters - 1:

            tau = tau_start * (tau_end / tau_start) ** progress

def smooth_slsqp(h_start, tau, maxiter=200, deadline=None):
    """Minimize smooth objective using SLSQP."""
    n = len(h_start)
    target = n / 2.0
    
    def obj_fn(x):
        return smooth_objective(x, tau)
    
    def obj_grad(x):
        return smooth_gradient(x, tau)
    
    def eq_fun(x):
        return x.sum() - target
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    x0 = h_start.copy()
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x)
            return h_new
    except Exception:
        pass
    return None

def active_descent(h, max_iters=100, deadline=None):
    """Active constraint gradient descent."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = exact_score(h)
    h = h.copy()
    
    for it in range(max_iters):
        if deadline is not None and time.perf_counter() > deadline:
            break
        
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        
        # Active set: constraints near the maximum
        thresh = c - max(1e-5, 0.01 * c)
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        
        # Weight active constraints by proximity to max
        w = profile[active] - thresh
        w = w / (w.sum() + 1e-15)
        
        grad = weighted_gradient(h, _expand_weights(active, w, len(profile)), n, dx)
        
        # Project gradient to feasible direction (zero sum)
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        
        # Backtracking line search
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = repair_feasible(h - alpha * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                best_h = h.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_h, best_c

def _expand_weights(active_idx, weights, M):

    full = np.zeros(M)
    full[active_idx] = weights
    return full

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_gradient(h, weights, n, dx):
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if abs(w) < 1e-15:
            continue
        d = m - (n - 1)
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:n] += w * dx * (1.0 - h[0:n-d])
            grad[0:n-d] -= w * dx * h[d:n]
        else:
            s = -d
            grad[0:n-s] += w * dx * (1.0 - h[s:n])
            grad[s:n] -= w * dx * h[0:n-s]
    return grad

def smooth_objective(h, tau):
    profile = compute_profile(h)
    mu = np.max(profile)
    shifted = np.clip((profile - mu) / tau, -80, 80)
    return mu + tau * np.log(np.sum(np.exp(shifted)))

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    mu = np.max(profile)
    shifted = np.clip((profile - mu) / tau, -80, 80)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    return weighted_gradient(h, weights, n, dx)

def fw_vertex(h, grad, n):
    target = n / 2.0
    order = np.argsort(grad)
    v = np.zeros(n)
    remaining = target
    for i in range(n):
        idx = order[i]
        if remaining >= 1.0 - 1e-12:
            v[idx] = 1.0
            remaining -= 1.0
        else:
            v[idx] = remaining
            remaining = 0.0
            break
    return v

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    h_new = np.interp(xt, full_xs, full_h)

def fw_anneal(h_start, tau_start, tau_end, n_iters, deadline):

    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    tau = tau_start

            break
        grad = smooth_gradient(h, tau)
        v = fw_vertex(h, grad, n)
        direction = v - h
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-10:
            break

        f_curr = smooth_objective(h, tau)
        best_alpha = 0.0
        best_f = f_curr
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05, 0.7, 0.4, 0.15, 0.02]:
            h_try = repair_feasible(h + alpha * direction)
            f_try = smooth_objective(h_try, tau)

def active_descent(h, max_iters=150, deadline=None):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = exact_score(h)
    h = h.copy()
    momentum = np.zeros(n)
    mu = 0.5
    stall = 0

    for it in range(max_iters):
        if deadline is not None and time.perf_counter() > deadline:
            break
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break

        thresh = c - max(5e-5, 0.02 * c)
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])

        w = (profile[active] - thresh)
        w = w / (w.sum() + 1e-15)

        full_w = np.zeros(len(profile))
        full_w[active] = w
        grad = weighted_gradient(h, full_w, n, dx)
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm

        momentum = mu * momentum + (1 - mu) * grad
        direction = momentum
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-14:
            break
        direction /= dnorm

        improved = False
        for alpha in [0.04, 0.025, 0.015, 0.01, 0.006, 0.003, 0.001, 0.0005]:
            h_try = repair_feasible(h - alpha * direction)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                best_h = h.copy()
                improved = True
                break

        if not improved:
            if stall > 15:
                break
            mu = max(0.3, mu * 0.95)

    return best_h, best_c

def epigraph_solve(h_start, maxiter=400, ftol=1e-12):

    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    target = n / 2.0
    dx = 2.0 / n
    M = 2 * n - 1

        return x[-1]

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - target

        g[:-1] = 1.0
        return g

    def ineq_fun(x):

        t = x[-1]
        profile = compute_profile(h)
        return t - profile

        G = np.zeros((M, n + 1))
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                G[m, :n] = -dx * (1.0 - 2.0 * h)
            elif d > 0:
                G[m, d:n] -= dx * (1.0 - h[0:n-d])
                G[m, 0:n-d] += dx * h[d:n]
            else:
                s = -d
                G[m, 0:n-s] -= dx * (1.0 - h[s:n])
                G[m, s:n] += dx * h[0:n-s]
            G[m, n] = 1.0
        return G

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}

    try:

                       bounds=bounds, constraints=cons,

        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            return h_new
    except Exception:
        pass
    return None

        h_pert = repair_feasible(best_h + eps * z)
        starts.append(h_pert)

    z = rng.randn(len(parent))
    h_sp = repair_feasible(symmetrize(best_h + 0.02 * z))
    starts.append(h_sp)

    # Stage 1: Smooth SLSQP with tau annealing from each start
    tau_schedule = [0.2, 0.1, 0.05, 0.025, 0.012, 0.006]

    for s_idx, h0 in enumerate(starts):
        if time_left() < 40:
            break
        h_curr = h0.copy()
        c_curr = exact_score(h_curr)

        for tau in tau_schedule:
            if time_left() < 25:
                break
            h_sol = smooth_slsqp(h_curr, tau, maxiter=200, ftol=1e-12)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < c_curr - 1e-13:
                    h_curr = h_sol
                    c_curr = c_sol
                else:
                    h_curr = h_sol

        c_final = exact_score(h_curr)
        if c_final < best_c - 1e-13:
            best_h = h_curr.copy()
            best_c = c_final

    # Stage 2: FW annealing from best
    if time_left() > 80:
        for tau_s in [0.25, 0.15, 0.1]:
            if time_left() < 40:
                break
            h_fw, c_fw = fw_anneal(best_h, tau_s, 0.003, 100, deadline)
            if c_fw < best_c - 1e-13:
                best_h = h_fw
                best_c = c_fw

    # Stage 3: Active constraint descent
    if time_left() > 50:
        h_ad, c_ad = active_descent(best_h, max_iters=200, deadline=deadline)
        if c_ad < best_c - 1e-13:
            best_h = h_ad
            best_c = c_ad

    # Stage 4: More smooth SLSQP from refined best
    if time_left() > 80:
        for tau in [0.015, 0.008, 0.004, 0.002, 0.001]:
            if time_left() < 30:
                break
            h_sol = smooth_slsqp(best_h, tau, maxiter=250, ftol=1e-13)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-14:
                    best_h = h_sol
                    best_c = c_sol

    # Stage 5: Epigraph polish
    if time_left() > 50:
        h_epi = epigraph_solve(best_h, maxiter=500, ftol=1e-12)
        if h_epi is not None:
            c_epi = exact_score(h_epi)
            if c_epi < best_c - 1e-14:
                best_h = h_epi
                best_c = c_epi

    # Stage 6: Active descent again
    if time_left() > 40:
        h_ad, c_ad = active_descent(best_h, max_iters=150, deadline=deadline)
        if c_ad < best_c - 1e-14:
            best_h = h_ad
            best_c = c_ad

    # Stage 7: Dimension scaling
    if time_left() > 60:
        n_curr = len(best_h)
        dim_candidates = []
        for delta in [-6, -4, -2, 2, 4, 6, 8, 12]:
            d = n_curr + delta
            if 30 <= d <= 180:
                dim_candidates.append(d)
        if n_curr // 2 >= 30:
            dim_candidates.append(n_curr // 2)
        dim_candidates = sorted(set(dim_candidates))

        for new_n in dim_candidates[:5]:
            if time_left() < 30:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_sol = smooth_slsqp(h_interp, 0.03, maxiter=150, ftol=1e-12)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-14:
                    best_h = h_sol
                    best_c = c_sol
            else:
                c_interp = exact_score(h_interp)
                if c_interp < best_c - 1e-14:
                    best_h = h_interp
                    best_c = c_interp

    # Stage 8: Final epigraph polish
    if time_left() > 40:
        h_epi = epigraph_solve(best_h, maxiter=600, ftol=1e-13)
        if h_epi is not None:
            c_epi = exact_score(h_epi)
            if c_epi < best_c - 1e-14:
                best_h = h_epi
                best_c = c_epi

    # Stage 9: Final active descent
    if time_left() > 20:
        h_ad, c_ad = active_descent(best_h, max_iters=100, deadline=deadline)
        if c_ad < best_c - 1e-14:
            best_h = h_ad
            best_c = c_ad

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
