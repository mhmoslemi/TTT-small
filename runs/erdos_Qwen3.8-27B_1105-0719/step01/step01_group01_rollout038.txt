    s = np.sum(exps)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    return val * (2.0 / len(h))

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    n1 = n - 1
    for i in range(2 * n - 1):
        k = i - n1
        w = weights[i]
        if w < 1e-12:
            continue
        # First term: (1-h[shifted]) for valid indices
        lo1 = max(0, k)
        hi1 = min(n1, n1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1-k:hi1+1-k])
        # Second term: -h[shifted] for valid indices
        lo2 = max(0, -k)
        hi2 = min(n1, n1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2+k:hi2+1+k]
    return grad * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))

        else:

        if len(eligible) > 0:

def slsqp_optimize(h0, betas, max_iter=200, bounds_list=None):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    if bounds_list is None:
        bounds_list = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0,
             'jac': lambda h: np.ones(n)}]
    for beta in betas:
        res = minimize(
            smooth_obj, best_h, args=(beta,),
            jac=smooth_grad,
            method='SLSQP',
            bounds=bounds_list,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-14}

        if res.x is not None:
            c5_new = compute_c5(res.x)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = res.x.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    return val * (2.0 / len(h))


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    n1 = n - 1
    for i in range(2 * n - 1):
        k = i - n1
        w = weights[i]
        if w < 1e-12:
            continue
        lo1 = max(0, k)
        hi1 = min(n1, n1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * (1.0 - h[lo1 - k:hi1 + 1 - k])
        lo2 = max(0, -k)
        hi2 = min(n1, n1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * h[lo2 + k:hi2 + 1 + k]
    return grad * dx


def eq_con_fun(h):
    return np.sum(h) - len(h) / 2.0


def eq_con_jac(h):
    return np.ones(len(h))


def project_feasible(y, n):
    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h0, betas, max_iter=250):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    for beta in betas:
        res = minimize(
            smooth_obj, best_h, args=(beta,),
            jac=smooth_grad,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        if res.x is not None:
            c5_new = compute_c5(res.x)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = project_feasible(res.x.copy(), n)
    return best_h, best_c5


def swap_local_search(h, n, n_iterations, rng, temperature=0.0):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iterations):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        best_h[i], best_h[j] = best_h[j], best_h[i]
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
        elif temperature > 0 and c5_new > best_c5:
            if rng.random() < np.exp(-(c5_new - best_c5) / temperature):
                best_c5 = c5_new
            else:
                best_h[i], best_h[j] = best_h[j], best_h[i]
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
    return best_h, best_c5


def binary_swap_local_search(h, n, n_iterations, rng, temperature=0.0):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    ones = np.where(best_h >= 0.5)[0].tolist()
    zeros = np.where(best_h < 0.5)[0].tolist()
    for _ in range(n_iterations):
        if not ones or not zeros:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        best_h[i], best_h[j] = best_h[j], best_h[i]
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        elif temperature > 0:
            delta = c5_new - best_c5
            if delta <= 0 or rng.random() < np.exp(-delta / temperature):
                best_c5 = c5_new
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
            else:
                best_h[i], best_h[j] = best_h[j], best_h[i]
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
    return best_h, best_c5


def reversal_search(h, n, n_iterations, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iterations):
        i = rng.randint(n)
        j = rng.randint(n)
        lo = min(i, j)
        hi = max(i, j)
        if hi - lo < 1:
            continue
        seg = best_h[lo:hi + 1].copy()
        best_h[lo:hi + 1] = seg[::-1]
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
        else:
            best_h[lo:hi + 1] = seg
    return best_h, best_c5


def targeted_transfer_search(h, n, n_iterations, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iterations):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if best_h[i] <= best_h[j]:
            continue
        delta = min(best_h[i] - best_h[j], 0.05 + 0.3 * rng.random())
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5


def make_alternating(n):
    h = np.zeros(n)
    h[::2] = 1.0
    h[1::2] = 0.0
    if h.sum() != n / 2.0:
        h[n // 2] = 1.0 if h.sum() < n / 2.0 else 0.0
    return h


def make_block(n, frac):
    h = np.zeros(n)
    h[:int(n * frac)] = 1.0
    return project_feasible(h, n)
