import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    return h


def compute_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    values = np.zeros(L)
    for delta in range(n):
        if delta == 0:
            values[n - 1] = dx * np.sum(h * (1.0 - h))
        else:
            m = n - delta
            values[n - 1 + delta] = dx * np.sum(h[:m] * (1.0 - h[delta:delta + m]))
            values[n - 1 - delta] = dx * np.sum(h[delta:delta + m] * (1.0 - h[:m]))
    return values


def compute_lag_grads(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grads = np.zeros((L, n))
    for delta in range(n):
        if delta == 0:
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            m = n - delta
            idx1 = np.arange(m)
            idx2 = np.arange(delta, delta + m)
            grads[n - 1 + delta, idx1] += dx * (1.0 - h[idx2])
            grads[n - 1 + delta, idx2] += -dx * h[idx1]
            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx1])
            grads[n - 1 - delta, idx1] += -dx * h[idx2]
    return grads


def project_feasible(y, n):
    h = np.clip(y, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    return h


def active_descent_v2(h, max_iters=300, time_limit=120.0):
    t0 = time.perf_counter()
    n = len(h)
    h = repair(h.copy())
    best = h.copy()
    best_score = exact_score(best)
    values = compute_lag_values(best)
    current_alpha = 0.005

    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        if time.perf_counter() - t0 > time_limit - 20:
            break

        f = np.max(values)
        tol = max(0.001 * f, 1e-7)
        active_mask = values >= f - tol
        n_active = np.sum(active_mask)
        if n_active < 2:
            active_idx = np.argsort(values)[::-1][:8]
        elif n_active > 60:
            active_idx = np.argsort(values)[::-1][:60]
        else:
            active_idx = np.where(active_mask)[0]

        grads = compute_lag_grads(best)
        ag = grads[active_idx, :]
        av = values[active_idx]
        gaps = f - av
        temp = 0.002 * f + 1e-9
        w = np.exp(-gaps / temp)
        w /= w.sum() + 1e-15

        direction = -ag.T @ w
        direction -= np.mean(direction)
        norm_d = np.linalg.norm(direction)
        if norm_d < 1e-14:
            break

        improved = False
        for trial in range(12):
            alpha = current_alpha * (0.6 ** trial)
            if alpha < 1e-7:
                break
            y = best + alpha * direction
            h_trial = project_feasible(y, n)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial
                best_score = score
                values = compute_lag_values(best)
                current_alpha = alpha * 1.5
                improved = True
                break
        if not improved:
            current_alpha *= 0.5
            if current_alpha < 1e-8:
                break

    return best, best_score


def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    h_start = repair(h_start)
    c_start = exact_score(h_start)

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        return t - compute_lag_values(h)

    def ineq_jac(x):
        h = x[:n]
        g = np.zeros((2 * n - 1, n + 1))
        g[:, n] = 1.0
        g[:, :n] = -compute_lag_grads(h)
        return g

    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    x0 = np.concatenate([h_start, [c_start + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c_start + 1.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_sol = repair(res.x[:n])
            s = exact_score(h_sol)
            if np.isfinite(s):
                return h_sol, s
    except Exception:
        pass
    return None, None


def slsqp_activeset(h_start, active_idx, maxiter=150):
    n = len(h_start)
    h_start = repair(h_start)
    c_start = exact_score(h_start)
    L_active = len(active_idx)

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        vals = compute_lag_values(h)
        return t - vals[active_idx]

    def ineq_jac(x):

        g = np.zeros((L_active, n + 1))
        g[:, n] = 1.0
        g[:, :n] = -grads[active_idx, :]
        return g

    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, c_start + 1.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}

    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

def cvx_trust_region(h, active_idx, lam=0.1):
    if not HAS_CP:
        return None
    n = len(h)
    grads = compute_lag_grads(h)
    values = compute_lag_values(h)
    g_act = grads[active_idx, :]
    q_act = values[active_idx]
    L = len(active_idx)

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = []
    for r in range(L):
        cons.append(q_act[r] + g_act[r] @ (h_var - h) <= t_var)
    cons.append(cp.sum(h_var) == n / 2.0)
    cons.append(h_var >= 0)
    cons.append(h_var <= 1)
    obj = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h))
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.asarray(h_var.value, dtype=np.float64)


def interpolate_dim(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_n) + 0.5) * 2.0 / new_n
    xs = np.concatenate([[0.0], x_src, [2.0]])
    ys = np.concatenate([[h[0]], h, [h[-1]]])
    return np.interp(x_tgt, xs, ys)


def block_perturbation(h, rng, block_size=5, magnitude=0.01):
    n = len(h)
    h_new = h.copy()
    n_blocks = n // block_size
    for _ in range(min(n_blocks, 8)):
        start = rng.randint(0, max(1, n - block_size))
        end = min(start + block_size, n)
        delta = rng.uniform(-magnitude, magnitude)
        h_new[start:end] += delta
    return repair(h_new)
