def exact_objective(h):

    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h):

    target = n / 2.0
    h[~np.isfinite(h)] = 0.0

def make_binary_start(n, rng, mode):

    if mode == 0:
        h = np.zeros(n)
        h[:m] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:

        h[::2] = 1.0
        if n % 4 == 0:
            h[1::4] = 0.0
            h[3::4] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 2:
        x = np.arange(n) / n
        freq = rng.choice([3, 5, 7, 11])
        phase = rng.uniform(0, 2 * np.pi)
        vals = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)

        h = np.zeros(n)
        h[idx] = 1.0
    elif mode == 3:
        h = np.zeros(n)
        block = max(2, n // 12)

            h[pick] = 0.0
    elif mode == 4:

        cnt = int(np.sum(h))

def lse_objective_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad[i] += w * dx * (1.0 - h[i + s])
                grad[i + s] -= w * dx * h[i]
        else:
            a = -s
            for i in range(a, n):
                grad[i] += w * dx * (1.0 - h[i - a])
                grad[i - a] -= w * dx * h[i]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_objective_and_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_objective_and_grad(x, self.tau)[1]


class EqConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))


def lse_optimize(h0, taus, deadline, maxiter_base=200):
    n = len(h0)
    h = h0.copy()
    curr_c = exact_objective(h)
    bounds = [(0.0, 1.0)] * n
    con = [EqConstraint(n / 2.0)]
    constraints = [{'type': 'eq', 'fun': con[0].fun, 'jac': con[0].jac}]
    for tau in taus:
        if time.time() > deadline - 8:
            break
        mi = min(maxiter_base, n * 3)
        try:
            res = minimize(LSEFunc(tau).value, h.copy(), method='SLSQP',
                           jac=LSEFunc(tau).gradient,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': mi, 'ftol': 1e-10})
            h_new = project_box_sum(np.array(res.x, dtype=np.float64))
            c_new = exact_objective(h_new)
            if c_new < curr_c - 1e-14:
                h = h_new
                curr_c = c_new
        except Exception:
            pass
    return h, curr_c


def epigraph_optimize(h0, deadline, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    n_prof = 2 * n - 1

    def fun(x):
        return x[n]

    def jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def constr_fun(x):
        h = x[:n]
        t = x[n]
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - profile

    def constr_jac(x):
        h = x[:n]
        n_prof_local = 2 * n - 1
        J = np.zeros((n_prof_local, n + 1))
        for j in range(n_prof_local):
            s = j - (n - 1)
            J[j, n] = 1.0
            if s >= 0:
                for i in range(n - s):
                    J[j, i] += dx * (1.0 - h[i + s])
                    J[j, i + s] -= dx * h[i]
            else:
                a = -s
                for i in range(a, n):
                    J[j, i] += dx * (1.0 - h[i - a])
                    J[j, i - a] -= dx * h[i]
        return J

    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = exact_objective(h0) + 0.05

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    try:
        res = minimize(fun, x0, method='SLSQP', jac=jac,
                       bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': constr_fun, 'jac': constr_jac},
                                    {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n/2.0,
                                     'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}],
                       options={'maxiter': maxiter, 'ftol': 1e-9})
        h_new = project_box_sum(np.array(res.x[:n], dtype=np.float64))
        c_new = exact_objective(h_new)
        return h_new, c_new
    except Exception:
        return h0.copy(), exact_objective(h0)


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_continuous_start(n, rng, mode):
    if mode == 0:
        h = np.full(n, 0.5)
        h += rng.normal(0, 0.1, n)
    elif mode == 1:
        x = np.arange(n) / n
        f1 = rng.choice([2, 3, 4, 5])
        f2 = rng.choice([3, 4, 5, 6, 7])
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.1, 0.4)
        a2 = rng.uniform(0.05, 0.2)
        h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + p1) + a2 * np.sin(2 * np.pi * f2 * x + p2)
    elif mode == 2:
        h = np.zeros(n)
        block = max(2, n // 6)
        for i in range(0, n, 2 * block):
            h[i:i+block] = rng.uniform(0.3, 1.0)
        h += rng.normal(0, 0.05, n)
    elif mode == 3:
        h = rng.uniform(0.2, 0.8, n)
    else:
        h = np.full(n, 0.5)
        h[rng.choice(n, n//2, replace=False)] += 0.3
        h[rng.choice(n, n//2, replace=False)] -= 0.3
    return project_box_sum(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load parent
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
    except Exception:
        pass

    if best_h is None:
        n_init = 48
        best_h = make_binary_start(n_init, rng, 0)
        best_c = exact_objective(best_h)

    parent_n = len(parent_h) if parent_h is not None else 48
    sizes = sorted(set([parent_n, 48, 52, 56, 60, 44, 50, 64]), reverse=True)
    sizes = [s for s in sizes if 24 <= s <= 96 and s % 2 == 0]

    # Phase 1: Binary SA for multiple sizes
    binary_best = {}
    for n in sizes:
        if time.time() > deadline - 80:
            break
        time_per_size = min(120.0, (deadline - 80 - time.time()) / max(len(sizes) - sizes.index(n), 1))
        local_best_b = None
        local_best_c = float('inf')

        n_starts = min(6, max(3, int(time_per_size / 15)))
        for s in range(n_starts):
            if time.time() > deadline - 80:
                break
            if s == 0 and parent_h is not None:
                h0 = transfer_to_size(parent_h, n)
                b0 = binary_threshold(h0)
            else:
                b0 = make_binary_start(n, rng, s % 7)
            if b0 is None:
                continue
            c0 = exact_objective(b0)
            if c0 < local_best_c:
                local_best_c = c0
                local_best_b = b0.copy()

            # SA from this start
            b_sa, c_sa = binary_sa(b0, deadline, seed + n * 100 + s, max_time_frac=0.7)
            if c_sa < local_best_c:
                local_best_c = c_sa
                local_best_b = b_sa.copy()

        # Greedy polish
        if local_best_b is not None:
            b_pol, c_pol = binary_greedy_polish(local_best_b, deadline, max_rounds=300)
            if c_pol < local_best_c:
                local_best_c = c_pol
                local_best_b = b_pol.copy()

        if local_best_b is not None:
            binary_best[n] = (local_best_b, local_best_c)
            if local_best_c < best_c:
                best_c = local_best_c
                best_h = local_best_b.copy()

    # Phase 2: Continuous optimization on best binary + parent transfers
    cont_best = {}
    sizes_to_cont = [n for n in sizes if n in binary_best or n == parent_n]
    sizes_to_cont = sizes_to_cont[:4]  # Limit to top 4 sizes

    for n in sizes_to_cont:
        if time.time() > deadline - 40:
            break

        # Start from best binary at this size
        if n in binary_best:
            h0 = binary_best[n][0].copy()
            # Smooth it slightly
            h0_smooth = project_box_sum(h0 + 0.05 * rng.normal(0, 1, n))
        elif parent_h is not None:
            h0 = transfer_to_size(parent_h, n)
            h0_smooth = h0.copy()
        else:
            h0 = make_continuous_start(n, rng, 0)
            h0_smooth = h0.copy()

        # LSE optimization
        taus = [0.05, 0.02, 0.008, 0.003, 0.001]
        h_ref, c_ref = lse_optimize(h0_smooth, taus, deadline, maxiter_base=150)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref.copy()

        # Also try from the binary directly (slightly perturbed)
        if n in binary_best:
            h_start2 = project_box_sum(binary_best[n][0] + 0.1 * rng.normal(0, 1, n))
            h_ref2, c_ref2 = lse_optimize(h_start2, taus[:3], deadline, maxiter_base=100)
            if c_ref2 < best_c:
                best_c = c_ref2
                best_h = h_ref2.copy()

        cont_best[n] = (h_ref, c_ref)

    # Phase 3: Epigraph polish on best continuous
    if best_h is not None and time.time() < deadline - 15:
        n_b = len(best_h)
        h_epi, c_epi = epigraph_optimize(best_h, deadline, maxiter=80)
        if c_epi < best_c:
            best_c = c_epi
            best_h = h_epi.copy()

    # Phase 4: More SA on best binary with longer run
    if best_h is not None and np.all((best_h == 0.0) | (best_h == 1.0)):
        if time.time() < deadline - 10:
            b_final, c_final = binary_sa(best_h, deadline, seed + 99999, max_time_frac=0.9)
            if c_final < best_c:
                best_c = c_final
                best_h = b_final.copy()

    # Phase 5: Final continuous polish
    if best_h is not None and not np.all((best_h == 0.0) | (best_h == 1.0)):
        if time.time() < deadline - 5:
            h_final, c_final = lse_optimize(best_h, [0.001, 0.0005], deadline, maxiter_base=100)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final.copy()

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
