```python
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def solve_linearized_lp(h_star, target_sum):
    n = len(h_star)
    corrs_star = compute_corrs(h_star)
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    for k in range(n_lags):
        g = grad_corr_k(h_star, k, n)
        A_ub[k, :n] = g
        A_ub[k, n] = -1.0
        b_ub[k] = np.dot(g, h_star) - corrs_star[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.clip(res.x[:n], 0.0, 1.0)
    return None


def solve_sca(h0, target_sum, max_iter=80, tol=1e-9):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, 2.0 / n)
    for it in range(max_iter):
        h_lp = solve_linearized_lp(h, target_sum)
        if h_lp is None:
            break
        h_lp = project_feasible(h_lp, target_sum)
        step = np.max(np.abs(h_lp - h))
        if step < tol:
            h = h_lp
            break
        h = h_lp
        c5_cur = c5_of_h(h, 2.0 / n)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5


def smooth_max_obj(x, dx, beta):
    corrs = compute_corrs(x)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(x, dx, beta):
    n = len(x)
    corrs = compute_corrs(x)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(x, k, n)
    return grad * dx


class SumConstraint:
    def __init__(self, target_sum):
        self.target_sum = target_sum

    def fun(self, x):
        return x.sum() - self.target_sum

    def jac(self, x):
        return np.ones(len(x))


class ObjFunc:
    def __init__(self, dx, beta):
        self.dx = dx
        self.beta = beta

    def value(self, x):
        return smooth_max_obj(x, self.dx, self.beta)

    def gradient(self, x):
        return smooth_max_grad(x, self.dx, self.beta)


def slsqp_opt(h0, dx, target_sum, beta, maxiter=400):
    n = len(h0)
    objf = ObjFunc(dx, beta)
    cf = SumConstraint(target_sum)
    cons = [{'type': 'eq', 'fun': cf.fun, 'jac': cf.jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objf.value, h0, jac=objf.gradient,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out


def active_set_refine(h, dx, target_sum, top_k, beta, maxiter):
    n = len(h)
    corrs = compute_corrs(h)
    sorted_idx = np.argsort(corrs)[::-1][:top_k]
    sorted_idx_list = list(sorted_idx)

    class ASObj:
        def __init__(self2, dx2, idx2, beta2):
            self2.dx = dx2
            self2.idx = idx2
            self2.beta = beta2

        def value(self2, x):
            c = compute_corrs(x)
            relevant = c[self2.idx]
            return logsumexp(self2.beta * relevant) / self2.beta * self2.dx

        def gradient(self2, x):
            c = compute_corrs(x)
            relevant = c[self2.idx]
            shifted = relevant - relevant.max()
            w = np.exp(self2.beta * shifted)
            w /= w.sum()
            grad = np.zeros(len(x))
            for i, k in zip(range(len(self2.idx)), self2.idx):
                if w[i] < 1e-14:
                    continue
                grad += w[i] * grad_corr_k(x, k, len(x))
            return grad * self2.dx

    ao = ASObj(dx, sorted_idx_list, beta)
    cf = SumConstraint(target_sum)
    cons = [{'type': 'eq', 'fun': cf.fun, 'jac': cf.jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(ao.value, h, jac=ao.gradient,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = h.copy()
    return h_out


def gen_mseq7():
    reg = 1
    seq = []
    for _ in range(127):
        seq.append(1.0 if (reg & 1) else -1.0)
        fb = (reg & 1) ^ ((reg >> 3) & 1)
        reg = (reg >> 1) | (fb << 6)
    return np.array(seq)


def mseq_to_h(mseq, n):
    d = mseq[:n]
    h = (d + 1.0) / 2.0
    return project_feasible(h, n / 2.0)


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    if n_ones > 0:
        h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def full_pipeline(h0, t0, budget_s):
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    h0 = project_feasible(h0, ts)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # SCA phase
    if remaining_time(t0, budget_s) > 15:
        h_sca, c5_sca = solve_sca(h0, ts, max_iter=60)
        if c5_sca < best_c5:
            best_c5 = c5_sca
            best_h = h_sca

    # SLSQP with increasing beta
    for beta in [100, 300, 800]:
        if remaining_time(t0, budget_s) < 8:
            break
        h_opt = slsqp_opt(best_h, dx, ts, beta, 350)
        c5_val = c5_of_h(h_opt, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Active set refinement
    if remaining_time(t0, budget_s) > 12:
        h_as = active_set_refine(best_h, dx, ts, top_k=15, beta=400, maxiter=250)
        c5_as = c5_of_h(h_as, dx)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as

    # Final SLSQP
    if remaining_time(t0, budget_s) > 10:
        h_final = slsqp_opt(best_h, dx, ts, 1500, 400)
        c5_final = c5_of_h(h_final, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    mseq7 = gen_mseq7()

    # Phase 1: initial_h_values
    try:
        h_init = np.array(initial_h_values, dtype=float)
        h_opt, c5_opt = full_pipeline(h_init, t0, budget_s)
        best_h, best_c5 = h_opt, c5_opt
    except Exception:
        pass

    # Phase 2: m-sequence starts for n=127, 126, 128
    for n_try in [127, 126, 128]:
        if remaining_time(t0, budget_s) < 40:
            break
        h_ms = mseq_to_h(mseq7, n_try)
        h_opt, c5_opt = full_pipeline(h_ms, t0, budget_s)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: more n values with SCA only (fast)
    for n_try in [125, 129, 130, 131, 132, 124, 133, 134]:
        if remaining_time(t0, budget_s) < 20:
            break
        ts_try = n_try / 2.0
        dx_try = 2.0 / n_try
        # Start from m-sequence
        h_ms = mseq_to_h(mseq7, n_try)
        h_sca, c5_sca = solve_sca(h_ms, ts_try, max_iter=40)
        if c5_sca < best_c5:
            # Polish with SLSQP
            h_pol = slsqp_opt(h_sca, dx_try, ts_try, 500, 250)
            c5_pol = c5_of_h(h_pol, dx_try)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
            else:
                best_c5 = c5_sca
                best_h = h_sca
        # Also try from interpolation of best
        if remaining_time(t0, budget_s) < 15 or best_h is None:
            continue
        n_b = len(best_h)
        x_orig = np.linspace(0.0, 2.0, n_b)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_sca2, c5_sca2 = solve_sca(h_interp, ts_try, max_iter=40)
        if c5_sca2 < best_c5:
            h_pol2 = slsqp_opt(h_sca2, dx_try, ts_try, 500, 250)
            c5_pol2 = c5_of_h(h_pol2, dx_try)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2

    # Phase 4: random binary starts with best n
    if best_h is not None and remaining_time(t0, budget_s) > 30:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        ts_best = n_best / 2.0
        for trial in range(40):
            if remaining_time(t0, budget_s) < 12:
                break
            h_bin = make_binary_start(n_best, ts_best, rng)
            h_sca = solve_sca(h_bin, ts_best, max_iter=40)[0]
            c5_sca = c5_of_h(h_sca, dx_best)
            if c5_sca < best_c5 + 0.002:
                h_pol = slsqp_opt(h_sca, dx_best, ts_best, 500, 200)
                c5_pol = c5_of_h(h_pol, dx_best)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 5: final intensive polish
    if best_h is not None and remaining_time(t0, budget_s) > 25:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        # High beta SLSQP
        h_hb = slsqp_opt(best_h, dx_f, ts_f, 2000, 500)
        c5_hb = c5_of_h(h_hb, dx_f)
        if c5_hb < best_c5:
            best_c5 = c5_hb
            best_h = h_hb
        # Active set with very few constraints
        if remaining_time(t0, budget_s) > 12:
            h_as = active_set_refine(best_h, dx_f, ts_f, top_k=8, beta=800, maxiter=300)
            c5_as = c5_of_h(h_as, dx_f)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as
        # SCA again from the polished point
        if remaining_time(t0, budget_s) > 10:
            h_sca_final, c5_sca_final = solve_sca(best_h, ts_f, max_iter=30)
            if c5_sca_final < best_c5:
                best_c5 = c5_sca_final
                best_h = h_sca_final
        # One final SLSQP
        if remaining_time(t0, budget_s) > 8:
            h_last = slsqp_opt(best_h, dx_f, ts_f, 3000, 400)
            c5_last = c5_of_h(h_last, dx_f)
            if c5_last < best_c5:
                best_c5 = c5_last
                best_h = h_last

    # Ensure feasibility
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, dx_final)
    return best_h, best_c5, n_final
```
