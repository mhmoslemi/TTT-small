def epigraph_sym_solve(q_init, dx, active_indices, t_init, maxiter):
    """Solve epigraph subproblem in symmetric space."""
    m = len(q_init)
    n = 2 * m
    dim = m + 1
    
    def obj(z):
        return z[m]
    
    def obj_grad(z):
        g = np.zeros(dim)
        g[m] = 1.0
        return g
    
    constraints = []
    for k in active_indices:
        def cfun(z, k=k):
            q = np.clip(z[:m], 0.0, 1.0)
            h = build_sym_h(q)
            corrs = np.correlate(h, 1.0 - h, mode='full')
            return z[m] - dx * corrs[k]
        
        def cjac(z, k=k):
            q = np.clip(z[:m], 0.0, 1.0)
            h = build_sym_h(q)
            offset = k - (n - 1)
            grad_h = np.zeros(n)
            if offset >= 0:
                grad_h[offset:] += (1.0 - h[:n - offset])
                grad_h[:n - offset] -= h[offset:]
            else:
                grad_h[:n + offset] += (1.0 - h[-offset:])
                grad_h[-offset:] -= h[:n + offset]
            grad_q = grad_h[:m] - grad_h[n-1:m-1:-1]
            j = np.zeros(dim)
            j[:m] = -dx * grad_q
            j[m] = 1.0
            return j
        
        constraints.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})
    
    bounds = [(0.0, 1.0)] * m + [(0.0, t_init + 1.0)]
    
    z0 = np.concatenate([q_init, [t_init]])
    
    result = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
    
    return result.x[:m]

    shifted = scaled - max_val
    exps = np.exp(beta * shifted)
    w = exps / exps.sum()
    val = max_val + np.log(exps.sum()) / beta
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        grad_h += w[k] * corr_grad_h(h, k, n)
    grad_q = grad_h[:m] - grad_h[n - 1:m - 1:-1]
    return val, dx * grad_q


def select_active(corrs, n, margin_frac=0.03, neighbor_pad=2, cap=50):
    """Select active lag indices for epigraph."""
    scaled_max = corrs.max()
    margin = max(1e-10, margin_frac * scaled_max)
    active = set()
    for k in range(len(corrs)):
        if corrs[k] >= scaled_max - margin:
            for d in range(-neighbor_pad, neighbor_pad + 1):
                idx = k + d
                if 0 <= idx < len(corrs):
                    active.add(idx)
    active_list = sorted(active)
    if len(active_list) > cap:
        sorted_by_val = sorted(active_list, key=lambda x: -corrs[x])
        active_list = sorted(sorted_by_val[:cap])
    return active_list


def epigraph_sym_solve(q_init, dx, active_k, t_init, maxiter=50, ftol=1e-10):
    """Solve epigraph subproblem in symmetric space (even n)."""
    m = len(q_init)
    n = 2 * m
    dim = m + 1

    def obj(z):
        return z[m]

    def obj_grad(z):
        g = np.zeros(dim)
        g[m] = 1.0
        return g

    constraints = []
    for k in active_k:
        def cfun(z, k=k):
            q = np.clip(z[:m], 0.0, 1.0)
            h = build_sym_h(q)
            corrs = np.correlate(h, 1.0 - h, mode='full')
            return z[m] - dx * corrs[k]

        def cjac(z, k=k):
            q = np.clip(z[:m], 0.0, 1.0)
            h = build_sym_h(q)
            gh = corr_grad_h(h, k, n)
            gq = gh[:m] - gh[n - 1:m - 1:-1]
            j = np.zeros(dim)
            j[:m] = -dx * gq
            j[m] = 1.0
            return j

        constraints.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})

    bounds = [(0.0, 1.0)] * m + [(0.0, t_init * 1.5 + 0.5)]
    z0 = np.concatenate([np.clip(q_init, 0.0, 1.0), [t_init]])

    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q_init.copy()

def full_space_smooth_solve(h0, dx, target_sum, beta, maxiter=300):
    """Full-space smooth max with SLSQP (equality constraint)."""
    n = len(h0)

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        scaled = dx * corrs
        return logsumexp(beta * scaled) / beta

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        scaled = dx * corrs
        shifted = scaled - scaled.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            grad += w[k] * corr_grad_h(h, k, n)
        return dx * grad

    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_target(res.x, target_sum)
    except Exception:
        return project_to_target(h0, target_sum)


def smooth_sym_optimize(q0, dx, betas, maxiter_each=200):
    """Run smooth max optimization with increasing beta."""
    q = q0.copy()
    for beta in betas:
        def f(z):
            v, _ = smooth_sym_val_grad(z, dx, beta)
            return v

        def g(z):
            _, gr = smooth_sym_val_grad(z, dx, beta)
            return gr

        bounds = [(0.0, 1.0)] * len(q)
        try:
            res = minimize(f, q, jac=g, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter_each,
                                                   'gtol': 1e-10})
            q = np.clip(res.x, 0.0, 1.0)
        except Exception:
            pass
    return q


def epigraph_loop(q0, dx, max_rounds=5, maxiter_each=40):
    """Iterative epigraph with active set expansion."""
    q = q0.copy()
    n = 2 * len(q)
    for _ in range(max_rounds):
        h = build_sym_h(q)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        scaled = dx * corrs
        t_cur = scaled.max()

        active = select_active(corrs, n, margin_frac=0.02, neighbor_pad=2, cap=45)
        q_new = epigraph_sym_solve(q, dx, active, t_cur + 1e-8,
                                   maxiter=maxiter_each)

        h_new = build_sym_h(q_new)
        t_new = np.correlate(h_new, 1.0 - h_new, mode='full').max() * dx

        if t_new < t_cur - 1e-12:
            q = q_new
        else:
            break
    return q


def transfer_to_n(h_src, n_new):
    """Interpolate h from its current length to n_new."""
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_src, h_src)
    return project_to_target(h_new, n_new / 2.0)


def make_random_sym_q(m, rng, sigma=0.15):
    x = np.linspace(0, 1, m)
    q = 0.5 + sigma * (np.sin(2 * np.pi * rng.uniform(0.5, 3.0) * x + rng.uniform(0, 2 * np.pi))
                       + 0.5 * np.sin(2 * np.pi * rng.uniform(3.0, 8.0) * x + rng.uniform(0, 2 * np.pi)))
    return np.clip(q, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_parent = None

    best_h = None
    best_score = float('inf')

    def accept(h_c):
        nonlocal best_h, best_score
        if h_c is None:
            return
        s = evaluate_exact(h_c)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_c.copy()

    # Initialize from parent
    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        h_parent = project_to_target(h_parent, n_p / 2.0)
        accept(h_parent)
    else:
        h_parent = None

    # Phase 1: Symmetric optimization from parent (even n)
    if h_parent is not None:
        n_p = len(h_parent)
        if n_p % 2 == 0:
            m_p = n_p // 2
            dx_p = 2.0 / n_p
            q_p = h_to_q(h_parent)

            # Smooth optimization
            if remaining() > 100:
                q_opt = smooth_sym_optimize(q_p, dx_p,
                                            betas=[60, 120, 240, 480, 900],
                                            maxiter_each=150)
                h_sym = build_sym_h(q_opt)
                accept(h_sym)

            # Epigraph refinement
            if remaining() > 80:
                q_eps = epigraph_loop(q_opt, dx_p, max_rounds=6, maxiter_each=40)
                h_eps = build_sym_h(q_eps)
                accept(h_eps)

            # More epigraph with tighter active set
            if remaining() > 60:
                h_cur = build_sym_h(q_eps)
                corrs = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                active_tight = select_active(corrs, n_p, margin_frac=0.005,
                                             neighbor_pad=1, cap=20)
                t_cur = (dx_p * corrs).max()
                q_tight = epigraph_sym_solve(q_eps, dx_p, active_tight,
                                             t_cur + 1e-9, maxiter=60, ftol=1e-12)
                accept(build_sym_h(q_tight))

        else:
            # Odd n parent
            m_p = (n_p - 1) // 2
            dx_p = 2.0 / n_p
            q_p = np.empty(m_p)
            for i in range(m_p):
                q_p[i] = 0.5 * (h_parent[i] + 1.0 - h_parent[n_p - 1 - i])
            q_p = np.clip(q_p, 0.0, 1.0)

            if remaining() > 100:
                q_opt = q_p.copy()
                for beta in [80, 160, 320, 640]:
                    def f_odd(z):
                        v, _ = smooth_sym_val_grad_odd(z, dx_p, beta)
                        return v
                    def g_odd(z):
                        _, gr = smooth_sym_val_grad_odd(z, dx_p, beta)
                        return gr
                    try:
                        res = minimize(f_odd, q_opt, jac=g_odd, method='L-BFGS-B',
                                       bounds=[(0.0, 1.0)] * m_p,
                                       options={'maxiter': 150, 'gtol': 1e-10})
                        q_opt = np.clip(res.x, 0.0, 1.0)
                    except Exception:
                        pass
                accept(build_sym_h_odd(q_opt))

    # Phase 2: Full-space SLSQP polish (can break symmetry)
    if best_h is not None and remaining() > 60:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_full = full_space_smooth_solve(best_h, dx_b, ts_b, beta=400, maxiter=400)
        accept(h_full)

    # Phase 3: Try different resolutions
    if best_h is not None and remaining() > 150:
        n_base = len(best_h)
        # Try nearby and smaller sizes
        candidates_n = []
        # Smaller (preferred)
        for dn in [1, 2, 3, 6]:
            for sign in [-1, 1]:
                nn = n_base + sign * dn
                if 20 <= nn <= 200:
                    candidates_n.append(nn)
        # Much smaller
        candidates_n.append(max(20, n_base // 2))
        candidates_n.append(max(20, n_base // 2 + 1))

        # Deduplicate and sort by closeness to n_base
        candidates_n = sorted(set(candidates_n), key=lambda x: abs(x - n_base))

        for n_try in candidates_n[:8]:
            if remaining() < 40:
                break
            if n_try == n_base:
                continue
            h_interp = transfer_to_n(best_h, n_try)
            dx_try = 2.0 / n_try
            ts_try = n_try / 2.0

            if n_try % 2 == 0:
                m_try = n_try // 2
                q_try = h_to_q(h_interp)

                # Quick smooth opt
                if remaining() > 30:
                    q_opt = smooth_sym_optimize(q_try, dx_try,
                                                betas=[100, 200, 400],
                                                maxiter_each=80)
                    h_cand = build_sym_h(q_opt)
                    s_cand = evaluate_exact(h_cand)

                    if s_cand < best_score + 1e-6:
                        accept(h_cand)
                        # Epigraph polish
                        if remaining() > 25:
                            q_eps = epigraph_loop(q_opt, dx_try,
                                                  max_rounds=3, maxiter_each=30)
                            accept(build_sym_h(q_eps))
            else:
                m_try = (n_try - 1) // 2
                q_try = np.empty(m_try)
                for i in range(m_try):
                    q_try[i] = 0.5 * (h_interp[i] + 1.0 - h_interp[n_try - 1 - i])
                q_try = np.clip(q_try, 0.0, 1.0)

                if remaining() > 30:
                    for beta in [100, 200, 400]:
                        def f_o(z):
                            v, _ = smooth_sym_val_grad_odd(z, dx_try, beta)
                            return v
                        def g_o(z):
                            _, gr = smooth_sym_val_grad_odd(z, dx_try, beta)
                            return gr
                        try:
                            res = minimize(f_o, q_try, jac=g_o, method='L-BFGS-B',
                                           bounds=[(0.0, 1.0)] * m_try,
                                           options={'maxiter': 80, 'gtol': 1e-10})
                            q_try = np.clip(res.x, 0.0, 1.0)
                        except Exception:
                            pass
                    accept(build_sym_h_odd(q_try))

    # Phase 4: Random restarts in symmetric space
    if best_h is not None and remaining() > 80:
        n_r = len(best_h)
        m_r = n_r // 2
        dx_r = 2.0 / n_r

        for trial in range(15):
            if remaining() < 20:
                break
            if m_r == 0:
                break
            q_rand = make_random_sym_q(m_r, rng, sigma=0.12)
            # Small smooth optimization
            q_opt_r = smooth_sym_optimize(q_rand, dx_r, betas=[80, 200],
                                          maxiter_each=60)
            h_r = build_sym_h(q_opt_r)
            s_r = evaluate_exact(h_r)
            if s_r < best_score + 1e-5:
                accept(h_r)
                if remaining() > 15:
                    q_eps_r = epigraph_loop(q_opt_r, dx_r, max_rounds=2,
                                            maxiter_each=25)
                    accept(build_sym_h(q_eps_r))

    # Phase 5: Final polish on best
    if best_h is not None and remaining() > 30:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0

        # Full space high beta
        if remaining() > 20:
            h_pol = full_space_smooth_solve(best_h, dx_f, ts_f, beta=800,
                                            maxiter=400)
            accept(h_pol)

        # Symmetric epigraph final
        if n_f % 2 == 0 and remaining() > 15:
            m_f = n_f // 2
            q_f = h_to_q(best_h)
            h_cur = build_sym_h(q_f)
            corrs = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            active_f = select_active(corrs, n_f, margin_frac=0.01,
                                     neighbor_pad=1, cap=30)
            t_f = (dx_f * corrs).max()
            q_fin = epigraph_sym_solve(q_f, dx_f, active_f, t_f + 1e-9,
                                       maxiter=80, ftol=1e-12)
            accept(build_sym_h(q_fin))

    # Ensure we have a valid result
    if best_h is None:
        n_def = 126
        x = np.linspace(0, 1, n_def // 2)
        q_def = np.clip(0.5 + 0.2 * np.sin(np.pi * x), 0.0, 1.0)
        best_h = build_sym_h(q_def)
        best_score = evaluate_exact(best_h)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_to_target(best_h, n_final / 2.0)
    best_score = evaluate_exact(best_h)

    if not np.isfinite(best_score) or best_score > 1.0:
        # Emergency fallback
        n_fb = 126
        q_fb = np.full(n_fb // 2, 0.5)
        best_h = build_sym_h(q_fb)
        best_score = evaluate_exact(best_h)

    return best_h, best_score, len(best_h)
