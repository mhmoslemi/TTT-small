def repair(h):
    """Clip to [0,1] and fix sum to n/2 via water-filling."""
    h = np.asarray(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-15:
        if diff > 0:
            # Need to add mass
            order = np.argsort(h)
            rem = diff
            for i in order:
                if rem <= 1e-15:
                    break
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
        else:
            # Need to remove mass
            order = np.argsort(-h)
            rem = -diff
            for i in order:
                if rem <= 1e-15:
                    break
                remv = min(rem, h[i])
                h[i] -= remv
                rem -= remv
    # Final fix

    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + residual)
    return h

    """Compute all lag values and gradients."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))

            values[idx] = dx * np.sum(h * (1.0 - h))

    n = len(h)
    values, grads = compute_all_lags(h)
    m = np.max(values)
    weights = np.exp(tau * (values - m))
    weights /= np.sum(weights)

Let me think about one more thing: the "pair-based local search" idea.

For a given lag d with high Q_d, we want to reduce the overlap. The overlap is:
Q_d = dx * Σ_{j=0}^{n-1-d} h[j] * (1-h[j+d])

To reduce this, for each term h[j]*(1-h[j+d]), we can:
- Decrease h[j] (if h[j+d] is high)
- Increase h[j+d] (if h[j] is high)
- Or both

But we need to maintain sum(h) = n/2, so any decrease in one place must be compensated by an increase elsewhere.

A simple approach: for a pair (j, j+d), if h[j] > h[j+d], try to decrease h[j] and increase h[j+d] (pushing them apart). The change δ should be small.

The change in Q_d from this move:
- Term j: h[j]*(1-h[j+d]) → (h[j]-δ)*(1-h[j+d]-δ) = h[j](1-h[j+d]) - δ(1-h[j+d]) - δ*h[j] + δ²
  Change: -δ(1-h[j+d]) - δ*h[j] + δ² = -δ(1-h[j+d]+h[j]) + δ²
- Term j-d (if exists, where j-d is the first index): h[j-d]*(1-h[j]) → h[j-d]*(1-(h[j]-δ)) = h[j-d](1-h[j]) + δ*h[j-d]
  Change: +δ*h[j-d]
- Term j+d (if j+d+d < n, i.e., j+2d < n): h[j+d]*(1-h[j+2d]) → (h[j+d]+δ)*(1-h[j+2d])
  Change: +δ*(1-h[j+2d])

So the total change in Q_d is approximately:
δ * [-(1-h[j+d]+h[j]) + h[j-d] + (1-h[j+2d])] (dropping the δ² term)

This is a linear function of δ. If the coefficient is negative, increasing δ reduces Q_d.

import numpy as np

import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair(h):
    h = np.asarray(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(h)
            rem = diff
            for i in order:
                if rem <= 1e-15:
                    break
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
        else:
            order = np.argsort(-h)
            rem = -diff
            for i in order:
                if rem <= 1e-15:
                    break
                remv = min(rem, h[i])
                h[i] -= remv
                rem -= remv
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            i = np.argmax(1.0 - h)
            h[i] = min(1.0, h[i] + residual)
        else:
            i = np.argmin(h)
            h[i] = max(0.0, h[i] + residual)
    return h


def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    values = np.zeros(L)
    grads = np.zeros((L, n))
    for d in range(n):
        if d == 0:
            values[n - 1] = dx * np.sum(h * (1.0 - h))
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            j = np.arange(n - d)
            jp = j + d
            values[n - 1 + d] = dx * np.sum(h[j] * (1.0 - h[jp]))
            grads[n - 1 + d, j] += dx * (1.0 - h[jp])
            grads[n - 1 + d, jp] -= dx * h[j]
            j2 = np.arange(d, n)
            j2p = j2 - d

            grads[n - 1 - d, j2] += dx * (1.0 - h[j2p])

    return values, grads

def subgradient_momentum(h0, tau, n_steps, alpha, beta, deadline):
    n = len(h0)
    h = h0.copy()
    best = h.copy()

        g -= np.mean(g)
        velocity = beta * velocity - alpha * g
        h_new = repair(h + velocity)
        sc = exact_score(h_new)
        if sc < best_sc - 1e-14:
            best = h_new.copy()
            best_sc = sc
        h = h_new
    return best, best_sc


def active_descent(h, max_steps=80, deadline=None):
    n = len(h)
    h = repair(h.copy())
    best = h.copy()
    best_score = exact_score(best)
    for _ in range(max_steps):
        if deadline is not None and time.perf_counter() > deadline:
            break
        values, grads = compute_lags(best)
        c = values.max()
        n_active = min(20, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_active]
        gaps = np.maximum(c - values[active_idx], 0.0)
        sigma = max(0.003 * c, 1e-6)
        weights = np.exp(-gaps / sigma)
        weights /= weights.sum() + 1e-15
        direction = -(weights @ grads[active_idx, :])
        direction -= np.mean(direction)
        improved = False
        for alpha in [0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair(best + alpha * direction)
            sc = exact_score(h_trial)
            if sc < best_score - 1e-13:
                best = h_trial
                best_score = sc
                improved = True
                break
        if not improved:
            break
    return best, best_score


def interpolate_dim(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_n) + 0.5) * 2.0 / new_n
    xs = np.concatenate([[0.0], x_src, [2.0]])
    ys = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_tgt, xs, ys)
    return repair(h_new)


def pair_local_search(h, max_iters=2000, deadline=None):
    n = len(h)
    h = repair(h.copy())
    best = h.copy()
    best_sc = exact_score(best)
    rng = np.random.RandomState(12345)
    for it in range(max_iters):
        if deadline is not None and time.perf_counter() > deadline:
            break
        if it % 20 == 0:
            values, _ = compute_lags(best)
            d = int(np.argmax(values[:n]) + n - 1) - (n - 1)
            if d <= 0:
                d = int(np.argmax(values[n:]))
        j = rng.randint(0, max(1, n - d))
        delta = rng.choice([-1, 1]) * rng.uniform(0.005, 0.03)
        h_trial = best.copy()
        h_trial[j] += delta
        h_trial[j + d] -= delta
        if h_trial[j] < 0 or h_trial[j] > 1 or h_trial[j + d] < 0 or h_trial[j + d] > 1:
            continue
        sc = exact_score(h_trial)
        if sc < best_sc - 1e-14:
            best = h_trial
            best_sc = sc
    return best, best_sc


def structured_perturb(h, rng, amplitude, freq):
    n = len(h)
    x = np.arange(n) * 2.0 / n
    pert = amplitude * np.sin(2 * np.pi * freq * x)
    h_new = repair(h + pert)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    deadline = t_start + budget_s - 10
    rng = np.random.RandomState(seed)

    def time_left():
        return deadline - time.perf_counter()

    # Load parent
    try:
        best_h = repair(np.asarray(initial_h_values, dtype=np.float64))
    except Exception:
        best_h = np.full(140, 0.5)

    # Try symmetrization
    h_sym = repair((best_h + best_h[::-1]) / 2.0)
    if exact_score(h_sym) < exact_score(best_h) - 1e-12:
        best_h = h_sym
    best_score = exact_score(best_h)
    best_n = len(best_h)

    # Phase 1: LSE-SLSQP annealing
    tau_schedule = [3, 6, 12, 25, 50, 100, 200, 400]
    for tau in tau_schedule:
        if time_left() < 80:
            break
        h_sol, sc = solve_lse_slsqp(best_h, tau=tau, mu=50.0, maxiter=250)
        if h_sol is not None and sc < best_score - 1e-12:
            best_h = h_sol
            best_score = sc
            best_n = len(h_sol)

    # Phase 2: Subgradient with momentum
    if time_left() > 60:
        dl = min(deadline, time.perf_counter() + min(120, time_left() * 0.5))
        h_sg, sc_sg = subgradient_momentum(best_h, tau=50, n_steps=3000,
                                           alpha=0.003, beta=0.9, deadline=dl)
        if sc_sg < best_score - 1e-12:
            best_h = h_sg
            best_score = sc_sg

    # Phase 3: Active-set SLSQP
    if time_left() > 80:
        values, _ = compute_lags(best_h)
        n_lags = len(values)
        for top_k in [30, 50, 80]:
            if time_left() < 40:
                break
            active = np.argsort(values)[::-1][:top_k]
            active = np.sort(active)
            h_sol, sc = solve_lse_active(best_h, active, tau=80, maxiter=200)
            if h_sol is not None and sc < best_score - 1e-12:
                best_h = h_sol
                best_score = sc
                best_n = len(h_sol)

    # Phase 4: Active descent
    if time_left() > 40:
        dl = min(deadline, time.perf_counter() + min(100, time_left() * 0.6))
        h_ad, sc_ad = active_descent(best_h, max_steps=200, deadline=dl)
        if sc_ad < best_score - 1e-12:
            best_h = h_ad
            best_score = sc_ad

    # Phase 5: Dimension changes
    if time_left() > 120:
        n_cur = len(best_h)
        dims = []
        for dn in [16, 28, 42, 56]:
            nd = n_cur + dn
            if nd <= 200:
                dims.append(nd)
        for dn in [16, 28, 42]:
            nd = n_cur - dn
            if nd >= 60:
                dims.append(nd)
        for nd in sorted(dims):
            if time_left() < 60:
                break
            h_interp = interpolate_dim(best_h, nd)
            sc_interp = exact_score(h_interp)
            if sc_interp < best_score + 0.01:
                # Try LSE-SLSQP at new dim
                h_sol, sc = solve_lse_slsqp(h_interp, tau=50, mu=50.0, maxiter=150)
                if h_sol is not None and sc < best_score - 1e-12:
                    best_h = h_sol
                    best_score = sc
                    best_n = len(h_sol)
                # Try active descent at new dim
                if time_left() > 40:
                    h_ad, sc_ad = active_descent(h_interp, max_steps=100,
                                                 deadline=min(deadline, time.perf_counter() + 60))
                    if sc_ad < best_score - 1e-12:
                        best_h = h_ad
                        best_score = sc_ad
                        best_n = len(h_ad)

    # Phase 6: Structured perturbations + re-optimize
    if time_left() > 80:
        for trial in range(6):
            if time_left() < 30:
                break
            freq = rng.choice([1, 2, 3, 4, 5, 6, 8])
            amp = rng.uniform(0.01, 0.04)
            h_pert = structured_perturb(best_h, rng, amp, freq)
            h_sol, sc = solve_lse_slsqp(h_pert, tau=50, mu=50.0, maxiter=150)
            if h_sol is not None and sc < best_score - 1e-12:
                best_h = h_sol
                best_score = sc
                best_n = len(h_sol)

    # Phase 7: Pair local search
    if time_left() > 50:
        dl = min(deadline, time.perf_counter() + min(150, time_left() * 0.7))
        h_ps, sc_ps = pair_local_search(best_h, max_iters=5000, deadline=dl)
        if sc_ps < best_score - 1e-12:
            best_h = h_ps
            best_score = sc_ps

    # Phase 8: Final high-precision LSE-SLSQP
    if time_left() > 40:
        for tau in [300, 500, 800]:
            if time_left() < 30:
                break
            h_sol, sc = solve_lse_slsqp(best_h, tau=tau, mu=200.0, maxiter=200)
            if h_sol is not None and sc < best_score - 1e-12:
                best_h = h_sol
                best_score = sc

    # Phase 9: Final active descent polish
    if time_left() > 20:
        h_ad, sc_ad = active_descent(best_h, max_steps=150, deadline=min(deadline, time.perf_counter() + 60))
        if sc_ad < best_score - 1e-12:
            best_h = h_ad
            best_score = sc_ad

    # Phase 10: One more subgradient pass with high tau
    if time_left() > 30:
        dl = min(deadline, time.perf_counter() + min(60, time_left() * 0.8))
        h_sg, sc_sg = subgradient_momentum(best_h, tau=200, n_steps=2000,
                                           alpha=0.001, beta=0.95, deadline=dl)
        if sc_sg < best_score - 1e-12:
            best_h = h_sg
            best_score = sc_sg

    # Final validation
    best_h = repair(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
