def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2

        else:
