    n = len(h)
    k = lag_idx - (n - 1)
    grad = np.zeros(n)

        # h[p] appears as h[i+k] when i = p-k, valid if 0 <= p-k <= n-1-k => k <= p <= n-1

        for p in range(k, n):
            grad[p] += (1.0 - h[p-k])
        for p in range(0, n - k):
            grad[p] -= h[p+k]
    else:
        m = -k
        # h[p] appears as h[j] when j = p, valid if 0 <= p <= n-1-m
        # h[p] appears as h[j+m] when j = p-m, valid if m <= p <= n-1

            grad[p] += (1.0 - h[p+m])
        for p in range(m, n):
            grad[p] -= h[p-m]
    
    return grad

def project_onto_feasible(y, target_sum):

    n = len(y)
    lo = np.min(y) - 1.0

    for _ in range(100):

        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    h = np.clip(y - t, 0.0, 1.0)
    # Fix any residual

    if abs(residual) > 1e-12:

def grad_lag(h, k):
    """Vectorized gradient of C(k) w.r.t. h. k is actual lag (negative to positive)."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:n] += (1.0 - h[0:n-k])
            g[0:n-k] -= h[k:n]
    else:

def project_feasible(y, target):
    """Project y onto {h: 0<=h<=1, sum(h)=target} via bisection."""
    n = len(y)
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(interior) > 0:
            h[interior] += residual / np.sum(interior)
    return np.clip(h, 0.0, 1.0)

def trust_region_minimax(h, max_iter=200, R_init=0.05):
    """Trust region method for the minimax problem."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5, _ = compute_c5_full(best_h)
    R = R_init
    
    for it in range(max_iter):
        c5, corr_arr = compute_c5_full(best_h)
        max_corr = np.max(corr_arr)
        
        # Active set: top 5 lags or within 5% of max
        sorted_idx = np.argsort(corr_arr)[::-1]
        active_count = min(5, n)
        active_idx = sorted_idx[:active_count]
        # Also include any within 2% of max
        close_idx = np.where(corr_arr >= max_corr - 0.02 * max_corr - 0.005)[0]
        active_idx = np.unique(np.concatenate([active_idx, close_idx]))
        
        # Build gradient matrix for active lags
        G = np.zeros((len(active_idx), n))
        for ai, idx in enumerate(active_idx):
            k = idx - (n-1)
            G[ai] = grad_lag(best_h, k)
        
        # Solve: min_d max_i (C[i] + G[i]@d) + (rho/2)||d||^2
        #        s.t. sum(d)=0, h+d in [0,1], ||d||<=R
        # Reformulate: min_{d,t} t + (rho/2)||d||^2
        #        s.t. t >= C[i] + G[i]@d, sum(d)=0, bounds
        
        rho = 50.0 / n
        best_d = np.zeros(n)
        best_subobj = np.inf
        
        # Try several random directions + gradient directions
        directions = []
        for ai in range(min(len(active_idx), 3)):
            d_dir = -G[ai].copy()
            d_dir -= np.mean(d_dir)
            norms = np.linalg.norm(d_dir)
            if norms > 1e-12:
                d_dir = d_dir / norms * R
                directions.append(d_dir)
        
        # Random directions
        rng_local = np.random.RandomState(it + 42)
        for _ in range(5):
            d_rand = rng_local.randn(n)
            d_rand -= np.mean(d_rand)
            nr = np.linalg.norm(d_rand)
            if nr > 1e-12:
                d_rand = d_rand / nr * R
                directions.append(d_rand)
        
        # Negative combined gradient
        g_comb = np.mean(G, axis=0)
        g_comb -= np.mean(g_comb)
        ng = np.linalg.norm(g_comb)
        if ng > 1e-12:
            directions.append(-g_comb / ng * R)
        
        for d_dir in directions:
            # Line search along d_dir from 0 to R
            for scale in [0.1, 0.3, 0.5, 0.7, 1.0]:
                d = d_dir * scale
                # Check bounds
                trial = best_h + d
                if np.min(trial) < -1e-12 or np.max(trial) > 1 + 1e-12:
                    continue
                if abs(np.sum(d)) > 1e-10:
                    d -= np.mean(d)
                    trial = best_h + d
                    if np.min(trial) < -1e-12 or np.max(trial) > 1 + 1e-12:
                        continue
                trial = np.clip(trial, 0.0, 1.0)
                trial = project_feasible(trial, target)
                trial_c5 = compute_c5_full(trial)[0]
                subobj = trial_c5 + rho * np.linalg.norm(trial - best_h) ** 2
                if subobj < best_subobj:
                    best_subobj = subobj
                    best_d = trial - best_h
        
        if np.linalg.norm(best_d) < 1e-12:
            R *= 0.5
            if R < 1e-6:
                break
            continue
        
        # Evaluate and accept
        trial = project_feasible(best_h + best_d, target)
        trial_c5 = compute_c5_full(trial)[0]
        
        if trial_c5 < best_c5 - 1e-14:
            best_h = trial
            best_c5 = trial_c5
            R = min(R * 1.3, 0.3)
        else:
            R *= 0.6
            if R < 1e-6:
                break
    
    return best_h, best_c5

def slsqp_polish(h, max_iter=300):
    """SLSQP polish using smoothed max objective."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def smooth_obj(hv, beta):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        shifted = beta * (corr - m)
        exps = np.exp(shifted - np.max(shifted))
        s = np.sum(exps)
        return (m + np.log(s) / beta) * dx
    
    def smooth_grad(hv, beta):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        shifted = beta * (corr - m)
        exps = np.exp(shifted - np.max(shifted))
        s = np.sum(exps)
        w = exps / s
        g = np.zeros(n)
        for idx in range(len(corr)):
            if w[idx] < 1e-12:
                continue
            k = idx - (n - 1)
            g += w[idx] * grad_lag(hv, k)
        return g * dx
    
    def eq_fun(hv):
        return np.sum(hv) - target
    
    def eq_jac(hv):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    best_h = h.copy()
    best_c5 = compute_c5_full(h)[0]
    
    for beta in [20, 100, 500, 2000, 10000]:
        try:
            res = minimize(
                smooth_obj, best_h, args=(beta,),
                jac=lambda x: smooth_grad(x, beta),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-14}
            )
            if res.x is not None:
                trial = project_feasible(res.x, target)
                trial_c5 = compute_c5_full(trial)[0]
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial
        except Exception:
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(60, 0.5)
    
    n = len(h_init)
    target = n / 2.0
    h_init = project_feasible(h_init, target)
    
    best_h = h_init.copy()
    best_c5 = compute_c5_full(h_init)[0]
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Phase 1: Gradient step method
    if time_left() > 60:
        h_new, c5_new = gradient_step_method(best_h, max_iter=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 2: Trust region
    if time_left() > 50:
        h_new, c5_new = trust_region_minimax(best_h, max_iter=150, R_init=0.04)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 3: Directed local search
    if time_left() > 30:
        n_iters = min(50000, int(time_left() * 60))
        h_new, c5_new = directed_local_search(best_h, n_iters, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 4: SLSQP polish
    if time_left() > 40:
        h_new, c5_new = slsqp_polish(best_h, max_iter=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 5: More gradient + local search alternating
    while time_left() > 80:
        # Gradient step
        h_new, c5_new = gradient_step_method(best_h, max_iter=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        if time_left() < 40:
            break
        # Local search
        n_iters = min(20000, int(time_left() * 30))
        h_new, c5_new = directed_local_search(best_h, n_iters, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 6: Random restarts with different starts
    if time_left() > 100:
        strategies = []
        # Binary starts
        for _ in range(3):
            strategies.append(make_binary_balanced(n, rng))
        # Wave starts
        for k in [3, 5, 7, 10]:
            for phase in [0, np.pi/3, np.pi/2, 2*np.pi/3]:
                strategies.append(make_wave(n, k, phase, rng))
        # Perturbed best
        for scale in [0.1, 0.2, 0.35]:
            h_pert = best_h + scale * rng.randn(n) * 0.1
            strategies.append(project_feasible(h_pert, target))
        # Flip parts of best
        for _ in range(3):
            h_flip = best_h.copy()
            mask = rng.random(n) < 0.15
            h_flip[mask] = 1.0 - h_flip[mask]
            strategies.append(project_feasible(h_flip, target))
        
        for h_start in strategies:
            if time_left() < 60:
                break
            h_start = project_feasible(h_start, target)
            h_try, c5_try = gradient_step_method(h_start, max_iter=100)
            if c5_try < best_c5:
                # Polish the improvement
                h_try, c5_try = directed_local_search(h_try, 5000, rng)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
    
    # Phase 7: Resolution exploration
    if time_left() > 80:
        n_current = len(best_h)
        n_candidates = []
        for n_try in [24, 30, 36, 40, 48, 50, 54, 60, 66, 72, 78, 84]:
            if n_try != n_current and n_try >= 20:
                n_candidates.append(n_try)
        
        for n_try in n_candidates:
            if time_left() < 50:
                break
            h_interp = resample_h(best_h, n_try)
            h_try, c5_try = gradient_step_method(h_interp, max_iter=150)
            if c5_try < best_c5:
                h_try, c5_try = directed_local_search(h_try, 10000, rng)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    n_current = n_try
            if time_left() < 40:
                break
    
    # Phase 8: Final polish
    if time_left() > 30:
        h_new, c5_new = gradient_step_method(best_h, max_iter=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    if time_left() > 20:
        n_iters = min(30000, int(time_left() * 40))
        h_new, c5_new = directed_local_search(best_h, n_iters, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    if time_left() > 15:
        h_new, c5_new = slsqp_polish(best_h, max_iter=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Final validation
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    final_c5 = compute_c5_full(best_h)[0]
    
    # Verify feasibility
    if not (np.all(best_h >= -1e-10) and np.all(best_h <= 1 + 1e-10)):
        best_h = np.clip(best_h, 0, 1)
        best_h = project_feasible(best_h, target_final)
        final_c5 = compute_c5_full(best_h)[0]
    
    if abs(np.sum(best_h) - target_final) > 1e-8:
        best_h = project_feasible(best_h, target_final)
        final_c5 = compute_c5_full(best_h)[0]
    
    return best_h, final_c5, len(best_h)
