def lag_gradient(x, lag, dx):
    n = len(x)

    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= x[lag:]

        s = n - a
        if s > 0:
            grad[:s] += 1.0 - x[a:]
            grad[a:] -= x[:s]

def smooth_max_grad(profile, tau):
    """Compute soft-max value and its gradient w.r.t. profile."""
    pmax = np.max(profile)
    if tau < 1e-10:
        idx = np.argmax(profile)
        return pmax, idx
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    wsum = np.sum(w)
    wn = w / wsum
    value = pmax + tau * np.log(wsum)
    return value, wn  # wn are the weights over lags

def obj_and_grad(x, tau, target):

def project_box_sum(h, target):
    n = len(h)

        diff = target - np.sum(h)

            break

            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            room = 1.0 - h[mask]

import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_to_polytope(h, target):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0).copy()
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, (1.0 - h[mask]).max())
            h[mask] += step
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = max(diff / cnt, -h[mask].min())
            h[mask] += step
    return h

def _lag_gradient(x, lag, dx):
    n = len(x)
    g = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            g[:s] -= x[lag:]
            g[lag:] += 1.0 - x[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            g[:s] += 1.0 - x[a:]
            g[a:] -= x[:s]
    return g * dx

def _profile(x):
    n = len(x)
    dx = 2.0 / n
    return np.correlate(x, 1.0 - x, mode='full') * dx

def _softmax.obj_grad(x, tau, target):
    n = len(x)
    dx = 2.0 / n
    prof = _profile(x)
    pmax = np.max(prof)
    if tau < 1e-11:
        m = int(np.argmax(prof))
        return pmax, _lag_gradient(x, m - (n - 1), dx)
    sh = np.clip((prof - pmax) / tau, -50, 50)
    w = np.exp(sh)
    ws = w.sum()
    wn = w / ws
    val = pmax + tau * np.log(ws)
    grad = np.zeros(n)
    for m in np.where(wn > 1e-7)[0]:
        grad += wn[m] * _lag_gradient(x, m - (n - 1), dx)
    return val, grad

def _slsqp_stage(x0, tau, target, maxiter):
    n = len(x0)
    bounds = [(0.0, 1.0)] * n
    def objf(x):
        v, g = _softmax_obj_grad(x, tau, target)
        se = np.sum(x) - target
        v += 5e4 * se * se
        g = g + 1e5 * se * np.ones(n)
        return v, g
    def eqf(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones(n)
    cons = {'type': 'eq', 'fun': eqf, 'jac': eqj}
    try:
        res = minimize(objf, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_polytope(res.x, target)
    except Exception:
        pass
    return x0

def _lbfgs_stage(x0, tau, target, maxiter):
    n = len(x0)
    bounds = [(0.0, 1.0)] * n
    def objf(x):
        v, _ = _softmax_obj_grad(x, tau, target)
        se = np.sum(x) - target
        return v + 1e4 * se * se
    def jacf(x):
        _, g = _softmax_obj_grad(x, tau, target)
