import cvxpy as cp
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n):
    m = n
    dx = 2.0 / m
    G = np.zeros((2 * m - 1, m))
    for i in range(2 * m - 1):
        L = i - (m - 1)
        if L >= 0:
            G[i, L:m] += dx * (1.0 - h[:m-L])
            G[i, 0:m-L] -= dx * h[L:m]
        else:
            Lp = -L
            G[i, 0:m-Lp] += dx * (1.0 - h[Lp:m])
            G[i, Lp:m] -= dx * h[0:m-Lp]
    return G

    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    R = initial_R
    rho = 2.0
    consecutive_fails = 0
    t_start = time.time()
    
    while time.time() - t_start < max_time:
        c5_cur, corr_dx = compute_c5(best_h)
        
        # Select active set: lags within band of max
        max_val = np.max(corr_dx)
        band = max(0.005, 0.03 * max_val)
        active_idx = np.where(corr_dx >= max_val - band)[0]
        if len(active_idx) == 0:

        G = build_gradient_matrix(best_h, n)
        F = corr_dx[active_idx]
        G_active = G[active_idx, :]

            u_var >= -best_h - 1e-9,

                    break

def smooth_obj_wrapper(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = beta * (corr - np.max(corr))
    exps = np.exp(shifted)
    val = np.max(corr) + np.log(np.sum(exps)) / beta
    return val * dx

def smooth_grad_wrapper(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    
    G = build_gradient_matrix_unscaled(h, n)
    grad = (weights @ G.T) * dx
    return grad

def build_gradient_matrix_unscaled(h, n):
    # Same as build_gradient_matrix but without dx factor
    m = n
    dx = 2.0 / m
    G = np.zeros((2 * m - 1, m))
    for i in range(2 * m - 1):
        L = i - (m - 1)
        if L >= 0:
            G[i, L:m] += dx * (1.0 - h[:m-L])
            G[i, 0:m-L] -= dx * h[L:m]
        else:
            Lp = -L
            G[i, 0:m-Lp] += dx * (1.0 - h[Lp:m])
            G[i, Lp:m] -= dx * h[0:m-Lp]
    return G

def slsqp_polish(h, betas, max_iter=200):

    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    
    def eq_con_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    
    for beta in betas:

            smooth_obj_wrapper, best_h, args=(beta,),
            jac=smooth_grad_wrapper,

def resample_h(h_old, n_old, n_new):

    dx_new = 2.0 / n_new
    
    # Create old sample positions (centers of intervals)
    x_old = (np.arange(n_old) + 0.5) * dx_old
    # Create new sample positions
    x_new = (np.arange(n_new) + 0.5) * dx_new
    
    # Interpolate
    h_interp = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    
    # Project to feasible
    h_new = project_feasible(h_interp, n_new)
    return h_new

def project_feasible(y, n):

    hi = float(np.max(y)) + 1.0

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    G = build_G(h, n)
    return weights @ G.T

def _eq_con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_con_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, betas, max_iter=200):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    
    for beta in betas:
        try:
            res = minimize(
                _smooth_obj, best_h, args=(beta,),
                jac=_smooth_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-15}
            )
            if res.x is not None:
                h_cand = project_feasible(res.x, n)
                c5_new, _ = compute_c5(h_cand)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cand
        except Exception:
            continue
    
    return best_h, best_c5

def targeted_anneal(h, max_time, rng):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5, _ = compute_c5(cur_h)
    
    T = 0.015
    T_min = 0.0005
    t_start = time.time()
    no_improve = 0
    eval_count = 0
    
    while time.time() - t_start < max_time and T > T_min:
        _, corr_dx = compute_c5(cur_h)
        max_val = np.max(corr_dx)
        band = max(0.003, 0.05 * max_val)
        active_mask = corr_dx >= max_val - band
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(corr_dx)])
        
        lag_idx = active_lags[rng.randint(len(active_lags))]
        L = lag_idx - (n - 1)
        h_new = None
        
        if L >= 0 and L < n:
            if L == 0:
                i, j = rng.choice(n, size=2, replace=False)
                mt = min(cur_h[i], 1.0 - cur_h[j])
                if mt > 1e-7:
                    delta = rng.uniform(0.03, 0.7) * mt
                    h_new = cur_h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
            else:
                wts = cur_h[L:n] * (1.0 - cur_h[:n - L])
                tw = wts.sum()
                if tw > 1e-12:
                    probs = wts / tw
                    pi = rng.choice(n - L, p=probs)
                    a = pi
                    b = pi + L
                    mt = min(cur_h[b], 1.0 - cur_h[a])
                    if mt > 1e-7:
                        delta = rng.uniform(0.03, 0.7) * mt
                        h_new = cur_h.copy()
                        h_new[b] -= delta
                        h_new[a] += delta
        elif L < 0:
            Lp = -L
            if Lp == 0:
                i, j = rng.choice(n, size=2, replace=False)
                mt = min(cur_h[i], 1.0 - cur_h[j])
                if mt > 1e-7:
                    delta = rng.uniform(0.03, 0.7) * mt
                    h_new = cur_h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
            elif Lp < n:
                wts = cur_h[:n - Lp] * (1.0 - cur_h[Lp:n])
                tw = wts.sum()
                if tw > 1e-12:
                    probs = wts / tw
                    pi = rng.choice(n - Lp, p=probs)
                    a = pi
                    b = pi + Lp
                    mt = min(cur_h[a], 1.0 - cur_h[b])
                    if mt > 1e-7:
                        delta = rng.uniform(0.03, 0.7) * mt
                        h_new = cur_h.copy()
                        h_new[a] -= delta
                        h_new[b] += delta
        
        if h_new is None:
            # Random fallback move
            i, j = rng.choice(n, size=2, replace=False)
            mt = min(cur_h[i], 1.0 - cur_h[j])
            if mt > 1e-7:
                delta = rng.uniform(0.03, 0.7) * mt
                h_new = cur_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
        
        if h_new is None:
            T *= 0.999
            continue
        
        c5_new, _ = compute_c5(h_new)
        delta_c = c5_new - cur_c5
        eval_count += 1
        
        if delta_c < 0 or (T > T_min and rng.random() < np.exp(-delta_c / T)):
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= 0.9997
        if no_improve == 4000:
            T = max(T, 0.008)
            no_improve = 0
    
    return best_h, best_c5

def pair_swap_search(h, n_passes=50):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if abs(best_h[i] - best_h[j]) < 1e-8:
                    continue
                h_new = best_h.copy()
                h_new[i], h_new[j] = best_h[j], best_h[i]
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_h = h_new
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    
    return best_h, best_c5

def resample_h(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_interp = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_interp, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(60, 0.5)
    
    best_h = h_init.copy()
    best_c5, _ = compute_c5(best_h)
    
    # Phase 1: SLP from initial
    rem = deadline - time.time()
    if rem > 80:
        t_limit = min(rem - 10, 150)
        h_opt, c5_opt = slp_polish(best_h, t_limit, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 2: SLSQP refinement
    rem = deadline - time.time()
    if rem > 50:
        h_opt, c5_opt = slsqp_polish(best_h, [30, 100, 500], 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 3: SLP again
    rem = deadline - time.time()
    if rem > 60:
        h_opt, c5_opt = slp_polish(best_h, min(rem - 10, 100), rng, initial_R=0.02)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 4: Targeted annealing
    rem = deadline - time.time()
    if rem > 120:
        at = min(rem - 40, 350)
        h_opt, c5_opt = targeted_anneal(best_h, at, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 5: SLP after annealing
    rem = deadline - time.time()
    if rem > 50:
        h_opt, c5_opt = slp_polish(best_h, min(rem - 10, 120), rng, initial_R=0.03)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 6: Pair swap search
    rem = deadline - time.time()
    if rem > 30:
        hp = int(min(rem * 2, 80))
        h_opt, c5_opt = pair_swap_search(best_h, hp)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 7: Try other resolutions
    rem = deadline - time.time()
    if rem > 60:
        n_cur = len(best_h)
        for n_try in [48, 54, 66, 72, 84, 42, 40]:
            if time.time() > deadline - 25:
                break
            if n_try == n_cur:
                continue
            h_interp = resample_h(best_h, n_cur, n_try)
            h_opt, c5_opt = slp_polish(h_interp, 25, rng, initial_R=0.03)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n_cur = n_try
            if time.time() > deadline - 25:
                break
    
    # Phase 8: Final SLP
    rem = deadline - time.time()
    if rem > 15:
        h_opt, c5_opt = slp_polish(best_h, min(rem - 3, 40), rng, initial_R=0.015)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 9: Final pair swap
    rem = deadline - time.time()
    if rem > 10:
        h_opt, c5_opt = pair_swap_search(best_h, 20)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    final_c5, _ = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    best_h = project_feasible(best_h, len(best_h))
    final_c5, _ = compute_c5(best_h)
    
    return best_h, final_c5, len(best_h)
