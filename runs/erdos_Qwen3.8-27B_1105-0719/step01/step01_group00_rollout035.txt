from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        mid = np.array([0.5])
        return np.concatenate([q, mid, 1.0 - q[::-1]])


def make_q_from_h(h, n):
    m = n // 2
    # For both even and odd n:
    # left = h[:m]
    # right_reversed = h[n-1:m-1:-1] (this gives h[n-1], h[n-2], ..., h[m] for even, or h[n-1], ..., h[m+1] for odd)
    q = 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])
    return np.clip(q, 0.0, 1.0)


def grad_q_lag(q, d, n, dx):
    """Gradient of F_k (with offset d) w.r.t. q, for symmetric h."""
    m = n // 2
    h = make_sym_h(q, n)
    i = np.arange(m, dtype=int)
    g = np.zeros(m)
    
    ip = i + d
    mask = (ip >= 0) & (ip < n)
    g[mask] += (1.0 - h[ip[mask]])
    
    im = i - d
    mask = (im >= 0) & (im < n)
    g[mask] -= h[im[mask]]
    
    j = n - 1 - i
    jp = j + d
    mask = (jp >= 0) & (jp < n)
    g[mask] -= (1.0 - h[jp[mask]])
    
    jm = j - d
    mask = (jm >= 0) & (jm < n)
    g[mask] += h[jm[mask]]
    
    return g * dx


def logsumexp_obj_grad(q, n, beta):
    """Smoothed max objective and gradient in q-space."""
    dx = 2.0 / n
    corrs = corr_values(make_sym_h(q, n))
    # Smoothed max: (1/beta) * log(sum(exp(beta * corrs)))
    val = (1.0 / beta) * logsumexp(beta * corrs)
    
    # Gradient
    shifted = beta * corrs - logsumexp(beta * corrs)
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n // 2)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        grad += w[k] * grad_q_lag(q, d, n, dx)
    return val, grad


def epigraph_solve(q0, n, active_lags, maxiter=500, ftol=1e-14):
    """Solve minimize t s.t. t >= F_k(q) for k in active_lags, 0<=q<=1."""
    m = n // 2
    dx = 2.0 / n
    h0 = make_sym_h(q0, n)
    corrs0 = corr_values(h0)
    t0 = corrs0.max()
    
    # Variables: [q_0, ..., q_{m-1}, t]
    y0 = np.concatenate([q0, [t0]])
    
    active_set = list(active_lags)
    
    def objective(y):
        return y[m]  # t
    
    def objective_grad(y):
        gy = np.zeros(m + 1)
        gy[m] = 1.0
        return gy
    
    def make_constraint(k):
        d = k - (n - 1)
        def con(y):
            q = y[:m]
            t = y[m]
            h = make_sym_h(q, n)
            # F_k = dx * sum valid h[i]*(1-h[i+d])
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                return t  # constraint is t >= 0, always satisfied
            idx = np.arange(i_lo, i_hi + 1)
            Fk = np.sum(h[idx] * (1.0 - h[idx + d])) * dx
            return t - Fk
        def con_jac(y):
            q = y[:m]
            g = grad_q_lag(q, d, n, dx)
            jac = np.zeros(m + 1)
            jac[:m] = -g
            jac[m] = 1.0
            return jac
        return {'type': 'ineq', 'fun': con, 'jac': con_jac}
    
    constraints = [make_constraint(k) for k in active_set]
    bounds = [(0.0, 1.0)] * m + [(0.0, 0.5)]
    
    try:
        result = minimize(objective, y0, jac=objective_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': ftol})
        if result.success or result.fun < t0:
            q_new = result.x[:m]
            q_new = np.clip(q_new, 0.0, 1.0)
            return q_new
    except Exception:
        pass
    return q0
