And #{i: h[i]=1, h[i+k]=0} = #{i: h[i+k]=1, h[i]=0} (reindex j=i+k: #{j: h[j]=1, h[j-k]=0} for k≤j<n).

So the positive and negative lag contributions are:
score(k) = dx * A(k) where A(k) = #{i: h[i]=1, h[i+k]=0} for k>0
score(-k) = dx * B(k) where B(k) = #{i: h[i]=1, h[i-k]=0} = #{i: h[i+k]=1, h[i]=0} for k>0

And A(k) + B(k) = #{i: h[i]≠h[i+k]} for 0≤i<n-k.

So we need both A(k) ≤ B and B(k) ≤ B, which means:
A(k) ≤ B AND #{i: h[i]≠h[i+k]} - A(k) ≤ B
→ #{i: h[i]≠h[i+k]} ≤ 2B AND A(k) ≤ B

The first condition (total transitions ≤ 2B) is necessary but not sufficient; we also need the "direction" to be balanced.

For the target C₅ = 0.38, n=140: B = floor(0.38 * 70) = floor(26.6) = 26.

    exp_prof = np.exp(temperature * (prof - prof.max()))
    smoothed = prof.max() + np.log(exp_prof.sum()) / temperature

    exp_weights = np.exp(temperature * (prof - prof.max()))
    weights = exp_weights / exp_weights.sum()

    # prof index m corresponds to shift delta = m - (n-1)

    #   d/dh[i] = dx * (1-h[i-d]) - dx * h[i+d] (if valid)

    # The gradient with respect to h[i]:

    # But this needs boundary handling. Let me just compute it numerically-free.
    
    # Simpler: use the formula that for the bilinear form sum h[i](1-h[i+k]):
    # ∂/∂h[j] = (1-h[j+k]) - h[j-k] (with the understanding that h[x]=0 for x out of range)

            #   If j <= n-1-delta: h[j] appears in the "h" role → contributes (1-h[j+delta])
            #   If j >= delta: h[j] appears in the "(1-h)" role (as h[j-delta] in the shifted) 
            #     Wait, h[j] appears as h[i+delta] when i = j-delta, so if j>=delta and j-delta <= n-1-delta:
            #     i.e., j >= delta and j <= n-1. Contradiction with first?
            #     j <= n-1-delta (first role) and j >= delta (second role). These can overlap!
            #     When both apply (delta <= j <= n-1-delta):
            #       contribution = (1-h[j+delta]) - h[j-delta]
            #     When only first (j < delta, but j <= n-1-delta):
            #       Wait, if j < delta, then j <= n-1-delta requires n-1-delta >= j >= 0, and delta > j.
            #       This means the second role doesn't apply (j-delta < 0).
            #     When only second (j > n-1-delta but j >= delta):
            #       j > n-1-delta means j+delta > n-1, so first role doesn't apply.
            
            # Let me just do it directly:
            for i in range(n - delta):
                # h[i] * (1-h[i+delta])
                # d/dh[i] from this term: (1-h[i+delta])
                # d/dh[i+delta] from this term: -h[i]

                grad[i] += w * (1.0 - h[i - d])

    grad -= grad.mean()  # Project to maintain sum
    
    # Step
    h_new = h - step_size * grad
    h_new = repair_feasible(h_new)
    return h_new

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def binary_sa(n, time_limit=3.0, seed=42):
    """Simulated annealing for binary sequences."""
    rng = np.random.RandomState(seed)
    target = n // 2
    h = np.zeros(n)
    h[rng.choice(n, target, replace=False)] = 1.0
    score = exact_score(h)
    best_h = h.copy()
    best_score = score
    T = 0.05  # initial temperature
    T_min = 0.001
    t0 = time.perf_counter()
    
    one_idx = np.where(h == 1.0)[0]
    zero_idx = np.where(h == 0.0)[0]
    
    while time.perf_counter() - t0 < time_limit:
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]
        
        h[i], h[j] = 0.0, 1.0
        new_score = exact_score(h)
        delta = new_score - best_score
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # Keep the move
            one_idx = np.where(h == 1.0)[0]
            zero_idx = np.where(h == 0.0)[0]
        else:
            # Revert
            h[i], h[j] = 1.0, 0.0
        
        T = max(T_min, T * 0.999)
    
    return best_h, best_score

def subgradient_optimize(h0, time_limit=10.0, temp_init=20.0, temp_final=200.0):
    """Subgradient descent with annealing temperature."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    step = 0.05
    
    while time.perf_counter() - t0 < time_limit:
        elapsed = time.perf_counter() - t0
        progress = elapsed / time_limit
        temp = temp_init + (temp_final - temp_init) * progress
        step_current = step * (1.0 - 0.5 * progress)
        
        prof = get_profile(h)
        exp_w = np.exp(temp * (prof - prof.max()))
        weights = exp_w / exp_w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        dx = 2.0 / n
        M = 2 * n - 1
        for m in range(M):
            w = weights[m] * dx
            if w < 1e-14:
                continue
            delta = m - (n - 1)
            if delta == 0:
                grad += w * (1.0 - 2.0 * h)
            elif delta > 0:
                grad[:n-delta] += w * (1.0 - h[delta:n])
                grad[delta:] += w * (-h[:n-delta])
            else:
                d = -delta
                grad[d:] += w * (1.0 - h[:n-d])
                grad[:n-d] += w * (-h[d:n])
        
        grad -= grad.mean()
        
        h_new = repair_feasible(h - step_current * grad)
        new_score = exact_score(h_new)
        
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
            h = h_new
        else:
            # Annealing: accept with some probability
            if np.random.random() < np.exp(-(new_score - best_score) / (step_current * 0.1)):
                h = h_new
            # else keep h
        
        # Periodically kick
        if int(progress * 20) > int((progress - 0.05) * 20):
            h = repair_feasible(h + 0.01 * np.random.randn(n))
    
    return best_h, best_score
