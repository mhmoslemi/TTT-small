    max_c = corrs.max()

    active = np.where(corrs >= threshold)[0]
    
    # Compute gradient: for each active lag k, add the subgradient
    grad = np.zeros(n)
    for k in active:
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx + offset] += (1.0 - h[idx])
            grad[idx] += (-h[idx + offset])
    grad /= len(active)
    grad *= dx
    return grad, max_c * dx

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def gradient_smooth_max(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj_val(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def obj_grad(x):
        return gradient_smooth_max(x, dx, beta)
    
    def cons_fun(x):
        return x.sum() - target_sum
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_val, h0, jac=obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def subgradient_descent(h0, dx, target_sum, max_iter, alpha_init, tol):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha_init
    no_improve = 0
    
    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
        
        threshold = max_c * 0.97
        active = np.where(corrs >= threshold)[0]
        
        grad = np.zeros(n)
        for k in active:
            grad += grad_corr_k(h, k, n)
        grad /= len(active)
        grad *= dx
        
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < c5_val:
            h = h_new
            alpha = min(alpha * 1.01, 0.5)
        else:
            alpha *= 0.97
        
        if alpha < tol:
            break
    
    return best_h, best_c5


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    if n_ones > 0:
        h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def binary_local_search(h0, dx, target_sum, rng, max_iter=2000, n_trials_per_iter=500):
    n = len(h0)
    h = np.where(h0 > 0.5, 1.0, 0.0).astype(float)
    if abs(h.sum() - target_sum) > 1e-10:
        h = make_binary_start(n, target_sum, rng)
    
    best_c5 = c5_of_h(h, dx)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iter):
        improved = False
        for _ in range(n_trials_per_iter):
            if len(ones) == 0 or len(zeros) == 0:
                break
            oi = rng.randint(len(ones))
            zi = rng.randint(len(zeros))
            o = ones[oi]

            h[o] = 0.0

                zeros = np.where(h == 0.0)[0]
                improved = True
            else:
                h[o] = 1.0
                h[z] = 0.0
        
        if not improved:

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def gradient_smooth_max(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


class SLSQPProblem:
    def __init__(self, dx, beta, target_sum):
        self.dx = dx
        self.beta = beta
        self.target_sum = target_sum

    def obj(self, x):
        corrs = compute_corrs(x)
        return logsumexp(self.beta * corrs) / self.beta * self.dx

    def grad(self, x):
        return gradient_smooth_max(x, self.dx, self.beta)

    def con_fun(self, x):
        return x.sum() - self.target_sum

    def con_jac(self, x):
        return np.ones(len(x))


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    prob = SLSQPProblem(dx, beta, target_sum)
    cons = [{'type': 'eq', 'fun': prob.con_fun, 'jac': prob.con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(prob.obj, h0, jac=prob.grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def subgradient_descent(h0, dx, target_sum, max_iter, alpha_init, tol, threshold_frac=0.97):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha_init
    no_improve = 0

    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            break

        threshold = max_c * threshold_frac
        active = np.where(corrs >= threshold)[0]

        grad = np.zeros(n)
        for k in active:
            grad += grad_corr_k(h, k, n)
        grad /= len(active)
        grad *= dx

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val:
            h = h_new
            alpha = min(alpha * 1.01, 0.5)

            alpha *= 0.97

    return best_h, best_c5
