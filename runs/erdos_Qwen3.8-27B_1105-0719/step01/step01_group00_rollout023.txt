    return best_h, best_c5


def make_thue_morse(n):
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t


def balance_binary(h, n):
    target = n / 2.0
    n_ones = int(h.sum() + 0.5)
    if n_ones > int(target):
        flip = n_ones - int(target)
        ones = np.where(h == 1)[0]
        if len(ones) >= flip:
            idx = np.random.RandomState(0).choice(ones, size=flip, replace=False)
            h[idx] = 0
    elif n_ones < int(target):
        flip = int(target) - n_ones
        zeros = np.where(h == 0)[0]
        if len(zeros) >= flip:
            idx = np.random.RandomState(0).choice(zeros, size=flip, replace=False)
            h[idx] = 1
    return h


def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def remaining():
        return budget_s - (time.time() - t0)

    def try_optimize_full(h0, n, time_limit_s=60):
        nonlocal best_h, best_c5
        dx = 2.0 / n
        ts = n / 2.0
        h0 = project_feasible(h0, ts)
        c5_0 = c5_of_h(h0, dx)
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()
        if remaining() < 5:
            return
        t_start = time.time()
        h_work = h0
        betas_phase1 = [30, 60, 120, 250, 500]
        for beta in betas_phase1:
            if time.time() - t_start > time_limit_s * 0.6:
                break
            if remaining() < 10:
                break
            h_opt, c5_opt = optimize_lv(h_work, dx, ts, beta, 250, [200, 2000])
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            if c5_opt < c5_of_h(h_work, dx):
                h_work = h_opt
        if remaining() > 15 and time.time() - t_start < time_limit_s * 0.8:
            h_f, c5_f = fista_smooth(h_work, dx, ts, 500, 3000, 0.5 / 500.0)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
                h_work = h_f
        if remaining() > 10 and time.time() - t_start < time_limit_s * 0.9:
            h_opt, c5_opt = optimize_lv(h_work, dx, ts, 1000, 300, [5000])
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_work = h_opt
        if remaining() > 10 and time.time() - t_start < time_limit_s:
            h_sg, c5_sg = subgrad_active(h_work, dx, ts, 3000, 0.1)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Phase 1: Optimize the given initial construction
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    try_optimize_full(h_init, n_init, 200)

    # Phase 2: Try Thue-Morse and other structured starts at various n
    n_candidates = [32, 48, 64, 96, 128, 160]
    for n_try in n_candidates:
        if remaining() < 30:
            break
        tm = make_thue_morse(n_try)
        tm = balance_binary(tm, n_try)
        tm = project_feasible(tm, n_try / 2.0)
        try_optimize_full(tm, n_try, 40)

    # Phase 3: Random binary starts
    for n_try in [64, 96, 128]:
        if remaining() < 30:
            break
        for trial in range(5):
            if remaining() < 15:
                break
            h_rand = (rng.rand(n_try) < 0.5).astype(float)
            h_rand = balance_binary(h_rand, n_try)
            h_rand = project_feasible(h_rand, n_try / 2.0)
            try_optimize_full(h_rand, n_try, 30)

    # Phase 4: Interpolate best to other n values
    if best_h is not None:
        for n_try in [48, 64, 80, 96, 128, 160, 192]:
            if remaining() < 25:
                break
            if n_try == len(best_h):
                continue
            h_interp = interp_to_n(best_h, n_try)
            try_optimize_full(h_interp, n_try, 35)

    # Phase 5: Refine the global best with higher precision
    if best_h is not None and remaining() > 40:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_ref, c5_ref = optimize_lv(best_h, dx_b, ts_b, 2000, 400, [10000])
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
        if remaining() > 20:
            h_f, c5_f = fista_smooth(best_h, dx_b, ts_b, 2000, 10000, 0.3 / 2000.0)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f
        if remaining() > 15:
            h_sg, c5_sg = subgrad_active(best_h, dx_b, ts_b, 5000, 0.05)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg
        if remaining() > 10:
            h_final, c5_final = optimize_lv(best_h, dx_b, ts_b, 5000, 300, [50000])
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final

    # Phase 6: Try perturbing the best solution
    if best_h is not None and remaining() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        for pert_scale in [0.05, 0.1, 0.2]:
            if remaining() < 15:
                break
            h_pert = best_h + rng.randn(n_b) * pert_scale
            h_pert = project_feasible(h_pert, ts_b)
            h_opt, c5_opt = optimize_lv(h_pert, dx_b, ts_b, 500, 200, [1000, 10000])
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Final evaluation
    if best_h is None:
        best_h = h_init
        best_c5 = c5_of_h(h_init, 2.0 / n_init)

    final_c5 = c5_of_h(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
