            if curr_c < best_c:

            b[j] = 0.0

import numpy as np

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -max(h) - 1.0, 1.0 - min(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    for _ in range(200):
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + residual / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + residual / cnt)
        residual = target - np.sum(h)
    return np.clip(h, 0.0, 1.0)


def transfer_to_size(h_src, n_tgt):

    h_new = np.interp(xt, xs, h_src)

def profile_grad(h, row_idx):
    n = len(h)
    dx = 2.0 / n
    s = row_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        a = max(0, 0)
        b = min(n, n - s)
        idx = np.arange(a, b)
        grad[idx] += dx * (1.0 - h[idx + s]) if s < n else 0
        if s > 0:
            idx2 = np.arange(0, n - s)
            grad[idx2 + s] -= dx * h[idx2]
    else:
        a = -s
        if a > 0:
            idx = np.arange(0, n - a)
            grad[idx] += dx * (1.0 - h[idx + a])
            idx2 = np.arange(a, n)
            grad[idx2] -= dx * h[idx2 - a]
    return grad


def lse_obj_and_grad(h, tau):
    n = len(h)
    profile = exact_profile(h)
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    dx = 2.0 / n
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            if s < n:
                idx = np.arange(0, n - s)
                grad[idx] += w * dx * (1.0 - h[idx + s])
            if s > 0:
                idx2 = np.arange(0, n - s)
                grad[idx2 + s] -= w * dx * h[idx2]
        else:
            a = -s
            if a > 0 and a < n:
                idx = np.arange(0, n - a)
                grad[idx] += w * dx * (1.0 - h[idx + a])
                idx2 = np.arange(a, n)
                grad[idx2] -= w * dx * h[idx2 - a]
    return obj, grad


def slsqp_optimize(h0, tau, maxiter=300, deadline=None):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj_fun(x):
        return lse_obj_and_grad(x, tau)[0]

    def grad_fun(x):
        return lse_obj_and_grad(x, tau)[1]

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(len(x))

    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    h_init = h0.copy()
    try:
        res = minimize(obj_fun, h_init, method='SLSQP', jac=grad_fun,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if deadline is not None and time.time() > deadline - 5:
            return h0
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        if np.all(np.isfinite(h_new)):
            return h_new
    except Exception:
        pass
    return h0.copy()

    if not np.isfinite(curr_c):

    taus = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0003, 0.0001]

        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c

def make_binary_from_continuous(h_cont):

def make_seed(n, rng, mode, h_parent=None):

        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)

    elif mode == 3:

    elif mode == 5:

    elif mode == 6:

                scored.append((sc, c))
        scored.sort(key=lambda x: x[0])

        n_top = min(4, len(scored))
        for idx in range(n_top):
            if time.time() > deadline - 15:
                break
            h0 = scored[idx][1]

            # Continuous refinement
            h_ref, c_ref = continuous_refine(h0, deadline)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
                best_n = n

            # Binary search from continuous
            if n % 2 == 0 and time.time() < deadline - 10:
                b = make_binary_from_continuous(h_ref)
                if b is not None:
                    c_b = exact_objective(b)
                    if c_b < best_c - 1e-12:
                        best_c = c_b
                        best_h = b.copy()
                        best_n = n
                    b, c_b = binary_greedy_swap(b, max_rounds=200, deadline=deadline)
                    if c_b < best_c - 1e-12:
                        best_c = c_b
                        best_h = b.copy()
                        best_n = n
                    if time.time() < deadline - 8:
                        b, c_b = binary_pair_swap(b, deadline, n_trials=500)
                        if c_b < best_c - 1e-12:
                            best_c = c_b
                            best_h = b.copy()
                            best_n = n
                    if time.time() < deadline - 6:
                        b, c_b = binary_anneal(b, deadline, n_steps=5000)
                        if c_b < best_c - 1e-12:
                            best_c = c_b
                            best_h = b.copy()
                            best_n = n
                        b, c_b = binary_greedy_swap(b, max_rounds=100, deadline=deadline)
                        if c_b < best_c - 1e-12:
                            best_c = c_b
                            best_h = b.copy()
                            best_n = n

    # Phase 3: Refine the global best further
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            b, c_b = binary_greedy_swap(best_h.copy(), max_rounds=300, deadline=deadline)
            if c_b < best_c - 1e-12:
                best_c = c_b
                best_h = b.copy()
            if time.time() < deadline - 8:
                b, c_b = binary_pair_swap(b, deadline, n_trials=1000)
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()
            if time.time() < deadline - 5:
                b, c_b = binary_anneal(b, deadline, n_steps=10000)
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()
                b, c_b = binary_greedy_swap(b, max_rounds=200, deadline=deadline)
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()
            # Try continuous from binary
            if time.time() < deadline - 5:
                h_cont = best_h.astype(np.float64).copy()
                h_cont = h_cont + rng.normal(0, 0.05, n_best)
                h_cont = project_box_sum(h_cont)
                h_ref, c_ref = continuous_refine(h_cont, deadline)
                if c_ref < best_c - 1e-12:
                    best_c = c_ref
                    best_h = h_ref.copy()
        else:
            h_ref, c_ref = continuous_refine(best_h.copy(), deadline)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
            # Binarize and refine
            if n_best % 2 == 0 and time.time() < deadline - 8:
                b = make_binary_from_continuous(h_ref)
                if b is not None:
                    b, c_b = binary_greedy_swap(b, max_rounds=200, deadline=deadline)
                    if c_b < best_c - 1e-12:
                        best_c = c_b
                        best_h = b.copy()
                    if time.time() < deadline - 5:
                        b, c_b = binary_anneal(b, deadline, n_steps=8000)
                        if c_b < best_c - 1e-12:
                            best_c = c_b
                            best_h = b.copy()
                        b, c_b = binary_greedy_swap(b, max_rounds=150, deadline=deadline)
                        if c_b < best_c - 1e-12:
                            best_c = c_b
                            best_h = b.copy()

    # Phase 4: Try transferring best to nearby sizes and refining
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        for dn in [-2, 2, -4, 4]:
            n_try = n_best + dn
            if n_try < 24 or n_try > 100 or n_try % 2 != 0:
                continue
            if time.time() > deadline - 8:
                break
            h_try = transfer_to_size(best_h, n_try)
            h_ref, c_ref = continuous_refine(h_try, deadline)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
                best_n = n_try
            if n_try % 2 == 0 and time.time() < deadline - 5:
                b = make_binary_from_continuous(h_ref)
                if b is not None:
                    b, c_b = binary_greedy_swap(b, max_rounds=100, deadline=deadline)
                    if c_b < best_c - 1e-12:
                        best_c = c_b
                        best_h = b.copy()
                        best_n = n_try

    # Phase 5: Final polish
    if best_h is not None and time.time() < deadline - 3:
        if not np.all((best_h == 0.0) | (best_h == 1.0)):
            n_final = len(best_h)
            h_polish = slsqp_optimize(best_h, 0.0001, maxiter=200, deadline=deadline)
            c_polish = exact_objective(h_polish)
            if c_polish < best_c - 1e-12:
                best_c = c_polish
                best_h = h_polish.copy()

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    if not np.isfinite(best_c) or not np.all(np.isfinite(best_h)):
        best_h = np.zeros(48)
        best_h[:24] = 1.0
        best_c = exact_objective(best_h)
        best_n = 48

    return best_h, best_c, best_n
