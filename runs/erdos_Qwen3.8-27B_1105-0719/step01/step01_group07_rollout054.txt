import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * dx
    return float(np.max(profile))


def full_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * dx


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def to_binary(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def random_binary(n, rng):
    m = n // 2
    idx = rng.choice(n, size=m, replace=False)
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

        if deadline is not None and time.time() > deadline - 2:
            break
        best_improve = 1e-14
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                impr = curr_c - c
                if impr > best_improve:
                    best_improve = impr
                    best_i, best_j = i, j
        if best_i >= 0:
            b[best_i] = 0.0
            b[best_j] = 1.0
            curr_c -= best_improve
            ones.remove(best_i)
            zeros.remove(best_j)
            zeros.append(best_i)
            ones.append(best_j)
        else:
            break
    return b, curr_c


def double_swap_optimize(b, max_rounds=30, deadline=None):
    n = len(b)
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline - 3:
            break
        best_improve = 1e-14
        best_move = None
        for a in range(len(ones)):
            for bidx in range(a + 1, len(ones)):
                i1, i2 = ones[a], ones[bidx]
                for c in range(len(zeros)):
                    for d in range(c + 1, len(zeros)):
                        j1, j2 = zeros[c], zeros[d]
                        b[i1] = 0.0
                        b[i2] = 0.0
                        b[j1] = 1.0
                        b[j2] = 1.0
                        cv = exact_objective(b)
                        b[i1] = 1.0
                        b[i2] = 1.0
                        b[j1] = 0.0
                        b[j2] = 0.0
                        impr = curr_c - cv
                        if impr > best_improve:
                            best_improve = impr
                            best_move = (i1, i2, j1, j2)
        if best_move is not None:
            i1, i2, j1, j2 = best_move
            b[i1] = 0.0
            b[i2] = 0.0
            b[j1] = 1.0
            b[j2] = 1.0
            curr_c -= best_improve
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b, curr_c


def triple_swap_optimize(b, max_rounds=10, deadline=None):
    n = len(b)
    curr_c = exact_objective(b)
    rng_local = np.random.RandomState(12345)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline - 3:
            break
        best_improve = 1e-14
        best_move = None
        n_trials = 2000
        for _t in range(n_trials):
            i_pick = rng_local.choice(ones, size=3, replace=False)
            j_pick = rng_local.choice(zeros, size=3, replace=False)
            for i in i_pick:
                b[i] = 0.0
            for j in j_pick:
                b[j] = 1.0
            cv = exact_objective(b)
            for i in i_pick:
                b[i] = 1.0
            for j in j_pick:
                b[j] = 0.0
            impr = curr_c - cv
            if impr > best_improve:
                best_improve = impr
                best_move = (tuple(i_pick), tuple(j_pick))
        if best_move is not None:
            i_pick, j_pick = best_move
            for i in i_pick:
                b[i] = 0.0
            for j in j_pick:
                b[j] = 1.0
            curr_c -= best_improve
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b, curr_c


def simulated_annealing(b, deadline, n_iter=500000, t_start=0.01, t_end=0.0001):
    n = len(b)
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    rng = np.random.RandomState(999)
    n_iter = int(n_iter)
    for step in range(n_iter):
        if time.time() > deadline - 5:
            break
        alpha = step / max(1, n_iter)
        T = t_start * (t_end / t_start) ** alpha
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        move_type = rng.rand()
        if move_type < 0.6:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
        elif move_type < 0.9:
            if len(ones) < 2 or len(zeros) < 2:
                break
            i_pick = rng.choice(ones, size=2, replace=False)
            j_pick = rng.choice(zeros, size=2, replace=False)
            for i in i_pick:
                b[i] = 0.0
            for j in j_pick:
                b[j] = 1.0
        else:
            if len(ones) < 3 or len(zeros) < 3:
                break
            i_pick = rng.choice(ones, size=3, replace=False)
            j_pick = rng.choice(zeros, size=3, replace=False)
            for i in i_pick:
                b[i] = 0.0
            for j in j_pick:
                b[j] = 1.0
        new_c = exact_objective(b)
        delta = new_c - curr_c
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            curr_c = new_c
            if curr_c < best_c - 1e-14:
                best_c = curr_c
                best_b = b.copy()
        else:
            if move_type < 0.6:
                b[i] = 1.0
                b[j] = 0.0
            elif move_type < 0.9:
                for i in i_pick:
                    b[i] = 1.0
                for j in j_pick:
                    b[j] = 0.0
            else:
                for i in i_pick:
                    b[i] = 1.0
                for j in j_pick:
                    b[j] = 0.0
    return best_b, best_c


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    c2n = 2 * n - 1
    for j in range(c2n):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))


def slsqp_optimize(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()


def refine_continuous(h0, deadline):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_objective(h)
    taus = [0.01, 0.005, 0.002, 0.0008, 0.0003, 0.0001]
    for tau in taus:
        if time.time() > deadline - 8:
            break
        h_new = slsqp_optimize(h, tau, maxiter=min(300, n * 3 + 30))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_structured_binary(n, rng, mode):
    if mode == 0:
        b = np.zeros(n)
        b[:n // 2] = 1.0
        b = b[rng.permutation(n)]
    elif mode == 1:
        b = np.zeros(n)
        block = max(2, n // 12)
        for i in range(0, n, 2 * block):
            b[i:i + block] = 1.0
        b = project_box_sum(b)
        b = to_binary(b)
    elif mode == 2:
        b = np.zeros(n)
        for i in range(0, n, 3):
            b[i] = 1.0
        b = to_binary(b)
    elif mode == 3:
        b = random_binary(n, rng)
    elif mode == 4:
        b = np.zeros(n)
        step = max(2, n // 10)
        for i in range(0, n, step):
            b[i] = 1.0
        b = to_binary(b)
    else:
        b = random_binary(n, rng)
    if b is None:
        b = random_binary(n, rng)
    return b
