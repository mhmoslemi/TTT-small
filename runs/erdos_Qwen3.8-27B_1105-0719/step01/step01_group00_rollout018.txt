def find_1d_minimax(a, b, x_lo, x_hi):
    """Find x in [x_lo, x_hi] minimizing max_i (a[i]*x + b[i])"""
    # Binary search for where the slope of the upper envelope changes sign
    # At a given x, the "subgradient" is the set of slopes of active lines
    # If all active lines have negative slope, decrease x
    # If all active lines have positive slope, increase x
    # Otherwise, x is optimal (or on flat region)
    
    lo, hi = x_lo, x_hi
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = a * mid + b
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        if active.sum() == 0:
            break
        slopes = a[active]
        if slopes.min() >= 0:
            # All active slopes non-negative, minimum is at or before mid
            hi = mid
        elif slopes.max() <= 0:
            # All active slopes non-positive, minimum is at or after mid
            lo = mid
        else:
            # Mixed slopes at mid, mid is optimal (within tolerance)
            hi = mid
            lo = mid
            break
    
    x_opt = (lo + hi) / 2
    # Check endpoints too
    candidates = [x_lo, x_hi, x_opt]
    best_x = min(candidates, key=lambda x: (a*x+b).max())
    return best_x

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break
        threshold = max_c * 0.97
        active = np.where(c >= threshold)[0]
        grad = np.zeros(n)
        for k in active:
            grad += grad_corr_k(h, k, n)
        grad /= len(active)
        grad *= dx
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_val:
            h = h_new
            alpha = min(alpha * 1.01, 0.3)
        else:
            alpha *= 0.95
        if alpha < tol:
            break
    return best_h, best_c5


class SlsqpObj:
    def __init__(self, dx, beta):
        self.dx = dx
        self.beta = beta

    def value(self, x):
        return smooth_max_c5(x, self.dx, self.beta)

    def gradient(self, x):
        return gradient_smooth_max(x, self.dx, self.beta)


class SlsqpCon:
    def __init__(self, target_sum):
        self.target_sum = target_sum

    def fun(self, x):
        return x.sum() - self.target_sum

    def jac(self, x):
        return np.ones(len(x))


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    objf = SlsqpObj(dx, beta)
    cf = SlsqpCon(target_sum)
    cons = [{'type': 'eq', 'fun': cf.fun, 'jac': cf.jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objf.value, h0, jac=objf.gradient,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    if n_ones > 0:
        h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def pair_optimize(h, i, j, n, corrs, d_range, n_grid=100):
    c = h[i] + h[j]
    t_lo = max(0.0, c - 1.0)
    t_hi = min(1.0, c)
    if t_hi - t_lo < 1e-13:
        return h[i], h[j], False
    v = h[i]
    m1 = (d_range >= -i) & (d_range <= n - 1 - i)
    m2 = (d_range >= i - (n - 1)) & (d_range <= i)
    m3 = (d_range >= -j) & (d_range <= n - 1 - j)
    m4 = (d_range >= j - (n - 1)) & (d_range <= j)
    G = np.zeros(2 * n - 1)
    G[m1] += -h[i + d_range[m1]]
    G[m2] += (1.0 - h[i - d_range[m2]])
    G[m3] += h[j + d_range[m3]]
    G[m4] += (1.0 - h[j - d_range[m4]])
    G[n - 1] = 0.0
    r0_base = corrs[n - 1] - v * (1.0 - v) - (c - v) * (1.0 - c + v)
    t_grid = np.linspace(t_lo, t_hi, n_grid)
    linear = corrs[None, :] + np.outer(t_grid - v, G)
    r0 = t_grid * (1.0 - t_grid) + (c - t_grid) * (1.0 - c + t_grid) + r0_base
    linear[:, n - 1] = r0
    f_vals = linear.max(axis=1)
    bi = int(f_vals.argmin())
    t_best = t_grid[bi]
    if 0 < bi < n_grid - 1:
        tr = np.linspace(t_grid[bi - 1], t_grid[bi + 1], 150)
        lr = corrs[None, :] + np.outer(tr - v, G)
        r0r = tr * (1.0 - tr) + (c - tr) * (1.0 - c + tr) + r0_base
        lr[:, n - 1] = r0r
        fr = lr.max(axis=1)
        t_best = tr[int(fr.argmin())]
    h[i] = t_best
    h[j] = c - t_best
    return t_best, c - t_best, True


def coordinate_descent(h0, dx, target_sum, n, max_time, rng, corrs=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    d_range = np.arange(-(n - 1), n)
    if corrs is None:
        corrs = compute_corrs(h)
    t0 = time.time()
    imp_tol = 1e-14
    sweep_no_improve = 0
    n_grid = 100
    sweep = 0
    while time.time() - t0 < max_time:
        sweep += 1
        sweep_best_c5 = best_c5
        order = np.random.permutation(n) if rng is not None else np.arange(n)
        for pos in range(n):
            if time.time() - t0 > max_time:
                break
            i = int(order[pos])
            c_i = h[i]
            if c_i < 1e-8 or c_i > 1.0 - 1e-8:
                candidates = [k for k in range(n) if k != i]
                if not candidates:
                    continue
                if rng is not None:
                    j = int(rng.choice(candidates))
                else:
                    j = (i + 1) % n
            else:
                others = [(k, abs(h[k] - 0.5)) for k in range(n) if k != i]
                if not others:
                    continue
                if rng is not None and len(others) > 20:
                    sample = rng.choice(len(others), size=min(20, len(others)), replace=False)
                    others = [others[k] for k in sample]
                j = int(min(others, key=lambda x: x[1])[0])
            t_new, j_new, changed = pair_optimize(h, i, j, n, corrs, d_range, n_grid)
            if not changed:
                continue
            corrs = compute_corrs(h)
            c5_val = corrs.max() * dx
            if c5_val < best_c5 - imp_tol:
                best_c5 = c5_val
                best_h = h.copy()
    return best_h, best_c5


def trust_epigraph(h_center, S_center, dx, target_sum, active_lags, corrs_current,
                   rho, alpha_h, maxiter):
    n = len(h_center)
    d_range = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1
    G_mat = np.zeros((len(active_lags), n))
    R0_vec = np.zeros(len(active_lags))
    for ai, r in enumerate(active_lags):
        d = d_range[r]
        R0_vec[ai] = corrs_current[r]
        b_lo = max(0, -d)
        b_hi = min(n - 1, n - 1 - d)
        if b_lo > b_hi:
            continue
        b_idx = np.arange(b_lo, b_hi + 1)
        s_idx = b_idx + d
        G_mat[ai, s_idx] += (1.0 - h_center[b_idx])
        G_mat[ai, b_idx] += (-h_center[s_idx])

    lower_h = np.maximum(0.0, h_center - rho)
    upper_h = np.minimum(1.0, h_center + rho)
    Gc = np.ones(n)
    Gc[n - 1] = 2.0
    A_mat = np.zeros((n, n_lags))
    A_mat[:, n - 1] = -dx
    for ai, r in enumerate(active_lags):
        A_mat[:, r] += dx * G_mat[ai]
    b_vec = np.zeros(n_lags)
    b_vec[n - 1] = dx * S_center
    for ai in range(len(active_lags)):
        b_vec[active_lags[ai]] = -(dx * R0_vec[ai])

    def obj(z):
        h_z = z[:n]
        t_z = z[n]
        return t_z + alpha_h * np.sum((h_z - h_center) ** 2)

    def obj_grad(z):
        h_z = z[:n]
        g = np.zeros(n + 1)
        g[:n] = 2.0 * alpha_h * (h_z - h_center)
        g[n] = 1.0
        return g

    def cn_eq(z):
        return z[:n].sum() - target_sum

    def cn_eq_jac(z):
        return np.concatenate([Gc, np.zeros(1)])

    def c_ineq(z):
        h_z = z[:n]
        t_z = z[n]
        linear = A_mat.dot(h_z) - b_vec
        q = np.maximum(0.0, A_mat.dot(np.abs(h_z - h_center))) * dx
        vals = linear + q
        return t_z - np.maximum(vals, 0.0)

    upper_t = S_center * 1.5 + 0.5
    bounds_list = [(lower_h[k], upper_h[k]) for k in range(n)]
    bounds_list.append((0.0, upper_t))
    cons_list = [
        {'type': 'eq', 'fun': cn_eq, 'jac': cn_eq_jac},
        {'type': 'ineq', 'fun': c_ineq, 'jac': None},
    ]
    z0 = np.zeros(n + 1)
    z0[:n] = h_center
    z0[n] = S_center
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds_list, constraints=cons_list,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.x is not None:
            h_candidate = project_feasible(res.x[:n], target_sum)
            c5_val = c5_of_h(h_candidate, dx)
            if c5_val < S_center - 1e-14:
                return h_candidate, c5_val
    except Exception:
        pass
    return None, None
