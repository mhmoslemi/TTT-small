import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y to [0,1]^n with sum = n/2"""
    target = n / 2.0
    # Bisection to find lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute residual among interior points
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            h[interior] += residual / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def sa_optimize(h0, n, t_start, t_end, steps_per_sec, rng):
    """Simulated annealing with exact objective"""
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    n_total = int((t_end - t_start) * steps_per_sec)
    T0 = 0.02
    Tf = 1e-7
    
    t = t_start
    step_count = 0
    no_improve = 0
    
    while step_count < n_total and t < t_end:
        # Temperature
        frac = step_count / max(1, n_total)
        T = T0 * (Tf / T0) ** frac
        
        # Generate move
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Transfer from j to i
        cap = min(1.0 - h[i], h[j])
        if cap < 1e-12:
            # Try reverse
            cap_r = min(1.0 - h[j], h[i])
            if cap_r < 1e-12:
                continue
            # Transfer from i to j
            delta = cap_r * rng.uniform(0.1, 1.0)
            h_new = h.copy()
            h_new[j] += delta
            h_new[i] -= delta
        else:
            delta = cap * rng.uniform(0.1, 1.0)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        
        c5_new = compute_c5(h_new)
        delta_e = c5_new - c5
        
        if delta_e < 0 or rng.uniform() < math.exp(-delta_e / T):
            h = h_new
            c5 = c5_new
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

        if step_count % 10000 == 0:
            t = time.time()
            # Adjust remaining steps if behind schedule
            remaining_time = t_end - t
            if remaining_time < 0.5:
                break
    
    return best_h, best_c5

- m = i: valid if max(0,lag) ≤ i ≤ min(n-1, n-1+lag). Contributes h[i]*h[i-lag] = τ*h[i-lag] (was h[i]*h[i-lag])

- m - lag = i, i.e., m = i+lag: valid if max(0,lag) ≤ i+lag ≤ min(n-1, n-1+lag), i.e., i ≥ 0 and i+lag < n (roughly). Contributes h[i+lag]*h[i] = h[i+lag]*τ
  Change: h[i+lag]*(τ - h[i])
- Similarly for j:
  m = j: change = (s-τ - h[j])*h[j-lag] = (h[i]-τ)*h[j-lag] = -(τ-h[i])*h[j-lag]
  m = j+lag: change = h[j+lag]*(s-τ - h[j]) = h[j+lag]*(h[i]-τ) = -h[j+lag]*(τ-h[i])

Total change in ac[k] = (τ - h[i]) * [h[i-lag] + h[i+lag] - h[j-lag] - h[j+lag]]
(only including terms where the indices are valid)

So: Δac[k] = (τ - h[i]) * A_k

where A_k = h[i-lag] + h[i+lag] - h[j-lag] - h[j+lag] (with terms set to 0 if index out of bounds)

And: ac_new[k] = ac[k] + (τ - h[i]) * A_k

This is linear in τ! Great.

- For lag = k - (n-1), A_k = (h[i-lag] if valid else 0) + (h[i+lag] if valid else 0) - (h[j-lag] if valid else 0) - (h[j+lag] if valid else 0)

```python
def compute_A_vector(h, i, j, n):
    """Compute A_k for all k=0,...,2n-2"""
    lags = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to (n-1)
    A = np.zeros(2*n-1)
    
    # h[i - lag] is valid when 0 <= i - lag < n
    valid_il = (i - lags >= 0) & (i - lags < n)
    A[valid_il] += h[i - lags[valid_il]]
    
    # h[i + lag] is valid when 0 <= i + lag < n
    valid_ipl = (i + lags >= 0) & (i + lags < n)
    A[valid_ipl] += h[i + lags[valid_ipl]]
    
    # h[j - lag]
    valid_jl = (j - lags >= 0) & (j - lags < n)
    A[valid_jl] -= h[j - lags[valid_jl]]
    
    # h[j + lag]
    valid_jpl = (j + lags >= 0) & (j + lags < n)
    A[valid_jpl] -= h[j + lags[valid_jpl]]
    
    return A
```
