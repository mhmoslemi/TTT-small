        G_h -= np.mean(G_h)

        G_norm = np.linalg.norm(G_h)
        if G_norm < 1e-15:
            break

        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003]:

                if score < best_score - 1e-14:

                    best_score = score
                    improved = True
                    break

def interpolate_to_dimension(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    # Cumulative sum for piecewise constant
    cumsum = np.concatenate([[0.0], np.cumsum(h_src) * dx_src])
    # Target cell edges
    x_src_edges = np.arange(n_src + 1) * dx_src
    x_tgt_edges = np.arange(n_tgt + 1) * (2.0 / n_tgt)
    # Interpolate cumulative values
    interp_vals = np.interp(x_tgt_edges, x_src_edges, cumsum)
    h_tgt = np.diff(interp_vals) / (2.0 / n_tgt)
    return repair_feasible(h_tgt)

    return (h + h[::-1]) / 2.0
