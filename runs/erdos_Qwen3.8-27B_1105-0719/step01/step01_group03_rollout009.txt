    h = h_init.copy()
    best_c = exact_score(h)
    for tau in tau_schedule:
        for _ in range(max_iter):
            profile = compute_profile(h)
            c = max(profile)
            if c < best_c:
                best_c = c
                h_best = h.copy()
            # Compute weights
            weights = softmax(profile / tau)  # or: exp((profile - max)/tau) / sum
            # Compute gradient

def fw_vertex(grad, n):
    """Find vertex of {h: 0<=h_i<=1, sum(h)=n/2} minimizing grad·h."""
    target = n / 2.0
    # Sort by grad ascending
    order = np.argsort(grad)
    v = np.zeros(n)
    # Fill 1s for smallest grad
    n_ones = int(np.floor(target))
    # If target is not integer, one entry gets fractional
    frac = target - n_ones
    if n_ones > 0:
        v[order[:n_ones]] = 1.0
    if frac > 1e-12:
        v[order[n_ones]] = frac
    return v
