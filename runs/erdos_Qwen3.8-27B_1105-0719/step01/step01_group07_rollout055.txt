def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def profile(h):

def lse_objective_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(prof))
    shifted = np.minimum((prof - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    
    # Gradient computation
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            for k in range(n - s):
                grad[k + s] += w * dx * (1.0 - h[k])
                grad[k] -= w * dx * h[k + s]
        else:
            a = -s
            for k in range(a, n):
                grad[k] += w * dx * (1.0 - h[k + a])
                grad[k + a] -= w * dx * h[k]
    return obj, grad

class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_objective_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_objective_grad(x, self.tau)[1]

class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))

def slsqp_refine(h0, tau, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_box_sum(np.array(res.x, dtype=np.float64))
    except Exception:
        return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(prof))


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def profile_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def lse_grad_vectorized(h, tau):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(prof))
    shifted = np.minimum((prof - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    # Vectorized gradient computation
    # For each position p, grad[p] gets contributions from various lags
    # Using the formula derived from the correlation structure
    for p in range(n):
        # k=0 contribution
        g = weights[n - 1] * dx * (1.0 - 2.0 * h[p])
        # k>0: h[p] in h[i+k], k=1..p
        if p > 0:
            ks = np.arange(1, p + 1)
            g += dx * np.sum(weights[n - 1 + ks] * (1.0 - h[p - ks]))
            # k>0: h[p] in (1-h[i]), k=1..n-1-p
            if n - 1 - p > 0:
                ks2 = np.arange(1, n - p)
                g -= dx * np.sum(weights[n - 1 + ks2] * h[p + ks2])
        else:
            if n - 1 - p > 0:
                ks2 = np.arange(1, n - p)
                g -= dx * np.sum(weights[n - 1 + ks2] * h[p + ks2])
        # k<0 (m>0): h[p] in h[i], m=1..p
        if p > 0:
            ms = np.arange(1, p + 1)
            g += dx * np.sum(weights[n - 1 - ms] * (1.0 - h[p - ms]))
        # k<0 (m>0): h[p] in (1-h[i-m]), m=1..n-1-p
        if n - 1 - p > 0:
            ms2 = np.arange(1, n - p)
            g -= dx * np.sum(weights[n - 1 - ms2] * h[p + ms2])
        grad[p] = g
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_grad_vectorized(x, self.tau)[0]
    def gradient(self, x):
        return lse_grad_vectorized(x, self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))


def slsqp_refine(h0, tau, maxiter=250):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_box_sum(np.array(res.x, dtype=np.float64))
    except Exception:
        return h0.copy()


def continuous_refine(h0, deadline):
    h = project_box_sum(h0.copy())
    curr_c = exact_objective(h)
    taus = [0.02, 0.008, 0.003, 0.001, 0.0004, 0.00015]
    for tau in taus:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_refine(h, tau, maxiter=200)
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def greedy_swap(b, deadline, max_rounds=150):
    n = len(b)
    curr = b.copy()
    curr_c = exact_objective(curr)
    ones = list(np.where(curr == 1.0)[0])
    zeros = list(np.where(curr == 0.0)[0])
    stalled = 0
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                curr[i] = 0.0
                curr[j] = 1.0
                c = exact_objective(curr)
                curr[i] = 1.0
                curr[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            curr[i] = 0.0
            curr[j] = 1.0
            curr_c = exact_objective(curr)
            ones = list(np.where(curr == 1.0)[0])
            zeros = list(np.where(curr == 0.0)[0])
            stalled = 0
        else:
            stalled += 1
            if stalled >= 3:
                break
    return curr, curr_c


def sa_binary(b, deadline, max_iters=40000, t_start=0.015, t_end=0.0005, allow_2swap=False):
    n = len(b)
    curr = b.copy()
    curr_c = exact_objective(curr)
    best = curr.copy()
    best_c = curr_c
    rng = np.random.RandomState(12345)
    ones_arr = np.where(curr == 1.0)[0]
    zeros_arr = np.where(curr == 0.0)[0]
    iters_done = 0
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        iters_done = it + 1
        T = t_start * (t_end / t_start) ** (it / float(max_iters))
        T = max(T, t_end)
        use_2swap = allow_2swap and (rng.random() < 0.25)
        if use_2swap and len(ones_arr) >= 2 and len(zeros_arr) >= 2:
            i1 = rng.choice(ones_arr)
            i2 = rng.choice(ones_arr)
            if i1 == i2:
                continue
            j1 = rng.choice(zeros_arr)
            j2 = rng.choice(zeros_arr)
            if j1 == j2:
                continue
            curr[i1] = 0.0
            curr[i2] = 0.0
            curr[j1] = 1.0
            curr[j2] = 1.0
            new_c = exact_objective(curr)
            delta = new_c - curr_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                curr_c = new_c
                ones_list = list(ones_arr)
                ones_list[ones_list.index(i1)] = j1
                ones_list[ones_list.index(i2)] = j2
                ones_arr = np.array(ones_list)
                zeros_list = list(zeros_arr)
                zeros_list[zeros_list.index(j1)] = i1
                zeros_list[zeros_list.index(j2)] = i2
                zeros_arr = np.array(zeros_list)
                if curr_c < best_c - 1e-14:
                    best = curr.copy()
                    best_c = curr_c
            else:
                curr[i1] = 1.0
                curr[i2] = 1.0
                curr[j1] = 0.0
                curr[j2] = 0.0
        else:
            i = rng.choice(ones_arr)
            j = rng.choice(zeros_arr)
            curr[i] = 0.0
            curr[j] = 1.0
            new_c = exact_objective(curr)
            delta = new_c - curr_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                curr_c = new_c
                idx_i = np.where(ones_arr == i)[0][0]
                idx_j = np.where(zeros_arr == j)[0][0]
                ones_arr[idx_i] = j
                zeros_arr[idx_j] = i
                if curr_c < best_c - 1e-14:
                    best = curr.copy()
                    best_c = curr_c
            else:
                curr[i] = 1.0
                curr[j] = 0.0
    return best, best_c


def iterated_sa(b, deadline, n_restarts=5, max_iters=15000):
    n = len(b)
    best = b.copy()
    best_c = exact_objective(best)
    rng = np.random.RandomState(999)
    for r in range(n_restarts):
        if time.time() > deadline - 3:
            break
        # Perturb best: flip 2-4 random positions
        start = best.copy()
        n_flips = rng.randint(2, 5)
        ones = list(np.where(start == 1.0)[0])
        zeros = list(np.where(start == 0.0)[0])
        for _ in range(n_flips):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            start[i] = 0.0
            start[j] = 1.0
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
        # Run SA from perturbed start
        sa_b, sa_c = sa_binary(start, deadline, max_iters=max_iters, t_start=0.012, t_end=0.0003, allow_2swap=True)
        if sa_c < best_c - 1e-14:
            best = sa_b
            best_c = sa_c
    return best, best_c


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_starts(n, rng):
    starts = []
    # Permutation of blocks
    h = np.zeros(n)
    h[:n // 2] = 1.0
    starts.append(h[rng.permutation(n)])
    # Sine waves
    x = np.linspace(0, 1, n)
    for freq in [3, 5, 8]:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        starts.append(project_box_sum(np.clip(h, 0, 1)))
    # Two blocks
    h = np.zeros(n)
    h[:n // 4] = 1.0
    h[3 * n // 4:] = 1.0
    starts.append(h[rng.permutation(n)])
    # Random
    starts.append(project_box_sum(rng.uniform(0, 1, n)))
    return starts


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 10
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c = float('inf')

    # Load parent
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:

            pc = exact_objective(parent_h)
            best_c = pc

    except Exception:
        pass

    if best_h is None:
        n0 = 48

        h_init[:n0 // 2] = 1.0
        best_h = project_box_sum(h_init)
        best_c = exact_objective(best_h)

    # Sizes to try (even, focused on smaller for speed)
    sizes = []
    for s in [parent_n, 48, 44, 40, 46, 36, 52, 32, 56, 60, 64]:
        if s not in sizes and 16 <= s <= 80 and s % 2 == 0:
            sizes.append(s)

    # Phase 1: Binary-focused search at multiple sizes
    for n in sizes:
        if time.time() > deadline - 40:
            break

        # Build candidates
        cands = []
        if parent_h is not None:
            cands.append(transfer_to_size(parent_h, n))
            h_rev = transfer_to_size(parent_h, n)[::-1]
            cands.append(project_box_sum(h_rev))
            # Rotations
            h_rot = transfer_to_size(parent_h, n)
            for sh in [1, 2, 3]:
                cands.append(project_box_sum(np.roll(h_rot, sh)))
        for s in make_starts(n, rng):
            cands.append(s)

        # Score
        scored = sorted([(exact_objective(c), c) for c in cands], key=lambda x: x[0])

        # Binary optimization from top candidates
        for idx in range(min(3, len(scored))):
            if time.time() > deadline - 20:

            b = binary_threshold(scored[idx][1])
            if b is None:

            c_b0 = exact_objective(b)
            if c_b0 < best_c - 1e-12:

            # Greedy swap
            b_g, c_g = greedy_swap(b.copy(), deadline, max_rounds=120)

            if time.time() < deadline - 12:

                b_isa, c_isa = iterated_sa(b_g.copy(), deadline, n_restarts=3, max_iters=10000)

                    best_c = c_isa
                    best_h = b_isa.copy()

        # Continuous refinement on best candidate
        if time.time() < deadline - 15:
            h_ref, c_ref = continuous_refine(scored[0][1], deadline)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
            # Binary from refined continuous
            if n % 2 == 0:
                b = binary_threshold(h_ref)
                if b is not None:
                    b_g, c_g = greedy_swap(b.copy(), deadline, max_rounds=80)
                    if c_g < best_c - 1e-12:
                        best_c = c_g
                        best_h = b_g.copy()

    # Phase 2: Final intensive search on best solution

            if c_isa < best_c - 1e-12:
                best_c = c_isa
                best_h = b_isa.copy()
            # One more greedy pass
            if time.time() < deadline - 4:
                b_g, c_g = greedy_swap(best_h.copy(), deadline, max_rounds=80)

                    best_c = c_g
                    best_h = b_g.copy()
        else:

        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, len(best_h)
