Yes! P_k(h) = P_k(1-h) for all k. So the complement doesn't change the score. That's an interesting symmetry but doesn't help directly.

What about reversal? If h_rev[i] = h[n-1-i], then:
P_k(h_rev) = dx * sum h_rev[i+delta]*(1-h_rev[i]) = dx * sum h[n-1-i-delta]*(1-h[n-1-i])

Let j = n-1-i, then h[n-1-i-delta] = h[j-delta] and h[n-1-i] = h[j].
So P_k(h_rev) = dx * sum h[j-delta]*(1-h[j]) = P_{-k}(h) (the profile at the opposite lag).

So reversal swaps the profile: P_k(h_rev) = P_{-k}(h). This means max P_k is the same for h and h_rev. So reversal doesn't change the score either!

This means the problem is symmetric under complement and reversal. The symmetrized version 0.5*(h + h_rev + (1-h) + (1-h_rev)) = 0.5*(h + h_rev + 1 - h + 1 - h_rev) = 0.5*(2) = 1, which is not useful.

But 0.5*(h + h_rev) is the "reversal-symmetrized" version, and since max P_k is invariant under reversal, symmetrizing by reversal should not change the score (or might slightly improve it due to the averaging effect on the profile).

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
        remaining = diff
        for i in idx:
            if remaining <= 1e-15:
                break
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
    else:
        idx = np.argsort(-h)
        remaining = -diff
        for i in idx:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[i])
            h[i] -= rem
            remaining -= rem
    return h

def grad_combined_top(h, top_k, weight='uniform'):
    prof = get_profile(h)
    n = len(h)
    dx = 2.0 / n
    active = np.argsort(prof)[::-1][:top_k]
    
    if weight == 'value':
        w = prof[active].copy()
        w = np.maximum(w, 0) + 1e-10
    else:
        w = np.ones(top_k)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx, k in enumerate(active):
        delta = k - (n - 1)
        if delta > 0:
            m = n - delta
            grad[0:m] -= dx * h[delta:n] * w[idx]
            grad[delta:n] += dx * (1.0 - h[0:m]) * w[idx]
        elif delta == 0:
            grad += dx * (1.0 - 2.0 * h) * w[idx]
        else:
            d = -delta
            m = n - d
            grad[0:m] += dx * (1.0 - h[d:n]) * w[idx]
            grad[d:n] -= dx * h[0:m] * w[idx]
    
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None, prof
    return -grad / norm, prof

def pair_1d(h, i, j, nsamp=6):
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return h, exact_score(h)
    
    best_s = exact_score(h)
    best_t = 0.0
    hi = h[i]
    hj = h[j]
    
    for k in range(1, nsamp + 1):
        frac = k / (nsamp + 1)
        t = t_lo + frac * (t_hi - t_lo)
        h[i] = hi + t
        h[j] = hj - t
        s = exact_score(h)
        if s < best_s - 1e-13:
            best_s = s
            best_t = t
    
    h[i] = hi + best_t
    h[j] = hj - best_t
    return h, best_s
