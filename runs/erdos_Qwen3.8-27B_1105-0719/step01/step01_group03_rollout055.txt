```python
import numpy as np
from scipy.optimize import minimize, lbfgsb
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_matrix(h, n, dx):
    """G[m, j] = dc_m/dh[j] where c is the profile."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # c = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d]) * dx
            # dc/dh[j] = dx*(1-h[j+d]) for j in [0, n-1-d]  (h[j] as first factor)
            #           - dx*h[j-d] for j in [d, n-1]         (h[j] as second factor: h[j-d]*(1-h[j]))
            G[m, 0:n-d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n-d]
        else:
            dp = -d
            # c = sum_{i=0}^{n-1-dp} h[i+dp]*(1-h[i]) * dx
            # dc/dh[j] = dx*(1-h[j-dp]) for j in [dp, n-1]  (h[j] as first factor)
            #           - dx*h[j+dp] for j in [0, n-1-dp]    (h[j] as second factor)
            G[m, dp:n] += dx * (1.0 - h[0:n-dp])
            G[m, 0:n-dp] += -dx * h[dp:n]
    return G

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)

                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:

    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:

        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def logsumexp_objective(h, beta, penalty=1e4):
    n = len(h)
    p = compute_profile(h)
    m = np.max(p)
    lse = m + np.log(np.sum(np.exp(beta * (p - m)))) / beta
    sum_violation = (h.sum() - n / 2.0) ** 2
    return lse + penalty * sum_violation

def logsumexp_grad(h, beta, penalty=1e4):
    n = len(h)
    dx = 2.0 / n
    p = compute_profile(h)
    m = np.max(p)
    weights = np.exp(beta * (p - m))
    weights /= weights.sum()
    G = grad_matrix(h, n, dx)
    grad_lse = G.T @ weights
    grad_pen = 2 * penalty * (h.sum() - n / 2.0)
    return grad_lse + grad_pen

def fista_optimize(h0, max_iters=5000, L_init=None):
    """FISTA (Nesterov accelerated projected gradient) for the minimax problem."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_c = exact_score(h)
    
    if L_init is None:
        # Estimate Lipschitz constant
        G = grad_matrix(h, n, dx)
        L = np.max(np.linalg.norm(G, axis=1))
        L *= 2.0  # Safety factor
    
    step = 1.0 / L
    
    for k in range(max_iters):
        # Compute gradient at y
        p = compute_profile(y)
        c = np.max(p)
        # Subgradient: average of active constraints
        thresh = c - 1e-4
        active = p >= thresh
        if not np.any(active):
            active = np.zeros(len(p), dtype=bool)
            active[np.argmax(p)] = True
        G = grad_matrix(y, n, dx)
        grad = G[active].mean(axis=0)
        # Remove mean (to respect the sum constraint approximately)
        grad -= grad.mean()
        
        # Gradient step
        new_h = y - step * grad
        # Project onto {h: 0<=h<=1, sum(h)=n/2}
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = repair_feasible(new_h)
        
        # Update
        t_new = (1 + np.sqrt(1 + 4 * t_k ** 2)) / 2
        y = new_h + ((t_k - 1) / t_new) * (new_h - h)
        y = np.clip(y, -0.5, 1.5)  # Keep in reasonable range
        h = new_h
        t_k = t_new
        
        c_new = exact_score(h)
        if c_new < best_c:
            best_c = c_new
    
    return h, best_c
```

```python
def grad_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # c = sum_{j=0}^{n-1-d} h[j]*(1-h[j+d]) * dx
```

One more thought: the "penalty" for the sum constraint in the LSE approach might not be the best. An alternative is to use the "augmented Lagrangian" or to simply project onto the sum constraint after each step.

For L-BFGS-B, the cleanest approach is:
- Objective: logsumexp(profile(h), beta)
- Constraints: 0 ≤ h[i] ≤ 1 (handled by bounds)
- After each L-BFGS-B call, project onto {sum(h) = n/2} ∩ {0 ≤ h ≤ 1}

This way, the sum constraint is handled exactly (via projection) rather than approximately (via penalty).

```python
def project_sum(h, n):
    """Project h onto {h: 0<=h<=1, sum(h)=n/2}."""
    return repair_feasible(h)

def lse_h_with_penalty(h, beta):
    """LSE objective with sum constraint penalty."""
    p = compute_profile(h)
    m = np.max(p)
    return m + np.log(np.sum(np.exp(beta*(p-m))))/beta

def lse_h_grad_with_penalty(h, beta):
    """Gradient of LSE objective."""
    n = len(h)
    dx = 2.0/n
    p = compute_profile(h)
    m = np.max(p)
    w = np.exp(beta*(p-m))
    w /= w.sum()
    G = grad_matrix_fixed(h, n, dx)
    return G.T @ w

def optimize_lse_penalized(h0, betas, maxiter=200):
    """L-BFGS-B on LSE objective, projecting onto sum constraint after each call."""
    h = h0.copy()
    for beta in betas:
        bounds = [(0.0, 1.0)] * len(h)
        try:
            res = minimize(lse_h_with_penalty, h, jac=lse_h_grad_with_penalty,
                          args=(beta,), method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h = project_sum(res.x, len(h))
        except:
            pass
    return h
```

```python
def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_matrix_fixed(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, 0:n-d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n-d]
        else:
            dp = -d
            G[m, 0:n-dp] += -dx * h[dp:n]
            G[m, dp:n] += dx * (1.0 - h[0:n-dp])
    return G

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()

    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lse_objective(h, beta, pen):
    p = compute_profile(h)
    m = np.max(p)
    shift = beta * (p - m)
    shift = np.clip(shift, -50, 50)
    lse = m + np.log(np.sum(np.exp(shift))) / beta
    sv = (h.sum() - len(h) / 2.0) ** 2
    return lse + pen * sv

def lse_gradient(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    p = compute_profile(h)
    m = np.max(p)
    shift = beta * (p - m)
    shift = np.clip(shift, -50, 50)
    w = np.exp(shift)
    w /= w.sum()
    G = grad_matrix_fixed(h, n, dx)
    grad_lse = G.T @ w
    grad_pen = 2.0 * pen * (h.sum() - n / 2.0)
    return grad_lse + grad_pen

def optimize_lse(h0, betas, pen=1e5, maxiter=200):
    h = h0.copy()
    for beta in betas:
        bounds = [(0.0, 1.0)] * len(h)
        try:
            res = minimize(lse_objective, h, jac=lse_gradient,
                          args=(beta, pen), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
            h = repair_feasible(res.x)
        except Exception:
            pass
    return h

def fista_optimize(h0, max_iters=5000, step=None):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_c = exact_score(h)
    if step is None:
        G = grad_matrix_fixed(h, n, dx)
        norms = np.linalg.norm(G, axis=1)
        step = 1.0 / (2.0 * np.max(norms))
    for k in range(max_iters):
        p = compute_profile(y)
        c = np.max(p)
        thresh = c - 1e-5
        active = p >= thresh
        if not np.any(active):
            active = np.zeros(len(p), dtype=bool)
            active[np.argmax(p)] = True
        G = grad_matrix_fixed(y, n, dx)
        grad = G[active].mean(axis=0)
        grad -= grad.mean()
        new_h = y - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = repair_feasible(new_h)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = new_h + ((t_k - 1.0) / t_new) * (new_h - h)
        y = np.clip(y, -0.5, 1.5)
        h = new_h
        t_k = t_new
        c_new = exact_score(h)
        if c_new < best_c:
            best_c = c_new
    return h, best_c

def binary_local_search(h_bin, rng, max_iters=3000):
    n = len(h_bin)
    h = h_bin.copy().astype(np.float64)
    h[h] >= 0.5 = 1.0
    h[h] < 0.5 = 0.0
    best_c = exact_score(h)
    for it in range(max_iters):
        improved = False
        for _ in range(100):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c = exact_score(h)
            if c < best_c - 1e-10:
                best_c = c
                improved = True
                break
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, best_c

def pair_local_search(h, rng, max_iters=2000):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    values = [0.0, 0.25, 0.5, 0.75, 1.0]
    for it in range(max_iters):
        improved = False
        for _ in range(200):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            a_save, b_save = h[i], h[j]
            mean_save = (a_save + b_save) / 2.0
            for va in values:
                for vb in values:
                    if abs((va + vb) / 2.0 - mean_save) > 0.3:
                        continue
                    h[i], h[j] = va, vb
                    h = repair_feasible(h)
                    c = exact_score(h)
                    if c < best_c - 1e-10:
                        best_c = c
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                h[i], h[j] = a_save, b_save
                h = repair_feasible(h)
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(66, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)
    best_n = len(parent)

    # Stage 1: LSE from parent with increasing beta
    if time_left() > 120:
        betas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
        h_lse = optimize_lse(best_h, betas, pen=1e5, maxiter=150)
        c_lse = exact_score(h_lse)
        if c_lse < best_c:
            best_h = h_lse
            best_c = c_lse

    # Stage 2: FISTA with corrected gradient
    if time_left() > 100:
        h_fista, c_fista = fista_optimize(best_h, max_iters=3000)
        if c_fista < best_c:
            best_h = h_fista
            best_c = c_fista

    # Stage 3: More LSE from improved point
    if time_left() > 80:
        betas = [500, 1000, 3000, 10000]
        h_lse = optimize_lse(best_h, betas, pen=1e6, maxiter=200)
        c_lse = exact_score(h_lse)
        if c_lse < best_c:
            best_h = h_lse
            best_c = c_lse

    # Stage 4: Try different resolutions
    if time_left() > 200:
        candidates = []
        base_n = len(best_h)
        for delta in [-6, -4, -2, 2, 4, 6, 8, 12, 16, 24, 34, 44]:
            d = base_n + delta
            if 30 <= d <= 160:
                candidates.append(d)
        candidates = sorted(set(candidates), key=lambda x: abs(x - base_n))
        for new_n in candidates[:6]:
            if time_left() < 50:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            betas = [50, 200, 1000, 5000]
            h_lse = optimize_lse(h_interp, betas, pen=1e5, maxiter=100)
            c_lse = exact_score(h_lse)
            if c_lse < best_c:
                best_h = h_lse
                best_c = c_lse
                best_n = new_n

    # Stage 5: FISTA again at best resolution
    if time_left() > 80:
        h_fista, c_fista = fista_optimize(best_h, max_iters=4000)
        if c_fista < best_c:
            best_h = h_fista
            best_c = c_fista

    # Stage 6: Binary local search
    if time_left() > 60:
        h_bin = (best_h >= 0.5).astype(np.float64)
        h_bin = repair_feasible(h_bin)
        h_bls, c_bls = binary_local_search(h_bin, rng, max_iters=2000)
        if c_bls < best_c:
            best_h = h_bls
            best_c = c_bls
        # Refine binary solution with LSE
        if time_left() > 40:
            h_ref = optimize_lse(h_bls, [100, 500, 2000], pen=1e5, maxiter=100)
            c_ref = exact_score(h_ref)
            if c_ref < best_c:
                best_h = h_ref
                best_c = c_ref

    # Stage 7: Random restarts
    if time_left() > 150:
        for trial in range(6):
            if time_left() < 40:
                break
            eps = 0.005 * (trial + 1)
            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            betas = [50, 200, 1000, 5000]
            h_lse = optimize_lse(h_pert, betas, pen=1e5, maxiter=80)
            c_lse = exact_score(h_lse)
            if c_lse < best_c:
                best_h = h_lse
                best_c = c_lse
                # Follow up with FISTA
                if time_left() > 60:
                    h_fista, c_fista = fista_optimize(best_h, max_iters=2000)
                    if c_fista < best_c:
                        best_h = h_fista
                        best_c = c_fista

    # Stage 8: Pair local search on continuous
    if time_left() > 60:
        h_ps, c_ps = pair_local_search(best_h, rng, max_iters=1000)
        if c_ps < best_c:
            best_h = h_ps
            best_c = c_ps

    # Stage 9: Final LSE polish
    if time_left() > 40:
        betas = [5000, 10000, 20000]
        h_final = optimize_lse(best_h, betas, pen=1e6, maxiter=100)
        c_final = exact_score(h_final)
        if c_final < best_c:
            best_h = h_final
            best_c = c_final

    # Stage 10: Final FISTA polish
    if time_left() > 30:
        h_fista, c_fista = fista_optimize(best_h, max_iters=2000)
        if c_fista < best_c:
            best_h = h_fista
            best_c = c_fista

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
```
