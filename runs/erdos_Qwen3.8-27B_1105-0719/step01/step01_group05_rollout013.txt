import numpy as np
from scipy.optimize import minimize
import time

def compute_profile(h):
    """Compute the full correlation profile."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return profile

def exact_c5(h):
    """Compute C5 = max of profile."""
    return float(np.max(compute_profile(h)))

def project_box_sum(h, target=None):
    """Project h to [0,1]^n with exact sum = target."""
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Spread the error across all non-boundary elements
        free = (h > 0.01) & (h < 0.99)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def binary_local_search(b, max_iter=5000, rng=None, time_limit=None):

    cooling = 0.999

            break

def slsqp_c_variable(h0, n, deadline, max_active=20):
    """SLSQP optimization with c-variable formulation."""
    profile0 = compute_profile(h0)
    c0 = float(np.max(profile0))
    
    # Variables: [h[0],...,h[n-1], c]
    x0 = np.append(h0, c0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    # Objective: minimize c
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    # Equality constraint: sum(h) = n/2
    target = 0.5 * n
    def eq_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    def make_ineq(r):
        """Inequality constraint: c - q_r(h) >= 0"""
        def ineq_fun(x):
            h = x[:n]
            c_var = x[-1]
            # Compute q_r
            profile_val = np.correlate(h, 1.0 - h, mode='full')[r] * (2.0 / n)
            return c_var - profile_val
        
        def ineq_jac(x):
            h = x[:n]
            dx = 2.0 / n
            e = r - (n - 1)
            g = np.zeros(n + 1)
            if e >= 0:
                # q_r = dx * sum_{i=0}^{n-1-e} h[i+e] * (1-h[i])
                s = n - e
                if s > 0:
                    # d/dh[i+e] = dx * (1-h[i])
                    g[e:e+s] += dx * (1.0 - h[:s])
                    # d/dh[i] = -dx * h[i+e]
                    g[:s] -= dx * h[e:e+s]
            else:
                # e < 0, s = -e
                s = -e
                # q_r = dx * sum_{i=s}^{n-1} h[i] * (1-h[i-s])
                # Let j = i-s, so j = 0 to n-1-s
                cs = n - s
                # d/dh[j+s] = dx * (1-h[j])
                g[s:s+cs] += dx * (1.0 - h[:cs])
                # d/dh[j] = -dx * h[j+s]
                g[:cs] -= dx * h[s:s+cs]
            g[-1] = 1.0  # d/dc = 1
            return g
        return ineq_fun, ineq_jac
    
    # Select top-k most active constraints
    profile_vals = np.correlate(h0, 1.0 - h0, mode='full') * (2.0 / n)
    top_k = np.argsort(-profile_vals)[:max_active]

    for r in top_k:
        f, j = make_ineq(int(r))
        constraints.append({'type': 'ineq', 'fun': f, 'jac': j})
    
    try:
        res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 500, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], target)
            c_new = exact_c5(h_new)
            if c_new < c0:
                return h_new, c_new

        pass
    
    return h0, c0
