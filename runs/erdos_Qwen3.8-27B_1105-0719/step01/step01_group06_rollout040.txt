from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    """Compute gradient of each lag's profile value with respect to h."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                # p_k = dx * sum_{i=delta}^{n-1} h[i]*(1-h[i-delta])
                # dp_k/dh[i] for i >= delta: dx*(1-h[i-delta])
                # dp_k/dh[i] for i < delta: -dx*h[i+delta]
                G[k, delta:] += dx * (1.0 - h[:n-delta])
                G[k, :n-delta] -= dx * h[delta:]
        else:
            d = -delta
            if d < n:
                # p_k = dx * sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
                # dp_k/dh[i] for i < n-d: dx*(1-h[i+d])
                # dp_k/dh[i] for i >= n-d: -dx*h[i-d]
                G[k, :n-d] += dx * (1.0 - h[d:])
                G[k, n-d:] -= dx * h[:n-d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()

    if diff > 0:

    if tau * max_p > 700:

    return np.log(np.sum(np.exp(tau * prof))) / tau

    prof = get_profile(h)
    max_p = np.max(prof)
    shifted = tau * (prof - max_p)
    weights = np.exp(shifted)
    weights /= weights.sum()
    G = get_lag_gradients(h)
    return weights @ G

def solve_epigraph(h_start, t_start, beta=0.0, maxiter=80, n_constraints=0, active_lags=None):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    
    if active_lags is None:
        if nc < M:
            prof = get_profile(h_start)
            active_lags = np.argsort(prof)[::-1][:nc]
        else:
            active_lags = np.arange(M)
    
    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))
    
    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):

            return h_sol

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n and delta > 0:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
            elif delta == 0:
                G[k, :] += dx * (1.0 - 2.0 * h)
        else:
            d = -delta
            if 0 < d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def lse_value_and_weights(h, tau):
    prof = get_profile(h)
    max_p = np.max(prof)
    shifted = tau * (prof - max_p)
    shifted = np.clip(shifted, -700, 700)
    weights = np.exp(shifted)
    weights /= weights.sum()
    lse_val = max_p + np.log(weights.sum() * len(prof) ** 0) / tau  # placeholder
    lse_val = max_p + np.log(np.sum(np.exp(shifted))) / tau
    return lse_val, weights

def lse_gradient(h, tau):
    _, weights = lse_value_and_weights(h, tau)
    G = get_lag_gradients(h)
    return weights @ G

def solve_lse_slsqp(h_start, tau, maxiter=100):
    n = len(h_start)
    def obj(h):
        v, _ = lse_value_and_weights(h, tau)
        return v
    def jac(h):
        return lse_gradient(h, tau)
    def con_fun(h):
        return h.sum() - n / 2.0
    def con_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj, h_start, jac=jac, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        if result.x is not None and np.all(np.isfinite(result.x)):

    except Exception:
        pass
    return None

def solve_epigraph(h_start, t_start, beta=0.0, maxiter=80, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))

    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    try:
        result = minimize(obj, x0, jac=obj_jac, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],

        if result.x is not None and np.all(np.isfinite(result.x)):

        pass

                h_sol = solve_epigraph(h_interp, best_score + 1e-6, beta=0.0, maxiter=mi, n_constraints=35)
                if h_sol is not None:
                    s = exact_score(h_sol)
                    if s < best_score - 1e-12:
                        best_h = h_sol
                        best_score = s

    # Stage 8: Final LSE-SLSQP with high temperature
    if tleft() > 20:
        for tau in [80, 150, 300]:
            if tleft() < 12:
                break
            mi = min(60, max(20, int(tleft() * 0.3)))
            h_sol = solve_lse_slsqp(best_h, tau=tau, maxiter=mi)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s

    # Stage 9: Final local descent
    if tleft() > 10:
        tl = min(10, tleft() * 0.5)
        h_fin, s_fin = local_descent(best_h, max_iters=60, time_limit=tl)
        if s_fin < best_score - 1e-12:
            best_h = h_fin
            best_score = s_fin

    # Final symmetrization check
    h_final_sym = symmetrize(best_h)
    s_final_sym = exact_score(h_final_sym)
    if s_final_sym < best_score - 1e-14:
        best_h = h_final_sym
        best_score = s_final_sym

    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)

    return (best_h, final_score, len(best_h))
