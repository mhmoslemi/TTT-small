    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0

    if abs(diff) < 1e-14:

    if diff > 0:

        order = np.argsort(-capacity)
        remaining = diff

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def profile_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def profile_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    w = weights
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        d = m - (n - 1)
        wc = w[m] * dx
        if d >= 0:
            if d < n:
                grad[d:n] += wc * (1.0 - h[0:n - d])
                grad[0:n - d] -= wc * h[d:n]

            grad[dp:n] += wc * (1.0 - h[0:n - dp])
            grad[0:n - dp] -= wc * h[dp:n]
    return profile, grad

def smax_weights(profile, tau):
    x = (profile - np.max(profile)) / tau

    return ex / ex.sum()

def smax_val(profile, tau):
    m = np.max(profile)
    return m + tau * np.log(np.sum(np.exp((profile - m) / tau)))

    nh = n // 2
    half = (h[:nh] + 1.0 - h[n - 1 - nh:n - 1:-1]) / 2.0

def from_symmetric_half(half):

    n = 2 * nh
    h = np.zeros(n)
    h[:nh] = half
    h[nh:] = 1.0 - half[::-1]

    n = len(h)
    if n == new_n:
        return h.copy()
