import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    """Returns the full correlation array (unscaled by dx)."""
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(y, target_sum):
    """Project y onto {0 <= h <= 1, sum(h) = target_sum}."""
    n = len(y)
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute residual among non-bound entries
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            per = residual / len(eligible)
            h[eligible] += per
            h = np.clip(h, 0.0, 1.0)
    return h

def slp_step(h, R, rho, active_tol, solver="CLARABEL"):
    """One SLP trust-region step using CVXPY."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Compute current correlations and gradient
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx  # scaled correlation values
    
    # Find active set
    F_max = np.max(F)
    active = np.where(F >= F_max - active_tol)[0]
    
    # Compute gradient of each active correlation w.r.t. h
    # For lag index m (0 to 2n-2), lag k = m - (n-1)
    # corr[m] = sum over valid (i,j) where j-i = k: h[i]*(1-h[j])
    # Actually: np.correlate(a,b,"full")[m] = sum_i a[i]*b[m-i] (with 0-indexing, valid range)
    
    # Let me use the definition: result[m] = sum_i a[i + (m - (n-1))] * b[i] for valid i
    # where a = h, b = 1-h
    # So result[m] = sum_i h[i + k] * (1-h[i]) where k = m - (n-1), i ranges so both indices valid
    
    # For k >= 0: i from 0 to n-1-k, result = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
    # For k < 0: i from -k to n-1, result = sum_{i=-k}^{n-1} h[i+k]*(1-h[i]) = sum_{j=0}^{n-1+|k|} h[j]*(1-h[j+|k|])
    
    # Gradient w.r.t. h[j]:
    # For k >= 0: d/dh[j] = (1-h[j-k]) if 0 <= j-k <= n-1-k (j is the "i+k" position)
    #                        - h[j+k] if 0 <= j+k <= n-1 (j is the "i" position)
    # For k < 0 (let k' = -k > 0): d/dh[j] = (1-h[j+k']) if 0 <= j+k' <= n-1 (j is the "i" position, h[i] where i=j)
    #                                          - h[j-k'] if 0 <= j-k' <= n-1-k' ... 
    
    # Actually let me just be very explicit. 
    # np.correlate(h, 1-h, mode="full")[m]:
    # Let n1 = len(h) = n, n2 = len(1-h) = n. Full length = n1+n2-1 = 2n-1.
    # The m-th element (0-indexed) is:
    # sum over i of h[i] * (1-h)[i + (m - (n-1))]  ... no wait.
    
    # From numpy docs: c_k = sum_n a_{n+k} * conj(v_n)
    # For real: c_k = sum_n a_{n+k} * v_n
    # where k ranges from -(n2-1) to n1-1
    # In "full" mode, the output is indexed so that k = -(n2-1) corresponds to index 0.
    # So output[m] corresponds to k = m - (n2-1) = m - (n-1)
    
    # So output[m] = sum_n h[n+k] * (1-h)[n] where k = m-(n-1)
    #              = sum_n h[n+k] * (1-h[n]) for valid n
    
    # For k >= 0: valid n is 0 to n-1-k (since n+k < n means n < n-k)
    #   output[m] = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])
    #   d/dh[j]: 
    #     If j = n+k for some valid n, i.e., k <= j <= n-1: contributes (1-h[j-k])
    #     If j = n for some valid n, i.e., 0 <= j <= n-1-k: contributes -h[j+k]
    #     So d/dh[j] = [k<=j<=n-1]*(1-h[j-k]) - [0<=j<=n-1-k]*h[j+k]
    
    # For k < 0, let k = -|k|:
    #   output[m] = sum_{n=|k|}^{n-1} h[n-|k|] * (1-h[n])
    #   d/dh[j]:
    #     If j = n-|k| for some valid n, i.e., |k| <= j <= n-1-|k|: contributes (1-h[j+|k|])
    #     If j = n for some valid n, i.e., |k| <= j <= n-1: contributes -h[j-|k|]
    #     So d/dh[j] = [|k|<=j<=n-1-|k|]*(1-h[j+|k|]) - [|k|<=j<=n-1]*h[j-|k|]
    
    # Let me compute gradients for active lags only
    n_lags = 2 * n - 1
    # G has shape (len(active), n)
    G = np.zeros((len(active), n))
    for idx, m in enumerate(active):
        k = m - (n - 1)
        if k >= 0:
            # d/dh[j] = [k<=j<=n-1]*(1-h[j-k]) - [0<=j<=n-1-k]*h[j+k]
            j_range1 = range(k, n)  # j from k to n-1
            for j in j_range1:
                G[idx, j] += (1.0 - h[j - k])
            j_range2 = range(0, n - k)  # j from 0 to n-1-k
            for j in j_range2:
                G[idx, j] -= h[j + k]
        else:
            abs_k = -k
            # d/dh[j] = [abs_k<=j<=n-1-abs_k]*(1-h[j+abs_k]) - [abs_k<=j<=n-1]*h[j-abs_k]
            j_range1 = range(abs_k, n - abs_k)  # j from |k| to n-1-|k|
            for j in j_range1:
                G[idx, j] += (1.0 - h[j + abs_k])
            j_range2 = range(abs_k, n)  # j from |k| to n-1
            for j in j_range2:
                G[idx, j] -= h[j - abs_k]
    
    # Build CVXPY problem
    u = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        t >= F[active] + G @ u,
        u >= -h,
        u <= 1.0 - h,
        cp.sum(u) == 0,
        cp.norm(u) <= R,
    ]
    
    objective = cp.Minimize(t + rho * cp.sum_squares(u))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver="SCS", verbose=False)
        except Exception:
            return None, None
    
    if u.value is None:
        return None, None
    
    u_opt = np.array(u.value).flatten()
    if not np.all(np.isfinite(u_opt)):
        return None, None
    
    # Clip to trust region
    norm_u = np.linalg.norm(u_opt)
    if norm_u > R * (1 + 1e-6):
        u_opt = u_opt * (R / norm_u)
    
    h_new = h + u_opt
    h_new = project_feasible(h_new, target_sum)
    
    return h_new, None

def local_search_binary(h, n_iterations, rng, n=0):
    """Local search using flips and swaps for near-binary solutions."""
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for it in range(n_iterations):
        # Try a random move
        move_type = rng.randint(3)
        h_new = best_h.copy()
        
        if move_type == 0:
            # Flip-like: change one value
            i = rng.randint(n)
            delta = 0.1 * rng.randn()
            h_new[i] = np.clip(best_h[i] + delta, 0, 1)
        elif move_type == 1:
            # Swap
            i, j = rng.choice(n, size=2, replace=False)
            h_new[i], h_new[j] = best_h[j], best_h[i]
        else:
            # Transfer
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(best_h[i], 1.0 - best_h[j])
            if max_transfer > 1e-6:
                transfer = max_transfer * rng.uniform(0.1, 1.0)
                h_new[i] = best_h[i] - transfer
                h_new[j] = best_h[j] + transfer
            else:
                continue
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5

def local_search_directed(h, n_iterations, rng):
    """Local search focusing on active lags."""
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for it in range(n_iterations):
        # Find the worst lag
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        worst_idx = np.argmax(corr)
        k = worst_idx - (n - 1)
        
        # For this lag, find the pair (i, j) with maximum contribution
        # corr[worst_idx] = sum h[i+k]*(1-h[i]) for valid i
        if k >= 0:
            terms = best_h[k:] * (1.0 - best_h[:n-k])
        else:
            abs_k = -k
            terms = best_h[abs_k:] * (1.0 - best_h[:n-abs_k])
        
        # Find the largest term and try to reduce it
        max_term_idx = np.argmax(terms)
        
        if k >= 0:
            i = max_term_idx + k  # h[i] is the "source"
            j = max_term_idx      # h[j] is the "sink" (1-h[j])
            # To reduce h[i]*(1-h[j]), we can decrease h[i] or increase h[j]
            # Transfer from i to j
            if i == j:
                continue
            max_transfer = min(best_h[i], 1.0 - best_h[j])
        else:
            abs_k = -k
            i = max_term_idx      # h[i] is the "source" (h[i+k] where i+k = max_term_idx, so i = max_term_idx)
            # Wait, for k<0: terms[i] = h[i+k]*(1-h[i]) = h[i-|k|]*(1-h[i])
            # So the "source" is at position i-|k| and "sink" (1-h) is at position i
            i_src = max_term_idx - abs_k  # position of h value that's multiplied
            j_sink = max_term_idx          # position of (1-h) value
            if i_src == j_sink:
                continue
            max_transfer = min(best_h[i_src], 1.0 - best_h[j_sink])
            i, j = i_src, j_sink
        
        if max_transfer > 1e-8:
            # Random fraction of max transfer
            frac = rng.uniform(0.05, 0.5)
            transfer = max_transfer * frac
            h_new = best_h.copy()
            h_new[i] = best_h[i] - transfer
            h_new[j] = best_h[j] + transfer
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
    
    return best_h, best_c5

    n = len(h)

    j = np.arange(n)

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(y, target_sum):
    n = len(y)
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_gradient(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            mask1 = (j >= k)
            mask2 = (j <= n - 1 - k)
            grad[m, mask1] += (1.0 - h[j[mask1] - k])
            grad[m, mask2] -= h[j[mask2] + k]
        else:
            abs_k = -k
            mask1 = (j <= n - 1 - abs_k)
            mask2 = (j >= abs_k)
            grad[m, mask1] += (1.0 - h[j[mask1] + abs_k])
            grad[m, mask2] -= h[j[mask2] - abs_k]
    return grad


def slp_step(h, R, rho, active_tol, solver_names=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    F_max = np.max(F)
    active = np.where(F >= F_max - active_tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(F)])
    
    grad = compute_gradient(h)
    G = grad[active]
    
    u = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        t >= F[active] + G @ u,
        u >= -h,
        u <= 1.0 - h,
        cp.sum(u) == 0,
        cp.norm(u) <= R,
    ]
    objective = cp.Minimize(t + rho * cp.sum_squares(u))
    prob = cp.Problem(objective, constraints)
    
    if solver_names is None:
        solver_names = ["CLARABEL", "ECOS", "SCS", "GLPK", "CBC"]
    
    solved = False
    for sname in solver_names:
        try:
            prob.solve(solver=sname, verbose=False)
            solved = True
            break
        except Exception:
            continue
    
    if not solved or u.value is None:
        return None
    
    u_opt = np.array(u.value).flatten()
    if not np.all(np.isfinite(u_opt)):
        return None
    
    norm_u = np.linalg.norm(u_opt)
    if norm_u > R * (1 + 1e-6):
        u_opt = u_opt * (R / norm_u)
    
    h_new = h + u_opt
    h_new = project_feasible(h_new, n / 2.0)
    return h_new

        k = worst_idx - (n - 1)
        
        if k >= 0:

        else:
            abs_k = -k

            j_sink = max_term_idx
        else:

            j_sink = max_term_idx + abs_k
        
        if i_src == j_sink:

                if k >= 0:

                    j_sink = idx2
                else:
                    i_src = idx2
                    j_sink = idx2 + abs_k

            else:
                continue

    deadline1 = t0 + budget_s * 0.30
    R = 0.06
    rho = 8.0
    active_tol = 0.015
    stall = 0
    
    while elapsed() < deadline1 and stall < 25:
        h_new = slp_step(best_h, R, rho, active_tol)
        if h_new is not None:
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
                stall = 0
                R = min(R * 1.15, 0.25)
                if R < 0.015:
                    active_tol = max(0.001, active_tol * 0.92)
            else:
                stall += 1
                R *= 0.7
                if R < 0.003:
                    break
        else:
            stall += 1
            R *= 0.7
    
    # Phase 2: Directed local search
    deadline2 = t0 + budget_s * 0.42
    while elapsed() < deadline2:
        h_opt, c5_opt = directed_local_search(best_h, 8000, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        else:
            break
    
    # Phase 3: SLP refinement with tighter parameters
    deadline3 = t0 + budget_s * 0.55
    R = 0.025
    rho = 30.0
    active_tol = 0.003
    stall = 0
    while elapsed() < deadline3 and stall < 20:
        h_new = slp_step(best_h, R, rho, active_tol)
        if h_new is not None:
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
                stall = 0
                R = min(R * 1.2, 0.08)
            else:
                stall += 1
                R *= 0.65
        else:
            stall += 1
    
    # Phase 4: Local search again
    deadline4 = t0 + budget_s * 0.65
    while elapsed() < deadline4:
        h_opt, c5_opt = directed_local_search(best_h, 5000, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        else:
            h_opt, c5_opt = binary_local_search(best_h, 5000, rng)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
            else:
                break
    
    # Phase 5: Try different resolutions
    deadline5 = t0 + budget_s * 0.88
    n_current = len(best_h)
    
    resolutions = [48, 54, 66, 72, 78, 84, 90, 96, 108, 120]
    resolutions.sort(key=lambda x: abs(x - n_current))
    
    for n_try in resolutions:
        if elapsed() > deadline5:
            break
        if n_try == n_current:
            continue
        
        x_old = np.linspace(0, 2, n_current, endpoint=False) + 1.0 / n_current
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        
        # SLP at new resolution
        R = 0.04
        rho = 10.0
        stall = 0
        max_slp_time = deadline5 - elapsed()
        while elapsed() < min(deadline5, time.time() + max(5, max_slp_time * 0.4)) and stall < 15:
            h_new = slp_step(h_interp, R, rho, 0.008)
            if h_new is not None:
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h_interp) - 1e-13:
                    h_interp = h_new
                    stall = 0
                    R = min(R * 1.2, 0.15)
                else:
                    stall += 1
                    R *= 0.7
            else:
                stall += 1
        
        # Quick local search
        if elapsed() < deadline5:
            h_opt, c5_opt = directed_local_search(h_interp, 3000, rng)
            if c5_opt < compute_c5(h_interp):
                h_interp = h_opt
        
        c5_new = compute_c5(h_interp)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_interp
            n_current = n_try
    
    # Phase 6: Final SLP polish
    deadline6 = t0 + budget_s * 0.95
    R = 0.01
    rho = 100.0
    active_tol = 0.001
    stall = 0
    while elapsed() < deadline6 and stall < 10:
        h_new = slp_step(best_h, R, rho, active_tol)
        if h_new is not None:
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
                stall = 0
            else:
                stall += 1
                R *= 0.5
        else:
            stall += 1
    
    # Phase 7: Final local search
    if elapsed() < budget_s - 5:
        n_iter = min(15000, int((budget_s - 5 - elapsed()) * 200))
        h_opt, c5_opt = directed_local_search(best_h, max(n_iter, 2000), rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_h = project_feasible(best_h, len(best_h) / 2.0)
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
