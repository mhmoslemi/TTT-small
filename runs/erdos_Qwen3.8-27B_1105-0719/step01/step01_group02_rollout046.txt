    n = len(b)
    cur = b.copy().astype(float)
    cur_score = exact_score(cur)
    best = cur.copy()
    best_score = cur_score
    
    T0 = 0.01  # initial temperature
    Tend = 0.0001
    decay = (Tend/T0) ** (1.0/n_iter)
    
    for it in range(n_iter):
        T = T0 * (decay ** it)
        i = rng.choice(np.where(cur==1)[0])
        j = rng.choice(np.where(cur==0)[0])
        cur[i] = 0
        cur[j] = 1
        new_score = exact_score(cur)
        delta = cur_score - new_score  # positive = improvement
        if delta > 0 or rng.random() < np.exp(delta/T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    
    return best, best_score

Now, let me also think about the continuous optimization approach. The LP-based cutting plane method:

At each iteration:
1. Compute the exact profile p(t) for all t.
2. Find the active set A = {t : p(t) ≥ C - ε}.
3. For each active t, compute the gradient ∇p(t) (which is a vector in R^n).
4. Solve the LP:
   minimize t
   subject to: ∇p(t_a) · x ≤ t - p(t_a) for all t_a ∈ A
                0 ≤ x ≤ 1
                sum(x) = n/2
5. Let x* be the LP solution.
6. Line search: h_new = h + α(x* - h), find best α.
7. Update.

The LP has n+1 variables (x_0,...,x_{n-1}, t) and |A| + n + 1 constraints. For n=140 and |A|~20, this is a small LP.

- Variables: x ∈ R^n, t ∈ R. Total: n+1 variables.

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))

            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
