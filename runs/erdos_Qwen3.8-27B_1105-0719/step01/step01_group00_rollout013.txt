            if c5_ep < best_c5 - 1e-10:
                best_c5 = c5_ep
                best_h = make_h_from_q(q_ep, n_parent).copy()
                q0 = q_ep

            h_new = make_h_from_q(q0, n_parent)
            cv_new = corr_dx(h_new)
            t_cur = cv_new[active_idx].max()
            hidden = np.where(cv_new - t_cur > 5e-5)[0]
            if len(hidden) == 0:
                break
            active_idx = np.unique(np.concatenate([active_idx, hidden]))
            if len(active_idx) > 200:
                cv_vals = cv_new[active_idx]
                top_mask = np.argsort(cv_vals)[-200:]
                active_idx = np.sort(active_idx[top_mask])

    # Phase 3: Refine with broader active set
    if time.time() - t0 < budget_s - 60:
        for margin_val in [0.05, 0.03, 0.01]:
            if time.time() - t0 > budget_s - 50:
                break
            h_cur = make_h_from_q(q0, n_parent)
            cv = corr_dx(h_cur)
            active_idx = select_active(cv, top_k=120, margin=margin_val)
            q_ep, c5_ep = run_epigraph(q0, n_parent, active_idx, 600)
            if c5_ep < best_c5 - 1e-10:
                best_c5 = c5_ep
                best_h = make_h_from_q(q_ep, n_parent).copy()
                q0 = q_ep

    # Phase 4: Random restarts in CP space
    for trial in range(40):
        if time.time() - t0 > budget_s - 40:
            break
        if trial < 10:
            scale = 0.03 + 0.02 * trial
        elif trial < 25:
            scale = 0.2 + 0.01 * (trial - 10)
        else:
            scale = 0.35 + 0.01 * (trial - 25)

        q_pert = q0 + rng.randn(m) * scale
        q_pert = np.clip(q_pert, 0.0, 1.0)

        q_opt, c5_val = run_smooth_opt(q_pert, n_parent, 150, 250)
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = make_h_from_q(q_opt, n_parent).copy()
            q0 = q_opt
            if time.time() - t0 < budget_s - 30:
                h_cur = make_h_from_q(q0, n_parent)
                cv = corr_dx(h_cur)
                active_idx = select_active(cv, top_k=70, margin=0.02)
                q_ep, c5_ep = run_epigraph(q0, n_parent, active_idx, 350)
                if c5_ep < best_c5 - 1e-10:
                    best_c5 = c5_ep
                    best_h = make_h_from_q(q_ep, n_parent).copy()
                    q0 = q_ep

    # Phase 5: Structured pattern restarts
    if time.time() - t0 < budget_s - 30:
        for pattern_seed in range(8):
            if time.time() - t0 > budget_s - 25:
                break
            rng_p = np.random.RandomState(seed + 1000 + pattern_seed)
            x = np.linspace(0, 2, m, endpoint=False)
            n_freq = rng_p.randint(1, 5)
            freq = n_freq / 2.0
            phase = rng_p.uniform(0, 2 * np.pi)
            amp = rng_p.uniform(0.1, 0.4)
            q_pat = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
            q_pat += rng_p.randn(m) * 0.05
            q_pat = np.clip(q_pat, 0.0, 1.0)

            q_opt, c5_val = run_smooth_opt(q_pat, n_parent, 200, 300)
            if c5_val < best_c5 - 1e-10:
                best_c5 = c5_val
                best_h = make_h_from_q(q_opt, n_parent).copy()
                q0 = q_opt
                if time.time() - t0 < budget_s - 20:
                    h_cur = make_h_from_q(q0, n_parent)
                    cv = corr_dx(h_cur)
                    active_idx = select_active(cv, top_k=70, margin=0.02)
                    q_ep, c5_ep = run_epigraph(q0, n_parent, active_idx, 350)
                    if c5_ep < best_c5 - 1e-10:
                        best_c5 = c5_ep
                        best_h = make_h_from_q(q_ep, n_parent).copy()
                        q0 = q_ep

    # Phase 6: Try nearby n values
    n_candidates = [n_parent, n_parent - 2, n_parent + 2, n_parent - 4, n_parent + 4]
    n_candidates = [n for n in n_candidates if n >= 50 and n % 2 == 0]

    for n_try in n_candidates:
        if time.time() - t0 > budget_s - 15:
            break
        if n_try == n_parent:
            continue
        m_try = n_try // 2
        x_orig = np.linspace(0, 2, best_h.size, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_orig, best_h)
        q_try = make_q_from_h(h_interp, n_try)

        q_opt, c5_val = run_smooth_opt(q_try, n_try, 200, 350)
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = make_h_from_q(q_opt, n_try).copy()
            q0 = q_opt
            if time.time() - t0 < budget_s - 10:
                h_cur = make_h_from_q(q0, n_try)
                cv = corr_dx(h_cur)
                active_idx = select_active(cv, top_k=70, margin=0.02)
                q_ep, c5_ep = run_epigraph(q0, n_try, active_idx, 350)
                if c5_ep < best_c5 - 1e-10:
                    best_c5 = c5_ep
                    best_h = make_h_from_q(q_ep, n_try).copy()
                    q0 = q_ep

    # Phase 7: Final polish
    if time.time() - t0 < budget_s - 8 and best_h is not None:
        n_b = len(best_h)
        m_b = n_b // 2
        q_b = make_q_from_h(best_h, n_b)

        q_opt, c5_val = run_smooth_opt(q_b, n_b, 2000, 500)
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = make_h_from_q(q_opt, n_b).copy()
            q_b = q_opt

        if time.time() - t0 < budget_s - 3:
            h_cur = make_h_from_q(q_b, n_b)
            cv = corr_dx(h_cur)
            active_idx = np.where(cv >= cv.max() - 0.005)[0]
            if len(active_idx) > 250:
                active_idx = np.argsort(cv)[-250:]
                active_idx = np.sort(active_idx)
            q_ep, c5_ep = run_epigraph(q_b, n_b, active_idx, 500)
            if c5_ep < best_c5 - 1e-10:
                best_c5 = c5_ep
                best_h = make_h_from_q(q_ep, n_b).copy()

    # Final validation
    h_final = np.clip(best_h, 0.0, 1.0)
    n_final = len(h_final)
    sum_err = abs(h_final.sum() - n_final / 2.0)
    if sum_err > 1e-8:
        # Small correction
        diff = n_final / 2.0 - h_final.sum()
        interior = (h_final > 1e-6) & (h_final < 1.0 - 1e-6)
        if interior.sum() > 0:
            h_final[interior] += diff / interior.sum()
            h_final = np.clip(h_final, 0.0, 1.0)
    c5_final = exact_score(h_final)

    return h_final, c5_final, n_final
