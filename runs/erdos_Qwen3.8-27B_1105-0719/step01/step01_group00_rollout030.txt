```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_all(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_sym(q):
    m = len(q)
    return np.concatenate([q, 1.0 - q[::-1]])


def q_from_h_sym(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)


def lag_grad_h(h, k_idx, n, dx):
    """Gradient of corr_values(h)[k_idx] * dx w.r.t. h."""
    d = k_idx - (n - 1)
    g = np.zeros(n)
    # Term 1: (1-h[i-d]) for valid i where i-d in [0,n-1] and i in [0,n-1]
    # Valid: max(0,d) <= i <= min(n-1, n-1+d)
    lo1 = max(0, d)
    hi1 = min(n-1, n-1+d)
    if lo1 <= hi1:
        g[lo1:hi1+1] += (1.0 - h[lo1-d:hi1+1-d])
    # Term 2: -h[i+d] for valid i where i in [0,n-1] and i+d in [0,n-1]
    # Valid: max(0,-d) <= i <= min(n-1, n-1-d)
    lo2 = max(0, -d)
    hi2 = min(n-1, n-1-d)
    if lo2 <= hi2:
        g[lo2:hi2+1] += (-h[lo2+d:hi2+1+d])
    return g * dx


def sym_epigraph_solve(q0, active_lags, n, dx, maxiter=500):
    """Solve min t s.t. t >= F_r(h(q)) for r in active_lags, 0<=q<=1."""
    m = len(q0)
    corrs = corr_all(build_h_sym(q0))
    t0 = corrs[active_lags].max() * dx
    
    def obj(y):
        return y[m]  # t
    
    def obj_grad(y):
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g
    
    def constraint_eval(q, t):
        h = build_h_sym(q)
        c = corr_all(h) * dx
        return t - c[active_lags]
    
    def constraint_jac(q, t):
        h = build_h_sym(q)
        jac = np.zeros((len(active_lags), m + 1))
        for idx, k in enumerate(active_lags):
            g_full = lag_grad_h(h, k, n, dx)
            jac[idx, :m] = -(g_full[:m] - g_full[m:][::-1])
            jac[idx, m] = 1.0
        return jac
    
    def make_con(k_idx):
        def fun(y):
            return constraint_eval(y[:m], y[m])[k_idx]
        def jacobian(y):
            return constraint_jac(y[:m], y[m])[k_idx]
        return {'type': 'ineq', 'fun': fun, 'jac': jacobian}
    
    cons = [make_con(i) for i in range(len(active_lags))]
    bounds = [(0.0, 1.0)] * m + [(0.0, t0 * 2.0 + 1.0)]
    y0 = np.concatenate([q0, [t0]])
    
    res = minimize(obj, y0, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return res

    """Epigraph solver with active constraint selection."""
    m = len(q0)

            active = active[top_idx]
        
        q = q_new
        t_bound = corrs_new.max() + 0.1
    
    return q, exact_score(build_h_sym(q))


def smooth_max_q(q, dx, beta, active_lags):
    """Smooth max surrogate in q-space."""
    h = build_h_sym(q)
    c = corr_all(h) * dx
    ca = c[active_lags]
    return logsumexp(beta * ca) / beta


def smooth_max_q_grad(q, dx, beta, active_lags, n):
    """Gradient of smooth max in q-space."""
    m = len(q)
    h = build_h_sym(q)
    c = corr_all(h) * dx
    ca = c[active_lags]
    w = np.exp(beta * (ca - ca.max()))
    w /= w.sum()
    
    g_q = np.zeros(m)
    for idx in range(len(active_lags)):
        if w[idx] < 1e-12:
            continue
        k = active_lags[idx]
        g_full = lag_grad_h(h, k, n, dx)
        g_q += w[idx] * (g_full[:m] - g_full[m:][::-1])
    return g_q


def smooth_solve_q(q0, n, dx, betas, maxiter_each=300):
    """Solve smooth max problem in q-space for increasing beta."""
    m = len(q0)
    all_lags = np.arange(2*n - 1)
    h = build_h_sym(q0)
    c = corr_all(h) * dx
    # Select top lags
    L = min(80, 2*n-1)
    active = np.argsort(c)[-L:]
    
    q = q0.copy()
    for beta in betas:
        def obj(y):
            h_y = build_h_sym(y)
            c_y = corr_all(h_y) * dx
            return logsumexp(beta * c_y[active]) / beta
        
        def obj_grad(y):
            return smooth_max_q_grad(y, dx, beta, active, n)
        
        bounds = [(0.0, 1.0)] * m
        res = minimize(obj, q, jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter_each, 'gtol': 1e-6})
        if res.x is not None:
            q = np.clip(res.x, 0.0, 1.0)
    
    return q, exact_score(build_h_sym(q))
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_all(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_sym(q):
    m = len(q)
    return np.concatenate([q, 1.0 - q[::-1]])


def q_from_h_sym(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)


def lag_grad_h(h, k_idx, n, dx):
    d = k_idx - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += (1.0 - h[lo1 - d:hi1 + 1 - d])
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += (-h[lo2 + d:hi2 + 1 + d])
    return g * dx


def smooth_obj_q(q, n, dx, beta, active):
    h = build_h_sym(q)
    c = corr_all(h) * dx
    ca = c[active]
    return logsumexp(beta * (ca - ca.max())) / beta + ca.max()


def smooth_grad_q(q, n, dx, beta, active):
    m = len(q)
    h = build_h_sym(q)
    c = corr_all(h) * dx
    ca = c[active]
    w = np.exp(beta * (ca - ca.max()))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w /= w_sum
    g_q = np.zeros(m)
    for i in range(len(active)):
        if w[i] < 1e-13:
            continue
        g_full = lag_grad_h(h, active[i], n, dx)
        g_q += w[i] * (g_full[:m] - g_full[m:][::-1])
    return g_q

    m = len(q0)
    c0 = corr_all(build_h_sym(q0)) * dx
    L = min(80, 2 * n - 1)
    active = np.argsort(c0)[-L:]
    
    q = q0.copy()
    for beta in betas:
        res = minimize(smooth_obj_q, q,
                      jac=lambda x: smooth_grad_q(x, n, dx, beta, active),
                      method='L-BFGS-B', bounds=[(0, 1)] * m,
                      options={'maxiter': maxiter, 'gtol': 1e-8})
        if res.x is not None:
            q = np.clip(res.x, 0.0, 1.0)
    return q


def epigraph_q(q0, n, dx, max_rounds=8, L_init=50, maxiter=400):
    m = len(q0)
    h = build_h_sym(q0)
    c = corr_all(h) * dx
    active = np.argsort(c)[-L_init:]
    q = q0.copy()
    
    for rnd in range(max_rounds):
        # Build constraint functions
        h_cur = build_h_sym(q)
        c_cur = corr_all(h_cur) * dx
        t0 = c_cur[active].max()
        
        def fobj(y):
            return y[m]
        
        def fjac(y):
            g = np.zeros(m + 1)
            g[m] = 1.0
            return g
        
        cons = []
        for idx in range(len(active)):
            k = active[idx]
            def _fun(y, kk=k):
                h_y = build_h_sym(y[:m])
                return y[m] - corr_all(h_y)[kk] * dx
            def _jac(y, kk=k):
                h_y = build_h_sym(y[:m])
                g_f = lag_grad_h(h_y, kk, n, dx)
                row = np.zeros(m + 1)
                row[:m] = -(g_f[:m] - g_f[m:][::-1])
                row[m] = 1.0
                return row
            cons.append({'type': 'ineq', 'fun': _fun, 'jac': _jac})
        
        bounds = [(0.0, 1.0)] * m + [(0.0, t0 + 1.0)]
        y0 = np.concatenate([q, [t0]])
        
        try:

            q_new = q

def full_epigraph(h0, n, dx, max_rounds=5, L_init=40, maxiter=300):
    """Full-dimensional epigraph (no symmetry)."""
    target = n / 2.0
    h = project_feasible(h0, target)
    c = corr_all(h) * dx
    active = np.argsort(c)[-L_init:]
    
    for rnd in range(max_rounds):
        t0 = c[active].max()
        
        def fobj(y):
            return y[n]
        
        def fjac(y):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        cons = []
        for idx in range(len(active)):
            k = active[idx]
            def _fun(y, kk=k):
                h_y = y[:n]
                return y[n] - corr_all(h_y)[kk] * dx
            def _jac(y, kk=k):
                h_y = y[:n]
                g_f = lag_grad_h(h_y, kk, n, dx)
                row = np.zeros(n + 1)
                row[:n] = -g_f
                row[n] = 1.0
                return row
            cons.append({'type': 'ineq', 'fun': _fun, 'jac': _jac})
        
        # Equality constraint: sum(h) = target
        def eq_fun(y):
            return y[:n].sum() - target
        def eq_jac(y):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
        
        bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
        y0 = np.concatenate([h, [t0]])
        
        try:
            res = minimize(fobj, y0, jac=fjac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x[:n], target)
        except Exception:
            h_new = h
        
        c_new = corr_all(h_new) * dx
        t_new = c_new[active].max()
        hidden = np.where(c_new > t_new + 1e-6)[0]
        hidden = hidden[~np.isin(hidden, active)]
        
        h = h_new
        if len(hidden) == 0:
            break

        if len(active) > L_init + 20:
```
