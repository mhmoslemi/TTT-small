    return np.correlate(h, 1.0 - h, mode='full') * dx

def profile_gradient(h, s):
    """Compute gradient of profile at signed lag s. s = t - (n-1) for profile index t."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        # valid i: 0 <= i <= n-1-s
        # first factor: grad[j] += (1-h[j+s]) for j in [0, n-1-s]
        for j in range(n - s):
            grad[j] += (1.0 - h[j + s])
        # second factor: grad[j] -= h[j-s] for j in [s, n-1]
        for j in range(s, n):
            grad[j] -= h[j - s]
    else:
        s_pos = -s
        # valid i: s_pos <= i <= n-1
        # first factor: grad[j] += (1-h[j-s_pos]) for j in [s_pos, n-1]
        for j in range(s_pos, n):

        # second factor: grad[j] -= h[j+s_pos] for j in [0, n-1-s_pos]
        for j in range(n - s_pos):
            grad[j] -= h[j + s_pos]
    return grad * dx

def lp_descent_step(h, profiles, active_indices, trust_radius=None):
    n = len(h)
    C = np.max(profiles)
    
    # Build LP: minimize t
    # vars: [d_0, ..., d_{n-1}, t]
    # constraints: grad_t · d - t <= -p_t for active t
    #              sum(d) = 0
    # bounds: -h[i] <= d[i] <= 1-h[i], t >= 0 (or -inf)
    
    m = len(active_indices)
    # Inequality constraints: A_ub @ x <= b_ub
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, t_idx in enumerate(active_indices):
        s = t_idx - (n - 1)
        g = profile_gradient(h, s)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -profiles[t_idx]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -h[i]
        hi = 1.0 - h[i]
        if trust_radius is not None:
            lo = max(lo, -trust_radius)
            hi = min(hi, trust_radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    d = res.x[:n]
    t_val = res.x[n]
    
    # Expected improvement: C - t_val (since we're minimizing t)
    # If t_val < C, we have improvement
    return d, t_val

For case 2, I can use the fact that the profile is quadratic: p_t(h + d) = p_t(h) + g_t·d + (1/2)d^T H_t d

The quadratic part is -sum h[i]*h[i+s].
The Hessian of -h[i]*h[i+s] has -1 at (i, i+s) and (i+s, i).

So H_s = -A where A has 1s at positions (i, i+s) and (i+s, i) for valid i.

This is a banded matrix (band width s). For s=1, it's like a tridiagonal matrix.

This means the QP is NOT convex! So I can't use standard QP solvers.

Let me also think about the gradient projection approach:
- g = gradient of the max profile (or sum of active gradients)
- We want to move in direction -g subject to sum(d)=0 and box constraints

import numpy as np
from scipy.optimize import linprog
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))


def compute_profiles(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def profile_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        sp = -s
        grad[sp:] += (1.0 - h[:n - sp])
        grad[:n - sp] -= h[sp:]
    return grad * dx


def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.array(h, dtype=np.float64, copy=True)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]) * 0.999)
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, (1.0 - np.max(h[mask])) * 0.999)
            h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    return h


def lp_descent_step(h, profiles, active_indices, trust_r):
    n = len(h)
    m = len(active_indices)
    if m == 0:
        return None
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, t_idx in enumerate(active_indices):
        s = t_idx - (n - 1)
        g = profile_gradient(h, s)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -profiles[t_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = []
    for i in range(n):
        lo = max(-h[i], -trust_r)
        hi = min(1.0 - h[i], trust_r)
        if lo >= hi:
            return None
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    d = res.x[:n]
    t_val = res.x[n]
    return d, t_val


def lp_local_optimize(h, max_rounds=300, deadline=None):
    n = len(h)
    C = exact_score(h)
    trust_r = 0.1
    stag = 0
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline - 10:
            break
        profiles = compute_profiles(h)
        C = float(np.max(profiles))
        delta = max(0.001, 0.01 * C * (1.0 + 0.1 * stag))
        active = np.where(profiles >= C - delta)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profiles)])
        if len(active) > 40:
            idx = np.argsort(-profiles[active])[:40]
            active = active[idx]
        result = lp_descent_step(h, profiles, active, trust_r)
        if result is None:
            trust_r *= 0.5
            stag += 1
            if trust_r < 0.002:
                break
            continue
        d, t_val = result
        if t_val >= C - 1e-10:
            trust_r *= 0.7
            stag += 1
            if trust_r < 0.002:
                break
            continue
        h_new = h + d
        h_new = project_box_sum(h_new)
        C_new = exact_score(h_new)
        if C_new < C - 1e-13:
            h = h_new
            C = C_new
            stag = 0
            trust_r = min(trust_r * 1.2, 0.15)
        else:
            h_half = project_box_sum(h + 0.5 * d)
            C_half = exact_score(h_half)
            if C_half < C - 1e-13:
                h = h_half
                C = C_half
                stag = 0
            else:
                trust_r *= 0.6
                stag += 1
                if trust_r < 0.002:
                    break
    return h


def gradient_projection_step(h, profiles, active_indices, trust_r):
    n = len(h)
    C = float(np.max(profiles))
    g_sum = np.zeros(n)
    for t_idx in active_indices:
        s = t_idx - (n - 1)
        g_sum += profile_gradient(h, s)
    g_sum /= len(active_indices)
    d = -g_sum + np.mean(g_sum)
    d = np.clip(d, -trust_r, trust_r)
    d -= np.mean(d)
    if np.max(np.abs(d)) < 1e-14:
        return None
    h_new = project_box_sum(h + d)
    C_new = exact_score(h_new)
    if C_new < C - 1e-13:
        return h_new, C_new
    h_half = project_box_sum(h + 0.5 * d)
    C_half = exact_score(h_half)
    if C_half < C - 1e-13:
        return h_half, C_half
    return None


def pairwise_transfer(h, n_trials, rng, deadline=None):
    n = len(h)
    C = exact_score(h)
    temp = 0.005
    for t in range(n_trials):
        if deadline is not None and time.time() > deadline - 15:
            break
        frac = t / float(n_trials)
        temp_t = temp * (1.0 - frac) * (1.0 - frac)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        cap = min(1.0 - h[i], h[j])
        if cap < 0.005:
            continue
        amt = rng.uniform(0.01, min(0.3, cap))
        h[i] += amt
        h[j] -= amt
        C_new = exact_score(h)
        if C_new < C - 1e-13:
            C = C_new
        else:
            delta = C_new - C
            if rng.random() < np.exp(-delta / max(temp_t, 1e-10)):
                C = C_new
            else:
                h[i] -= amt
                h[j] += amt
    return h, C


def gradient_informed_transfer(h, n_trials, rng, deadline=None):
    n = len(h)
    C = exact_score(h)
    profiles = compute_profiles(h)
    max_idx = np.argmax(profiles)
    s_max = max_idx - (n - 1)
    g_max = profile_gradient(h, s_max)
    delta = max(0.001, 0.005 * C)
    active = np.where(profiles >= np.max(profiles) - delta)[0]
    g_avg = np.zeros(n)
    for t_idx in active:
        s = t_idx - (n - 1)
        g_avg += profile_gradient(h, s)
    g_avg /= max(1, len(active))
    for t in range(n_trials):
        if deadline is not None and time.time() > deadline - 15:
            break
        hi_candidates = np.where(g_avg > np.percentile(g_avg, 75))[0]
        lo_candidates = np.where(g_avg < np.percentile(g_avg, 25))[0]
        if len(hi_candidates) == 0 or len(lo_candidates) == 0:
            j = rng.randint(n)
            i = rng.randint(n)
        else:
            j = hi_candidates[rng.randint(len(hi_candidates))]
            i = lo_candidates[rng.randint(len(lo_candidates))]
        if i == j:
            continue
        cap = min(1.0 - h[i], h[j])
        if cap < 0.005:
            continue
        amt = rng.uniform(0.02, min(0.15, cap))
        h[i] += amt
        h[j] -= amt
        C_new = exact_score(h)
        if C_new < C - 1e-13:
            C = C_new
        else:
            h[i] -= amt
            h[j] += amt
    return h, C


def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)


def smooth_obj_grad(x, n, tau):
    dx = 2.0 / n
    r = compute_profiles(x)
    r_max = np.max(r)
    shifted = np.clip((r - r_max) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    log_sum = np.log(sum_exp)
    value = r_max + tau * log_sum
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        g = profile_gradient(x, k)
        grad += w * g
    return value, grad


def smooth_slsqp(h, tau, maxiter=80, deadline=None):
    n = len(h)
    dx = 2.0 / n

    def obj_fun(x):
        v, g = smooth_obj_grad(x, n, tau)
        sv = np.sum(x) - n / 2.0
        v = v + 1e4 * sv * sv
        g = g + 2e4 * sv * np.ones(n)
        return v, g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_fun, h, method='SLSQP', jac=True, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if deadline is not None and time.time() > deadline - 10:
            return h
        h_new = project_box_sum(res.x)
        return h_new
    except Exception:
        return h


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b


def binary_swap_improve(b, max_rounds, rng):
    n = len(b)
    cur = b.copy()
    cur_c = exact_score(cur)
    for _ in range(max_rounds):
        best_delta = 0.0
        best_i, best_j = -1, -1
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = exact_score(cur)
                delta = cur_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                cur[i] = 1.0
                cur[j] = 0.0
        if best_i >= 0:
            cur[best_i] = 0.0
            cur[best_j] = 1.0
            cur_c -= best_delta
        else:
            break
    return cur, cur_c
