We want Δcorr < 0, so we want g[b] < g[a].

So a good targeted move is: find a pair (a, b) with g[a] >> g[b] for some active lag, and transfer from a to b.

    n = len(h)
    grads = np.zeros((2*n-1, n))
    for j in range(2*n-1):

            # Second term: p from 0 to n-1-k_abs

def sa_targeted(h, rng, n_steps=5000, T_init=0.015, T_min=0.001, cooling=0.998):
    n = len(h)
    h_cur = h.copy()
    c5_cur = score_c5(h_cur)
    c5_best = c5_cur
    h_best = h_cur.copy()
    T = T_init
    dx = 2.0 / n
    for step in range(n_steps):
        if T < T_min:
            break
        corr = compute_corr(h_cur)
        max_corr = np.max(corr)
        active_idx = np.where(corr >= max_corr - 0.5 * max_corr * 0.08)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        active_j = active_idx[rng.randint(len(active_idx))]
        s = active_j - n + 1
        g = _grad_single(h_cur, s)
        move_type = rng.random()
        if move_type < 0.5:
            pos_indices = np.where(g > 0.1)[0]
            neg_indices = np.where(g < -0.1)[0]
            if len(pos_indices) > 0 and len(neg_indices) > 0:
                i = pos_indices[rng.randint(len(pos_indices))]
                j = neg_indices[rng.randint(len(neg_indices))]
            else:
                i, j = rng.choice(n, size=2, replace=False)
            transfer = rng.uniform(0.02, 0.25) * min(h_cur[i], 1.0 - h_cur[j])
            if transfer < 1e-6:
                T *= cooling
                continue
            h_new = h_cur.copy()
            h_new[i] -= transfer
            h_new[j] += transfer
        elif move_type < 0.8:
            i, j = rng.choice(n, size=2, replace=False)
            transfer = rng.uniform(0.01, 0.15) * min(h_cur[i], 1.0 - h_cur[j])
            if transfer < 1e-6:
                T *= cooling
                continue
            h_new = h_cur.copy()
            h_new[i] -= transfer
            h_new[j] += transfer
        else:
            i, j = rng.choice(n, size=2, replace=False)
            h_new = h_cur.copy()
            h_new[i], h_new[j] = h_cur[j], h_cur[i]
        if h_new.min() < -1e-9 or h_new.max() > 1.0 + 1e-9:
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
        else:
            h_new = project_feasible(h_new, n)
        c5_new = score_c5(h_new)
        delta_c5 = c5_new - c5_cur
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            h_cur = h_new
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_cur.copy()
        T *= cooling
    return h_best, c5_best

def sa_random(h, rng, n_steps=3000, T_init=0.03, T_min=0.002, cooling=0.997):
    n = len(h)
    h_cur = h.copy()
    c5_cur = score_c5(h_cur)
    c5_best = c5_cur
    h_best = h_cur.copy()

            break
        move_type = rng.random()
        if move_type < 0.6:
            i, j = rng.choice(n, size=2, replace=False)
            transfer = rng.uniform(0.01, 0.2) * min(h_cur[i], 1.0 - h_cur[j])
            if transfer < 1e-6:
                T *= cooling
                continue
            h_new = h_cur.copy()
            h_new[i] -= transfer
            h_new[j] += transfer
        else:
            i, j = rng.choice(n, size=2, replace=False)
            h_new = h_cur.copy()

        h_new = np.clip(h_new, 0.0, 1.0)

        c5_new = score_c5(h_new)

        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):

            if c5_cur < c5_best:

                h_best = h_cur.copy()

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    pos_old = np.linspace(0, 2.0, n_old, endpoint=False)
    pos_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(pos_new, pos_old, h_old)
    return project_feasible(h_interp, n_new)

def generate_starts(h_parent, n, rng):
    starts = [h_parent.copy()]
    h_smooth = np.convolve(h_parent, np.ones(max(1, n // 10)) / max(1, n // 10), mode='same')
    h_smooth = project_feasible(h_smooth, n)
    starts.append(h_smooth)
    for k in [3, 5, 8]:
        x = np.arange(n) / n * 2.0
        wave = 0.5 + 0.3 * np.cos(2 * np.pi * k * x / 2.0)
        wave = project_feasible(wave, n)
        starts.append(wave)
    h_rand = rng.randint(0, 2, n).astype(float)
    h_rand = project_feasible(h_rand, n)
    starts.append(h_rand)
    h_pert = h_parent + 0.1 * rng.randn(n)
    h_pert = project_feasible(h_pert, n)
    starts.append(h_pert)
    return starts

def fast_local_optimize(h, budget=30):
    n = len(h)
    c5 = score_c5(h)
    h = h.copy()
    t0 = time.time()
    for it in range(200):
        if time.time() - t0 > budget:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        active = np.where(corr >= max_corr - 0.1 * max_corr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad_sum = np.zeros(n)
        for j in active:
            s = j - n + 1
            grad_sum += _grad_single(h, s)
        grad_sum /= len(active)
        grad_sum -= np.mean(grad_sum)
        step_size = 0.01
        for _ in range(5):
            h_trial = project_feasible(h - step_size * grad_sum, n)
            c5_trial = score_c5(h_trial)
            if c5_trial < c5 - 1e-14:
                h = h_trial
                c5 = c5_trial
                break
            step_size *= 0.5
        if step_size < 1e-6:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    global maxcorr_tol
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(60, 0.5)
    n = len(h_init)
    h_init = project_feasible(h_init, n)

    best_h = h_init.copy()
    best_c5 = score_c5(h_init)

    # Phase 1: SLP from parent
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_slp, c5_slp = slp_optimize(h_init, budget=min(80, rem - 5), rho_init=15.0, R_init=0.05)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp

    # Phase 2: Targeted SA from best, then SLP polish
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        sa_budget = min(200, rem - 20)
        h_sa, c5_sa = sa_targeted(best_h, rng, n_steps=min(20000, int(sa_budget * 30)), T_init=0.012, T_min=0.0008, cooling=0.9985)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            h_pol, c5_pol = fast_local_optimize(best_h, budget=min(20, rem - 5))
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 3: Multiple SA restarts from diverse starts
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        starts = generate_starts(best_h, n, rng)
        for si, h_start in enumerate(starts):
            if time.time() - t0 > budget_s - 30:
                break
            rem_start = budget_s - (time.time() - t0)
            if rem_start < 15:
                break
            sa_steps = min(8000, int(rem_start * 0.4 * 30))
            h_sa, c5_sa = sa_targeted(h_start, rng, n_steps=sa_steps, T_init=0.01, T_min=0.001, cooling=0.998)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa
            rem_start = budget_s - (time.time() - t0)
            if rem_start > 10 and c5_sa < best_c5 + 0.005:
                h_pol, c5_pol = fast_local_optimize(h_sa, budget=min(10, rem_start - 3))
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 4: SLP refinement on best
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_slp, c5_slp = slp_optimize(best_h, budget=min(80, rem - 5), rho_init=10.0, R_init=0.04)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp

    # Phase 5: Try other resolutions
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        n_current = len(best_h)
        for n_try in [n_current - 4, n_current + 4, n_current + 8, n_current - 8]:
            if n_try < 20 or time.time() - t0 > budget_s - 25:
                continue
            h_res = resample_h(best_h, n_try)
            c5_res = score_c5(h_res)
            rem_try = budget_s - (time.time() - t0)
            if rem_try > 15:
                h_pol, c5_pol = fast_local_optimize(h_res, budget=min(15, rem_try - 3))
                if c5_pol < c5_res:
                    c5_res = c5_pol
                    h_res = h_pol
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res

    # Phase 6: Final SLP + SA polish
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_slp, c5_slp = slp_optimize(best_h, budget=min(40, rem - 5), rho_init=5.0, R_init=0.03)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp

    rem = budget_s - (time.time() - t0)
    if rem > 10:
        n_f = len(best_h)
        sa_steps = min(5000, int(rem * 20))
        h_sa, c5_sa = sa_targeted(best_h, rng, n_steps=sa_steps, T_init=0.005, T_min=0.0005, cooling=0.999)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Final evaluation
    final_c5 = score_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = score_c5(best_h)

    return best_h, final_c5, len(best_h)
