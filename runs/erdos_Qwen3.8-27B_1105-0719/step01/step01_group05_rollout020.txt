```python
from scipy.optimize import minimize
import time

def exact_objective(h):

    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return h

    idx_i, idx_j = np.triu_indices(n, k=1)
    n_pairs = len(idx_i)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return h

def binary_local_search(b, deadline, max_rounds=100000):
    """Local search on binary sequences using swap moves."""
    n = len(b)
    b = b.copy()
    c5, profile = exact_objective(b)
    
    ones = list(np.where(b == 1)[0])
    zeros = list(np.where(b == 0)[0])
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 1:
            break
        
        best_c = c5
        best_swap = None
        best_profile = profile
        
        # Try all swaps
        for i in zeros:
            for j in ones:
                # Apply swap
                b[i] = 1
                b[j] = 0
                new_c5, new_profile = exact_objective(b)
                # Undo swap
                b[i] = 0
                b[j] = 1
                
                if new_c5 < best_c - 1e-15:
                    best_c = new_c5
                    best_swap = (i, j)
                    best_profile = new_profile
        
        if best_swap is not None:
            i, j = best_swap
            b[i] = 1
            b[j] = 0
            c5 = best_c
            profile = best_profile
            ones = list(np.where(b == 1)[0])
            zeros = list(np.where(b == 0)[0])
        else:
            break
    
    return b, c5
```
