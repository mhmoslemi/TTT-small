import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    """Compute C5 = max of correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    """Project vec onto the feasible set: 0<=h_i<=1, sum(h_i)=n/2"""
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add 'diff' total, distribute to elements with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        # Need to remove -diff total, take from elements with most value
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    # Fix any residual
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.zeros(M)

        d = m - (n - 1)
        if d == 0:
            profile[m] = dx * np.sum(h * (1 - h))
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:

            G[m, idx1] += dx * (1.0 - h[idx2])
            G[m, idx2] += -dx * h[idx1]
    return profile, G

def sca_step(h0, radius):
    """One SCA step: solve LP and return proposed direction."""
    n = len(h0)
    p, G = build_profile_and_gradient(h0)
    
    # LP: min t s.t. G@m @ z - t <= G@m @ h0 - p[m], sum(z) = n/2, box constraints
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - p
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - radius)
        hi = min(1.0, h0[i] + radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        z = res.x[:n]
        return z
    return None

def line_search(h0, z_target, c_current, alphas=None):
    """Line search from h0 towards z_target."""
    if alphas is None:
        alphas = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    best_h = h0.copy()
    best_c = c_current
    
    for alpha in alphas:
        candidate = h0 + alpha * (z_target - h0)
        candidate = repair_feasible(candidate)
        c = exact_score(candidate)
        if c < best_c - 1e-14:
            best_c = c
            best_h = candidate.copy()
    
    return best_h, best_c

def interpolate_to_n(h, new_n):
    """Interpolate h to new dimension."""
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
