```python
def linearized_direction(h0, r, max_cons=None):
    n = len(h0)
    dx = 2.0 / n
    profile = exact_profile(h0)
    c = max(profile)
    
    # Select active or near-active constraints
    if max_cons is None:
        max_cons = 20
    # Top K profile values
    k = min(max_cons, 2*n-1)
    top_idx = np.argsort(profile)[::-1][:k]
    
    # Compute gradients for top K
    kks = np.arange(-(n-1), n)
    grads = np.zeros((k, n))
    for idx, jj in enumerate(top_idx):
        kk = jj - (n-1)
        for m in range(n):
            a = m + kk
            if 0 <= a < n:
                if a == m:  # kk=0
                    grads[idx, m] += dx * (1 - 2*h0[m])
                else:
                    grads[idx, a] += dx * (1 - h0[m])
                    grads[idx, m] -= dx * h0[a]
    
    # Solve LP: min t s.t. profile[top_idx] + grads @ (h-h0) <= t, 0<=h<=1, sum(h)=n/2
    # Variables: h (n) + t (1) = n+1
    from scipy.optimize import linprog
    
    # c vector: minimize t → c = [0,...,0, 1]
    c_vec = np.zeros(n+1)
    c_vec[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each constraint j: grads[j] @ h - t <= profile[j] - grads[j] @ h0
    # → grads[j] @ h - t <= profile[j] - grads[j] @ h0
    A_ub = np.zeros((k, n+1))
    b_ub = np.zeros(k)
    for idx in range(k):
        A_ub[idx, :n] = grads[idx]
        A_ub[idx, n] = -1.0
        b_ub[idx] = profile[top_idx[idx]] - np.dot(grads[idx], h0)
    
    # Bounds: 0 <= h <= 1, t unbounded below (or t >= 0)
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        direction = h_new - h0
        return direction, res.x[n]
    return None, None
```

```python
def exact_eval(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if target is None:
        target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += residual / cnt
    return np.clip(result, 0.0, 1.0)


def binary_fast_best_swap(b, deadline, max_rounds=1000):
    """Fast greedy best-swap using O(n) delta per swap."""
    n = len(b)
    dx = 2.0 / n
    b = b.copy().astype(np.float64)
    profile = exact_profile(b)
    curr_max = float(np.max(profile))
    kks = np.arange(-(n - 1), n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 1e-15
        best_i, best_j = -1, -1
        best_profile = None
        
        for i in ones:
            # Precompute terms for position i
            i_src_m = (i - kks >= 0) & (i - kks < n)
            i_sink_m = (i + kks >= 0) & (i + kks < n)
            i_contrib = np.zeros(2 * n - 1)
            i_contrib[i_src_m] = -(1.0 - b[i - kks[i_src_m]])
            i_contrib[i_sink_m] += b[i + kks[i_sink_m]]
            
            for j in zeros:

                j_sink_m = (j + kks >= 0) & (j + kks < n)
                # delta = dx * (i_contrib + j_contrib)

                delta = i_contrib * dx

                delta[j_sink_m] -= dx * b[j + kks[j_sink_m]]
                
                new_max = float(np.max(profile + delta))

                if improve > best_improve:

                    best_i, best_j = i, j

        b[best_j] = 1.0
```

Wait, are they independent? If we swap i→j and i'→j', the delta for the first swap assumes h[i]=1, h[j]=0, and the delta for the second assumes h[i']=1, h[j']=0. Since i≠i' and j≠j' (all four positions are distinct: i,i' are 1s and j,j' are 0s), the deltas are indeed independent. So delta_2swap = delta_swap1 + delta_swap2.

This means we can precompute the "contribution" of flipping each position, and then the 2-swap delta is just the sum of four contributions. This makes 2-swap evaluation O(1) per pair of pairs after precomputation!

Let me formalize:
- flip_out[i] = contribution to delta_profile when h[i] goes from 1 to 0
- flip_in[j] = contribution to delta_profile when h[j] goes from 0 to 1

Then:
- Single swap i→j: delta = flip_out[i] + flip_in[j]
- Double swap (i,i')→(j,j'): delta = flip_out[i] + flip_out[i'] + flip_in[j] + flip_in[j']

Precomputing flip_out for all 1-positions and flip_in for all 0-positions is O(n * n) = O(n²) total.

Then:
- Best single swap: O(n² * n) = O(n³) [576 * 95 for n=48]
- Best double swap: O(n⁴ * n) in naive approach, which is too much.
  But we can do: for each pair (i,i'), compute combined_out = flip_out[i] + flip_out[i']. Then for each pair (j,j'), delta = combined_out + flip_in[j] + flip_in[j']. Finding the best is O(C(24,2)² * n) = O(276² * 95) ≈ 7.3M. That's borderline.

    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if target is None:
        target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += residual / cnt
    return np.clip(result, 0.0, 1.0)


def binary_threshold(h):
    n = len(h)
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


class _LSEFunc:
    def __init__(self, tau):
        self.tau = tau

    def value(self, x):
        return _lse_obj(x, self.tau)

    def gradient(self, x):
        return _lse_grad(x, self.tau)


def _lse_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    s = float(np.sum(np.exp(shifted)))
    return m + tau * np.log(s)


def _lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_vals / s
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-12:
            continue
        kk = j - (n - 1)
        m_start = max(0, -kk)
        m_end = min(n, n - kk)
        if m_start >= m_end:
            continue
        ms = np.arange(m_start, m_end)
        as_ = ms + kk
        if kk == 0:
            grad[ms] += w * dx * (1.0 - 2.0 * h[ms])
        else:
            grad[as_] += w * dx * (1.0 - h[ms])
            grad[ms] -= w * dx * h[as_]
    return grad


class _SumEq:
    def __init__(self, target):
        self.target = target

    def fun(self, x):
        return np.sum(x) - self.target

    def jac(self, x):
        return np.ones(len(x))


def lse_refine(h0, tau, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj_func = _LSEFunc(tau)
    con = _SumEq(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj_func.value, h0.copy(), method='SLSQP',
                       jac=obj_func.gradient, bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()


def continuous_refine(h0, deadline):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_eval(h)
    taus = [0.02, 0.008, 0.003, 0.001, 0.0004, 0.00015, 0.00006]
    for tau in taus:
        if time.time() > deadline - 5:
            break
        h_new = lse_refine(h, tau, maxiter=min(300, n * 8))
        c_new = exact_eval(h_new)
        if c_new < curr_c - 1e-14:
            h = h_new
            curr_c = c_new
    return h, curr_c


def transfer_to_size(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)


def make_random_binary(n, rng):
    b = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    b[idx] = 1.0
    return b
