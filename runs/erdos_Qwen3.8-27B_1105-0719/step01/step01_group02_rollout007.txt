Wait: flipping i from 1 to 0 is the NEGATIVE of flipping i from 0 to 1 (since the operation is the same, just reversed). And flipping j from 0 to 1 is δ⁺_s(j).

where δ⁺_s(i) is the change in P_s when h[i] increases by 1 (from 0 to 1, or more generally the derivative-like quantity).

However, for a given h, we can compute δ⁺_s(i) for all (i,s) pairs in O(n²) time, and then for any swap (i,j), the change is just δ⁺_s(j) - δ⁺_s(i) for each s.

So the algorithm is:
1. Compute current profile P (O(n log n) via FFT/correlate)
2. Compute the "influence" matrix D[i,s] = δ⁺_s(i) for all i,s (O(n²))
3. For each candidate swap (i,j) with h[i]=1, h[j]=0:
   ΔP = D[j,:] - D[i,:]  (the change in all lags)
   new_P = P + ΔP
   new_max = max(new_P)
4. Pick the swap that minimizes new_max
5. Apply, repeat

For n=140: 5000 swaps × 279 lags = 1.4M operations per step. In Python, this might take a few seconds per step. If we need ~50-100 steps to converge, that's 100-1500 seconds. Tight but doable.

def binary_local_search(h, max_steps=10000, time_limit=60):
    n = len(h)
    start_time = time.time()
    h = h.copy()
    P = np.correlate(h, 1-h, 'full')
    cur_max = P.max()
    
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        
        # Compute influence matrix D
        # D[i, k] where k is numpy index 0..2n-2
        # D[i, n-1+s] = (1-h[i-s])*(s<=i) - h[i+s]*(i+s<n) for s=0..n-1
        # D[i, n-1-s] = same for s>0
        
        D = np.zeros((n, 2*n-1))
        for s in range(n):
            # Positive lag: D[:, n-1+s]
            # (1-h[i-s]) for i>=s, else 0
            temp = np.zeros(n)
            temp[s:] = 1 - h[:n-s]  # h[i-s] for i=s..n-1
            D[:, n-1+s] = temp
            
            # h[i+s] for i+s<n, i.e., i<n-s, else 0
            temp2 = np.zeros(n)
            temp2[:n-s] = h[s:]  # h[i+s] for i=0..n-s-1
            D[:, n-1+s] -= temp2
            
            if s > 0:
                D[:, n-1-s] = D[:, n-1+s]  # same value
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # For all pairs (i,j), compute new_max
        # new_max[i,j] = max(P + D[j,:] - D[i,:])
        # = max_k (P[k] + D[j,k] - D[i,k])
        
        # A[i,k] = P[k] - D[ones[i],k], shape (len(ones), 2n-1)
        # B[j,k] = D[zeros[j],k], shape (len(zeros), 2n-1)
        # C[a,b] = max_k (A[a,k] + B[b,k])
        
        A = P[None, :] - D[ones, :]  # (n/2, 2n-1)
        B = D[zeros, :]  # (n/2, 2n-1)
        C = A[:, None, :] + B[None, :, :]  # (n/2, n/2, 2n-1)
        new_maxes = C.max(axis=2)  # (n/2, n/2)
        
        best_val = new_maxes.min()
        if best_val >= cur_max - 1e-10:
            break  # No improvement possible
        
        # Find the best pair
        best_idx = np.unravel_index(new_maxes.argmin(), new_maxes.shape)
        i = ones[best_idx[0]]
        j = zeros[best_idx[1]]
        
        h[i] = 0
        h[j] = 1
        P = P + D[j, :] - D[i, :]
        cur_max = P.max()
    
    return h, cur_max * (2.0/n)

def exact_objective(h):
    n = len(h)

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    if target is None:
        target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-12:

        if diff > 0:
