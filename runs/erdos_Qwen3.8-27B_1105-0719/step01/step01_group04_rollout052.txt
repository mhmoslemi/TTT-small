```python
def newton_polish(h, active_idx, max_iter=20):
    """
    Solve: g_k(h) = C for k in active_idx, sum(h) = n/2, minimize ||delta||²
    Using Gauss-Newton linearization.
    """
    n = len(h)
    h = h.copy()
    C = exact_score(h)
    
    for it in range(max_iter):
        vals = compute_lag_values(h)
        active_vals = vals[active_idx]
        # Residual: active_vals should all equal C
        # Also: sum(h) - n/2 = 0
        # Linearize: vals_k(h + delta) ≈ vals_k(h) + grad_k(h) . delta
        # Want: vals_k(h) + grad_k(h).delta = C for all k in active
        # And: sum(h) + sum(delta) = n/2, i.e., sum(delta) = 0 (if already feasible)
        
        # Set up: [grads; ones] @ delta = [C - active_vals; 0]
        # This is (|A|+1) x n system, underdetermined. Solve min ||delta|| s.t. ...
        
        grads = compute_lag_gradients(h)
        G = np.vstack([grads[active_idx, :], np.ones((1, n))])
        rhs = np.concatenate([C - active_vals, [0.0]])
        
        # Least squares solution
        delta, _, _, _ = np.linalg.lstsq(G, rhs, rcond=None)
        
        # Scale down for safety
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            h_try = repair_feasible(h + alpha * delta)
            s_try = exact_score(h_try)
            if s_try < C - 1e-15:
                h = h_try
                C = s_try
                break
        else:
            break
    
    return h, exact_score(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h


def compute_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def compute_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    grads[n - 1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        grads[n - 1 + d, d:] += dx * (1.0 - h[:-d])
        grads[n - 1 + d, :n - d] += -dx * h[d:]
        grads[n - 1 - d, :n - d] += dx * (1.0 - h[d:])
        grads[n - 1 - d, d:] += -dx * h[:-d]
    return grads


def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)


def solve_slqp_active(h, active_idx, maxiter=300, ftol=1e-15, t_margin=0.01):
    n = len(h)
    h = repair_feasible(h)

    def obj_func(x):
        return x[n]

    def obj_jac_func(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        t = x[n]
        vals = compute_lag_values(h_x)
        return t - vals[active_idx]

    def ineq_jac(x):
        h_x = x[:n]
        _, grads = compute_lag_gradients(h_x)
        g = np.zeros((len(active_idx), n + 1))
        g[:, n] = 1.0
        g[:, :n] = -grads[active_idx, :]
        return g

    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    c_start = exact_score(h)
    t0 = c_start + t_margin
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    try:
        result = minimize(
            obj_func, x0, jac=obj_jac_func, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if result.x is not None:
            h_sol = repair_feasible(result.x[:n])
            score = exact_score(h_sol)
            if np.isfinite(score):
                return h_sol, score
    except Exception:
        pass
    return None, None


def get_active_set(h, top_k=15, gap_frac=0.0):

        top_vals = vals[active]
        order = np.argsort(top_vals)[::-1][:top_k]
        active = active[order]
    return np.sort(active)

def newton_polish(h, active_idx, max_iter=30):
    """Gauss-Newton step to equalize active lags."""
    n = len(h)
    h = h.copy()
    C = exact_score(h)
    
    for _ in range(max_iter):
        vals = compute_lag_values(h)
        active_vals = vals[active_idx]
        C_cur = np.max(vals)
        # Linearize: want grad_k . delta = C_target - vals_k for active k
        # Also sum(delta) = 0
        grads = compute_lag_gradients(h)
        G = np.vstack([grads[active_idx, :], np.ones((1, n))])
        C_target = C_cur
        rhs = np.concatenate([C_target - active_vals, [0.0]])
        try:
            delta, _, _, _ = np.linalg.lstsq(G, rhs, rcond=None)
        except Exception:
            break
        # Ensure zero mean for sum constraint
        delta -= np.mean(delta)
        improved = False
        for alpha in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            h_try = repair_feasible(h + alpha * delta)
            s_try = exact_score(h_try)
            if s_try < C - 1e-15:
                h = h_try
                C = s_try
                improved = True
                break
        if not improved:
            break
    return h, exact_score(h)

        c = np.max(vals)
        n_act = min(top_k, 2 * n - 1)
        idx = np.argsort(vals)[::-1][:n_act]
        gaps = np.maximum(c - vals[idx], 0.0)
        
        # Multiple weight schemes
        for temp in [temp_scale, 2.0 * temp_scale, 0.5 * temp_scale]:
            if step < max_steps - 1:
                temp = temp * (1.0 + 0.1 * step)
            weights = np.exp(-gaps / (temp * c + 1e-12))
            weights /= np.sum(weights) + 1e-15
            
            _, grads = compute_lag_gradients(best)
            direction = -np.sum(weights[:, None] * grads[idx, :], axis=0)
            direction -= np.mean(direction)
            
            improved = False
            for alpha in [0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]:
                h_try = repair_feasible(best + alpha * direction)
                score = exact_score(h_try)
                if score < best_score - 1e-14:
                    best = h_try
                    best_score = score
                    improved = True
                    break
            if improved:
                break
    
    return best, best_score


def slqp_full(h, maxiter=400, ftol=1e-15):
    """SLSQP with all constraints (for small n)."""
    n = len(h)
    if n > 100:
        return None, None
    h = repair_feasible(h)
    n_lags = 2 * n - 1
    all_idx = np.arange(n_lags)
    return solve_slqp_active(h, all_idx, maxiter=maxiter, ftol=ftol)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    def hard_deadline():
        return budget_s - 10.0

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_h = repair_feasible(parent)
    else:
        best_h = repair_feasible(np.full(70, 0.5))

    best_score = exact_score(best_h)
    best_n = len(best_h)

    # Try symmetrization
    h_sym = repair_feasible((best_h + best_h[::-1]) / 2.0)
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-14:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: Tighten at parent dimension
    if time_left() > 60:
        h_new, s_new = tighten_loop(best_h, budget_sec=min(200, time_left() - 20))
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new
            best_n = len(h_new)

    # Newton polish at current dimension
    if time_left() > 30:
        vals = compute_lag_values(best_h)
        c_max = np.max(vals)
        for gap_frac in [0.0, 0.002, 0.005]:
            if time_left() < 20:
                break
            active = get_active_set(best_h, top_k=10, gap_frac=gap_frac)
            h_new, s_new = newton_polish(best_h, active, max_iter=20)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new

    # Phase 2: Active descent
    if time_left() > 30:
        for top_k in [8, 12, 16]:
            if time_left() < 20:
                break
            h_new, s_new = active_descent(best_h, max_steps=40, top_k=top_k)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new

    # Phase 3: Dimension exploration
    if time_left() > 60:
        n_cur = len(best_h)
        # Try coarser dimensions
        for nd in [70, 84, 98, int(n_cur * 0.75), int(n_cur * 0.6)]:
            if time_left() < 30:
                break
            if nd < 30 or nd >= n_cur:
                continue
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.005:
                # Optimize at this dimension
                if nd <= 80:
                    h_sol, score = slqp_full(h_interp, maxiter=300)
                else:
                    active = get_active_set(h_interp, top_k=12, gap_frac=0.01)
                    h_sol, score = solve_slqp_active(h_interp, active, maxiter=200)
                if h_sol is not None and score < best_score - 1e-14:
                    best_h = h_sol
                    best_score = score
                    best_n = len(h_sol)
                else:
                    # Try newton polish
                    active = get_active_set(h_interp, top_k=10, gap_frac=0.005)
                    h_sol, score = newton_polish(h_interp, active, max_iter=15)
                    if score < best_score - 1e-14:
                        best_h = h_sol
                        best_score = score
                        best_n = len(h_sol)

        # Try finer dimensions (upsample)
        for nd in [int(n_cur * 1.2), int(n_cur * 1.5), n_cur + 28]:
            if time_left() < 30:
                break
            if nd > 200 or nd <= n_cur:
                continue
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.008:
                active = get_active_set(h_interp, top_k=12, gap_frac=0.01)
                h_sol, score = solve_slqp_active(h_interp, active, maxiter=150)
                if h_sol is not None and score < best_score - 1e-14:
                    best_h = h_sol
                    best_score = score
                    best_n = len(h_sol)
                    h_new, s_new = newton_polish(h_sol, active, max_iter=10)
                    if s_new < best_score - 1e-14:
                        best_h = h_new
                        best_score = s_new

    # Phase 4: Multi-start perturbation
    if time_left() > 50:
        for trial in range(10):
            if time_left() < 30:
                break
            eps = rng.uniform(0.001, 0.008)
            z = rng.uniform(-1.0, 1.0, size=len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            # Quick SLSQP
            active = get_active_set(h_pert, top_k=10, gap_frac=0.01)
            h_sol, score = solve_slqp_active(h_pert, active, maxiter=100, ftol=1e-14)
            if h_sol is not None and score < best_score - 1e-14:
                best_h = h_sol
                best_score = score
                best_n = len(h_sol)
            # Also try the perturbed point directly with active descent
            if score is None or score > best_score:
                h_ad, s_ad = active_descent(h_pert, max_steps=20, top_k=10)
                if s_ad < best_score - 1e-14:
                    best_h = h_ad
                    best_score = s_ad

    # Phase 5: Final tightening cycle
    if time_left() > 40:
        h_new, s_new = tighten_loop(best_h, budget_sec=min(150, time_left() - 15))
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new
            best_n = len(h_new)

    # Phase 6: Final newton polish + active descent
    if time_left() > 20:
        vals = compute_lag_values(best_h)
        active = get_active_set(best_h, top_k=8, gap_frac=0.001)
        h_new, s_new = newton_polish(best_h, active, max_iter=25)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new
        h_new, s_new = active_descent(best_h, max_steps=30, top_k=10, temp_scale=0.002)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new

    # Phase 7: One more SLSQP pass
    if time_left() > 20:
        vals = compute_lag_values(best_h)
        c_max = np.max(vals)
        active = get_active_set(best_h, top_k=15, gap_frac=0.005)
        h_sol, score = solve_slqp_active(best_h, active, maxiter=300, ftol=1e-15)
        if h_sol is not None and score < best_score - 1e-14:
            best_h = h_sol
            best_score = score
        # Newton again
        if time_left() > 10:
            active2 = get_active_set(best_h, top_k=7, gap_frac=0.0)
            h_new, s_new = newton_polish(best_h, active2, max_iter=20)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    if not np.isfinite(best_score):
        best_h = repair_feasible(np.full(50, 0.5))
        best_score = exact_score(best_h)

    return best_h, best_score, len(best_h)
```
