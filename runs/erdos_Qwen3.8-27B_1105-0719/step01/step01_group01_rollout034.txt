```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    """Smooth max of correlation * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    log_sum = m + np.log(np.sum(np.exp(shifted))) / beta
    return log_sum * dx

def smooth_max_grad(h, beta):
    """Gradient of smooth max objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-15:

        if lag >= 0:

            # d/dh[i] for i in [0, n-1-lag]: (1-h[i+lag])
            # d/dh[i+lag] for i in [0, n-1-lag]: -h[i]
            i_max = n - 1 - lag
            grad[0:i_max+1] += w * (1.0 - h[lag:i_max+1+lag])
            grad[lag:i_max+1+lag] -= w * h[0:i_max+1]
        else:

            # where lag < 0, so i+lag = i-|lag|
            # = sum_{i=|lag|}^{n-1} h[i] * (1-h[i-|lag|])

            grad[0:n-abs_lag] -= w * h[abs_lag:n]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    val = m + np.log(np.sum(np.exp(shifted))) / beta
    return val * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(2 * n - 1)  # gradient w.r.t. correlation values
    # Actually we need gradient w.r.t. h
    # corr[k] = sum valid h[i]*(1-h[i+lag]) where lag = k-(n-1)
    # d(corr[k])/d(h[x]) = sum over valid pairs involving x
    
    h_grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            i_end = n - lag
            # d/dh[i]: (1-h[i+lag]) for i in [0, i_end-1]
            # d/dh[i+lag]: -h[i] for i in [0, i_end-1], i.e., d/dh[x]: -h[x-lag] for x in [lag, n-1]
            h_grad[:i_end] += w * (1.0 - h[lag:i_end+lag])
            h_grad[lag:] -= w * h[:i_end]
        else:
            # lag < 0, let a = -lag
            a = -lag
            # corr = sum_{i=a}^{n-1} h[i]*(1-h[i-a])
            # d/dh[i]: (1-h[i-a]) for i in [a, n-1]
            # d/dh[i-a]: -h[i] for i in [a, n-1], i.e., d/dh[x]: -h[x+a] for x in [0, n-1-a]
            h_grad[a:] += w * (1.0 - h[:n-a])
            h_grad[:n-a] -= w * h[a:]
    
    return h_grad * dx

def eq_con_fun(h):
    return np.sum(h) - len(h) / 2.0

def eq_con_jac(h):
    return np.ones(len(h))

def project_feasible(y, n):
    """Project y to [0,1]^n with sum = n/2."""
    target = n / 2.0
    # Bisection on threshold
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute residual among interior points
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(interior) > 0:
            h[interior] += residual / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
        else:
            # All at boundaries, need to adjust
            # Find a point to adjust
            if residual > 0:
                # Need to increase sum, find a point with h < 1
                candidates = np.where(h < 1.0 - 1e-10)[0]
                if len(candidates) > 0:
                    h[candidates] += residual / len(candidates)
            else:
                candidates = np.where(h > 1e-10)[0]
                if len(candidates) > 0:
                    h[candidates] += residual / len(candidates)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_refine(h0, betas, max_iter=300):
    """Run SLSQP with increasing beta."""
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    
    for beta in betas:
        try:
            res = minimize(
                smooth_obj, best_h, args=(beta,),
                jac=smooth_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-14}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_proj = project_feasible(res.x, n)
                c5_new = compute_c5(h_proj)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_proj
        except Exception:
            pass
    
    return best_h, best_c5

def local_search(h, n_iterations, rng, step_size=0.05):
    """Local search with perturb moves."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for _ in range(n_iterations):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = step_size * (1 if rng.random() < 0.5 else -1)
        
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Check bounds
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5

def sa_search(h, n_iterations, rng, T_init=0.01, T_min=0.001):
    """Simulated annealing with perturb moves."""
    n = len(h)
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    
    T = T_init
    cool_rate = (T_min / T_init) ** (1.0 / max(n_iterations, 1))
    
    for it in range(n_iterations):
        T *= cool_rate
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Random perturbation
        magnitude = 0.02 + 0.08 * rng.random()
        delta = magnitude * (1 if rng.random() < 0.5 else -1)
        
        h_new = cur_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        
        c5_new = compute_c5(h_new)
        delta_c5 = c5_new - cur_c5
        
        if delta_c5 < 0 or (T > 1e-8 and rng.random() < np.exp(-delta_c5 / T)):
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return best_h, best_c5

def binary_sa(n, n_iterations, rng, T_init=3.0, T_min=0.1):
    """Simulated annealing for binary solutions."""
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    best_h = h.copy()
    best_mc = max_corr
    
    T = T_init
    cool = (T_min / T_init) ** (1.0 / n_iterations)
    
    for it in range(n_iterations):
        T *= cool
        # Pick random 1 and 0
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode="full")
        new_mc = np.max(corr)
        delta = new_mc - max_corr
        
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            max_corr = new_mc
            if new_mc < best_mc:
                best_mc = new_mc
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_mc
```

```python
def exact_max_grad(h):
    """Compute subgradient of max(corr) at h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Find all lags achieving (near) the max
    active = np.where(corr > max_val - 1e-10)[0]
    # Average the gradients of active lags
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            i_end = n - lag
            grad[:i_end] += (1.0 - h[lag:i_end+lag])
            grad[lag:] -= h[:i_end]
        else:
            a = -lag
            grad[a:] += (1.0 - h[:n-a])
            grad[:n-a] -= h[a:]
    grad /= len(active)
    return grad * dx
```
