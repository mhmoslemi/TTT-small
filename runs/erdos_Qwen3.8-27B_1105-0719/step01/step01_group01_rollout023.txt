def active_lag_local_search(h, n_iterations, rng, tol=0.01):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_corr = get_correlation(best_h)
    best_max_corr = np.max(best_corr)
    
    for it in range(n_iterations):
        # Find active lags
        max_val = np.max(best_corr)
        active_mask = best_corr > max_val - tol * max_val
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) == 0:
            break
        
        # Pick a random active lag
        lag_idx = active_lags[rng.randint(len(active_lags))]
        k = lag_idx - (n - 1)  # actual lag
        
        # For this lag, find good transfer pairs
        if k >= 0:
            # corr = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
            # To decrease: transfer from i to i+k where h[i] is high and h[i+k] is low

    """Take a subgradient step to reduce max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Find active lags (within tolerance of max)
    max_val = np.max(corr)
    tol = 0.05 * max_val if max_val > 0 else 0.01
    active = np.where(corr >= max_val - tol)[0]
    
    # Compute subgradient (average gradient over active lags)
    sgrad = np.zeros(n)
    for p in active:
        k = p - (n - 1)
        grad_k = compute_lag_gradient(h, k, n)
        sgrad += grad_k
    sgrad /= len(active)
    
    # Take step
    h_new = h - step_size * sgrad
    # Project to feasible set
    h_new = project_feasible(h_new, n)
    return h_new

                    best_h = res.x.copy()
        except:
            pass
    return best_h, best_c5

def subgradient_refine(h, n_iter, step_init=0.01, rng=None):
    """Subgradient method targeting active lags."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = step_init
    
    for it in range(n_iter):
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_val = float(np.max(corr))
        
        # Find active lags (within 5% of max)
        tol = 0.05 * max_val
        active = np.where(corr >= max_val - tol)[0]
        
        if len(active) == 0:
            break
        
        # Compute subgradient direction
        sgrad = np.zeros(n)
        for p in active:
            k = p - (n - 1)
            # Compute gradient of corr[p] w.r.t. h
            if k >= 0:
                # corr = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
                # For each m:
                #   if m >= k and m < n: (1-h[m-k])  [h[m] as h[i+k]]
                #   if m < n-k: -h[m+k]  [h[m] as h[i]]
                for m in range(n):
                    g = 0.0
                    if m >= k:
                        g += (1.0 - best_h[m-k])
                    if m < n - k:
                        g += (-1.0) * best_h[m+k]
                    sgrad[m] += g
            else:
                kp = -k
                # corr = sum_{i=0}^{n-1-kp} h[i]*(1-h[i+kp])
                for m in range(n):
                    g = 0.0
                    if m < n - kp:
                        g += (1.0 - best_h[m+kp])
                    if m >= kp:
                        g += (-1.0) * best_h[m-kp]
                    sgrad[m] += g
        
        sgrad /= len(active)
        
        # Take step and project
        h_new = best_h - step * sgrad
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            step = min(step * 1.1, 0.1)  # Increase step slightly
        else:
            step *= 0.7  # Decrease step
        
        if step < 1e-8:
            break
    
    return best_h, best_c5

def targeted_local_search(h, n_iter, rng):
    """Local search that targets the worst lag."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    best_max = float(np.max(best_corr))
    
    for it in range(n_iter):
        # Find the worst lag
        worst_p = int(np.argmax(best_corr))
        k = worst_p - (n - 1)
        
        # Also consider nearby lags
        active = np.where(best_corr >= best_max - 0.01 * best_max)[0]
        
        # For the worst lag, find the best (src, dst) pair to transfer
        best_improvement = 0
        best_move = None
        
        # Sample some random pairs
        for _ in range(5):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            
            # Try transferring from i to j
            for direction in [1, -1]:
                if direction == 1:
                    # h[i] -= t, h[j] += t
                    max_t = min(best_h[i], 1.0 - best_h[j])
                else:
                    # h[i] += t, h[j] -= t
                    max_t = min(1.0 - best_h[i], best_h[j])
                
                if max_t < 1e-10:
                    continue
                
                # Try a few step sizes
                for frac in [0.1, 0.3, 0.5, 0.8]:
                    t = frac * max_t
                    h_try = best_h.copy()
                    if direction == 1:
                        h_try[i] -= t
                        h_try[j] += t
                    else:
                        h_try[i] += t
                        h_try[j] -= t
                    
                    if h_try[i] < -1e-12 or h_try[i] > 1 + 1e-12:
                        continue
                    if h_try[j] < -1e-12 or h_try[j] > 1 + 1e-12:
                        continue
                    
                    new_corr = np.correlate(h_try, 1.0 - h_try, mode="full")
                    new_max = float(np.max(new_corr))
                    improvement = best_max - new_max
                    
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (i, j, direction, t)
        
        if best_move is not None and best_improvement > 1e-12:
            i, j, direction, t = best_move
            best_h[i] += t * direction
            best_h[j] -= t * direction
            best_h = np.clip(best_h, 0, 1)
            best_corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            best_max = float(np.max(best_corr))
    
    return best_h, best_max * dx
