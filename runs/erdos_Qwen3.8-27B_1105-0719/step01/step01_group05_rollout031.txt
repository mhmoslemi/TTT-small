def solve_1d_convex_min(a, b, c, delta_lo, delta_hi, max_iter=50):
    """Minimize g(δ) = max_k(a[k]δ² + b[k]δ + c[k]) over [delta_lo, delta_hi].
    Each f_k is convex (a[k] >= 0) or linear (a[k] = 0).
    Returns (delta_opt, g_opt)."""
    m = len(a)
    if m == 0:
        return 0.0, 0.0
    if delta_lo >= delta_hi:
        d = delta_lo
        val = np.max(a * d * d + b * d + c)
        return d, val
    
    lo = delta_lo
    hi = delta_hi
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        vals = a * mid * mid + b * mid + c
        max_val = np.max(vals)
        # Active set: those with value close to max
        active = vals >= max_val - 1e-12
        if not np.any(active):
            active = np.ones(m, dtype=bool)
        derivs = 2.0 * a[active] * mid + b[active]
        min_d = np.min(derivs)
        max_d = np.max(derivs)
        if min_d > 1e-12:
            hi = mid
        elif max_d < -1e-12:
            lo = mid
        else:
            break
    d_opt = (lo + hi) / 2.0
    g_opt = np.max(a * d_opt * d_opt + b * d_opt + c)
    return d_opt, g_opt
