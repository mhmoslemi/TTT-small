import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add diff total, spread to entries with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        # Need to remove -diff total
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def fw_vertex(g, n):
    """Find v in P_n that minimizes g^T v.
    P_n = {v : 0 <= v_i <= 1, sum v_i = n/2}
    """
    target = n / 2.0
    order = np.argsort(g)
    v = np.zeros(n)
    remaining = target
    for idx in order:
        if remaining >= 1.0 - 1e-15:
            v[idx] = 1.0
            remaining -= 1.0
        else:
            v[idx] = remaining
            remaining = 0.0
            break
    return v

def smooth_profile(h, tau):
    """Log-sum-exp of profile entries, giving smooth approximation to max."""
    p = profile(h)
    mu = np.max(p)
    if tau < 1e-10:
        return mu
    shifted = (p - mu) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    return mu + tau * np.log(np.sum(np.exp(shifted)))

def smooth_grad(h, tau):
    """Gradient of smooth profile objective."""
    n = len(h)
    dx = 2.0 / n
    p = profile(h)
    mu = np.max(p)
    shifted = (p - mu) / tau
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w /= w.sum()
    G = build_grad_matrix(h, n, dx)
    return G.T @ w

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def fw_anneal(h, tau_start=0.2, tau_end=0.01, n_iters=200, time_fn=None):
    """Frank-Wolfe with annealing temperature."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    tau = tau_start
    
    for it in range(n_iters):
        if time_fn is not None and time_fn() > 0.9:
            break
        
        # Current smooth objective and gradient
        g = smooth_grad(h, tau)
        
        # Frank-Wolfe: find vertex minimizing g^T v
        v = fw_vertex(g, n)
        
        # Check if we're at a vertex (no improvement possible)
        g_dot = g @ (v - h)
        if g_dot >= -1e-12:
            # Can't improve via FW step, reduce tau
            tau = max(tau_end, tau * 0.7)
            continue
        
        # Line search: try several alpha values
        dirv = v - h
        best_alpha = 0.0
        best_f = smooth_profile(h, tau)
        
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h_try = h + alpha * dirv
            f_try = smooth_profile(h_try, tau)
            if f_try < best_f - 1e-12:
                best_f = f_try
                best_alpha = alpha
        
        if best_alpha > 0:
            h = h + best_alpha * dirv
            h = np.clip(h, 0, 1)
            h = repair_feasible(h)
            c = exact_score(h)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
        
        # Anneal
        if it < n_iters // 3:
            tau = tau_start  # Keep warm
        elif it < 2 * n_iters // 3:
            tau = max(tau_end, tau_start * 0.5)
        else:
            tau = max(tau_end, tau * 0.85)
    
    return best_h, best_c

def active_set_descent(h, max_iters=100):
    """Subgradient descent focusing on active (near-max) profile entries."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        p = profile(h)
        c = np.max(p)
        
        # Active set: entries within threshold of max
        thresh = c - 0.005
        active = np.where(p >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(p)])
        
        # Weighted gradient
        w = p[active] - (thresh - 0.001)
        w = np.maximum(w, 0.0)
        if w.sum() < 1e-15:
            w = np.ones(len(active)) / len(active)
        else:
            w /= w.sum()
        
        G = build_grad_matrix(h, n, dx)
        grad = G[active, :].T @ w
        
        # Center gradient (to preserve sum constraint approximately)
        grad -= grad.mean()
        
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        
        # Backtracking line search
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = h - alpha * grad
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                improved = True
                break
        
        if not improved:
            break
    
    return h, best_c

def slsqp_epigraph(h_start, maxiter=200, tol=1e-10):
    """SLSQP with epigraph formulation."""
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        p = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - p
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_grad_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            if h_new.sum() > 0 and np.all(h_new >= -1e-12):
                return h_new
    except Exception:
        pass
    return None

def exact_score(h):

def profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_rows(h, n, dx, rows):

    for ii, m in enumerate(rows):

            G[ii, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[ii, d:n] += dx * (1.0 - h[0:n - d])
            G[ii, 0:n - d] += -dx * h[d:n]
        else:
            dp = -d
            G[ii, 0:n - dp] += dx * (1.0 - h[dp:n])
            G[ii, dp:n] += -dx * h[0:n - dp]

def fw_vertex(g, n):
    target = n / 2.0
    order = np.argsort(g)
    v = np.zeros(n)
    remaining = target
    for idx in order:
        if remaining >= 1.0 - 1e-15:
            v[idx] = 1.0
            remaining -= 1.0
        else:
            v[idx] = remaining
            remaining = 0.0

def smooth_value(h, tau):
    p = profile(h)
    mu = np.max(p)
    if tau < 1e-10:
        return mu
    shifted = np.clip((p - mu) / tau, -80, 80)
    return mu + tau * np.log(np.sum(np.exp(shifted)))

def smooth_grad_vec(h, tau):

    rows = np.arange(M)
    G = build_grad_rows(h, n, dx, rows)

def interpolate_to_n(h, new_n):

        for alpha in [0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002]:
            h_try = repair_feasible(h - alpha * grad)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                improved = True
                break
        if not improved:
            break
    return h, best_c

def slsqp_full(h_start, maxiter, tol):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    rows_all = np.arange(M)

    def s_obj(x):
        return x[-1]

    def s_obj_jac(x):
        arr = np.zeros(n + 1)
        arr[-1] = 1.0
        return arr

    def s_eq(x):
        return x[:-1].sum() - n / 2.0

    def s_eq_jac(x):
        arr = np.zeros(n + 1)
        arr[:-1] = 1.0
        return arr

    def s_ineq(x):
        hv = x[:-1]
        pv = np.correlate(hv, 1.0 - hv, mode="full") * dx
        return x[-1] - pv

    def s_ineq_jac(x):
        hv = x[:-1]
        G = build_grad_rows(hv, n, dx, rows_all)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': s_eq, 'jac': s_eq_jac},
        {'type': 'ineq', 'fun': s_ineq, 'jac': s_ineq_jac}
    ]
    try:
        res = minimize(s_obj, x0, method='SLSQP', jac=s_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            return repair_feasible(res.x[:-1])
    except Exception:
        pass
    return None

def slsqp_cycling(h_start, K, n_cycles, maxiter_per_cycle, tol):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    best_c = exact_score(h)
    for _ in range(n_cycles):
        p = profile(h)
        c_cur = np.max(p)
        rows_k = np.argsort(-p)[:K].copy()
        c0 = c_cur

        def c_obj(x):
            return x[-1]

        def c_obj_jac(x):
            arr = np.zeros(n + 1)
            arr[-1] = 1.0
            return arr

        def c_eq(x):
            return x[:-1].sum() - n / 2.0

        def c_eq_jac(x):
            arr = np.zeros(n + 1)
            arr[:-1] = 1.0
            return arr

        def c_ineq(x):
            hv = x[:-1]
            pv = np.correlate(hv, 1.0 - hv, mode="full") * dx
            return x[-1] - pv[rows_k]

        def c_ineq_jac(x):
            hv = x[:-1]
            G = build_grad_rows(hv, n, dx, rows_k)
            J = np.zeros((K, n + 1))
            J[:, :-1] = -G
            J[:, -1] = 1.0
            return J

        x0 = np.concatenate([h, [c0 + 1e-5]])
        bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
        cons = [
            {'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac},
            {'type': 'ineq', 'fun': c_ineq, 'jac': c_ineq_jac}
        ]
        try:
            res = minimize(c_obj, x0, method='SLSQP', jac=c_obj_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_cycle, 'ftol': tol})
            if res.x is not None and np.isfinite(res.x).all():
                h_new = repair_feasible(res.x[:-1])
                c_new = exact_score(h_new)
                if c_new < best_c - 1e-14:
                    h = h_new
                    best_c = c_new
                else:
                    return h, best_c
            else:
                return h, best_c
        except Exception:
            break
    return h, best_c

def symmetrize(h):
    return (h + h[::-1]) / 2.0
