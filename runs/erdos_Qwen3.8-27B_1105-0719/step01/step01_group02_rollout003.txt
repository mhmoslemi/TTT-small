def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx

def grad_profile(h, k):
    """Analytical gradient of raw profile element at lag k."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # c[k] = sum_{i=0}^{n-k-1} h[i+k](1-h[i])
        # d/dh[j]: if j >= k, contributes (1-h[j-k]) (from h[j] as first factor)
        #          if j < n-k, contributes -h[j+k] (from h[j] in second factor)
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        m = -k
        # c[-m] = sum_{i=0}^{n-m-1} h[i](1-h[i+m])
        # d/dh[j]: if j < n-m, contributes (1-h[j+m]) (from h[j] as first factor)
        #          if j >= m, contributes -h[j-m] (from h[j] in second factor)
        g[:n-m] += (1.0 - h[m:])

def active_set_step(h, prof, c_max, n_active, trust_radius):
    """Find a descent direction using LP on active set gradients."""
    n = len(h)
    dx = 2.0 / n
    # Find active lags
    active = np.where(prof >= c_max - 0.005)[0]
    active = active[-n_active:]  # take at most n_active

    # Total gradient count
    m = len(active)
    # LP: min t + eps * sum|d_i|
    # s.t. g_k · d <= t for all k in active
    #      sum d = 0
    #      -trust_radius <= d_i <= trust_radius
    # Variables: [t, d_0,...,d_{n-1}, a_0,...,a_{n-1}] where a_i = |d_i|
    n_var = 1 + n + n

    # Just do: min t s.t. g_k · d <= t, sum d = 0, box on d.
    # This is a standard LP.
    
    # Simpler LP:
    # Variables: [t, d_0, ..., d_{n-1}]
    # min t
    # s.t. g_k · d - t <= 0 for each active k
    #      sum d_i = 0
    #      -R <= d_i <= R

    if np.any(neg_mask):
        alpha_max = min(alpha_max, np.min(h[neg_mask] / (-d[neg_mask])))
    alpha_max = max(alpha_max, 0.0)
