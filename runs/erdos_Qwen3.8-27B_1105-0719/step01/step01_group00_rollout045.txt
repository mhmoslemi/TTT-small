from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0

        h = np.clip(x - mid, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(np.max(compute_corrs(h))) * dx


def grad_corr_k(h, k, n):
    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        i_lo, i_hi = 0, n - 1 - d
    else:
        i_lo, i_hi = -d, n - 1
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] -= (1.0 - h[idx + d])
    grad[idx + d] += (1.0 - h[idx])
    return grad


def smooth_max_value(h, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta


def smooth_max_grad(h, n, beta):
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad

    def obj(x):
        return smooth_max_value(x, beta) * dx
    
    def grad(x):
        return smooth_max_grad(x, n, beta) * dx
    
    def cons_fun(x):
        return x.sum() - target_sum
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

In the discrete case with n points on [0,2]:

This is a quadratic form in h: c(k) = sum_i h[i] - sum_i h[i] h[i+d]

The first term is constant (= n/2 * dx = 1). So:

So minimizing max_k c(k) is equivalent to minimizing max_k (1 - Q(d_k)) = 1 - min_k Q(d_k)

Which means we want to MAXIMIZE min_k Q(d_k)!

So the problem becomes: max_h min_d Q(d) subject to 0 ≤ h ≤ 1, sum(h)*dx = 1

where Q(d) = sum_{i: i+d valid} h[i] h[i+d] * dx

This is a max-min problem. The inner Q(d) is a quadratic form (specifically, a bilinear form), and we want to maximize the minimum over d.

def threshold_reshape(h, target_sum):
    """Push h towards binary: zero out small values, cap large values, keep one fractional."""
    n = len(h)
    h_t = h.copy()
    threshold = 0.5
    h_t[h_t < threshold] = 0.0
    h_t[h_t >= threshold] = 1.0
    n_ones = int(h_t.sum())
    if n_ones >= int(target_sum) + 1:
        # Too many ones, demote the smallest "one"
        one_mask = h_t == 1.0
        if one_mask.sum() > 0:
            vals = h[one_mask]
            order = np.argsort(vals)
            for idx in order:
                actual_idx = np.where(one_mask)[0][idx]
                h_t[actual_idx] = 0.0
                if int(h_t.sum()) <= int(target_sum):
                    break
    frac_needed = target_sum - h_t.sum()
    if frac_needed > 1e-10:
        zero_mask = h_t == 0.0
        if zero_mask.sum() > 0:
            vals = h[zero_mask]
            order = np.argsort(-vals)
            actual_idx = np.where(zero_mask)[0][order[0]]
            h_t[actual_idx] = min(1.0, frac_needed)
    return project_feasible(h_t, target_sum)
