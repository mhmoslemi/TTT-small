import cvxpy as cp

def solve_trust_subproblem(h_center, S_center, dx, target_sum, active_lags, corrs, rho, alpha_h, maxiter=None):
    n = len(h_center)
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [t >= S_center - 1, t <= S_center + 1]
    constraints.append(cp.sum(h) == target_sum)
    constraints.append(h >= np.maximum(0, h_center - rho))
    constraints.append(h <= np.minimum(1, h_center + rho))
    
    for r in active_lags:
        # Build quadratic model
        d = r - (n - 1)
        R0 = corrs[r]
        base, shifted = get_pairs(n, d)
        G = np.zeros(n)
        diag = np.zeros(n)
        for b in base:
            j = b + d
            G[b] -= h_center[j]
            G[j] += 1 - h_center[b]
            diag[b] += 1
            diag[j] += 1
        # Constraint: t >= dx * (R0 + G^T(h - h_center) + 0.5 * diag^T (h-h_center)^2)

        model = R0 + G @ delta + 0.5 * (diag * delta**2).sum()

            prob.solve(solver=solver)
            if h.value is not None and not np.any(np.isnan(h.value)):
                return h.value
        except:
            continue
    
    return None

def get_lag_pairs(n, d):
    """Get valid (base, shifted) index pairs for lag offset d."""
    b_min = max(0, -d)
    b_max = min(n - 1, n - 1 - d)
    if b_min > b_max:
        return np.array([], dtype=int), np.array([], dtype=int)
    base = np.arange(b_min, b_max + 1)
    shifted = base + d
    return base, shifted

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_exact(h, dx):
    return compute_corrs(h).max() * dx

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:

        h = z[:n]
        return z[n] + alpha * np.sum((h-h_c)**2)
