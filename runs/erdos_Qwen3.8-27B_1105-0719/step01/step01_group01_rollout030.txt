def compute_c5(h):

def project_feasible(y, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)

def smooth_obj(h, beta):

    bounds = [(0.0, 1.0)] * n

    for beta in betas:
        res = minimize(
            smooth_obj, best_h, args=(beta,),
            jac=smooth_grad,

We want this to be negative (descent). So we want (h[j]-h[i])(grad[i] - grad[j]) < 0, i.e., (h[i]-h[j])(grad[i] - grad[j]) > 0.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

def _eq_con_fun(h):
    return np.sum(h) - h.size / 2.0

def _eq_con_jac(h):
    return np.ones(h.size)

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    return val * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        if k >= 0:
            if k < n:
                grad[:n-k] -= w * h[k:]
                grad[k:] += w * (1.0 - h[:n-k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= w * h[:n-kk]
                grad[:n-kk] += w * (1.0 - h[kk:])
    return grad * dx

def slsqp_round(h0, betas, max_iter=300, ftol=1e-12):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    for beta in betas:
        res = minimize(
            smooth_obj, best_h, args=(beta,),
            jac=smooth_grad,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': ftol}
        )
        if res.x is not None:
            h_cand = project_feasible(res.x, n)
            c5_new = compute_c5(h_cand)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cand
    return best_h, best_c5

def swap_search(h, max_passes=2000, min_improve=1e-14):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = len(h)
    for _ in range(max_passes):
        best_delta = min_improve
        best_i, best_j = -1, -1
        for i in range(n):
            for j in range(i + 1, n):
                if abs(best_h[i] - best_h[j]) < 1e-10:
                    continue
                best_h[i], best_h[j] = best_h[j], best_h[i]
                c5_new = compute_c5(best_h)
                best_h[i], best_h[j] = best_h[j], best_h[i]
                delta = best_c5 - c5_new
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i >= 0:
            best_h[best_i], best_h[best_j] = best_h[best_j], best_h[best_i]
            best_c5 -= best_delta
        else:
            break
    return best_h, best_c5

    target = n // 2

        h[rem] = 0
    elif num_ones < target:

        add = rng.choice(pos, target - num_ones, replace=False)
        h[add] = 1
    cur = compute_c5(h)

    best = cur
    temp = t0
    cool = (t1 / t0) ** (1.0 / n_iter)
