                h[idx] += residual2
                h[idx] = min(max(h[idx], 0.0), 1.0)
                break
    return h


def compute_grad_at_active(h, active_idx):
    n = len(h)
    grad = np.zeros(n)
    for idx in active_idx:
        k = idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:n] += (1.0 - h[0:n - k])
                grad[0:n - k] -= h[k:n]
        else:
            m = -k
            if n - m > 0:
                grad[0:n - m] += (1.0 - h[m:n])
                grad[m:n] -= h[0:n - m]
    return grad


def active_set_descent(h, max_iters=500, active_tol=0.05, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n

    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(best_h)
        max_corr = np.max(corr)
        c5_val = max_corr * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val

        threshold = max_corr - active_tol
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = compute_grad_at_active(best_h, active)

        c_obj = grad
        A_eq = np.ones((1, n))
        b_eq = np.array([0.0])
        bounds = [(-best_h[i], 1.0 - best_h[i]) for i in range(n)]

        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

        if res.status != 0 or res.x is None:
            break

        delta = res.x
        if np.max(np.abs(delta)) < 1e-14:
            break

        best_alpha = 0.0
        best_trial = c5_val
        for alpha in np.concatenate([np.linspace(0.02, 0.5, 12), np.linspace(0.5, 1.0, 10)]):
            h_try = best_h + alpha * delta
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            val = compute_c5(h_try)
            if val < best_trial - 1e-14:
                best_trial = val
                best_alpha = alpha

        if best_alpha > 0:
            best_h = project_feasible(best_h + best_alpha * delta, n)
            new_c5 = compute_c5(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = best_h.copy()
            else:
                break
        else:
            break

        if it > 5 and c5_val - best_c5 < 1e-14:
            break

    return best_h, best_c5


def transfer_search(h, n_iters, rng, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    deltas = np.array([0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5])

    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        max_transfer = min(best_h[j], 1.0 - best_h[i])
        if max_transfer < 1e-6:
            continue
        for d in deltas:
            amt = min(d, max_transfer)
            h_try = best_h.copy()
            h_try[i] += amt
            h_try[j] -= amt
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            val = compute_c5(h_try)
            if val < best_c5 - 1e-14:
                best_c5 = val
                best_h = h_try
                break
            # Also try reverse
            max_transfer_r = min(best_h[i], 1.0 - best_h[j])
            if max_transfer_r < 1e-6:
                continue
            amt_r = min(d, max_transfer_r)
            h_try2 = best_h.copy()
            h_try2[j] += amt_r
            h_try2[i] -= amt_r
            h_try2 = np.clip(h_try2, 0.0, 1.0)
            h_try2 = project_feasible(h_try2, n)
            val2 = compute_c5(h_try2)
            if val2 < best_c5 - 1e-14:
                best_c5 = val2
                best_h = h_try2
                break

    return best_h, best_c5


def swap_search(h, n_iters, rng, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or abs(best_h[i] - best_h[j]) < 1e-6:
            continue
        h_try = best_h.copy()
        h_try[i], h_try[j] = best_h[j], best_h[i]
        val = compute_c5(h_try)
        if val < best_c5 - 1e-14:
            best_c5 = val
            best_h = h_try
    return best_h, best_c5


def simulated_annealing(h, T0, T_end, n_iters, rng, deadline=None):
    n = len(h)
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    T = T0
    cooling = (T_end / T0) ** (1.0 / max(n_iters, 1))
    deltas_pool = np.array([0.02, 0.05, 0.1, 0.15, 0.25, 0.4])

    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = rng.randint(2)
        if direction == 0:
            max_t = min(cur_h[j], 1.0 - cur_h[i])
        else:
            max_t = min(cur_h[i], 1.0 - cur_h[j])
        if max_t < 1e-6:
            continue
        d = delta_pool_choice(rng, max_t, deltas_pool)
        h_try = cur_h.copy()
        if direction == 0:
            h_try[i] += d
            h_try[j] -= d
        else:
            h_try[j] += d
            h_try[i] -= d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        new_c5 = compute_c5(h_try)
        delta_c = new_c5 - cur_c5
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            cur_h = h_try
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        T *= cooling

    return best_h, best_c5


def delta_pool_choice(rng, max_t, deltas_pool):
    valid = deltas_pool[deltas_pool <= max_t]
    if len(valid) == 0:
        return min(deltas_pool[0], max_t)
    return rng.choice(valid)


def resample_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new) + (1.0 / n_new)
    h_interp = np.interp(x_new, x_old, h)
    h_interp = project_feasible(h_interp, n_new)
    return h_interp


def generate_start(h_parent, rng, strategy):
    n = len(h_parent)
    if strategy == 0:
        return project_feasible(h_parent + 0.1 * rng.randn(n), n)
    elif strategy == 1:
        h_try = h_parent.copy()
        mask = rng.random(n) < 0.15
        h_try[mask] = 1.0 - h_try[mask]
        return project_feasible(h_try, n)
    elif strategy == 2:
        shift = rng.randint(1, n)
        return project_feasible(np.roll(h_parent, shift), n)
    elif strategy == 3:
        h_try = (h_parent > 0.5).astype(float)
        idx = rng.choice(n, size=2, replace=False)
        h_try[idx[0]] = 0.5
        h_try[idx[1]] = 0.5
        return project_feasible(h_try, n)
    elif strategy == 4:
        h_try = np.zeros(n)
        ones = rng.choice(n, size=n // 2, replace=False)
        h_try[ones] = 1.0
        return project_feasible(h_try, n)
    else:
        h_try = 0.3 + 0.4 * np.sin(2.0 * np.pi * rng.randint(3, 12) * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        return project_feasible(h_try, n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(60, 0.5)

    best_h = project_feasible(h_init, len(h_init))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: Active set descent from initial
    h_opt, c5_opt = active_set_descent(best_h, max_iters=300, active_tol=0.1, deadline=deadline)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 2: Transfer local search
    rem = budget_s - (time.time() - t0)
    n_iter = min(50000, int(rem * 60))
    h_opt, c5_opt = transfer_search(best_h, n_iter, rng, deadline)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 3: Active set descent again (tighter tolerance)
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_opt, c5_opt = active_set_descent(best_h, max_iters=400, active_tol=0.02, deadline=deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        h_opt, c5_opt = active_set_descent(best_h, max_iters=200, active_tol=0.0, deadline=deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Swap search
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        n_iter = min(30000, int(rem * 50))
        h_opt, c5_opt = swap_search(best_h, n_iter, rng, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Simulated annealing for global exploration
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        sa_time = min(rem * 0.4, 300)
        sa_deadline = t0 + (budget_s - 10.0 - rem * 0.6)
        n_sa = int(sa_time * 3000)
        h_opt, c5_opt = simulated_annealing(best_h, 0.04, 0.002, n_sa, rng, sa_deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            # Polish after SA
            h_opt2, c5_opt2 = active_set_descent(best_h, max_iters=300, active_tol=0.05, deadline=deadline)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    # Phase 6: Multi-start with perturbation
    rem = budget_s - (time.time() - t0)
    n_starts = min(15, int(rem / 30))
    for s in range(n_starts):
        if time.time() > deadline - 15:
            break
        h_pert = generate_start(best_h, rng, s % 6)
        h_opt, c5_opt = active_set_descent(h_pert, max_iters=200, active_tol=0.1, deadline=deadline)
        if time.time() > deadline - 10:
            break
        h_opt, c5_opt = transfer_search(h_opt, min(5000, int((deadline - time.time()) * 40)), rng, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Multi-resolution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        n_current = len(best_h)
        candidates = []
        if n_current != 24:
            candidates.append(24)
        if n_current != 30:
            candidates.append(30)
        if n_current != 36:
            candidates.append(36)
        if n_current != 48:
            candidates.append(48)
        if n_current != 72:
            candidates.append(72)
        if n_current != 84:
            candidates.append(84)

        for n_try in candidates:
            if time.time() > deadline - 25:
                break
            h_interp = resample_h(best_h, n_try)
            h_opt, c5_opt = active_set_descent(h_interp, max_iters=200, active_tol=0.08, deadline=deadline)
            if time.time() > deadline - 15:
                break
            h_opt, c5_opt = transfer_search(h_opt, min(3000, int((deadline - time.time()) * 30)), rng, deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n = n_try

    # Phase 8: Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h_opt, c5_opt = active_set_descent(best_h, max_iters=300, active_tol=0.01, deadline=deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        if time.time() < deadline - 5:
            h_opt, c5_opt = transfer_search(best_h, min(20000, int((deadline - time.time()) * 80)), rng, deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        if time.time() < deadline - 3:
            h_opt, c5_opt = active_set_descent(best_h, max_iters=200, active_tol=0.0, deadline=deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
