import time

        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def lag_grad(h, r, n):
    d = r - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        nv = n - d
        b = np.arange(nv)
        j = b + d
        np.add.at(grad, j, 1.0 - h[b])
        np.add.at(grad, b, -h[j])
    else:
        dd = -d
        b = np.arange(dd, n)
        j = b - dd
        np.add.at(grad, j, 1.0 - h[b])
        np.add.at(grad, b, -h[j])
    return grad

def active_set_select(corrs, k_min=3, frac=0.95):
    """Select active correlation indices."""
    max_c = corrs.max()
    # Include all that are within frac of max, at least k_min
    active = np.where(corrs >= max_c * frac)[0]
    if len(active) < k_min:
        # Take top k_min
        active = np.argsort(corrs)[-k_min:]
    return active

class ActiveSetSolver:
    def __init__(self, h, dx, target_sum):
        self.n = len(h)
        self.dx = dx
        self.target_sum = target_sum
        self.h = h.copy()
        self.corrs = np.correlate(h, 1.0 - h, mode='full')
        self.active = active_set_select(self.corrs)
    
    def _solve_subproblem(self, active, h_center, rho, maxiter=200):
        n = self.n
        dx = self.dx
        corrs = np.correlate(h_center, 1.0 - h_center, mode='full')
        # Precompute gradients for active lags
        grads = np.array([lag_grad(h_center, r, n) for r in active])  # shape (K, n)
        
        def obj(v):
            h = v[:n]
            t = v[n]
            return t
        
        def obj_grad(v):
            h = v[:n]
            t = v[n]
            g = np.zeros(n+1)
            g[n] = 1.0
            return g
        
        cons = []
        # Sum constraint
        sum_fun = lambda v: v[:n].sum() - self.target_sum
        sum_jac = lambda v: np.concatenate([np.ones(n), [0.0]])
        cons.append({'type': 'eq', 'fun': sum_fun, 'jac': sum_jac})
        
        # Active constraint (linearized): t >= dx * [R_r(h0) + grad_r^T (h-h0)]
        h0 = h_center
        R0 = corrs[active]  # raw values
        for i, r in enumerate(active):
            grad_r = grads[i]
            def c_fun(v, g=grad_r, R0i=R0[i]):
                h = v[:n]
                t = v[n]
                return t - self.dx * (R0i + g @ (h - h0))
            def c_jac(v, g=grad_r):
                h = v[:n]
                t = v[n]
                j = np.zeros(n+1)
                j[:n] = -self.dx * g
                j[n] = 1.0
                return j
            cons.append({'type': 'ineq', 'fun': c_fun, 'jac': c_jac})
        
        # Bounds
        lower = np.concatenate([np.maximum(0, h_center - rho), [0.0]])
        upper = np.concatenate([np.minimum(1, h_center + rho), [self.corrs.max() * self.dx + 1.0]])
        bounds = list(zip(lower, upper))
        
        # Initial point
        t0 = self.corrs.max() * self.dx + 1e-6
        v0 = np.concatenate([h_center, [t0]])
        
        try:
            res = minimize(obj, v0, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-12})

    solver = ActiveLPSolver(h0, active, n, dx, target_sum, rho)
    cons = [
        {'type': 'ineq', 'fun': solver.ineq_fun, 'jac': solver.ineq_jac},
        {'type': 'eq', 'fun': solver.eq_fun, 'jac': solver.eq_jac}
    ]
    bounds = list(zip(solver.lower, solver.upper))
    try:
        res = minimize(solver.fun, solver.x0, jac=solver.fun_jac,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

def lag_grad(h, r, n):
    d = r - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        nv = n - d
        b = np.arange(nv)
        j = b + d
        grad[j] += (1.0 - h[b])
        grad[b] -= h[j]
    else:
        dd = -d
        b = np.arange(dd, n)
        j = b - dd
        grad[j] += (1.0 - h[b])
        grad[b] -= h[j]
    return grad
