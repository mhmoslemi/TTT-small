        else:

    h_final = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_final.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmin(h_final)
            h_final[idx] = min(1.0, h_final[idx] + diff)
        else:
            idx = np.argmax(h_final)
            h_final[idx] = max(0.0, h_final[idx] + diff)
    return h_final


def compute_L_values(h):
    n = len(h)
    dx = 2.0 / n
    L = np.zeros(n)
    cumsum = np.zeros(n + 1)
    cumsum[1:] = np.cumsum(h)
    for m in range(n):
        if m == 0:
            A = float(np.dot(h, h))
        else:
            A = float(np.dot(h[:n - m], h[m:]))
        P_m = cumsum[m]
        S_m = cumsum[n] - cumsum[n - m]
        L[m] = dx * (A + min(P_m, S_m))
    return L


def compute_L_gradient_single(h, m):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 2.0 * dx * h
    else:
        grad[:n - m] += dx * h[m:]
        grad[m:] += dx * h[:n - m]
        P_m = np.sum(h[:m])
        S_m = np.sum(h[n - m:])
        if P_m < S_m:
            grad[:m] += dx
        elif S_m < P_m:
            grad[n - m:] += dx
        else:
            grad[:m] += 0.5 * dx
            grad[n - m:] += 0.5 * dx
    return grad


def softmin_value_and_grad(h, beta):
    n = len(h)
    L = compute_L_values(h)
    L_min = np.min(L)
    exps = -beta * (L - L_min)
    exps = np.clip(exps, -500.0, 0.0)
    exps_arr = np.exp(exps)
    Z = np.sum(exps_arr)
    softmin_val = L_min - np.log(Z) / beta
    w = exps_arr / Z
    grad = np.zeros(n)
    active = w > 1e-8
    for m in np.where(active)[0]:
        grad += w[m] * compute_L_gradient_single(h, m)
    return softmin_val, grad, L, w


def nesterov_projected_gradient(h0, beta_start, beta_end, n_iter, time_limit):
    t0 = time.perf_counter()
    n = len(h0)
    h = repair_feasible(h0.copy())
    best_h = h.copy()
    best_score = exact_score(best_h)
    momentum = np.zeros(n)
    mu = 0.85
    alpha = 0.03
    best_alpha = alpha
    prev_h = h.copy()

    for i in range(n_iter):
        if time.perf_counter() - t0 > time_limit:
            break
        beta_t = beta_start + (beta_end - beta_start) * (i / max(n_iter - 1, 1))
        _, grad, _, _ = softmin_value_and_grad(h, beta_t)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        grad = grad / grad_norm
        momentum = mu * momentum + grad
        h_new = project_feasible(h + alpha * momentum)
        score_new = exact_score(h_new)
        if score_new < best_score - 1e-12:
            best_h = h_new.copy()
            best_score = score_new
            best_alpha = alpha
            if alpha < 0.05:
                alpha *= 1.05
        else:
            alpha *= 0.7
            if alpha < 0.001:
                alpha = 0.001
        if score_new <= exact_score(prev_h) + 1e-10:
            h = h_new
        else:
            h = prev_h
        prev_h = h.copy()
    return best_h, best_score


def slsqp_softmin(h_start, beta, maxiter=200):
    n = len(h_start)

    def obj_func(x):
        val, _, _, _ = softmin_value_and_grad(x, beta)
        return -val

    def jac_func(x):
        _, g, _, _ = softmin_value_and_grad(x, beta)
        return -g

    def eq_fun(x):
        return float(np.sum(x) - n / 2.0)

    def eq_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj_func, h_start, jac=jac_func, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        if result.x is not None:
            h_sol = repair_feasible(result.x)
            score = exact_score(h_sol)
            if np.isfinite(score):
                return h_sol, score
    except Exception:
        pass
    return None, None


def slsqp_regulated(h_start, h_ref, beta, lam, maxiter=200):
    n = len(h_start)

    def obj_func(x):
        val, _, _, _ = softmin_value_and_grad(x, beta)
        reg = lam * np.sum((x - h_ref) ** 2)
        return -val + reg

    def jac_func(x):
        _, g, _, _ = softmin_value_and_grad(x, beta)
        return -g + 2.0 * lam * (x - h_ref)

    def eq_fun(x):
        return float(np.sum(x) - n / 2.0)

    def eq_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj_func, h_start, jac=jac_func, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        if result.x is not None:
            h_sol = repair_feasible(result.x)
            score = exact_score(h_sol)
            if np.isfinite(score):
                return h_sol, score
    except Exception:
        pass
    return None, None


def active_descent_min(h0, max_steps=50, time_limit=20):
    t0 = time.perf_counter()
    n = len(h0)
    best = repair_feasible(h0.copy())
    best_score = exact_score(best)

    for step in range(max_steps):
        if time.perf_counter() - t0 > time_limit:
            break
        L = compute_L_values(best)
        min_L = np.min(L)
        spread = max(np.max(L) - min_L, 0.01)
        threshold = min_L + 0.02 * spread
        active = np.where(L <= threshold)[0]
        if len(active) < 2:
            active = np.argsort(L)[:5]

        gaps = np.maximum(min_L + 0.003 - L[active], 0.0)
        weights = gaps ** 2
        if weights.sum() < 1e-15:
            weights = np.ones(len(active)) / len(active)
        else:
            weights = weights / weights.sum()

        direction = np.zeros(n)
        for i, m in enumerate(active):
            direction += weights[i] * compute_L_gradient_single(best, m)
        direction -= np.mean(direction)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-10:
            break
        direction /= dnorm

        improved = False
        for alpha in [0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]:
            h_trial = project_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-13:
                best = h_trial
                best_score = score
                improved = True
                break
        if not improved:
            break
    return best, best_score


def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_h = repair_feasible(parent.copy())
    else:
        best_h = np.full(70, 0.5)

    best_score = exact_score(best_h)

    # Symmetrize
    h_sym = repair_feasible(symmetrize(best_h))
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-13:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: Nesterov PG with annealed beta
    if tl() > 100:
        h_res, s_res = nesterov_projected_gradient(
            best_h, beta_start=50.0, beta_end=300.0, n_iter=300, time_limit=80)
        if s_res < best_score - 1e-13:
            best_h = h_res
            best_score = s_res

    # Phase 2: SLSQP softmin
    if tl() > 120:
        for beta in [200.0, 500.0, 1000.0]:
            if tl() < 40:
                break
            h_sol, score = slsqp_softmin(best_h, beta, maxiter=200)
            if h_sol is not None and score < best_score - 1e-13:
                best_h = h_sol
                best_score = score

    # Phase 3: Active descent
    if tl() > 30:
        h_ref, s_ref = active_descent_min(best_h, max_steps=60, time_limit=25)
        if s_ref < best_score - 1e-13:
            best_h = h_ref
            best_score = s_ref

    # Phase 4: Regulated SLSQP (keeps near current best)
    if tl() > 80:
        for lam in [0.01, 0.005, 0.001, 0.0005]:
            if tl() < 30:
                break
            h_sol, score = slsqp_regulated(best_h, best_h, 500.0, lam, maxiter=150)
            if h_sol is not None and score < best_score - 1e-13:
                best_h = h_sol
                best_score = score

    # Phase 5: Dimension transfer up
    if tl() > 60:
        n_cur = len(best_h)
        for nd in [min(n_cur + 20, 200)]:
            if tl() < 30:
                break
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.01:
                h_sol, score = slsqp_softmin(h_interp, 300.0, maxiter=150)
                if h_sol is not None and score < best_score - 1e-13:
                    best_h = h_sol
                    best_score = score
                else:
                    h_ref2, s_ref2 = active_descent_min(h_interp, max_steps=30, time_limit=15)
                    if s_ref2 < best_score - 1e-13:
                        best_h = h_ref2
                        best_score = s_ref2

    # Phase 6: Another Nesterov pass with higher beta
    if tl() > 50:
        h_res, s_res = nesterov_projected_gradient(
            best_h, beta_start=300.0, beta_end=800.0, n_iter=200, time_limit=40)
        if s_res < best_score - 1e-13:
            best_h = h_res
            best_score = s_res

    # Phase 7: Perturbed multi-start
    if tl() > 60:
        for trial in range(8):
            if tl() < 20:
                break
            eps = rng.uniform(0.003, 0.01)
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.std(z) + 1e-10)
            h_pert = repair_feasible(best_h + eps * z)
            h_sol, score = slsqp_softmin(h_pert, 500.0, maxiter=100)
            if h_sol is not None and score < best_score - 1e-13:
                best_h = h_sol
                best_score = score
            else:
                h_ref3, s_ref3 = active_descent_min(h_pert, max_steps=20, time_limit=10)
                if s_ref3 < best_score - 1e-13:
                    best_h = h_ref3
                    best_score = s_ref3

    # Phase 8: Final SLSQP polish with very high beta
    if tl() > 30:
        h_sol, score = slsqp_softmin(best_h, 2000.0, maxiter=300)
        if h_sol is not None and score < best_score - 1e-13:
            best_h = h_sol
            best_score = score

    # Phase 9: Final active descent
    if tl() > 15:
        h_ref, s_ref = active_descent_min(best_h, max_steps=40, time_limit=12)
        if s_ref < best_score - 1e-13:
            best_h = h_ref
            best_score = s_ref

    # Phase 10: One last Nesterov with very high beta
    if tl() > 20:
        h_res, s_res = nesterov_projected_gradient(
            best_h, beta_start=800.0, beta_end=1500.0, n_iter=100, time_limit=15)
        if s_res < best_score - 1e-13:
            best_h = h_res
            best_score = s_res

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
