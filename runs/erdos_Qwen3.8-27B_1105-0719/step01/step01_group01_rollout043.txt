    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

    corr = compute_corr(h)
    val, _ = smooth_max(corr, beta)
    return val * (2.0 / len(h))

def binary_anneal(n, n_iters, rng, initial=None):
    """Anneal a binary sequence to minimize max correlation."""
    if initial is not None:
        h = initial.copy().astype(float)
    else:
        h = np.zeros(n)

        h[ones] = 1.0

    best_val = np.max(corr)
    
    # Energy function: use sum of top values for smoother landscape
    def energy(c):
        sorted_c = np.sort(c)[::-1]
        return np.max(c) * 10 + np.sum(sorted_c[:5])
    
    current_energy = energy(corr)
    T_start = 10.0
    T_end = 0.1
    
    for it in range(n_iters):
        T = T_start * (T_end / T_start) ** (it / n_iters)
        # Pick a random pair to swap (one 1 and one 0)
        ones_mask = h > 0.5
        zeros_mask = h < 0.5
        if not np.any(ones_mask) or not np.any(zeros_mask):
            break
        i = rng.choice(np.where(ones_mask)[0])
        j = rng.choice(np.where(zeros_mask)[0])
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        corr_new = compute_corr(h)
        energy_new = energy(corr_new)
        delta = energy_new - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_energy = energy_new
            if np.max(corr_new) < best_val:
                best_val = np.max(corr_new)
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            corr_new = corr  # restore
    
    corr_best = compute_corr(best_h)
    return best_h, np.max(corr_best)

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(interior):
            h[interior] += residual / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_run(h0, betas, max_iter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for beta in betas:
        try:
            res = minimize(
                smooth_obj, best_h, args=(beta,),
                jac=smooth_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-14}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_candidate = project_feasible(res.x, n)
                c5_new = compute_c5(h_candidate)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_candidate
        except Exception:
            pass
    return best_h, best_c5

def binary_best_improve(h, max_passes=100):
    """Best-improvement local search for binary sequences."""
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_max = np.max(corr)
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]
    
    for _ in range(max_passes):
        improved = False
        for i in ones_idx:
            for j in zeros_idx:
                if i == j:
                    continue
                h[i], h[j] = 0.0, 1.0
                new_corr = compute_corr(h)
                new_max = np.max(new_corr)
                if new_max < best_max:
                    best_max = new_max
                    corr = new_corr
                    ones_idx = np.where(h > 0.5)[0]
                    zeros_idx = np.where(h < 0.5)[0]
                    improved = True
                    break
                h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, best_max

def binary_anneal(n, n_iters, rng):
    """Simulated annealing for binary sequences."""
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    corr = compute_corr(h)
    best_max = np.max(corr)
    best_h = h.copy()
    
    T = 5.0
    T_decay = (0.05 / 5.0) ** (1.0 / max(n_iters, 1))
    
    for _ in range(n_iters):
        T *= T_decay
        ones_mask = h > 0.5
        zeros_mask = h < 0.5
        if not np.any(ones_mask) or not np.any(zeros_mask):
            break
        i = rng.choice(np.where(ones_mask)[0])
        j = rng.choice(np.where(zeros_mask)[0])
        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr(h)
        new_max = np.max(new_corr)
        delta = new_max - np.max(corr)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-20)):
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
            corr = new_corr
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_max

def local_search_continuous(h, n_iters, rng, step_size=0.05):
    """Local search: random paired moves maintaining sum constraint."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iters):
        i, j = rng.choice(n, size=2, replace=False)
        # Try moving mass from j to i (or vice versa)
        delta = step_size * rng.choice([-1, 1])
        new_i = best_h[i] + delta
        new_j = best_h[j] - delta
        if 0 <= new_i <= 1 and 0 <= new_j <= 1:
            h_new = best_h.copy()
            h_new[i] = new_i
            h_new[j] = new_j
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    return best_h, best_c5

    rem = budget_s - (time.time() - t0)

            break

                best_bin_max = max_v

            if elapsed > time_per_n * 0.5:

        if best_bin_h is not None:
            binary_results[n] = (best_bin_h, best_bin_max)
