import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:

    """Compute profile values and gradient matrix.
    P[m] = profile value at index m
    G[m, p] = dP[m]/dh[p]
    Returns (P, G) where P has length 2n-1 and G has shape (2n-1, n)
    """
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.zeros(M)
    G = np.zeros((M, n))
    
    for m in range(M):
        s = m - (n - 1)
        if s >= 0:
            a = s
            # R_s = sum_{i=a}^{n-1} h[i]*(1-h[n-1+a-i])
            # grad[p] = dx*(1-h[p]-h[n-1+a-p]) for p in [a, n-1]
            idx = np.arange(a, n)
            j = n - 1 + a - idx
            P[m] = dx * np.sum(h[idx] * (1.0 - h[j]))
            G[m, idx] = dx * (1.0 - h[idx] - h[j])
        else:
            a = -s
            # R_{-a} = sum_{i=0}^{n-1-a} h[i]*(1-h[n-1-a-i])
            # grad[p] = dx*(1-h[p]-h[n-1+a-p]) for p in [0, n-1-a]
            idx = np.arange(0, n - a)
            j = n - 1 - a - idx
            P[m] = dx * np.sum(h[idx] * (1.0 - h[j]))
            G[m, idx] = dx * (1.0 - h[idx] - h[j])
    return P, G

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff2 / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def build_profile_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.zeros(M)
    G = np.zeros((M, n))
    for m in range(M):
        s = m - (n - 1)
        if s >= 0:
            a = s
            if a >= n:
                continue
            idx = np.arange(a, n)
            j = n - 1 + a - idx
            P[m] = dx * np.sum(h[idx] * (1.0 - h[j]))
            G[m, idx] = dx * (1.0 - h[idx] - h[j])
        else:
            a = -s
            if a >= n:
                continue
            idx = np.arange(0, n - a)
            j = n - 1 - a - idx
            P[m] = dx * np.sum(h[idx] * (1.0 - h[j]))

    return P, G

def sca_propose(h, r):
    """Solve one SCA LP. Returns z_star or None."""
    n = len(h)
    P, G = build_profile_and_grad(h)
    M = 2 * n - 1
    # LP: min tau s.t. G@z - tau <= G@h - P, 1'z = n/2, bounds
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = G @ h - P
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - r)
        hi = min(1.0, h[i] + r)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, z_star, current_c):
    """Exact line search from h toward z_star. Returns (best_candidate, best_c) or (h, current_c)."""
    alphas = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    best_c = current_c
    best_h = h

        c_val = exact_score(cand)
        if c_val < best_c - 1e-12:

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def local_slsqp_polish(h, maxiter=200):
    """SLSQP polish as backup."""
    n = len(h)
    dx = 2.0 / n
    c0 = exact_score(h)
    t0 = c0 + 1e-4

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        hv = x[:-1]
        t = x[-1]
        profile = np.correlate(hv, 1.0 - hv, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        hv = x[:-1]
        P, G = build_profile_and_grad(hv)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})

            h_new = repair_feasible(res.x[:-1])
            if h_new.sum() > 0:
                return h_new
    except Exception:
        pass
    return None
