import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(h):
    """Project h onto {0<=h<=1, sum=h = n/2}"""
    n = len(h)
    h = np.asarray(h, dtype=np.float64).copy()
    target = n / 2.0
    # Bisection on shift parameter
    # h_i(a) = clip(h_i + a, 0, 1)
    # Find a such that sum h_i(a) = target
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    a = (lo + hi) / 2
    result = np.clip(h + a, 0, 1)
    # Fine adjustment
    diff = target - result.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            order = np.argsort(1.0 - result)
        else:
            order = np.argsort(result)
        for i in range(n):
            idx = order[i]
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
            else:
                add = max(diff, -result[idx])
            result[idx] += add
            diff -= add
            if abs(diff) < 1e-15:
                break
    return np.clip(result, 0, 1)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def gradient_rows(h, rows):
    """Compute gradient matrix G[rows, :] where G[m, j] = dr_m/dh_j"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(rows), n))
    for k, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[k, d:] += dx * (1.0 - h[:n-d])
            G[k, :n-d] += -dx * h[d:]
        else:
            s = -d
            G[k, :n-s] += dx * (1.0 - h[s:])
            G[k, s:] += -dx * h[:n-s]
    return G

def logsumexp_objective(h, tau):
    """Compute tau * log(sum exp(r_m/tau)) and its gradient"""
    profile = compute_profile(h)
    max_r = np.max(profile)
    shifted = (profile - max_r) / tau
    weights = np.exp(shifted)
    weights /= weights.sum()
    obj = tau * (np.log(weights.sum()) + max_r / tau)  # = max_r + tau * log(sum exp((r_m-max_r)/tau))
    # Actually: tau * log(sum exp(r_m/tau)) = max_r + tau * log(sum exp((r_m - max_r)/tau))
    g = gradient_rows(h, np.arange(len(profile)))
    grad = weights @ g  # sum_m w_m * g_m
    return obj, grad

def active_set_gradient(h, epsilon=1e-4):
    """Compute weighted gradient focusing on active (near-max) lags"""
    profile = compute_profile(h)
    max_r = np.max(profile)
    active = np.where(profile >= max_r - epsilon)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    weights = profile[active] - (max_r - epsilon)
    weights = np.maximum(weights, 1e-10)
    weights /= weights.sum()
    G = gradient_rows(h, active)
    grad = weights @ G
    return grad, active, profile

def projected_gradient_step(h, grad, alpha):
    """Take a gradient step and project back to feasible set"""
    # Project gradient onto null space of sum constraint
    grad_feasible = grad - grad.mean()
    h_new = h - alpha * grad_feasible
    return repair_feasible(h_new)

def line_search(h, grad, alpha0):
    """Backtracking line search"""
    best = exact_score(h)
    grad_feasible = grad - grad.mean()
    alpha = alpha0
    for _ in range(30):

def lse_descent(h, max_iters=200, tau0=0.05, tau_min=0.001):
    """Log-sum-exp guided descent with adaptive temperature"""
    n = len(h)
    best_h = h.copy()

        # Compute weights

        max_shifted = np.max(shifted)
        weights = np.exp(shifted - max_shifted)
        weights /= weights.sum()

        grad = weights @ G
        grad_feasible = grad - grad.mean()
        norm = np.linalg.norm(grad_feasible)

            break

                improved = True

def active_descent(h, max_iters=300, epsilon=1e-5):
    """Active-set projected gradient descent"""
    best_h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        max_r = np.max(profile)
        active = np.where(profile >= max_r - epsilon)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])

        w = profile[active] - (max_r - epsilon)
        w = np.maximum(w, 1e-10)

        grad = w @ G

def slsqp_polish(h, maxiter=300):
    """SLSQP epigraph polish"""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h_ = x[:-1]
        t = x[-1]
        profile = np.correlate(h_, 1 - h_, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h_ = x[:-1]
        profile = np.correlate(h_, 1 - h_, mode="full") * dx
        # Only include active constraints for efficiency
        active = np.where(profile >= np.max(profile) - 0.01)[0]
        G = gradient_rows(h_, active)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            return h_new
    except Exception:
        pass
    return None

    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(h):
    n = len(h)
    h = np.asarray(h, dtype=np.float64).copy()
    target = n / 2.0

        if s < target:
            lo = mid
        else:
            hi = mid

    diff = target - result.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - result)
        else:
            order = np.argsort(result)
        for i in range(n):
            idx = order[i]
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
            else:
                add = max(diff, -result[idx])
            if add == 0:
                continue
            result[idx] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
    return np.clip(result, 0.0, 1.0)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def gradient_rows(h, rows):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(rows), n))
    for k, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[k, d:] += dx * (1.0 - h[:n - d])
            G[k, :n - d] += -dx * h[d:]
        else:
            s = -d
            G[k, :n - s] += dx * (1.0 - h[s:])
            G[k, s:] += -dx * h[:n - s]
    return G

def lse_descent(h, max_iters=300, tau0=0.04, tau_min=0.0005):
    n = len(h)
    M = 2 * n - 1
    best_h = h.copy()

        max_s = np.max(shifted)
        weights = np.exp(shifted - max_s)
        weights /= weights.sum()

                best_c = s
                best_h = h.copy()
                improved = True
                break
