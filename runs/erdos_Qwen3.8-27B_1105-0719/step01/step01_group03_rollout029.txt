```python
def binary_local_search(h_binary, max_iters=100000, t0=1.0, t1=0.001):
    """Simulated annealing on binary sequences with fixed weight."""
    n = len(h_binary)
    dx = 2.0 / n
    h = h_binary.copy().astype(np.float64)
    # Ensure binary
    h = (h > 0.5).astype(np.float64)
    # Ensure exact count
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        for idx in ones[target:]:
            h[idx] = 0
            zeros = np.append(zeros, idx)
            zeros.sort()
    elif len(ones) < target:
        for idx in zeros[:target - len(ones)]:
            h[idx] = 1
            ones = np.append(ones, idx)
            ones.sort()
    
    def compute_profile_binary(h):
        # P[d] for d in -(n-1)..(n-1)
        # P[0] = 0 for binary
        # P[d] = sum_{i} h[i]*(1-h[i+d])*dx for d>0
        # P[-d] = sum_{i} h[i+d]*(1-h[i])*dx for d>0
        profile = np.zeros(2*n-1)
        for d in range(1, n):
            profile[n-1+d] = np.sum(h[:n-d] * (1-h[d:])) * dx
            profile[n-1-d] = np.sum(h[d:] * (1-h[:n-d])) * dx
        return profile
    
    P = compute_profile_binary(h)
    best_max = np.max(P)
    current_max = best_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        temp = t0 * (t1/t0) ** (it / max_iters)
        if temp < 1e-10:
            break
        
        # Random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Compute delta
        # Flipping i: 1→0, flipping j: 0→1
        delta = np.zeros(2*n-1)

        for d in range(1, n):

            if i < n - d:

            if i >= d:
                delta[n-1+d] += h[i-d] * dx
            # d < 0: P[n-1-d]
            if i >= d:
                delta[n-1-d] -= (1 - h[i-d]) * dx
            if i < n - d:
                delta[n-1-d] += h[i+d] * dx
        # Effect of flipping j (0→1):
```

```python
        # As j+d=i: h[i-d]*(1-0) = h[i-d], change = +h[i-d] - 0 = +h[i-d]
        # But wait, before the flip, the j+d=i term was h[i-d]*(1-1) = 0
        # After flip, it's h[i-d]*(1-0) = h[i-d]
        # So change for P[n-1+d] = -(1-h[i+d])*[d<=n-1-i] + h[i-d]*[d<=i]
        
        # For direction = -1 (1→0):
        delta[n-1+d] += direction * (-(1-h[i+d]) if d <= n-1-i else 0 + h[i-d] if d <= i else 0)
        
        # For shift -d: P[n-1-d] = sum_{j=0}^{n-1-d} h[j+d]*(1-h[j])*dx
        # Contribution of position i:
        # As j+d=i (j=i-d, if i-d >= 0): h[i]*(1-h[i-d])
        # As j=i (if i <= n-1-d): h[i+d]*(1-h[i])
        
        # Before flip (h[i]=1):
        # As j+d=i: 1*(1-h[i-d]) = 1-h[i-d]
        # As j=i: h[i+d]*(1-1) = 0
        # After flip (h[i]=0):
        # As j+d=i: 0*(1-h[i-d]) = 0, change = -(1-h[i-d])
        # As j=i: h[i+d]*(1-0) = h[i+d], change = +h[i+d]
        
        delta[n-1-d] += direction * (-(1-h[i-d]) if d <= i else 0 + h[i+d] if d <= n-1-i else 0)
    
    return delta * dx
```

```python
def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):

            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
```

Now let me think about the LSE optimization more carefully. The LSE objective is:
f_τ(h) = max(profile) + (1/τ) * log(sum(exp(τ * (profile - max(profile)))))

The gradient is:
∇f_τ(h) = G^T @ (softmax(τ * profile) * ... )

Wait, let me be more careful. Let p = profile(h), m = max(p).
f_τ(h) = m + (1/τ) * log(Σ_k exp(τ(p_k - m)))

∂f_τ/∂h = ∂m/∂h + (1/τ) * Σ_k [τ * exp(τ(p_k-m)) / Σ_j exp(τ(p_j-m))] * ∂p_k/∂h

The first term (∂m/∂h) is the gradient of the max, which is the gradient of whichever p_k achieves the max. The second term is a weighted average of the gradients.

For large τ, the softmax concentrates on the max element, so the total gradient is dominated by the gradient of the max element. For small τ, it's a smoother average.

Actually, the issue is that ∂m/∂h is not well-defined at points where multiple p_k achieve the max. But in practice, the LSE handles this smoothly.

```python
def greedy_binary_search(h, max_passes=100):

    M = 2*n - 1

    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for p in range(max_passes):
        # Compute influence for all positions
        infl = compute_influence(h, dx)
        
        # Find best swap using linear approximation
        best_improvement = 0
        best_swap = None
        
        for i in ones:
            for j in zeros:
                delta_approx = infl[i] - infl[j]
                new_max_approx = max(current_max, np.max(P + delta_approx))
                # Actually, we want the max of the new profile
                # The new profile is P + delta_approx (approximately)
                # new max = max_k (P[k] + delta_approx[k])
                new_max_approx = float(np.max(P + delta_approx))
                improvement = current_max - new_max_approx
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_swap = (i, j)
        
        if best_swap is None or best_improvement < 1e-12:
            break
        
        i, j = best_swap
        h[i] = 0
        h[j] = 1
        
        # Update P exactly
        P = profile_binary(h)
        current_max = float(np.max(P))
        
        if current_max < best_max:
            best_max = current_max
            best_h = h.copy()
        
        # Update ones and zeros
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return best_h, best_max
```
