from scipy.optimize import linprog

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        d = k - (n - 1)
        if d == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            L = n - d
            G[k, 0:L] += dx * (1.0 - h[d:d+L])
            G[k, d:n] -= dx * h[0:L]
        else:
            c = -d
            L = n - c
            G[k, c:n] += dx * (1.0 - h[0:L])
            G[k, 0:L] -= dx * h[c:n]
    return G

def repair(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])

    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair(h_new)

    n = len(h)
    M = 2 * n - 1
    if theta_ub is None:
        theta_ub = exact_score(h) + 0.5
    
    p = compute_profile(h)
    G = compute_gradient(h)
    
    # LP: minimize theta
    # Variables: x[0..n-1], theta
    # Constraints: G[k,:]·x - theta <= p[k] - G[k,:]·h
    #   0 <= x[i] <= 1
    #   |x[i] - h[i]| <= radius (trust region)
    #   sum(x) = n/2
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize theta
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = p - G @ h
    
    # Trust region constraints: h[i] - radius <= x[i] <= h[i] + radius
    # Already handled by bounds, but add explicit for clarity
    # Actually, we'll just modify bounds to incorporate trust region
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lb = max(0.0, h[i] - radius)
        ub = min(1.0, h[i] + radius)
        bounds.append((lb, ub))
    bounds.append((0.0, theta_ub))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    return res.x[:n], res.x[n]

        s = exact_score(trial)
        if s < best_s - 1e-12:

            best_trial = trial
    return best_trial, best_s

            L = n - d
            G[k, 0:L] += dx * (1.0 - h[d:d+L])
            G[k, d:n] -= dx * h[0:L]
        else:
            c = -d
            L = n - c
            G[k, c:n] += dx * (1.0 - h[0:L])
            G[k, 0:L] -= dx * h[c:n]
    return G

def repair(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair(h_new)

def solve_lp_trust(h, radius):
    n = len(h)
    M = 2 * n - 1
    p = compute_profile(h)
    G = compute_gradient(h)
    theta_ub = float(np.max(p)) + 1.0

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = p - G @ h

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = []
    for i in range(n):
        lb = max(0.0, h[i] - radius)
        ub = min(1.0, h[i] + radius)
        bounds.append((lb, ub))
    bounds.append((0.0, theta_ub))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None
    return res.x[:n], res.x[n]

def line_search(h, direction, current_score):
    alphas = np.array([1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.04, 0.02])
    best_trial = h
    best_s = current_score
    for a in alphas:
        trial = repair(h + a * direction)
        s = exact_score(trial)
        if s < best_s - 1e-12:
            best_s = s
            best_trial = trial.copy()
    return best_trial, best_s

def lp_optimize(h_start, score_start, t0, budget_s, max_time):
    h = h_start.copy()
    score = score_start
    radius = 0.35
    stalled = 0

    while True:
        elapsed = time.perf_counter() - t0
        if budget_s - elapsed < 8 or elapsed > t0 + max_time:
            break

        result = solve_lp_trust(h, radius)
        if result is None:
            radius *= 0.6
            stalled += 1
            if stalled > 8 or radius < 0.005:
                break
            continue

        x_lp, theta_lp = result
        direction = x_lp - h
        max_step = float(np.max(np.abs(direction)))

        if max_step < 1e-9:
            break

        h_new, s_new = line_search(h, direction, score)
        if s_new < score - 1e-11:
            h = h_new
            score = s_new
            stalled = 0
            radius = min(0.5, radius * 1.15)
        else:
            stalled += 1
            radius *= 0.6
            if stalled > 10 or radius < 0.005:
                break

    return h, score

def gradient_descent_step(h, profile, G, max_lags=20):
    n = len(h)
    score = float(np.max(profile))
    idx = np.argsort(profile)[::-1][:max_lags]
    w = profile[idx].copy()
    w = np.maximum(w, 0.0)
    wsum = w.sum()
    if wsum < 1e-15:
        return h, score
    w /= wsum
    direction = np.sum(w[:, None] * G[idx, :], axis=0)
    direction -= direction.mean()
    norm = np.linalg.norm(direction)
    if norm < 1e-14:
        return h, score
    direction /= norm
    best_h = h
    best_s = score
    for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003]:
        trial = repair(h - alpha * direction)
        s = exact_score(trial)
        if s < best_s - 1e-12:
            best_s = s
            best_h = trial.copy()
    return best_h, best_s

def make_perturbation(base, seed_val, eps_scale=1.0):
    rng = np.random.RandomState(seed_val)
    n = len(base)
    z = rng.randn(n)
    z -= z.mean()
    return repair(base + eps_scale * eps_scale * 0.03 * z)

def make_symmetric(base):
    return repair(0.5 * (base + base[::-1].copy()))

def make_antisymmetric(base):
    return repair(0.5 * (base + 1.0 - base[::-1].copy()))

def block_construct(n, nblocks=4):
    h = np.zeros(n)
    block_size = n // nblocks
    for b in range(nblocks):
        start = b * block_size
        end = start + block_size
        if b % 2 == 0:
            h[start:end] = 0.7
        else:
            h[start:end] = 0.3
    return repair(h)

def smooth_construct(n, freq=2, phase=0.0, amp=0.4):
    x = np.linspace(0, 2, n, endpoint=False) + 2.0/n
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    return repair(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t0)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(140, 0.5)

    best_h = repair(parent)
    best_score = exact_score(best_h)

    h_sym = make_symmetric(best_h)
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-13:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: LP optimization from best
    if tleft() > 40:
        tb = min(100, int(tleft() * 0.35))
        h_c, s_c = lp_optimize(best_h, best_score, time.perf_counter(), tleft(), tb)
        if s_c < best_score - 1e-12:
            best_h, best_score = h_c, s_c

    # Phase 2: Gradient refinement
    if tleft() > 20:
        prof = compute_profile(best_h)
        G = compute_gradient(best_h)
        gd_h, gd_s = gradient_descent_step(best_h, prof, G, max_lags=25)
        if gd_s < best_score - 1e-12:
            best_h, best_score = gd_h, gd_s
        # Second pass
        if tleft() > 15:
            prof = compute_profile(best_h)
            G = compute_gradient(best_h)
            gd_h, gd_s = gradient_descent_step(best_h, prof, G, max_lags=15)
            if gd_s < best_score - 1e-12:
                best_h, best_score = gd_h, gd_s

    # Phase 3: Try transformed starting points
    if tleft() > 50:
        transforms = [
            make_antisymmetric(best_h),
            repair(0.7 * best_h + 0.3 * best_h[::-1]),
            repair(0.6 * best_h + 0.4 * (1.0 - best_h[::-1])),
        ]
        for ti, h_tr in enumerate(transforms):
            if tleft() < 30:
                break
            s_tr = exact_score(h_tr)
            if s_tr < best_score - 1e-12:
                best_h, best_score = h_tr, s_tr
            if tleft() < 25:
                break
            tb = min(30, int(tleft() * 0.15))
            h_c, s_c = lp_optimize(h_tr, s_tr, time.perf_counter(), tleft(), tb)
            if s_c < best_score - 1e-12:
                best_h, best_score = h_c, s_c

    # Phase 4: Random perturbations + LP
    if tleft() > 60:
        nbest = len(best_h)
        for p in range(8):
            if tleft() < 30:
                break
            h_pert = make_perturbation(best_h, seed + 100 + p, eps_scale=1.0 + 0.5 * p)
            s_pert = exact_score(h_pert)
            tb = min(20, int(tleft() * 0.1))
            h_c, s_c = lp_optimize(h_pert, s_pert, time.perf_counter(), tleft(), tb)
            if s_c < best_score - 1e-12:
                best_h, best_score = h_c, s_c

    # Phase 5: Dimension search
    if tleft() > 70:
        n_parent = len(best_h)
        dims_to_try = []
        for scale in [0.7, 0.85, 1.15, 1.3, 1.5]:
            d = int(round(n_parent * scale))
            d = max(40, min(250, d))
            if d != n_parent and d not in dims_to_try:
                dims_to_try.append(d)
        for new_n in dims_to_try:
            if tleft() < 35:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            tb = min(30, int(tleft() * 0.15))
            h_c, s_c = lp_optimize(h_interp, s_interp, time.perf_counter(), tleft(), tb)
            if s_c < best_score - 1e-12:
                best_h, best_score = h_c, s_c

    # Phase 6: Structured constructions at current dimension
    if tleft() > 50:
        nbest = len(best_h)
        structured = []
        for nb in [3, 4, 5, 6]:
            structured.append(block_construct(nbest, nb))
        for f in [1, 2, 3]:
            for ph in [0.0, np.pi/4, np.pi/2]:
                structured.append(smooth_construct(nbest, f, ph, 0.35))
        for h_st in structured:
            if tleft() < 20:
                break
            s_st = exact_score(h_st)
            if s_st < best_score:
                tb = min(15, int(tleft() * 0.1))
                h_c, s_c = lp_optimize(h_st, s_st, time.perf_counter(), tleft(), tb)
                if s_c < best_score - 1e-12:
                    best_h, best_score = h_c, s_c

    # Phase 7: Final LP refinement with smaller radius
    if tleft() > 15:
        tb = min(20, int(tleft() * 0.5))
        h_c, s_c = lp_optimize(best_h, best_score, time.perf_counter(), tleft(), tb)
        if s_c < best_score - 1e-12:
            best_h, best_score = h_c, s_c

    # Phase 8: Final gradient polish
    if tleft() > 8:
        prof = compute_profile(best_h)
        G = compute_gradient(best_h)
        gd_h, gd_s = gradient_descent_step(best_h, prof, G, max_lags=20)
        if gd_s < best_score - 1e-12:
            best_h, best_score = gd_h, gd_s

    best_h = repair(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
