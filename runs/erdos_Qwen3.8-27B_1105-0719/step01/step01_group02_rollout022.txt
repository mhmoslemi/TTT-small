```python
def lp_step(h, c, delta=0.0, r=0.1, method='highs'):
    n = len(h)
    dx = 2.0 / n
    profiles = np.correlate(h, 1.0-h, mode='full') * dx
    
    # Get all gradients (2n-1 vectors of size n)
    # Active set
    active = np.where(profiles >= c - delta)[0]
    
    # Build constraint matrix
    # Variables: d_0...d_{n-1}, t (n+1 vars)
    # Minimize t: c_obj = [0,...,0,1]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    A_ub = []
    b_ub = []
    for k in active:
        s = (n-1) - k  # signed offset
        g = profile_gradient(h, s) * dx
        row = np.concatenate([g, [-1.0]])
        A_ub.append(row)
        b_ub.append(0.0)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -min(r, h[i])
        hi = min(r, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))  # t unbounded
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method=method)
    
    if res.status != 0:
        return None, None
    
    d = res.x[:n]
    t_opt = res.x[n]
    return d, t_opt
```

```python
        row = np.zeros(nvars)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
    A_ub = np.array(A_rows)
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        lo = -min(r, h[i] + 1e-12)
        hi = min(r, 1.0 - h[i] - 1e-12)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, float('inf')
    if res.status != 0:
        return None, float('inf')
    d = res.x[:n]
    t_opt = res.x[n]
    if t_opt >= -1e-9:
        return None, float('inf')
    max_abs_d = np.max(np.abs(d)) + 1e-12
    alpha_max = min(1.0, r / max_abs_d)
    c_cur = exact_score(h)
    best_h = None
    best_c = c_cur
    for alpha in [alpha_max, alpha_max * 0.5, alpha_max * 0.25, alpha_max * 0.1]:
        if alpha <= 1e-7:
            break
        h_new = project_valid(h + alpha * d)
        c_new = exact_score(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    if best_h is not None:
        return best_h, best_c
    return None, c_cur


def pair_swap_optimize(h, n_trials, rng, alpha_scale=0.15):
    n = len(h)
    c_cur = exact_score(h)
    h_cur = h.copy()
    for t in range(n_trials):
        frac = t / float(n_trials)
        scale = alpha_scale * (0.5 + 1.0 * (1.0 - frac))
        i, j = rng.choice(n, 2, replace=False)
        if h_cur[i] > h_cur[j]:
            i, j = j, i
        cap = min(h_cur[j], 1.0 - h_cur[i])
        if cap < 1e-6:
            continue
        alpha = scale * cap
        if alpha > cap:
            alpha = cap
        if alpha < 1e-6:
            continue
        h_cur[i] += alpha
        h_cur[j] -= alpha
        h_cur = project_valid(h_cur)
        c_new = exact_score(h_cur)
        if c_new < c_cur - 1e-13:
            c_cur = c_new
        else:
            h_cur[i] -= alpha
            h_cur[j] += alpha
            h_cur = project_valid(h_cur)
    return h_cur, c_cur


def deterministic_pair_swap(h, n_rounds=10, rng=None):
    n = len(h)
    h_cur = h.copy()
    c_cur = exact_score(h_cur)
    if rng is None:
        rng = np.random.RandomState(0)
    for rnd in range(n_rounds):
        improved = False
        idx = rng.permutation(n)
        for i in idx:
            for j in idx:
                if i == j:
                    continue
                if h_cur[i] > h_cur[j]:
                    continue
                cap = min(h_cur[j], 1.0 - h_cur[i])
                if cap < 1e-5:
                    continue
                for frac in [0.3, 0.7, 1.0]:
                    alpha = frac * cap
                    h_cur[i] += alpha
                    h_cur[j] -= alpha
                    h_cur = project_valid(h_cur)
                    c_new = exact_score(h_cur)
                    if c_new < c_cur - 1e-12:
                        c_cur = c_new
                        improved = True
                        break
                    else:
                        h_cur[i] -= alpha
                        h_cur[j] += alpha
                        h_cur = project_valid(h_cur)
        if not improved:
            break
    return h_cur, c_cur


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b


def binary_swap_improve(b, max_rounds=200, rng_seed=42):
    n = len(b)
    cur = b.copy()
    c_cur = exact_score(cur)
    rng = np.random.RandomState(rng_seed)
    for _ in range(max_rounds):
        best_delta = 0.0
        best_i, best_j = -1, -1
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(ones) > 200:
            break
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = exact_score(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                delta = c_cur - c
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i >= 0:
            cur[best_i] = 0.0
            cur[best_j] = 1.0
            c_cur -= best_delta
        else:
            break
    return cur, c_cur


def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_valid(new_h)


def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        parent = project_valid(parent)
        pc = exact_score(parent)
        best_h = parent.copy()
        best_c = pc
    except Exception:
        best_h = np.ones(70) * 0.5
        best_h = project_valid(best_h)
        best_c = exact_score(best_h)

    n0 = len(best_h)

    # Phase 1: LP-based optimization on the parent size
    h_cur = best_h.copy()
    c_cur = best_c

    delta_schedule = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    r_schedule = [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03]

    for di, delta in enumerate(delta_schedule):
        if time.time() > deadline - 120:
            break
        r = r_schedule[min(di, len(r_schedule) - 1)]
        progress = False
        for it in range(200):
            if time.time() > deadline - 120:
                break
            h_new, c_new = lp_step(h_cur, delta, r, max_active=150)
            if h_new is not None:
                h_cur = h_new
                c_cur = c_new
                progress = True
                if c_cur < best_c - 1e-13:
                    best_c = c_cur
                    best_h = h_cur.copy()
            else:
                # Try full LP
                if r > 0.05:
                    h_new2, c_new2 = lp_step_full(h_cur, r * 0.7)
                    if h_new2 is not None:
                        h_cur = h_new2
                        c_cur = c_new2
                        if c_cur < best_c - 1e-13:
                            best_c = c_cur
                            best_h = h_cur.copy()
                    else:
                        # Perturb slightly and retry
                        h_cur = project_valid(h_cur + 0.01 * rng.randn(n0))
                        c_cur = exact_score(h_cur)
                        continue
                else:
                    break
        if not progress and di < len(delta_schedule) - 1:
            continue

    # Phase 2: Pair swap optimization
    if time.time() < deadline - 60:
        h_sw, c_sw = pair_swap_optimize(h_cur, n_trials=3000, rng=rng, alpha_scale=0.12)
        if c_sw < c_cur - 1e-13:
            h_cur = h_sw
            c_cur = c_sw
            if c_cur < best_c - 1e-13:
                best_c = c_cur
                best_h = h_cur.copy()

    # Phase 3: More LP steps at fine scale
    if time.time() < deadline - 60:
        for delta in [0.0003, 0.0001]:
            for r in [0.05, 0.03]:
                for it in range(100):
                    if time.time() > deadline - 60:
                        break
                    h_new, c_new = lp_step(h_cur, delta, r, max_active=100)
                    if h_new is not None:
                        h_cur = h_new
                        c_cur = c_new
                        if c_cur < best_c - 1e-13:
                            best_c = c_cur
                            best_h = h_cur.copy()
                    else:
                        break

    # Phase 4: Binary threshold + swap
    if time.time() < deadline - 40 and n0 % 2 == 0:
        b = binary_threshold(h_cur)
        if b is not None:
            b_c = exact_score(b)
            if b_c < best_c - 1e-13:
                best_c = b_c
                best_h = b.copy()
            b2, b2_c = binary_swap_improve(b, max_rounds=300, rng_seed=seed)
            if b2_c < best_c - 1e-13:
                best_c = b2_c
                best_h = b2.copy()

    # Phase 5: Try different sizes
    if time.time() < deadline - 90:
        sizes_to_try = []
        for s in [n0, n0 - 4, n0 + 4, 120, 130, 150, 160, 100, 110]:
            if s >= 20 and s != n0 and s % 2 == 0 and s not in sizes_to_try:
                sizes_to_try.append(s)

        for s in sizes_to_try:
            if time.time() > deadline - 40:
                break
            h_tr = transfer_to_size(best_h, s)
            c_tr = exact_score(h_tr)
            if c_tr < best_c + 0.002:
                # Try LP on this size
                for delta in [0.003, 0.001, 0.0005]:
                    if time.time() > deadline - 40:
                        break
                    for r in [0.15, 0.08]:
                        for it in range(80):
                            if time.time() > deadline - 40:
                                break
                            h_new, c_new = lp_step(h_tr, delta, r, max_active=100)
                            if h_new is not None:
                                h_tr = h_new
                                c_tr = c_new
                                if c_tr < best_c - 1e-13:
                                    best_c = c_tr
                                    best_h = h_tr.copy()
                            else:
                                break
                if c_tr < best_c + 0.001:
                    h_sw2, c_sw2 = pair_swap_optimize(h_tr, n_trials=1500, rng=rng, alpha_scale=0.1)
                    if c_sw2 < best_c - 1e-13:
                        best_c = c_sw2
                        best_h = h_sw2.copy()

    # Final: ensure feasibility and compute exact score
    best_h = project_valid(best_h)
    best_c = exact_score(best_h)

    return (best_h, best_c, len(best_h))
```
