def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    """Project h to [0,1]^n with sum = n/2 using bisection on shift."""
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    # Bisection for lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo = -2.0
    hi = 2.0
    # First find bounds
    for _ in range(50):
        s = np.sum(np.clip(h + lo, 0, 1))
        if s > target:
            lo -= 1.0
        else:
            break
    for _ in range(50):
        s = np.sum(np.clip(h + hi, 0, 1))
        if s < target:
            hi += 1.0
        else:
            break
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0, 1)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            result[mask] += residual / cnt
    return np.clip(result, 0.0, 1.0)

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    idx = np.argsort(h)[::-1][:n//2]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0

    """Greedy binary swap search."""
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):

                b[i] = 1.0
                b[j] = 0.0

def project_box_sum(h):

def make_binary(n, rng, mode):

        x = rng.permutation(n)
        b = np.zeros(n)
        b[x[:n//2]] = 1.0
    elif mode == 1:  # Tent map
        x = np.arange(n) / n
        b = (2 * np.abs(2 * x - 1) < 1).astype(float)
        # Adjust to have exactly n/2 ones
        b = _fix_count(b, n, rng)
    elif mode == 2:  # Sawtooth
        x = np.arange(n) / n
        b = ((x * 3) % 1 < 1/3 + 1/6).astype(float)  # Not exact
        b = _fix_count(b, n, rng)
    elif mode == 3:  # Modular
        p = rng.choice([2,3,5,7,11,13])
        b = ((np.arange(n) * p) % n < n//2 - n//4).astype(float)
        b = _fix_count(b, n, rng)
    elif mode == 4:  # Alternating blocks
        block = max(2, n // 8)
        b = np.zeros(n)
        for i in range(0, n, 2*block):
            b[i:i+block] = 1.0
        b = _fix_count(b, n, rng)
    elif mode == 5:  # Two-gap pattern
        b = np.zeros(n)
        gap1 = n // 3
        gap2 = n // 3
        b[:n//4] = 1.0
        b[gap1:gap1+n//4] = 1.0
        b = _fix_count(b, n, rng)
    else:
        b = np.zeros(n)
        b[::2] = 1.0
        b = _fix_count(b, n, rng)
    return b

def _fix_count(b, n, rng):
    """Adjust binary vector to have exactly n/2 ones."""
    target = n // 2
    cnt = int(b.sum())
    if cnt == target:
        return b
    b = b.copy()
    if cnt > target:
        idx = np.where(b == 1.0)[0]
        to_remove = cnt - target
        perm = rng.permutation(len(idx))[:to_remove]
        b[idx[perm]] = 0.0
    else:
        idx = np.where(b == 0.0)[0]
        to_add = target - cnt
        perm = rng.permutation(len(idx))[:to_add]
        b[idx[perm]] = 1.0
    return b

def swap_search(b, max_rounds=500, deadline=None, rng=None):

    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is None:

        i, j = best_swap
        b[i] = 0.0
        b[j] = 1.0
        curr_c = exact_objective(b)
        ones = list(np.where(b == 1.0)[0])
        zeros = list(np.where(b == 0.0)[0])
    return b, curr_c

                if src_sum == 0 or src_sum == blen:

                    if imp > best_improve:

    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
                best_n = len(parent_h)
    except Exception:
        pass

    if best_h is None:
        parent_h = make_binary(48, rng, 0)
        best_c = exact_objective(parent_h)
        best_h = parent_h.copy()
        best_n = 48

    parent_n = len(parent_h)

    # Sizes to try (even, structured)
    all_sizes = set()
    all_sizes.add(parent_n)
    for s in [40, 44, 48, 52, 56, 60, 64, 72, 80, 88, 96, 104, 112, 120, 128, 144, 160]:
        all_sizes.add(s)
    sizes = sorted([s for s in all_sizes if 24 <= s <= 160 and s % 2 == 0])

    # Allocate time per size
    n_sizes = len(sizes)
    time_per_size = (budget_s - 30) / n_sizes
    # Give more time to parent size and nearby sizes
    size_priority = {}
    for idx, s in enumerate(sizes):
        dist = abs(s - parent_n)
        if dist == 0:
            size_priority[s] = 2.0
        elif dist <= 8:
            size_priority[s] = 1.5
        elif dist <= 24:
            size_priority[s] = 1.0
        else:
            size_priority[s] = 0.6

    total_weight = sum(size_priority[s] for s in sizes)
    time_alloc = {}
    for s in sizes:
        time_alloc[s] = time_per_size * size_priority[s] / total_weight * n_sizes

    # Process sizes in priority order
    sizes_ordered = sorted(sizes, key=lambda s: -size_priority[s])

    for n in sizes_ordered:
        if time.time() > deadline - 5:
            break
        size_deadline = min(deadline - 5, t_start + time_alloc[n])

        # Generate candidates
        cands = []
        if parent_h is not None:
            transferred = transfer_to_size(parent_h, n)
            b_t = threshold_to_binary(transferred)
            if b_t is not None:
                cands.append(b_t)
            cands.append(b_t)
            # Perturbed parent
            h_pert = project_box_sum(transferred + rng.normal(0, 0.02, n))
            b_p = threshold_to_binary(h_pert)
            if b_p is not None:
                cands.append(b_p)

        for mode in range(7):
            cands.append(make_binary(n, rng, mode))

        # Score candidates
        scored = []
        for c in cands:
            if c is None:
                continue
            c = project_box_sum(c)
            # Make binary
            b = threshold_to_binary(c)
            if b is None:
                continue
            sc = exact_objective(b)
            scored.append((sc, b))
        scored.sort(key=lambda x: x[0])

        # Remove duplicates (same binary vector)
        seen = set()
        unique_scored = []
        for sc, b in scored:
            key = tuple(b.astype(int))
            if key not in seen:
                seen.add(key)
                unique_scored.append((sc, b))
        scored = unique_scored

        # Refine top candidates with ILS
        n_top = min(4, len(scored))
        for idx in range(n_top):
            if time.time() > size_deadline:
                break
            sc, b0 = scored[idx]
            if sc >= best_c and idx > 0:
                break

            # ILS
            b_ils, c_ils = iterated_local_search(b0, size_deadline, rng,
                                                  max_iters=60, perturbation=3)
            if c_ils < best_c - 1e-12:
                best_c = c_ils
                best_h = b_ils.copy()
                best_n = n

            # Block swap refinement
            if time.time() < size_deadline - 2:
                b_blk, c_blk = block_swap_search(b_ils, max_rounds=30, deadline=size_deadline)
                if c_blk < best_c - 1e-12:
                    best_c = c_blk
                    best_h = b_blk.copy()
                    best_n = n

            # Also try SA from the best
            if time.time() < size_deadline - 3:
                b_sa, c_sa = simulated_annealing(b_ils, size_deadline, rng, max_iter=150000)
                if c_sa < best_c - 1e-12:
                    best_c = c_sa
                    best_h = b_sa.copy()
                    best_n = n
                # Follow up with swap search
                b_fin, c_fin = swap_search(b_sa, max_rounds=100, deadline=size_deadline)
                if c_fin < best_c - 1e-12:
                    best_c = c_fin
                    best_h = b_fin.copy()
                    best_n = n

    # Final phase: intensive search on best solution
    if best_h is not None and time.time() < deadline - 20:
        n_best = len(best_h)
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            # More ILS iterations with varying perturbation sizes
            for pert in [2, 4, 5, 7, 10]:
                if time.time() > deadline - 10:
                    break
                b_new, c_new = iterated_local_search(best_h, deadline - 10, rng,
                                                     max_iters=30, perturbation=pert)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = b_new.copy()
                    best_n = n_best

            # Block swap refinement
            b_blk, c_blk = block_swap_search(best_h, max_rounds=50, deadline=deadline - 5)
            if c_blk < best_c - 1e-12:
                best_c = c_blk
                best_h = b_blk.copy()

            # One more SA + swap
            if time.time() < deadline - 10:
                b_sa, c_sa = simulated_annealing(best_h, deadline - 10, rng, max_iter=200000)
                b_fin, c_fin = swap_search(b_sa, max_rounds=150, deadline=deadline - 5)
                if c_fin < best_c - 1e-12:
                    best_c = c_fin
                    best_h = b_fin.copy()
                    best_n = n_best

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
