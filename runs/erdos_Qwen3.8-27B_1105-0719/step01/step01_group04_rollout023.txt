        if score < best_score - 1e-10:
            best_h = h_sol
            best_score = score
            best_n = len(h_sol)

    # Final validation
    best_h = project_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
