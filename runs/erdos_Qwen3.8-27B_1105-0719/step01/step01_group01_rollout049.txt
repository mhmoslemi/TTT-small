  Wait, G[i, idx] = 1 - h[i+k] - h[i-k]. If h[p] increases by τ:
  - G[i, idx] decreases by τ if i+k = p, i.e., i = p-k
  - G[i, idx] decreases by τ if i-k = p, i.e., i = p+k
  
  And for h[q] decreasing by τ:
  - G[i, idx] increases by τ if i+k = q, i.e., i = q-k
  - G[i, idx] increases by τ if i-k = q, i.e., i = q+k

So for each of the n_corr lags, at most 4 entries in G change. Total: 4 * n_corr = O(n) updates per move. Much better than rebuilding G!

```python
def update_G_after_move(G, h, p, q, tau, n):
    """Update G after h[p] += tau and h[q] -= tau."""
    n_corr = 2 * n - 1
    for idx in range(n_corr):
        k = idx - (n - 1)
        # h[p] increased by tau: affects G[p-k, idx] and G[p+k, idx]
        if 0 <= p - k < n:
            G[p - k, idx] -= tau  # h[i-k] = h[p] appeared, now larger
        if 0 <= p + k < n:
            G[p + k, idx] -= tau  # h[i+k] = h[p] appeared, now larger
        # h[q] decreased by tau: affects G[q-k, idx] and G[q+k, idx]
        if 0 <= q - k < n:
            G[q - k, idx] += tau  # h[i-k] = h[q] appeared, now smaller
        if 0 <= q + k < n:
            G[q + k, idx] += tau  # h[i+k] = h[q] appeared, now smaller
```

```python
def compute_c5(h):

    dx = 2.0 / n

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
            r2 = target - np.sum(h)
            if abs(r2) > 1e-12:
                interior = np.where((h > 0) & (h < 1))[0]
                if len(interior) > 0:

    G = np.ones((n, n_corr))
    for idx in range(n_corr):

        lo1 = max(0, -k)

        hi2 = min(n - 1, k + n - 1)

    if cap <= 1e-14:
        return 0.0

    fcap = float(np.max(R + cap * d))

        return 0.0

        if d[int(np.argmax(R + mid * d))] < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

            if 1.0 - h[p] < 1e-10:

        h[q] -= tau

def resample_h(h_old, n_new):
    """Resample h from n_old points to n_new points, preserving area."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    # For each new interval [j*dx_new, (j+1)*dx_new], compute average height
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        # Find overlapping old intervals
        i_start = int(x_start / dx_old)
        i_end = int(np.ceil(x_end / dx_old)) - 1  # inclusive
        i_start = max(0, i_start)
        i_end = min(n_old - 1, i_end)
        total_overlap = 0.0
        total_area = 0.0
        for i in range(i_start, i_end + 1):
            old_start = i * dx_old
            old_end = (i + 1) * dx_old
            ov_start = max(x_start, old_start)
            ov_end = min(x_end, old_end)
            if ov_end > ov_start:
                total_area += h_old[i] * (ov_end - ov_start)
                total_overlap += (ov_end - ov_start)
        if total_overlap > 0:
            h_new[j] = total_area / total_overlap
    h_new = np.clip(h_new, 0.0, 1.0)

    return h_new

def slsqp_refine(h, betas, max_iter=200):
    """SLSQP refinement with smoothed max."""
    n = len(h)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    def smooth_obj(x, beta):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        s = np.sum(np.exp(beta * (corr - m)))
        return (m + np.log(s) / beta) * dx
    
    def smooth_grad(x, beta):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        w = exps / np.sum(exps)
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            k = idx - (n - 1)
            wi = w[idx]
            if wi < 1e-13:
                continue
            lo1 = max(0, k)
            hi1 = min(n - 1, n - 1 + k)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += wi * (1.0 - x[lo1+k:hi1+1+k])
            lo2 = max(0, -k)
            hi2 = min(n - 1, n - 1 - k)
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= wi * x[lo2-k:hi2+1-k]
        return grad * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    for beta in betas:
        res = minimize(smooth_obj, best_h, args=(beta,), jac=smooth_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.success or (res.x is not None):
            c5 = compute_c5(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x.copy()
    return best_h, best_c5
```
