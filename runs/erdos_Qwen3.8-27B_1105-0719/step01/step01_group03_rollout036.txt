  = -argmax_{||v||=1} min_k (-∇f_k · v)  [for active k]

By the minimax theorem, this equals:
d = -∑ λ_k ∇f_k where λ minimizes ||∑ λ_k ∇f_k|| subject to λ ≥ 0, ∑λ = 1

For a small active set, I can solve this QP exactly.

def optimal_active_descent_direction(G_active):
    """Find optimal weights and direction for active-set descent.
    G_active has shape (m, n) where m is the number of active constraints.
    Returns (weights, direction)."""
    m = G_active.shape[0]
    n = G_active.shape[1]
    
    # Center the gradients (remove mean to maintain sum constraint)
    G_centered = G_active - np.mean(G_active, axis=1, keepdims=True)
    
    # Solve: min_w ||G_centered.T @ w||^2 s.t. w >= 0, sum(w) = 1
    # This is: min w' (G G') w s.t. w >= 0, 1'w = 1
    # where G = G_centered
    
    # For small m, use scipy
    from scipy.optimize import minimize as sp_minimize
    
    def q_obj(w):
        v = G_centered.T @ w
        return v @ v
    
    def q_jac(w):
        v = G_centered.T @ w
        return 2.0 * G_centered @ v
    
    w0 = np.ones(m) / m
    bounds = [(0, 1)] * m
    constraints = [{'type': 'eq', 'fun': lambda w: w.sum() - 1.0, 'jac': lambda w: np.ones(m)}]
    
    try:
        res = sp_minimize(q_obj, w0, method='SLSQP', jac=q_jac,
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 100, 'ftol': 1e-12})
        w = res.x
        if w.sum() > 0:
            w = w / w.sum()
        else:
            w = w0
    except Exception:
        w = w0
    
    direction = G_centered.T @ w
    return w, direction

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def softmax_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    max_p = np.max(profile)
    shifted = np.clip(beta * (profile - max_p), -400, 400)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = (max_p + np.log(sum_exp)) / beta
    w = exp_s / sum_exp
    G = build_gradient_matrix(h, n, dx)
    grad = G.T @ w
    return obj, grad

def pgd_softmax(h0, beta, max_iter=200):
    n = len(h0)
    h = h0.copy()
    best_obj, _ = softmax_obj_grad(h, beta)
    best_h = h.copy()
    for _ in range(max_iter):
        obj, grad = softmax_obj_grad(h, beta)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        alpha = 1.0 / (beta * 4.0 + 1.0)
        for _ in range(40):
            h_try = project_feasible(h - alpha * grad)
            obj_try, _ = softmax_obj_grad(h_try, beta)
            if obj_try <= obj - 1e-4 * alpha * gn * gn:
                break
            alpha *= 0.5
        else:
            break
        h = h_try
    if exact_score(h) < exact_score(best_h):
        best_h = h.copy()
    return best_h

def active_descent(h, max_iters=500, rng=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best:
            best = c
        thresh = c - 1e-3
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        slack = profile[active] - (c - 1e-3)
        w = 1.0 / (1.0 + (1.0 - slack / max(1e-3, 1e-30)) * 200.0)
        w = w / np.sum(w)
        G = build_gradient_matrix(h, n, dx)
        grad = G[active, :].T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        grad /= gn
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]:
            h_try = project_feasible(h - alpha * grad)
            s = exact_score(h_try)
            if s < best - 1e-14:
                h = h_try
                best = s
                improved = True
                break
        if not improved:
            # Try wider active set
            thresh2 = c - 5e-3
            active2 = np.where(profile >= thresh2)[0]
            if len(active2) > len(active):
                w2 = np.ones(len(active2)) / len(active2)
                if rng is not None:
                    w2 = w2 + rng.rand(len(active2)) * 0.1
                    w2 = w2 / w2.sum()
                grad2 = G[active2, :].T @ w2
                grad2 -= np.mean(grad2)
                gn2 = np.linalg.norm(grad2)
                if gn2 > 1e-13:
                    grad2 /= gn2
                    for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
                        h_try = project_feasible(h - alpha * grad2)
                        s = exact_score(h_try)
                        if s < best - 1e-14:
                            h = h_try
                            best = s
                            improved = True
                            break
            if not improved:
                break
    return h, best

def fine_descent(h, max_iters=1000, rng=None):
    """Very fine active-set with shrinking step sizes."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = exact_score(h)
    step_scale = 1.0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best:
            best = c
        thresh = c - 2e-4
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        w = np.ones(len(active)) / len(active)
        G = build_gradient_matrix(h, n, dx)
        grad = G[active, :].T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        improved = False
        base_steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        for alpha in [a * step_scale for a in base_steps]:
            if alpha < 1e-7:
                continue
            h_try = project_feasible(h - alpha * grad)
            s = exact_score(h_try)
            if s < best - 1e-15:
                h = h_try
                best = s
                improved = True
                break
        if improved:
            step_scale = max(0.5, step_scale * 0.95)
        else:
            break
    return h, best

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def slsqp_polish(h0, maxiter=500, ftol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    c0 = exact_score(h0)
    M = 2 * n - 1
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            return h_new
    except Exception:
        pass
    return None
