def subgrad_line_search(h0, dx, target_sum, max_iter, top_m=8):
    """Subgradient method with backtracking line search and momentum."""
    n = len(h0)
    h = h0.copy()
    h_mom = h0.copy()  # momentum
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    gamma = 0.3  # step size
    momentum = 0.7
    no_improve = 0
    
    for t in range(max_iter):
        corrs = compute_corrs(h)
        c5_cur = corrs.max() * dx
        
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        
        # Compute subgradient using top-m
        top_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += grad_corr_k(h, k, n)
        grad /= len(top_idx)
        grad *= dx
        
        # Line search
        best_val = c5_cur
        best_h_new = h.copy()
        gam = gamma
        for ls in range(15):
            h_trial = h - gam * grad
            h_trial = project_feasible(h_trial, target_sum)
            val = c5_of_h(h_trial, dx)
            if val < best_val:
                best_val = val
                best_h_new = h_trial
                break
            gam *= 0.5
            if gam < 1e-10:
                break
        
        # Momentum update
        h_mom = momentum * h_mom + (1 - momentum) * best_h_new
        h = project_feasible(h_mom, target_sum)
        c5_h = c5_of_h(h, dx)
        if c5_h > best_val:
            h = best_h_new  # revert if momentum made it worse
        
        gamma = min(gamma * 1.01, 1.0)
    
    return best_h, best_c5

c_k = sum_n h[n+k] - sum_n h[n+k]*h[n] (times dx)

The Hessian of sum_n h[n+k]*h[n] with respect to h is:

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def gradient_smooth_max(h, dx, beta, active_idx=None):
    n = len(h)
    corrs = compute_corrs(h)
    if active_idx is not None:
        c_sub = corrs[active_idx]
        shifted = c_sub - c_sub.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(c_sub) / len(c_sub)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for i in range(len(active_idx)):
            if w[i] < 1e-15:
                continue
            grad += w[i] * grad_corr_k(h, active_idx[i], n)
        return grad * dx
    else:
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)

        if w_sum < 1e-30:

        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt, c5_val = slsqp_optimize(h_interp, dx_try, ts_try, 300, 500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining_time(t0, budget_s) > 30:
                h_as2, c5_as2 = active_set_refine(h_opt, dx_try, ts_try,
                                                   top_k=4, beta=1500, maxiter=400, n_rounds=3)
                if c5_as2 < best_c5:
                    best_c5 = c5_as2
                    best_h = h_as2

    # Phase 7: Final high-precision polish
    if remaining_time(t0, budget_s) > 20 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_fin, c5_fin = slsqp_optimize(best_h, dx_b, ts_b, 3000, 800)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
        if remaining_time(t0, budget_s) > 15:
            h_as3, c5_as3 = active_set_refine(best_h, dx_b, ts_b,
                                               top_k=3, beta=5000, maxiter=500, n_rounds=3)
            if c5_as3 < best_c5:
                best_c5 = c5_as3
                best_h = h_as3
        if remaining_time(t0, budget_s) > 10:
            h_sg2, c5_sg2 = subgrad_polish(best_h, dx_b, ts_b, 5000, top_m=3)
            if c5_sg2 < best_c5:
                best_c5 = c5_sg2
                best_h = h_sg2

    return best_h, best_c5, len(best_h)
