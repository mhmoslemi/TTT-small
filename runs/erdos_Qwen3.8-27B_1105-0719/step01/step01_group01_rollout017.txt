    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = smooth_max(corr, beta)

def smooth_grad(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    grad = np.zeros(n)

        m_lo2 = max(0, -k)

        else:

**Approach: Frank-Wolfe / Conditional Gradient**

Since the feasible set is a polytope (intersection of the hyperplane sum=n/2 with the cube [0,1]^n), we can use the Frank-Wolfe algorithm:
1. Compute subgradient g at current h
2. Find s in argmin_{s ∈ F} g · s (where F is the feasible set)
3. Update: h_new = (1-γ)h + γs

The linear minimization over F = {s: sum(s)=n/2, 0≤s≤1} can be done by sorting: s_i = 1 if w_i is among the largest n/2 values, s_i = 0 otherwise, with one fractional value to hit the exact sum.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_and_lags(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    lags = np.arange(-(n-1), n)  # k values
    return corr, lags

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

def smooth_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = smooth_max(corr, beta)
    return val * (2.0 / len(h))

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, weights = smooth_max(corr, beta)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        m_lo1 = max(0, k)
        m_hi1 = min(n - 1, n - 1 + k)
        if m_lo1 <= m_hi1:
            grad[m_lo1:m_hi1 + 1] += w * (1.0 - h[m_lo1 - k:m_hi1 + 1 - k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, n - 1 - k)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2 + 1] -= w * h[m_lo2 + k:m_hi2 + 1 + k]
    return grad * dx

def active_gradient(h):
    """Compute gradient of the max (using uniform weights on active lags)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Active lags: those within a small tolerance of max
    active = np.where(corr >= m - 1e-10)[0]
    weights = np.zeros(2*n-1)
    weights[active] = 1.0 / len(active)
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        m_lo1 = max(0, k)
        m_hi1 = min(n - 1, n - 1 + k)
        if m_lo1 <= m_hi1:
            grad[m_lo1:m_hi1 + 1] += w * (1.0 - h[m_lo1 - k:m_hi1 + 1 - k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, n - 1 - k)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2 + 1] -= w * h[m_lo2 + k:m_hi2 + 1 + k]
    return grad * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            per = residual / len(eligible)
            h[eligible] = np.clip(h[eligible] + per, 0.0, 1.0)
    return h

def slsqp_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, betas, max_iter_per_beta=200):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac}]
    for beta in betas:
        try:
            res = minimize(
                smooth_obj, best_h, args=(beta,),
                jac=smooth_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter_per_beta, 'ftol': 1e-16}
            )
            if res.x is not None:

                c5_new = compute_c5(h_new)

                    best_c5 = c5_new
                    best_h = h_new
        except:
            pass
    return best_h, best_c5

def transfer_local_search(h, n_iter, rng, use_annealing=True, T0=0.005, T_end=0.0001):
    """Local search using mass transfer moves."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5
    
    for it in range(n_iter):
        if use_annealing:
            frac = it / max(1, n_iter - 1)
            T = T0 * (T_end / T0) ** frac
        else:
            T = 0
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Transfer from the larger to the smaller (or random direction)
        if current_h[i] >= current_h[j]:
            max_tr = current_h[i] - current_h[j]
            di, dj = i, j  # decrease i, increase j
        else:
            max_tr = current_h[j] - current_h[i]
            di, dj = j, i
        
        if max_tr < 1e-9:
            continue
        
        # Try multiple step sizes
        best_delta = 0
        best_eps = 0
        for frac in [0.2, 0.5, 0.8, 1.0]:
            eps = max_tr * frac * (0.5 + 0.5 * rng.random())
            if eps < 1e-9:
                continue
            new_h = current_h.copy()
            new_h[di] -= eps
            new_h[dj] += eps
            new_c5 = compute_c5(new_h)
            delta = new_c5 - current_c5
            if delta < best_delta:
                best_delta = delta
                best_eps = eps
        
        if best_eps == 0:
            continue
        
        # Acceptance criterion
        if best_delta < -1e-15:
            accept = True
        elif T > 0 and best_delta > 0:
            accept = rng.random() < np.exp(-best_delta / T)
        else:
            accept = False
        
        if accept:
            current_h[di] -= best_eps
            current_h[dj] += best_eps
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    return best_h, best_c5

def gradient_local_search(h, n_steps, lr_init=0.01, beta=200):
    """Gradient descent with projection."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current = h.copy()
    lr = lr_init
    
    for s in range(n_steps):
        grad = smooth_grad(current, beta)
        grad = grad - np.mean(grad)  # project to sum=0 subspace
        new = current - lr * grad
        new = project_feasible(new, n)
        new_c5 = compute_c5(new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new.copy()
        
        if new_c5 <= compute_c5(current) + 1e-15:
            current = new
            lr *= 0.95  # slightly decrease for stability
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    
    return best_h, best_c5

def frank_wolfe_step(h, grad, n):
    """Find the FW vertex: minimize grad·s over s in {sum=s/n, 0<=s<=1}."""
    # The minimizer of grad·s subject to sum(s)=n/2, 0<=s<=1
    # is: s_i = 1 if grad_i is among the smallest n/2 values
    #      s_i = 0 if grad_i is among the largest n/2 values
    #      s_i = fractional for the boundary value
    sorted_idx = np.argsort(grad)
    s = np.zeros(n)
    s[sorted_idx[:n//2]] = 1.0
    # Check sum
    # Since we need exactly n/2, and we set n/2 values to 1, sum = n/2. 
    # But this only works if n is even. For odd n, we need one fractional value.
    # Actually, n/2 might not be integer. Let's handle generally.
    target = n / 2.0
    if int(target) == target:
        s[sorted_idx[:int(target)]] = 1.0
    else:
        k = int(target)
        s[sorted_idx[:k]] = 1.0
        s[sorted_idx[k]] = target - k
    return s
