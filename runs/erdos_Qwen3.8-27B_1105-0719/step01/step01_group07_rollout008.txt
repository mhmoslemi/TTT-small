import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argpartition(h, -m)[-m:]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def binary_sa_search(b0, deadline, rng, n_iter=200000):
    n = len(b0)
    b = b0.copy()
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    T = 0.004
    T_min = 1e-8
    cool = 0.99995
    for it in range(n_iter):
        if time.time() > deadline - 5:
            break
        r = rng.random()
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if r < 0.55:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            b[i] = 0.0
            b[j] = 1.0
            new_c = exact_objective(b)
            delta = new_c - curr_c
            if delta < -1e-15 or (T > T_min and rng.random() < np.exp(-delta / T)):
                curr_c = new_c
                if new_c < best_c - 1e-15:
                    best_c = new_c
                    best_b = b.copy()
            else:
                b[i] = 1.0
                b[j] = 0.0
        elif r < 0.85:
            if len(ones) < 2 or len(zeros) < 2:
                T *= cool
                continue
            i1, i2 = int(ones[rng.randint(len(ones))]), int(ones[rng.randint(len(ones))])
            if i1 == i2:
                T *= cool
                continue
            j1, j2 = int(zeros[rng.randint(len(zeros))]), int(zeros[rng.randint(len(zeros))])
            if j1 == j2:
                T *= cool
                continue
            b[i1] = 0.0; b[i2] = 0.0; b[j1] = 1.0; b[j2] = 1.0
            new_c = exact_objective(b)
            delta = new_c - curr_c
            if delta < -1e-15 or (T > T_min and rng.random() < np.exp(-delta / T)):
                curr_c = new_c
                if new_c < best_c - 1e-15:
                    best_c = new_c
                    best_b = b.copy()
            else:
                b[i1] = 1.0; b[i2] = 1.0; b[j1] = 0.0; b[j2] = 0.0
        else:
            # Triple swap: 3 ones <-> 3 zeros
            if len(ones) < 3 or len(zeros) < 3:
                T *= cool
                continue
            ii = rng.choice(ones, size=3, replace=False)
            jj = rng.choice(zeros, size=3, replace=False)
            b[ii] = 0.0
            b[jj] = 1.0
            new_c = exact_objective(b)
            delta = new_c - curr_c
            if delta < -1e-15 or (T > T_min and rng.random() < np.exp(-delta / T)):
                curr_c = new_c
                if new_c < best_c - 1e-15:
                    best_c = new_c
                    best_b = b.copy()
            else:
                b[ii] = 1.0
                b[jj] = 0.0
        T *= cool
        if it % 100000 == 99999 and T < 0.001:
            T = 0.003
    return best_b, best_c


def binary_hill_climb(b0, deadline, rng, max_rounds=500):
    n = len(b0)
    b = b0.copy()
    curr_c = exact_objective(b)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        best_improve = 1e-14
        best_move = None
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        # Sampled single swaps
        n_single = min(len(ones) * len(zeros), 4000)
        for _ in range(n_single):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            b[i] = 0.0; b[j] = 1.0
            new_c = exact_objective(b)
            b[i] = 1.0; b[j] = 0.0
            imp = curr_c - new_c
            if imp > best_improve:
                best_improve = imp
                best_move = ('single', i, j)
        # Sampled double swaps
        n_double = min(len(ones) * len(zeros), 2000)
        for _ in range(n_double):
            i1 = int(ones[rng.randint(len(ones))])
            i2 = int(ones[rng.randint(len(ones))])
            j1 = int(zeros[rng.randint(len(zeros))])
            j2 = int(zeros[rng.randint(len(zeros))])
            if i1 == i2 or j1 == j2:
                continue
            b[i1] = 0.0; b[i2] = 0.0; b[j1] = 1.0; b[j2] = 1.0
            new_c = exact_objective(b)
            b[i1] = 1.0; b[i2] = 1.0; b[j1] = 0.0; b[j2] = 0.0
            imp = curr_c - new_c
            if imp > best_improve:
                best_improve = imp
                best_move = ('double', i1, i2, j1, j2)
        if best_move is not None:
            if best_move[0] == 'single':
                b[best_move[1]] = 0.0
                b[best_move[2]] = 1.0
            else:
                b[best_move[1]] = 0.0; b[best_move[2]] = 0.0
                b[best_move[3]] = 1.0; b[best_move[4]] = 1.0
            curr_c = exact_objective(b)
        else:
            break
    return b, curr_c


def lse_objective(h, tau):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(prof))
    shifted = np.minimum((prof - m) / tau, 0.0)
    return m + tau * np.log(np.sum(np.exp(shifted)))


def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(prof))
    shifted = np.minimum((prof - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return grad


def continuous_refine(h0, deadline, rng, max_taus=8):
    n = len(h0)
    h = project_box_sum(h0.copy())
    curr_c = exact_objective(h)
    bounds = [(0.0, 1.0)] * n
    taus = [0.008, 0.003, 0.001, 0.0004, 0.00015, 0.00006, 0.00002, 0.000008]
    for idx in range(min(max_taus, len(taus))):
        if time.time() > deadline - 8:
            break
        tau = taus[idx]
        try:
            def obj(x):
                return lse_objective(x, tau)
            def obj_grad(x):
                return lse_gradient(x, tau)
            def cons_fun(x):
                return np.sum(x) - n / 2.0
            def cons_jac(x):
                return np.ones(n)
            con = {'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}
            res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=[con],
                           options={'maxiter': min(500, n * 8), 'ftol': 1e-13})
            if res.success or res.x is not None:
                h_new = project_box_sum(np.array(res.x, dtype=np.float64))
                c_new = exact_objective(h_new)
                if c_new < curr_c - 1e-14:
                    h = h_new
                    curr_c = c_new
        except Exception:
            pass
    return h, curr_c


def transfer_size(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return project_box_sum(h_src.copy())
    x_s = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_t = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_new = np.interp(x_t, x_s, h_src)
    return project_box_sum(h_new)


def make_starts(n, rng, parent_h=None):
    starts = []
    if parent_h is not None:
        starts.append(transfer_size(parent_h, n))
        starts.append(project_box_sum(transfer_size(parent_h, n) + 0.02 * rng.randn(n)))
        starts.append(project_box_sum(transfer_size(parent_h, n) + 0.05 * rng.randn(n)))
    # Structured starts
    x = (np.arange(n) + 0.5) / n
    for freq in [2, 3, 4, 5, 6]:
        for amp in [0.4, 0.5, 0.45]:
            ph = rng.uniform(0, 2 * np.pi)
            h = 0.5 + amp * np.sin(2 * np.pi * freq * x + ph)
            starts.append(project_box_sum(h))
    # Alternating blocks
    block = max(2, n // 6)
    h = np.zeros(n)
    for i in range(0, n, 2 * block):
        h[i:i + block] = 1.0
    starts.append(project_box_sum(h))
    # Random
    starts.append(project_box_sum(0.3 + 0.4 * rng.random(n)))
    return starts
