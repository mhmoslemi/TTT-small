import numpy as np
from scipy.optimize import linprog
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = 1.0, -1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-14 and n > 0:
            free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
            if len(free) > 0:
                h[free[0]] += diff2
            else:
                h[0] = np.clip(h[0] + diff2, 0.0, 1.0)
    return h

def lag_gradient(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] += (1.0 - h[lag:])
            if lag > 0:
                grad[lag:] += -h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[a:] += (1.0 - h[:s])
            grad[:s] += -h[a:]
    return grad * dx

def find_active_lags(profile, n, tol_frac=0.03):
    pm = np.max(profile)
    tol = max(1e-6, tol_frac * pm)
    active = np.where(profile >= pm - tol)[0]
    if len(active) > 25:
        idx = np.argsort(profile)[::-1][:25]
        active = idx
    return active

def lp_descent_direction(h, active_indices, dx, trust=0.5):
    n = len(h)
    A_rows = []
    for idx in active_indices:
        lag = idx - (n - 1)
        g = lag_gradient(h, lag, dx)
        A_rows.append(g)
    A = np.array(A_rows) if A_rows else np.array([[1.0]] * n)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(A_rows) + 1, n + 1))
    b_ub = np.zeros(len(A_rows) + 1)
    for i in range(len(A_rows)):
        A_ub[i, :n] = A[i]
        A_ub[i, n] = -1.0
        b_ub[i] = 0.0
    A_ub[len(A_rows), :n] = np.ones(n)
    b_ub[len(A_rows)] = 0.0
    bounds_lp = []
    for i in range(n):
        if h[i] < 1e-8:
            bounds_lp.append((0.0, trust))
        elif h[i] > 1.0 - 1e-8:
            bounds_lp.append((-trust, 0.0))
        else:
            bounds_lp.append((-trust, trust))
    bounds_lp.append((None, None))
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
    if res is not None and res.status == 0 and res.x is not None:
        d = res.x[:n]
        t = res.x[n]
        if t < -1e-10:
            d = d - d.mean()
            return d, float(t)
    return None, 0.0

def line_search(h, d, c_current, dx, max_step=1.0):
    n = len(h)
    if np.max(np.abs(d)) < 1e-14:
        return h, c_current
    scale = max_step / np.max(np.abs(d))
    d = d * scale
    alphas = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02]
    best_h = h.copy()
    best_c = c_current
    for a in alphas:
        h_try = h + a * d
        h_try = project_box_sum(h_try, 0.5 * n)
        c_try, _ = exact_objective(h_try)
        if c_try < best_c - 1e-13:
            best_c = c_try
            best_h = h_try.copy()
    return best_h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=100, rng=None, deadline=None):
    n = len(b)
    if n % 2 != 0:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(42)
    c5, _ = exact_objective(b)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline - 5:
            break
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 2000:
            for i in ones:
                for j in zeros:
                    b[i] = 0.0
                    b[j] = 1.0
                    nc, _ = exact_objective(b)
                    if nc < c5 - 1e-13:
                        c5 = nc
                        improved = True
                        ones = list(np.where(b > 0.5)[0])
                        zeros = list(np.where(b < 0.5)[0])
                        break
                    else:
                        b[i] = 1.0
                        b[j] = 0.0
                if improved:
                    break
        else:
            n_tries = min(2000, n_pairs)
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]

                b[i], b[j] = 0.0, 1.0
                nc, _ = exact_objective(b)
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = list(np.where(b > 0.5)[0])
                    zeros = list(np.where(b < 0.5)[0])
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5

def lp_descent_loop(h, deadline, max_iters=5000, tol_frac=0.03):
    n = len(h)
    dx = 2.0 / n
    c_cur, profile = exact_objective(h)
    trust = 0.4
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        active = find_active_lags(profile, n, tol_frac)
        d, t_val = lp_descent_direction(h, active, dx, trust)
        if d is None:
            stall += 1
            trust *= 0.8
            if stall > 15 or trust < 0.01:
                break
            continue
        h_new, c_new = line_search(h, d, c_cur, dx, max_step=trust)
        if c_new < c_cur - 1e-13:
            h = h_new
            c_cur = c_new
            profile = np.correlate(h, 1.0 - h, mode='full') * dx
            trust = min(trust * 1.1, 0.8)
            stall = 0
        else:
            stall += 1
            trust *= 0.75
            if stall > 20 or trust < 0.005:
                break
    return h, c_cur

def make_sturmian(n, alpha, phi, rng=None):
    if n % 2 != 0:
        n -= 1
    target = n // 2
    x = np.arange(n)
    s = np.abs(np.sin(2 * np.pi * (alpha * x / n + phi)))
    idx = np.argsort(s)[:target]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def make_harmonic(n, freq, phase, amp):
    if n % 2 != 0:
        n -= 1
    x = np.arange(n, dtype=np.float64) / n
    h = 0.5 + amp * np.cos(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, 0.5 * n)

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def sa_refine(h, deadline, rng, n_steps=3000, T_init=0.003, cooling=0.995):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    c_cur = exact_objective(cur)[0]
    best = cur.copy()
    c_best = c_cur
    T = T_init
    for _ in range(n_steps):
        if time.time() > deadline - 3:
            break
        a = rng.randint(n)
        b = rng.randint(n)
        while b == a:
            b = rng.randint(n)
        delta = rng.choice([0.01, 0.02, 0.04, 0.08, 0.15])
        if (cur[a] - delta < 0) or (cur[b] + delta > 1):
            continue
        proposal = cur.copy()
        proposal[a] -= delta
        proposal[b] += delta
        c_new, _ = exact_objective(proposal)

        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-10)):
            cur = proposal
            c_cur = c_new
            if c_new < c_best - 1e-13:
                c_best = c_new
                best = cur.copy()
        T *= cooling
    return best, c_best

                        best_h = b.copy()
                    if time.time() < deadline - 10:
                        h_lp6, c_lp6 = lp_descent_loop(b.copy(), deadline, max_iters=1000, tol_frac=0.02)
                        if c_lp6 < best_c - 1e-12:
                            best_c = c_lp6
                            best_h = h_lp6

    # Phase 7: One more SA burst if time remains
    if time.time() < deadline - 8:
        h_sa2, c_sa2 = sa_refine(best_h.copy(), deadline, rng, n_steps=1500, T_init=0.001)
        if c_sa2 < best_c - 1e-12:
            best_c = c_sa2
            best_h = h_sa2
        if time.time() < deadline - 5:
            h_lp7, c_lp7 = lp_descent_loop(h_sa2.copy(), deadline, max_iters=500, tol_frac=0.02)
            if c_lp7 < best_c - 1e-12:
                best_c = c_lp7
                best_h = h_lp7

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)

    return (best_h, best_c, best_n)
