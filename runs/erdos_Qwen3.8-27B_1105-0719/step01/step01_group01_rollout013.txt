    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for step in range(n_steps):
        # Compute correlation
        one_minus_h = 1.0 - best_h
        corr = np.correlate(best_h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find active lags (within some tolerance of max)
        max_corr = np.max(corr)
        active = corr > max_corr - 1e-6  # or some tolerance
        
        # Compute subgradient: average gradient of active lags
        grad = np.zeros(n)
        for idx in np.where(active)[0]:
            k = idx - (n - 1)  # lag
            # c_k = sum_i h[i+k]*(1-h[i]) for k>=0, or h[i]*(1-h[i+k]) for k<0
            # d(c_k)/d(h[j]) = ...

            #   d/d(h[j]) = (if j = i+k for some i) (1-h[i]) + (if j = i for some i) -h[i+k]

                term2 = -best_h[j+k] if 0 <= j+k < n else 0.0

        new_h = best_h - lr * grad
        new_h = project_feasible(new_h, n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(y, n):
    """Project y onto {h : 0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            h[interior] += residual / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_corr(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(np.clip(shifted, -100, 0))
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

For a transfer of eps from j to i (h[i] += eps, h[j] -= eps):

where g(p,k) = dc_k/dh[p] = (1-h[p-k] if valid) - (h[p+k] if valid)

We want to minimize max_k (c_k + eps * [g(i,k) - g(j,k)])

This is a 1D minimization problem in eps for given i, j. And we want to find the best (i, j) pair.

For efficiency, we can:
1. Precompute g(p,k) for all p, k (this is an n × (2n-1) matrix)
2. For a given (i,j), the change in each lag is eps * (g(i,k) - g(j,k))
3. The new max is max_k (c_k + eps * (g(i,k) - g(j,k)))
4. We want the smallest eps >= 0 (or <= 0) that reduces the max

This is a piecewise linear function of eps, and we can find the optimal eps by solving a small LP or by bisection.

But precomputing the full g matrix is O(n²) and the search over (i,j) pairs is O(n²) × (cost of finding optimal eps). For n=60, n² = 3600 pairs, and each pair requires O(n) work to find the optimal eps. Total: O(n³) = 216,000 per "round." That's very fast.

    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        
        # Find all lags near the max (within 1% of max)
        threshold = max_corr - 0.01 * max_corr
        active_idx = np.where(corr >= threshold)[0]
        
        # Compute gradient for the active lags (averaged)
        grad = np.zeros(n)
        for idx in active_idx:
            k = idx - (n - 1)
            for p in range(n):
                g = 0.0
                if 0 <= p-k < n:
                    g += 1.0 - h[p-k]
                if 0 <= p+k < n:
                    g -= h[p+k]
                grad[p] += g
        grad /= len(active_idx)
        
        # Find best source (positive grad, h > 0) and best target (negative grad, h < 1)
        # Source: we want to decrease h[j], so we want grad[j] > 0 (decreasing h[j] reduces the objective)
        # Target: we want to increase h[i], so we want grad[i] < 0
        sources = np.where((grad > 0.01) & (h > 0.01))[0]
        targets = np.where((grad < -0.01) & (h < 0.99))[0]
        
        if len(sources) == 0 or len(targets) == 0:
            break
        
        # Pick the best source and target
        j = sources[np.argmax(grad[sources])]
        i = targets[np.argmin(grad[targets])]
        
        # Determine step size
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps < 1e-10:
            break

        best_c5_try = c5
        best_eps = 0
        for frac in [0.1, 0.3, 0.5, 0.8]:
            eps = frac * max_eps
            h_try = h.copy()
            h_try[i] += eps
            h_try[j] -= eps
            c5_try = compute_c5(h_try)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_eps = eps

            h[i] += best_eps
            h[j] -= best_eps
            best_c5 = best_c5_try
        else:
            # Try smaller steps or different pairs
            # If no improvement, break to avoid infinite loop
            break

For a random binary string, #(0,1 pairs at dist k) ≈ (n-k)/4 and #(1,0 pairs at dist k) ≈ (n-k)/4.
Max is at k=1: ≈ (n-1)/4. C₅ ≈ (n-1)/4 * 2/n ≈ 1/2.

But we can do better! The key is to arrange the ones and zeros so that transitions are "spread out" across all lags.

For a perfectly periodic pattern like 1111000011110000..., the transitions only occur at specific lags.

Actually, I think the optimal binary arrangement is related to "balanced sequences" or "Barker sequences" or "Golay sequences" - sequences where the autocorrelation is minimized.

Wait, this is actually related to the problem of finding a binary sequence with minimum maximum aperiodic autocorrelation! The aperiodic autocorrelation at lag k is:
R(k) = sum_i h[i]*h[i+k] - sum_i h[i]*(1-h[i+k]) - sum_i (1-h[i])*h[i+k]

Hmm, that's not quite the same. Let me think differently.

For binary h, let's define:
R(k) = sum_i (2h[i]-1)*(2h[i+k]-1) for i where both are valid.

This is the standard (aperiodic) autocorrelation of the ±1 sequence.

Our c_k = sum_i h[i+k]*(1-h[i]) = sum_i h[i+k] - sum_i h[i+k]*h[i]

Let s = 2h-1 (so s ∈ {-1, 1}). Then h = (s+1)/2.
c_k = sum_i (s[i+k]+1)/2 * (1-s[i])/2 = (1/4) sum_i (s[i+k]+1)(1-s[i])
= (1/4) sum_i (s[i+k] - s[i+k]*s[i] + 1 - s[i])
= (1/4) [(N-k) + sum_i s[i+k] - sum_i s[i] - sum_i s[i+k]*s[i]]
= (1/4) [(N-k) + sum_i s[i+k] - sum_i s[i] - R(k)]

For a "balanced" sequence (sum s = 0), this simplifies to:
c_k ≈ (1/4) [(N-k) - R(k)] (approximately, ignoring boundary effects)

So minimizing max_k c_k is equivalent to maximizing min_k R(k) (or at least making R(k) large for all k).

This is the opposite of what we want in compressed sensing/spectroscopy (where we want small autocorrelation)! We want LARGE autocorrelation at all lags, which means the sequence should be "smooth" - adjacent elements should be similar.

For a binary sequence, large R(k) means that h[i] and h[i+k] tend to be the same for all k. This is achieved by long runs of 1s followed by long runs of 0s (a "block" structure).

For example, if h = [1,1,...,1, 0,0,...,0] (n/2 ones followed by n/2 zeros):
- For small k: most pairs have h[i]=h[i+k], so R(k) is large, c_k is small
- For k = n/2: h[i] and h[i+n/2] are always different, so c_k = n/2

So the max is at k = n/2: c_{n/2} = n/2. C₅ = (n/2)*(2/n) = 1. That's terrible!

So a single block is bad. We need multiple blocks.

What about alternating blocks? Like 1100110011... (blocks of length 2):
For k=1: within a block, h[i]=h[i+1], so no contribution. At block boundaries, there are transitions. With n/4 boundaries (n/8 of type 1→0 and n/8 of type 0→1), c_1 ≈ n/8 * ... this is getting complicated.

Let me just focus on the computational approach rather than trying to find the optimal structure analytically.

Let me write the code now. I'll aim for a clean, efficient implementation:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            h[interior] += residual / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(np.clip(shifted, -500, 0))
    s = np.sum(exps)
    if s < 1e-300:
        s = 1e-300
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, _ = smooth_max_and_weights(corr, beta)
    return val * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    _, weights = smooth_max_and_weights(corr, beta)
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-14:
            continue
        k = i - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n-kp]
    return grad * dx

def slsqp_run(h0, betas, max_iter=200):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    
    def eq_fun(h):
        return np.sum(h) - n / 2.0
    
    def eq_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    for beta in betas:
        try:
            res = minimize(
                smooth_obj, best_h, args=(beta,),
                jac=smooth_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-15}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_try = np.clip(res.x, 0, 1)
                # Fix sum
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
        except Exception:
            pass
    return best_h, best_c5

def greedy_local_search(h, max_rounds=50000):
    """Greedily reduce the worst lag using targeted transfers."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        
        if c5 <= best_c5 + 1e-12:
            # No progress, check if really stuck
            pass
        
        # Find active lags (within 2% of max)
        threshold = max_corr * 0.98
        active_idx = np.where(corr >= threshold)[0]
        
        # Compute averaged gradient over active lags
        grad = np.zeros(n)
        for idx in active_idx:
            k = idx - (n - 1)
            if k == 0:
                grad += (1.0 - 2.0 * h)
            elif k > 0:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
            else:
                kp = -k
                grad[:n-kp] += (1.0 - h[kp:])
                grad[kp:] -= h[:n-kp]
        grad /= len(active_idx)
        
        # Find best transfer: increase h where grad is most negative, decrease where most positive
        # Only consider interior points
        can_incr = (h < 1.0 - 1e-6)
        can_decr = (h > 1e-6)
        
        targets = np.where(can_incr & (grad < 0))[0]
        sources = np.where(can_decr & (grad > 0))[0]
        
        if len(targets) == 0 or len(sources) == 0:
            break
        
        i = targets[np.argmin(grad[targets])]
        j = sources[np.argmax(grad[sources])]
        
        # Try several step sizes
        max_eps = min(1.0 - h[i], h[j])
        if max_eps < 1e-8:
            break
        
        improved = False
        for frac in [0.5, 0.25, 0.1, 0.05]:
            eps = frac * max_eps
            h_try = h.copy()
            h_try[i] += eps
            h_try[j] -= eps
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        
        if not improved:
            break
    
    return h, best_c5

def random_transfer_search(h, n_iters, rng):
    """Randomized local search with transfer moves."""
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(n_iters):
        i, j = rng.choice(n, size=2, replace=False)
        h_i, h_j = h[i], h[j]
        if abs(h_i - h_j) < 1e-8:
            continue
        
        # Try a few epsilon values and directions
        max_eps_f = min(h_i, 1.0 - h_j)  # transfer from i to j
        max_eps_b = min(h_j, 1.0 - h_i)  # transfer from j to i
        
        for eps in [0.5 * max_eps_f, 0.1 * max_eps_f, 0.5 * max_eps_b, 0.1 * max_eps_b]:
            if eps < 1e-8:
                continue
            # Try both directions
            h_try = h.copy()
            h_try[i] -= eps
            h_try[j] += eps
            if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    h = h_try
                    best_c5 = c5_try
                    break
    
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(60, 0.5)
    
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Phase 1: SLSQP optimization
    if time_left() > 60:
        h_opt, c5_opt = slsqq_run(best_h, [10, 50, 200, 1000, 5000, 20000], max_iter=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 2: Greedy local search
    if time_left() > 20:
        h_opt, c5_opt = greedy_local_search(best_h, max_rounds=200000)
