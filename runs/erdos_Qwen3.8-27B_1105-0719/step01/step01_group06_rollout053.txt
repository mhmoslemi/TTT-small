def solve_epigraph_slqp(h_start, t_start, maxiter, n_constraints, beta):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    prof = get_profile(h_start)
    if nc < M:
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def _obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))

    def _obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof_local = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof_local[active_lags]

    def _cons_jac(x):

        G = build_gradient_matrix(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]

            z = np.random.randn(n)
            z -= z.mean()
            starts.append(repair_feasible(best_h + eps * z))

        for start in starts:
            if time_left() < 25:
                break
            mi = max(8, min(30, int(time_left() * 0.08)))
            h_new, s_new = lp_trust_region_loop(start, mi, t_global, budget_s)
            if s_new < best_score - 1e-12:
                best_h = h_new
                best_score = s_new

    # Stage 6: Gradient descent polish on best
    if time_left() > 15:
        h_gd2, s_gd2 = weighted_gradient_descent(best_h, 60, t_global, budget_s)
        if s_gd2 < best_score - 1e-12:
            best_h = h_gd2
            best_score = s_gd2

    # Stage 7: SLSQP epigraph with top-K constraints (backup, different from previous)
    if time_left() > 20:
        n = len(best_h)
        for nc in [20, 40, 2 * n - 1]:
            if time_left() < 12:
                break
            if nc >= 2 * n - 1:
                nc = 2 * n - 1
            mi = min(50, max(10, int(time_left() * 0.3)))
            h_sol = solve_epigraph_slqp(best_h, best_score + 1e-6, mi, nc, 0.0)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s

    # Stage 8: Final LP + gradient polish
    if time_left() > 12:
        h_final_lp, s_final_lp = lp_trust_region_loop(best_h, 20, t_global, budget_s)
        if s_final_lp < best_score - 1e-12:
            best_h = h_final_lp
            best_score = s_final_lp
    if time_left() > 8:
        h_fin_gd, s_fin_gd = weighted_gradient_descent(best_h, 40, t_global, budget_s)
        if s_fin_gd < best_score - 1e-12:
            best_h = h_fin_gd
            best_score = s_fin_gd

    # Final repair and evaluation
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
