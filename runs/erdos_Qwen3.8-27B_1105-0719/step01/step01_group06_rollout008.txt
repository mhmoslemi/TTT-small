            best_local_t = t
    if best_local_t != 0.0:
        step = (t_hi - t_lo) / n_coarse
        lo = max(t_lo, best_local_t - step * 1.5)
        hi = min(t_hi, best_local_t + step * 1.5)
        if hi > lo + 1e-14:
            ts_fine = np.linspace(lo, hi, n_fine)
            for t in ts_fine:
                h_t = h + t * d
                if h_t.min() < -1e-9 or h_t.max() > 1.0 + 1e-9:
                    continue
                s = exact_score(h_t)
                if s < best_local_s:
                    best_local_s = s
                    best_local_t = t
    if best_local_s < current_score - 1e-14:
        h_best = repair_feasible(h + best_local_t * d)
        return h_best, exact_score(h_best)
    return h, current_score

def compute_lag_gradients(h, k):
    n = len(h)
    dx = 2.0 / n
    d = np.zeros(n)
    delta = k - (n - 1)
    if delta == 0:
        d = dx * (1.0 - 2.0 * h)
    elif 0 < delta < n:
        d[delta:] += dx * (1.0 - h[:n - delta])
        d[:n - delta] -= dx * h[delta:]
    else:
        sep = -delta
        if sep < n:
            d[:n - sep] += dx * (1.0 - h[sep:])
            d[sep:] -= dx * h[:n - sep]
    return d

def gen_gradient_dir(h, profile, n_active=15):
    n = len(h)
    M = len(profile)
    pmax = profile.max()
    tol = 0.02 * pmax
    active = np.where(profile > pmax - tol)[0]
    if len(active) == 0:
        active = np.argsort(profile)[::-1][:n_active]
    d = np.zeros(n)
    w = profile[active]
    w = np.maximum(w - (pmax - tol), 0)
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.ones(len(active))
    else:
        w /= wsum
    for i, k in enumerate(active):
        gk = compute_lag_gradients(h, k)
        d += w[i] * gk
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm > 1e-14:
        d /= norm
    return d

def gen_anti_dir(h, profile, k):
    """Direction that specifically tries to reduce lag k."""
    n = len(h)
    delta = k - (n - 1)
    d = np.zeros(n)
    if delta > 0 and delta < n:
        # P_k = dx * sum h[u+delta]*(1-h[u])
        # To reduce: lower h[u+delta] where h[u] is high, or lower h[u] where h[u+delta] is high
        for u in range(n - delta):
            both_high = h[u] * h[u + delta]
            d[u + delta] -= both_high
            d[u] -= both_high
    elif delta < 0:
        sep = -delta
        if sep < n:
            for u in range(n - sep):
                both_high = h[u] * h[u + sep]
                d[u] -= both_high
                d[u + sep] -= both_high
    else:
        d = -(h * (1 - h))
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm > 1e-14:
        d /= norm
    return d

def gen_pairwise_dir(h, rng, n=140):
    i = rng.integers(n)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def gen_block_dir(h, rng, n=140):
    size = rng.integers(max(2, n // 20), max(3, n // 4))
    i0 = rng.integers(0, max(1, n - size))
    j0 = rng.integers(0, max(1, n - size))
    if abs(i0 - j0) < size:
        j0 = (i0 + size + size // 2) % max(1, n - size)
    d = np.zeros(n)
    d[i0:i0 + size] = -1.0 / size
    d[j0:j0 + size] = 1.0 / size
    d -= d.mean()
    return d

def gen_lowfreq_dir(h, rng, n=140):
    x = np.arange(n)
    freq = rng.integers(1, max(2, n // 8))
    phase = rng.uniform(0, 2 * np.pi)
    d = np.sin(2 * np.pi * freq * x / n + phase)
    d -= d.mean()
    return d

def gen_random_dir(h, rng, n=140):
    d = rng.standard_normal(n)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm > 1e-14:
        d /= norm
    return d

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_global_start = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t_global_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(64, 0.5)

    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_score = exact_score(parent)
    n = len(best_h)

    # Phase 1: Intensive line search with diverse directions (70% of time)
    phase1_time = budget_s * 0.65
    phase2_time = budget_s * 0.20
    phase3_time = budget_s * 0.15

    t0 = time.perf_counter()
    stall_count = 0
    max_stall = 500

    while time.perf_counter() - t0 < phase1_time and tleft() > 10:
        profile = get_profile(best_h)
        pmax = profile.max()

        # Choose direction type based on stall count
        r = rng.random()
        if stall_count < max_stall:
            if r < 0.35:
                d = gen_gradient_dir(best_h, profile, n_active=10 + rng.integers(0, 10))
            elif r < 0.55:
                active_k = np.argmax(profile)
                d = gen_anti_dir(best_h, profile, active_k)
            elif r < 0.70:
                top_k = np.argsort(profile)[::-1][:5]
                k_sel = top_k[rng.integers(5)]
                d = gen_anti_dir(best_h, profile, k_sel)
            elif r < 0.82:
                d = gen_block_dir(best_h, rng, n)
            elif r < 0.90:
                d = gen_pairwise_dir(best_h, rng, n)
            elif r < 0.95:
                d = gen_lowfreq_dir(best_h, rng, n)
            else:
                d = gen_random_dir(best_h, rng, n)
        else:
            # Stuck: use more aggressive exploration
            if r < 0.3:
                d = gen_random_dir(best_h, rng, n)
            elif r < 0.5:
                d = gen_block_dir(best_h, rng, n)
            elif r < 0.7:
                d = gen_lowfreq_dir(best_h, rng, n)
            elif r < 0.85:
                d = gen_pairwise_dir(best_h, rng, n)
            else:
                d = gen_anti_dir(best_h, profile, rng.integers(len(profile)))

        if np.linalg.norm(d) < 1e-14:
            continue

        h_new, s_new = exact_line_search(best_h, d, n_coarse=50, n_fine=30)
        if s_new < best_score - 1e-13:
            best_h = h_new
            best_score = s_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 200 == 0:
                # Perturb and retry
                eps = 0.005 + 0.003 * rng.random()
                z = rng.standard_normal(n)
                z -= z.mean()
                h_pert = repair_feasible(best_h + eps * z)
                s_pert = exact_score(h_pert)
                if s_pert < best_score - 1e-13:
                    best_h = h_pert
                    best_score = s_pert
                    stall_count = 0

    # Phase 2: Try different dimensions
    if tleft() > 20:
        base_n = len(best_h)
        dims_to_try = []
        for dn in [-10, -5, 5, 10, 20, 40]:
            dn_new = base_n + dn
            if 32 <= dn_new <= 300 and dn_new != base_n:
                dims_to_try.append(dn_new)
        dims_to_try = sorted(set(dims_to_try))

        for new_n in dims_to_try:
            if tleft() < 15:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            if s_interp > best_score + 0.005:
                continue
            nn = new_n
            for _ in range(200):
                if tleft() < 10:
                    break
                profile = get_profile(h_interp)
                r = rng.random()
                if r < 0.4:
                    d = gen_gradient_dir(h_interp, profile, n_active=12)
                elif r < 0.6:
                    d = gen_anti_dir(h_interp, profile, np.argmax(profile))
                elif r < 0.75:
                    d = gen_block_dir(h_interp, rng, nn)
                elif r < 0.88:
                    d = gen_lowfreq_dir(h_interp, rng, nn)
                else:
                    d = gen_random_dir(h_interp, rng, nn)
                if np.linalg.norm(d) < 1e-14:
                    continue
                h_new, s_new = exact_line_search(h_interp, d, n_coarse=40, n_fine=20)
                if s_new < s_interp - 1e-13:
                    h_interp = h_new
                    s_interp = s_new
            if s_interp < best_score - 1e-12:
                best_h = h_interp
                best_score = s_interp

    # Phase 3: Final fine refinement with small steps
    if tleft() > 10:
        n_fin = len(best_h)
        t3_start = time.perf_counter()
        t3_budget = min(phase3_time, tleft())
        for _ in range(2000):
            if time.perf_counter() - t3_start > t3_budget:
                break
            profile = get_profile(best_h)
            pmax = profile.max()
            tol = 0.01 * pmax
            active = np.where(profile > pmax - tol)[0]
            if len(active) == 0:
                active = np.argsort(profile)[::-1][:8]

            # Average gradient of active lags
            d = np.zeros(n_fin)
            for k in active:
                d += compute_lag_gradients(best_h, k)
            d -= d.mean()
            norm = np.linalg.norm(d)
            if norm < 1e-14:
                continue
            d /= norm

            # Try small and medium steps
            improved = False
            for alpha_scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
                # Scale d so max |d_i| is reasonable
                d_scaled = d * alpha_scale
                h_new, s_new = exact_line_search(best_h, d_scaled, n_coarse=30, n_fine=20)
                if s_new < best_score - 1e-13:
                    best_h = h_new
                    best_score = s_new
                    improved = True
                    break

            if not improved:
                # Try a random perturbation of the gradient
                z = rng.standard_normal(n_fin) * 0.3
                z -= z.mean()
                d_mix = 0.7 * d + 0.3 * z
                d_mix -= d_mix.mean()
                if np.linalg.norm(d_mix) < 1e-14:
                    continue
                d_mix /= np.linalg.norm(d_mix)
                h_new, s_new = exact_line_search(best_h, d_mix, n_coarse=30, n_fine=20)
                if s_new < best_score - 1e-13:
                    best_h = h_new
                    best_score = s_new

    # Final: try a short SLSQP polish
    if tleft() > 30:
        n_sl = len(best_h)
        M = 2 * n_sl - 1
        profile = get_profile(best_h)
        pmax = profile.max()
        nc = min(M, 40)
        active_lags = np.argsort(profile)[::-1][:nc]
        dx = 2.0 / n_sl

        def obj_sl(x):
            return x[n_sl]

        def cons_fun_sl(x):
            h = x[:n_sl]
            t = x[n_sl]
            eq = h.sum() - n_sl / 2.0
            prof = np.correlate(h, 1.0 - h, mode="full") * dx
            ineq = t - prof[active_lags]
            return np.concatenate([[eq], ineq])

        bounds = [(0.0, 1.0)] * n_sl + [(0.0, pmax + 0.5)]
        x0 = np.concatenate([best_h, [pmax + 0.001]])

        try:
            result = minimize(
                obj_sl, x0,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'ineq', 'fun': cons_fun_sl}],
                options={'maxiter': min(100, int(tleft() * 2)), 'ftol': 1e-12}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                h_sol = repair_feasible(result.x[:n_sl])
                s_sol = exact_score(h_sol)
                if s_sol < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s_sol
        except Exception:
            pass

    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
