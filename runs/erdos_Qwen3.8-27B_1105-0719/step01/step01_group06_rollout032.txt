            # Check if we improved
            s_trial = exact_score(h_trial)
            if s_trial < best_exact - 1e-13:

def solve_epigraph(h_start, t_start, maxiter=80, n_constraints=0, beta=0.0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)
    
    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        n = len(h)
        M = 2 * n - 1
        dx = 2.0 / n
        G = np.zeros((len(active_lags), n))
        for idx, k in enumerate(active_lags):
            delta = k - (n - 1)
            d = abs(delta)
            if d >= n:
                continue
            if delta >= 0:
                G[idx, d:n] += dx * (1.0 - h[0:n-d])
                G[idx, 0:n-d] += -dx * h[d:n]
            else:
                G[idx, 0:n-d] += dx * (1.0 - h[d:n])
                G[idx, d:n] += -dx * h[0:n-d]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -G
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(
            obj, x0, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = project_box_sum(result.x[:n], n / 2.0)
            return h_sol
    except Exception:
        pass
    return None

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return project_box_sum(h_sym, len(h) / 2.0)

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    return project_box_sum(h_new, new_len / 2.0)

def local_gradient_descent(h_start, max_iters=100, time_limit=30):
    """Weighted gradient descent on active lags."""
    t0 = time.perf_counter()
    h = h_start.copy()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()

        if time.perf_counter() - t0 > time_limit:
            break

            best_h = h.copy()

        pmax = prof[active].max()

            d = abs(delta)
            if d >= n:

            if delta >= 0:
                grad[d:n] += w * (1.0 - h[0:n-d])
                grad[0:n-d] += w * (-h[d:n])
            else:
                grad[0:n-d] += w * (1.0 - h[d:n])
                grad[d:n] += w * (-h[0:n-d])
        
        # Gradient step with projection
        h_trial = project_box_sum(h - step * grad, target_sum)
        s_trial = exact_score(h_trial)

            best_score = s_trial
            best_h = h_trial.copy()
            h = h_trial
        else:
            step *= 0.7
            if step < 0.005:
                break
    return best_h, best_score

        dx = 2.0 / n
        G = np.zeros((len(active_lags), n))
        for idx, k in enumerate(active_lags):
            delta = k - (n - 1)
            d = abs(delta)
            if d >= n:
                continue
            if delta >= 0:
                G[idx, d:n] += dx * (1.0 - h[0:n-d])
                G[idx, 0:n-d] += -dx * h[d:n]
            else:
                G[idx, 0:n-d] += dx * (1.0 - h[d:n])
                G[idx, d:n] += -dx * h[0:n-d]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -G
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, max(t_start + 1.0, 0.01))]
    x0 = np.concatenate([h_start, [max(t_start, 0.001)]])
    try:
        result = minimize(
            obj, x0, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return project_box_sum(result.x[:n], n / 2.0)
    except Exception:
        pass
    return None

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return project_box_sum(h_sym, len(h) / 2.0)

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    return project_box_sum(h_new, new_len / 2.0)

def local_gradient_descent(h_start, max_iters=150, time_limit=30.0):
    t0 = time.perf_counter()
    h = h_start.copy()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = exact_score(h)
    step = 1.0
    stale = 0
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        prof = get_profile(h)
        score = float(np.max(prof))
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 12:
                step *= 0.6
                stale = 0
                if step < 0.003:
                    break
        A = max(8, min(35, n // 3))
        active = np.argsort(prof)[::-1][:A]
        pmax = prof[active].max()
        weights = np.exp((prof[active] - pmax) * 8.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights /= wsum
        grad = np.zeros(n)
        dx = 2.0 / n
        for idx, k in enumerate(active):
            if weights[idx] < 1e-8:
                continue
            delta = k - (n - 1)
            d = abs(delta)
            if d >= n:
                continue
            w = weights[idx] * dx
            if delta >= 0:
                grad[d:n] += w * (1.0 - h[0:n-d])
                grad[0:n-d] += w * (-h[d:n])
            else:
                grad[0:n-d] += w * (1.0 - h[d:n])
                grad[d:n] += w * (-h[0:n-d])
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_trial = project_box_sum(h - step * grad / gnorm, target_sum)
        s_trial = exact_score(h_trial)
        if s_trial < best_score - 1e-13:
            h = h_trial
        else:
            h = h  # stay
    return best_h, best_score

def try_binary_push(h, strength=0.1, n_tries=5):
    n = len(h)
    target_sum = n / 2.0
    best = h.copy()
    best_s = exact_score(h)
    for _ in range(n_tries):
        h_push = np.where(h > 0.5, np.minimum(1.0, h + strength), np.maximum(0.0, h - strength))
        h_push = project_box_sum(h_push, target_sum)
        s = exact_score(h_push)
        if s < best_s - 1e-13:
            best = h_push
            best_s = s
            h = h_push
        else:
            break
    return best, best_s

def make_structured_start(n, seed):
    rng = np.random.RandomState(seed)
    pattern = rng.randint(0, 4)
    if pattern == 0:
        h = np.full(n, 0.5)
        h[:n//2] = 0.9
        h[n//2:] = 0.1
    elif pattern == 1:
        n_blocks = max(3, n // 20)
        block_len = n // n_blocks
        h = np.zeros(n)
        for b in range(n_blocks):
            if rng.rand() > 0.5:
                h[b*block_len:(b+1)*block_len] = 1.0
    elif pattern == 2:
        t = np.linspace(0, 4*np.pi, n)
        h = 0.5 + 0.3 * np.sin(t) + 0.1 * np.sin(2*t)
    else:
        h = rng.rand(n) * 0.4 + 0.3
    return project_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_global_start = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t_global_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(64, 0.5)

    n = len(parent)
    target_sum = n / 2.0
    parent = project_box_sum(parent, target_sum)
    best_h = parent.copy()
    best_score = exact_score(parent)

    h_sym = symmetrize(best_h)
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-14:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: Quick SLSQP from parent
    if tleft() > 30:
        mi = min(100, max(20, int(tleft() * 0.15)))
        h_sol = solve_epigraph(best_h, best_score + 5e-6, maxiter=mi, n_constraints=min(60, 2*n-1))
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-14:
                best_h = h_sol
                best_score = s

    # Phase 2: LSE projected gradient descent
    if tleft() > 100:
        tl = min(200, tleft() * 0.3)
        betas = [3.0, 5.0, 8.0, 12.0, 20.0, 35.0, 60.0]
        h_lse, s_lse = lse_descent(best_h, betas, steps_per_beta=120, time_limit=tl)
        if s_lse < best_score - 1e-14:
            best_h = h_lse
            best_score = s_lse

    # Phase 3: SLSQP full polish
    if tleft() > 50:
        mi = min(200, max(40, int(tleft() * 0.25)))
        h_sol = solve_epigraph(best_h, best_score + 5e-7, maxiter=mi, n_constraints=0)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-14:
                best_h = h_sol
                best_score = s

    # Phase 4: Local gradient descent
    if tleft() > 30:
        tl = min(50, tleft() * 0.3)
        h_loc, s_loc = local_gradient_descent(best_h, max_iters=200, time_limit=tl)
        if s_loc < best_score - 1e-14:
            best_h = h_loc
            best_score = s_loc

    # Phase 5: SLSQP top-K (various K)
    if tleft() > 40:
        for nc in [30, 50, 100, 2*n-1]:
            if tleft() < 20:
                break
            if nc >= 2 * n - 1:
                nc = 2 * n - 1
            mi = min(120, max(25, int(tleft() * 0.2)))
            h_sol = solve_epigraph(best_h, best_score + 5e-7, maxiter=mi, n_constraints=nc)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-14:
                    best_h = h_sol
                    best_score = s

    # Phase 6: Binary push exploration
    if tleft() > 30:
        for strength in [0.05, 0.1, 0.15, 0.2, 0.3]:
            if tleft() < 15:
                break
            h_bp, s_bp = try_binary_push(best_h, strength=strength, n_tries=8)
            if s_bp < best_score - 1e-14:
                best_h = h_bp
                best_score = s_bp
                if tleft() > 20:
                    h_sol = solve_epigraph(best_h, best_score + 5e-7, maxiter=50, n_constraints=40)
                    if h_sol is not None:
                        s = exact_score(h_sol)
                        if s < best_score - 1e-14:
                            best_h = h_sol
                            best_score = s

    # Phase 7: Multi-start perturbations
    if tleft() > 60:
        for p in range(12):
            if tleft() < 25:
                break
            eps = 0.005 + 0.008 * p
            z = np.random.randn(n)
            z -= z.mean()
            h_pert = project_box_sum(best_h + eps * z, target_sum)
            s_pert = exact_score(h_pert)
            mi = min(60, max(15, int(tleft() * 0.12)))
            nc = min(40, 2*n-1)
            h_sol = solve_epigraph(h_pert, s_pert + 5e-6, maxiter=mi, n_constraints=nc)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-14:
                    best_h = h_sol
                    best_score = s
            if s_pert < best_score - 1e-14:
                best_h = h_pert
                best_score = s_pert

    # Phase 8: Structured starts at same dimension
    if tleft() > 80:
        for sid in range(6):
            if tleft() < 30:
                break
            h_struct = make_structured_start(n, seed + 100 + sid)
            s_struct = exact_score(h_struct)
            if tleft() > 20:
                h_lse, s_lse = lse_descent(h_struct, [5.0, 15.0, 30.0], steps_per_beta=60, time_limit=15)
                if s_lse < s_struct:
                    s_struct = s_lse
                    h_struct = h_lse
            mi = min(60, max(15, int(tleft() * 0.15)))
            h_sol = solve_epigraph(h_struct, s_struct + 5e-6, maxiter=mi, n_constraints=40)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-14:
                    best_h = h_sol
                    best_score = s

    # Phase 9: Dimension search
    if tleft() > 70:
        dims = []
        for d in range(40, 320, 20):
            if d != n:
                dims.append(d)
        np.random.shuffle(dims)
        for new_n in dims:
            if tleft() < 30:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            if tleft() > 20:
                h_lse, s_lse = lse_descent(h_interp, [5.0, 20.0], steps_per_beta=40, time_limit=12)
                if s_lse < s_interp:
                    s_interp = s_lse
                    h_interp = h_lse
            mi = min(50, max(10, int(tleft() * 0.15)))
            nc = min(35, 2*new_n-1)
            h_sol = solve_epigraph(h_interp, s_interp + 5e-6, maxiter=mi, n_constraints=nc)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-14:
                    best_h = h_sol
                    best_score = s

    # Phase 10: Final intensive polish
    if tleft() > 20:
        h_loc, s_loc = local_gradient_descent(best_h, max_iters=80, time_limit=min(15, tleft()*0.4))
        if s_loc < best_score - 1e-14:
            best_h = h_loc
            best_score = s_loc
        if tleft() > 10:
            mi = min(100, max(20, int(tleft() * 0.5)))
            h_sol = solve_epigraph(best_h, best_score + 3e-7, maxiter=mi, n_constraints=0)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-14:
                    best_h = h_sol
                    best_score = s

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
