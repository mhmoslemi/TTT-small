5. If the cone is non-empty (shouldn't happen at a true local minimum, but might due to numerical issues), move in that direction.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(y, n):
    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        elig = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(elig) > 0:
            h[elig] += residual / len(elig)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    val = m + np.log(np.sum(exps)) / beta
    return val * dx


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        w = weights[idx]
        if w < 1e-14:
            continue
        if k >= 0:
            lo1 = k
            hi1 = n - 1
            if lo1 <= hi1:
                grad[lo1:hi1 + 1] += w * (1.0 - h[0:hi1 + 1 - k])
            lo2 = 0
            hi2 = n - 1 - k
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] -= w * h[k:hi2 + 1 + k]
        else:
            j = -k
            lo1 = 0
            hi1 = n - 1 - j
            if lo1 <= hi1:
                grad[lo1:hi1 + 1] += w * (1.0 - h[j:hi1 + 1 + j])
            lo2 = j
            hi2 = n - 1
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] -= w * h[0:hi2 + 1 - j]
    return grad * dx


def slsqp_optimize(h0, betas, max_iter=500):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    for beta in betas:
        try:
            res = minimize(
                smooth_obj, best_h.copy(), args=(beta,),
                jac=smooth_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-16}
            )
            if res.x is not None:
                hx = project_feasible(res.x, n)
                c5_new = compute_c5(hx)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = hx
        except Exception:
            pass
    return best_h, best_c5


def systematic_local_search(h, n, max_rounds=200, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for r in range(max_rounds):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                # Try swap
                if abs(best_h[i] - best_h[j]) < 1e-8:
                    continue
                best_h[i], best_h[j] = best_h[j], best_h[i]
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    improved = True
                else:
                    best_h[i], best_h[j] = best_h[j], best_h[i]
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def continuous_local_search(h, n, max_rounds=100, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    fractions = [0.0, 0.25, 0.5, 0.75, 1.0]
    for r in range(max_rounds):
        improved = False
        for i in range(n):
            old_val = best_h[i]
            for frac in fractions:
                if abs(frac - old_val) < 1e-8:
                    continue
                best_h[i] = frac
                best_h = project_feasible(best_h, n)
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    old_val = best_h[i]
                    improved = True
                else:
                    best_h[i] = old_val
                    best_h = project_feasible(best_h, n)
            if improved:
                improved = False
        if not improved:
            break
    return best_h, best_c5


def projected_subgradient(h, n, n_iter, beta, lr0, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    for t in range(n_iter):
        grad = smooth_grad(best_h, beta)
        # Remove mean to preserve sum constraint direction
        grad = grad - np.mean(grad)
        lr = lr0 / (1.0 + 0.01 * t)
        v = 0.9 * v - lr * grad
        candidate = best_h + v
        candidate = project_feasible(candidate, n)
        c5_new = compute_c5(candidate)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = candidate
    return best_h, best_c5


def simulated_annealing(h, n, n_iter, t_start, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    total = float(n_iter)
    for t in range(n_iter):
        frac = float(t) / total
        temp = t_start * (t_end / t_start) ** frac
        move_type = rng.randint(3)
        cand = cur_h.copy()
        if move_type == 0:
            i = rng.randint(n)
            delta = 0.05 * rng.randn()
            cand[i] += delta
            cand = project_feasible(cand, n)
        elif move_type == 1:
            i, j = rng.choice(n, size=2, replace=False)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = rng.choice(n, size=2, replace=False)
            delta = 0.1 * (cand[i] - cand[j])
            cand[i] -= delta
            cand[j] += delta
            cand = project_feasible(cand, n)
        c5_new = compute_c5(cand)
        delta_c = c5_new - cur_c5
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(temp, 1e-10)):
            cur_h = cand
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = cand.copy()
    return best_h, best_c5


def try_resolution(h_source, n_target, n_source, rng, time_limit):
    x_old = np.linspace(0, 2, n_source, endpoint=False)
    x_new = np.linspace(0, 2, n_target, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_source)
    h_interp = project_feasible(h_interp, n_target)
    c5_init = compute_c5(h_interp)
    h_opt, c5_opt = slsqp_optimize(h_interp, [20, 100, 500, 2000], max_iter=400)
    if c5_opt < c5_init:
        h_opt, c5_opt = continuous_local_search(h_opt, n_target, 50, rng)
    rem = time_limit - (time.time() - time_limit + 0)
    return h_opt, c5_opt, n_target
