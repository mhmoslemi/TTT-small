    n = len(h)
    target = n / 2.0

            break

        else:

            if cnt == 0:
                break

from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def profile_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def swap_delta_profile(b, i, j, n):
    """
    Compute the change in profile when swapping b[i] (1->0) and b[j] (0->1).
    Returns the delta array (length 2n-1) and the new profile max.
    """
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    
    # For position i changing from 1 to 0:
    # As first factor: valid shifts s where 0 <= i <= n-1 and 0 <= i+s <= n-1
    #   s range: max(-(n-1), -i) to min(n-1, n-1-i)
    #   contribution change: (0-1)*(1-b[i+s]) = -(1-b[i+s])
    # As second factor: valid shifts s where 0 <= i-s <= n-1 and 0 <= i <= n-1
    #   s range: max(-(n-1), i-(n-1)) to min(n-1, i)
    #   contribution change: b[i-s]*(1-0) - b[i-s]*(1-1) = b[i-s]
    #   Wait: the term is b[i-s]*(1-b[i]). Before: b[i-s]*(1-1)=0. After: b[i-s]*(1-0)=b[i-s].
    #   So change = b[i-s]

    # As second factor: term b[j-s]*(1-b[j]). Before: b[j-s]*(1-0)=b[j-s]. After: b[j-s]*(1-1)=0.
    #   Change = -b[j-s]

    s_max2 = min(n-1, i)

    s_max4 = min(n-1, j)

def binary_sa_search(b, deadline, max_iters=50000):
    """Simulated annealing search for binary solutions."""
    n = len(b)
    b = b.copy()

import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def profile_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau

    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]

    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target

    def fun(self, x):
        return np.sum(x) - self.target

    def jac(self, x):
        return np.ones(len(x))


def slsqp_optimize(h0, tau, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def compute_swap_deltas(b, n):
    dx = 2.0 / n
    m = 2 * n - 1
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]
    delta_remove = np.zeros((len(ones), m))
    delta_add = np.zeros((len(zeros), m))
    for idx, i in enumerate(ones):
        s_min = max(-(n - 1), -i)
        s_max = min(n - 1, n - 1 - i)
        for s in range(s_min, s_max + 1):
            delta_remove[idx, s + n - 1] -= (1.0 - b[i + s])
        s_min2 = max(-(n - 1), i - (n - 1))
        s_max2 = min(n - 1, i)
        for s in range(s_min2, s_max2 + 1):
            delta_remove[idx, s + n - 1] += b[i - s]
        delta_remove[idx] *= dx
    for idx, j in enumerate(zeros):
        s_min = max(-(n - 1), -j)
        s_max = min(n - 1, n - 1 - j)
        for s in range(s_min, s_max + 1):
            delta_add[idx, s + n - 1] += (1.0 - b[j + s])
        s_min2 = max(-(n - 1), j - (n - 1))
        s_max2 = min(n - 1, j)
        for s in range(s_min2, s_max2 + 1):
            delta_add[idx, s + n - 1] -= b[j - s]
        delta_add[idx] *= dx
    return ones, zeros, delta_remove, delta_add


def binary_pairwise_search(b, deadline, max_rounds=200):
    n = len(b)
    b = b.copy()
    curr_profile = profile_values(b)
    curr_c = float(np.max(curr_profile))
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        ones, zeros, delta_remove, delta_add = compute_swap_deltas(b, n)
        best_improve = 1e-14
        best_oi = -1
        best_zi = -1
        for oi in range(len(ones)):
            for zi in range(len(zeros)):
                new_c = float(np.max(curr_profile + delta_remove[oi] + delta_add[zi]))
                improve = curr_c - new_c
                if improve > best_improve:
                    best_improve = improve
                    best_oi = oi
                    best_zi = zi
        if best_oi >= 0:
            i = ones[best_oi]
            j = zeros[best_zi]
            b[i] = 0.0
            b[j] = 1.0
            curr_profile = profile_values(b)
            curr_c = float(np.max(curr_profile))
        else:
            break
    return b, curr_c


def binary_sa_search(b_init, deadline, max_iters=30000, rng=None):
    n = len(b_init)
    b = b_init.copy()
    curr_profile = profile_values(b)
    curr_c = float(np.max(curr_profile))
    best_c = curr_c
    best_b = b.copy()
    if rng is None:
        rng = np.random.RandomState(999)
    T = 0.008
    cooling = 0.997
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        dx = 2.0 / n
        m = 2 * n - 1
        delta = np.zeros(m)
        s_min = max(-(n - 1), -i)
        s_max = min(n - 1, n - 1 - i)
        for s in range(s_min, s_max + 1):
            delta[s + n - 1] -= (1.0 - b[i + s])
        s_min2 = max(-(n - 1), i - (n - 1))
        s_max2 = min(n - 1, i)
        for s in range(s_min2, s_max2 + 1):
            delta[s + n - 1] += b[i - s]
        s_min = max(-(n - 1), -j)
        s_max = min(n - 1, n - 1 - j)
        for s in range(s_min, s_max + 1):
            delta[s + n - 1] += (1.0 - b[j + s])
        s_min2 = max(-(n - 1), j - (n - 1))
        s_max2 = min(n - 1, j)
        for s in range(s_min2, s_max2 + 1):
            delta[s + n - 1] -= b[j - s]
        delta *= dx
        new_profile = curr_profile + delta
        new_c = float(np.max(new_profile))
        dc = new_c - curr_c
        if dc <= 0:
            accept = True
        else:
            if dc > 0.03:
                accept = False
            else:
                accept = rng.uniform() < np.exp(-dc / max(T, 1e-15))
        if accept:
            b[i] = 0.0
            b[j] = 1.0
            curr_profile = new_profile
            curr_c = new_c
            if curr_c < best_c:
                best_c = curr_c
                best_b = b.copy()
        T *= cooling
    return best_b, best_c


def binary_multi_swap_search(b, deadline, max_rounds=40):
    n = len(b)
    b = b.copy()
    curr_profile = profile_values(b)
    curr_c = float(np.max(curr_profile))
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        ones, zeros, delta_remove, delta_add = compute_swap_deltas(b, n)
        no = len(ones)
        nz = len(zeros)
        if no < 2 or nz < 2:
            break
        oi_pairs = np.triu_indices(no, 1)
        zi_pairs = np.triu_indices(nz, 1)
        base_deltas = delta_remove[oi_pairs[0]] + delta_remove[oi_pairs[1]]
        pair_adds = delta_add[zi_pairs[0]] + delta_add[zi_pairs[1]]
        best_improve = 1e-14
        best_oi = -1
        best_zi = -1
        for oi in range(len(oi_pairs[0])):
            bd = base_deltas[oi][None, :]
            new_profiles = curr_profile[None, :] + bd + pair_adds
            new_maxes = np.max(new_profiles, axis=1)
            best_zi_local = int(np.argmin(new_maxes))
            improve = curr_c - new_maxes[best_zi_local]
            if improve > best_improve:
                best_improve = improve
                best_oi = oi
                best_zi = best_zi_local
        if best_oi >= 0:
            i1 = ones[oi_pairs[0][best_oi]]
            i2 = ones[oi_pairs[1][best_oi]]
            j1 = zeros[zi_pairs[0][best_zi]]
            j2 = zeros[zi_pairs[1][best_zi]]
            b[i1] = 0.0
            b[i2] = 0.0
            b[j1] = 1.0
            b[j2] = 1.0
            curr_profile = profile_values(b)
            curr_c = float(np.max(curr_profile))
        else:
            break
    return b, curr_c


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6, 7, 8])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
        if np.sum(h) > n // 2:
            h[-1] = 0.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def cyclic_shift(h, k):
    return np.roll(h, k)


def refine_continuous(h0, deadline, tau_list=None):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_objective(h)
    if tau_list is None:
        tau_list = [0.02, 0.008, 0.003, 0.001, 0.0004, 0.00015, 0.00006, 0.00002]
    for tau in tau_list:
        if time.time() > deadline - 8:
            break
        h_new = slsqp_optimize(h, tau, maxiter=min(300, n * 4))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            best_c = pc
            best_h = parent_h.copy()
            best_n = len(parent_h)
    except Exception:
        pass

    if best_h is None:
        n_init = 48
        best_h = make_start(n_init, rng, 0)
        best_c = exact_objective(best_h)
        best_n = n_init

    parent_n = len(parent_h) if parent_h is not None else 48

    sizes = sorted(set([parent_n,
                        max(8, parent_n - 8), max(8, parent_n - 4),
                        parent_n + 4, parent_n + 8, parent_n + 12,
                        parent_n + 16]))
    sizes = [s for s in sizes if 8 <= s <= 100 and s % 2 == 0]
    sizes = sorted(set(sizes))

    all_continuous = []
    all_binary = []

    for n in sizes:
        if time.time() > deadline - 30:
            break

        cands = []
        if parent_h is not None:
            cands.append(transfer_to_size(parent_h, n))
            cands.append(project_box_sum(transfer_to_size(parent_h, n) + 0.02 * rng.randn(n)))
            cands.append(project_box_sum(transfer_to_size(parent_h, n) + 0.05 * rng.randn(n)))
            for shift in [1, 2, 3]:
                cands.append(transfer_to_size(cyclic_shift(parent_h, shift), n))
        for mode in range(6):
            cands.append(make_start(n, rng, mode))

        scored = []
        for c in cands:
            c_score = exact_objective(c)
            scored.append((c_score, c))
        scored.sort(key=lambda x: x[0])

        n_top = min(4, len(scored))
        local_best_h = scored[0][1]
        local_best_c = scored[0][0]

        for idx in range(n_top):
            if time.time() > deadline - 20:
                break
            h0 = scored[idx][1]
            h_ref, c_ref = refine_continuous(h0, deadline)
            if c_ref < local_best_c:
                local_best_c = c_ref
                local_best_h = h_ref
            all_continuous.append((n, h_ref.copy(), c_ref))
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
                best_n = n

        if n % 2 == 0:
            if time.time() > deadline - 15:
                break
            b = binary_threshold(local_best_h)
            if b is not None:
                c_b_init = exact_objective(b)
                all_binary.append((n, b.copy(), c_b_init))
                if c_b_init < best_c - 1e-12:
                    best_c = c_b_init
                    best_h = b.copy()
                    best_n = n
                b, c_b = binary_pairwise_search(b, deadline, max_rounds=150)
                c_b = exact_objective(b)
                all_binary.append((n, b.copy(), c_b))
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()
                    best_n = n

    # Phase 2: SA refinement on best binary solutions
    all_binary.sort(key=lambda x: x[2])
    top_binaries = all_binary[:min(6, len(all_binary))]

    for (n_b, b_b, c_b) in top_binaries:
        if time.time() > deadline - 15:
            break
        for restart in range(3):
            if time.time() > deadline - 8:
                break
            b_sa, c_sa = binary_sa_search(b_b, deadline, max_iters=20000,
                                          rng=np.random.RandomState(seed + restart * 137 + n_b))
            c_sa = exact_objective(b_sa)
            all_binary.append((n_b, b_sa, c_sa))
            if c_sa < best_c - 1e-12:
                best_c = c_sa
                best_h = b_sa.copy()
                best_n = n_b

    # Phase 3: Multi-swap on best binary
    all_binary.sort(key=lambda x: x[2])
    for (n_b, b_b, c_b) in all_binary[:min(4, len(all_binary))]:
        if time.time() > deadline - 8:
            break
        b_ms, c_ms = binary_multi_swap_search(b_b, deadline, max_rounds=30)
        c_ms = exact_objective(b_ms)
        all_binary.append((n_b, b_ms, c_ms))
        if c_ms < best_c - 1e-12:
            best_c = c_ms
            best_h = b_ms.copy()
            best_n = n_b

    # Phase 4: Epigraph polish on best continuous
    all_continuous.sort(key=lambda x: x[2])
    for (n_c, h_c, c_c) in all_continuous[:min(3, len(all_continuous))]:
        if time.time() > deadline - 12:
            break
        h_epi = epigraph_optimize(h_c, maxiter=80)
        c_epi = exact_objective(h_epi)
        all_continuous.append((n_c, h_epi, c_epi))
        if c_epi < best_c - 1e-12:
            best_c = c_epi
            best_h = h_epi.copy()
            best_n = n_c
        # Also try threshold + binary search on epigraph result
        if n_c % 2 == 0 and time.time() < deadline - 10:
            b_e = binary_threshold(h_epi)
            if b_e is not None:
                b_e, c_be = binary_pairwise_search(b_e, deadline, max_rounds=80)
                c_be = exact_objective(b_e)
                if c_be < best_c - 1e-12:
                    best_c = c_be
                    best_h = b_e.copy()
                    best_n = n_c

    # Phase 5: Final SA with more iterations on absolute best binary
    if best_h is not None and np.all((best_h == 0.0) | (best_h == 1.0)):
        if time.time() < deadline - 5:
            b_final, c_final = binary_sa_search(best_h, deadline, max_iters=50000,
                                                rng=np.random.RandomState(seed + 7777))
            c_final = exact_objective(b_final)
            if c_final < best_c - 1e-12:
                best_c = c_final
                best_h = b_final.copy()
    else:
        # Final LSE refinement
        if time.time() < deadline - 8:
            h_fin, c_fin = refine_continuous(best_h, deadline,
                                             tau_list=[0.001, 0.0004, 0.0001, 0.00004])
            if c_fin < best_c - 1e-12:
                best_c = c_fin
                best_h = h_fin.copy()
            if len(best_h) % 2 == 0 and time.time() < deadline - 5:
                b_fin = binary_threshold(best_h)
                if b_fin is not None:
                    b_fin, c_bf = binary_pairwise_search(b_fin, deadline, max_rounds=100)
                    c_bf = exact_objective(b_fin)
                    if c_bf < best_c - 1e-12:
                        best_c = c_bf
                        best_h = b_fin.copy()

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
