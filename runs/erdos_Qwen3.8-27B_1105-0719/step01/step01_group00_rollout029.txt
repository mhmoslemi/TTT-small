    def __init__(self, n, dx, active_indices, t_upper):
        self.n = n
        self.dx = dx
        self.active = active_indices
        self.t_upper = t_upper
        self.m = n // 2
    
    def fun(self, x):
        # x = [q_0, ..., q_{m-1}, t]
        return x[-1]
    
    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[-1] = 1.0
        return g
    
    def constraint(self, x, k_idx):
        # Returns f_k(q) - t (should be ≤ 0)
        q = x[:self.m]
        t = x[-1]
        h = build_sym_h(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return corrs[self.active[k_idx]] * self.dx - t
    
    def constraint_jac(self, x, k_idx):
        q = x[:self.m]
        h = build_sym_h(q, self.n)
        k = self.active[k_idx]
        d = k - (self.n - 1)
        i_lo = max(0, -d)
        i_hi = min(self.n - 1, self.n - 1 - d)

    def __init__(self, n, dx, active_indices):
        self.n = n
        self.dx = dx
        self.active = np.array(active_indices)
        self.m = n // 2
    
    def fun(self, x):
        return x[self.m]  # t is the last variable
    
    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[self.m] = 1.0
        return g
    
    def cons_fun(self, x):
        q = x[:self.m]
        t = x[self.m]
        h = build_sym_h(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return t - corrs[self.active] * self.dx  # should be >= 0
    
    def cons_jac(self, x):
        q = x[:self.m]
        h = build_sym_h(q, self.n)
        J = np.zeros((len(self.active), self.m + 1))
        for row, k in enumerate(self.active):
            d = k - (self.n - 1)
            i_lo = max(0, -d)
            i_hi = min(self.n - 1, self.n - 1 - d)
            G = np.zeros(self.n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G[idx + d] += (1.0 - h[idx])
                G[idx] += (-h[idx + d])
            J[row, :self.m] = -(G[:self.m] - G[self.n-1-self.m:self.n][::-1]) * self.dx
            J[row, self.m] = 1.0
        return J

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def build_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[n-1-m:n] = 1.0 - q[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h


def extract_q(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1-m:n][::-1])
    return np.clip(q, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G_all[k, idx + d] += one_minus_h[idx]
        G_all[k, idx] += (-h[idx + d])
    return corrs, G_all


def q_gradient_from_full(G_all, n):
    m = n // 2
    return G_all[:, :m] - G_all[:, n-1-m:n][::-1, :]


class SmoothQProb:
    def __init__(self, n, dx, beta):
        self.n = n
        self.dx = dx
        self.beta = beta
        self.m = n // 2

    def value(self, q):
        h = build_sym_h(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        f = corrs * self.dx
        return float(logsumexp(self.beta * f) / self.beta)

    def gradient(self, q):
        h = build_sym_h(q, self.n)
        corrs, G_all = compute_full_gradient_h(h, self.n)
        f = corrs * self.dx
        shifted = self.beta * f - (self.beta * f).max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w /= w_sum
        grad_q = (w @ q_gradient_from_full(G_all, self.n)) * self.dx
        return grad_q


class EpigraphQProb:
    def __init__(self, n, dx, active_indices):
        self.n = n
        self.dx = dx
        self.active = np.asarray(active_indices, dtype=int)
        self.m = n // 2

    def fun(self, x):
        return x[self.m]

    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[self.m] = 1.0
        return g

    def cons_fun(self, x):
        q = x[:self.m]
        t = x[self.m]
        h = build_sym_h(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return t - corrs[self.active] * self.dx

    def cons_jac(self, x):
        q = x[:self.m]
        h = build_sym_h(q, self.n)
        J = np.zeros((len(self.active), self.m + 1))
        G_all = np.zeros((len(self.active), self.n))
        for row, k in enumerate(self.active):
            d = k - (self.n - 1)
            i_lo = max(0, -d)
            i_hi = min(self.n - 1, self.n - 1 - d)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G_all[row, idx + d] += (1.0 - h[idx])
                G_all[row, idx] += (-h[idx + d])
        Gq = (G_all[:, :self.m] - G_all[:, self.n-1-self.m:self.n][::-1, :])
        J[:, :self.m] = -Gq * self.dx
        J[:, self.m] = 1.0
        return J


def optimize_smooth_q(q0, n, dx, beta, maxiter=400):
    prob = SmoothQProb(n, dx, beta)
    bounds = [(0.0, 1.0)] * prob.m
    try:
        res = minimize(prob.value, q0.copy(), jac=prob.gradient,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-15})
        q_out = np.clip(res.x, 0.0, 1.0)
    except Exception:
        q_out = q0.copy()
    h_out = build_sym_h(q_out, n)
    c5 = compute_c5(h_out)
    return q_out, c5


def optimize_epigraph_q(q0, n, dx, t0, budget, max_rounds=6, active_cap=120):
    m = n // 2
    h0 = build_sym_h(q0, n)
    c5_init = compute_c5(h0)
    corrs0, _ = compute_full_gradient_h(h0, n)
    f_all = corrs0 * dx
    t_val = c5_init
    active = list(np.argsort(-f_all)[:min(active_cap, 2 * n - 1)])
    best_q = q0.copy()
    best_c5 = c5_init

    for rnd in range(max_rounds):
        if time.time() - t0 > budget - 20:
            break
        prob = EpigraphQProb(n, dx, active)
        x0 = np.concatenate([q0, [t_val]])
        t_upper = max(t_val + 0.05, 1.0)
        bounds = [(0.0, 1.0)] * m + [(0.0, t_upper)]
        cons = [{'type': 'ineq', 'fun': prob.cons_fun, 'jac': prob.cons_jac}]
        try:
            res = minimize(prob.fun, x0, jac=prob.jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-14})
            q_new = np.clip(res.x[:m], 0.0, 1.0)
            t_new = res.x[m]
        except Exception:
            q_new = q0.copy()
            t_new = t_val

        h_new = build_sym_h(q_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_q = q_new.copy()

        corrs_new, _ = compute_full_gradient_h(h_new, n)
        f_new = corrs_new * dx
        violations = np.where(f_new > t_new + 1e-8)[0]
        if len(violations) == 0:
            break
        active = list(set(active + list(violations)))
        if len(active) > active_cap:
            active_indices_all = np.argsort(-f_new)
            active = list(active_indices_all[:active_cap])
        t_val = max(t_new, f_new.max())

    return best_q, best_c5


def make_random_q(n, rng, mode='uniform'):
    m = n // 2
    if mode == 'uniform':
        return rng.uniform(0, 1, m)
    elif mode == 'binary':
        return (rng.uniform(0, 1, m) > 0.5).astype(float)
    elif mode == 'sine':
        f = rng.uniform(1, 5, 1)
        phase = rng.uniform(0, 2 * np.pi, 1)
        x = np.arange(m)
        return np.clip(0.5 + 0.4 * np.sin(2 * np.pi * f[0] * x / m + phase[0]), 0, 1)
    elif mode == 'two_block':
        q = np.full(m, 0.5)
        split = rng.randint(m // 4, 3 * m // 4)
        q[:split] = 1.0
        q[split:] = 0.0
        noise = rng.normal(0, 0.1, m)
        return np.clip(q + noise, 0, 1)
    return np.full(m, 0.5)


def remaining_time(t0, budget):
    return budget - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    h_parent = project_feasible(h_parent, n_parent / 2.0)
    best_c5 = compute_c5(h_parent)
    best_h = h_parent.copy()

    # Convert to q-space
    n = n_parent
    dx = 2.0 / n
    m = n // 2
    q0 = extract_q(h_parent, n)

    # Phase 1: Smooth optimization with increasing beta
    betas = [30, 60, 120, 250, 500, 800, 1200, 2000]
    for beta in betas:
        if remaining_time(t0, budget_s) < 40:
            break
        maxit = 500 if beta <= 200 else 300
        q_opt, c5_val = optimize_smooth_q(q0, n, dx, beta, maxiter=maxit)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = build_sym_h(q_opt, n)
        q0 = q_opt

    # Phase 2: Epigraph polishing on the main n
    if remaining_time(t0, budget_s) > 80:
        q_epi, c5_epi = optimize_epigraph_q(q0, n, dx, t0, budget_s, max_rounds=8, active_cap=150)
        if c5_epi < best_c5:

            q0 = q_epi

    if remaining_time(t0, budget_s) > 50:

            q0 = q_opt

    # Phase 3: Try different n values
