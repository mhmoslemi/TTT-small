import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    """Project onto {0<=h<=1, sum(h)=n/2} using bisection on shift."""
    n = len(vec)
    h = np.asarray(vec, dtype=np.float64).copy()
    target = n / 2.0
    
    # Bisection to find shift a such that sum(clip(h+a, 0, 1)) = target
    def sum_clipped(a):
        return np.clip(h + a, 0.0, 1.0).sum()
    
    # Find bracket
    a_lo, a_hi = -2.0, 2.0
    while sum_clipped(a_lo) > target + 1e-12:
        a_lo *= 2
    while sum_clipped(a_hi) < target - 1e-12:
        a_hi *= 2
    
    # Bisection
    for _ in range(60):
        a_mid = (a_lo + a_hi) / 2
        if sum_clipped(a_mid) > target:
            a_hi = a_mid
        else:
            a_lo = a_mid
    
    h_new = np.clip(h + (a_lo + a_hi) / 2, 0.0, 1.0)
    
    # Fix residual
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            # Add to entries with most room
            room = 1.0 - h_new
            order = np.argsort(-room)
            remaining = diff
            for i in range(n):
                idx = order[i]
                add = min(remaining, room[idx])
                h_new[idx] += add
                remaining -= add
                if remaining <= 1e-15:
                    break
        else:
            order = np.argsort(h_new)
            remaining = -diff
            for i in range(n):
                idx = order[i]
                rem = min(remaining, h_new[idx])
                h_new[idx] -= rem
                remaining -= rem
                if remaining <= 1e-15:

    return h_new

def profile_gradient(h):
    """Compute the full gradient matrix G where G[m,j] = dr_m/dh_j."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    
    # Zero lag (m = n-1)
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    
    # Positive lags d = 1, ..., n-1 (m = n-1+d)
    for d in range(1, n):
        m = n - 1 + d
        G[m, d:] += dx * (1.0 - h[:n-d])
        G[m, :n-d] -= dx * h[d:]
    
    # Negative lags d = -1, ..., -(n-1) (m = n-1+d, i.e., m = n-2, ..., 0)
    for s in range(1, n):
        m = n - 1 - s
        G[m, :n-s] += dx * (1.0 - h[s:])
        G[m, s:] -= dx * h[:n-s]
    
    return G

def active_gradient(h, profile, threshold=1e-6):
    """Compute weighted gradient for active set."""
    c = np.max(profile)
    active = profile >= (c - threshold)
    if not np.any(active):
        active = np.zeros(len(profile), dtype=bool)
        active[np.argmax(profile)] = True
    
    # Weights proportional to how close to max
    w = profile[active] - (c - threshold)
    w = np.maximum(w, 0)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones_like(w)
        w_sum = len(w)
    w = w / w_sum
    
    G = profile_gradient(h)
    grad = G[active, :].T @ w
    # Project onto sum-zero subspace
    grad -= grad.mean()
    return grad

def line_search(h, grad, c_best, max_alpha=0.1, min_alpha=1e-8):

    n = len(h)
    # Step direction projected to feasible set
    # We need to move along -grad while maintaining feasibility
    # Use the projection approach
    
    best_h = h
    best_c = c_best
    alpha = max_alpha

            if alpha < min_alpha * 10:

                alpha = min(alpha * 1.5, max_alpha)
            else:
                break
        else:
            alpha *= 0.5
        if alpha < min_alpha:
            break
    
    return best_h, best_c

def optimize_active(h, max_iters=200, time_limit=30.0):

    t_start = time.perf_counter()

    h_cur = h.copy()
    c_best = exact_score(h_cur)
    
    for it in range(max_iters):

            break
        
        profile = compute_profile(h_cur)
        c = np.max(profile)

            h_cur = h_cur  # already updated

        thresh = max(1e-6, 0.01 * (c - 0.38))
        grad = active_gradient(h_cur, profile, threshold=thresh)

        if grad_norm < 1e-12:
            break
        grad /= grad_norm

        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = repair_feasible(h_cur - alpha * grad)
            c_try = exact_score(h_try)
            if c_try < c_best - 1e-12:
                h_cur = h_try
                c_best = c_try
                improved = True
                break

                    if c_try < c_best + 0.001:  # Accept slightly worse
                        h_cur = h_try
                        c_best = max(c_best, c_try)  # Keep track of best
                        break
    
    return h_cur, c_best

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def repair_feasible(vec):
    n = len(vec)
    h = np.asarray(vec, dtype=np.float64).copy()
    target = n / 2.0
    # Water-filling: find shift a such that sum(clip(h+a, 0, 1)) = target
    # Use bisection
    lo, hi = -3.0, 3.0

    s_hi = np.clip(h + hi, 0, 1).sum()
    while s_lo > target:

        s_lo = np.clip(h + lo, 0, 1).sum()
    while s_hi < target:

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.asarray(vec, dtype=np.float64).copy()
    target = n / 2.0
    lo, hi = -3.0, 3.0
    while np.clip(h + lo, 0, 1).sum() > target:
        lo *= 2.0
    while np.clip(h + hi, 0, 1).sum() < target:
        hi *= 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmin(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def make_profile_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - 1 + d
        G[m, d:] += dx * (1.0 - h[:n - d])
        G[m, :n - d] -= dx * h[d:]
        m2 = n - 1 - d
        G[m2, :n - d] += dx * (1.0 - h[d:])
        G[m2, d:] -= dx * h[:n - d]
    return G


def lse_optimize(h0, tau_list, steps_per_tau, time_limit):
    t_start = time.perf_counter()
    h = h0.copy()
    n = len(h)
    best_c = exact_score(h)
    for tau in tau_list:
        for _ in range(steps_per_tau):
            if time.perf_counter() - t_start > time_limit:
                return h, best_c
            profile = compute_profile(h)
            max_p = np.max(profile)
            shifted = np.clip((profile - max_p) / tau, -100, 100)
            w = np.exp(shifted)
            ws = w.sum()
            if ws < 1e-30:
                w = np.zeros(len(profile))
                w[int(np.argmax(profile))] = 1.0
            else:
                w /= ws
            G = make_profile_gradient(h)
            grad = G.T @ w
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            grad /= gn
            improved = False

                h_try = repair_feasible(h - alpha * grad)

    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

    prob = EpiProblem(n)
