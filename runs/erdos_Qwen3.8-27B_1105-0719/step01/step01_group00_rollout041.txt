import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_to_target(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def build_sym_h(q, n):
    m = n // 2
    h = np.zeros(n)
    h[:m] = q
    h[n - 1: n - 1 - m: -1] = 1.0 - q
    if n % 2 == 1:
        h[m] = 0.5
    return h


def get_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + (1.0 - h[n - 1: n - 1 - m: -1]))
    return np.clip(q, 0.0, 1.0)


def corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def sym_lse_obj(q, n, beta):
    h = build_sym_h(q, n)
    c = corr_array(h)
    return float(logsumexp(beta * c) / beta)


def sym_lse_grad(q, n, beta):
    m = n // 2
    h = build_sym_h(q, n)
    c = corr_array(h)
    shifted = c - c.max()
    exp_s = np.exp(beta * shifted)
    w = exp_s / exp_s.sum()
    dx = 2.0 / n
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        wk = w[k] * dx
        if d == 0:
            grad_h += wk * (1.0 - 2.0 * h)
        elif d > 0:
            grad_h[d:n] += wk * (1.0 - h[0:n - d])
            grad_h[0:n - d] -= wk * h[d:n]
        else:
            da = -d
            grad_h[0:n - da] += wk * (1.0 - h[da:n])
            grad_h[da:n] -= wk * h[0:n - da]
    grad_q = np.zeros(m)
    for i in range(m):
        grad_q[i] = grad_h[i] - grad_h[n - 1 - i]
    return grad_q


def sym_lse_optimize(q0, n, beta, maxiter, time_limit):
    m = n // 2
    t_start = time.time()

    def obj(x):
        if time.time() - t_start > time_limit:
            raise KeyboardInterrupt
        return sym_lse_obj(x, n, beta)

    def grad(x):
        if time.time() - t_start > time_limit:
            raise KeyboardInterrupt
        return sym_lse_grad(x, n, beta)

    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(obj, q0, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        if res.success or res.fun < float('inf'):
            q_opt = np.clip(res.x, 0.0, 1.0)
            return q_opt
    except (KeyboardInterrupt, Exception):
        pass
    return q0


def epigraph_sym_solve(q0, n, max_const, maxiter, time_limit, t_upper):
    m = n // 2
    h0 = build_sym_h(q0, n)
    c0 = corr_array(h0)
    c_max = c0.max()
    margin = max(1e-9, 0.03 * c_max)
    active = np.where(c0 >= c_max - margin)[0]
    k_list = sorted(active.tolist())
    if len(k_list) > max_const:
        idx = np.argsort(c0)[::-1][:max_const]
        k_list = sorted(idx.tolist())
    k_set = set()
    for k in k_list:
        for dk in range(-3, 4):
            kk = k + dk
            if 0 <= kk <= 2 * n - 2:
                k_set.add(kk)
    k_final = sorted(k_set)[:max_const]

    dx = 2.0 / n
    v0 = np.zeros(m + 1)
    v0[:m] = q0
    v0[m] = c_max + 1e-8

    t_start = time.time()

    def obj(v):
        if time.time() - t_start > time_limit:
            raise KeyboardInterrupt
        return v[m]

    def obj_grad(v):
        if time.time() - t_start > time_limit:
            raise KeyboardInterrupt
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g

    def constraints(v):
        if time.time() - t_start > time_limit:
            raise KeyboardInterrupt
        qv = v[:m]
        h = build_sym_h(qv, n)
        c = corr_array(h)
        cons = []
        for k in k_final:
            cv = lambda kk: v[m] - c[kk], k=kk
            cons.append({'type': 'ineq', 'fun': cv})
        return cons

    bounds = [(0.0, 1.0)] * m + [(0.0, t_upper)]

    try:
        res = minimize(obj, v0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and res.x is not None:
            q_out = np.clip(res.x[:m], 0.0, 1.0)
            return q_out
    except (KeyboardInterrupt, Exception):
        pass
    return q0

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_to_target(h_new, target)


def make_sym_random(n, rng, sigma):
    m = n // 2
    q = np.full(m, 0.5)
    t = np.linspace(0, 1, m)

    q = np.clip(q, 0.0, 1.0)
    return q
