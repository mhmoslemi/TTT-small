def simulated_annealing_binary(n, t_start, t_end, deadline):

    # Initialize: random binary vector with n/2 ones

    np.random.shuffle(pos)
    h = np.zeros(n)
    h[:n//2] = 1.0
    
    curr_c = exact_objective(h)
    best_h = h.copy()
    best_c = curr_c
    
    T = t_start
    total_iters = 1000000  # or based on time
    decay = (t_end / t_start) ** (1.0 / total_iters)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

            h[i] = 1.0
            h[j] = 0.0

Alternatively, I can use a "pair swap" approach where I maintain the profile explicitly and update it incrementally after each swap.

When I swap h[i] (1→0) and h[j] (0→1):
- For each lag s, P[s] changes by:
  - Removing contribution of h[i]=1: -dx * (1-h[i+s]) for the "first factor" at position i
  - Adding contribution of h[j]=1: +dx * (1-h[j+s]) for the "first factor" at position j
  - And the "second factor" contributions

This is O(n) per swap to update the profile, which for n=48 is 95 operations. Much faster than calling np.correlate.

class BinaryProfile:
    def __init__(self, h):
        self.h = h.astype(np.float64)
        self.n = len(h)
        self.dx = 2.0 / self.n
        self.P = self._compute()
        self.ones = set(np.where(h == 1)[0].tolist())
        self.zeros = set(np.where(h == 0)[0].tolist())
    
    def _compute(self):
        return np.correlate(self.h, 1.0 - self.h, mode='full') * self.dx
    
    def eval_max(self):
        return float(np.max(self.P))
    
    def swap(self, i, j):
        """Swap h[i]=1 to 0 and h[j]=0 to 1. Returns improvement (negative = better)."""
        n = self.n
        dx = self.dx
        h = self.h
        old_max = self.eval_max()
        
        self.h[i] = 0.0
        self.h[j] = 1.0
        self.ones.discard(i)
        self.ones.add(j)
        self.zeros.discard(j)
        self.zeros.add(i)
        
        # Update profile incrementally
        # P changes because h[i] went from 1 to 0 and h[j] went from 0 to 1
        # For each lag s, the change comes from positions i and j in both "first" and "second" factor roles
        
        # For position k changing from old_val to new_val:
        # It appears in P_s for lags where it's in the "first" or "second" position
        # First position: P_s includes h[k]*(1-h[k+s]) for lag s where k is valid source
        # Second position: P_s includes h[k-s]*(1-h[k]) for lag s where k is valid target
        
        # Change at position i (1→0):
        # As first factor (source): for lag s where i is in [0, n-1-s], i.e., s in [0, n-1-i] (positive lags)
        #   or i is in [s, n-1] for negative lag s, i.e., |s| in [0, i] 
        #   Contribution was dx*(1-h[i+s]), now 0. Change: -dx*(1-h[i+s])
        # As second factor (target): for lag s where i-s >= 0 and i-s <= n-1-s, i.e., s in [i-(n-1), i]
        #   Contribution was -dx*h[i-s] (from the -h[i] part of (1-h[i]))
        #   Wait, let me think again...
        
        # P_s = dx * sum_{i valid} h[src_i] * (1 - h[tgt_i])
        # where src_i and tgt_i depend on the lag
        
        # For np.correlate(h, 1-h, 'full'), the standard formula is:
        # output[k] = sum_i h[i] * (1-h)[i + k - (n-1)]
        #           = sum_i h[i] * (1 - h[i + k - (n-1)])
        
        # So position p in h appears as:
        # - "first" factor: h[p] * (1-h[p + s]) where s = k-(n-1), valid when 0<=p<n and 0<=p+s<n
        #   i.e., s in [-p, n-1-p], and the output index is k = s + n - 1
        # - "second" factor: h[p-s] * (1-h[p]) where the output index is k = s + n - 1
        #   valid when 0<=p-s<n and 0<=p<n, i.e., s in [p-(n-1), p]

        # First factor: for s in [-i, n-1-i], P_{s+n-1} changes by -dx*(1-h[i+s])
        # Second factor: for s in [i-(n-1), i], P_{s+n-1} changes by +dx*h[i+s]
        #   (because (1-h[i]) goes from 0 to 1, so h[i+s]*(1-h[i]) goes from 0 to h[i+s])
        #   Wait: the term is h[i+s] * (1-h[i]). When h[i] was 1, this was h[i+s]*0 = 0.
        #   When h[i] becomes 0, this becomes h[i+s]*1 = h[i+s].
        #   So the change is +dx * h[i+s] for s in [i-(n-1), i].
        
        # Change when h[j] goes from 0 to 1:
        # First factor: for s in [-j, n-1-j], P_{s+n-1} changes by +dx*(1-h[j+s])

```python
import numpy as np
import time

def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_objective(h):
    return float(np.max(exact_profile(h)))

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    x = np.asarray(x, dtype=np.float64).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    h = np.clip(x + lam, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def gradient_at_lag(h, lag_idx):
    n = len(h)
    dx = 2.0 / n
    s = lag_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            grad[i] += dx * (1.0 - h[i + s])
            grad[i + s] -= dx * h[i]
    else:
        a = -s
        for i in range(a, n):
            grad[i] += dx * (1.0 - h[i - a])
            grad[i - a] -= dx * h[i]
    return grad

def active_set_gp(h0, deadline, max_rounds=50):
    """Active set projected gradient with backtracking."""
    h = h0.copy()
    n = len(h)
    p = 2 * n - 1
    curr_c = exact_objective(h)
    best_h = h.copy()
    best_c = curr_c
    
    for round_num in range(max_rounds):
        if time.time() > deadline - 3:
            break
        P = exact_profile(h)
        c = float(np.max(P))
        
        # Active set: lags within eps of max
        eps = max(0.001 * (1 + round_num * 0.1), 1e-4)
        active = np.where(P >= c - eps)[0]
        
        # Compute weighted average gradient
        weights = (c - P[active] + eps * 0.5)  # heavier for lags closer to max
        # Actually, for those AT the max, weight is eps*0.5; for those at c-eps, weight is eps
        # So all positive. Let me use a different weighting:
        # weight = 1 for active max, proportional to (c - P[j]) + eps for others
        weights = np.ones(len(active))
        # For lags that are exactly at max, give them weight 2
        exact_max = P[active] >= c - 1e-12
        weights[exact_max] = 2.0
        weight_sum = np.sum(weights)
        weights /= weight_sum
        
        g = np.zeros(n)
        for idx, w in zip(active, weights):
            g += w * gradient_at_lag(h, idx)
        
        # Remove mean to preserve sum
        g -= np.mean(g)
        
        # Backtracking line search
        alpha = 0.05
        for bt in range(20):
            h_new = project_feasible(h - alpha * g)
            c_new = exact_objective(h_new)
            if c_new < c - 1e-13:
                h = h_new
                curr_c = c_new
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = h.copy()
                break
            alpha *= 0.7
        else:
            # No improvement found
            if round_num > 3:
                break
    
    return best_h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def binary_hill_climb(b, deadline, max_rounds=500):
    """Greedy 1-swap hill climbing for binary vectors."""
    n = len(b)
    b = b.astype(np.float64).copy()
    curr_c = exact_objective(b)
    best_h = b.copy()
    best_c = curr_c
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                impr = curr_c - c
                b[i] = 1.0
                b[j] = 0.0
                if impr > best_improve:
                    best_improve = impr
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            if curr_c < best_c:
                best_c = curr_c
                best_h = b.copy()
        else:
            break
    
    return best_h, best_c

def binary_sa(b0, deadline, T0=0.05, T1=0.001):

    n = len(b0)
    b = b0.astype(np.float64).copy()
    rng = np.random.RandomState(12345)
    curr_c = exact_objective(b)
    best_h = b.copy()
    best_c = curr_c
    
    # Estimate iterations per second to set total budget

    if max_iters > 0:

        if i == j:
            continue

        b[j] = 1.0
        new_c = exact_objective(b)
        
        if new_c <= curr_c or rng.random() < np.exp(-(new_c - curr_c) / T):
            curr_c = new_c
            if curr_c < best_c - 1e-12:
                best_c = curr_c
                best_h = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
        
        T = max(T1, T * decay)
    
    return best_h, best_c

def transfer_to_size(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return project_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_feasible(h_new)

def make_starts(n, rng):
    """Generate diverse starting vectors."""
    starts = []
    # Block
    h = np.zeros(n)
    h[:n // 2] = 1.0
    starts.append(h)
    # Shuffled block
    h = np.zeros(n)
    h[:n // 2] = 1.0
    starts.append(h[rng.permutation(n)])
    # Alternating
    h = np.zeros(n)
    h[::2] = 1.0
    if np.sum(h) != n // 2:
        h[rng.randint(n)] = 0.0  # fix if needed (odd n)
    starts.append(project_feasible(h))
    # Sine
    x = np.linspace(0, 1, n)
    for freq in [2, 3, 4]:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        starts.append(project_feasible(h))
    # Random
    starts.append(project_feasible(rng.uniform(0, 1, n)))
    # Two blocks
    block = max(2, n // 6)
    h = np.zeros(n)
    h[:block] = 1.0
    h[n - block:] = 1.0
    mid = n // 2 - block
    if mid > 0:
        h[n // 4:n // 4 + mid] = 1.0
    starts.append(project_feasible(h))
    return starts
