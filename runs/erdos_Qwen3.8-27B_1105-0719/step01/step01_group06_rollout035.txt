def coord_descent_1D(h, j, profile, dx, n):

    # We want to minimize max_m (profile[m] + (t - h[j]) * c_m) over t in [0,1]

    # P_m(u) = profile[m] + u * c_m

def feasible(V, A, B):
    """Check if exists t in [0,1] such that A[m] + t*B[m] <= V for all m."""
    # For B[m] > 0: t <= (V - A[m]) / B[m]
    # For B[m] < 0: t >= (V - A[m]) / B[m]
    # For B[m] == 0: need A[m] <= V
    
    pos = B > 1e-12
    neg = B < -1e-12
    zer = ~(pos | neg)
    
    if np.any(A[zer] > V + 1e-12):
        return False
    
    t_hi = 1.0
    if np.any(pos):
        t_hi = min(t_hi, np.min((V - A[pos]) / B[pos]))
    
    t_lo = 0.0
    if np.any(neg):
        t_lo = max(t_lo, np.max((V - A[neg]) / B[neg]))
    
    return t_lo <= t_hi + 1e-12

def find_optimal_t(V, A, B, h_j):
    """Find a feasible t in [0,1] for the given V."""
    pos = B > 1e-12
    neg = B < -1e-12
    
    t_hi = 1.0
    if np.any(pos):
        t_hi = min(t_hi, np.min((V - A[pos]) / B[pos]))
    
    t_lo = 0.0
    if np.any(neg):
        t_lo = max(t_lo, np.max((V - A[neg]) / B[neg]))
    
    if t_lo <= t_hi:
        return (t_lo + t_hi) / 2  # any feasible point works
    else:
        return h_j  # shouldn't happen

def solve_1D_minimax(A, B, t_lo=0.0, t_hi=1.0):
    """
    Minimize max_m (A[m] + t*B[m]) over t in [t_lo, t_hi].
    A, B are arrays of length M.
    Returns the optimal t.
    """
    # Bisection on V
    # Lower bound: max_m min_{t in [t_lo,t_hi]} (A[m]+t*B[m])
    # For each m: if B[m] >= 0, min is at t_lo: A[m]+t_lo*B[m]
    #            if B[m] < 0, min is at t_hi: A[m]+t_hi*B[m]
    V_lo = np.max(np.where(B >= 0, A + t_lo * B, A + t_hi * B))
    # Upper bound: max_m max(t in [t_lo,t_hi]) (A[m]+t*B[m]) = max(max(A+t_lo*B), max(A+t_hi*B))
    V_hi = np.max([np.max(A + t_lo * B), np.max(A + t_hi * B)])
    
    # Bisection
    for _ in range(60):
        V_mid = (V_lo + V_hi) / 2.0
        if feasible_1D(V_mid, A, B, t_lo, t_hi):
            V_hi = V_mid
        else:
            V_lo = V_mid
    
    # Find optimal t at V_hi
    return find_t(V_hi, A, B, t_lo, t_hi)

def feasible_1D(V, A, B, t_lo, t_hi):
    """Check if exists t in [t_lo, t_hi] with A[m]+t*B[m] <= V for all m."""
    pos = B > 1e-14
    neg = B < -1e-14
    zer = ~(pos | neg)
    
    if np.any(A[zer] > V + 1e-14):
        return False
    
    lo = t_lo
    if np.any(neg):
        lo = max(lo, np.max((V - A[neg]) / B[neg]))
    
    hi = t_hi
    if np.any(pos):
        hi = min(hi, np.min((V - A[pos]) / B[pos]))
    
    return lo <= hi + 1e-12

def find_t(V, A, B, t_lo, t_hi):
    """Find a feasible t."""
    pos = B > 1e-14
    neg = B < -1e-14
    
    lo = t_lo
    if np.any(neg):
        lo = max(lo, np.max((V - A[neg]) / B[neg]))
    
    hi = t_hi
    if np.any(pos):
        hi = min(hi, np.min((V - A[pos]) / B[pos]))
    
    if lo <= hi:
        return (lo + hi) / 2.0
    else:
        # Shouldn't happen, return midpoint
        return (t_lo + t_hi) / 2.0
