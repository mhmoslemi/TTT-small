        lo_from_pos = -h[d > 1e-15] / pos
        tau_hi = min(tau_hi, np.min(hi_from_pos))
        tau_lo = max(tau_lo, np.max(lo_from_pos))
    if len(neg) > 0:
        hi_from_neg = -h[d < -1e-15] / neg
        lo_from_neg = (1.0 - h[d < -1e-15]) / neg
        tau_hi = min(tau_hi, np.min(hi_from_neg))
        tau_lo = max(tau_lo, np.max(lo_from_neg))
    if tau_hi <= tau_lo + 1e-15:
        return None
    return (tau_lo, tau_hi)

def line_optimize(h, d):
    """
    Find τ that minimizes F(τ) = max_k (a_k τ² + b_k τ + c_k) over feasible range.
    Returns (new_h, new_score) or (h, current_score) if no improvement.
    """
    fr = find_feasible_range(h, d)
    if fr is None:
        return h, exact_score(h)
    tau_lo, tau_hi = fr
    
    # Compute coefficients
    eps = min(0.05, 0.15 * (tau_hi - tau_lo))
    if eps < 1e-10:
        return h, exact_score(h)
    a, b, c = compute_direction_coeffs(h, d, eps)
    
    # Grid search for approximate minimum
    n_grid = 50
    taus = np.linspace(tau_lo, tau_hi, n_grid)
    vals = (a[None, :] * taus[:, None]**2 + b[None, :] * taus[:, None] + c[None, :]).max(axis=1)
    idx = np.argmin(vals)
    
    # Also check τ=0
    f0 = c.max()
    if vals[idx] >= f0 - 1e-12:
        return h, f0
    
    # Refine around the best point
    left = max(tau_lo, taus[max(0, idx - 3)])
    right = min(tau_hi, taus[min(n_grid - 1, idx + 3)])
    
    for _ in range(25):
        if right - left < 1e-9:
            break
        m1 = left + (right - left) * 0.382
        m2 = left + (right - left) * 0.618
        f1 = (a * m1**2 + b * m1 + c).max()
        f2 = (a * m2**2 + b * m2 + c).max()
        if f1 < f2:
            right = m2
        else:
            left = m1
    
    best_tau = (left + right) / 2.0
    f_best = (a * best_tau**2 + b * best_tau + c).max()
    
    if f_best < f0 - 1e-11:
        new_h = repair_feasible(h + best_tau * d)
        return new_h, f_best
    return h, f0

def gradient_direction(h, n_top=10):
    """Compute a smart descent direction based on active lags."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    prof = get_profile(h)
    active = np.argsort(prof)[::-1][:n_top]
    
    # Compute gradient for active lags
    grad = np.zeros(n)
    for k in active:
        delta = k - (n - 1)
        w = prof[k] / max(prof.max(), 1e-15)
        if delta == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif delta > 0:
            # P_k = dx * sum_{u=0}^{n-1-delta} h_{u+delta}*(1-h_u)
            # d/dh_v = dx*(1-h_{v-delta}) for v>=delta, minus dx*h_{v+delta} for v<=n-1-delta
            if delta < n:
                grad[delta:] += w * dx * (1.0 - h[:n-delta])
                grad[:n-delta] -= w * dx * h[delta:]
        else:
            d = -delta
            if d < n:
                grad[:n-d] += w * dx * (1.0 - h[d:])
                grad[d:] -= w * dx * h[:n-d]
    
    # Project to zero-sum
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return np.zeros(n)
    return -grad / norm

def block_direction(h, rng):
    """Generate a random block exchange direction."""
    n = len(h)
    block_size = rng.integers(2, max(3, n // 10))
    i = rng.integers(0, n - block_size)
    j = rng.integers(0, n - block_size)
    if abs(i - j) < block_size:
        return None
    d = np.zeros(n)
    d[i:i+block_size] = -1.0 / block_size
    d[j:j+block_size] = 1.0 / block_size
    # Ensure we're moving from high to low on average
    if h[i:i+block_size].mean() < h[j:j+block_size].mean():
        d = -d
    return d

def random_pair_direction(h, rng):
    """Generate a random pair exchange direction."""
    n = len(h)
    # Sample pairs weighted by |h[i] - h[j]|
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return None
    d = np.zeros(n)
    if h[i] > h[j]:
        d[i] = -1.0
        d[j] = 1.0
    else:
        d[i] = 1.0
        d[j] = -1.0
    return d

def solve_epigraph(h_start, t_start, maxiter=60):
    """SLSQP epigraph solver."""
    n = len(h_start)
    M = 2 * n - 1
    
    def obj(x):
        return x[n]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        n_ = len(h)
        dx = 2.0 / n_
        M_ = 2 * n_ - 1
        J = np.zeros((1 + M_, n_ + 1))
        J[0, :n_] = 1.0
        for k in range(M_):
            delta = k - (n_ - 1)
            if delta == 0:
                J[1 + k, :n_] = -dx * (1.0 - 2.0 * h)
            elif delta > 0 and delta < n_:
                J[1 + k, delta:n_] = -dx * (1.0 - h[:n_-delta])
                J[1 + k, :n_-delta] += dx * h[delta:n_]
            else:
                d = -delta
                if d < n_:
                    J[1 + k, :n_-d] = -dx * (1.0 - h[d:])
                    J[1 + k, d:n_] += dx * h[:n_-d]
            J[1 + k, n_] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 1.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(
            obj, x0, jac=obj_jac, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def make_binary(n, rng):
    """Create a random binary vector with exactly n/2 ones."""
    idx = rng.permutation(n)[:n//2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.ascontiguousarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def symmetrize(h):
    return repair_feasible(0.5 * (h + h[::-1].copy()))

def find_feasible_range(h, d):
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    tau_lo = 0.0
    tau_hi = np.inf
    if np.any(pos_mask):
        tau_hi = min(tau_hi, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
        tau_lo = max(tau_lo, np.max(-h[pos_mask] / d[pos_mask]))
    if np.any(neg_mask):
        tau_hi = min(tau_hi, np.min(-h[neg_mask] / d[neg_mask]))
        tau_lo = max(tau_lo, np.max((1.0 - h[neg_mask]) / d[neg_mask]))
    if not np.isfinite(tau_hi) or tau_hi - tau_lo < 1e-14:
        return None
    return (tau_lo, tau_hi)

def line_optimize(h, d):
    fr = find_feasible_range(h, d)
    if fr is None:
        return h, exact_score(h)
    tau_lo, tau_hi = fr
    eps = 0.1
    prof_0 = get_profile(h)
    prof_p = get_profile(h + eps * d)
    prof_m = get_profile(h - eps * d)
    a = (prof_p - 2.0 * prof_0 + prof_m) / (2.0 * eps * eps)
    b = (prof_p - prof_m) / (2.0 * eps)
    c = prof_0
    f0 = c.max()
    n_grid = 48
    taus = np.linspace(tau_lo, tau_hi, n_grid)
    quad = a[None, :] * taus[:, None] ** 2 + b[None, :] * taus[:, None] + c[None, :]
    vals = quad.max(axis=1)
    idx = int(np.argmin(vals))
    if vals[idx] >= f0 - 1e-12:
        return h, f0
    left = max(tau_lo, taus[max(0, idx - 2)])
    right = min(tau_hi, taus[min(n_grid - 1, idx + 2)])
    for _ in range(28):
        if right - left < 1e-10:
            break
        m1 = left + (right - left) * 0.381966
        m2 = left + (right - left) * 0.618034
        f1 = np.max(a * m1 * m1 + b * m1 + c)
        f2 = np.max(a * m2 * m2 + b * m2 + c)
        if f1 <= f2:
            right = m2
        else:
            left = m1
    best_tau = 0.5 * (left + right)
    f_best = float(np.max(a * best_tau * best_tau + b * best_tau + c))
    if f_best < f0 - 1e-12:
        new_h = repair_feasible(h + best_tau * d)
        actual = exact_score(new_h)
        if actual < f0 - 1e-12:
            return new_h, actual
    return h, f0

def gradient_direction(h, n_top=8):
    n = len(h)
    dx = 2.0 / n
    prof = get_profile(h)
    amax = prof.max()
    w = np.exp((prof - amax) / max(amax * 0.05, 1e-6))
    w /= w.sum()
    active_idx = np.argsort(prof)[::-1][:n_top]
    grad = np.zeros(n)
    for k in active_idx:
        delta = k - (n - 1)
        wk = w[k]
        if delta == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif 0 < delta < n:
            grad[delta:] += wk * dx * (1.0 - h[:n - delta])
            grad[:n - delta] -= wk * dx * h[delta:]
        else:
            dd = -delta
            if dd < n:
                grad[:n - dd] += wk * dx * (1.0 - h[dd:])
                grad[dd:] -= wk * dx * h[:n - dd]
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    return -grad / norm

def random_pair_direction(h, rng):
    n = len(h)
    for _ in range(3):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            d = np.zeros(n)
            if h[i] > h[j]:
                d[i] = -1.0
                d[j] = 1.0
            else:
                d[i] = 1.0
                d[j] = -1.0
            return d
    return None

def block_direction(h, rng):
    n = len(h)
    for _ in range(3):
        bs = rng.randint(2, max(3, n // 8))
        i = rng.randint(0, n - bs)
        j = rng.randint(0, n - bs)
        if abs(i - j) < bs:
            continue
        d = np.zeros(n)
        d[i:i + bs] = -1.0 / bs
        d[j:j + bs] = 1.0 / bs
        if h[i:i + bs].mean() < h[j:j + bs].mean():
            d = -d
        return d
    return None

def lowfreq_direction(h, rng):
    n = len(h)
    freq = rng.randint(1, 4)
    phase = rng.uniform(0, 2 * np.pi)
    d = np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-12:
        return None
    return d / norm

def variance_direction(h, rng):
    d = h - h.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-12:
        return None
    sign = 1.0 if rng.random() < 0.5 else -1.0
    return sign * d / norm

def solve_epigraph(h_start, t_start, maxiter=60):
    n = len(h_start)
    M = 2 * n - 1
    dx = 2.0 / n

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * dx
        return np.concatenate([[eq], t - prof])

    def cons_jac(x):
        h = x[:n]
        J = np.zeros((1 + M, n + 1))
        J[0, :n] = 1.0
        for k in range(M):
            delta = k - (n - 1)
            if delta == 0:
                J[1 + k, :n] = -dx * (1.0 - 2.0 * h)
            elif 0 < delta < n:
                J[1 + k, delta:n] = -dx * (1.0 - h[:n - delta])
                J[1 + k, :n - delta] += dx * h[delta:n]
            else:
                dd = -delta
                if dd < n:
                    J[1 + k, :n - dd] = -dx * (1.0 - h[dd:])
                    J[1 + k, dd:n] += dx * h[:n - dd]
            J[1 + k, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 0.5)]
    x0 = np.concatenate([h_start, [t_start]])
    try:
        result = minimize(
            obj, x0, jac=obj_jac, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return repair_feasible(np.clip(np.interp(x_tgt, x_ext, f_ext), 0.0, 1.0))

def make_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def make_stripe_binary(n, rng):
    h = np.zeros(n)
    period = rng.randint(4, max(5, n // 15))
    fill = max(1, period // 2)
    for start in range(0, n, period):
        h[start:start + fill] = 1.0
    ones = np.sum(h)
    if ones > n // 2:
        idx = np.where(h == 1.0)[0]
        extra = ones - n // 2
        h[idx[:extra]] = 0.0
    elif ones < n // 2:
        idx = np.where(h == 0.0)[0]
        missing = n // 2 - ones
        h[idx[:missing]] = 1.0
    return h
