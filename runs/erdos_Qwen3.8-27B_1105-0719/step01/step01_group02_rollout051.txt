        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def flip_delta(b, i, direction):
    """
    Compute the change in each raw profile entry when b[i] changes by `direction`.
    direction = -1 means b[i] goes from 1 to 0.
    direction = +1 means b[i] goes from 0 to 1.
    Returns a vector of length 2n-1 with the changes.
    """
    n = len(b)
    delta = np.zeros(2 * n - 1)
    bi = b[i]
    
    # For k >= 0 (output index t = n-1+k, k = 0, ..., n-1):
    # p[t] = sum_{m=0}^{n-1-k} b[m+k] * (1-b[m])
    # Change at position i:
    #   As "a" (b[m+k] where m+k=i, m=i-k): valid if 0<=i-k<=n-1-k, i.e., k<=i<=n-1
    #     contribution: direction * (1-b[i-k])
    #   As "b" (1-b[m] where m=i): valid if 0<=i<=n-1-k
    #     contribution: direction * (-b[i+k])
    
    # Vectorized for k in [0, n-1]:
    k = np.arange(n)
    # "a" role: valid where k <= i
    mask_a = k <= i
    if np.any(mask_a):
        # i-k ranges from i to i-k_max
        idx = i - k[mask_a]  # these are valid (0 <= i-k <= n-1-k since k<=i and i<=n-1)
        delta[n-1+k[mask_a]] += direction * (1 - b[idx])
    
    # "b" role: valid where i <= n-1-k, i.e., k <= n-1-i
    mask_b = k <= (n - 1 - i)
    if np.any(mask_b):
        idx = i + k[mask_b]
        delta[n-1+k[mask_b]] += direction * (-b[idx])
    
    # For k < 0 (d = -k, output index t = n-1+d, d = 1, ..., n-1):
    # p[t] = sum_{m=0}^{n-1-d} b[m] * (1-b[m+d])
    # Change at position i:
    #   As "a" (b[m] where m=i): valid if 0<=i<=n-1-d, i.e., d <= n-1-i
    #     contribution: direction * (1-b[i+d])
    #   As "b" (1-b[m+d] where m+d=i, m=i-d): valid if 0<=i-d<=n-1-d, i.e., d<=i<=n-1
    #     contribution: direction * (-b[i-d])
    
    d = np.arange(1, n)
    # "a" role: valid where d <= n-1-i
    mask_a = d <= (n - 1 - i)
    if np.any(mask_a):
        idx = i + d[mask_a]
        delta[n-1 - d[mask_a]] += direction * (1 - b[idx])
    
    # "b" role: valid where d <= i
    mask_b = d <= i
    if np.any(mask_b):
        idx = i - d[mask_b]
        delta[n-1 - d[mask_b]] += direction * (-b[idx])
    
    return delta
