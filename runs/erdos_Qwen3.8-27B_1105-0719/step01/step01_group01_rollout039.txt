        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    return (m + np.log(np.sum(exps)) / beta) * dx


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    w = exps / np.sum(exps) * dx
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        wi = w[i]
        if wi < 1e-14:
            continue
        a = max(0, -k)
        b = min(n - 1, n - 1 - k)
        if a <= b:
            grad[a:b + 1] += wi * (1.0 - h[a + k:b + 1 + k])
        a2 = max(0, k)
        b2 = n - 1
        if a2 <= b2:
            grad[a2:b2 + 1] -= wi * h[a2 - k:b2 + 1 - k]
    return grad


def eq_fun(h):
    return np.sum(h) - len(h) / 2.0


def eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, betas, max_iter=300, ftol=1e-13):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for beta in betas:
        try:
            res = minimize(smooth_obj, best_h, args=(beta,),
                           jac=smooth_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': ftol})
            if res.x is not None:
                c5_new = compute_c5(res.x)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = res.x.copy()
        except Exception:
            pass
    return best_h, best_c5


def full_swap_search(h, max_passes=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if abs(best_h[i] - best_h[j]) < 1e-10:
                    continue
                h_try = best_h.copy()
                h_try[i], h_try[j] = best_h[j], best_h[i]
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def push_pull_search(h, n_iterations, rng, delta=0.05):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iterations):
        i = rng.randint(n)
        if best_h[i] >= 1.0 - 1e-8 or best_h[i] <= 1e-8:
            continue
        j = rng.randint(n)
        if i == j:
            continue
        if best_h[i] > 0.5:
            d = min(delta, best_h[i] - 1e-8)
            max_up = 1.0 - 1e-8 - best_h[j]
            if max_up <= 0:
                continue
            d = min(d, max_up)
            h_try = best_h.copy()
            h_try[i] -= d
            h_try[j] += d
        else:
            d = min(delta, 1.0 - 1e-8 - best_h[i])
            max_down = best_h[j] - 1e-8
            if max_down <= 0:
                continue
            d = min(d, max_down)
            h_try = best_h.copy()
            h_try[i] += d
            h_try[j] -= d
        if d < 1e-10:
            continue
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5


def randomized_swap_search(h, n_iterations, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iterations):
        i, j = rng.choice(n, size=2, replace=False)
        if abs(best_h[i] - best_h[j]) < 1e-10:
            continue
        h_try = best_h.copy()
        h_try[i], h_try[j] = best_h[j], best_h[i]
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(60, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing beta
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_opt, c5_opt = slsqp_optimize(best_h, [10, 50, 100, 500, 2000, 10000], max_iter=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Full swap search (cyclic, exact)
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        n_passes = min(100, int(rem / 0.5))
        h_opt, c5_opt = full_swap_search(best_h, max_passes=n_passes)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_opt, c5_opt = slsqp_optimize(best_h, [1000, 5000, 20000, 50000], max_iter=400, ftol=1e-14)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: More full swap search
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        n_passes = min(100, int(rem / 0.5))
        h_opt, c5_opt = full_swap_search(best_h, max_passes=n_passes)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Push-pull search
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        n_iter = min(50000, int(rem * 60))
        h_opt, c5_opt = push_pull_search(best_h, n_iter // 2, rng, delta=0.05)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        h_opt, c5_opt = push_pull_search(best_h, n_iter // 2, rng, delta=0.02)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: SLSQP again
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_opt, c5_opt = slsqp_optimize(best_h, [5000, 20000, 100000], max_iter=500, ftol=1e-14)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Full swap search again
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        n_passes = min(100, int(rem / 0.5))
        h_opt, c5_opt = full_swap_search(best_h, max_passes=n_passes)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 8: Random restarts
    rem = budget_s - (time.time() - t0)
    n_restarts = min(40, int(rem / 10))
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 15:
            break
        strategy = i % 7
        if strategy == 0:
            h_pert = best_h + 0.04 * rng.randn(n)
        elif strategy == 1:
            h_pert = best_h.copy()
            idx = rng.choice(n, size=3, replace=False)
            for k in idx:
                h_pert[k] = 1.0 - h_pert[k]
        elif strategy == 2:
            shift = rng.randint(1, n)
            h_pert = np.roll(best_h, shift)
        elif strategy == 3:
            h_pert = best_h + 0.1 * rng.randn(n)
        elif strategy == 4:
            h_pert = best_h.copy()
            mask = rng.random(n) < 0.1
            h_pert[mask] = rng.uniform(0, 1, int(mask.sum()))
        elif strategy == 5:
            h_pert = best_h + 0.01 * rng.randn(n)
        else:
            h_pert = best_h.copy()
            idx = rng.choice(n, size=5, replace=False)
            for k in idx:
                h_pert[k] = 1.0 - h_pert[k]

        h_pert = project_feasible(h_pert)
        h_opt, c5_opt = slsqp_optimize(h_pert, [50, 500, 5000], max_iter=250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

        rem2 = budget_s - (time.time() - t0)
        if rem2 > 10:
            h_opt2, c5_opt2 = full_swap_search(h_opt, max_passes=min(50, int(rem2 / 0.5)))
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    # Phase 9: Try other resolutions
    rem = budget_s - (time.time() - t0)
    for n_try in [54, 66, 72, 48]:
        if time.time() - t0 > budget_s - 40:
            break
        cur_n = len(best_h)
        x_old = np.linspace(0, 2, cur_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp)
        h_opt, c5_opt = slsqp_optimize(h_interp, [30, 300, 3000, 10000], max_iter=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        rem2 = budget_s - (time.time() - t0)
        if rem2 > 10:
            h_opt2, c5_opt2 = full_swap_search(h_opt, max_passes=min(50, int(rem2 / 0.5)))
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    # Phase 10: Final SLSQP + search
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h_opt, c5_opt = slsqp_optimize(best_h, [10000, 50000, 100000, 200000], max_iter=500, ftol=1e-15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 11: Final full swap search
    rem = budget_s - (time.time() - t0)
    if rem > 3:
        n_passes = max(1, min(200, int(rem / 0.3)))
        h_opt, c5_opt = full_swap_search(best_h, max_passes=n_passes)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 12: Final push-pull polish
    rem = budget_s - (time.time() - t0)
    if rem > 3:
        n_iter = max(1000, min(100000, int(rem * 100)))
        h_opt, c5_opt = push_pull_search(best_h, n_iter, rng, delta=0.01)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
