    n = len(b)
    b = b.copy()
    profile = profile.copy()
    c5 = float(np.max(profile))
    ones = np.where(b > 0.5)[0]
    zeros = np.where(b < 0.5)[0]
    
    if rng is None:
        rng = np.random.RandomState(42)

        if deadline is not None and time.time() > deadline:
            break
        
        improved = False
        n_z = len(zeros)
        n_o = len(ones)
        
        # Try all zero-one pairs
        for a in range(n_z):
            i = zeros[a]
            # Precompute the i-part of the delta
            i_part = A[i] - B[i]  # This is (1-b[i-k]) - b[i+k]
            for bb in range(n_o):
                j = ones[bb]
                delta = (i_part + B[j] - A[j]) * dx
                new_c5 = (profile + delta).max()
                if new_c5 < c5 - 1e-14:
                    # Apply swap
                    b[i] = 1.0
                    b[j] = 0.0
                    profile += delta
                    c5 = new_c5
                    # Update precomputed terms
                    # A and B change for positions i and j
                    # Actually, A and B depend on b, so I need to update them
                    # A[i] changes because b[i] changed
                    # A[j] changes because b[j] changed
                    # B[i] changes because b[i] changed
                    # B[j] changes because b[j] changed
                    # Also, A[p] and B[p] for other p might change if they reference i or j
                    # This is expensive... let me just recompute A and B.
                    # For n=70, recomputing A and B takes O(n^2) = 4900 operations, which is fast.
                    A, B, _ = precompute_swap_terms(b, dx)
                    ones = np.where(b > 0.5)[0]
                    zeros = np.where(b < 0.5)[0]
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # No improving swap found; try simulated annealing
            found = False
            for _ in range(300):
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
                delta = (A[i] - B[i] + B[j] - A[j]) * dx
                new_c5 = (profile + delta).max()
                if new_c5 < c5 + 0.005:

                    b[j] = 0.0

                    found = True
                    break
            if not found:

def precompute_swap_terms(b, dx):
    n = len(b)
    k_vals = np.arange(-(n-1), n)  # (2n-1,)
    # For A: A[p, k] = 1-b[p+k_val[k]] if 0<=p+k_val[k]<n
    # p ranges 0..n-1, k ranges 0..2n-2
    # p+k_val[k] = p + (k - (n-1)) = p + k - n + 1
    # We need 0 <= p + k - n + 1 < n, i.e., n-1-p <= k <= 2n-2-p
    p_idx = np.arange(n)[:, None]  # (n, 1)
    k_idx = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    val = p_idx + k_idx - (n-1)  # (n, 2n-1), this is p + k_val
    mask = (val >= 0) & (val < n)  # (n, 2n-1)
    A = np.zeros((n, 2*n-1))
    B = np.zeros((n, 2*n-1))
    A[mask] = 1.0 - b[val[mask]]
    B[mask] = b[val[mask]]
    return A, B
