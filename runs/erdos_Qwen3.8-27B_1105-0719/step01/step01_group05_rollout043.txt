        idx = j - lag

import numpy as np

def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(prof)) * dx

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            c = int(np.sum(mask))
            if c == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / c, 1.0)
        else:
            mask = h > 0.0
            c = int(np.sum(mask))
            if c == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / c, 0.0)
    return h

    dx = 2.0 / n
    h = project_box_sum(h, 0.5 * n).copy()
    lags = np.arange(-(n - 1), n)
    rows = np.arange(n)[:, None]
    ip = rows + lags[None, :]
    im = rows - lags[None, :]
    vp = (ip >= 0) & (ip < n)
    vm = (im >= 0) & (im < n)
    ipc = np.clip(ip, 0, n - 1)
    imc = np.clip(im, 0, n - 1)
    G = np.where(vp, 1.0 - h[ipc], 0.0) - np.where(vm, h[imc], 0.0)
    prof = np.correlate(h, 1.0 - h, mode='full')
    cur = float(np.max(prof)) * dx
    n21 = 2 * n - 1
    for rnd in range(max_rounds):
        if deadline and time.time() > deadline - 5:
            break
        best_c = cur
        best_mv = None
        for i in range(n):
            hij = h[i]
            for j in range(n):
                if i == j:
                    continue
                hjj = h[j]
                L = max(-hij, hjj - 1.0)
                U = min(1.0 - hij, hjj)
                if U - L < 1e-13:
                    continue
                b = G[i] - G[j]
                lo, hi = L, U
                cand_lams = set()
                cand_lams.add(L)
                cand_lams.add(U)
                cand_lams.add((L + U) * 0.5)
                for _ in range(50):
                    mid = (lo + hi) * 0.5
                    cand_lams.add(mid)
                    vals = prof + b * mid
                    mv = float(np.max(vals))
                    active = vals >= mv - 1e-13
                    ba = b[active]
                    if ba.size == 0:
                        break
                    mn_b = float(np.min(ba))
                    mx_b = float(np.max(ba))
                    if mn_b > 1e-13:
                        hi = mid
                    elif mx_b < -1e-13:
                        lo = mid
                    else:
                        break
                best_lam = L
                best_f = float(np.max(prof + b * L))
                for cl in cand_lams:
                    cl_c = max(L, min(U, cl))
                    f = float(np.max(prof + b * cl_c))
                    if f < best_f - 1e-15:
                        best_f = f
                        best_lam = cl_c
                fn = best_f * dx
                if fn < best_c - tol:
                    best_c = fn
                    best_mv = (i, j, best_lam)
        if best_mv is not None:
            i, j, lam = best_mv
            b = G[i] - G[j]
            old_hi, old_hj = h[i], h[j]
            h[i] = old_hi + lam
            h[j] = old_hj - lam
            prof = prof + lam * b
            kp_arr = i - np.arange(n) + (n - 1)
            km_arr = np.arange(n) - i + (n - 1)
            vp2 = (kp_arr >= 0) & (kp_arr < n21)
            vm2 = (km_arr >= 0) & (km_arr < n21)
            G[np.where(vp2)[0], kp_arr[vp2]] -= lam
            G[np.where(vm2)[0], km_arr[vm2]] -= lam
            kp_arr = j - np.arange(n) + (n - 1)
            km_arr = np.arange(n) - j + (n - 1)
            vp3 = (kp_arr >= 0) & (kp_arr < n21)
            vm3 = (km_arr >= 0) & (km_arr < n21)
            G[np.where(vp3)[0], kp_arr[vp3]] += lam
            G[np.where(vm3)[0], km_arr[vm3]] += lam
            cur = best_c
        else:
            break
    return h, cur

def make_starts(n, rng, n_starts=8):
    target = 0.5 * n
    starts = []
    h = np.zeros(n)
    h[::2] = 1.0
    starts.append(project_box_sum(h, target))
    h = np.zeros(n)
    h[:int(target)] = 1.0
    starts.append(project_box_sum(h, target))
    h = np.zeros(n)
    idx = rng.choice(n, size=int(round(target)), replace=False)
    h[idx] = 1.0
    starts.append(project_box_sum(h, target))
    x = np.arange(n, dtype=np.float64)
    for freq in [2, 3]:
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
        starts.append(project_box_sum(h, target))
    h = np.full(n, 0.5)
    h[:] += rng.randn(n) * 0.2
    starts.append(project_box_sum(h, target))
    h = np.zeros(n)
    q = n // 4
    h[:q] = 1.0
    h[2 * n // 3:2 * n // 3 + q] = 1.0
    starts.append(project_box_sum(h, target))
    h = np.zeros(n)
    third = n // 3
    h[:third] = 1.0
    h[third * 2:n] = 1.0
    starts.append(project_box_sum(h, target))
    return starts

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0 = exact_c5(init)
            best_h = init.copy()
            best_c = c0
    except Exception:
        pass

    if best_h is None:
        best_h = np.zeros(70)
        best_h[::2] = 1.0
        best_c = exact_c5(best_h)

    solvers = ['ECOS', 'GLPK', 'PDLP', 'MOSEK', 'SCS']

    # Phase 1: LP-based binary search on multiple n values
    n_list = [60, 64, 68, 70, 72, 76, 80]
    n0_init = len(best_h)
    n_list.sort(key=lambda x: abs(x - n0_init))

    lp_data_cache = {}
    for n_try in n_list:
        if time.time() > deadline - 60:
            break
        if n_try not in lp_data_cache:
            lp_data_cache[n_try] = build_lp_data(n_try)
        data = lp_data_cache[n_try]

        lo_T = 0.378
        hi_T = best_c + 0.001
        best_h_n = None
        best_c_n = hi_T

        for it in range(28):
            if time.time() > deadline - 40:
                break
            mid_T = 0.5 * (lo_T + hi_T)
            h_sol = solve_lp(n_try, mid_T, data, solvers)
            if h_sol is not None:
                h_sol = project_box_sum(h_sol, 0.5 * n_try)
                c_val = exact_c5(h_sol)
                if c_val <= mid_T + 0.005:
                    hi_T = mid_T
                    best_c_n = min(best_c_n, c_val)
                    best_h_n = h_sol.copy()
                else:
                    hi_T = min(hi_T, c_val)
                    best_h_n = h_sol.copy()
                    best_c_n = c_val
            else:
                lo_T = mid_T

        if best_h_n is not None and best_c_n < best_c:
            best_c = best_c_n
            best_h = best_h_n.copy()
            best_n = n_try

    # Phase 2: Pair transfer refinement on best solution
    if time.time() < deadline - 30:
        h_ref, c_ref = pair_optimize(best_h, max_rounds=300, tol=1e-11, deadline=deadline)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref

    # Phase 3: Multi-start pair transfer at best n and nearby sizes
    best_n = len(best_h)
    sizes_to_try = sorted(set([best_n] + [s for s in [56, 58, 60, 62, 64, 66, 68, 70, 72, 74, 76, 78, 80] if abs(s - best_n) <= 12]))

    for n_try in sizes_to_try:
        if time.time() > deadline - 20:
            break
        starts = make_starts(n_try, rng)
        if best_h is not None and len(best_h) != n_try:
            starts.append(downsample(best_h, n_try))

        for hs in starts:
            if time.time() > deadline - 10:
                break
            h_opt, c_opt = pair_optimize(hs, max_rounds=200, tol=1e-11, deadline=deadline)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Perturbation + pair transfer
    if time.time() < deadline - 15:
        n_best = len(best_h)
        for pert_scale in [0.05, 0.1, 0.15, 0.2]:
            if time.time() > deadline - 8:
                break
            h_pert = best_h + rng.randn(n_best) * pert_scale
            h_pert = project_box_sum(h_pert, 0.5 * n_best)
            h_opt, c_opt = pair_optimize(h_pert, max_rounds=200, tol=1e-11, deadline=deadline)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final LP squeeze at best n
    if time.time() < deadline - 10:
        n_best = len(best_h)
        if n_best in lp_data_cache:
            data = lp_data_cache[n_best]
        else:
            lp_data_cache[n_best] = build_lp_data(n_best)
            data = lp_data_cache[n_best]
        h_fin = solve_lp(n_best, best_c + 0.0001, data, solvers)
        if h_fin is not None:
            h_fin = project_box_sum(h_fin, 0.5 * n_best)
            c_fin = exact_c5(h_fin)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin
                # Pair refine the LP result
                if time.time() < deadline - 5:
                    h_r, c_r = pair_optimize(h_fin, max_rounds=200, tol=1e-11, deadline=deadline)
                    if c_r < best_c:
                        best_c = c_r
                        best_h = h_r

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c = exact_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    return (best_h, best_c, len(best_h))
