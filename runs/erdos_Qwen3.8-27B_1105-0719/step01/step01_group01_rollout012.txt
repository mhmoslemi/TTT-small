import time

def compute_all_grads(h):
    """Compute gradient of each correlation term.
    Returns array of shape (2n-1, n) where row m corresponds to lag m-(n-1).
    For memory efficiency with n up to ~80, this is (159, 80) = ~12720 floats. Fine.
    """
    n = len(h)
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for k in range(-(n-1), n):
        m = k + n - 1  # index in full correlation
        i_lo = max(0, -k)
        i_hi = min(n-1, n-1-k)
        i_range = range(i_lo, i_hi+1)
        h_i = h[i_lo:i_hi+1]
        h_j = h[i_lo+k:i_hi+1+k]
        # grad w.r.t. h[i] for i in [i_lo, i_hi]: (1 - h[i+k])
        grads[m, i_lo:i_hi+1] += (1.0 - h_j)
        # grad w.r.t. h[i+k] for i in [i_lo, i_hi]: -h[i]
        # indices: i_lo+k to i_hi+k
        grads[m, i_lo+k:i_hi+1+k] += -h_i
    return grads

def project_to_feasible(y, n):
    """Project y onto {h : 0<=h[i]<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on threshold
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute among free variables
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.sum(free) > 0:
            h[free] += residual / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            # All at boundaries - use smallest adjustment
            if residual > 0:
                # Need to increase some
                order = np.argsort(h)  # smallest first (closest to 0)
                for idx in order:

                    h[idx] += add
                    residual -= add
                    if residual <= 1e-12:
                        break
            else:

                for idx in order:
                    sub = min(-residual, h[idx])
                    h[idx] -= sub
                    residual += sub
                    if residual >= -1e-12:
                        break
    return h

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

def slsqp_step(h0, beta, max_iter=200, n_max=None):
    """One SLSQP optimization step with smooth max objective."""
    n = len(h0)
    dx = 2.0 / n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        v, _ = smooth_max(corr, beta)
        return v * dx
    
    def grad(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        _, w = smooth_max(corr, beta)
        # Compute weighted gradient
        g = np.zeros(n)
        for m in range(len(corr)):
            if w[m] < 1e-14:

            k = m - (n - 1)
            i_lo = max(0, -k)
            i_hi = min(n-1, n-1-k)
            h_i = h[i_lo:i_hi+1]
            h_j = h[i_lo+k:i_hi+1+k]
            g[i_lo:i_hi+1] += w[m] * (1.0 - h_j)
            g[i_lo+k:i_hi+1+k] += -w[m] * h_i
        return g * dx
    
    def eq_fun(h):
        return np.sum(h) - n / 2.0
    
    def eq_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return res.x

            h_i = h[i_lo:i_hi+1]
            h_j = h[i_lo+k:i_hi+1+k]
            g[i_lo:i_hi+1] += (1.0 - h_j)
            g[i_lo+k:i_hi+1+k] += -h_i
        g /= len(active)
        
        # Projected step
        h_new = project_to_feasible(h - alpha * g, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        alpha *= decay
    
    return best_h, best_c5

def active_lag_local_search(h, n_iter=20000, rng=None):
    """Local search that targets active lags."""
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    
    for _ in range(n_iter):
        corr = compute_corr_full(best_h)
        c_max = np.max(corr)
        
        # Pick an active lag
        active = np.where(corr > c_max - 0.1)[0]
        m = active[rng.randint(len(active))]
        k = m - (n-1)
        
        # Find worst pair for this lag
        i_lo = max(0, -k)
        i_hi = min(n-1, n-1-k)
        if i_lo > i_hi:
            continue
        
        # Contributions: h[i] * (1-h[i+k]) for i in [i_lo, i_hi]
        contribs = best_h[i_lo:i_hi+1] * (1.0 - best_h[i_lo+k:i_hi+1+k])
        # Pick the worst (largest) contribution
        worst_local = np.argmax(contribs)
        i = i_lo + worst_local
        j = i + k
        
        # Propose: transfer from i to j (reduce h[i], increase h[j])
        # This reduces the specific term h[i](1-h[j])
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer < 1e-10:
            # Try reverse
            i, j = j, i
            max_transfer = min(best_h[i], 1.0 - best_h[j])
            if max_transfer < 1e-10:
                continue
        
        # Try different transfer amounts
        delta = max_transfer * rng.uniform(0.3, 1.0)
        h_new = best_h.copy()
        if best_h[i] >= best_h[j]:
            h_new[i] = best_h[i] - delta
            h_new[j] = best_h[j] + delta
        else:
            h_new[i] = best_h[i] + delta
            h_new[j] = best_h[j] - delta
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5

def random_swap_search(h, n_iter=10000, rng=None):

def interpolate_to_n(h_old, n_new):
    """Interpolate step function from n_old to n_new points."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 2.0/(2*n_new)  # centers
    # Actually, use edges for step function interpolation
    # Old: each h[i] covers [2*i/n_old, 2*(i+1)/n_old)
    # New: each h_new[j] covers [2*j/n_new, 2*(j+1)/n_new)
    # For each new cell, compute the weighted average of old cells that overlap
    
    n_new_arr = np.zeros(n_new)
    for j in range(n_new):
        x_start = 2.0 * j / n_new
        x_end = 2.0 * (j+1) / n_new
        # Find overlapping old cells
        i_start = max(0, int(x_start * n_old / 2.0))
        i_end = min(n_old - 1, int((x_end - 1e-15) * n_old / 2.0))
        total_weight = 0
        total_val = 0
        for i in range(i_start, i_end+1):
            old_start = 2.0 * i / n_old
            old_end = 2.0 * (i+1) / n_old
            overlap = min(x_end, old_end) - max(x_start, old_start)
            if overlap > 0:
                total_weight += overlap
                total_val += h_old[i] * overlap
        if total_weight > 0:
            n_new_arr[j] = total_val / total_weight
        else:
            n_new_arr[j] = 0.5
    return project_to_feasible(n_new_arr, n_new)

import numpy as np

import time

# Module-level state for constraint functions (avoids closures)
_state_n = 60

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

def _slsqp_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v, _ = smooth_max(corr, beta)
    return v * dx

def _slsqp_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, w = smooth_max(corr, beta)
    g = np.zeros(n)
    for m in range(len(corr)):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        i_lo = max(0, -k)

        g[i_lo + k: i_hi + 1 + k] += -w[m] * h_i

    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

                residual -= add

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h_bin = h_new
            best_c5 = c5_new
    return best_h_bin, best_c5

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    n_new_arr = np.zeros(n_new)
    for j in range(n_new):
        x_start = 2.0 * j / n_new
        x_end = 2.0 * (j + 1) / n_new
        i_start = max(0, int(x_start * n_old / 2.0))
        i_end = min(n_old - 1, int((x_end - 1e-15) * n_old / 2.0))
        total_w = 0.0
        total_v = 0.0
        for i in range(i_start, i_end + 1):
            old_start = 2.0 * i / n_old
            old_end = 2.0 * (i + 1) / n_old
            overlap = min(x_end, old_end) - max(x_start, old_start)
            if overlap > 0:
                total_w += overlap
                total_v += h_old[i] * overlap
        n_new_arr[j] = total_v / total_w if total_w > 0 else 0.5
    return project_feasible(n_new_arr, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(60, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with increasing beta
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, [15, 80, 400, 2000], max_iter=180)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Subgradient descent
    if rem() > 25:
        h_opt, c5_opt = subgradient_descent(best_h, 400, 0.004, 0.985, tol_active=0.06)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Active lag local search
    if rem() > 15:
        n_iter = min(40000, int(rem() * 35))
        h_opt, c5_opt = active_lag_local_search(best_h, n_iter, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: SLSQP high-beta polish
    if rem() > 20:
        h_opt, c5_opt = slsqp_optimize(best_h, [3000, 20000], max_iter=250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Binary search for good structure
    if rem() > 30:
        target_n = n // 2
        h_bin = np.zeros(n)
        sorted_idx = np.argsort(-best_h)
        h_bin[sorted_idx[:target_n]] = 1.0
        h_bin_c5 = compute_c5(h_bin)
        h_bin_opt, h_bin_c5 = binary_local_search(h_bin, 15000, rng)
        if h_bin_c5 < best_c5:
            best_c5 = h_bin_c5
            best_h = h_bin_opt
        # Blend binary with continuous
        if rem() > 20:
            h_blend = 0.5 * best_h + 0.5 * h_bin_opt
            h_blend = project_feasible(h_blend, n)
            h_blend_opt, c5_blend = slsqp_optimize(h_blend, [100, 1000], max_iter=150)
            if c5_blend < best_c5:
                best_c5 = c5_blend
                best_h = h_blend_opt

    # Phase 6: More active search
    if rem() > 15:
        n_iter = min(50000, int(rem() * 40))
        h_opt, c5_opt = active_lag_local_search(best_h, n_iter, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Resolution exploration
    if rem() > 70:
        for n_try in [30, 36, 42, 48, 54, 72, 84, 96]:
            if rem() < 25:
                break
            try:
                h_interp = interpolate_to_n(best_h, n_try)
                h_interp, c5_interp = slsqp_optimize(h_interp, [50, 500, 3000], max_iter=120)
                if rem() > 15:
                    h_interp, c5_interp = active_lag_local_search(h_interp, 8000, rng)
                if c5_interp < best_c5:
                    best_c5 = c5_interp
                    best_h = h_interp
                    n = n_try
            except Exception:
                pass

    # Phase 8: Final SLSQP polish
    if rem() > 12:
        n_f = len(best_h)
        h_opt, c5_opt = slsqp_optimize(best_h, [5000, 50000], max_iter=400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 9: Final local search
    if rem() > 5:
        n_f = len(best_h)
        n_iter = min(40000, int(rem() * 60))
        h_opt, c5_opt = active_lag_local_search(best_h, n_iter, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 10: One more subgradient pass on final
    if rem() > 5:
        h_opt, c5_opt = subgradient_descent(best_h, 200, 0.002, 0.99, tol_active=0.03)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final validation and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
