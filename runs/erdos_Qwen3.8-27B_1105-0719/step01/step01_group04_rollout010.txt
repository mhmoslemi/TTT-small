= (1/2)[∫h(x)h(x+k)dx + ∫h(x)h(x-k)dx]

import numpy as np
from scipy.optimize import minimize, brentq
from scipy.special import expit
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += min(diff2, 1.0 - h[i])
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += max(diff2, -h[i])
                break
    return h


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def compute_grads(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grads = np.zeros((L, n))
    for delta in range(0, n):
        if delta == 0:
            grads[n - 1] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - delta)
            grads[n - 1 + delta, idx] += dx * (1.0 - h[idx + delta])
            grads[n - 1 + delta, idx + delta] += -dx * h[idx]
            idx2 = np.arange(delta, n)
            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
            grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    return grads


def logit_project(z, n):
    target = n / 2.0
    def expr(c):
        return float(np.sum(expit(z + c)) - target)
    lo, hi = z.min() - 40.0, -z.min() + 40.0
    lo = min(lo, -40.0)
    hi = max(hi, 40.0)
    f_lo = expr(lo)
    f_hi = expr(hi)
    if f_lo * f_hi > 0:
        return 0.5 * np.ones(n)
    c = brentq(expr, lo, hi, xtol=1e-14, maxiter=200)
    return expit(z + c)


def softmax_obj_grad(z, n, beta):
    h = logit_project(z, n)
    h = np.clip(h, 1e-15, 1.0 - 1e-15)
    values = compute_profile(h)
    bv = beta * values
    bv_max = bv.max()
    exp_vals = np.exp(bv - bv_max)
    Z = exp_vals.sum()
    w = exp_vals / Z
    J_val = (bv_max + np.log(Z)) / beta
    grads = compute_grads(h)
    g_h = w @ grads
    sigp = h * (1.0 - h)
    T = sigp.sum()
    if T < 1e-10:
        g_z = g_h
    else:
        g_z = g_h - sigp * (sigp @ g_h) / T
    return J_val, g_z


def optimize_logit(h_init, beta_schedule, iters_per_beta=200, time_budget=300.0, t_start=0.0):
    n = len(h_init)
    h_clip = np.clip(h_init, 1e-8, 1.0 - 1e-8)
    z = np.log(h_clip / (1.0 - h_clip))
    z -= z.mean()
    best_h = h_init.copy()
    best_score = exact_score(h_init)
    for beta in beta_schedule:
        for it in range(iters_per_beta):
            if time.perf_counter() - t_start > time_budget:
                return best_h, best_score
            J_val, g_z = softmax_obj_grad(z, n, beta)
            if not np.isfinite(J_val) or not np.all(np.isfinite(g_z)):
                break
            lr = 0.5 / (beta * 0.05 + 1.0)
            lr = min(lr, 0.3)
            z_new = z - lr * g_z
            z_new = np.clip(z_new, -25.0, 25.0)
            z_new -= z_new.mean()
            h_new = logit_project(z_new, n)
            h_new = repair_feasible(h_new)
            s = exact_score(h_new)
            if s < best_score:

                best_h = h_new.copy()
                z = z_new
            else:

def active_descent(h, max_steps=80, time_budget=120.0, t_start=0.0):

def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    dx_src = 2.0 / n
    dx_tgt = 2.0 / new_len
    cumsum = np.concatenate([[0.0], np.cumsum(vec)])
    result = np.zeros(new_len)
    for j in range(new_len):
        x_lo = j * dx_tgt
        x_hi = (j + 1) * dx_tgt
        i_lo = int(x_lo / dx_src)
        i_hi = int(np.ceil(x_hi / dx_src - 1e-12))
        i_lo = max(0, min(i_lo, n - 1))
        i_hi = max(0, min(i_hi, n))
        area = 0.0
        for i in range(i_lo, i_hi):
            cell_lo = i * dx_src
            cell_hi = (i + 1) * dx_src
            ov_lo = max(x_lo, cell_lo)
            ov_hi = min(x_hi, cell_hi)
            if ov_hi > ov_lo:
                area += vec[i] * (ov_hi - ov_lo)
        result[j] = area / dx_tgt
    return result


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def slsqp_solve(h_start, maxiter=150):
    n = len(h_start)
    dx = 2.0 / n
    L = 2 * n - 1
    h_start = repair_feasible(h_start)
    c_start = exact_score(h_start)
    t0 = c_start + 0.03

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def eq_fun(x):
        return x[:n].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        vals = compute_profile(h)
        return t - vals

    def ineq_jac(x):
        h = x[:n]
        g = np.zeros((L, n + 1))
        g[:, n] = 1.0
        g[:, :n] = -compute_grads(h)
        return g

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = minimize(
            obj, x0, jac=obj_jac, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
    except Exception:
        return None, None
    if result.x is not None and np.all(np.isfinite(result.x)):
        h_sol = repair_feasible(result.x[:n])
        score = exact_score(h_sol)
        if np.isfinite(score):
            return h_sol, score
    return None, None
