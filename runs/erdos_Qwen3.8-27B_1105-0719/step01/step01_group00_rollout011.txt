def project_feasible(x, target_sum):
    """Project x to feasible set: 0<=x<=1, sum(x)=target_sum"""
    x = np.clip(x, 0.0, 1.0)
    # Bisection to find shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            # Distribute diff using available headroom
            if diff > 0:
                headroom = 1.0 - h[mask]
                total = headroom.sum()
                if total > 0:
                    h[mask] += diff * headroom / total
            else:
                h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def get_lag_pairs(n, k):
    """Get valid (base, shifted) index pairs for lag k.
    k is the raw index from np.correlate (0 to 2n-2).
    offset = k - (n-1)
    """
    offset = k - (n - 1)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return np.array([]), np.array([])
    base = np.arange(i_lo, i_hi + 1)
    shifted = base + offset
    return base, shifted

    """Gradient of raw correlation k with respect to h."""
    base, shifted = get_lag_pairs(n, k)

        return np.zeros(n)
    grad = np.zeros(n)
    np.add.at(grad, base, -h[shifted])
    np.add.at(grad, shifted, 1.0 - h[base])
    return grad

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import cvxpy as cp
import time


def project_feasible(x, target_sum):
    """Project x onto {0<=x<=1, sum(x)=target_sum}."""
    x = np.clip(x, 0.0, 1.0).astype(float)
    n = len(x)
    # Bisection for the shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max()) * dx


def lag_grad(h, n, k):
    """Gradient of raw correlation at index k."""
    offset = k - (n - 1)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return np.zeros(n)
    base = np.arange(i_lo, i_hi + 1)
    shifted = base + offset
    grad = np.zeros(n)
    np.add.at(grad, base, -h[shifted])
    np.add.at(grad, shifted, 1.0 - h[base])
    return grad


def solve_qp_subproblem(h0, active_lags, dx, target_sum, alpha_h, rho, solver_list):
    """Solve the trust-region QP subproblem using cvxpy."""
    n = len(h0)
    corrs = compute_corrs(h0)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target_sum,
        h_var >= 0,
        h_var <= 1.0,
        h_var >= h0 - rho,
        h_var <= h0 + rho,
        t_var >= 0,
        t_var <= corrs.max() + 1.0,
    ]
    
    for k in active_lags:
        Gk = lag_grad(h0, n, k)
        R0 = corrs[k]
        constraints.append(t_var >= dx * (R0 + Gk @ (h_var - h0)))
    
    objective = cp.Minimize(t_var + alpha_h * cp.sum_squares(h_var - h0))
    prob = cp.Problem(objective, constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate') and h_var.value is not None:
                h_sol = np.asarray(h_var.value).flatten()
                if np.all(np.isfinite(h_sol)):
                    return h_sol
        except Exception:
            continue
    return None

def slsqp_smooth_max(h0, dx, target_sum, beta, maxiter):
    """SLSQP with smooth max objective."""
    n = len(h0)
    
    def obj_val(x):
        return logsumexp(beta * compute_corrs(x)) / beta * dx

    def obj_grad(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            Gk = lag_grad(x, n, k)
            grad += w[k] * Gk
        return grad * dx

    def constr_fun(x):
        return x.sum() - target_sum

    def constr_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_val, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
        return h_out, c5_of_h(h_out, dx)
    except Exception:
        return h0, c5_of_h(h0, dx)
