import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def evaluate_exact(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_target(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum."""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target_sum - h.sum()
    if abs(resid) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += resid / mask.sum()
    return np.clip(h, 0.0, 1.0)

def build_symh(q, n):
    """Build h from symmetric q. n=2m, len(q)=m."""
    m = len(q)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def q_from_h(h):
    """Extract q from complement-palindromic h."""
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    return np.clip(q, 0.0, 1.0)

def corr_at_lag(h, d):
    """Compute correlation at lag d (positive or negative).
    Positive lag d: sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
    Negative lag -d: sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
    """
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    ad = abs(d)
    if ad >= n:
        return 0.0
    if d > 0:
        return float(np.sum(h[ad:] * (1.0 - h[:n-ad])))
    else:
        return float(np.sum(h[:n-ad] * (1.0 - h[ad:])))

def corr_lags_from_eval_index(h, eval_indices):
    """Given evaluator indices (0 to 2n-2), return the correlation values.
    Evaluator index k corresponds to lag d = k - (n-1)."""
    n = len(h)
    corrs = np.zeros(len(eval_indices))
    for idx, k in enumerate(eval_indices):
        d = k - (n - 1)
        corrs[idx] = corr_at_lag(h, d)
    return corrs

def grad_corr_lag(h, d, n):
    """Gradient of corr_at_lag(h, d) w.r.t. h."""
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        hi = n - d
        grad[d:] += (1.0 - h[:hi])
        grad[:hi] -= h[d:]
    else:
        hi = n - (-d)
        grad[:hi] += (1.0 - h[-d:])
        grad[-d:] -= h[:hi]
    return grad

def grad_corr_lag_q(q, d, n):
    """Gradient of corr_at_lag(build_symh(q,n), d) w.r.t. q."""
    gh = grad_corr_lag(build_symh(q, n), d, n)
    m = len(q)
    gq = gh[:m] - gh[m:][::-1]
    return gq

def project_to_target(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

        if mask.sum() > 0:

    if d > 0:

    else:
        return float(np.sum(h[:n-ad] * (1.0 - h[ad:])))

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def evaluate_exact(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_to_target(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target_sum - h.sum()
    if abs(resid) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += resid / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_symh(q, n):
    m = len(q)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def q_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    return np.clip(q, 0.0, 1.0)


def corr_at_lag(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    ad = abs(d)
    if ad >= n:
        return 0.0
    if d > 0:
        return float(np.sum(h[ad:] * (1.0 - h[:n - ad])))
    else:
        return float(np.sum(h[:n - ad] * (1.0 - h[ad:])))


def grad_corr_lag_h(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        hi = n - d
        if hi > 0:
            grad[d:] += (1.0 - h[:hi])
            grad[:hi] -= h[d:]
    else:
        ad = -d
        hi = n - ad
        if hi > 0:
            grad[:hi] += (1.0 - h[ad:])
            grad[ad:] -= h[:hi]
    return grad


def select_active_set(h, n, top_k, pad=3):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    num_lags = 2 * n - 1
    top_k = min(top_k, num_lags)
    top_idx = np.argsort(corrs)[-top_k:][::-1]
    all_idx = set()
    for idx in top_idx:
        for p in range(-pad, pad + 1):
            ni = int(idx) + p
            if 0 <= ni < num_lags:
                all_idx.add(ni)
    return sorted(all_idx)


def smooth_value_grad_sym(q, n, dx, beta):
    h = build_symh(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= wsum
    val = float(logsumexp(beta * corrs) / beta)
    m = len(q)
    gq = np.zeros(m)
    num_lags = 2 * n - 1
    for idx in range(num_lags):
        if w[idx] < 1e-14:
            continue
        d = idx - (n - 1)
        gh = grad_corr_lag_h(h, d)
        gq += w[idx] * dx * (gh[:m] - gh[m:][::-1])
    return val, gq


def smooth_optimize_sym(q0, n, dx, beta, maxiter=400):
    m = len(q0)

    def _fun(x):
        return smooth_value_grad_sym(x, n, dx, beta)[0]

    def _grad(x):
        return smooth_value_grad_sym(x, n, dx, beta)[1]

    try:
        res = minimize(_fun, q0, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0.copy()


class EpigraphSym:
    def __init__(self, lag, n, dx):
        self.lag = lag
        self.n = n
        self.dx = dx

    def fun(self, x):
        m = self.n // 2
        q = x[:m]
        h = build_symh(q, self.n)
        return x[m] - self.dx * corr_at_lag(h, self.lag)

    def jac(self, x):
        n = self.n
        m = n // 2
        q = x[:m]
        h = build_symh(q, n)
        gh = grad_corr_lag_h(h, self.lag)
        gq = gh[:m] - gh[m:][::-1]
        j = np.zeros(m + 1)
        j[:m] = -self.dx * gq
        j[m] = 1.0
        return j


class EpigraphFull:
    def __init__(self, lag, n, dx):
        self.lag = lag
        self.n = n
        self.dx = dx

    def fun(self, x):
        h = x[:self.n]
        return x[self.n] - self.dx * corr_at_lag(h, self.lag)

    def jac(self, x):
        n = self.n
        h = x[:n]
        gh = grad_corr_lag_h(h, self.lag)
        j = np.zeros(n + 1)
        j[:n] = -self.dx * gh
        j[n] = 1.0
        return j


class SumEq:
    def __init__(self, target, n):
        self.target = target
        self.n = n

    def fun(self, x):
        return x[:self.n].sum() - self.target

    def jac(self, x):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j

    return x[-1]


def obj_last_grad(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g


def solve_epigraph_sym(q0, n, active_lags, dx, maxiter=300, ftol=1e-12):
    m = n // 2
    h = build_symh(q0, n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corrs) * dx) + 1e-7
    x0 = np.empty(m + 1)
    x0[:m] = q0
    x0[m] = t0

    constraints = [EpigraphSym(r - (n - 1), n, dx) for r in active_lags]
    bounds = [(0.0, 1.0)] * m + [(0.0, 3.0)]

    try:
        res = minimize(obj_last, x0, jac=obj_last_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            q_out = np.clip(res.x[:m], 0.0, 1.0)
            return q_out
    except Exception:
        pass
    return None


def solve_epigraph_full(h0, n, active_lags, dx, target, maxiter=300, ftol=1e-12):
    corrs = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = float(np.max(corrs) * dx) + 1e-7
    x0 = np.empty(n + 1)
    x0[:n] = h0
    x0[n] = t0

    constraints = [SumEq(target, n)]
    for r in active_lags:
        constraints.append(EpigraphFull(r - (n - 1), n, dx))

    bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]

    try:
        res = minimize(obj_last, x0, jac=obj_last_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_out = project_to_target(res.x[:n], target)
            return h_out
    except Exception:
        pass
    return None


def check_hidden_violations(q, n, dx, active_lags, tol=1e-10):
    h = build_symh(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    t_val = float(np.max(corrs))
    active_set = set(active_lags)
    hidden = []
    for r in range(2 * n - 1):
        if r not in active_set:
            if corrs[r] >= corrs[np.array(sorted(active_lags))].max() - 1e-12:
                hidden.append(r)
    return hidden, t_val

def coord_descent_sym(q, n, dx, active_lags, max_sweeps=10):

        Any_improve = False
        for j in range(m):
            lo_b, hi_b = 0.0, 1.0
            best_qj = q[j]

            grid = np.linspace(lo_b, hi_b, 201)

                q_trial = q.copy()
                q_trial[j] = qj_val

                cmax = float(np.max(c_trial))
                if cmax < best_obj - 1e-14:

                    best_qj = qj_val
            if best_qj != q[j]:
                q[j] = best_qj
                Any_improve = True
        if not Any_improve:
