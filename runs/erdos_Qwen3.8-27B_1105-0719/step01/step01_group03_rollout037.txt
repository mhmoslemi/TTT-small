from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def project_feasible(z, target):
    n = len(z)
    z = np.asarray(z, dtype=np.float64)
    lo, hi = -z.max() - 2.0, -z.min() + 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    a = (lo + hi) * 0.5
    h = np.clip(z + a, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_profile(h, n, dx, rows=None):
    M = 2 * n - 1
    if rows is None:
        rows = np.arange(M)
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[idx, d:] = dx * (1.0 - h[:n - d])
            G[idx, :n - d] = -dx * h[d:]
        else:
            s = -d
            G[idx, :n - s] = dx * (1.0 - h[s:])
            G[idx, s:] = -dx * h[:n - s]
    return G

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    p = profile(h)
    max_p = np.max(p)
    logits = (p - max_p) / tau
    exp_l = np.exp(logits)
    Z = exp_l.sum()
    weights = exp_l / Z
    obj = tau * np.log(Z) + max_p
    G = grad_profile(h, n, dx)
    grad = G.T @ weights
    return obj, grad

def nesterov_descent(h0, target, taus, steps_per_tau, alpha_init, time_limit, t_start):

    alpha = alpha_init

            if time.perf_counter() - t_start > time_limit:
                break

            grad = (grad + grad[::-1]) * 0.5
            grad -= grad.mean()
            norm_g = np.linalg.norm(grad)
            if norm_g < 1e-13:
                break

            a = alpha
            improved = False

                h_new = project_feasible(y - a * grad, target)
                obj_new, _ = smooth_obj_grad(h_new, tau)

                    improved = True
                    break
                a *= 0.5

def slsqp_polish(h0, target, maxiter=500):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def _obj(x):
        return x[-1]

    def _obj_g(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def _ceq(x):
        return x[:-1].sum() - target

    def _ceq_g(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def _cineq(x):
        h = x[:-1]
        t = x[-1]
        return t - np.correlate(h, 1 - h, mode="full") * dx

    def _cineq_g(x):
        h = x[:-1]
        G = grad_profile(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]

        {'type': 'eq', 'fun': _ceq, 'jac': _ceq_g},
        {'type': 'ineq', 'fun': _cineq, 'jac': _cineq_g}
    ]
