import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def compute_weighted_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    w0 = weights[n - 1]
    if abs(w0) > 1e-15:
        grad += w0 * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[n - 1 + d]
        if abs(wp) > 1e-15:
            idx = np.arange(n - d)
            grad[idx] += wp * dx * (1.0 - h[idx + d])
            grad[idx + d] -= wp * dx * h[idx]
        wn = weights[n - 1 - d]
        if abs(wn) > 1e-15:
            idx2 = np.arange(d, n)
            grad[idx2] += wn * dx * (1.0 - h[idx2 - d])
            grad[idx2 - d] -= wn * dx * h[idx2]
    return grad


def soft_obj_and_jac(h, beta):
    n = len(h)
    values = compute_profile(h)
    max_v = np.max(values)
    shifted = beta * (values - max_v)
    # Stable: clip to avoid underflow issues
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return max_v, np.zeros(n)
    logsumexp_val = max_v + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    grad = compute_weighted_grad(h, weights)
    return logsumexp_val, grad


def cdf_transfer(h, new_len):
    n = len(h)
    dx = 2.0 / n
    cumsum = np.cumsum(np.concatenate([[0.0], h])) * dx
    x_old = np.linspace(0.0, 2.0, n + 1)
    x_new = np.linspace(0.0, 2.0, new_len + 1)
    cumsum_new = np.interp(x_new, x_old, cumsum)
    h_new = np.diff(cumsum_new) / (2.0 / new_len)
    return repair_feasible(h_new)


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def soft_solve(h_start, beta, maxiter=150, ftol=1e-12):
    n = len(h_start)
    h_start = repair_feasible(h_start)

    def obj(x):
        val, _ = soft_obj_and_jac(x, beta)
        return val

    def jac(x):
        _, g = soft_obj_and_jac(x, beta)
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            obj, h_start, jac=jac,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x)
    except Exception:
        pass
    return None


def active_polish(h, max_steps=50, time_limit=None, t_start=None):
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    for _ in range(max_steps):
        if time_limit is not None and t_start is not None:
            if time.perf_counter() - t_start > time_limit - 10:
                break
        values = compute_profile(best)
        c = np.max(values)
        gaps = np.maximum(c - values, 0.0)
        sigma = max(0.002, 0.001 * c)
        w = np.exp(-gaps / sigma)
        w_sum = w.sum()
        if w_sum < 1e-10:
            break
        w = w / w_sum
        grad = compute_weighted_grad(best, w)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair_feasible(best - alpha * grad)
            s = exact_score(h_trial)
            if s < best_score - 1e-14:
                best = h_trial
                best_score = s
                improved = True
                break
        if not improved:
            break
    return best, best_score
