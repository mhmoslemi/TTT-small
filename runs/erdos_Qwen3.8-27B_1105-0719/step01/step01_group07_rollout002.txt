def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    # Bisection on lambda: find lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo = -h.max() - 2
    hi = 1 - h.min() + 2
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix residual
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_new < 1 - 1e-12
        else:
            mask = h_new > 1e-12
        if np.sum(mask) > 0:
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0, 1)
    return h_new
