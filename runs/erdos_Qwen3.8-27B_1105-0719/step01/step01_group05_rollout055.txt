def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = min(diff / count, 1.0 - h[mask]).max() if count > 0 else 0
            # Actually, just use the simple approach
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        # Distribute to the element with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _gradient_lag(x, lag, dx):
    n = len(x)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= x[lag:]
            grad[lag:] += 1.0 - x[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - x[a:]
            grad[a:] -= x[:s]
    return grad * dx

def slsqp_direct(h_start, deadline, max_iter=200, ftol=1e-14):
    """Direct SLSQP: minimize c s.t. c >= q_r(h) for all r, sum(h)=n/2, 0<=h<=1."""
    n = len(h_start)
    h_start = project_box_sum(h_start, 0.5 * n)
    c0, _ = exact_objective(h_start)
    target = 0.5 * n
    n_c = 2 * n - 1
    
    # Initial point
    z0 = np.concatenate([h_start, [c0 + 1e-4]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    def obj_fun(z):
        return z[n]
    
    def obj_jac(z):
        j = np.zeros(n + 1)
        j[n] = 1.0
        return j
    
    def eq_fun(z):
        return np.sum(z[:n]) - target
    
    def eq_jac(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    def ineq_fun(z):
        h = z[:n]
        c = z[n]
        dx = 2.0 / n
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        return c - profile
    
    def ineq_jac(z):
        h = z[:n]
        dx = 2.0 / n
        jac = np.zeros((n_c, n + 1))
        jac[:, n] = 1.0
        for r in range(n_c):
            lag = r - (n - 1)
            jac[r, :n] = -_gradient_lag(h, lag, dx)
        return jac
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    best_h = h_start.copy()
    best_c = c0
    
    try:
        res = minimize(obj_fun, z0, jac=obj_jac, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], target)
            c_new, _ = exact_objective(h_new)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new
    except Exception:
        pass
    
    return best_h, best_c

    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _gradient_lag(x, lag, dx):
    n = len(x)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= x[lag:]
            grad[lag:] += 1.0 - x[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - x[a:]
            grad[a:] -= x[:s]
    return grad * dx

def slsqp_direct(h_start, max_iter=150, ftol=1e-14):
    global _g_n, _g_target, _g_nc
    n = len(h_start)
    h_start = project_box_sum(h_start, 0.5 * n)
    c0, _ = exact_objective(h_start)
    _g_n = n
    _g_target = 0.5 * n
    _g_nc = 2 * n - 1

    z0 = np.concatenate([h_start, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac}
    ]

    best_h = h_start.copy()
    best_c = c0
    try:
        res = minimize(_obj_fun, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], 0.5 * n)
            c_new, _ = exact_objective(h_new)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new
    except Exception:
        pass
    return best_h, best_c

def slsqp_active(h_start, top_k, max_iter=200):
    """SLSQP using only the top-k most active constraints for speed."""
    global _g_n, _g_target, _g_nc
    n = len(h_start)
    h_start = project_box_sum(h_start, 0.5 * n)
    c0, profile = exact_objective(h_start)
    _g_n = n
    _g_target = 0.5 * n
    _g_nc = 2 * n - 1

    z0 = np.concatenate([h_start, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    # Find top-k active lags
    top_indices = np.argsort(profile)[-top_k:]
    top_indices = np.sort(top_indices)

    def _a_ineq_fun(z):
        h = z[:n]
        c = z[n]
        dx = 2.0 / n
        full_profile = np.correlate(h, 1.0 - h, mode='full') * dx
        return c - full_profile[top_indices]

    def _a_ineq_jac(z):
        h = z[:n]
        dx = 2.0 / n
        jac = np.zeros((len(top_indices), n + 1))
        jac[:, n] = 1.0
        for i, r in enumerate(top_indices):
            lag = r - (n - 1)
            jac[i, :n] = -_gradient_lag(h, lag, dx)
        return jac

    constraints = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _a_ineq_fun, 'jac': _a_ineq_jac}
    ]

    best_h = h_start.copy()
    best_c = c0
    try:
        res = minimize(_obj_fun, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], 0.5 * n)
            c_new, _ = exact_objective(h_new)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new
    except Exception:
        pass
    return best_h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=60, rng=None):
    n = len(b)
    if n % 2 != 0 or b is None:
        return b.copy(), exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(0)
    b = b.copy()
    c5, _ = exact_objective(b)
    for rnd in range(max_rounds):
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 5000:
            for i in ones:
                new_improved = False
                for j in zeros:
                    b[i] = 0.0
                    b[j] = 1.0
                    nc, _ = exact_objective(b)
                    if nc < c5 - 1e-13:
                        c5 = nc
                        improved = True
                        ones = list(np.where(b > 0.5)[0])
                        zeros = list(np.where(b < 0.5)[0])
                        new_improved = True
                        break
                    b[i] = 1.0
                    b[j] = 0.0
                if new_improved:
                    break
        else:
            n_tries = 800
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc, _ = exact_objective(b)
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = list(np.where(b > 0.5)[0])
                    zeros = list(np.where(b < 0.5)[0])
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5

def enumerate_binary(n):
    if n % 2 != 0:
        return None, float('inf')
    half = n // 2
    best_pattern = None
    best_c = float('inf')
    dx = 2.0 / n
    for combo in combinations(range(n), half):
        h = np.zeros(n)
        for i in combo:
            h[i] = 1.0
        c, _ = exact_objective(h)
        if c < best_c - 1e-15:
            best_c = c
            best_pattern = h.copy()
    return best_pattern, best_c

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        cnt = int(round(target))
        idx = rng.choice(n, size=min(cnt, n), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(2, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.2
    elif mode == 5:
        x = np.arange(n, dtype=np.float64)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3.0 * x / n) + 0.2 * np.cos(2 * np.pi * 5.0 * x / n + rng.uniform(0, 1))
    elif mode == 6:
        h = np.zeros(n)
        q = n // 3
        h[:q] = 1.0
        h[n//2:n//2+q//2] = 1.0
        h[-q//2:] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def gradient_descent_step(h, max_steps=50, deadline=None):
    """Gradient descent using active constraint gradients."""
    n = len(h)
    dx = 2.0 / n
    h = project_box_sum(h, 0.5 * n)
    best_c, profile = exact_objective(h)
    best_h = h.copy()
    target = 0.5 * n

    for step in range(max_steps):
        if deadline is not None and time.time() > deadline - 2:
            break
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        pmax = float(np.max(profile))
        if pmax < best_c - 1e-13:
            best_c = pmax
            best_h = h.copy()

        # Find active lags (top 5)
        top_k = min(5, len(profile))
        active_idx = np.argsort(profile)[-top_k:]
        active_idx = np.sort(active_idx)

        # Weighted negative gradient
        weights = np.zeros(len(profile))
        weights[active_idx] = 1.0 / top_k
        d = np.zeros(n)
        for r in active_idx:
            lag = r - (n - 1)
            d -= weights[r] * _gradient_lag(h, lag, dx)

        # Normalize direction
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm

        # Line search
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = h - alpha * d
            h_trial = project_box_sum(h_trial, target)
            c_trial, _ = exact_objective(h_trial)
            if c_trial < pmax - 1e-12:
                h = h_trial
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    # Phase 0: Get initial values
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0, _ = exact_objective(init)
            best_h = init.copy()
            best_c = c0
            best_n = n0
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 70
        best_h = make_start(n0, rng, mode=3)
        best_c, _ = exact_objective(best_h)
        best_n = n0

    # Phase 1: Direct SLSQP on parent
    if time.time() < deadline - 200:
        h_opt, c_opt = slsqp_direct(best_h, max_iter=200)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Active-constraint SLSQP (top-k)
    if time.time() < deadline - 180:
        n_cur = len(best_h)
        for topk in [10, 20, 30]:
            if time.time() > deadline - 160:
                break
            h_opt, c_opt = slsqp_active(best_h, top_k=topk, max_iter=150)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 3: Binary refinement
    if len(best_h) % 2 == 0 and time.time() < deadline - 100:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=80, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b
            # SLSQP from binary
            if time.time() < deadline - 80:
                h_opt, c_opt = slsqp_direct(b, max_iter=150)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                h_opt2, c_opt2 = slsqp_active(b, top_k=15, max_iter=100)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2

    # Phase 4: Enumeration for small n
    small_ns = [8, 10, 12, 14, 16, 18]
    for n_enum in small_ns:
        if time.time() > deadline - 60:
            break
        b_pat, c_enum = enumerate_binary(n_enum)
        if b_pat is not None and c_enum < best_c:
            best_c = c_enum
            best_h = b_pat.copy()
            best_n = n_enum
            # Refine with SLSQP
            if time.time() < deadline - 40:
                h_opt, c_opt = slsqp_direct(b_pat, max_iter=200)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                    best_n = n_enum
        elif b_pat is not None and c_enum < best_c + 0.01:
            # Close - try SLSQP refinement
            if time.time() < deadline - 30:
                h_opt, c_opt = slsqp_direct(b_pat, max_iter=150)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                    best_n = n_enum

    # Phase 5: Try different sizes with multiple starts
    n_best = len(best_h)
    size_candidates = set()
    for delta in [-4, -2, 2, 4, 6, 8, -6, -8]:
        s = n_best + delta
        if 8 <= s <= 120 and s % 2 == 0:
            size_candidates.add(s)
    # Add some fixed sizes
    for s in [20, 24, 28, 32, 36, 40, 44, 48, 52, 56, 60, 64, 68, 72, 80]:
        size_candidates.add(s)
    size_candidates = sorted(size_candidates, key=lambda x: abs(x - n_best))

    for n_try in size_candidates:
        if n_try == n_best or n_try < 8:
            continue
        if time.time() > deadline - 30:
            break
        # Downsample current best
        h_ds = downsample(best_h, n_try)
        c_ds, _ = exact_objective(h_ds)

        # Also try a random start
        h_rand = make_start(n_try, rng, mode=rng.randint(0, 7))
        c_rand, _ = exact_objective(h_rand)

        h_start = h_ds if c_ds <= c_rand else h_rand

        # Direct SLSQP
        if time.time() < deadline - 15:
            h_opt, c_opt = slsqp_direct(h_start, max_iter=120)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                best_n = n_try

            # Active SLSQP
            if time.time() < deadline - 15:
                h_opt, c_opt = slsqp_active(h_start, top_k=15, max_iter=100)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                    best_n = n_try

            # Binary refinement
            b = binary_threshold(h_opt)
            if b is not None and time.time() < deadline - 10:
                b, c_b = binary_swap_improve(b, max_rounds=40, rng=rng)
                if c_b < best_c:
                    best_c = c_b
                    best_h = b
                    best_n = n_try

    # Phase 6: Gradient descent refinement
    if time.time() < deadline - 20:
        h_opt, c_opt = gradient_descent_step(best_h, max_steps=100, deadline=deadline)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final SLSQP rounds
    if time.time() < deadline - 15:
        h_opt, c_opt = slsqp_direct(best_h, max_iter=200)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if time.time() < deadline - 15:
        h_opt, c_opt = slsqp_active(best_h, top_k=20, max_iter=150)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final binary + SLSQP
    if len(best_h) % 2 == 0 and time.time() < deadline - 10:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=50, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)

    # Early success check
    if best_c > 0.38085 and best_h is not None:
        # Try one more aggressive round
        if time.time() < deadline - 10:
            for attempt in range(3):
                h_pert = best_h + rng.randn(len(best_h)) * 0.01
                h_pert = project_box_sum(h_pert, 0.5 * len(best_h))
                h_opt, c_opt = slsqp_direct(h_pert, max_iter=100)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()

    # Final revalidation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)

    return (best_h, final_c, len(best_h))
