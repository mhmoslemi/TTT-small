def project_box_sum(h, target=None):

def gradient_step(h, max_rounds=20, rng=None):
    """Descend on the active constraint gradient with projection."""
    n = len(h)
    target = 0.5 * n
    dx = 2.0 / n
    c_best = exact_objective(h)
    h_best = h.copy()
    
    for _ in range(max_rounds):
        profile = get_profile(h_best)
        active_r = int(np.argmax(profile))
        # Gradient of the active constraint
        e = active_r - (n - 1)
        grad = np.zeros(n)
        if e >= 0:
            s = n - e
            if s > 0:
                grad[:s] += dx * (1.0 - h_best[e:])
                grad[e:] -= dx * h_best[:s]
        else:
            a = -e
            s = n - a
            if s > 0:
                grad[a:] += dx * (1.0 - h_best[:s])
                grad[:s] -= dx * h_best[a:]

        for alpha in [0.01, 0.005, 0.002, 0.001]:
            h_trial = h_best - alpha * grad
            h_trial = project_box_sum(h_trial, target)
            c_trial = exact_objective(h_trial)
            if c_trial < c_best - 1e-14:
                c_best = c_trial
                h_best = h_trial
                break
    
    return h_best, c_best
