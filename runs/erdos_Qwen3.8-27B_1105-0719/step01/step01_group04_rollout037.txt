def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += min(diff2, 1.0 - h[i])
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += max(diff2, -h[i])
                break
    return h

def compute_F(h):
    n = len(h)
    F = np.zeros(n)
    F[0] = np.dot(h, h)
    cumsum = np.cumsum(h)
    total = cumsum[-1]
    P = np.zeros(n)
    P[1:] = cumsum[:-1]
    S = np.zeros(n)
    S[1:] = total - cumsum[:-1]
    for m in range(1, n):
        A_m = np.dot(h[:n-m], h[m:])
        F[m] = A_m + min(P[m], S[m])
    return F

def smooth_obj_and_grad(h, beta, beta_e):
    n = len(h)
    F = np.zeros(n)
    F[0] = np.dot(h, h)
    cumsum = np.cumsum(h)

    S = np.zeros(n)
    S[1:] = total - cumsum[:-1]

    for m in range(1, n):
        A_m = np.dot(h[:n-m], h[m:])
        diff_ps = P[m] - S[m]
        if abs(diff_ps) < 1e-12:
            sm = min(P[m], S[m])
        else:
            m0 = min(P[m], S[m])
            sm = m0 - (1.0 / beta_e) * np.log(1.0 + np.exp(-beta_e * abs(diff_ps)))
        F[m] = A_m + sm

    F_min = np.min(F)
    shifted = F - F_min
    exp_neg = np.exp(-beta * shifted)
    sum_exp = np.sum(exp_neg)
    val = -F_min + np.log(sum_exp) / beta

    weights = exp_neg / sum_exp
    grad = np.zeros(n)

    h_cr = repair_feasible(1.0 - best_h[::-1])
    s_cr = exact_score(h_cr)
    if s_cr < best_score - 1e-13:
        best_h = h_cr
        best_score = s_cr

    # Phase 1: Smooth SLSQP with annealed beta
    betas = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0]
    beta_e_list = [5.0, 10.0, 30.0, 50.0, 50.0, 50.0]
    for i, (beta, beta_e) in enumerate(zip(betas, beta_e_list)):
        if tl() < 30:
            break
        maxiter = 300 if i >= 3 else 200
        h_sol, score = solve_smooth_slsqp(best_h, beta, beta_e, maxiter)
        if h_sol is not None and score < best_score - 1e-12:
            best_h = h_sol
            best_score = score

    # Phase 2: Active set descent
    if tl() > 30:
        h_ref, score_ref = active_set_descent(best_h, max_steps=100, gap_tol=0.5)
        if score_ref < best_score - 1e-12:
            best_h = h_ref
            best_score = score_ref

    # Phase 3: More SLSQP with high beta
    if tl() > 40:
        for beta in [50.0, 200.0, 500.0]:
            if tl() < 25:
                break
            h_sol, score = solve_smooth_slsqp(best_h, beta, 50.0, maxiter=250)
            if h_sol is not None and score < best_score - 1e-12:
                best_h = h_sol
                best_score = score

    # Phase 4: Active descent with wider gap
    if tl() > 20:
        h_ref, score_ref = active_set_descent(best_h, max_steps=60, gap_tol=2.0)
        if score_ref < best_score - 1e-12:
            best_h = h_ref
            best_score = score_ref

    # Phase 5: Dimension transfers
    if tl() > 60:
        n_cur = len(best_h)
        dims_to_try = []
        for dn in [-20, -10, 10, 20, 40]:
            nd = n_cur + dn
            if 40 <= nd <= 300:
                dims_to_try.append(nd)
        if n_cur * 2 <= 280:
            dims_to_try.append(n_cur * 2)
        for nd in dims_to_try:
            if tl() < 30:
                break
            h_interp = interpolate_to_dimension(best_h, nd)
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.015:
                for beta in [10.0, 50.0]:
                    if tl() < 20:
                        break
                    h_sol, score = solve_smooth_slsqp(h_interp, beta, 50.0, maxiter=150)
                    if h_sol is not None and score < best_score - 1e-12:
                        best_h = h_sol
                        best_score = score
                if tl() > 15:
                    h_ref, score_ref = active_set_descent(best_h, max_steps=40, gap_tol=0.5)
                    if score_ref < best_score - 1e-12:
                        best_h = h_ref
                        best_score = score_ref

    # Phase 6: Multi-start with perturbations
    if tl() > 60:
        for trial in range(12):
            if tl() < 25:
                break
            eps = rng.uniform(0.003, 0.015)
            z = rng.standard_normal(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_sol, score = solve_smooth_slsqp(h_pert, 30.0, 50.0, maxiter=100)
            if h_sol is not None and score < best_score - 1e-12:
                best_h = h_sol
                best_score = score

    # Phase 7: Rounding + refinement
    if tl() > 30:
        n_cur = len(best_h)
        for thresh in [0.4, 0.5, 0.6]:
            if tl() < 20:
                break
            binary = (best_h > thresh).astype(np.float64)
            binary = repair_feasible(binary)
            h_bin_ref = repair_feasible(interpolate_to_dimension(binary, n_cur))
            h_sol, score = solve_smooth_slsqp(h_bin_ref, 20.0, 50.0, maxiter=150)
            if h_sol is not None:
                h_ref, score_ref = active_set_descent(h_sol, max_steps=40, gap_tol=0.5)
                if score_ref < best_score - 1e-12:
                    best_h = h_ref
                    best_score = score_ref

    # Phase 8: Final active descent
    if tl() > 15:
        h_ref, score_ref = active_set_descent(best_h, max_steps=80, gap_tol=0.3)
        if score_ref < best_score - 1e-12:
            best_h = h_ref
            best_score = score_ref

    # Phase 9: Final SLSQP
    if tl() > 20:
        h_sol, score = solve_smooth_slsqp(best_h, 100.0, 50.0, maxiter=200)
        if h_sol is not None and score < best_score - 1e-12:
            best_h = h_sol
            best_score = score
        h_ref, score_ref = active_set_descent(best_h, max_steps=50, gap_tol=0.5)
        if score_ref < best_score - 1e-12:
            best_h = h_ref
            best_score = score_ref

    # Phase 10: Try slightly different active gap for final push
    if tl() > 10:
        for gap in [0.1, 0.3, 1.0, 3.0]:
            if tl() < 8:
                break
            h_ref, score_ref = active_set_descent(best_h, max_steps=30, gap_tol=gap)
            if score_ref < best_score - 1e-12:
                best_h = h_ref
                best_score = score_ref

    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
