import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add total diff; add to elements with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        # Need to remove total -diff
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    return h

def build_gradient_matrix(h, n, dx):
    """Build the gradient matrix G where G[m, j] = dP[m]/dh[j]."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # Positive lag d
            # grad[j] = (1-h[j+d]) * [j <= n-d-1] - h[j-d] * [j >= d]
            G[m, 0:n-d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n-d]
        else:
            # Negative lag, a = -d
            a = -d
            # grad[j] = (1-h[j-a]) * [j >= a] - h[j+a] * [j <= n-a-1]
            G[m, a:n] += dx * (1.0 - h[0:n-a])
            G[m, 0:n-a] += -dx * h[a:n]
    return G

def sca_lp_propose(h, r):
    """Solve the SCA LP to propose a new point z within trust region of radius r."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    profile = compute_profile(h)
    G = build_gradient_matrix(h, n, dx)
    
    # LP: minimize t
    # s.t. G[m]·z - t <= G[m]·h - P[m] for all m
    #      sum z = n/2
    #      max(0, h-r) <= z <= min(1, h+r)
    #      -inf <= t <= inf
    
    # Variable ordering: [z_0, ..., z_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: G[m]·z - t <= G[m]·h - P[m]
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - profile
    
    # Equality: sum z = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    bounds = []
    for j in range(n):
        lo = max(0.0, h[j] - r)
        hi = min(1.0, h[j] + r)
        bounds.append((lo, hi))
    bounds.append((-1.0, 10.0))  # t bounds
    
    # Solve
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            z = res.x[:n]
            if np.all(np.isfinite(z)):
                return z
    except Exception:
        pass
    
    # Fallback: try cvxpy
    try:
        import cvxpy as cp
        z_var = cp.Variable(n)
        t_var = cp.Variable()
        lo = np.maximum(0.0, h - r)
        hi = np.minimum(1.0, h + r)
        constraints = [
            cp.sum(z_var) == n / 2.0,
            G @ z_var - t_var <= G @ h - profile,
            z_var >= lo,
            z_var <= hi
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=cp.ECOS, tol=1e-9)
        if z_var.value is not None and np.all(np.isfinite(z_var.value)):
            return np.array(z_var.value)
    except Exception:
        pass
    
    return None

def line_search(cur_h, target_z, cur_score, max_alphas=12):
    """Line search from cur_h toward target_z, returning best feasible point."""
    n = len(cur_h)
    best_h = cur_h.copy()
    best_c = cur_score
    
    # Generate alpha values
    alphas = np.array([1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.025, 0.01])
    if len(alphas) > max_alphas:
        alphas = alphas[:max_alphas]
    
    diff = target_z - cur_h
    for alpha in alphas:
        candidate = cur_h + alpha * diff
        candidate = repair_feasible(candidate)
        c_new = exact_score(candidate)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = candidate.copy()
    
    return best_h, best_c

    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n - d])
            G[m, 0:n - d] += -dx * h[d:n]
        else:
            a = -d
            G[m, a:n] += dx * (1.0 - h[0:n - a])
            G[m, 0:n - a] += -dx * h[a:n]
    return G

def sca_lp_propose(h, r):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    G = build_gradient_matrix(h, n, dx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.vstack([G, np.zeros((1, n))])
    A_ub[:, n] = -1.0
    b_ub = G @ h - profile
    b_ub = np.append(b_ub, 0.0)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for j in range(n):
        lo = max(0.0, h[j] - r)
        hi = min(1.0, h[j] + r)
        bounds.append((lo, hi))
    bounds.append((-10.0, 10.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and np.all(np.isfinite(res.x[:n])):
            return res.x[:n]
    except Exception:
        pass
    try:
        import cvxpy as cp
        z_var = cp.Variable(n)
        t_var = cp.Variable()
        lo = np.maximum(0.0, h - r)
        hi = np.minimum(1.0, h + r)
        constraints = [
            cp.sum(z_var) == n / 2.0,
            G @ z_var - t_var <= G @ h - profile,
            z_var >= lo,
            z_var <= hi
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=cp.ECOS, tol=1e-10)
        if z_var.value is not None and np.all(np.isfinite(z_var.value)):
            return np.array(z_var.value, dtype=np.float64)
    except Exception:
        pass
    return None

def line_search(cur_h, z_target, cur_score, num_alphas=14):
    n = len(cur_h)
    z = repair_feasible(z_target)
    best_h = cur_h.copy()
    best_c = cur_score
    alphas = np.linspace(0.05, 1.0, num_alphas)
    for alpha in alphas:
        candidate = (1.0 - alpha) * cur_h + alpha * z
        candidate = np.clip(candidate, 0.0, 1.0)
        diff = n / 2.0 - candidate.sum()
        if abs(diff) > 1e-10:
            candidate = repair_feasible(candidate)
        c_new = exact_score(candidate)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = candidate.copy()
    return best_h, best_c

def sca_refine(h0, time_limit, initial_radius=0.25):
    cur_h = repair_feasible(h0.copy())
    cur_c = exact_score(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    radii = [0.3, 0.2, 0.12, 0.08, 0.05, 0.035, 0.025, 0.015, 0.01, 0.007, 0.004, 0.002, 0.001]
    r_idx = 0
    stagnation = 0
    max_stag = 3
    t_start = time.perf_counter()
    total_iters = 0
    while time.perf_counter() - t_start < time_limit:
        total_iters += 1
        r = radii[min(r_idx, len(radii) - 1)]
        z = sca_lp_propose(cur_h, r)
        if z is not None:
            new_h, new_c = line_search(cur_h, z, cur_c)
            if new_c < cur_c - 1e-12:
                cur_h = new_h
                cur_c = new_c
                if cur_c < best_c - 1e-13:
                    best_h = cur_h.copy()
                    best_c = cur_c
                stagnation = 0
                if r_idx >= 3 and stagnation == 0:
                    r_idx = max(0, r_idx - 1)
            else:
                stagnation += 1
                if stagnation >= max_stag:
                    r_idx = min(r_idx + 1, len(radii) - 1)
                    stagnation = 0
        else:
            r_idx = min(r_idx + 1, len(radii) - 1)
            stagnation += 1
            if stagnation >= max_stag:
                z = repair_feasible(cur_h + 0.008 * np.random.randn(len(cur_h)))
                new_h, new_c = line_search(cur_h, z, cur_c)
                if new_c < cur_c - 1e-12:
                    cur_h = new_h
                    cur_c = new_c
                    if cur_c < best_c - 1e-13:
                        best_h = cur_h.copy()
                        best_c = cur_c
                stagnation = 0
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def generate_wave_start(n, freq, phase, amplitude, offset):
    x = (np.arange(n) + 0.5) / n
    h = offset + amplitude * np.cos(2.0 * np.pi * freq * x + phase)
    return repair_feasible(np.clip(h, 0.0, 1.0))

def slsqp_polish(h0, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    
    def ep_obj(x):
        return x[-1]
    def ep_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_f(x):
        return x[:-1].sum() - n / 2.0
    def eq_j(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_f(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof
    def ineq_j(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    cons = [
        {'type': 'eq', 'fun': eq_f, 'jac': eq_j},
        {'type': 'ineq', 'fun': ineq_f, 'jac': ineq_j}
    ]
    try:
        res = minimize(ep_obj, x0, method='SLSQP', jac=ep_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            return h_new
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start) - 5.0

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    
    if parent is None or len(parent) < 4 or not np.all(np.isfinite(parent)):
        parent = np.full(66, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    n_par = len(best_h)
    starts = [best_h.copy()]
    
    for freq in [1, 2, 3, 4, 5]:
        for phase in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0]:
            for amp in [0.4, 0.45, 0.5]:
                hw = generate_wave_start(n_par, freq, phase, amp, 0.5)
                starts.append(hw)
    
    for _ in range(6):
        z = rng.randn(n_par)
        z = z / (np.max(np.abs(z)) + 1e-10)
        hr = repair_feasible(np.clip(0.5 + 0.35 * z, 0.0, 1.0))
        starts.append(hr)
    
    for eps in [0.005, 0.01, 0.02, 0.04]:
        z = rng.randn(n_par)
        z -= z.mean()
        hp = repair_feasible(best_h + eps * z)
        starts.append(hp)
    
    scored = []
    for s in starts:
        c = exact_score(s)
        scored.append((c, s))
    scored.sort(key=lambda x: x[0])

    n_starts_to_try = min(10, len(scored))
    per_start_time = max(20.0, (time_left() - 40.0) / (n_starts_to_try * 1.5))
    
    for idx in range(n_starts_to_try):
        if time_left() < 35:
            break
        c_s, h_s = scored[idx]
        t_limit = min(per_start_time, time_left() - 15)
        if t_limit < 10:
            break
        h_ref, c_ref = sca_refine(h_s, t_limit)
        if c_ref < best_c - 1e-12:
            best_h = h_ref
            best_c = c_ref

    if time_left() > 50:
        dims = set()
        for d_offset in [-6, -4, -2, 2, 4, 6, 10, -10, 14]:
            d = best_h.size + d_offset
            if 28 <= d <= 150:
                dims.add(d)
        dims.add(64)
        dims.add(72)
        dims = sorted(dims)
        tried = set([len(best_h)])
        
        for new_n in dims:
            if time_left() < 35:
                break
            if new_n in tried:
                continue
            tried.add(new_n)
            h_interp = interpolate_to_n(best_h, new_n)
            t_lim = min(35.0, time_left() - 15)
            if t_lim < 10:
                break
            h_ref, c_ref = sca_refine(h_interp, t_lim)
            if c_ref < best_c - 1e-12:
                best_h = h_ref
                best_c = c_ref

    if time_left() > 25:
        h_sol = slsqp_polish(best_h, maxiter=400)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-13:
                best_h = h_sol
                best_c = c_sol

    if time_left() > 15:
        h_ref, c_ref = sca_refine(best_h, min(20.0, time_left() - 3))
        if c_ref < best_c - 1e-13:
            best_h = h_ref
            best_c = c_ref

    if time_left() > 15:
        for eps in [0.003, 0.006, 0.01]:
            if time_left() < 8:
                break
            z = rng.randn(len(best_h))
            z -= z.mean()
            hp = repair_feasible(best_h + eps * z)
            h_sol = slsqp_polish(hp, maxiter=200)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-13:
                    best_h = h_sol
                    best_c = c_sol

    if time_left() > 10:
        h_ref, c_ref = sca_refine(best_h, min(12.0, time_left() - 2))
        if c_ref < best_c - 1e-13:
            best_h = h_ref
            best_c = c_ref

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_res = evaluate_erdos_solution(best_h)
        if isinstance(eval_res, (float, int)):
            final_c = float(eval_res)
        elif isinstance(eval_res, tuple) and len(eval_res) >= 2:
            final_c = float(eval_res[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
