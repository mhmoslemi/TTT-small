```python
def exact_score(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def repair_feasible(vec):

def compute_lag_values_and_grads(h):

def trust_region_step(h_cur, lambda_tr, n_active=None, tol=1e-6):

    n = len(h_cur)
    values, grads = compute_lag_values_and_grads(h_cur)
    
    # Select active lags
    if n_active is not None and n_active < 2*n-1:

        sorted_idx = np.argsort(values)[::-1][:n_active]
        sorted_idx = np.sort(sorted_idx)
        vals_act = values[sorted_idx]
        grads_act = grads[sorted_idx, :]
    else:

        vals_act = values
        grads_act = grads

    """Polish using weighted gradient descent on active lags."""

    h = repair_feasible(h)
    best = h.copy()

        # Softmax weights
        weights = np.exp(-gaps / (temperature * c + 1e-10))
        weights /= np.sum(weights) + 1e-15
        
        # Direction: negative weighted sum of gradients
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        direction -= np.mean(direction)  # preserve sum constraint

                best_score = score
                improved = True
                break
        if not improved:
            break
    
    return best, best_score
```

```python
    # gradients for negative lag (values[n-1-delta])
    grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])     # first factor
    grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]            # second factor
```

For the positive lag:
- `idx` = [0, ..., n-1-delta], contribution to grads at positions `idx`: dx*(1-h[idx+delta]) ✓
- `idx + delta` = [delta, ..., n-1], contribution to grads at positions `idx+delta`: -dx*h[idx] ✓

For the negative lag:
- `idx2` = [delta, ..., n-1], contribution to grads at positions `idx2`: dx*(1-h[idx2-delta]) ✓
- `idx2 - delta` = [0, ..., n-1-delta], contribution to grads at positions `idx2-delta`: -dx*h[idx2] ✓

```python
import numpy as np
import cvxpy as cp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h


def compute_lag_data(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    values[n - 1] = dx * np.sum(h * (1.0 - h))
    grads[n - 1, :] = dx * (1.0 - 2.0 * h)
    for delta in range(1, n):
        idx = np.arange(n - delta)
        idx2 = np.arange(delta, n)
        values[n - 1 + delta] = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
        values[n - 1 - delta] = dx * np.sum(h[idx2] * (1.0 - h[idx2 - delta]))
        grads[n - 1 + delta, idx] = dx * (1.0 - h[idx + delta])
        grads[n - 1 + delta, idx + delta] = -dx * h[idx]
        grads[n - 1 - delta, idx2] = dx * (1.0 - h[idx2 - delta])
        grads[n - 1 - delta, idx2 - delta] = -dx * h[idx2]
    return values, grads


def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)


def trust_region_solve(h_cur, lam, n_active=None, tv_weight=0.0):
    n = len(h_cur)
    values, grads = compute_lag_data(h_cur)
    n_lags = 2 * n - 1

    if n_active is not None and n_active < n_lags:
        sorted_idx = np.argsort(values)[::-1][:n_active]
        sorted_idx = np.sort(sorted_idx)
    else:
        sorted_idx = np.arange(n_lags)

    vals_act = values[sorted_idx]
    grads_act = grads[sorted_idx, :]
    n_act = len(sorted_idx)

    h_var = cp.Variable(n)
    t_var = cp.Variable()

    offset = vals_act - grads_act @ h_cur
    constraints = [offset + grads_act @ h_var <= t_var]
    constraints.append(cp.sum(h_var) == n / 2.0)
    constraints.append(h_var >= 0.0)
    constraints.append(h_var <= 1.0)

    obj = t_var + lam * cp.sum_squares(h_var - h_cur)
    if tv_weight > 0:
        tv = cp.sum(cp.abs(h_var[1:] - h_var[:-1]))
        obj = obj + tv_weight * tv

    prob = cp.Problem(cp.Minimize(obj), constraints)

    solved = False
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solved = True
                break
        except Exception:
            continue
    if not solved:
        try:
            prob.solve()
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solved = True
        except Exception:
            pass

    if not solved or h_var.value is None:
        return None

    h_new = np.array(h_var.value, dtype=np.float64)
    return repair_feasible(h_new)


def trust_region_optimize(h_start, lam=0.1, max_iters=30, n_active=None, tv_weight=0.0):
    h_cur = repair_feasible(h_start.copy())
    best = h_cur.copy()
    best_score = exact_score(best)
    cur_lam = lam

    for it in range(max_iters):
        h_new = trust_region_solve(h_cur, cur_lam, n_active=n_active, tv_weight=tv_weight)
        if h_new is None:
            break
        score_new = exact_score(h_new)
        if score_new < best_score - 1e-14:
            h_cur = h_new
            best = h_new.copy()
            best_score = score_new
            cur_lam = max(0.01, cur_lam * 0.7)
        else:
            step = 0.5
            improved = False
            for step in [0.5, 0.25, 0.125]:
                h_inter = repair_feasible(h_cur + step * (h_new - h_cur))
                s_inter = exact_score(h_inter)
                if s_inter < best_score - 1e-14:
                    h_cur = h_inter
                    best = h_inter.copy()
                    best_score = s_inter
                    improved = True
                    break
            if not improved:
                cur_lam *= 2.0
                if cur_lam > 50.0:
                    break

    return best, best_score


def active_descent_momentum(h_start, max_steps=60, n_top=20, temp=0.003, momentum=0.5):
    n = len(h_start)
    best = repair_feasible(h_start.copy())
    best_score = exact_score(best)
    vel = np.zeros(n)

    for step in range(max_steps):
        values, grads = compute_lag_data(best)
        c = np.max(values)
        n_act = min(n_top, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_act]
        gaps = np.maximum(c - values[active_idx], 0.0)
        weights = np.exp(-gaps / (temp * c + 1e-10))
        weights /= np.sum(weights) + 1e-15
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        direction -= np.mean(direction)

        norm_dir = np.linalg.norm(direction)
        if norm_dir < 1e-12:
            break

        vel = momentum * vel + direction
        vel -= np.mean(vel)

        improved = False
        for alpha in [0.04, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = repair_feasible(best + alpha * vel)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        if not improved:
            vel *= 0.5
            for alpha in [0.005, 0.002, 0.001]:
                h_trial = repair_feasible(best + alpha * direction)
                score = exact_score(h_trial)
                if score < best_score - 1e-14:
                    best = h_trial.copy()
                    best_score = score
                    improved = True
                    break
            if not improved:
                break

    return best, best_score


def coordinate_descent(h_start, max_rounds=30):
    n = len(h_start)
    best = repair_feasible(h_start.copy())
    best_score = exact_score(best)

    for rnd in range(max_rounds):
        values, grads = compute_lag_data(best)
        c = np.max(values)
        n_act = min(30, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_act]
        gaps = np.maximum(c - values[active_idx], 0.0)
        weights = np.exp(-gaps / (0.003 * c + 1e-10))
        weights /= np.sum(weights) + 1e-15
        grad_combo = np.sum(weights[:, None] * grads[active_idx, :], axis=0)

        improved_any = False
        for i in range(n):
            if abs(grad_combo[i]) < 1e-8:
                continue
            # Move h[i] in direction that reduces grad_combo[i] * dh[i]
            if grad_combo[i] > 0:
                # Decrease h[i]
                for delta_h in [0.02, 0.01, 0.005, 0.001]:
                    h_trial = best.copy()
                    h_trial[i] = max(0.0, h_trial[i] - delta_h)
                    # Compensate sum
                    comp = delta_h
                    # Spread comp to other elements (those with room)
                    others = np.argsort(best[np.not_equal(np.arange(n), i)])
                    spread = 0.0
                    for j in others:
                        rem = comp - spread
                        add = min(rem, 1.0 - best[j])
                        if add <= 0:
                            break
                        h_trial[j] += add
                        spread += add
                    if spread < comp - 1e-10:
                        continue
                    h_trial = repair_feasible(h_trial)
                    score = exact_score(h_trial)
                    if score < best_score - 1e-14:
                        best = h_trial
                        best_score = score
                        improved_any = True
                        break
            else:
                # Increase h[i]
                for delta_h in [0.02, 0.01, 0.005, 0.001]:
                    h_trial = best.copy()
                    h_trial[i] = min(1.0, h_trial[i] + delta_h)
                    comp = delta_h
                    others = np.argsort(best[np.not_equal(np.arange(n), i)])
                    spread = 0.0
                    for j in others:
                        rem = comp - spread
                        sub = min(rem, best[j])
                        if sub <= 0:
                            break
                        h_trial[j] -= sub
                        spread += sub
                    if spread < comp - 1e-10:
                        continue
                    h_trial = repair_feasible(h_trial)
                    score = exact_score(h_trial)
                    if score < best_score - 1e-14:
                        best = h_trial
                        best_score = score
                        improved_any = True
                        break

        if not improved_any:
            break

    return best, best_score
```
