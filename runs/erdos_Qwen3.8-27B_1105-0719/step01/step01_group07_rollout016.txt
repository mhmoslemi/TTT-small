def binary_sa(n, deadline, rng, max_moves=2000000):
    """
    Binary simulated annealing with n/2 ones and n/2 zeros.
    """
    m = n // 2
    # Initialize: first m are ones
    h = np.zeros(n)
    h[:m] = 1.0
    curr_obj = exact_objective(h)
    best_h = h.copy()
    best_obj = curr_obj
    
    T = 0.005  # initial temperature
    T_min = 1e-6
    T_factor = 0.9997  # cooling rate
    
    moves_since_best = 0
    best_moves_between = max(1, max_moves // 50)
    
    for it in range(max_moves):
        if time.time() > deadline - 5:
            break
        if it % 5000 == 0 and time.time() > deadline - 5:
            break
        
        # Pick a random 1 and a random 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        new_obj = exact_objective(h)
        
        delta = new_obj - curr_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            curr_obj = new_obj
            if curr_obj < best_obj:
                best_obj = curr_obj
                best_h = h.copy()
                moves_since_best = 0
            else:
                moves_since_best += 1
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
            moves_since_best += 1
        
        T *= T_factor
        
        # Reheat if stuck
        if moves_since_best > best_moves_between:
            T = max(T, 0.003)
            moves_since_best = 0
    
    return best_h, best_obj

    w = 0.05 * (U - L)
    gl, gu = max(L, best_x - w), min(U, best_x + w)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(45):
        if gu - gl < 1e-14:
            break
        x1 = gu - gr * (gu - gl)
        x2 = gl + gr * (gu - gl)
        f1 = F(x1)
        f2 = F(x2)
        if f1 < f2:
            gu = x2
        else:
            gl = x1
    xg = (gl + gu) / 2.0
    fg = F(xg)
    if fg < best_F:
        best_x = xg
        best_F = fg
    h_new = h.copy()
    h_new[i] = float(np.clip(best_x, L, U))
    h_new[j] = float(np.clip(S - best_x, L, U))
    obj_new = exact_objective(h_new)
    if obj_new < curr_obj - 1e-15:
        return h_new, obj_new
    return h, curr_obj


def anti_cd(h, deadline, max_sweeps=80):
    n = len(h)
    m = n // 2
    obj = exact_objective(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline - 12:
            break
        improved = False
        order = list(range(m))
        if sweep > 5:
            order = order[::-1]
        for i in order:
            if time.time() > deadline - 12:
                break
            j = n - 1 - i
            h, obj = line_search(h, i, j, obj)
            if obj < exact_objective(h) + 1:
                improved = True
        if not improved:
            # try all pairs in anti structure
            any_imp = False
            for i in range(m):
                if time.time() > deadline - 12:
                    break
                j = n - 1 - i
                h_old_obj = obj
                h_new, obj_new = line_search(h, i, j, obj, n_grid=48)
                if obj_new < obj - 1e-14:
                    h = h_new
                    obj = obj_new
                    any_imp = True
            if not any_imp:
                break
    return h, obj


def general_cd(h, obj, deadline, rng, n_pairs=400, max_sweeps=150):
    n = len(h)
    best_h = h.copy()
    best_obj = obj
    for sweep in range(max_sweeps):
        if time.time() > deadline - 12:
            break
        improved = False
        for _ in range(n_pairs):
            if time.time() > deadline - 12:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_new, obj_new = line_search(best_h, i, j, best_obj)
            if obj_new < best_obj - 1e-14:
                best_h = h_new
                best_obj = obj_new
                improved = True
        if not improved and sweep > 3:
            break
    return best_h, best_obj


def binary_sa(n, deadline, rng, max_moves=5000000, h_init=None):
    m = n // 2
    if h_init is not None:
        h = h_init.copy()
    else:
        h = np.zeros(n)
        h[:m] = 1.0
    curr_obj = exact_objective(h)
    best_h = h.copy()
    best_obj = curr_obj
    T = 0.004
    T_min = 1e-7
    T_factor = 0.99985
    moves_since_best = 0
    reheat_threshold = 200000
    for it in range(max_moves):
        if time.time() > deadline - 6:
            break
        if it % 8000 == 0:
            if time.time() > deadline - 6:
                break
            if moves_since_best > reheat_threshold:
                T = max(T, 0.002)
                moves_since_best = 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_obj = exact_objective(h)
        delta = new_obj - curr_obj
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            curr_obj = new_obj
            if curr_obj < best_obj:
                best_obj = curr_obj
                best_h = h.copy()
                moves_since_best = 0
            else:
                moves_since_best += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            moves_since_best += 1
        T *= T_factor
    return best_h, best_obj


def transfer_to_size(h_src, n_tgt, anti=False):
    n_src = len(h_src)
    if n_src == n_tgt:
        result = project_box(h_src.copy())
        if anti:
            result = to_anti(result)
        return result
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) / n_tgt
    h_new = np.interp(x_tgt, x_src, h_src)
    h_new = project_box(h_new)
    if anti:
        h_new = to_anti(h_new)
    return h_new
