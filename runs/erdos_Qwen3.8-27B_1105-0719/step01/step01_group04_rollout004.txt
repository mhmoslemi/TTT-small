Let me think about what's different and potentially more effective than the previous approach:

1. **Soft-max objective**: Instead of epigraph constraints (which are many inequalities), we optimize a single smooth scalar J_beta(h) = (1/beta)*log(sum exp(beta*C_r)). This is C^∞ and the gradient naturally weights near-worst lags.

2. **Annealing schedule**: Start with low beta (more uniform weighting of all lags) and increase to high beta (focusing on the true max). This is like simulated annealing for the max function.

3. **Different optimizer approach**: Instead of SLSQP with many constraints, we use SLSQP with just bounds and one equality constraint. The soft-max objective handles the "max" smoothing internally.

4. **Portfolio and dimension transfer**: Try multiple starting dimensions and transfer solutions between them.

Let me think about the implementation:

For the soft-max objective:
- Compute all 2n-1 correlation values C_r
- J_beta = (1/beta) * log(sum exp(beta * C_r))
- dJ_beta/dh_i = sum_r exp(beta*C_r) * (dC_r/dh_i) / sum exp(beta*C_r)
- This simplifies to: dJ/dh_i = Σ_r w_r * (dC_r/dh_i) where w_r = exp(beta*C_r) / Σ exp(beta*C_r)

This is a weighted average of the per-lag gradients, with weights concentrated on the worst lags as beta increases.

For the gradient of C_r with respect to h:
- C_{n-1} = dx * Σ h_i(1-h_i), so dC/dh_i = dx*(1-2h_i)
- C_{n-1+k} = dx * Σ_{i=0}^{n-1-k} h_i(1-h_{i+k}), so dC/dh_i = dx*(1-h_{i+k}) for i < n-k, and dC/dh_{i+k} = -dx*h_i
- C_{n-1-k} = dx * Σ_{i=k}^{n-1} h_i(1-h_{i-k}), similar

def exact_score(h):

    dx = 2.0 / n

            h[i] += add
            rem -= add
    else:

        rem = -diff

            if rem <= 1e-15:

            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    return h

def compute_profile(h):
    """Compute all 2n-1 correlation values and their gradients."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    # Center (lag 0 in correlation sense, index n-1)
    values[n-1] = dx * np.sum(h * (1.0 - h))
    grads[n-1, :] = dx * (1.0 - 2.0 * h)
    
    for delta in range(1, n):
        # Positive lag: C_{n-1+delta} = dx * sum_{i=0}^{n-1-delta} h_i * (1-h_{i+delta})
        idx = np.arange(n - delta)
        vals_pos = h[idx] * (1.0 - h[idx + delta])
        values[n - 1 + delta] = dx * np.sum(vals_pos)
        grads[n - 1 + delta, idx] += dx * (1.0 - h[idx + delta])
        grads[n - 1 + delta, idx + delta] += -dx * h[idx]
        
        # Negative lag: C_{n-1-delta} = dx * sum_{i=delta}^{n-1} h_i * (1-h_{i-delta})
        idx2 = np.arange(delta, n)
        vals_neg = h[idx2] * (1.0 - h[idx2 - delta])
        values[n - 1 - delta] = dx * np.sum(vals_neg)
        grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
        grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    
    return values, grads

def soft_max_obj_and_grad(h, beta):
    """Compute soft-max objective and gradient."""
    values, grads = compute_profile(h)
    # Stable log-sum-exp
    max_val = np.max(values)
    shifted = beta * (values - max_val)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum = max_val * beta + np.log(sum_exp)
    obj = log_sum / beta
    
    # Weighted gradient
    weights = exp_shifted / sum_exp
    grad = weights @ grads  # shape (n,)
    
    return obj, grad

def cdf_transfer(h, new_n):
    """Transfer h to new dimension using CDF (cumulative) interpolation."""
    n = len(h)
    dx = 2.0 / n
    # Build CDF of h
    cdf = np.cumsum(h) * dx
    cdf_full = np.concatenate([[0.0], cdf])  # length n+1
    
    # For new dimension, sample at uniform points
    x_new = (np.arange(new_n + 1)) * (2.0 / new_n)
    # Interpolate CDF at new points
    y_new = np.interp(x_new, np.linspace(0, 2, n + 1), cdf_full)
    # New h values are differences
    h_new = np.diff(y_new) / (2.0 / new_n)
    return h_new

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def active_descent(h, max_steps=50, sigma_base=0.005):
    """Exact nonsmooth descent on the max objective."""
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for step in range(max_steps):
        values, grads = compute_profile(best)
        c = np.max(values)
        # Find near-worst lags
        threshold = c - sigma_base * c
        active_mask = values >= threshold
        if not np.any(active_mask):
            active_mask = values >= np.percentile(values, 90)
        
        # Weight by exponential of gap
        gaps = c - values[active_mask]
        weights = np.exp(-gaps / max(sigma_base * c, 1e-10))
        weights /= np.sum(weights) + 1e-15
        
        # Direction: negative weighted gradient
        direction = -np.sum(weights[:, None] * grads[active_mask, :], axis=0)
        direction -= np.mean(direction)  # preserve sum
        
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        
        if not improved:
            # Try smaller sigma to focus more
            sigma_base *= 0.5
            if sigma_base < 1e-6:
                break
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    return h


def compute_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def compute_grads(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    # Center lag
    grads[n - 1, :] = dx * (1.0 - 2.0 * h)
    for delta in range(1, n):
        idx = np.arange(n - delta)
        grads[n - 1 + delta, idx] += dx * (1.0 - h[idx + delta])
        grads[n - 1 + delta, idx + delta] += -dx * h[idx]
        idx2 = np.arange(delta, n)
        grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
        grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    return grads


def soft_max_obj(x, beta):
    vals = compute_values(x)
    mv = np.max(vals)
    shifted = beta * (vals - mv)
    log_sum = beta * mv + np.log(np.sum(np.exp(shifted)))
    return log_sum / beta


def soft_max_grad(x, beta):
    vals = compute_values(x)
    mv = np.max(vals)
    shifted = beta * (vals - mv)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    grads = compute_grads(x)
    return w @ grads


def eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def eq_jac(x):
    return np.ones(len(x))


def solve_soft(h_start, beta, maxiter):
    n = len(h_start)
    h_start = repair_feasible(h_start)

    def obj(x):
        v = soft_max_obj(x, beta)
        return v if np.isfinite(v) else 1e10

    def jac(x):
        g = soft_max_grad(x, beta)
        return g if np.all(np.isfinite(g)) else np.zeros(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h_start, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
    except Exception:
        return None, None
    if res.x is not None:
        h_sol = repair_feasible(res.x)
        s = exact_score(h_sol)
        if np.isfinite(s):
            return h_sol, s
    return None, None


def active_descent(h, max_steps=50, sigma=0.004):
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_s = exact_score(best)
    for _ in range(max_steps):
        vals = compute_values(best)
        c = np.max(vals)
        thresh = c - sigma * c
        mask = vals >= thresh
        if np.count_nonzero(mask) < 3:
            mask = vals >= np.percentile(vals, 85)
        gaps = c - vals[mask]
        w = np.exp(-gaps / max(sigma * c, 1e-10))
        w /= np.sum(w) + 1e-15
        grads = compute_grads(best)
        direction = -(w @ grads[mask, :])
        direction -= np.mean(direction)
        improved = False
        for alpha in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]:
            ht = repair_feasible(best + alpha * direction)
            s = exact_score(ht)
            if s < best_s - 1e-14:
                best = ht.copy()
                best_s = s
                improved = True
                break
        if not improved:
            sigma *= 0.6
            if sigma < 5e-5:
                break
    return best, best_s


def cdf_transfer(h, new_n):
    n = len(h)
    dx = 2.0 / n
    cdf = np.concatenate([[0.0], np.cumsum(h) * dx])
    x_src = np.linspace(0, 2, n + 1)
    x_tgt = np.linspace(0, 2, new_n + 1)
    y_tgt = np.interp(x_tgt, x_src, cdf)
    h_new = np.diff(y_tgt) / (2.0 / new_n)
    return h_new


def symmetrize(h):
    return (h + h[::-1]) / 2.0
