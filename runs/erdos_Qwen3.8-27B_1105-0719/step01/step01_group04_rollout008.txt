import numpy as np
import time
import cvxpy as cp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.argsort(h)
        rem = diff
        for i in idx:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        idx = np.argsort(-h)
        rem = -diff
        for i in idx:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            for i in range(len(h)):
                if h[i] < 1.0:
                    h[i] += min(diff2, 1.0 - h[i])
                    break
        else:
            for i in range(len(h)):
                if h[i] > 0.0:
                    h[i] += max(diff2, -h[i])
                    break
    return h

def project_feasible(y):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2} via scalar shift."""
    n = len(y)
    target = n / 2.0
    lo = y.min() - n
    hi = y.max() + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    h = np.clip(y - tau, 0.0, 1.0)
    # Fix tiny residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
            if len(idx) > 0:
                h[idx[0]] += min(diff, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx[0]] += max(diff, -h[idx[0]])
    return h

def compute_lag_values(h):
    """Compute all lag overlap values."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    for delta in range(0, n):
        if delta == 0:
            values[n - 1] = dx * np.sum(h * (1.0 - h))
        else:
            idx = np.arange(n - delta)
            values[n - 1 + delta] = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
            idx2 = np.arange(delta, n)
            values[n - 1 - delta] = dx * np.sum(h[idx2] * (1.0 - h[idx2 - delta]))
    return values

def compute_grad(h, active_idx):
    """Compute gradients only for given active indices. Returns (n_active, n) matrix."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    grads = np.zeros((len(active_idx), n))
    for row, k in enumerate(active_idx):
        delta = k - (n - 1)  # shift in samples: k-(n-1) is the shift
        if delta == 0:
            grads[row, :] = dx * (1.0 - 2.0 * h)
        elif delta > 0:
            idx = np.arange(n - delta)
            grads[row, idx] += dx * (1.0 - h[idx + delta])
            grads[row, idx + delta] += -dx * h[idx]
        else:
            d = -delta
            idx = np.arange(n - d)
            grads[row, idx] += dx * (1.0 - h[idx + d])
            grads[row, idx + d] += -dx * h[idx]
    return grads

def smoothed_minimax_grad(h, lam):
    """Compute gradient of smoothed minimax f_lambda(h) = (1/lam)*log(sum(exp(lam*Q_k)))."""
    n = len(h)
    values = compute_lag_values(h)
    max_val = np.max(values)
    # Numerically stable log-sum-exp
    shifted = lam * (values - max_val)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    # Get active indices (all, but weighted by softmax)
    active_idx = np.arange(len(values))
    G = compute_grad(h, active_idx)
    # Gradient = sum_k weights[k] * grad_k
    grad_f = weights @ G
    # Value of smoothed minimax
    f_val = max_val + np.log(np.sum(exp_shifted)) / lam
    return f_val, grad_f, values, weights

def fista_smoothed(h_start, lam, max_iter=5000, tol=1e-14):
    """FISTA on smoothed minimax with projection."""
    h = h_start.copy()
    y = h_start.copy()
    t_k = 1.0
    f_prev = np.inf
    L = 1.0  # Lipschitz constant estimate
    
    for it in range(max_iter):
        f_val, grad, values, weights = smoothed_minimax_grad(y, lam)
        # Gradient step
        z = y - grad / L
        # Project
        h_new = project_feasible(z)
        f_new_val, _, _, _ = smoothed_minimax_grad(h_new, lam)
        # Backtracking on L if needed
        if f_new_val > f_val + np.dot(grad, h_new - y) / L + 0.5 * L * np.dot(h_new - y, h_new - y) + 1e-10:
            L *= 1.5
            continue
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        h = h_new
        t_k = t_new
        if abs(f_prev - f_new_val) < tol:
            break
        f_prev = f_new_val
    
    return h

def active_descent(h, max_steps=80, time_limit=None, t_start=None):
    """Active-set weighted gradient descent polishing."""
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for step in range(max_steps):
        if time_limit is not None and (time.perf_counter() - t_start) > time_limit:
            break
        values = compute_lag_values(best)
        c = np.max(values)
        n_active = min(25, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_active]
        gaps = np.maximum(c - values[active_idx], 0.0)
        temperature = 0.002 * c + 1e-10
        weights = np.exp(-gaps / temperature)
        weights /= np.sum(weights) + 1e-15
        G = compute_grad(best, active_idx)
        direction = -weights @ G
        direction -= np.mean(direction)
        improved = False
        for alpha in [0.04, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-13:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        if not improved:
            for alpha in [0.0001, 5e-5, 2e-5, 1e-5]:
                h_trial = repair_feasible(best + alpha * direction)
                score = exact_score(h_trial)
                if score < best_score - 1e-13:
                    best = h_trial.copy()
                    best_score = score
                    break
            else:
                break
    return best, best_score

def trust_region_cvxpy(h_cur, values, lam, n_active_max=80, time_limit=45):
    """Solve trust-region QP using CVXPY."""
    n = len(h_cur)
    n_lags = len(values)
    
    active_idx = np.argsort(values)[::-1][:min(n_active_max, n_lags)]
    G = compute_grad(h_cur, active_idx)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
    ]
    for i, k in enumerate(active_idx):
        rhs = t_var - values[k] + G[i, :] @ h_cur
        constraints.append(G[i, :] @ h_var <= rhs)
    
    obj = cp.Minimize(t_var + (lam / 2.0) * cp.sum_squares(h_var - h_cur))
    prob = cp.Problem(obj, constraints)
    
    t0 = time.perf_counter()
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9)
        except Exception:
            return None
    if time.perf_counter() - t0 > time_limit:
        return None
    if h_var.value is None:
        return None
    h_new = repair_feasible(np.array(h_var.value, dtype=np.float64))
    return h_new

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)
