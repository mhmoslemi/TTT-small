def project_binary(h, n):

    n = len(h)
    m = n // 2
    idx = np.argsort(h)[::-1][:m]

    b[idx] = 1.0

def project_box_sum(h):

    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)

    """Transfer solution to target size via interpolation."""
    n_src = len(h_src)
    if n_src == n_target:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_src)

def lse_optimize(h0, tau, maxiter=200, deadline=None):
    """LSE smoothed optimization."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0, 1)
        profile = np.correlate(x, 1.0 - x, mode='full') * dx
        m = float(np.max(profile))
        shifted = (profile - m) / tau
        shifted = np.minimum(shifted, 0.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        obj = m + tau * np.log(sum_exp)
        weights = exp_vals / sum_exp
        
        # Gradient
        grad = np.zeros(n)
        for j in range(2 * n - 1):
            w = weights[j]
            if w < 1e-13:
                continue
            s = j - (n - 1)
            if s >= 0:
                k1 = np.arange(s, n)
                grad[k1] += w * dx * (1.0 - x[k1 - s])
                k2 = np.arange(0, n - s)
                grad[k2] -= w * dx * x[k2 + s]
            else:
                a = -s
                k1 = np.arange(0, n - a)
                grad[k1] += w * dx * (1.0 - x[k1 + a])
                k2 = np.arange(a, n)
                grad[k2] -= w * dx * x[k2 - a]
        return obj, grad
    
    # Use SLSQP
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    try:
        x0 = project_box_sum(h0.copy())
        res = minimize(obj_and_grad, x0, method='SLSQP', jac=True,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if n == 0:
        return h
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    h = np.clip(h, 0.0, 1.0)
    return h


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def transfer_to_size(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)


def greedy_construct_binary(n, seed):
    h = np.zeros(n)
    target = n // 2
    rng = np.random.RandomState(seed)
    order = rng.permutation(n)
    for _ in range(target):
        best_pos = -1
        best_c = float('inf')
        for pos in order:
            if h[pos] == 0:
                h[pos] = 1.0
                c = exact_objective(h)
                h[pos] = 0.0
                if c < best_c:
                    best_c = c
                    best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1.0
        else:
            break
    return h


def single_swap_local_search(b, deadline, max_rounds=5000):
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b, curr_c

def perturb_binary(b, k, rng):
    b = b.copy()
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for _ in range(k):
        if not ones or not zeros:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
    return b

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


class LSEObjFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(np.clip(x, 0, 1), self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(np.clip(x, 0, 1), self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))


def slsqp_lse(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    objf = LSEObjFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(objf.value, project_box_sum(h0.copy()), method='SLSQP',
                       jac=objf.gradient, bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(np.array(res.x, dtype=np.float64))
    except Exception:
        return project_box_sum(h0.copy())


def lse_refine(h0, deadline):

        c_new = exact_objective(h_new)

            h = h_new
