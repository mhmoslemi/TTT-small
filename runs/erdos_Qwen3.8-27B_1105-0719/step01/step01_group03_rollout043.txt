    _CUR['dx'] = 2.0 / float(n)

    _CUR['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_obj, 'jac': sum_eq_jac}]
    try:
        res = minimize(
            smooth_obj, h0, method='SLSQP', jac=smooth_jac,
            bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': float(ftol), 'disp': False}
        )
        x = res.x
        if x is None or not np.all(np.isfinite(x)):
            return None
        return project_feasible(x, target)
    except Exception:
        return None


def smooth_anneal_start(h, tau_list, maxiter, deadline):
    h = project_feasible(np.asarray(h, dtype=np.float64).reshape(-1), len(h) / 2.0)
    best_h = h.copy()
    best_c = exact_c5(h)
    h_cur = h
    c_cur = best_c
    for tau in tau_list:
        if deadline is not None and time_remains(deadline) < 20.0:
            break
        F_cur = smooth_value(h_cur, tau)
        h_new = smooth_solve(h_cur, tau, maxiter, 1e-10)
        if h_new is None:
            continue
        c_new = exact_c5(h_new)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        F_new = smooth_value(h_new, tau)
        if c_new < c_cur - 1e-12 or F_new < F_cur - 1e-8 or F_new <= F_cur + 1e-10:
            h_cur = project_feasible(h_new, len(h_new) / 2.0)
            c_cur = exact_c5(h_cur)
    return h_cur, best_h, best_c


def epi_obj(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    return float(x[-1])


def epi_obj_jac(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    g = np.zeros(x.size, dtype=np.float64)
    g[-1] = 1.0
    return g


def epi_eq_fun(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    return float(np.sum(x[:-1]) - _EPI['target'])


def epi_eq_jac(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    g = np.zeros(x.size, dtype=np.float64)
    g[:-1] = 1.0
    return g


def epi_ineq_fun(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    active = _EPI['active']
    if active is None or len(active) == 0:
        return np.zeros(0, dtype=np.float64)
    t = float(x[-1])
    prof = exact_profile(x[:-1])
    return t - prof[active]


def epi_ineq_jac(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    active = _EPI['active']
    if active is None or len(active) == 0:
        return np.zeros((0, x.size), dtype=np.float64)
    G = full_gradient_matrix(x[:-1])
    J = np.hstack((-G[active], np.ones((len(active), 1), dtype=np.float64)))
    return J


def active_epi_solve(h0, maxiter=80, ftol=1e-10, rounds=2, deadline=None):
    h = project_feasible(np.asarray(h0, dtype=np.float64).reshape(-1), len(h0) / 2.0)
    n = h.size
    target = 0.5 * float(n)
    best = exact_c5(h)
    thresholds = (2e-4, 1e-4, 5e-5, 2e-5)
    rounds = int(min(rounds, len(thresholds)))
    for r in range(rounds):
        if deadline is not None and time_remains(deadline) < 25.0:
            break
        prof = exact_profile(h)
        c = best
        th = thresholds[r]
        active = np.where(prof >= c - th)[0]
        if active.size < 3:
            k = min(8, max(3, n // 10))
            active = np.argsort(prof)[-k:]
        if active.size > 30:
            active = np.argsort(prof)[::-1][:30]
        _EPI['n'] = n
        _EPI['target'] = target
        _EPI['active'] = active
        x0 = np.concatenate((h, [c + 1e-7]))
        bounds = [(0.0, 1.0)] * n + [(0.0, c + 1.0)]
        cons = [
            {'type': 'eq', 'fun': epi_eq_fun, 'jac': epi_eq_jac},
            {'type': 'ineq', 'fun': epi_ineq_fun, 'jac': epi_ineq_jac}
        ]
        try:
            res = minimize(
                epi_obj, x0, method='SLSQP', jac=epi_obj_jac,
                bounds=bounds, constraints=cons,
                options={'maxiter': int(maxiter), 'ftol': float(ftol), 'disp': False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h2 = project_feasible(res.x[:-1], target)
                c2 = exact_c5(h2)
                if c2 < best:
                    h = h2
                    best = c2
        except Exception:
            pass
    return h, best


def active_descent(h, max_iters=50):
    h = project_feasible(np.asarray(h, dtype=np.float64).reshape(-1), len(h) / 2.0)
    best = exact_c5(h)
    steps = (0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005)
    no_improve = 0
    for _ in range(int(max_iters)):
        prof = exact_profile(h)
        c = float(np.max(prof))
        if c < best - 1e-13:
            best = c
        tau_w = max(1e-4, 0.005 * c)
        z = (prof - c) / tau_w
        z = np.clip(z, -30.0, 0.0)
        w = np.exp(z)
        s = float(np.sum(w))
        if s <= 1e-300 or not np.isfinite(s):
            w = np.zeros_like(prof)
            w[int(np.argmax(prof))] = 1.0
            s = 1.0
        else:
            w = w / s
        grad = weighted_gradient(h, w)
        grad -= grad.mean()
        norm = float(np.linalg.norm(grad))
        if norm < 1e-12:
            break
        grad /= norm
        ok = False
        for alpha in steps:
            h_try = project_feasible(h - alpha * grad, len(h) / 2.0)
            ct = exact_c5(h_try)
            if ct < best - 1e-12:
                h = h_try
                best = ct
                ok = True
                break
        if ok:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                break
    return h, best


def linear_min_vertex(g, target):
    g = np.asarray(g, dtype=np.float64).reshape(-1)
    n = g.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if not np.all(np.isfinite(g)):
        return project_feasible(np.zeros(n, dtype=np.float64), target)
    v = np.zeros(n, dtype=np.float64)
    order = np.argsort(g)
    rem = float(target)
    for idx in order:
        if rem <= 1e-14:
            break
        if rem + 1e-14 >= 1.0:
            v[idx] = 1.0
            rem -= 1.0
        else:
            v[idx] = max(0.0, min(1.0, rem))
            rem = 0.0
    return project_feasible(v, target)


def fw_stage(h, tau, max_iters=40, deadline=None):
    h = project_feasible(np.asarray(h, dtype=np.float64).reshape(-1), len(h) / 2.0)
    best = exact_c5(h)
    best_h = h.copy()
    no = 0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(int(max_iters)):
        if deadline is not None and time_remains(deadline) < 25.0:
            break
        F, w, prof = smooth_parts(h, tau)
        c = float(np.max(prof))
        if c < best - 1e-13:
            best = c
            best_h = h.copy()
        grad = weighted_gradient(h, w)
        v = linear_min_vertex(grad, len(h) / 2.0)
        if float(np.linalg.norm(v - h)) < 1e-10:
            break
        if float(np.dot(grad, v - h)) > 1e-10:
            no += 1
            if no > 6:
                break
            continue
        a = 0.0
        b = 1.0
        c1 = b - invphi * (b - a)
        d2 = a + invphi * (b - a)
        fc = smooth_value(np.clip(h + c1 * (v - h), 0.0, 1.0), tau)
        fd = smooth_value(np.clip(h + d2 * (v - h), 0.0, 1.0), tau)
        for __ in range(22):
            if fc <= fd:
                b = d2
                d2 = c1
                fd = fc
                c1 = b - invphi * (b - a)
                fc = smooth_value(np.clip(h + c1 * (v - h), 0.0, 1.0), tau)
            else:
                a = c1
                c1 = d2
                fc = fd
                d2 = a + invphi * (b - a)
                fd = smooth_value(np.clip(h + d2 * (v - h), 0.0, 1.0), tau)
        alpha = 0.5 * (a + b)
        local_best = best
        local_h = best_h
        vals = (alpha, 1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05)
        for al in vals:
            al = float(al)
            if al < 0.0 or al > 1.0:
                continue
            ht = project_feasible(h + al * (v - h), len(h) / 2.0)
            ct = exact_c5(ht)
            if ct < local_best - 1e-13:
                local_best = ct
                local_h = ht.copy()
        if local_best < best - 1e-12:
            h = local_h
            best = local_best
            best_h = local_h.copy()
            no = 0
            continue
        ht = project_feasible(h + alpha * (v - h), len(h) / 2.0)
        Fh = smooth_value(ht, tau)
        h = ht
        no += 1
        if no > 12:
            break
    return best_h, best


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    budget = float(budget_s)
    safety = 15.0 if budget > 100.0 else 5.0
    deadline = t_start + max(2.0, budget - safety)
    rng = np.random.RandomState(int(seed))

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).reshape(-1)
    except Exception:
        parent = None

    if parent is None or parent.size < 8 or not np.all(np.isfinite(parent)):
        parent = np.full(64, 0.5, dtype=np.float64)
    parent = project_feasible(parent, parent.size / 2.0)

    best_h = parent.copy()
    best_c = float("inf")
    best_n = int(parent.size)

    best_h, best_c, best_n = better_candidate(parent, best_h, best_c, best_n)
    best_h, best_c, best_n = better_candidate(symmetrize(parent), best_h, best_c, best_n)
    best_h, best_c, best_n = better_candidate(threshold_start(parent), best_h, best_c, best_n)
    if time_remains(deadline) > 20.0:
        best_h, best_c, best_n = better_candidate(np.power(np.clip(parent, 0.0, 1.0), 1.5), best_h, best_c, best_n)

    fixed1 = [32, 40, 48, 56, 64, 72, 80, 88, 96, 104, 112, 120, 140]
    dims1 = make_dim_list(best_n, fixed1)
    if len(dims1) > 16:
        dims1 = dims1[:16]
    quick_tau = [0.2, 0.1, 0.05, 0.02]
    for n in dims1:
        if time_remains(deadline) < 60.0:
            break
        n = int(n)
        h0 = resample_h(best_h, n)
        best_h, best_c, best_n = better_candidate(h0, best_h, best_c, best_n)
        final_h, local_h, local_c = smooth_anneal_start(h0, quick_tau, 50, deadline)
        best_h, best_c, best_n = better_candidate(local_h, best_h, best_c, best_n)
        best_h, best_c, best_n = better_candidate(final_h, best_h, best_c, best_n)

    if time_remains(deadline) > 120.0:

            extra2.append(best_n + off)
        extra2.append(best_n)
        dims2 = make_dim_list(best_n, extra2)

            n = int(n)

            best_h, best_c, best_n = better_candidate(h0, best_h, best_c, best_n)

            dim_h = h0.copy()
            dim_c = exact_c5(dim_h)

            variants = make_variants(h0, rng)
            max_starts = 4 if time_remains(deadline) > 260.0 else 2
            for s in variants[:max_starts]:
                if time_remains(deadline) < 40.0:
                    break
                final_h, local_h, local_c = smooth_anneal_start(s, focus_tau, 80, deadline)
                if local_c < dim_c:
                    dim_h = local_h.copy()
                    dim_c = local_c
                cf = exact_c5(final_h)
                if cf < dim_c:
                    dim_h = final_h.copy()
                    dim_c = cf
                best_h, best_c, best_n = better_candidate(local_h, best_h, best_c, best_n)
                best_h, best_c, best_n = better_candidate(final_h, best_h, best_c, best_n)

            if time_remains(deadline) > 50.0:

        best_h, best_c, best_n = better_candidate(final_h, best_h, best_c, best_n)
        if time_remains(deadline) > 45.0:
            h_new = smooth_solve(best_h, 0.002, 100, 1e-11)
            if h_new is not None:
                best_h, best_c, best_n = better_candidate(h_new, best_h, best_c, best_n)

    if time_remains(deadline) > 30.0:
        best_h, best_c, best_n = better_candidate(symmetrize(best_h), best_h, best_c, best_n)

    try:
        eval_val = evaluate_erdos_solution(best_h)
        if isinstance(eval_val, tuple):

        if isinstance(eval_val, (list, np.ndarray)):
            eval_val = eval_val[0]
        if np.isfinite(eval_val):
            final_c = min(float(final_c), float(eval_val))
    except Exception:
        pass

    if not np.isfinite(final_c):
        final_c = exact_c5(best_h)

    return best_h, float(final_c), int(best_h.size)
