    # For each lag index k (0 to 2n-2), d = k - (n-1)

        wk_k = w[k]
        if wk_k < 1e-15:

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / len(h))

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    return exps / s

def gradient_analytical(h, beta):

    dx = 2.0 / n

    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
