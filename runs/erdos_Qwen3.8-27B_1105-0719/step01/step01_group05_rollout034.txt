import numpy as np
from scipy.optimize import minimize
import time

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def profile_gradient(h, r, dx):
    n = len(h)
    k = r - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        s = n - k
        if s > 0:
            grad[:s] -= dx * h[k:]
            grad[k:k+s] += dx * (1.0 - h[:s])
    else:
        a = -k
        s = n - a
        if s > 0:
            grad[:s] += dx * (1.0 - h[a:])
            grad[a:a+s] -= dx * h[:s]
    return grad

def project_box_sum(h, target):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff/cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff/cnt, 0.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmax(h < 1.0) if diff > 0 else np.argmin(h > 0.0)
        # Actually let's just add to first feasible
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                h[np.where(mask)[0][0]] += min(diff, 1.0 - h[np.where(mask)[0][0]])
        else:
            mask = h > 0.0
            if np.any(mask):
                h[np.where(mask)[0][0]] += max(diff, -h[np.where(mask)[0][0]])
    return h

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        # Alternating
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        # Block
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        # Random
        h = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        # Cosine
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(0, 3)
        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.4 * np.cos(2*np.pi*freq*x/n + phase)
    elif mode == 4:
        # Two blocks
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[n//2:n//2+q] = 1.0
    elif mode == 5:
        # Noisy uniform
        h = np.full(n, 0.5) + rng.randn(n) * 0.2
    else:
        h = np.full(n, 0.3 + 0.4 * (np.arange(n) / n))
    return project_box_sum(h, target)

def subgradient_refine(h, max_iters=5000, target_c=0.38, deadline=None):
    """Projected subgradient method for the maximin problem."""
    n = len(h)
    dx = 2.0 / n
    target = 0.5 * n
    h = project_box_sum(h.copy(), target)
    best_h = h.copy()
    best_c = exact_objective(h)
    
    base_step = 0.01
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Find active lags (within tolerance of max)
        tol = max(0.005, 0.05 * (c - 0.3))
        active = np.where(profile >= c - tol)[0]
        
        # Average gradient over active lags
        grad = np.zeros(n)
        for r in active:
            grad += profile_gradient(h, r, dx)
        grad /= len(active)
        
        # Step size: Polyak-like
        gap = c - target_c
        if gap <= 0:
            break
        gnorm = np.dot(grad, grad)
        if gnorm < 1e-20:
            break
        alpha = min(0.1, gap / max(gnorm, 1e-10))
        
        # Projected step
        h_new = project_box_sum(h - alpha * grad, target)
        c_new = exact_objective(h_new)
        
        if c_new <= c + 1e-12:
            h = h_new
        else:
            # Try smaller step
            alpha *= 0.3
            h_new = project_box_sum(h - alpha * grad, target)
            c_new = exact_objective(h_new)
            if c_new <= c + 1e-12:
                h = h_new
            else:
                # Reduce step size
                alpha *= 0.5
                base_step *= 0.98
                h = project_box_sum(h - alpha * grad, target)
    
    return best_h, best_c

def slsqp_full_constraint(h_start, deadline, max_iter=200):
    """SLSQP with all constraints: minimize c s.t. sum(h)=n/2, 0<=h<=1, profile[r]-c<=0"""
    n = len(h_start)
    dx = 2.0 / n
    target = 0.5 * n
    n_profile = 2 * n - 1
    
    h0 = project_box_sum(h_start, target)
    c0 = exact_objective(h0) + 1e-6
    
    # Variable vector: [h_0, ..., h_{n-1}, c]
    x0 = np.concatenate([h0, [c0]])
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(1e-8, 1.0)]
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    # Equality constraint: sum(h) - target = 0
    def eq_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    # Inequality constraints: c - profile[r] >= 0
    # (SLSQP requires fun >= 0 for inequalities)
    for r in range(n_profile):
        def ineq_fun(x, r=r):
            h = x[:n]
            c_val = x[-1]
            return c_val - compute_profile(h)[r]
        def ineq_jac(x, r=r):
            h = x[:n]
            g = np.zeros(n + 1)
            pg = profile_gradient(h, r, dx)
            g[:n] = -pg
            g[-1] = 1.0
            return g
        constraints.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.x is not None:
            h_new = project_box_sum(res.x[:n], target)
            c_new = exact_objective(h_new)
            return h_new, c_new
    except Exception:
        pass
    return h0, c0
