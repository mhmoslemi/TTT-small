import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(y, n):
    target = n / 2.0
    y = np.array(y, dtype=float)
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    y = np.clip(y, -2.0, 3.0)
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            adj = residual / len(eligible)
            h[eligible] = np.clip(h[eligible] + adj, 0.0, 1.0)
    return h

def pair_transfer_optimize(h, n_iter, rng, verbose=False):
    """Iteratively try pair transfers to reduce C5."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for it in range(n_iter):
        improved = False
        # Try random pairs
        for _ in range(100):
            i, j = rng.choice(n, size=2, replace=False)
            if abs(best_h[i] - best_h[j]) < 1e-10:
                continue
            # Try several transfer amounts
            cap_f = min(1.0 - best_h[i], best_h[j])  # transfer j -> i
            cap_b = min(1.0 - best_h[j], best_h[i])  # transfer i -> j
            for cap, inc, dec in [(cap_f, i, j), (cap_b, j, i)]:
                if cap < 1e-10:
                    continue
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    tau = cap * frac
                    new_h = best_h.copy()
                    new_h[inc] += tau
                    new_h[dec] -= tau
                    c5_new = compute_c5(new_h)
                    if c5_new < best_c5 - 1e-12:
                        best_h = new_h
                        best_c5 = c5_new
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c5

def subgradient_step(h, n, step_size, rng):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Find active positions (within tolerance of max)
    tol = 1e-6
    active = np.where(corr >= max_val - tol)[0]
    
    # Compute subgradient (average of gradients at active positions)
    grad = np.zeros(n)
    for pos in active:
        l = pos - (n - 1)
        for m in range(n):
            # h[m] as first factor: valid if 0 <= m+l < n
            if 0 <= m + l < n:
                grad[m] += (1.0 - h[m + l]) / len(active)
            # h[m] as second factor: valid if 0 <= m-l < n
            if 0 <= m - l < n:
                grad[m] -= h[m - l] / len(active)
    
    # Projected subgradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
