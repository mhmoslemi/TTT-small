def swap_delta(h, prof, a, b, n):
    """Compute the change in profile when h[a]:1->0 and h[b]:0->1."""
    # For each lag t (shift), the valid i range is [max(0,-t), min(n-1,n-1-t)]
    # Delta contributions:
    # 1. a was 1, now 0 (as first factor): -(1-h[a+t]) for t where a is valid first factor
    #    a valid as first factor at lag t: max(0,-t) <= a <= min(n-1,n-1-t)
    #    i.e., -min(a, n-1-a) <= t <= min(n-1-a, ...) 
    #    Simplify: t must satisfy a >= max(0,-t) and a <= n-1-t
    #    a >= 0 (always) and a >= -t => t >= -a
    #    a <= n-1 => a+t <= n-1 => t <= n-1-a
    #    Also need 0 <= a+t < n which gives -a <= t <= n-1-a (same)
    #    
    # 2. b was 0, now 1 (as first factor): +(1-h[b+t]) for t where b is valid first factor
    #    -b <= t <= n-1-b
    #    
    # 3. a was 1, now 0 (as second factor, i+t=a, i=a-t): +(1-0)=1... wait
    #    Actually: the term is h[i]*(1-h[i+t]). When i+t=a and h[a] changes from 1 to 0:
    #    (1-h[a]) changes from 0 to 1
    #    So the term h[a-t]*(1-h[a]) changes from h[a-t]*0 to h[a-t]*1
    #    Delta: +h[a-t] for t where a-t is valid as i
    #    a-t valid as i: 0 <= a-t < n and 0 <= (a-t)+t = a < n (true)
    #    So: 0 <= a-t < n => a-n+1 <= t <= a
    #    
    # 4. b was 0, now 1 (as second factor, i+t=b, i=b-t): 
    #    (1-h[b]) changes from 1 to 0
    #    Term h[b-t]*(1-h[b]) changes from h[b-t]*1 to h[b-t]*0
    #    Delta: -h[b-t] for t where b-t valid as i
    #    0 <= b-t < n => b-n+1 <= t <= b
    
    delta = np.zeros(2*n - 1)
    # Contribution 1: a as first factor, -(1-h[a+t])
    t_lo = max(-(n-1), -a)
    t_hi = min(n-1, n-1-a)
    for t in range(t_lo, t_hi + 1):
        j = a + t  # this is the index in h that we look up
        if 0 <= j < n:
            delta[t + (n-1)] += -(1 - h[j])
    
    # Contribution 2: b as first factor, +(1-h[b+t])
    t_lo = max(-(n-1), -b)
    t_hi = min(n-1, n-1-b)
    for t in range(t_lo, t_hi + 1):
        j = b + t
        if 0 <= j < n:
            delta[t + (n-1)] += (1 - h[j])
    
    # Contribution 3: a as second factor (i+t=a), +h[a-t]
    t_lo = max(-(n-1), a - (n-1))  # a-t >= 0 => t <= a; a-t <= n-1 => t >= a-n+1
    t_hi = min(n-1, a)
    for t in range(t_lo, t_hi + 1):
        i = a - t
        if 0 <= i < n:
            delta[t + (n-1)] += h[i]
    
    # Contribution 4: b as second factor (i+t=b), -h[b-t]
    t_lo = max(-(n-1), b - (n-1))
    t_hi = min(n-1, b)
    for t in range(t_lo, t_hi + 1):
        i = b - t
        if 0 <= i < n:
            delta[t + (n-1)] += -h[i]
    
    return delta

For the continuous case, the objective is:
f(h) = max_k g_k(h) where g_k(h) = sum_i h[i]*(1-h[i+k]) * dx

This is a DC (difference of convex) function since each g_k is bilinear.

One approach: **Proximal gradient method with the max smoothed by a softmax approximation.**
f_ν(h) = (1/ν) * log(sum_k exp(ν * g_k(h)))

As ν → ∞, f_ν → f. We can use a moderate ν (like 50-100) for a smooth approximation and do gradient descent.

The gradient of f_ν is:
∇f_ν = (1/ν) * sum_k exp(ν*g_k) * ∇g_k / sum_k exp(ν*g_k)

This is a weighted average of the gradients of the g_k's, weighted by how close each g_k is to the maximum.

This is essentially what the previous code does in `local_descent`, but with a softmax weighting instead of uniform weighting over top-A lags.

Let me also think about the gradient computation more carefully. I'll use a vectorized approach:

For the weighted gradient with weights w (length M):
grad[j] = dx * Σ_k w_k * [(1-h[j+k]) * 1{0≤j+k<n} - h[j-k] * 1{0≤j-k<n}]

I can split this into two parts:
Part A: dx * Σ_k w_k * (1-h[j+k]) * 1{0≤j+k<n}
Part B: dx * Σ_k w_k * h[j-k] * 1{0≤j-k<n}

For Part A: Let u = 1-h. Then Part A[j] = dx * Σ_k w_k * u[j+k] * 1{0≤j+k<n}
This is like a "correlation" where for each shift k, we add w_k * u shifted by -k.

I can compute this by:
- For k ≥ 0: u[j+k] is valid for j = 0 to n-1-k. Spawn: for j in [0, n-1-k], add w_k * u[j+k] to PartA[j].
  This is: PartA[0:n-k] += w_k * u[k:n]

- For k < 0 (let d = -k): u[j-d] is valid for j = d to n-1. For j in [d, n-1], add w_k * u[j-d] to PartA[j].
  This is: PartA[d:n] += w_k * u[0:n-d]

Similarly for Part B:
- For k ≥ 0: h[j-k] valid for j = k to n-1. PartB[k:n] += w_k * h[0:n-k]
- For k < 0 (d=-k): h[j+d] valid for j = 0 to n-1-d. PartB[0:n-d] += w_k * h[d:n]

So:
grad = dx * (PartA - PartB)

This can be computed in a loop over k (M=279 iterations), each doing O(n) numpy ops. Total: 279 * O(n) = 279 * 140 ≈ 39K numpy scalar ops. Should take about 1-2ms.

Let me implement this:

```python
def weighted_gradient(h, w, dx):
    """Compute weighted gradient of the overlap profile."""
    n = len(h)
    M = 2 * n - 1
    nc = n - 1
    grad_A = np.zeros(n)
    grad_B = np.zeros(n)
    
    for m in range(M):
        k = m - nc
        wk = w[m]
        if wk < 1e-15:
            continue
        if k >= 0:
            # Part A: grad_A[0:n-k] += wk * (1-h[k:n])
            grad_A[:n-k] += wk * (1 - h[k:])
            # Part B: grad_B[k:] += wk * h[:n-k]
            grad_B[k:] += wk * h[:n-k]
        else:
            d = -k
            # Part A: grad_A[d:] += wk * (1-h[0:n-d])
            grad_A[d:] += wk * (1 - h[:n-d])
            # Part B: grad_B[:n-d] += wk * h[d:]
            grad_B[:n-d] += wk * h[d:]
    
    return dx * (grad_A - grad_B)
```

This is clean and should be fast. Let me verify the correctness:

For k=0 (m=nc): 
- Part A: grad_A[0:n] += wk * (1-h[0:n]) → grad_A += wk * (1-h)
- Part B: grad_B[0:n] += wk * h[0:n] → grad_B += wk * h
- Net: dx * wk * (1-h-h) = dx * wk * (1-2h) ✓ (matches the lag-0 derivative)

For k=1 (m=nc+1):
- Part A: grad_A[0:n-1] += wk * (1-h[1:n])
  This means for j in [0, n-2], grad_A[j] += wk * (1-h[j+1]) ✓
- Part B: grad_B[1:] += wk * h[0:n-1]
  This means for j in [1, n-1], grad_B[j] += wk * h[j-1] ✓

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def compute_swap_delta(h, a, b, n):
    M = 2 * n - 1
    nc = n - 1
    delta = np.zeros(M)
    t_lo = max(-a, -nc)
    t_hi = min(n - 1 - a, nc)
    if t_lo <= t_hi:
        tv = np.arange(t_lo, t_hi + 1)
        delta[tv + nc] -= (1.0 - h[a + tv])
    t_lo = max(-b, -nc)
    t_hi = min(n - 1 - b, nc)
    if t_lo <= t_hi:
        tv = np.arange(t_lo, t_hi + 1)
        delta[tv + nc] += (1.0 - h[b + tv])
    t_lo = max(a - nc, -nc)
    t_hi = min(a, nc)
    if t_lo <= t_hi:
        tv = np.arange(t_lo, t_hi + 1)
        delta[tv + nc] += h[a - tv]
    t_lo = max(b - nc, -nc)
    t_hi = min(b, nc)
    if t_lo <= t_hi:
        tv = np.arange(t_lo, t_hi + 1)
        delta[tv + nc] -= h[b - tv]
    return delta

def binary_local_search(h_bin, time_limit=20.0, n_trials=80, seed=42):
    n = len(h_bin)
    h = h_bin.copy().astype(np.float64)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    prof = get_profile(h)
    best = h.copy()
    best_score = float(np.max(prof))
    t0 = time.perf_counter()
    rng = np.random.default_rng(seed)
    stuck = 0
    for it in range(200000):
        if time.perf_counter() - t0 > time_limit:
            break
        current_max = float(np.max(prof))
        if current_max < best_score - 1e-12:
            best_score = current_max
            best = h.copy()
            stuck = 0
        else:
            stuck += 1
        best_imp = 0.0
        best_a, best_b = -1, -1
        best_dv = None
        nt = n_trials if stuck < 200 else min(n_trials, 300)
        for _ in range(nt):
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            sv = compute_swap_delta(h, a, b, n)
            nm = float(np.max(prof + sv))
            imp = current_max - nm
            if imp > best_imp:
                best_imp = imp
                best_a, best_b = a, b
                best_dv = sv
        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            prof += best_dv
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            if stuck < 500:
                a = ones[rng.integers(len(ones))]
                b = zeros[rng.integers(len(zeros))]
                sv = compute_swap_delta(h, a, b, n)
                h[best_a] = 0.0 if best_a >= 0 else h[a]
                h[a] = 0.0
                h[b] = 1.0
                prof += sv
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                break
    return best, best_score

def weighted_gradient(h, w, dx):
    n = len(h)
    M = 2 * n - 1
    nc = n - 1
    gA = np.zeros(n)
    gB = np.zeros(n)
    for m in range(M):
        k = m - nc
        wk = w[m]
        if wk < 1e-18:
            continue
        if k >= 0:
            gA[:n - k] += wk * (1.0 - h[k:])
            gB[k:] += wk * h[:n - k]
        else:
            d = -k
            gA[d:] += wk * (1.0 - h[:n - d])
            gB[:n - d] += wk * h[d:]
    return dx * (gA - gB)

def softmax_grad_descent(h0, time_limit=30.0, nu=40.0, max_iters=50000, seed=42):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    rng = np.random.default_rng(seed)
    step_scale = 1.0 / (nu + 10.0)
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        prof = get_profile(h)
        score = float(np.max(prof))
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
        ps = prof - np.max(prof)
        w = np.exp(nu * ps)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones_like(prof) / len(prof)
        else:
            w /= wsum
        grad = weighted_gradient(h, w, dx)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        alpha = step_scale
        improved = False
        for _ in range(15):
            ht = repair_feasible(h - alpha * grad)
            st = exact_score(ht)
            if st < best_score - 1e-11:
                h = ht
                best_h = h.copy()
                best_score = st
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break
        if not improved and it % 200 == 199:
            eps = 0.002
            z = rng.standard_normal(n)
            z -= z.mean()
            h = repair_feasible(h + eps * z)
    return best_h, best_score

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_solve(h0, t_init, time_limit=30.0, maxiter=80):
    n = len(h0)
    M = 2 * n - 1

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        n_inner = len(h)
        dx = 2.0 / n_inner
        G = np.zeros((M, n_inner))
        for m in range(M):
            k = m - (n_inner - 1)
            if k >= 0:
                if k < n_inner:
                    G[m, :n_inner - k] += dx
                    G[m, k:] -= dx
            else:
                d = -k
                if d < n_inner:
                    G[m, d:] += dx
                    G[m, :n_inner - d] -= dx
        J = np.zeros((1 + M, n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -G
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_init + 1.0)]
    x0 = np.concatenate([h0, [t_init]])
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair_feasible(res.x[:n])
    except Exception:
        pass
    return None

    def tleft():
