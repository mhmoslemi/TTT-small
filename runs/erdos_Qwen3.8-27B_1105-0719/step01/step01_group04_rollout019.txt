    n = len(h)
    F = np.zeros(n)
    # Prefix sums
    P = np.zeros(n + 1)
    P[1:] = np.cumsum(h)
    # Suffix sums: S_m = sum(h[n-m:])
    # S_m = P[n] - P[n-m]
    # A_m = sum(h[:n-m] * h[m:])
    for m in range(n):
        if m == 0:
            A_m = np.dot(h, h)
            b = 0.0
        else:
            A_m = np.dot(h[:n-m], h[m:])
            p_m = P[m]
            s_m = P[n] - P[n-m]
            b = min(p_m, s_m)
        F[m] = A_m + b
    return F

    """Compute G[m, k] = dF_m/dh_k for all m, k."""
    n = len(h)
    G = np.zeros((n, n))
    P = np.zeros(n + 1)
    P[1:] = np.cumsum(h)
    total = P[n]
    
    for m in range(n):

            # dA_0/dh_k = 2*h[k], db_0/dh_k = 0
            G[0, :] = 2.0 * h

            G[m, :n-m] += h[m:]
            G[m, m:] += h[:n-m]

            if p_m < s_m - 1e-15:
                G[m, :m] += 1.0
            elif s_m < p_m - 1e-15:
                G[m, n-m:] += 1.0
            else:
                G[m, :m] += 0.5
                G[m, n-m:] += 0.5
    return G

    obj = -logsumexp / beta  # This is the smooth min

    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient
    G = compute_grad_matrix(x)
    grad = -np.dot(weights, G)  # d/dx of -smooth_min = -sum(w_m * G[m,:])
    
    return obj, grad
