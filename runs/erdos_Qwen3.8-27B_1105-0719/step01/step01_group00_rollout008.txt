import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def build_h_from_q(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, np.array([0.5]), 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    if n % 2 == 0:
        q = 0.5 * (h[:m] + 1.0 - h[n - 1:n - 1 - m:-1])
    else:
        q = 0.5 * (h[:m] + 1.0 - h[n - 1:n - m:-1])
    return np.clip(q, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    dx = 2.0 / len(h)
    return float(compute_corrs(h).max() * dx)


def grad_corr_h(h, d):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n, d + n)
    if lo1 < hi1:
        grad[lo1:hi1] += 1.0 - h[lo1 - d:hi1 - d]
    lo2 = max(0, -d)
    hi2 = min(n, n - d)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + d:hi2 + d]
    return grad


def grad_corr_q(q, n, r):
    m = n // 2
    h = build_h_from_q(q, n)
    d = r - (n - 1)
    gh = grad_corr_h(h, d)
    return gh[:m] - gh[m:n][::-1]


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def smooth_obj_q(x, n, beta, dx):
    q = x
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h) * dx
    shifted = corrs - corrs.max()
    return logsumexp(beta * shifted) / beta + corrs.max()


def smooth_grad_q(x, n, beta, dx):
    m = n // 2
    q = x
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h) * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(m)
    active = np.where(w > 1e-12)[0]
    for r in active:
        grad += w[r] * grad_corr_q(q, n, r) * dx
    return grad


def epigraph_solve(q0, n, active_lags, t0_val, maxiter):
    m = n // 2
    dx = 2.0 / n
    h0 = build_h_from_q(q0, n)
    corrs0 = compute_corrs(h0) * dx
    t_init = corrs0.max() + 1e-6

    def epig_fun(xv):
        return xv[m]

    def epig_jac(xv):
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g

    cons = []
    for r in active_lags:
        def cf(xv, rr=r):
            q = xv[:m]
            h = build_h_from_q(q, n)
            return xv[m] - compute_corrs(h)[rr] * dx

        def cj(xv, rr=r):
            q = xv[:m]
            gc = grad_corr_q(q, n, rr) * dx
            g = np.zeros(m + 1)
            g[:m] = -gc
            g[m] = 1.0
            return g

        cons.append({'type': 'ineq', 'fun': cf, 'jac': cj})

    x0 = np.concatenate([q0, [t_init]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    try:
        res = minimize(epig_fun, x0, jac=epig_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        q_out = q0.copy()
    return q_out


def epigraph_loop(q0, n, max_rounds, time_limit_fn, max_active=80):
    m = n // 2
    dx = 2.0 / n
    q = q0.copy()
    best_q = q.copy()
    best_score = exact_score(build_h_from_q(q, n))

    for rnd in range(max_rounds):
        if time_limit_fn() < 5:
            break
        h = build_h_from_q(q, n)
        corrs = compute_corrs(h) * dx
        max_c = corrs.max()
        t_current = max_c

            best_score = max_c
            best_q = q.copy()

        # Select active lags: top max_active plus some neighbors
        sorted_idx = np.argsort(corrs)[::-1]
        active = sorted_idx[:max_active]
        # Add neighbors of top ones
        for r in sorted_idx[:20]:
            if r > 0 and r - 1 not in active:
                active = np.append(active, r - 1)
            if r < len(corrs) - 1 and r + 1 not in active:
                active = np.append(active, r + 1)
        active = np.unique(active)

        maxiter = 400 if len(active) < 50 else 300
        q_new = epigraph_solve(q, n, active, t_current, maxiter)

        # Check hidden violations
        h_new = build_h_from_q(q_new, n)
        corrs_new = compute_corrs(h_new) * dx
        t_new = corrs_new.max()

        violations = np.where(corrs_new > t_new + 1e-8)[0]
        if len(violations) > 0:
            active_v = np.unique(np.concatenate([active, violations]))
            if len(active_v) <= len(corrs):
                q_new = epigraph_solve(q_new, n, active_v, t_new, maxiter)
                h_new = build_h_from_q(q_new, n)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_q = q_new.copy()
        q = q_new

    return best_q, best_score


def smooth_optimize_q(q0, n, betas, maxiter_each):
    m = n // 2
    dx = 2.0 / n
    q = q0.copy()
    best_q = q.copy()
    best_score = exact_score(build_h_from_q(q, n))

    for beta in betas:
        def obj(xv):
            return smooth_obj_q(xv, n, beta, dx)

        def jac(xv):
            return smooth_grad_q(xv, n, beta, dx)

        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(obj, q, jac=jac, method='SLSQP',
                           bounds=bounds, options={'maxiter': maxiter_each, 'ftol': 1e-14})
            q_cand = np.clip(res.x, 0.0, 1.0)
        except Exception:
            q_cand = q.copy()

        score_cand = exact_score(build_h_from_q(q_cand, n))
        if score_cand < best_score - 1e-14:
            best_score = score_cand
            best_q = q_cand.copy()
        q = q_cand

    return best_q, best_score


def subgradient_q(q0, n, max_iter, alpha_init=0.3):
    m = n // 2
    dx = 2.0 / n
    q = q0.copy()
    best_q = q.copy()
    best_score = exact_score(build_h_from_q(q, n))
    alpha = alpha_init
    no_improve = 0

    for it in range(max_iter):
        h = build_h_from_q(q, n)
        corrs = compute_corrs(h) * dx
        max_c = corrs.max()
        threshold = max_c * 0.95
        active = np.where(corrs >= threshold)[0]

        grad = np.zeros(m)
        for r in active:
            grad += grad_corr_q(q, n, r) * dx
        grad /= len(active)

        q_new = np.clip(q - alpha * grad, 0.0, 1.0)
        score_new = exact_score(build_h_from_q(q_new, n))

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_q = q_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if score_new < max_c * dx:
            q = q_new
            alpha = min(alpha * 1.03, 0.5)
        else:
            alpha *= 0.95

        if no_improve > 3000 or alpha < 1e-8:
            break

    return best_q, best_score
