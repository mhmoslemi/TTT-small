def coord_solve_j(h, j, n):
    """Find optimal h[j] given all other coordinates fixed."""
    a = np.zeros(2*n-1)  # coefficient of h[j] for each lag
    b = np.zeros(2*n-1)  # constant part
    
    for d in range(-(n-1), n):
        # Compute a[d] and b[d]
        # corrs[n-1+d] = sum_i h[i]*(1-h[i-d]) for valid i
        # h[j] contributes to this when:
        #   i = j (if valid range for this d) OR i-d = j (if valid range)
        
        # For d >= 0: i ranges from d to n-1
        #   i=j valid if d <= j <= n-1
        #   i-d=j valid if d <= j+d <= n-1, i.e., j <= n-1-d
        # For d < 0: i ranges from 0 to n-1+d
        #   i=j valid if j <= n-1+d
        #   i-d=j valid if 0 <= j+d <= n-1+d, i.e., -d <= j <= n-1
        
        m = n - 1 + d  # index in corrs array
        val = 0.0
        coeff = 0.0
        
        if d >= 0:
            i_min, i_max = d, n - 1
        else:
            i_min, i_max = 0, n - 1 + d
        
        for i in range(i_min, i_max + 1):
            idx2 = i - d
            if i == j:
                # h[j]*(1-h[idx2])
                val += h[j] * (1 - h[idx2])
                coeff += (1 - h[idx2])
            elif idx2 == j:
                # h[i]*(1-h[j])
                val += h[i] * (1 - h[j])
                coeff += (-h[i])
            else:
                val += h[i] * (1 - h[idx2])
        
        a[m] = coeff
        b[m] = val - coeff * h[j]
    
    # Now find t in [0,1] minimizing max_m (a[m]*t + b[m])
    t_star = solve_1d_minmax(a, b)
    return t_star

    val_0 = b.max()
    val_1 = (a + b).max()

    # Otherwise, use binary search on subgradient

        return (a * t + b).max()

    lo, hi = 0.0, 1.0

def subgradient_step(h, dx, n, step_size):
    """One subgradient step."""
    corrs = compute_corrs(h)
    max_c = corrs.max()
    tol = max_c * 0.01
    active = np.where(corrs >= max_c - tol)[0]
    
    grad = np.zeros(n)
    for m in active:
        d = m - (n - 1)
        if d >= 0:
            # grad[j] += (1-h[j-d]) for d<=j, and -h[j+d] for j+d<n
            for j in range(d, n):
                grad[j] += (1.0 - h[j - d])
            for j in range(0, n - d):
                grad[j] += (-h[j + d])
        else:

            for j in range(0, n - dp):
                grad[j] += (1.0 - h[j + dp])
            for j in range(dp, n):
                grad[j] += (-h[j - dp])

    grad *= dx
    
    h_new = h - step_size * grad
    target_sum = n / 2.0

    return h_new

        vals = a * mid + b
        mv = vals.max()
        active = a[vals >= mv - 1e-12]
        if active.min() >= -1e-12:
            hi = mid
        elif active.max() <= 1e-12:
            lo = mid
        else:
            break
    return (lo + hi) / 2


def compute_coeff_a(h, j, n):
    a = np.zeros(2 * n - 1)
    d_max1 = min(j, n - 1)
    if d_max1 >= 0:
        a[n - 1:n + d_max1] += 1.0 - h[j:j - d_max1 - 1:-1]
    d_max2 = min(n - 1 - j, n - 1)
    if d_max2 >= 0:
        a[n - 1:n + d_max2] += -h[j:j + d_max2 + 1]
    dp_max1 = min(n - 1 - j, n - 1)
    if dp_max1 >= 1:
        a[n - 2:n - 2 - dp_max1:-1] += 1.0 - h[j + 1:j + 1 + dp_max1]
    dp_max2 = min(j, n - 1)
    if dp_max2 >= 1:
        a[n - 2:n - 2 - dp_max2:-1] += -h[j - 1:j - 1 - dp_max2:-1]
    return a


def coord_descent(h, n, max_passes=200, tol=1e-13):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = c5_val(h, dx)
    best_h = h.copy()
    no_improve = 0

    for p in range(max_passes):
        corrs = compute_corrs(h)

        for j in range(n):
            hj_old = h[j]
            a = compute_coeff_a(h, j, n)
            b = corrs - a * hj_old
            t_star = solve_1d_minmax(a, b)
            if abs(t_star - hj_old) > 1e-14:
                h[j] = t_star
                corrs += a * (t_star - hj_old)

        h = project_feasible(h, target_sum)
        new_c5 = c5_val(h, dx)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 15:
            break

    return best_h, best_c5


def local_search(h, n, dx, num_iters=500):
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = c5_val(h, dx)
    best_h = h.copy()
    deltas = np.array([0.05, 0.02, 0.01, 0.1, 0.005, 0.15, 0.2])

    for it in range(num_iters):
        improved = False
        j = it % n
        for d in deltas:
            for sign in [1.0, -1.0]:
                h_try = h.copy()
                h_try[j] = np.clip(h[j] + sign * d, 0.0, 1.0)
                h_try = project_feasible(h_try, target_sum)
                c5_try = c5_val(h_try, dx)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    h = h_try
                    improved = True
        if not improved and it > 2 * n:
            break
    return best_h, best_c5


def subgradient_descend(h, n, max_iter=3000):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = c5_val(h, dx)
    best_h = h.copy()
    alpha = 0.3
    no_improve = 0

    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        threshold = max_c * 0.95
        active = np.where(corrs >= threshold)[0]

        grad = np.zeros(n)
        for m in active:
            d = m - (n - 1)
            if d >= 0:
                if d == 0:
                    grad += 1.0 - 2.0 * h
                else:
                    for j in range(d, n):
                        grad[j] += (1.0 - h[j - d])
                    for j in range(0, n - d):
                        grad[j] += (-h[j + d])
            else:
                dp = -d
                for j in range(0, n - dp):
                    grad[j] += (1.0 - h[j + dp])
                for j in range(dp, n):
                    grad[j] += (-h[j - dp])

        grad /= len(active)
        grad *= dx

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = c5_val(h_new, dx)

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            h = h_new
            alpha = min(alpha * 1.02, 0.5)
            no_improve = 0
        else:
            alpha *= 0.97
            no_improve += 1
        if no_improve > 2000 or alpha < 1e-6:
            break

    return best_h, best_c5


def make_random_start(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = n // 2
    h[idx[:n_ones]] = 1.0
    frac = n / 2.0 - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def make_periodic_start(n, period, fraction, rng):
    h = np.zeros(n)
    pattern = np.zeros(period)
    n_fill = int(period * fraction)
    idx = rng.permutation(period)
    pattern[idx[:n_fill]] = 1.0
    h = np.tile(pattern, n // period + 1)[:n]
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Optimize initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        dx0 = 2.0 / n0
        h0 = project_feasible(h0, n0 / 2.0)

        h_opt, c5_opt = coord_descent(h0, n0, max_passes=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n0

        if remaining() > 20:
            h_ls, c5_ls = local_search(best_h, best_n, 2.0 / best_n, 300)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

        if remaining() > 20:
            h_opt2, c5_opt2 = coord_descent(best_h, best_n, max_passes=200)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    # Phase 2: Try various n values with diverse starts
    n_list = [56, 60, 64, 72, 80, 84, 90, 96, 100, 108, 112, 120, 126, 128, 140, 144, 160]

    for n_try in n_list:
        if remaining() < 25:
            break
        dx_try = 2.0 / n_try
        ts = n_try / 2.0

        n_starts = 4 if remaining() > 300 else (2 if remaining() > 80 else 1)

        for trial in range(n_starts):
            if remaining() < 15:
                break

            if trial == 0:
                h_start = make_random_start(n_try, rng)
            elif trial == 1:
                h_start = make_periodic_start(n_try, max(2, n_try // 7), 0.5, rng)
            elif trial == 2:
                h_start = make_periodic_start(n_try, max(2, n_try // 5), 0.5, rng)
            else:
                h_start = rng.random(n_try)
                h_start = project_feasible(h_start, ts)

            h_opt, c5_opt = coord_descent(h_start, n_try, max_passes=80)

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

                if remaining() > 30:
                    h_ls, c5_ls = local_search(best_h, best_n, 2.0 / best_n, 200)
                    if c5_ls < best_c5:
                        best_c5 = c5_ls
                        best_h = h_ls

                if remaining() > 25:
                    h_opt2, c5_opt2 = coord_descent(best_h, best_n, max_passes=150)
                    if c5_opt2 < best_c5:
                        best_c5 = c5_opt2
                        best_h = h_opt2

    # Phase 3: Interpolate best solution to nearby n values
    if best_h is not None and remaining() > 40:
        n_b = len(best_h)
        x_orig = np.linspace(0.0, 2.0, n_b, endpoint=False)
        for dn in [-2, -1, 1, 2, 3, 4, 5]:
            n_new = n_b + dn
            if n_new < 20 or remaining() < 20:
                continue
            x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
            h_interp = np.interp(x_new, x_orig, best_h, period=2.0)
            h_interp = project_feasible(h_interp, n_new / 2.0)
            h_opt, c5_opt = coord_descent(h_interp, n_new, max_passes=100)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_new
                if remaining() > 20:
                    h_opt2, c5_opt2 = coord_descent(best_h, best_n, max_passes=150)
                    if c5_opt2 < best_c5:
                        best_c5 = c5_opt2
                        best_h = h_opt2

    # Phase 4: Subgradient polishing
    if best_h is not None and remaining() > 30:
        n_b = len(best_h)
        h_sg, c5_sg = subgradient_descend(best_h, n_b, max_iter=2000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
        h_opt, c5_opt = coord_descent(best_h, n_b, max_passes=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: More diverse random restarts with best n
    if best_h is not None and remaining() > 50:
        n_b = best_n
        for trial in range(10):
            if remaining() < 15:
                break
            h_rand = make_random_start(n_b, rng)
            h_opt, c5_opt = coord_descent(h_rand, n_b, max_passes=60)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                h_opt2, c5_opt2 = coord_descent(h_opt, n_b, max_passes=150)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2

    # Phase 6: Final intensive polish
    if best_h is not None and remaining() > 20:
        n_b = len(best_h)
        h_final, c5_final = coord_descent(best_h, n_b, max_passes=300)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        if remaining() > 10:
            h_ls, c5_ls = local_search(best_h, n_b, 2.0 / n_b, 400)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls
        if remaining() > 8:
            h_final2, c5_final2 = coord_descent(best_h, n_b, max_passes=200)
            if c5_final2 < best_c5:
                best_c5 = c5_final2
                best_h = h_final2

    return best_h, best_c5, len(best_h)
