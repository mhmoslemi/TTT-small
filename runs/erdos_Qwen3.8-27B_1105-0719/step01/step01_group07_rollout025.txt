    dx = 2.0 / n

def project_box_sum(h, target):
    h = np.array(h, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.0
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(200):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        cnt = float(np.sum(mask))

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = get_profile(h)
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    
    # Gradient
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            # C = sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])
            # dC/dh[i] for i in [0, n-1-s]: (1-h[i+s])*dx
            # dC/dh[i] for i in [s, n-1]: -h[i-s]*dx
            k1 = np.arange(0, n - s)
            grad[k1] += w * dx * (1.0 - h[k1 + s])
            k2 = np.arange(s, n)
            grad[k2] -= w * dx * h[k2 - s]
        else:
            a = -s
            # C = sum_{i=a}^{n-1} h[i] * (1-h[i-a])
            # dC/dh[i] for i in [a, n-1]: (1-h[i-a])*dx
            # dC/dh[i] for i in [0, n-1-a]: -h[i+a]*dx
            k1 = np.arange(a, n)
            grad[k1] += w * dx * (1.0 - h[k1 - a])
            k2 = np.arange(0, n - a)
            grad[k2] -= w * dx * h[k2 + a]
    return obj, grad

    """Epigraph: x = [h(0..n-1), t], minimize t"""
    h = x[:n]
    t = x[n]
    profile = get_profile(h)
    # Constraint: profile[j] - t <= 0 for all j
    return t, profile - t

def epigraph_jac(x, n):
    """Jacobian for epigraph: rows = 2n-1 profile constraints + 1 sum constraint, cols = n+1"""
    h = x[:n]
    n_prof = 2 * n - 1
    jac = np.zeros((n_prof + 1, n + 1))
    dx = 2.0 / n

            k1 = np.arange(0, n - s)
            jac[j, k1] = dx * (1.0 - h[k1 + s])
            k2 = np.arange(s, n)
            jac[j, k2] -= dx * h[k2 - s]

            k2 = np.arange(0, n - a)
            jac[j, k2] -= dx * h[k2 + a]

    jac[n_prof, :n] = 1.0
    jac[n_prof, n] = 0.0
    
    return jac

import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h, target):
    h = np.array(h, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.0
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        cnt = float(np.sum(mask))
        if cnt > 0:
            result[mask] += diff / cnt
    return np.clip(result, 0.0, 1.0)


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = get_profile(h)
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumEq:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        return np.sum(x) - self.n / 2.0
    def jac(self, x):
        return np.ones(self.n)


def lse_slsqp(h0, tau, maxiter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    func = LSEFunc(tau)
    con = SumEq(n)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(func.value, h0.copy(), method='SLSQP', jac=func.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h0.copy()


def epi_ineq_fun(x, n):
    return x[n] - get_profile(x[:n])


def epi_ineq_jac(x, n):
    h = x[:n]
    n_prof = 2 * n - 1
    jac = np.zeros((n_prof, n + 1))
    dx = 2.0 / n
    for j in range(n_prof):
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            jac[j, k1] = -dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            jac[j, k2] += dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            jac[j, k1] = -dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            jac[j, k2] += dx * h[k2 - a]
        jac[j, n] = 1.0
    return jac


def epi_eq_fun(x, n):
    return np.sum(x[:n]) - n / 2.0


def epi_eq_jac(x, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def epi_obj_val(x, n):
    return x[n]


def epi_obj_grad(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


class EpiIneq:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        return epi_ineq_fun(x, self.n)
    def jac(self, x):
        return epi_ineq_jac(x, self.n)


class EpiEqC:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        return epi_eq_fun(x, self.n)
    def jac(self, x):
        return epi_eq_jac(x, self.n)


class EpiObj:
    def __init__(self, n):
        self.n = n
    def val(self, x):
        return epi_obj_val(x, self.n)
    def grad(self, x):
        return epi_obj_grad(x, self.n)


def epigraph_slsqp(h0, t0, maxiter=300):
    n = len(h0)
    x0 = np.concatenate([h0.copy(), [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    obj = EpiObj(n)
    ineq = EpiIneq(n)
    eqc = EpiEqC(n)
    constraints = [
        {'type': 'ineq', 'fun': ineq.fun, 'jac': ineq.jac},
        {'type': 'eq', 'fun': eqc.fun, 'jac': eqc.jac}
    ]
    try:
        res = minimize(obj.val, x0, method='SLSQP', jac=obj.grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_box_sum(res.x[:n], n / 2.0)
        return h_new
    except Exception:
        return h0.copy()


def de_single_eval(h, n):
    h = project_box_sum(h, n / 2.0)
    return exact_objective(h)


def de_optimize(h0, pop_size, n_gen, deadline, rng):
    n = len(h0)
    target = n / 2.0
    population = np.zeros((pop_size, n))
    fitness = np.zeros(pop_size)
    population[0] = h0.copy()
    fitness[0] = exact_objective(h0)
    for i in range(1, pop_size):
        if i == 1:
            population[i] = np.sort(h0)[::-1].copy()
        elif i == 2:
            population[i] = np.roll(h0, n // 3)
        elif i == 3:
            population[i] = np.roll(h0, -n // 3)
        else:
            population[i] = h0 + rng.normal(0, 0.15, n)
        population[i] = project_box_sum(population[i], target)
        fitness[i] = exact_objective(population[i])

    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_fit = fitness[best_idx]

    for gen in range(n_gen):
        if time.time() > deadline - 5:
            break
        best_idx = np.argmin(fitness)
        for i in range(pop_size):
            if time.time() > deadline - 5:
                break
            r1, r2 = rng.choice(pop_size, 2, replace=False)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i:
                r2 = (r2 + 1) % pop_size
            F = rng.uniform(0.4, 0.9)
            CR = rng.uniform(0.3, 0.8)
            v = population[best_idx] + F * (population[r1] - population[r2])
            crossover_mask = rng.random(n) < CR
            if not np.any(crossover_mask):
                crossover_mask[rng.randint(n)] = True
            trial = np.where(crossover_mask, v, population[i])
            trial = project_box_sum(trial, target)
            trial_fit = exact_objective(trial)
            if trial_fit <= fitness[i]:
                population[i] = trial
                fitness[i] = trial_fit
                if trial_fit < best_fit:
                    best_fit = trial_fit
                    best_h = trial.copy()

    return best_h, best_fit


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def binary_swap_search(b, deadline, max_rounds=500):
    n = len(b)
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        found = False
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
            found = True
        if not found:
            break
    return b, curr_c


def binary_block_swap(b, deadline, max_rounds=100):
    n = len(b)
    curr_c = exact_objective(b)
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        found = False
        best_improve = 1e-14
        best_move = None
        for blen in [2, 3]:
            for i in range(0, n - blen + 1):
                src_sum = np.sum(b[i:i + blen])
                if src_sum == 0 or src_sum == blen:
                    continue
                for j in range(0, n - blen + 1):
                    if abs(i - j) < blen:
                        continue
                    dst_sum = np.sum(b[j:j + blen])
                    if dst_sum != src_sum:
                        continue
                    old_i = b[i:i + blen].copy()
                    old_j = b[j:j + blen].copy()
                    b[i:i + blen] = old_j
                    b[j:j + blen] = old_i
                    c = exact_objective(b)
                    b[i:i + blen] = old_i
                    b[j:j + blen] = old_j
                    improve = curr_c - c
                    if improve > best_improve:
                        best_improve = improve
                        best_move = (i, j, blen, old_i, old_j)
        if best_move is not None:
            i, j, bl, old_i, old_j = best_move
            b[i:i + bl] = old_j
            b[j:j + bl] = old_i
            curr_c = exact_objective(b)
            found = True
        if not found:
            break
    return b, curr_c


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy(), n_target / 2.0)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new, n_target / 2.0)


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6, 7, 8])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        h = np.zeros(n)
        h[::3] = 1.0
        h[1::3][::2] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, n / 2.0)


def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new, len(h) / 2.0)


def refine_lse(h0, deadline, max_tau_iter=5):
    h = project_box_sum(h0.copy(), len(h0) / 2.0)
    n = len(h)
    curr_c = exact_objective(h)
    taus = [0.005, 0.002, 0.0007, 0.00025, 0.0001]
    for tau in taus:
        if time.time() > deadline - 8:
            break
        h_new = lse_slsqp(h, tau, maxiter=min(500, n * 8 + 50))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-14:
            h = h_new
            curr_c = c_new
    return h, curr_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h, len(parent_h) / 2.0)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
                best_n = len(parent_h)
    except Exception:
        pass

    if best_h is None:
        best_h = make_start(48, rng, 0)
        best_c = exact_objective(best_h)
        best_n = 48

    parent_n = len(parent_h) if parent_h is not None else best_n

    sizes = set()
    for s in [parent_n, parent_n - 4, parent_n - 8, parent_n + 4, parent_n + 8,
              32, 36, 40, 44, 48, 52, 56, 60, 64, 68, 72]:
        if 16 <= s <= 100 and s % 2 == 0:
            sizes.add(s)
    sizes = sorted(sizes)

    # Phase 1: Quick candidate generation and LSE refinement per size
    lse_results = {}
    for n in sizes:
        if time.time() > deadline - 120:
            break
        cands = []
        if parent_h is not None:
            cands.append(transfer_to_size(parent_h, n))
            cands.append(random_perturb(transfer_to_size(parent_h, n), 0.02, rng))
            cands.append(random_perturb(transfer_to_size(parent_h, n), 0.05, rng))
            cands.append(transfer_to_size(np.roll(parent_h, n - len(parent_h) // 3), n))
        for mode in range(7):
            cands.append(make_start(n, rng, mode))

        scored = []
        for c in cands:
            c_score = exact_objective(c)
            scored.append((c_score, c))
        scored.sort(key=lambda x: x[0])

        local_best_h = scored[0][1]
        local_best_c = scored[0][0]

        n_refine = min(3, len(scored))
        for idx in range(n_refine):
            if time.time() > deadline - 80:
                break
            h_ref, c_ref = refine_lse(scored[idx][1], deadline)
            if c_ref < local_best_c:
                local_best_h = h_ref
                local_best_c = c_ref

        lse_results[n] = (local_best_h, local_best_c)
        if local_best_c < best_c:
            best_c = local_best_c
            best_h = local_best_h.copy()
            best_n = n

    # Phase 2: DE optimization on top sizes
    top_sizes = [n for n, (h, c) in sorted(lse_results.items(), key=lambda x: x[1][1])[:6]]
    for n in top_sizes:
        if time.time() > deadline - 150:
            break
        h_init, c_init = lse_results[n]
        de_budget = 60
        pop_size = max(15, min(25, n // 2))
        max_gen = max(50, min(500, int(de_budget * 50)))
        h_de, c_de = de_optimize(h_init, pop_size, max_gen, deadline - 100, rng)
        if c_de < best_c:
            best_c = c_de
            best_h = h_de.copy()
            best_n = n
            lse_results[n] = (h_de, c_de)
        else:
            if c_de < lse_results[n][1]:
                lse_results[n] = (h_de, c_de)

        # Run DE from a different start too
        if time.time() > deadline - 100:
            break
        h_init2 = random_perturb(h_init, 0.08, rng)
        h_de2, c_de2 = de_optimize(h_init2, pop_size // 2 + 5, max_gen // 2, deadline - 80, rng)
        if c_de2 < best_c:
            best_c = c_de2
            best_h = h_de2.copy()
            best_n = n

    # Phase 3: Epigraph SLSQP polish
    for n in top_sizes[:4]:
        if time.time() > deadline - 80:
            break
        h_cur = lse_results.get(n, (best_h, best_c))
        if len(h_cur[0]) == n:
            h_e0, _ = h_cur
        else:
            h_e0 = transfer_to_size(best_h, n)
        t0 = exact_objective(h_e0) + 0.005
        h_epi = epigraph_slsqp(h_e0, t0, maxiter=200)
        c_epi = exact_objective(h_epi)
        if c_epi < best_c:
            best_c = c_epi
            best_h = h_epi.copy()
            best_n = n

    # Phase 4: Binary search on best continuous solutions
    if time.time() < deadline - 40:
        for n in top_sizes[:5]:
            if time.time() > deadline - 15:
                break
            h_cur = lse_results.get(n, (best_h, best_c))
            if len(h_cur[0]) != n:
                h_cur = (transfer_to_size(best_h, n), exact_objective(transfer_to_size(best_h, n)))
            b = binary_threshold(h_cur[0])
            if b is None:
                continue
            c_b = exact_objective(b)
            if c_b < best_c:
                best_c = c_b
                best_h = b.copy()
                best_n = n
            b_swapped, c_swapped = binary_swap_search(b.copy(), deadline, max_rounds=200)
            if c_swapped < best_c:
                best_c = c_swapped
                best_h = b_swapped.copy()
                best_n = n
            if time.time() < deadline - 10:
                b_block, c_block = binary_block_swap(b_swapped.copy(), deadline, max_rounds=50)
                if c_block < best_c:
                    best_c = c_block
                    best_h = b_block.copy()
                    best_n = n

    # Phase 5: Final LSE + epigraph polish on global best
    if time.time() < deadline - 10 and best_h is not None:
        n_best = len(best_h)
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            b, c_b = binary_swap_search(best_h.copy(), deadline, max_rounds=300)
            if c_b < best_c:
                best_c = c_b
                best_h = b.copy()
        else:
            h_ref, c_ref = refine_lse(best_h.copy(), deadline)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()
            if time.time() < deadline - 6:
                t0 = exact_objective(best_h) + 0.003
                h_epi = epigraph_slsqp(best_h, t0, maxiter=150)
                c_epi = exact_objective(h_epi)
                if c_epi < best_c:
                    best_c = c_epi
                    best_h = h_epi.copy()
            if len(best_h) % 2 == 0 and time.time() < deadline - 5:
                b = binary_threshold(best_h)
                if b is not None:
                    b, c_b = binary_swap_search(b, deadline, max_rounds=150)
                    if c_b < best_c:
                        best_c = c_b
                        best_h = b.copy()

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h, best_n / 2.0)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
