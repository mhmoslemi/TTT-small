def lag_gradient(h, k):
    """Compute gradient of overlap at lag k with respect to h.
    k is the actual shift (can be negative)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        # p = dx * sum_{i=k}^{n-1} h[i]*(1-h[i-k])
        for j in range(k, n):
            grad[j] += dx * (1.0 - h[j-k])
            if j + k < n:
                grad[j] -= dx * h[j+k]
    else:

        for j in range(0, n - kp):
            grad[j] += dx * (1.0 - h[j+kp])
            if j >= kp:

    return grad

def active_set_lp(h, active_lags, delta=0.1):
    """Solve the active-set LP to find a descent direction.
    
    Variables: d[0],...,d[n-1], t
    Minimize t
    Subject to:
      grad_l · d + p_l <= t  for each active lag l
      sum(d) = 0
      -h[j] <= d[j] <= 1-h[j]
      -delta <= d[j] <= delta
    """
    n = len(h)
    # Number of variables: n + 1 (d[0..n-1] and t)
    # c = [0, 0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag l with shift k:
    #   grad_k · d - t <= -p_k  i.e.,  [grad_k, -1] · [d, t] <= -p_k
    rows = []
    rhs = []
    profile = compute_profile(h)
    for k in active_lags:
        g = lag_gradient(h, k)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        rows.append(row)
        rhs.append(-profile[k + (n-1)])  # p_k = profile[k+(n-1)]
    
    # sum(d) = 0 -> this is equality
    # -h[j] <= d[j] -> d[j] >= -h[j] -> -d[j] <= h[j]
    # d[j] <= 1-h[j]
    # -delta <= d[j] -> -d[j] <= delta
    # d[j] <= delta
    
    A_ub = []
    b_ub = []
    if rows:
        A_ub.extend(rows)
        b_ub.extend(rhs)
    
    # d[j] <= 1-h[j] and d[j] <= delta
    for j in range(n):
        row = np.zeros(n+1)
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(min(1.0 - h[j], delta))
        
        row2 = np.zeros(n+1)
        row2[j] = -1.0
        A_ub.append(row2)
        b_ub.append(min(h[j], delta))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None

To minimize p[0], we want to maximize sum h[i]^2, which means making h as binary as possible.

Wait, but we want to MINIMIZE the max of the profile. So we want all profile entries to be small. For p[0], making h binary minimizes it (p[0] = 0 for binary h). But making h binary might make other lags large.

import numpy as np
from scipy.optimize import linprog
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    return h

def lag_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            grad[j] += (1.0 - h[j - k])
            if j + k < n:
                grad[j] -= h[j + k]
    else:
        m = -k
        for j in range(0, n - m):
            grad[j] += (1.0 - h[j + m])
            if j >= m:
                grad[j] -= h[j - m]
    return grad

def active_set_polish(h, max_iters=100, delta_init=0.05, active_count=15, tol=1e-12):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c = exact_objective(h)
    delta = delta_init
    no_improve = 0

    for it in range(max_iters):
        profile = compute_profile(h)
        C = np.max(profile)

        order = np.argsort(-profile)
        active_idx = order[:active_count]
        active_lags = active_idx - (n - 1)

        nv = n + 1
        c_obj = np.zeros(nv)
        c_obj[-1] = 1.0

        A_ub_list = []
        b_ub_list = []

        for k in active_lags:
            g = lag_gradient(h, k) * dx
            row = np.zeros(nv)
            row[:n] = g
            row[n] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(-profile[k + (n - 1)])

        for j in range(n):
            ub_d = min(1.0 - h[j], delta)
            row = np.zeros(nv)
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(ub_d)

            row2 = np.zeros(nv)
            row2[j] = -1.0
            A_ub_list.append(row2)
            b_ub_list.append(min(h[j], delta))

        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)

        A_eq = np.zeros((1, nv))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * nv, method='highs')

        if not res.success:
            delta *= 0.5
            if delta < 1e-7:
                break
            continue

        d = res.x[:n]
        t_pred = res.x[n]

        if t_pred >= C - tol:
            delta *= 0.6
            no_improve += 1
            if delta < 1e-7 or no_improve > 20:
                break
            continue

        best_alpha = 0.0
        best_c_ls = cur_c
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            h_trial = h + alpha * d
            h_trial = project_box_sum(h_trial)
            c_trial = exact_objective(h_trial)
            if c_trial < best_c_ls:
                best_c_ls = c_trial
                best_alpha = alpha

        if best_alpha > 0:
            h = project_box_sum(h + best_alpha * d)
            cur_c = exact_objective(h)
            no_improve = 0
            delta = min(delta * 1.4, delta_init)
        else:
            delta *= 0.5
            no_improve += 1

    return h, cur_c

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

def binary_local_search(h, max_rounds=2000, rng_seed=42):
    n = len(h)
    if n % 2 != 0:
        return h, exact_objective(h)
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    cur = b.copy()
    cur_c = exact_objective(cur)
    rg = np.random.RandomState(rng_seed)

    for _ in range(max_rounds):
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-15
        best_i, best_j = -1, -1
        n_attempts = min(300, len(ones) * len(zeros))
        for _ in range(n_attempts):
            i = rg.choice(ones)
            j = rg.choice(zeros)
            cur[i] = 0.0
            cur[j] = 1.0
            c = exact_objective(cur)
            cur[i] = 1.0
            cur[j] = 0.0
            delta = cur_c - c
            if delta > best_delta:
                best_delta = delta
                best_i, best_j = i, j
        if best_i >= 0:
            cur[best_i] = 0.0
            cur[best_j] = 1.0
            cur_c -= best_delta
        else:
            break
    return cur, cur_c

def smoothed_max_polish(h, max_iters=200, tau_init=0.01, tau_min=0.001):
    """Polish using log-sum-exp smoothing with projected gradient."""
    n = len(h)
    h = h.copy()
    cur_c = exact_objective(h)
    tau = tau_init
    lr = 0.02

    for it in range(max_iters):
        profile = compute_profile(h)
        C = np.max(profile)
        shifted = (profile - C) / tau
        shifted = np.clip(shifted, -500, 500)
        exp_s = np.exp(shifted)
        weights = exp_s / exp_s.sum()

        # Gradient of log-sum-exp objective
        grad = np.zeros(n)
        for i in range(2 * n - 1):
            k = i - (n - 1)
            w = weights[i]
            if w < 1e-14:
                continue
            g = lag_gradient(h, k) * (2.0 / n)
            grad += w * g
        grad = grad / tau

        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_box_sum(h_new)
        c_new = exact_objective(h_new)

        if c_new < cur_c:
            h = h_new
            cur_c = c_new
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.7
            tau *= 0.9

        if tau < tau_min and lr < 1e-5:
            break

    return h, cur_c
