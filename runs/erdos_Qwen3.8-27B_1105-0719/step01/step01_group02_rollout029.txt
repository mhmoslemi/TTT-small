    dx = 2.0 / n

        target_sum = n / 2.0
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    for it in range(n_iter):
        t += 1
        profile = get_profile(h)
        pmax = np.max(profile)
        shifted = (profile - pmax) / tau
        shifted = np.clip(shifted, -500, 500)
        exp_s = np.exp(shifted)
        log_sum = np.log(np.sum(exp_s))
        weights = exp_s / np.sum(exp_s)
        
        # Gradient: g[i] = dx * sum_t weights[t] * ((1-h[i-s_t])*[valid] - h[i+s_t]*[valid])
        # where s_t = t - (n-1)
        g = np.zeros(n)
        for t_idx in range(2*n-1):
            w = weights[t_idx]
            if w < 1e-10:
                continue
            s = t_idx - (n-1)
            # (1-h[i-s]) term: i-s valid means 0<=i-s<n, i.e., s<=i<n+s
            # h[i+s] term: i+s valid means 0<=i+s<n, i.e., -s<=i<n-s

                if 0 <= i-s < n:
                    g[i] += w * (1.0 - h[i-s])
                if 0 <= i+s < n:
                    g[i] -= w * h[i+s]
        g *= dx
        
        # Sum constraint gradient
        sum_err = np.sum(h) - target_sum
        g += 1.0 * sum_err  # penalty for sum violation
        
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project

        # Check improvement
        old_c = exact_objective(h)
        new_c = exact_objective(h_new)
        if new_c < old_c:
            h = h_new
        # else: keep old h, but still update m, v (or reset?)

        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), 1.0 - np.max(h[mask]))
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) if diff > 0 else (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_profile(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        n = n // 2 * 2
        h = h[:n]
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b
