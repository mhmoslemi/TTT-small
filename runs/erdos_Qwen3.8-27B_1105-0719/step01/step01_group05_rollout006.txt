def exact_objective(h):
    """Compute C5 = max of correlation profile."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target):
    """Project h onto {0 ≤ h_i ≤ 1, sum(h) = target}."""
    n = len(h)
    h = np.clip(np.nan_to_num(h), 0.0, 1.0)
    # Bisection on a constant shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:

    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[a:] += 1.0 - h[:s]

minimize ∑_{k∈A} λ_k * C_k(h) ≈ ∑_{k∈A} λ_k * C_k(h_0) + ∑_{k∈A} λ_k * ∇C_k(h_0)^T (h - h_0)
= constant + (∑_{k∈A} λ_k * ∇C_k(h_0))^T h

So the Frank-Wolfe step is: find the binary vector b with n/2 ones that minimizes (∑_{k∈A} λ_k * ∇C_k(h_0))^T b.

This is equivalent to: pick the n/2 coordinates with the smallest values of the combined gradient.

Then move: h_new = (1-γ) * h + γ * b, where γ is a step size.

This is much better because:
1. The "vertex search" is trivial (sort and pick)
2. The step is guaranteed to be feasible
3. It naturally handles the combinatorial structure

```python
def frank_wolfe_step(h, profile, active_tol, dx, gamma):
    n = len(h)
    target = 0.5 * n
    c_max = np.max(profile)
    active = np.where(profile >= c_max - active_tol)[0]
    
    # Combined gradient
    g = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        g += lag_gradient(h, lag, dx)
    g /= len(active)
    
    # Find best vertex: pick n/2 smallest gradient coordinates
    # (we want to minimize g^T b, so pick where g is smallest)
    k = int(target)  # number of ones
    indices = np.argpartition(g, k)[:k]
    b = np.zeros(n)
    b[indices] = 1.0
    
    # Frank-Wolfe step
    h_new = (1 - gamma) * h + gamma * b
    h_new = project_box_sum(h_new, target)
    c_new, _ = exact_objective(h_new)
    
    return h_new, c_new
```

def interpolate(ha, hb, deadline, n_alphas=20):

    target = 0.5 * n
    best_h = project_box_sum(ha, target)
    best_c = exact_objective(best_h)[0]

            best_h = h

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

    best_n = len(best_h)
    return (best_h, best_c, best_n)
