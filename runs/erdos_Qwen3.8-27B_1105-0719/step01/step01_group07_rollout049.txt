import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(profile)) * dx

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad

class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]

class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))

def slsqp_optimize(h0, tau, maxiter=300, gtol=1e-8):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': gtol})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()

def sim_annealing(b, deadline, rng, t_start=1.0, t_end=0.01, max_iters=500000):
    """Simulated annealing with single swap, 2-swap, and reversal moves"""
    n = len(b)
    b = b.copy().astype(np.float64)
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]
    
    t = t_start
    cooling = (t_end / t_start) ** (1.0 / max(1, max_iters))
    
    n_accepted = 0
    n_rejected = 0
    
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        
        # Choose move type
        move_type = rng.randint(3)
        
        if move_type == 0:
            # Single swap
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
            new_c = exact_objective(b)
            b[i] = 1.0
            b[j] = 0.0
        elif move_type == 1:
            # 2-swap: exchange 2 ones with 2 zeros
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(len(ones), size=2, replace=False)
            j1, j2 = rng.choice(len(zeros), size=2, replace=False)
            i1, i2 = ones[i1], ones[i2]
            j1, j2 = zeros[j1], zeros[j2]
            b[i1] = 0.0; b[i2] = 0.0
            b[j1] = 1.0; b[j2] = 1.0
            new_c = exact_objective(b)
            b[i1] = 1.0; b[i2] = 1.0
            b[j1] = 0.0; b[j2] = 0.0
        else:
            # Reversal: reverse a contiguous segment
            if n < 4:
                continue
            i = rng.randint(n)
            j = rng.randint(n)
            if i > j:
                i, j = j, i
            if j - i < 2:
                continue
            # Reverse segment [i, j]
            seg = b[i:j+1].copy()
            b[i:j+1] = b[j:i-1:-1]
            new_c = exact_objective(b)
            b[i:j+1] = seg  # restore
        
        # Accept/reject
        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            # Accept
            if move_type == 0:
                b[i] = 0.0; b[j] = 1.0
                ones = np.where(b == 1.0)[0]
                zeros = np.where(b == 0.0)[0]
            elif move_type == 1:
                b[i1] = 0.0; b[i2] = 0.0
                b[j1] = 1.0; b[j2] = 1.0
                ones = np.where(b == 1.0)[0]
                zeros = np.where(b == 0.0)[0]
            else:
                # Already applied
                pass
            curr_c = new_c
            n_accepted += 1
            if curr_c < best_c - 1e-15:
                best_c = curr_c
                best_b = b.copy()
        else:
            n_rejected += 1
        
        t *= cooling
    
    return best_b, best_c

    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)

def make_start(n, rng, mode):

import numpy as np

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad

    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]

class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))

    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))

            i1, i2 = ones_idx[rng.randint(len(ones_idx))]
            while i2 == i1:
                i2 = ones_idx[rng.randint(len(ones_idx))]
            j1, j2 = zeros_idx[rng.randint(len(zeros_idx))]
            while j2 == j1:
                j2 = zeros_idx[rng.randint(len(zeros_idx))]
            b[i1] = 0.0
            b[i2] = 0.0
            b[j1] = 1.0
            b[j2] = 1.0
            new_c = exact_objective(b)
            b[i1] = 1.0
            b[i2] = 1.0
            b[j1] = 0.0
            b[j2] = 0.0
            is_single = False

        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            if move_type == 0:
                b[single_i] = 0.0
                b[single_j] = 1.0
            elif move_type == 1 or move_type == 3:
                b[i1] = 0.0
                b[i2] = 0.0
                b[j1] = 1.0
                b[j2] = 1.0
            else:
                a2 = a
                b[a2:a2 + width] = b[a2:a2 + width][::-1]
            curr_c = new_c
            if curr_c < best_c - 1e-15:
                best_c = curr_c
                best_b = b.copy()
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > stall_threshold:
                t = max(t_final, t * 2.0)
                stall_count = 0
        t *= cooling
    return best_b, best_c

def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)

def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.randint(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        h = np.zeros(n)
        period = max(3, n // 5)
        for i in range(n):
            if (i % period) < period // 2:
                h[i] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)

def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    margin = 6
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
                best_n = len(parent_h)
    except Exception:
        pass

    if best_h is None:
        n0 = 48
        best_h = make_start(n0, rng, 0)
        best_c = exact_objective(best_h)
        best_n = n0

    parent_n = len(parent_h) if parent_h is not None else 48

    sizes = []
    for s in range(max(36, parent_n - 8), min(68, parent_n + 12) + 1, 2):
        sizes.append(s)

    # Phase 1: Continuous refinement to generate good starting points
    best_per_size = {}
    for n in sizes:
        if time.time() > deadline - margin - 120:
            break
        cands = []
        if parent_h is not None:
            ph = transfer_to_size(parent_h, n)
            cands.append(ph)
            cands.append(random_perturb(ph, 0.015, rng))
            cands.append(random_perturb(ph, 0.04, rng))
            cands.append(random_perturb(ph, 0.08, rng))
            for shift in [1, 2, 3]:
                cands.append(project_box_sum(np.roll(ph, shift)))
        for mode in range(7):
            cands.append(make_start(n, rng, mode))

        scored = []
        for c in cands:
            c_score = exact_objective(c)
            scored.append((c_score, c))
        scored.sort(key=lambda x: x[0])

        local_best_h = scored[0][1]
        local_best_c = scored[0][0]

        for idx in range(min(3, len(scored))):
            if time.time() > deadline - margin - 60:
                break
            h_ref, c_ref = refine_continuous(scored[idx][1], deadline - 80, max_taus=4)
            if c_ref < local_best_c:
                local_best_c = c_ref
                local_best_h = h_ref.copy()

        best_per_size[n] = (local_best_c, local_best_h.copy())
        if local_best_c < best_c - 1e-14:
            best_c = local_best_c
            best_h = local_best_h.copy()
            best_n = n

    # Phase 2: Binary optimization for each size
    # Sort sizes by their continuous best (try best continuous first)
    sorted_sizes = sorted(best_per_size.keys(), key=lambda s: best_per_size[s][0])

    for n in sorted_sizes:
        if time.time() > deadline - margin - 15:
            break
        if n % 2 != 0:
            continue

        c_cont, h_cont = best_per_size[n]
        b = binary_from_continuous(h_cont)
        if b is None:
            continue

        # Greedy single swap
        b1, c1 = greedy_single_swap(b, deadline - 40, max_rounds=60)

        # SA round 1
        b_sa1, c_sa1 = sa_binary(b1, deadline - 30, rng, t_init=0.05, t_final=0.0008, max_iters=250000)

        # Greedy 2-swap
        b2s, c2s = greedy_2swap(b_sa1, deadline - 15, max_rounds=25)

        # SA round 2 (shorter, lower temp)
        b_sa2, c_sa2 = sa_binary(b2s, deadline - 8, rng, t_init=0.02, t_final=0.0003, max_iters=150000)

        # Final greedy single swap
        b_fin, c_fin = greedy_single_swap(b_sa2, deadline - 4, max_rounds=50)

        # Track best from this size
        candidates_here = [(c1, b1), (c_sa1, b_sa1), (c2s, b2s), (c_sa2, b_sa2), (c_fin, b_fin)]
        candidates_here.sort(key=lambda x: x[0])
        local_best_c, local_best_b = candidates_here[0]

        if local_best_c < best_c - 1e-14:
            best_c = local_best_c
            best_h = local_best_b.copy()
            best_n = n

    # Phase 3: Additional 2-swap rounds on the global best
    if best_h is not None and time.time() < deadline - margin - 5:
        b = best_h.copy().astype(np.float64)
        if np.all((b == 0.0) | (b == 1.0)):
            # Multiple rounds of 2-swap + SA
            for _ in range(3):
                if time.time() > deadline - margin - 3:
                    break
                b, c = greedy_2swap(b, deadline - 5, max_rounds=20)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = b.copy()
                if time.time() < deadline - margin - 5:
                    b_sa, c_sa = sa_binary(b, deadline - 4, rng, t_init=0.01, t_final=0.0002, max_iters=100000)
                    if c_sa < best_c - 1e-14:
                        best_c = c_sa
                        best_h = b_sa.copy()
                        b = b_sa.copy()
            b, c = greedy_single_swap(b, deadline - 3, max_rounds=80)
            if c < best_c - 1e-14:
                best_c = c
                best_h = b.copy()

    # Phase 4: Try continuous refinement around best binary
    if best_h is not None and time.time() < deadline - margin:
        n_best = len(best_h)
        h_cont = best_h.copy()
        if np.all((h_cont == 0.0) | (h_cont == 1.0)):
            h_cont = h_cont + rng.normal(0, 0.005, n_best)
            h_cont = project_box_sum(h_cont)
        h_ref, c_ref = refine_continuous(h_cont, deadline - 2, max_taus=4)
        if c_ref < best_c - 1e-14:
            best_c = c_ref
            best_h = h_ref.copy()
        if len(best_h) % 2 == 0:
            b = binary_from_continuous(best_h)
            if b is not None and time.time() < deadline - margin:
                b, c_b = greedy_single_swap(b, deadline - 2, max_rounds=40)
                if c_b < best_c - 1e-14:
                    best_c = c_b
                    best_h = b.copy()

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
