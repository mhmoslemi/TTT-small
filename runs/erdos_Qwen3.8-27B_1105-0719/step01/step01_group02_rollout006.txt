def build_grad_matrix(h, shifts):
    """Build gradient matrix for given shifts.
    shifts: array of signed shift values (can be negative, zero, or positive)
    Returns G of shape (len(shifts), n) where G[j, i] = d(profile_k)/d(h[i]) * dx
    """
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(shifts), n))
    for row, k in enumerate(shifts):
        k = int(k)
        for i in range(n):
            g = 0.0
            m = i - k
            if 0 <= m < n:
                g += (1.0 - h[m])
            m2 = i + k
            if 0 <= m2 < n:
                g -= h[m2]
            G[row, i] = g
    G *= dx
    return G
