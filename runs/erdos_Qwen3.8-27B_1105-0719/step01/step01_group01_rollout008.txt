```python
def slp_step(h, R, rho, active_frac=0.1):
    n = len(h)
    corr = compute_corr(h)
    C = np.max(corr)
    # Active set: lags where corr is within active_frac of max
    threshold = C - active_frac * C
    active = np.where(corr >= threshold)[0]
    
    G = compute_grad_matrix(h)
    G_active = G[active, :]
    c_active = corr[active]
    
    u = cp.Variable(n)
    t = cp.Variable()
    
    obj = cp.Minimize(t + rho * cp.sum_squares(u))
    cons = [
        G_active @ u - t <= -c_active,  # c_active + G_active @ u <= t
        cp.sum(u) == 0,
        u >= -h,
        u <= 1 - h
    ]
    # Trust region: ||u||_2 <= R
    cons.append(cp.norm(u, 2) <= R)
    
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.SCS, max_iter=10000)
    except:
        try:
            prob.solve(solver=cp.CLARABEL)
        except:
            return None
    
    if u.value is None:
        return None
    
    u_opt = np.array(u.value).flatten()
    # Project to feasible
    h_new = h + u_opt
    h_new = np.clip(h_new, 0, 1)
    # Fix sum
    target = n / 2.0
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-10:
        # Distribute correction
        free = np.where((h_new > 1e-8) & (h_new < 1 - 1e-8))[0]
        if len(free) > 0:
            h_new[free] += diff / len(free)
            h_new = np.clip(h_new, 0, 1)
        else:
            h_new += diff / n
            h_new = np.clip(h_new, 0, 1)
    
    return h_new
```

```python
def local_search(h, n_iters, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for it in range(n_iters):
        # Choose move type
        if rng.random() < 0.5:
            # Swap
            i, j = rng.choice(n, size=2, replace=False)
            if abs(best_h[i] - best_h[j]) < 1e-6:
                continue
            new_h = best_h.copy()
            new_h[i], new_h[j] = best_h[j], best_h[i]
        else:
            # Transfer
            i, j = rng.choice(n, size=2, replace=False)
            if best_h[i] <= best_h[j]:
                # Transfer from j to i
                max_d = min(best_h[j], 1 - best_h[i])
            else:
                # Transfer from i to j
                max_d = min(best_h[i], 1 - best_h[j])
            if max_d < 1e-8:
                continue
            d = rng.uniform(0.01, max_d)
            if best_h[i] > best_h[j]:
                new_h = best_h.copy()
                new_h[i] -= d
                new_h[j] += d
            else:
                new_h = best_h.copy()
                new_h[i] += d
                new_h[j] -= d
```

```python
def slp_polish(h0, max_iter=100, R0=0.05, rho=0.1, time_limit=60):
    t0 = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    R = R0
    n = len(h)
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        corr = compute_corr(h)
        C_current = np.max(corr)
        
        # Compute gradient
        G = compute_grad_matrix(h)
        
        # Set up QP
        u = cp.Variable(n)
        t = cp.Variable()
        
        obj = cp.Minimize(t + rho * cp.sum_squares(u))
        cons = [
            G @ u - t <= -corr,
            cp.sum(u) == 0,
            u >= -h,
            u <= 1.0 - h,
            cp.norm2(u) <= R * R
        ]
        
        prob = cp.Problem(obj, cons)
        
        solved = False
        for solver in [cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved or u.value is None:
            R *= 0.5
            if R < 1e-4:
                break
            continue
        
        u_opt = np.array(u.value).flatten()
        
        # Check step size
        u_norm = np.linalg.norm(u_opt)
```
