    n = len(h)

    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair(vec):
    n = len(vec)
    target = n / 2.0
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    if not np.isfinite(h).all():
        h = np.full(n, 0.5)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(h + lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def smooth_value(h, tau):
    p = get_profile(h)
    mu = np.max(p)
    shifted = (p - mu) / tau
    shifted = np.clip(shifted, -400, 400)
    return mu + tau * np.log(np.sum(np.exp(shifted)))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    p = get_profile(h)
    mu = np.max(p)
    shifted = (p - mu) / tau
    shifted = np.clip(shifted, -400, 400)
    w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for m in range(M):
        wm = w[m]
        if wm < 1e-14:
            continue
        d = m - (n - 1)
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:n] += wm * dx * (1.0 - h[0:n-d])
            grad[0:n-d] -= wm * dx * h[d:n]
        else:
            s = -d
            grad[0:n-s] += wm * dx * (1.0 - h[s:n])
            grad[s:n] -= wm * dx * h[0:n-s]
    return grad

def fw_vertex(grad):
    n = len(grad)
    target = n / 2.0
    v = np.zeros(n)
    idx = np.argsort(grad)
    if target == int(target):
        v[idx[:int(target)]] = 1.0
    else:
        remaining = target
        for i in range(n):
            if remaining >= 1.0 - 1e-12:
                v[idx[i]] = 1.0
                remaining -= 1.0
            elif remaining > 1e-12:
                v[idx[i]] = remaining
                remaining = 0.0
            else:
                break
    return v

def projected_grad_step(h, grad, alpha):
    h_new = h - alpha * grad
    return repair(h_new)

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair(h_new)

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            s = -d
            G[m, 0:n-s] += dx * (1.0 - h[s:n])
            G[m, s:n] += -dx * h[0:n-s]
    return G

def active_set_descent(h, max_iters=100, tol=1e-10):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h.copy()
    best = exact_score(h)
    for it in range(max_iters):
        p = get_profile(h)
        c = np.max(p)
        if c < best - 1e-12:
            best = c
        thresh = c - 1e-7
        active = np.where(p >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(p)])
        if len(active) > 30:
            order = np.argsort(-p[active])
            active = active[order[:30]]
        G = build_gradient_matrix(h, n, dx)
        G_act = G[active, :]
        w_raw = p[active] - thresh + 1e-10
        w = w_raw / w_raw.sum()
        grad = G_act.T @ w
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_try = repair(h - alpha * grad)
            s = exact_score(h_try)
            if s < best - 1e-12:
                h = h_try
                best = s
                improved = True
                break
        if not improved:
            # Try anti-gradient (push away from worst)
            p2 = get_profile(h)
            worst = np.argmax(p2)
            G2 = build_gradient_matrix(h, n, dx)
            g2 = G2[worst, :]
            g2 -= g2.mean()
            n2 = np.linalg.norm(g2)
            if n2 > 1e-14:
                g2 /= n2
                for alpha in [0.005, 0.002, 0.001]:
                    h_try = repair(h - alpha * g2)
                    s = exact_score(h_try)
                    if s < best - 1e-12:
                        h = h_try
                        best = s
                        improved = True
                        break
        if not improved:
            break
    return h, best

def slsqp_solve(h_start, maxiter=300, tol=1e-10):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - prof

    def ineq_jac(x):
        h = x
        G = build_gradient_matrix(np.clip(h, 0, 1), n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 3.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol, 'disp': False})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair(res.x[:-1])
            if np.all(h_new >= -1e-10) and np.all(h_new <= 1 + 1e-10):
                return h_new
    except Exception:
        pass
    return None

def fw_smooth_anneal(h0, tau_start, tau_end, total_iters, deadline):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    for it in range(total_iters):
        if time.perf_counter() > deadline:
            break
        frac = it / max(total_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 0.003)
        grad = smooth_grad(h, tau)
        # FW step
        v = fw_vertex(grad)
        alpha_fw = 2.0 / (it + 2)
        h_fw = (1 - alpha_fw) * h + alpha_fw * v
        h_fw = repair(h_fw)
        # Also try gradient step
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-10:
            alpha_g = 0.05 / g_norm
            h_g = repair(h - alpha_g * grad)
        else:
            h_g = h_fw
        # Evaluate both
        candidates = [h_fw, h_g]
        best_local = h
        best_local_c = exact_score(h)
        for cand in candidates:
            c = exact_score(cand)
            if c < best_local_c:
                best_local_c = c
                best_local = cand
        h = best_local
        if best_local_c < best_c - 1e-12:
            best_c = best_local_c
            best_h = h.copy()
    return best_h, best_c

def gradient_descent_smooth(h0, tau, n_iters, deadline, lr=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    if lr is None:
        lr = 0.3
    for it in range(n_iters):
        if time.perf_counter() > deadline:
            break
        grad = smooth_grad(h, tau)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        step = lr / (g_norm + 0.1)
        step = min(step, 0.5)
        h_new = repair(h - step * grad)
        c_new = exact_score(h_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        elif it % 10 == 0:
            # Occasionally accept non-improving to escape
            h = h_new
        else:
            h = h_new
    return best_h, best_c

def binary_search_init(n, rng, n_trials=200):
    """Generate binary candidates via random + local search."""
    target = n // 2
    best_h = None
    best_c = 1e9
    for _ in range(n_trials):
        if target * 2 == n:
            idx = rng.choice(n, size=target, replace=False)
            h = np.zeros(n)
            h[idx] = 1.0
        else:
            h = np.full(n, 0.5)
            h[:target] = 1.0
            h[target] = 0.5
            remaining = n // 2.0 - target
            if remaining > 0:
                h[target + 1] = 0.5
            h = repair(h)
        # Local search: try moving 1s
        for _ in range(200):
            i = rng.randint(n)
            if h[i] > 0.9:
                j = rng.randint(n)
                if h[j] < 0.1:
                    h_try = h.copy()
                    h_try[i] = 0.0
                    h_try[j] = 1.0
                    c_try = exact_score(h_try)
                    if c_try < best_c:
                        best_c = c_try
                        best_h = h_try.copy()
            elif h[i] < 0.1:
                j = rng.randint(n)
                if h[j] > 0.9:
                    h_try = h.copy()
                    h_try[i] = 1.0
                    h_try[j] = 0.0
                    c_try = exact_score(h_try)
                    if c_try < best_c:
                        best_c = c_try
                        best_h = h_try.copy()
    if best_h is not None:
        return best_h, best_c
    return None, 1e9

def cyclotomic_init(p):
    """Use quadratic residues mod p for initial binary candidate."""
    if p % 2 == 0:
        p += 1
    qr = set()
    for x in range(1, p):
        qr.add((x * x) % p)
    qr = sorted(qr)
    n = p
    target = n / 2.0
    h = np.zeros(n)
    for q in qr:
        if q < n:
            h[q] = 1.0
    h = repair(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s * 0.92
    rng = np.random.RandomState(seed)

    # Get parent
    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    n_parent = 66
    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(n_parent, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent
    best_n = len(parent)

    def time_left():
        return deadline - time.perf_counter()

    # Initialize with various starting points
    starts = [best_h.copy()]

    # Symmetrization
    h_sym = repair((best_h + best_h[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-10:
        best_h = h_sym
        best_c = c_sym
        starts[0] = h_sym.copy()

    # Random perturbations
    for eps in [0.02, 0.05, 0.1, 0.15]:
        z = rng.randn(best_n)
        z -= z.mean()
        h_p = repair(best_h + eps * z)
        starts.append(h_p)

    # Cyclotomic initialization
    for p in [61, 67, 71, 53]:
        if time_left() < 30:
            break
        h_cyc = cyclotomic_init(p)
        c_cyc = exact_score(h_cyc)
        starts.append(h_cyc)

    # Binary local search
    if time_left() > 60:
        h_bin, c_bin = binary_search_init(best_n, rng, n_trials=50)
        if h_bin is not None:
            starts.append(h_bin)

    # Stage 1: FW + smooth gradient from best starts
    for s_idx, h0 in enumerate(starts[:8]):
        if time_left() < 60:
            break
        h0 = repair(h0)
        # Determine which n to use
        n0 = len(h0)
        # FW annealing
        tau_start = 0.3
        tau_end = 0.005
        n_fw = max(30, int(min(120, time_left() / 5.0)))
        h_res, c_res = fw_smooth_anneal(h0, tau_start, tau_end, n_fw, deadline)
        if c_res < best_c - 1e-10:
            best_c = c_res
            best_h = h_res.copy()
            best_n = n0

    # Stage 2: Gradient descent with varying tau from current best
    for tau in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
        if time_left() < 30:
            break
        n_iters = max(20, int(min(80, time_left() / 8.0)))
        h_res, c_res = gradient_descent_smooth(best_h, tau, n_iters, deadline)
        if c_res < best_c - 1e-10:
            best_c = c_res
            best_h = h_res.copy()

    # Stage 3: Active-set descent on exact objective
    if time_left() > 20:
        h_res, c_res = active_set_descent(best_h, max_iters=80)
        if c_res < best_c - 1e-12:
            best_c = c_res
            best_h = h_res.copy()

    # Stage 4: SLSQP epigraph polish
    if time_left() > 80:
        h_sol = slsqp_solve(best_h, maxiter=200, tol=1e-10)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-10:
                best_c = c_sol
                best_h = h_sol.copy()

    # Stage 5: FW from perturbed best at smaller tau
    for eps in [0.01, 0.03, 0.05]:
        if time_left() < 40:
            break
        z = rng.randn(best_n)
        z -= z.mean()
        h_pert = repair(best_h + eps * z)
        n_fw = max(20, int(min(60, time_left() / 8.0)))
        h_res, c_res = fw_smooth_anneal(h_pert, 0.1, 0.003, n_fw, deadline)
        if c_res < best_c - 1e-10:
            best_c = c_res
            best_h = h_res.copy()
            best_n = len(h_res)

    # Stage 6: Active-set + SLSQP again
    if time_left() > 30:
        h_res, c_res = active_set_descent(best_h, max_iters=60)
        if c_res < best_c - 1e-12:
            best_c = c_res
            best_h = h_res.copy()

    if time_left() > 60:
        h_sol = slsqp_solve(best_h, maxiter=250, tol=1e-11)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-10:
                best_c = c_sol
                best_h = h_sol.copy()

    # Stage 7: Try different dimensions
    if time_left() > 120:
        dims_to_try = []
        for d in [best_n - 4, best_n - 8, best_n + 4, best_n + 8, best_n - 2, best_n + 2,
                  40, 44, 48, 52, 56, 60, 64, 68, 72, 76, 80]:
            if 24 <= d <= 160 and d != best_n and d not in dims_to_try:
                dims_to_try.append(d)
        dims_to_try.sort(key=lambda d: abs(d - best_n))
        for new_n in dims_to_try[:6]:
            if time_left() < 60:
                break
            h_interp = interp_to_n(best_h, new_n)
            c_interp = exact_score(h_interp)
            # Quick FW + descent
            tau_start = 0.2
            tau_end = 0.005
            n_fw = max(20, int(min(50, time_left() / 10.0)))
            h_res, c_res = fw_smooth_anneal(h_interp, tau_start, tau_end, n_fw, deadline)
            if c_res < best_c - 1e-10:
                h_res2, c_res2 = active_set_descent(h_res, max_iters=40)
                if c_res2 < c_res:
                    h_res, c_res = h_res2, c_res2
                best_c = c_res
                best_h = h_res.copy()
                best_n = new_n

    # Stage 8: More aggressive active-set with multiple restarts
    if time_left() > 30:
        for _ in range(3):
            if time_left() < 20:
                break
            z = rng.randn(best_n) * 0.01
            h_pert = repair(best_h + z)
            h_res, c_res = active_set_descent(h_pert, max_iters=50)
            if c_res < best_c - 1e-12:
                best_c = c_res
                best_h = h_res.copy()

    # Stage 9: Final SLSQP
    if time_left() > 40:
        h_sol = slsqp_solve(best_h, maxiter=400, tol=1e-12)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_c = c_sol
                best_h = h_sol.copy()

    # Final active-set polish
    if time_left() > 15:
        h_res, c_res = active_set_descent(best_h, max_iters=40)
        if c_res < best_c - 1e-12:
            best_c = c_res
            best_h = h_res.copy()

    # Final repair and validation
    best_h = repair(best_h)
    final_c = exact_score(best_h)
    if final_c > best_c + 1e-10:
        # Repair made it worse slightly; use pre-repair if feasible
        pass
    else:
        best_c = final_c

    # Try the evaluator if available
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            final_c = float(eval_result[0])
    except Exception:
        pass

    if final_c < best_c:
        best_c = final_c

    return best_h, best_c, len(best_h)
