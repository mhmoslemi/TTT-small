    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:

            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count

def gradient_for_lag(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= h[lag:]
            grad[lag:] += 1.0 - h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - h[a:]
            grad[a:] -= h[:s]
    return grad * dx

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    pmax = np.max(profile)
    if tau < 1e-12:
        m_idx = int(np.argmax(profile))
        lag = m_idx - (n - 1)
        grad = gradient_for_lag(h, lag, dx)
        return pmax, grad
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    value = pmax + tau * np.log(wsum)
    wnorm = weights / wsum
    grad = np.zeros(n)
    active = np.where(wnorm > 1e-6)[0]
    for m in active:
        lag = m - (n - 1)
        grad += wnorm[m] * gradient_for_lag(h, lag, dx)
    return value, grad

def slsqp_refine(h_start, n, deadline, tau_list=None, max_iter=300):
    if tau_list is None:
        tau_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, _ = exact_objective(best_h)
    bounds = [(0.0, 1.0)] * n
    target = 0.5 * n
    
    for tau in tau_list:
        if time.time() > deadline - 5:
            break
        def obj_grad(x):
            val, grad = smooth_obj_and_grad(x, tau)
            se = np.sum(x) - target
            val += 1e4 * se * se
            grad = grad + 2e4 * se * np.ones(n)
            below = np.maximum(0.0, -x)
            above = np.maximum(0.0, x - 1.0)
            val += 1e4 * (np.sum(below*below) + np.sum(above*above))
            grad = grad - 2e4*below + 2e4*above
            return val, grad
        
        try:
            mi = max(50, min(max_iter, n*2+50))
            res = minimize(obj_grad, best_h, method='SLSQP',
                         bounds=bounds, jac=True,
                         options={'maxiter': mi, 'ftol': 1e-15})
            if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                c, _ = exact_objective(cand)
                if c < best_c - 1e-14:

                    best_h = cand
        except Exception:

    return best_h, best_c

def binary_1for1(b, max_rounds=100, deadline=None):

    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:

        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])

            best_c = c5

    if len(ones) < 2 or len(zeros) < 2:
        return b, c5
    for _ in range(n_tries):
        if deadline is not None and time.time() > deadline:
            break
        i1, i2 = rng.choice(len(ones), 2, replace=False)
        j1, j2 = rng.choice(len(zeros), 2, replace=False)
        pi1, pi2 = ones[i1], ones[i2]
        pj1, pj2 = zeros[j1], zeros[j2]
        b[pi1] = 0.0
        b[pi2] = 0.0
        b[pj1] = 1.0
        b[pj2] = 1.0
        nc, _ = exact_objective(b)
        if nc < c5 - 1e-14:
            c5 = nc
            ones = np.where(b > 0.5)[0]
            zeros = np.where(b < 0.5)[0]
        else:
            b[pi1] = 1.0
            b[pi2] = 1.0
            b[pj1] = 0.0
            b[pj2] = 0.0
    return b, c5

def binary_combined_search(b, deadline, rng):
    n = len(b)
    c5 = exact_objective(b)[0]
    round_num = 0
    stall_rounds = 0
    while time.time() < deadline - 10:
        round_num += 1
        b, c5 = binary_1for1(b, max_rounds=100, deadline=deadline)
        if time.time() > deadline - 10:
            break
        n_tries = max(5000, 40000 - round_num * 1000)
        b, c5_2 = binary_2for2_random(b, n_tries, rng, deadline)
        if c5_2 < c5 - 1e-14:
            c5 = c5_2
            stall_rounds = 0
        else:
            stall_rounds += 1
            if stall_rounds >= 4:
                break
        b, c5 = binary_1for1(b, max_rounds=50, deadline=deadline)
    return b, c5

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(round(target))] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        cnt = int(round(target))
        idx = rng.choice(n, size=min(cnt, n), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[n // 2:n // 2 + q] = 1.0
    elif mode == 5:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.2
    elif mode == 6:
        # quadratic residue inspired
        h = np.zeros(n)
        p = n // 2
        residues = sorted(set((i * i) % p for i in range(1, p)))
        for r in residues[:int(round(target))]:
            if r < n:
                h[r] = 1.0
            if r + p < n:
                h[r + p] = 1.0
        cnt = int(round(target))
        if int(np.sum(h)) < cnt:
            remaining = [i for i in range(n) if h[i] == 0]
            for idx in remaining[:cnt - int(np.sum(h))]:
                h[idx] = 1.0
        elif int(np.sum(h)) > cnt:
            ones = [i for i in range(n) if h[i] == 1]
            for idx in ones[int(np.sum(h)) - cnt:]:
                h[idx] = 0.0
    elif mode == 7:
        # golden ratio spacing
        h = np.zeros(n)
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        positions = sorted(set(int((i * golden) % n) for i in range(int(round(target)))))
        for pos in positions[:int(round(target))]:
            h[pos] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def optimize_periodic(p, n, deadline, rng):
    m = n // p
    dx = 2.0 / n
    target_pp = p / 2.0
    bounds = [(0.0, 1.0)] * p

    def unpack(x):
        return np.tile(x, m)

    def periodic_obj_grad(x, tau_val):
        h = unpack(x)
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        pmax = np.max(profile)
        shifted = (profile - pmax) / tau_val
        weights = np.exp(np.clip(shifted, -50, 50))
        wsum = np.sum(weights)
        value = pmax + tau_val * np.log(wsum)
        wnorm = weights / wsum
        grad_h = np.zeros(n)
        active = np.where(wnorm > 1e-5)[0]
        for idx in active:
            lag = idx - (n - 1)
            grad_h += wnorm[idx] * gradient_for_lag(h, lag, dx)
        grad_x = np.zeros(p)
        for j in range(p):
            grad_x[j] = np.sum(grad_h[j::p])
        se = np.sum(x) - target_pp
        value += 1e4 * se * se
        grad_x += 2e4 * se * np.ones(p)
        return value, grad_x

    x0 = np.full(p, target_pp / p) + rng.randn(p) * 0.1
    x0 = np.clip(x0, 0.0, 1.0)
    diff = target_pp - np.sum(x0)
    mask = (x0 < 1.0) & (x0 > 0.0)
    if np.sum(mask) > 0:
        x0[mask] += diff / np.sum(mask)
    x0 = np.clip(x0, 0.0, 1.0)

    best_x = x0.copy()
    best_c = exact_objective(unpack(x0))[0]

    for tau in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline - 5:
            break
        def obj_grad(x):
            return periodic_obj_grad(x, tau)
        try:
            res = minimize(obj_grad, best_x, method='SLSQP',
                          bounds=bounds, jac=True,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
                cand = np.clip(res.x, 0.0, 1.0)
                c = exact_objective(unpack(cand))[0]
                if c < best_c - 1e-14:
                    best_c = c
                    best_x = cand
        except Exception:
            pass

    # Try binary version of periodic
    b_periodic = (unpack(best_x) >= 0.5).astype(np.float64)
    cnt = int(np.round(0.5 * n))
    if int(np.sum(b_periodic)) != cnt:
        idx = np.argsort(-unpack(best_x))
        b_periodic = np.zeros(n)
        b_periodic[idx[:cnt]] = 1.0
    c_bin = exact_objective(b_periodic)[0]
    if c_bin < best_c:
        return b_periodic, c_bin
    return unpack(best_x), best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load parent
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0, _ = exact_objective(init)
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 70
        best_h = make_start(n0, rng, mode=0)
        best_c, _ = exact_objective(best_h)

    n0 = len(best_h)

    # Phase 1: SLSQP on parent
    if time.time() < deadline - 120:
        h_opt, c_opt = slsqp_refine(best_h, n0, deadline,
                                    tau_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Binary search with 1-for-1 + 2-for-2
    if n0 % 2 == 0 and time.time() < deadline - 80:
        b = (best_h >= 0.5).astype(np.float64)
        cnt = int(round(0.5 * n0))
        if int(np.sum(b)) != cnt:
            idx = np.argsort(-best_h)
            b = np.zeros(n0)
            b[idx[:cnt]] = 1.0
        b, c_b = binary_combined_search(b.copy(), deadline - 80, rng)
        if c_b < best_c:
            best_c = c_b
            best_h = b
        # SLSQP from binary
        if time.time() < deadline - 60:
            h_opt, c_opt = slsqp_refine(b, n0, deadline,
                                        tau_list=[0.002, 0.001, 0.0005, 0.0002, 0.0001])
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            # Another binary round from SLSQP result
            if time.time() < deadline - 40 and len(h_opt) % 2 == 0:
                b2 = (h_opt >= 0.5).astype(np.float64)
                cnt2 = int(round(0.5 * len(h_opt)))
                if int(np.sum(b2)) != cnt2:
                    idx2 = np.argsort(-h_opt)
                    b2 = np.zeros(len(h_opt))
                    b2[idx2[:cnt2]] = 1.0
                b2, c_b2 = binary_combined_search(b2, deadline - 30, rng)
                if c_b2 < best_c:
                    best_c = c_b2
                    best_h = b2

    # Phase 3: Try periodic solutions for current n
    if n0 > 8 and time.time() < deadline - 60:
        for p_candidate in [2, 3, 4, 5, 6, 7, 8, 10, 12, 14, 16]:
            if time.time() > deadline - 30:
                break
            if n0 % p_candidate == 0 and p_candidate >= 4:
                h_per, c_per = optimize_periodic(p_candidate, n0, deadline, rng)
                if c_per < best_c:
                    best_c = c_per
                    best_h = h_per

    # Phase 4: Try different n values
    sizes_to_try = []
    for s in range(30, 130):
        if s % 2 == 0 and 16 <= s <= 200:
            sizes_to_try.append(s)
    # Prioritize sizes near n0 and those with many divisors
    def size_priority(s):
        divs = sum(1 for d in range(1, s + 1) if s % d == 0)
        return abs(s - n0) * 10 - divs * 3
    sizes_to_try.sort(key=size_priority)

    for n_try in sizes_to_try[:20]:
        if n_try == n0:
            continue
        if time.time() > deadline - 20:
            break
        # Try a few starts
        for mode in [0, 2, 5, 6]:
            if time.time() > deadline - 15:
                break
            h_start = make_start(n_try, rng, mode=mode)
            h_ds = downsample(best_h, n_try)
            c_ds, _ = exact_objective(h_ds)
            c_st, _ = exact_objective(h_start)
            if c_ds < c_st:
                h_start = h_ds
            h_opt, c_opt = slsqp_refine(h_start, n_try, deadline,
                                        tau_list=[0.005, 0.002, 0.001, 0.0005])
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            # Binary refinement
            if n_try % 2 == 0 and c_opt < best_c + 0.01 and time.time() < deadline - 10:
                b_try = (h_opt >= 0.5).astype(np.float64)
                cnt_try = int(round(0.5 * n_try))
                if int(np.sum(b_try)) != cnt_try:
                    idx_try = np.argsort(-h_opt)
                    b_try = np.zeros(n_try)
                    b_try[idx_try[:cnt_try]] = 1.0
                b_try, c_btry = binary_combined_search(b_try, min(deadline - 10, time.time() + 20), rng)
                if c_btry < best_c:
                    best_c = c_btry
                    best_h = b_try

    # Phase 5: Final polish on best solution
    if time.time() < deadline - 10:
        n_best = len(best_h)
        # SLSQP polish
        h_opt, c_opt = slsqp_refine(best_h, n_best, deadline,
                                    tau_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        # Final binary attempt
        if len(best_h) % 2 == 0:
            b_final = (best_h >= 0.5).astype(np.float64)
            cnt_f = int(round(0.5 * len(best_h)))
            if int(np.sum(b_final)) != cnt_f:
                idx_f = np.argsort(-best_h)
                b_final = np.zeros(len(best_h))
                b_final[idx_f[:cnt_f]] = 1.0
            b_final, c_final = binary_combined_search(b_final, deadline, rng)
            if c_final < best_c:
                best_c = c_final
                best_h = b_final

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)

    try:
        eval_c = evaluate_erdos_solution(best_h)
        if eval_c is not None and eval_c < best_c:
            best_c = float(eval_c)
    except Exception:
        pass

    return (best_h, best_c, best_n)
