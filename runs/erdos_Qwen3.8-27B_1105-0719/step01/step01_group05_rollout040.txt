import numpy as np

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(profile) * dx), profile

def project_box_sum(h, target):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final fixup
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Find an element to adjust
        for idx in range(n):
            if (diff > 0 and h[idx] < 1.0) or (diff < 0 and h[idx] > 0.0):
                h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
                break
    return h

    best_c = np.max(profile)

                h[j] -= d

                update_C_for_change(C, i, d, n)

                best_c_sweep = f_new

    profile = np.correlate(h, 1.0 - h, mode='full')
    best_c = np.max(profile)
    
    return h, best_c

    C = np.zeros((n, 2*n-1))
    offsets = np.arange(2*n-1) - (n-1)  # lags
    for k in range(n):

        idx1 = k + offsets
        valid1 = (idx1 >= 0) & (idx1 < n)
        C[k, valid1] += (1.0 - h[idx1[valid1]])
        # Second term: -h[k-lag] where k-lag in [0,n-1]
        idx2 = k - offsets
        valid2 = (idx2 >= 0) & (idx2 < n)
        C[k, valid2] -= h[idx2[valid2]]
    return C

        m2 = k - i + n - 1

    """Minimize max(profile + c*delta) over delta in [L,U]. Returns (delta, value)."""
    if U - L < 1e-14:
        return L, float(np.max(profile + c * L))

        return L, f_L
    elif f_U <= f_d and f_U <= f_L:
        return U, f_U
    return d, f_d

                update_C_for_change(C, i, d, n)
                update_C_for_change(C, j, -d, n)
                best_c_sweep = f_new
                improved = True
        
        if best_c_sweep < best_c:

    if rng is None:
        rng = np.random.RandomState(42)
    
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 3.0:
            break
        
        if sweep > 0:
            rng.shuffle(pairs)
        
        improved = False
        
        for i, j in pairs:
            hi = h[i]
            hj = h[j]
            L = max(-hi, hj - 1.0)
            U = min(1.0 - hi, hj)
            if U - L < 1e-13:
                continue
            
            c = C[i] - C[j]
            d, f_new = ternary_min(profile, c, L, U, niter)
            
            if f_new < best_c - 1e-12:
                h[i] += d
                h[j] -= d
                profile += c * d
                update_C_for_change(C, i, d, n)
                update_C_for_change(C, j, -d, n)
                best_c = f_new
                improved = True
        
        if not improved:
            break
    
    h = project_box_sum(h, 0.5 * n)
    profile = np.correlate(h, 1.0 - h, mode='full')
    return h, float(np.max(profile))

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        cnt = int(round(target))
        idx = rng.choice(n, size=min(cnt, n), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(2, 6)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.1
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)

def perturb(h, rng, sigma):
    n = len(h)
    h_new = h + rng.randn(n) * sigma
    return project_box_sum(h_new, 0.5 * n)
