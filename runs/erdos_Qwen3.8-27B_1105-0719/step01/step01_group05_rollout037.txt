def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)

        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count

    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)

def swap_delta_profile(b, p, q):
    """
    Compute the change in the full correlation profile when b[p] goes 1->0 and b[q] goes 0->1.
    Returns the delta profile (change in each entry of the full correlation).
    """
    n = len(b)
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    
    # b[p] changes from 1 to 0, b[q] changes from 0 to 1
    # For each lag ℓ (from -(n-1) to n-1), compute the change
    
    for ell in range(-(n-1), n):
        if ell == 0:
            continue  # corr[0] = ∑ h[i](1-h[i]) changes by: for binary, was 0, still 0
            # Actually for binary, h[i](1-h[i]) = 0 always, so no change. But wait:
            # If b[p] was 1, h[p](1-h[p]) = 0. After change to 0, h[p](1-h[p]) = 0. No change.
            # If b[q] was 0, h[q](1-h[q]) = 0. After change to 1, h[q](1-h[q]) = 0. No change.
            # So delta at lag 0 is always 0 for binary. Good.

            d = 0.0
            if p >= ell:

            if p + ell < n:

            # b[q] contribution:
            if q >= ell:

            if q + ell < n:

            delta[ell + n - 1] = d * dx

def swap_delta_raw(b, p, q, n):
    """
    Compute the raw (unnormalized) change in the full correlation profile
    when b[p] goes 1->0 and b[q] goes 0->1.
    Returns an array of length 2*n-1.
    """
    delta = np.zeros(2 * n - 1)
    # b[p]: 1 -> 0
    # Positive lag ell: contribution from b[p]
    #   if p >= ell: -(1 - b[p-ell])
    #   if p+ell < n: +b[p+ell]
    # b[q]: 0 -> 1
    # Positive lag ell: contribution from b[q]
    #   if q >= ell: +(1 - b[q-ell])
    #   if q+ell < n: -b[q+ell]
    
    # Vectorized: for ell = 1..n-1
    ells = np.arange(1, n)
    
    # Positive lags
    # b[p] 1->0:
    mask_p_left = ells <= p
    mask_p_right = (p + ells) < n
    d_pos = np.zeros(n - 1)
    d_pos[mask_p_left] += -(1 - b[p - ells[mask_p_left]])
