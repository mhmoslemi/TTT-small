import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_grad_exact(h):
    """Compute gradient of C5 = dx * max_k C[k] at the active lag."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    
    # Find active lags (within tolerance of max)
    tol = 1e-8 + 1e-6 * max_val
    active = np.where(corr >= max_val - tol)[0]
    
    # For each active lag, compute gradient and average
    grad = np.zeros(n)
    for lag_idx in active:
        # lag_idx corresponds to offset k = lag_idx - (n-1)
        k = lag_idx - (n - 1)
        # C[k] = sum_i h[i] * (1 - h[i+k]) for valid i
        # dC[k]/dh[j] = (1 - h[j+k]) - h[j-k]  (boundary terms = 0)
        g_local = np.ones(n)
        # Subtract h[j+k] for valid j+k
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            g_local[lo1:hi1] -= h[lo1 + k: hi1 + k]
        # Subtract h[j-k] for valid j-k
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            g_local[lo2:hi2] -= h[lo2 - k: hi2 - k]
        grad += g_local
    
    grad /= len(active)
    return grad * dx, corr, max_val, active

def project_feasible(y, n):
    """Project y onto {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0
    # Bisection on threshold t: h = clip(y - t, 0, 1)
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute among non-bound values
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            adj = residual / len(eligible)
            h[eligible] = np.clip(h[eligible] + adj, 0.0, 1.0)
            # If still off, try more

            if abs(residual2) > 1e-12:
                # Spread over all
                h += residual2 / n
                h = np.clip(h, 0.0, 1.0)
    return h

def gradient_descent_step(h, max_iter=2000, tol=1e-14):
    """Active-set gradient descent with backtracking line search."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for it in range(max_iter):
        # Compute gradient at active lag
        grad, corr, max_val, active = compute_grad_exact(best_h)
        
        # Check if gradient is near zero (at optimum or at a point where 
        # moving in any direction doesn't help)
        # The effective gradient (projected to preserve sum) is:
        grad_proj = grad - np.mean(grad)  # Project onto sum=0 subspace
        grad_norm = np.linalg.norm(grad_proj)
        if grad_norm < tol:
            break
        
        # Direction: negative projected gradient
        direction = -grad_proj
        
        # Backtracking line search
        alpha = 0.5
        for bt in range(30):
            h_try = best_h + alpha * direction
            # Project to feasible set
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_h = h_try
                best_c5 = c5_try
                break
            alpha *= 0.5
        else:
            # No improvement found
            break
    
    return best_h, best_c5

def compute_c5(h):

def compute_gradient_at_active(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    tol = 1e-10 + 1e-8 * max_val
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        g = np.ones(n)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            g[lo1:hi1] -= h[lo1 + k:hi1 + k]
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 - k:hi2 - k]
        grad += g
    grad /= max(len(active), 1)

def project_feasible(y, n):

def smooth_max_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        g = np.ones(n)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            g[lo1:hi1] -= h[lo1 + k:hi1 + k]
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 - k:hi2 - k]
        grad += w * g
    return grad * dx


def slsqp_smooth(h0, beta, max_iter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj_fun(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = beta * (corr - m)
        exps = np.exp(shifted)
        return (m + np.log(np.sum(exps)) / beta) * (2.0 / n)

    def obj_grad(x):
        return smooth_max_gradient(x, beta)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    res = minimize(
        obj_fun, h0, jac=obj_grad,
        method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
        options={'maxiter': max_iter, 'ftol': 1e-16}
    )
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_out = project_feasible(res.x, n)
        c5_out = compute_c5(h_out)
        return h_out, c5_out
    return h0, compute_c5(h0)

                best_n = len(best_h)
            if remaining() > 15:
                n_ls = min(30000, int(remaining() * 35))
                h_opt, c5_opt = targeted_search(best_h, n_ls, rng, scale=0.02)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                    best_n = len(best_h)

    # Phase 6: Multi-resolution search
    if remaining() > 70:
        for n_try in [48, 56, 72, 84, 36]:
            if remaining() < 25:
                break
            if n_try == best_n:
                continue
            x_old = np.linspace(0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c5_opt = slsqp_smooth(h_interp, 500, max_iter=200)
            h_opt, c5_opt = grad_descent(h_opt, max_iter=4000)
            n_ls = min(15000, int(remaining() * 15))
            h_opt, c5_opt = targeted_search(h_opt, n_ls, rng, scale=0.03)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 7: Random restarts from perturbed best
    if remaining() > 60:
        for restart in range(6):
            if remaining() < 20:
                break
            strategy = restart % 3
            if strategy == 0:
                h_pert = best_h + 0.06 * rng.randn(best_n)
            elif strategy == 1:
                h_pert = best_h.copy()
                mask = rng.random(best_n) < 0.1
                h_pert[mask] = 1.0 - h_pert[mask]
            else:
                shift = rng.randint(1, best_n)
                h_pert = np.roll(best_h, shift)
            h_pert = project_feasible(h_pert, best_n)
            h_opt, c5_opt = slsqp_smooth(h_pert, 1000, max_iter=150)
            h_opt, c5_opt = grad_descent(h_opt, max_iter=3000)
            if remaining() > 15:
                h_opt, c5_opt = targeted_search(h_opt, min(15000, int(remaining() * 20)), rng, scale=0.025)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = len(best_h)

    # Phase 8: Final SLSQP + gradient polish
    if remaining() > 25:
        h_opt, c5_opt = slsqp_smooth(best_h, 50000, max_iter=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(best_h)
    if remaining() > 10:
        h_opt, c5_opt = grad_descent(best_h, max_iter=5000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(best_h)

    # Phase 9: Final targeted search
    if remaining() > 8:
        n_iter = min(40000, int(remaining() * 40))
        h_opt, c5_opt = targeted_search(best_h, n_iter, rng, scale=0.015)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(best_h)

    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
