import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] = np.minimum(h[mask] + diff / np.sum(mask), 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] = np.maximum(h[mask] + diff / np.sum(mask), 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def get_coord_gradient(h, j):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    k_vals = np.arange(n - 1, -(n), -1, dtype=np.int64)
    idx1 = j + k_vals
    valid1 = (idx1 >= 0) & (idx1 <= n - 1)
    idx2 = j - k_vals
    valid2 = (idx2 >= 0) & (idx2 <= n - 1)
    g = np.zeros(2 * n - 1)
    if np.any(valid1):
        g[valid1] += u[idx1[valid1]]
    if np.any(valid2):
        g[valid2] -= h[idx2[valid2]]
    return g * dx

def coord_1d_solve(a, b, s, target, penalty):
    def obj(t):
        return np.max(a * t + b) + penalty * (s + t - target) ** 2
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coordinate_descent(h, n_sweeps=40, penalty=5e3, tol=1e-11):
    n = len(h)
    target = 0.5 * n
    h = project_box_sum(h, target)
    best_c, _ = exact_objective(h)
    for sweep in range(n_sweeps):
        profile = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        sum_h = np.sum(h)
        max_change = 0.0
        for j in range(n):
            g = get_coord_gradient(h, j)
            a = g
            b = profile - g * h[j]
            s = sum_h - h[j]
            new_t = coord_1d_solve(a, b, s, target, penalty)
            delta = abs(new_t - h[j])
            h[j] = new_t
            sum_h += new_t - h[j] + h[j] - new_t
            # recompute sum_h properly
            sum_h = np.sum(h)  # expensive but correct; let's avoid
            max_change = max(max_change, delta)
        h = project_box_sum(h, target)
        c, _ = exact_objective(h)
        if c < best_c:
            best_c = c
        if max_change < tol:
            break
    return h, best_c

def coordinate_descent_fast(h, n_sweeps=40, penalty=5e3, tol=1e-11):
    n = len(h)
    target = 0.5 * n
    h = project_box_sum(h, target)
    best_c, _ = exact_objective(h)
    for sweep in range(n_sweeps):
        profile = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_change = 0.0
        for j in range(n):
            s = np.sum(h) - h[j]
            g = get_coord_gradient(h, j)
            a = g
            b = profile - g * h[j]
            new_t = coord_1d_solve(a, b, s, target, penalty)
            delta = abs(new_t - h[j])
            h[j] = new_t
            max_change = max(max_change, delta)
        h = project_box_sum(h, target)
        c, _ = exact_objective(h)
        if c < best_c:
            best_c = c
        if max_change < tol:
            break
    return h, best_c

def slsqp_refine(h_start, tau_list, deadline, max_iter=150):
    n = len(h_start)
    dx = 2.0 / n
    target = 0.5 * n
    best_h = project_box_sum(h_start, target)
    best_c, _ = exact_objective(best_h)
    bounds = [(0.0, 1.0)] * n

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

    for tau in tau_list:
        if time.time() > deadline - 3:
            break

        def obj_grad(x):
            profile = np.correlate(x, 1.0 - x, mode='full') * dx
            pmax = np.max(profile)
            shifted = np.clip((profile - pmax) / tau, -80, 80)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            value = pmax + tau * np.log(wsum)
            wnorm = weights / wsum
            grad = np.zeros(n)
            for m in range(2 * n - 1):
                if wnorm[m] < 1e-8:
                    continue
                k = n - 1 - m
                idx1 = np.arange(max(0, -k), min(n, n - k))
                idx2 = np.arange(max(0, k), min(n, n + k))
                u = 1.0 - x
                if len(idx1) > 0:
                    grad[idx1] += wnorm[m] * u[idx1 + k] * dx
                if len(idx2) > 0:
                    grad[idx2] -= wnorm[m] * x[idx2 - k] * dx
            se = np.sum(x) - target
            value += 1e5 * se * se
            grad += 2e5 * se * np.ones(n)
            below = np.maximum(0.0, -x)
            above = np.maximum(0.0, x - 1.0)
            value += 1e5 * (np.sum(below * below) + np.sum(above * above))
            grad -= 2e5 * below
            grad += 2e5 * above
            return value, grad

        try:
            mi = min(max_iter, n * 2 + 50)
            res = minimize(obj_grad, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraint, jac=True,
                           options={'maxiter': mi, 'ftol': 1e-12})
            if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                c, _ = exact_objective(cand)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = cand
        except Exception:
            pass
    return best_h, best_c

def binary_local_search(b, max_rounds=50, rng=None):
    n = len(b)
    if n % 2 != 0:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(0)
    c5, _ = exact_objective(b)
    ones = list(np.where(b > 0.5)[0])
    zeros = list(np.where(b < 0.5)[0])
    for rnd in range(max_rounds):
        improved = False
        n_pairs = len(ones) * len(zeros)
        n_tries = min(n_pairs, 2000) if n_pairs <= 10000 else 1000
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = b[j], b[i]
            nc, _ = exact_objective(b)
            if nc < c5 - 1e-13:
                c5 = nc
                improved = True
                ones = list(np.where(b > 0.5)[0])
                zeros = list(np.where(b < 0.5)[0])
            else:
                b[i], b[j] = b[j], b[i]
        if not improved:
            break
    return b, c5

def binary_local_search_full(b, max_rounds=30, rng=None):

def interpolate_to(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def threshold_binary(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def make_random_binary(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    return h

def make_structured(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        idx = rng.choice(n, size=int(round(target)), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(2, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.zeros(n)
        q = n // 4
        offset = rng.randint(0, n // 2)
        h[offset:offset + q] = 1.0
        h[offset + n // 2:offset + n // 2 + q] = 1.0
    else:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.1
    return project_box_sum(h, target)

        pass

    if best_h is None:

    n0 = len(best_h)

    # Phase 1: Coordinate descent at initial size
