The issue with this approach is that it can be very slow to converge, especially when the improvement per step is tiny (like 1e-5).

**Better approach: Direct QP on active lags**

Since the number of active lags is small (say 3-10), I can try to directly minimize the maximum of those specific quadratic functions. This is a non-convex problem, but for small perturbations, I can linearize:

Linearize f_k(h + d) ≈ f_k(h) + ∇f_k(h)ᵀ d

Then solve: minimize t subject to f_k(h) + ∇f_k(h)ᵀ d ≤ t for active k, ||d|| ≤ δ, 0 ≤ h+d ≤ 1, ∑(h+d) = n/2

This is a linear program in (t, d)! And since n is small (70), this is very tractable.

After solving the LP, I take the step d (possibly scaled down) and re-evaluate the true (nonlinear) objective. If it improved, accept; else, shrink δ.

This is essentially a **linearized subgradient method** with exact line search, which should be more effective than plain subgradient.

Let me implement this:

```python
def lp_direction(h, active_lags, profile, dx, trust_radius, target, n):
    """Solve LP to find descent direction for max of active lag objectives."""
    from scipy.optimize import linprog
    
    # Variables: d_0, ..., d_{n-1}, t
    # d represents the step, t is the upper bound on the active objectives
    
    # Objective: minimize t (or minimize t + small_penalty * ||d||²... but that's QP)
    # Actually, minimize t - beta * ||d|| (to encourage movement)
    # Or just minimize t, and the constraint ||d|| <= radius will bound the step
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Constraints: for each active lag k:
    # f_k(h) + grad_k^T d <= t
    # => grad_k^T d - t <= -f_k(h)
    
    A_ub = []
    b_ub = []
    
    for lag in active_lags:
        g = lag_gradient(h, lag, dx)  # shape (n,)
        row = np.append(g, -1.0)  # grad^T d - t
        A_ub.append(row)
        b_ub.append(-profile[lag])  # Wait, profile[lag] is already * dx
    
    # Hmm, I need to be careful. Let me use:
    # f_k(h) = profile[lag_index] (already includes dx)
    # Linearization: f_k(h+d) ≈ f_k(h) + grad_k^T d
    # Constraint: f_k(h) + grad_k^T d <= t
    # => grad_k^T d - t <= -f_k(h)
    
    # But I want to allow t to be below the current max, so I should add a "descent target"
    # Actually, I want to minimize t, so the LP will find the best t achievable
    # with the given step direction, bounded by trust radius.
    
    # Trust region: ||d||_inf <= trust_radius
    # => d_i <= trust_radius, -d_i <= trust_radius
    # => d_i - 0 <= trust_radius
    # => -d_i <= trust_radius
    
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(trust_radius)
        
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_ub.append(row)
        b_ub.append(trust_radius)
    
    # Box constraints on h+d: 0 <= h_i + d_i <= 1
    # => d_i >= -h_i, d_i <= 1 - h_i
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_ub.append(row)
        b_ub.append(h[i])  # -d_i <= h_i => d_i >= -h_i
        
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - h[i])  # d_i <= 1 - h_i
    
    # Sum constraint: sum(h+d) = target => sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # t is free (no bounds on t)
    # d is bounded by the constraints above
    
    bounds = [(None, None)] * n + [(None, None)]  # all free, bounded by A_ub
    
    result = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n]
        t_new = result.x[-1]
        return d, t_new
    return None, None
```

This LP has n+1 = 71 variables and about (active_lags + 3*n) constraints. For active_lags=10 and n=70, that's about 220 constraints. This is very fast to solve.

The key insight: the LP gives us a direction d that, to first order, reduces all active lag objectives simultaneously. The exact line search then checks if the true (nonlinear) objective actually decreases.

Let me now put together the full algorithm:

```python
def active_lag_optimize(h, deadline, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    dx = 2.0 / n
    
    best_h = h.copy()
    best_c, best_profile = exact_objective(h)
    
    trust = 0.05
    stall_count = 0
    
    while time.time() < deadline - 5:
        c, profile = exact_objective(best_h)
        if c <= best_c:
            best_c = c
            best_profile = profile
        else:
            break
        
        # Find active lags (within 0.01*c of max)
        tol = max(0.002, 0.03 * c)
        active_idx = np.where(profile >= best_c - tol)[0]
        active_lags = active_idx - (n - 1)
        
        # Limit to top 15 active lags
        if len(active_idx) > 15:
            sorted_idx = active_idx[np.argsort(-profile[active_idx])][:15]
            active_lags = sorted_idx - (n - 1)
        
        # Solve LP for descent direction
        d, t_new = lp_direction(best_h, active_lags, profile, dx, trust, target, n)
        
        if d is None:
            # Fallback: negative gradient of max lag
            g = lag_gradient(best_h, active_lags[0], dx)
            g -= g.mean()  # ensure sum zero
            norm = np.max(np.abs(g))
            if norm > 0:
                d = -trust * g / norm
            else:
                break
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_box_sum(best_h + alpha * d, target)
            c_try, _ = exact_objective(h_try)
            if c_try < best_c - 1e-12:
                best_h = h_try
                best_c = c_try
                improved = True
                if alpha >= 0.5:
                    trust = min(0.2, trust * 1.3)
                break
        
        if not improved:
            trust *= 0.6
            stall_count += 1
            if stall_count > 30 or trust < 1e-6:
                break
        else:
            stall_count = 0
    
    return best_h, best_c

Let me also think about the binary local search more carefully.

For a binary solution with n=70 (35 ones, 35 zeros):
- The overlap at lag k is an integer (count of 1-0 pairs at that lag)
- The objective is (max_count) * dx
- To reduce the max_count by 1, I need to find a swap that reduces the count at the worst lag

Single swap: flip position a (1→0) and position b (0→1).
- This changes the count at lag k by:
  - -1 if h[a]=1, h[a+k]=0 (we removed a 1-0 pair at lag k) [for k>0, a+k<n]
  - +1 if h[a-k]=1, h[a]=0 → wait, a is being changed to 0
  
Let me think more carefully. After the swap:
- h'[a] = 0 (was 1)
- h'[b] = 1 (was 0)

For lag k > 0, the count is:
count(k) = #{i : 0≤i≤n-1-k, h'[i]=1, h'[i+k]=0}

The change in count(k) due to the swap:
- Position a: was contributing h[a]=1. After swap, h'[a]=0.
  - If a ≤ n-1-k: we lose the pair (a, a+k) if h[a+k]=0. Change: -1 (if h[a+k]=0)
  - If a-k ≥ 0: we gain the pair (a-k, a) if h[a-k]=1. Change: +1 (if h[a-k]=1)
    Wait no. Before: h[a-k] and h[a]=1, so this was NOT a 1-0 pair (it was 1-1).
    After: h[a-k] and h'[a]=0. If h[a-k]=1, this IS a 1-0 pair. Change: +1 (if h[a-k]=1)
    
- Position b: was contributing h[b]=0. After swap, h'[b]=1.
  - If b ≤ n-1-k: we gain the pair (b, b+k) if h[b+k]=0. Change: +1 (if h[b+k]=0)
    Before: h[b]=0, so this was NOT a 1-0 pair.
    After: h'[b]=1, h[b+k]=0. This IS a 1-0 pair. Change: +1.
  - If b-k ≥ 0: we lose the pair (b-k, b) if h[b-k]=1. Change: -1 (if h[b-k]=1)
    Before: h[b-k]=1, h[b]=0. This WAS a 1-0 pair.
    After: h[b-k]=1, h'[b]=1. This is NOT a 1-0 pair. Change: -1.

So the total change in count(k) for a single swap (a:1→0, b:0→1) at lag k>0:
Δcount(k) = -[h[a+k]=0 and a≤n-1-k] + [h[a-k]=1 and a≥k] + [h[b+k]=0 and b≤n-1-k] - [h[b-k]=1 and b≥k]

This is at most ±2 per lag. For the swap to reduce the maximum, we need Δcount(k*) < 0 at the lag k* achieving the maximum, AND Δcount(k) < 0 (or ≤ 0) for all other lags k where count(k) = max_count.

This gives a very targeted local search! I only need to check the change at the currently-maximizing lags.

                        # Actually let me just use the same formula with the profile index

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    
    # Bisection for the shift
    def clipped_sum(shift):
        return np.sum(np.clip(h - shift, 0.0, 1.0))
    
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if clipped_sum(mid) > target:
            lo = mid
        else:
            hi = mid
    shift = (lo + hi) / 2.0
    result = np.clip(h - shift, 0.0, 1.0)
    
    # Fix residual
    diff = target - np.sum(result)
    if abs(diff) > 1e-12 and n > 0:
        if diff > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            result[idx[0]] = np.clip(result[idx[0]] + diff, 0.0, 1.0)
    return result

def lag_gradient(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] += (1.0 - h[lag:])
            grad[lag:] -= h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[a:] += (1.0 - h[:s])
            grad[:s] -= h[a:]
    return grad * dx

def lp_descent_direction(h, active_idx, profile, dx, trust, target, n):
    """Solve LP: min t s.t. grad_k^T d - t <= -profile[k], ||d||_inf <= trust, box, sum=0"""
    # Variables: [d_0, ..., d_{n-1}, t]
    nv = n + 1
    
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub_rows = []
    b_ub_vals = []
    
    for idx in active_idx:
        lag = idx - (n - 1)
        g = lag_gradient(h, lag, dx)
        row = np.zeros(nv)
        row[:n] = g
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(-profile[idx])
    
    # Trust region: |d_i| <= trust
    for i in range(n):
        row = np.zeros(nv); row[i] = 1.0
        A_ub_rows.append(row); b_ub_vals.append(trust)
        row = np.zeros(nv); row[i] = -1.0
        A_ub_rows.append(row); b_ub_vals.append(trust)
    
    # Box: 0 <= h_i + d_i <= 1
    for i in range(n):
        row = np.zeros(nv); row[i] = -1.0
        A_ub_rows.append(row); b_ub_vals.append(h[i])
        row = np.zeros(nv); row[i] = 1.0
        A_ub_rows.append(row); b_ub_vals.append(1.0 - h[i])
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Sum constraint: sum(d) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * nv
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        d = res.x[:n].copy()
        t_val = res.x[-1]
        return d, t_val
    return None, None

def active_lag_optimize(h, deadline, max_iters=500):
    n = len(h)
    target = 0.5 * n
    dx = 2.0 / n
    
    h = h.copy()
    best_c, best_prof = exact_objective(h)
    trust = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        
        c, profile = exact_objective(h)
        
        # Find active lags
        tol = max(0.003, 0.05 * c)
        active_mask = profile >= c - tol
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) > 12:
            # Keep top 12 by profile value
            order = np.argsort(-profile[active_idx])[:12]
            active_idx = active_idx[order]
        
        # Solve LP
        d, t_new = lp_descent_direction(h, active_idx, profile, dx, trust, target, n)
        
        if d is None:
            # Fallback: negative gradient of top lag
            top_lag = int(np.argmax(profile)) - (n - 1)
            g = lag_gradient(h, top_lag, dx)
            g -= g.mean()
            gm = np.max(np.abs(g))
            if gm < 1e-15:
                break
            d = -trust * g / gm
        
        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05]:
            h_try = project_box_sum(h + alpha * d, target)
            c_try, _ = exact_objective(h_try)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                improved = True
                if alpha >= 0.5:
                    trust = min(0.15, trust * 1.2)
                elif alpha >= 0.1:
                    trust = min(0.15, trust * 1.05)
                break
        
        if not improved:
            trust *= 0.65
            stall += 1
            if stall > 40 or trust < 1e-7:
                break
        else:
            stall = 0
    
    return h, best_c

def binary_local_search(b, deadline, max_rounds=200):

    b = b.astype(np.float64).copy()

def binary_sa(b, deadline, t0=0.02, cooling=0.95, max_iter=5000):
    """Simulated annealing for binary solution."""
    n = len(b)
    b = b.astype(np.float64).copy()
    c = float(np.max(np.correlate(b, 1-b, mode='full'))) * (2.0/n)
    best_b = b.copy()
    best_c = c
    T = t0
    rng = np.random.RandomState(0)
    
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        if it % 100 == 99:
            T *= cooling
        
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        
        a = ones[rng.randint(len(ones))]
        b_idx = zeros[rng.randint(len(zeros))]
        
        b[a] = 0.0
        b[b_idx] = 1.0
        nc = float(np.max(np.correlate(b, 1-b, mode='full'))) * (2.0/n)
        delta = nc - c
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c = nc
            if nc < best_c:
                best_c = nc
                best_b = b.copy()
        else:
            b[a] = 1.0
            b[b_idx] = 0.0
    
    return best_b, best_c
