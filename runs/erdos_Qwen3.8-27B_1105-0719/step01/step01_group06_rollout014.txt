    h = np.clip(h, 0.0, 1.0)
    n = len(h)

        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    # Final fix for floating point
    residual = target - h.sum()
    if abs(residual) > 0:
        # Apply to the element with most room
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + residual)
    return h

    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    z = beta * profile
    z_max = np.max(z)
    lse = (z_max + np.log(np.sum(np.exp(z - z_max)))) / beta
    sum_viol = h.sum() - n / 2.0
    return lse + mu * sum_viol * sum_viol

    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    z = beta * profile
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / np.sum(exp_z)  # softmax weights
    
    grad = np.zeros(n)
    sum_viol = h.sum() - n / 2.0
    grad += 2.0 * mu * sum_viol

        if wk < 1e-15:
            continue
        delta = k - (n - 1)

            if d < n:
                grad[d:n] += wk * dx * (1.0 - h[0:n-d])
                grad[0:n-d] -= wk * dx * h[d:n]
    
    return grad

    betas = np.geomspace(beta_start, beta_end, n_steps)

            lse_objective, h,

- For k < 0 (d = -k): profile[k] = sum_{n=d}^{n-1} h[n-d] * (1-h[n]) * dx

Now the gradient:
For k >= 0:
∂profile[k]/∂h[j]:
- h[j] appears as h[n+k] when n = j-k, contributing (1-h[j-k])*dx (valid if 0 <= j-k <= n-1-k, i.e., k <= j <= n-1)
- h[j] appears in (1-h[n]) when n = j, contributing -h[j+k]*dx (valid if 0 <= j+k < n, i.e., j < n-k)

So:
∂profile[k]/∂h[j] = (1-h[j-k])*dx (if k <= j < n) + (-h[j+k])*dx (if j + k < n)
= (1-h[j-k])*dx (if k <= j) - h[j+k]*dx (if j < n-k)

For k = 0:
∂profile[0]/∂h[j] = (1-h[j])*dx - h[j]*dx = (1-2h[j])*dx ✓

- h[j] appears in (1-h[n]) when n = j, contributing -h[j-d]*dx (valid if j >= d)
