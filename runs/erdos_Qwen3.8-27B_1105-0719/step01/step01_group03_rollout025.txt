import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final adjustment
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def fw_oracle(g, n):
    """Solve min g^T v s.t. sum v = n/2, 0 <= v <= 1"""
    target = n / 2.0
    order = np.argsort(g)
    v = np.zeros(n)
    filled = 0.0
    for i in range(n):
        idx = order[i]
        if filled >= target - 1e-14:
            break
        add = min(1.0, target - filled)
        v[idx] = add
        filled += add
    return v

def smooth_profile_grad(h, tau):
    """Compute gradient of log-sum-exp of profile."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    mu = np.max(profile)
    shifted = (profile - mu) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(M) / M
    else:
        w /= w_sum
    
    # Compute gradient: sum_m w_m * R_m
    grad = np.zeros(n)
    for m in range(M):
        d = m - (n - 1)
        if abs(w[m]) < 1e-15:
            continue
        wm = w[m]
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:n] += wm * dx * (1.0 - h[0:n-d])
            grad[0:n-d] -= wm * dx * h[d:n]
        else:
            s = -d
            grad[0:n-s] += wm * dx * (1.0 - h[s:n])
            grad[s:n] -= wm * dx * h[0:n-s]
    return grad

def fw_step(h, tau, max_line_alphas=8):
    """One Frank-Wolfe step with line search on exact objective."""
    n = len(h)
    g = smooth_profile_grad(h, tau)
    v = fw_oracle(g, n)
    dir_vec = v - h
    if np.linalg.norm(dir_vec) < 1e-12:
        return h
    # Line search
    best_h = h
    best_c = exact_c5(h)
    alphas = np.array([1.0/(k+2) for k in range(max_line_alphas)])
    # Also try some larger steps
    alphas = np.unique(np.concatenate([alphas, [0.5, 0.7, 0.9]]))
    alphas = np.sort(alphas)[::-1]
    for alpha in alphas:
        h_try = h + alpha * dir_vec
        h_try = repair_feasible(h_try)
        c_try = exact_c5(h_try)
        if c_try < best_c - 1e-12:
            best_h = h_try
            best_c = c_try
    return best_h

def fw_anneal(h0, tau_start, tau_end, max_iters, deadline_fn):
    """Frank-Wolfe with tau annealing."""
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_c5(h)
    for it in range(max_iters):
        if deadline_fn() < 5:
            break
        tau = tau_start * (tau_end / tau_start) ** (it / max(1, max_iters - 1))
        h = fw_step(h, tau)
        c = exact_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def active_set_descent(h, max_iters=100, threshold=1e-5, deadline_fn=None):
    """Subgradient descent on the active set of profile lags."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_c5(h)
    
    for it in range(max_iters):
        if deadline_fn is not None and deadline_fn() < 5:
            break
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-12:
            best_c = c
        
        # Find active lags
        active_thresh = c - threshold
        active = np.where(profile >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        
        # Build gradient for active set
        # Weight active lags equally (or by proximity to max)
        w = np.zeros(M)
        w[active] = 1.0 / len(active)
        
        grad = np.zeros(n)
        for m in active:
            d = m - (n - 1)
            wm = w[m]
            if d == 0:
                grad += wm * dx * (1.0 - 2.0 * h)
            elif d > 0:
                grad[d:n] += wm * dx * (1.0 - h[0:n-d])
                grad[0:n-d] -= wm * dx * h[d:n]
            else:
                s = -d
                grad[0:n-s] += wm * dx * (1.0 - h[s:n])
                grad[s:n] -= wm * dx * h[0:n-s]
        
        # Project gradient to sum-zero direction
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        
        # Line search
        improved = False
        for alpha in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001]:
            h_try = repair_feasible(h - alpha * grad)
            c_try = exact_c5(h_try)
            if c_try < best_c - 1e-12:
                h = h_try
                best_c = c_try
                improved = True
                break
        
        if not improved:
            # Try smaller threshold
            threshold = max(1e-7, threshold * 0.5)
            if threshold < 1e-7:
                break
    
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_solve(h_start, maxiter=200, tol=1e-10, deadline_fn=None):
    """SLSQP epigraph solve."""
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_c5(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = compute_profile(h)
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        J[:, -1] = 1.0
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                J[m, :n] = -dx * (1.0 - 2.0 * h)
            elif d > 0:
                J[m, d:n] += -dx * (1.0 - h[0:n-d])
                J[m, 0:n-d] += dx * h[d:n]
            else:
                s = -d
                J[m, 0:n-s] += -dx * (1.0 - h[s:n])
                J[m, s:n] += dx * h[0:n-s]
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            if h_new.sum() > 0 and np.all(h_new >= -1e-10) and np.all(h_new <= 1 + 1e-10):
                return h_new
    except Exception:
        pass
    return None

def smooth_slsqp(h_start, tau, maxiter=100, deadline_fn=None):
    """SLSQP on the smooth log-sum-exp objective."""
    n = len(h_start)
    c0 = exact_c5(h_start)
    
    def obj(x):
        h = np.clip(x, 0, 1)
        profile = compute_profile(h)
        mu = np.max(profile)
        shifted = (profile - mu) / tau
        shifted = np.clip(shifted, -50, 0)
        return mu + tau * np.log(np.exp(shifted).sum())
    
    def obj_jac(x):
        h = np.clip(x, 0, 1)
        return smooth_profile_grad(h, tau)
    
    def eq_fun(x):
        return x.sum() - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    x0 = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x)
            c_new = exact_c5(h_new)
            if c_new < c0 + 1e-6:
                return h_new
    except Exception:
        pass
    return None

    top_starts = sorted(starts, key=lambda x: exact_c5(x))[:5]
    for s in top_starts:
        if time_left() < 20:
            break
        for tau_s, tau_e, n_it in [(0.25, 0.004, 50), (0.15, 0.003, 70)]:
            if time_left() < 10:
                break
            h_fw, c_fw = fw_anneal(s, tau_s, tau_e, n_it, time_left)
            if c_fw < best_c - 1e-12:
                best_c = c_fw
                best_h = h_fw

    # === Phase 3: Active-set refinement ===
    if time_left() > 20:
        h_as, c_as = active_set_optimize(best_h, max_iters=150, thresh=5e-5, deadline_fn=time_left)
        if c_as < best_c - 1e-13:
            best_c = c_as
            best_h = h_as

    # === Phase 4: 1-h transform and refine ===
    if time_left() > 30:
        h_inv2 = repair_feasible(1.0 - best_h)
        h_as2, c_as2 = active_set_optimize(h_inv2, max_iters=100, thresh=3e-5, deadline_fn=time_left)
        if c_as2 < best_c - 1e-13:
            best_c = c_as2
            best_h = h_as2

    # === Phase 5: Smooth SLSQP ===
    if time_left() > 30:
        for tau in [0.06, 0.03, 0.015, 0.008, 0.004]:
            if time_left() < 15:
                break
            h_ss = smooth_slsqp(best_h, tau, maxiter=80)
            if h_ss is not None:
                c_ss = exact_c5(h_ss)
                if c_ss < best_c - 1e-12:
                    best_c = c_ss
                    best_h = h_ss

    # === Phase 6: Active-set with tighter threshold ===
    if time_left() > 20:
        h_as, c_as = active_set_optimize(best_h, max_iters=200, thresh=1e-6, deadline_fn=time_left)
        if c_as < best_c - 1e-13:
            best_c = c_as
            best_h = h_as

    # === Phase 7: SLSQP epigraph polish ===
    if time_left() > 40:
        h_sol = slsqp_epigraph(best_h, maxiter=400, tol=1e-13)
        if h_sol is not None:
            c_sol = exact_c5(h_sol)
            if c_sol < best_c - 1e-13:
                best_c = c_sol
                best_h = h_sol

    # === Phase 8: Swap-based local search ===
    if time_left() > 30:
        h_sw, c_sw = swap_local_search(best_h, max_swaps=300, deadline_fn=time_left)
        if c_sw < best_c - 1e-13:
            best_c = c_sw
            best_h = h_sw
        # Follow up with active set
        if time_left() > 20:
            h_as, c_as = active_set_optimize(best_h, max_iters=100, thresh=1e-6, deadline_fn=time_left)
            if c_as < best_c - 1e-13:
                best_c = c_as
                best_h = h_as

    # === Phase 9: Perturbation + refinement cycles ===
    if time_left() > 40:
        for eps in [0.005, 0.01, 0.015, 0.02, 0.03]:
            if time_left() < 25:
                break
            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_as, c_as = active_set_optimize(h_pert, max_iters=80, thresh=3e-5, deadline_fn=time_left)
            if c_as < best_c - 1e-13:
                best_c = c_as
                best_h = h_as
            # Try smooth SLSQP from perturbed
            if time_left() > 20:
                h_ss = smooth_slsqp(h_pert, 0.01, maxiter=60)
                if h_ss is not None:
                    c_ss = exact_c5(h_ss)
                    if c_ss < best_c - 1e-13:
                        best_c = c_ss
                        best_h = h_ss

    # === Phase 10: 1-h + active set again ===
    if time_left() > 25:
        h_inv3 = repair_feasible(1.0 - best_h)
        h_as3, c_as3 = active_set_optimize(h_inv3, max_iters=100, thresh=1e-6, deadline_fn=time_left)
        if c_as3 < best_c - 1e-13:
            best_c = c_as3
            best_h = h_as3

    # === Phase 11: Dimension exploration ===
    if time_left() > 50:
        n_cur = len(best_h)
        dims = []
        for d in [n_cur-4, n_cur-8, n_cur+4, n_cur+8, n_cur+12, n_cur-12, n_cur+16]:
            if 30 <= d <= 130 and d != n_cur:
                dims.append(d)
        dims.sort(key=lambda d: abs(d - n_cur))
        for new_n in dims[:3]:
            if time_left() < 35:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            # Quick active set
            h_as, c_as = active_set_optimize(h_interp, max_iters=60, thresh=3e-5, deadline_fn=time_left)
            if c_as < best_c - 1e-12:
                best_c = c_as
                best_h = h_as
            if time_left() > 25:
                h_sol = slsqp_epigraph(h_as, maxiter=200, tol=1e-12)
                if h_sol is not None:
                    c_sol = exact_c5(h_sol)
                    if c_sol < best_c - 1e-12:
                        best_c = c_sol
                        best_h = h_sol

    # === Phase 12: Final polish sequence ===
    if time_left() > 20:
        h_as, c_as = active_set_optimize(best_h, max_iters=100, thresh=5e-7, deadline_fn=time_left)
        if c_as < best_c - 1e-13:
            best_c = c_as
            best_h = h_as

    if time_left() > 25:
        h_ss = smooth_slsqp(best_h, 0.003, maxiter=80)
        if h_ss is not None:
            c_ss = exact_c5(h_ss)
            if c_ss < best_c - 1e-13:
                best_c = c_ss
                best_h = h_ss
            h_as, c_as = active_set_optimize(best_h, max_iters=60, thresh=5e-7, deadline_fn=time_left)
            if c_as < best_c - 1e-13:
                best_c = c_as
                best_h = h_as

    if time_left() > 30:
        h_sol = slsqp_epigraph(best_h, maxiter=500, tol=1e-14)
        if h_sol is not None:
            c_sol = exact_c5(h_sol)
            if c_sol < best_c - 1e-13:
                best_c = c_sol
                best_h = h_sol
        # One more active set pass
        if time_left() > 15:
            h_as, c_as = active_set_optimize(best_h, max_iters=80, thresh=1e-7, deadline_fn=time_left)
            if c_as < best_c - 1e-13:
                best_c = c_as
                best_h = h_as

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_c5(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
