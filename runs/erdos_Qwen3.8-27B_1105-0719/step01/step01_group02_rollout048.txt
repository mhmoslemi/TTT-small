def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    # Final fix
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        if diff > 0:

        else:

            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_for_lag(h, abs_k):
    """Gradient of profile at lag with magnitude abs_k, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    m = abs_k
    if m == 0:
        # p_0 = dx * sum h[i]*(1-h[i])
        # dp_0/dh[i] = dx * (1 - 2*h[i])
        return dx * (1.0 - 2.0 * h)

        # Second term: -h[j-m] for m <= j <= n-1
