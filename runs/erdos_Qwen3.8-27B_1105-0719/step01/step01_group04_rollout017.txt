def trust_region_step(h_cur, values, grads, lam, active_idx, max_iters=50):
    """Solve the trust region QP subproblem."""
    n = len(h_cur)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Linearized constraints for active lags
    constraints = []
    for idx in active_idx:
        f_k = values[idx]
        g_k = grads[idx]
        # f_k + g_k · (h - h_cur) <= t
        constraints.append(f_k + g_k @ (h_var - h_cur) <= t_var)
    
    constraints.append(cp.sum(h_var) == n / 2.0)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    obj = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h_cur))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None
    
    if h_var.value is None:
        return None
    return np.asarray(h_var.value, dtype=np.float64).flatten()

        gaps = c - values[active]
        w = np.exp(-gaps / (0.003 * max(c, 1e-10) + 1e-15))

        direction = -np.sum(w[:, None] * grads[active], axis=0)

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            for i in range(n):
                if h[i] < 1.0:
                    h[i] += min(diff2, 1.0 - h[i])
                    break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    h[i] += max(diff2, -h[i])
                    break
    return h


def compute_lags_and_grads(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    for delta in range(0, n):
        if delta == 0:
            values[n - 1] = dx * np.sum(h * (1.0 - h))
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            values[n - 1 + delta] = dx * np.sum(h[:n - delta] * (1.0 - h[delta:]))
            grads[n - 1 + delta, :n - delta] += dx * (1.0 - h[delta:])
            grads[n - 1 + delta, delta:] += -dx * h[:n - delta]
            values[n - 1 - delta] = dx * np.sum(h[delta:] * (1.0 - h[:n - delta]))
            grads[n - 1 - delta, delta:] += dx * (1.0 - h[:n - delta])
            grads[n - 1 - delta, :n - delta] += -dx * h[delta:]
    return values, grads


def interpolate_h(h, new_n):
    n = len(h)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_n) + 0.5) * 2.0 / new_n
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[h[0]], h, [h[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)


def solve_qp(h_cur, values, grads, lam, active_idx):
    n = len(h_cur)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = []
    for idx in active_idx:
        f_k = values[idx]
        g_k = grads[idx]
        # f_k + g_k . (h - h_cur) <= t  =>  g_k . h <= t - f_k + g_k . h_cur
        rhs = t_var - f_k + g_k @ h_cur
        constraints.append(g_k @ h_var <= rhs)
    constraints.append(cp.sum(h_var) == n / 2.0)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    obj = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h_cur))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None
    if h_var.value is None:
        return None
    return np.asarray(h_var.value, dtype=np.float64).flatten()

    t_start = time.perf_counter()
    best = h.copy()
    best_score = exact_score(best)

    for step in range(max_steps):
        if time.perf_counter() - t_start > time_limit:
            break
        values, grads = compute_lags_and_grads(best)
        c = np.max(values)

        threshold = c - 0.004 * max(c, 0.01)
        active = np.where(values >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(values)])
        if len(active) > 30:
            active = np.argsort(values)[::-1][:30]

        gaps = c - values[active]
        w = np.exp(-gaps / (0.003 * max(c, 1e-10) + 1e-15))
        w /= w.sum() + 1e-30
        direction = -np.sum(w[:, None] * grads[active], axis=0)
        direction -= np.mean(direction)

                best_score = s

                break
