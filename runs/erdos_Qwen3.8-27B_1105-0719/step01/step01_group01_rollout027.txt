import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def soft_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

def obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, weights = soft_max(corr, beta)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        # First term: +w*(1-h[j-k]) for j in [max(0,k), min(n-1, n-1+k)]
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1-k:hi1+1-k])
        # Second term: -w*h[j+k] for j in [max(0,-k), min(n-1, n-1-k)]
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2+k:hi2+1+k]
    return val * dx, grad * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible[0]] += residual
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_run(h0, betas, max_iter=300, ftol=1e-14):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0, 'jac': lambda h: np.ones(n)}]
    
    for beta in betas:
        def fun(h):
            v, _ = obj_and_grad(h, beta)
            return v
        def jac(h):
            _, g = obj_and_grad(h, beta)
            return g
        
        res = minimize(fun, best_h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.success or (res.x is not None and compute_c5(res.x) < best_c5):
            c5_new = compute_c5(res.x)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = res.x.copy()
    return best_h, best_c5

        # Or position i where h[i-k_worst] is high and h[i] is low
        # Transfer mass from bad to good

        i = rng.randint(n)
        # Try transferring from i to j (where j = i + k_worst if valid)
        moves = []
        
        # Move 1: transfer from i to i+k (reduces c(k) if h[i] > h[i+k])
        j1 = i + k_worst
        if 0 <= j1 < n and best_h[i] > best_h[j1]:
            moves.append((i, j1, best_h[i] - best_h[j1]))
        
        # Move 2: transfer from i-k to i (reduces c(k) if h[i-k] > h[i])
        j2 = i - k_worst
        if 0 <= j2 < n and best_h[j2] > best_h[i]:
            moves.append((j2, i, best_h[j2] - best_h[i]))
        
        # Random swap as fallback
        if not moves:
            j = rng.randint(n)
            if i == j or abs(best_h[i] - best_h[j]) < 1e-8:
                continue
            moves.append((i, j, abs(best_h[i] - best_h[j])))
        
        # Pick a move and apply it
        src, dst, delta = moves[rng.randint(len(moves))]
        # Transfer some fraction of the difference
        eps = min(delta * rng.uniform(0.3, 0.7), 0.5)
        if eps < 1e-10:
            continue

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def soft_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

def obj_func(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = soft_max(corr, beta)
    return val * dx

def grad_func(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, weights = soft_max(corr, beta)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1-k:hi1+1-k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2+k:hi2+1+k]
    return grad * dx

def eq_fun(h, n):
    return np.sum(h) - n / 2.0

def eq_jac(h, n):
    return np.ones(n)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible[0]] += residual
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, betas, max_iter=300):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    for beta in betas:
        res = minimize(obj_func, best_h, args=(beta,), jac=grad_func,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            c5_new = compute_c5(res.x)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = res.x.copy()
    return best_h, best_c5

def gradient_descent(h0, beta, n_steps, lr_init, lr_end, rng):
    """Projected gradient descent on smooth max."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        t = step / max(n_steps - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** t
        g = grad_func(h, beta)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5

def shift_targeted_search(h, n_iter, rng):
    """Local search that targets the worst shift."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_corr = compute_corr(best_h)
    
    for it in range(n_iter):
        # Find worst shift (among non-zero shifts)
        # Zero shift is always 0 for binary, so focus on others
        k_worst = int(np.argmax(best_corr)) - (n - 1)
        
        # Strategy 1: Transfer along the worst shift
        # Find positions i where h[i] is high and h[i+k_worst] is low
        if 0 < k_worst < n:
            valid_range = range(n - k_worst)
        elif -n < k_worst < 0:
            valid_range = range(-k_worst, n)
            k_worst = -k_worst  # work with positive shift
            # In this case, we transfer from i to i-k (i.e., from i+k_original to i)
        
        # Simplified: pick random i, try transferring to i ± k_worst
        i = rng.randint(n)
        candidates = []
        for delta in [k_worst, -k_worst, 1, -1]:
            j = i + delta
            if 0 <= j < n and i != j:
                candidates.append((i, j))
        
        # Also try random pair
        j_rand = rng.randint(n)
        if j_rand != i:
            candidates.append((i, j_rand))
        
        improved = False
        for (a, b) in candidates:
            if abs(best_h[a] - best_h[b]) < 1e-8:
                continue
            # Transfer from higher to lower
            if best_h[a] > best_h[b]:
                src, dst = a, b
            else:
                src, dst = b, a
            delta = best_h[src] - best_h[dst]
            eps = delta * rng.uniform(0.3, 0.8)
            if eps < 1e-10:
                continue
            h_new = best_h.copy()
            h_new[src] -= eps
            h_new[dst] += eps
            h_new = np.clip(h_new, 0.0, 1.0)
            # Fix sum constraint
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
                best_corr = compute_corr(best_h)
                improved = True
                break
        if not improved and it % 100 == 0:
            # Occasionally do a random swap
            a, b = rng.choice(n, size=2, replace=False)
            if abs(best_h[a] - best_h[b]) > 1e-6:
                h_new = best_h.copy()
                h_new[a], h_new[b] = best_h[b], best_h[a]
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
    
    return best_h, best_c5

def binary_round_and_refine(h, n, betas):
    """Round to binary, then allow a few fractional values."""
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Fix sum: need exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        # Flip the smallest fractional values to 0
        fracs = np.argsort(h)[::-1]  # indices sorted by h value descending
        flips_needed = n_ones - n // 2
        for i in range(flips_needed):
            h_bin[fracs[i]] = 0.0
    elif n_ones < n // 2:
        fracs = np.argsort(h)  # ascending
        flips_needed = n // 2 - n_ones
        for i in range(flips_needed):
            h_bin[fracs[i]] = 1.0
    
    h_bin = project_feasible(h_bin, n)
    return slsqp_optimize(h_bin, betas, max_iter=200)
