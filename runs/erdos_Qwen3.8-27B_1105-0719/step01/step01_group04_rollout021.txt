def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    return h

    dx = 2.0 / n

            h_ref, sc = active_descent(best_h, max_steps=40, sigma=sig)
            if sc < best_score - 1e-13:
                best_h = h_ref
                best_score = sc

    # Phase 3: Focused epigraph on top lags
    if tl() > 60:
        for k in [6, 10, 15]:
            if tl() < 30:
                break
            h_sol, score = focused_epigraph(best_h, top_k=k, maxiter=80)
            if h_sol is not None and score < best_score - 1e-13:
                best_h = h_sol
                best_score = score

    # Phase 4: Dimension transfer (larger)
    if tl() > 80:
        n_cur = len(best_h)
        for nd in [n_cur + 20, n_cur + 40, min(n_cur * 2, 220)]:
            if tl() < 50:
                break
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.04:
                h_sol, score = solve_soft_max(h_interp, [20, 50, 100, 200], maxiter_per_beta=50)
                if score < best_score - 1e-13:
                    best_h = h_sol
                    best_score = score
                h_sol2, score2 = focused_epigraph(best_h, top_k=8, maxiter=60)
                if h_sol2 is not None and score2 < best_score - 1e-13:
                    best_h = h_sol2
                    best_score = score2

    # Phase 5: Dimension transfer (smaller)
    if tl() > 60:
        n_cur = len(best_h)
        for nd in [max(n_cur - 30, 60), max(n_cur - 15, 70)]:
            if tl() < 35:
                break
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.02:
                h_sol, score = solve_soft_max(h_interp, [30, 80, 150], maxiter_per_beta=50)
                if score < best_score - 1e-13:
                    best_h = h_sol
                    best_score = score

    # Phase 6: Random perturbation multi-start
    if tl() > 70:
        for trial in range(8):
            if tl() < 35:
                break
            eps = rng.uniform(0.002, 0.012)
            z = rng.uniform(-1.0, 1.0, size=len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_sol, score = solve_soft_max(h_pert, [20, 60, 120], maxiter_per_beta=40)
            if score < best_score - 1e-13:
                best_h = h_sol
                best_score = score
            h_sol2, score2 = focused_epigraph(best_h, top_k=8, maxiter=50)
            if h_sol2 is not None and score2 < best_score - 1e-13:
                best_h = h_sol2
                best_score = score2

    # Phase 7: Final soft-max with high beta
    if tl() > 50:
        h_sol, score = solve_soft_max(best_h, [50, 100, 200, 400], maxiter_per_beta=80)
        if score < best_score - 1e-13:
            best_h = h_sol
            best_score = score

    # Phase 8: Final focused epigraph
    if tl() > 30:
        h_sol, score = focused_epigraph(best_h, top_k=10, maxiter=100)
        if h_sol is not None and score < best_score - 1e-13:
            best_h = h_sol
            best_score = score

    # Phase 9: Final active descent with multiple sigmas
    if tl() > 25:
        for sig in [0.001, 0.002, 0.005, 0.01]:
            if tl() < 12:
                break
            h_ref, sc = active_descent(best_h, max_steps=30, sigma=sig)
            if sc < best_score - 1e-13:
                best_h = h_ref
                best_score = sc

    # Phase 10: One last soft-max + epigraph combo
    if tl() > 30:
        h_sol, score = solve_soft_max(best_h, [100, 300], maxiter_per_beta=60)
        if score < best_score - 1e-13:
            best_h = h_sol
            best_score = score
        h_sol2, score2 = focused_epigraph(best_h, top_k=12, maxiter=80)
        if h_sol2 is not None and score2 < best_score - 1e-13:
            best_h = h_sol2
            best_score = score2
        h_ref, sc = active_descent(best_h, max_steps=30, sigma=0.002)
        if sc < best_score - 1e-13:
            best_h = h_ref
            best_score = sc

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)

    return best_h, best_score, len(best_h)
