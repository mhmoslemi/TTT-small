def active_set_lp_polish(h, max_iter=50, n_active=5, max_step=0.5):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_c = exact_objective(h_cur)
    
    for it in range(max_iter):
        profile = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        M = np.max(profile)
        # Find active lags (within tolerance of max)
        tol = 1e-6 * M + 1e-10
        active = np.where(profile >= M - tol)[0]
        if len(active) > n_active:
            active = active[np.argsort(-profile[active])[:n_active]]
        
        # Compute gradients for active lags
        # grad_k[j] = d/dh_j of p_k
        grads = []
        for idx in active:
            k = idx - (n - 1)  # actual lag
            g = np.zeros(n)
            if k == 0:
                g[:] = 1.0 - 2.0 * h_cur
            elif k > 0:
                # p_k = dx * sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
                for pos in range(n - k):
                    g[pos + k] += (1.0 - h_cur[pos])  # derivative of h[pos+k]*(1-h[pos]) w.r.t. h[pos+k]
                    g[pos] -= h_cur[pos + k]  # derivative w.r.t. h[pos]
            else:
                m = -k
                for pos in range(m, n):
                    g[pos - m] += (1.0 - h_cur[pos])  # derivative of h[pos-m]*(1-h[pos]) w.r.t. h[pos-m]
                    g[pos] -= h_cur[pos - m]  # derivative w.r.t. h[pos]
            grads.append(g * dx)
        
        # LP: minimize t subject to:
        #   sum_k in active: grad_k · d <= t - profile[k] (linearized decrease)
        #   sum(d) = 0
        #   -1 <= d[j] <= 1 for all j
        # Actually, let's minimize: max_k (profile[k] + grad_k · d)
        # Subject to: sum(d) = 0, -bound <= d[j] <= bound
        # 
        # This is: minimize t
        #   s.t. grad_k · d - t <= -profile[k] for all active k
        #        sum(d) = 0
        #        -bound <= d[j] <= bound
        #
        # Variables: d[0..n-1], t (total n+1 variables)
        # Inequality constraints: A_ineq * x <= b_ineq
        # Equality: A_eq * x = b_eq
        
        from scipy.optimize import linprog
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize t
        
        # Inequality: for each active k: grad_k · d - t <= -profile[idx]
        # i.e., grad_k[0], ..., grad_k[n-1], -1 (for t) as row
        A_ineq = []
        b_ineq = []
        for gi, idx in zip(grads, active):
            row = np.append(gi, -1.0)
            A_ineq.append(row)
            b_ineq.append(-profile[idx])
        A_ineq = np.array(A_ineq)
        b_ineq = np.array(b_ineq)
        
        # Equality: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds: -bound <= d[j] <= bound, t unbounded (or bounded)
        bound = max_step
        bounds = [(-bound, bound)] * n + [(None, None)]
        
        res = linprog(c_vec, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        d = res.x[:n]
        
        # Line search
        # Find max alpha such that 0 <= h_cur + alpha*d
        alpha_max = max_step
        for j in range(n):
            if d[j] > 1e-12:
                alpha_max = min(alpha_max, (1.0 - h_cur[j]) / d[j])
            elif d[j] < -1e-12:
                alpha_max = min(alpha_max, -h_cur[j] / d[j])
        alpha_max = min(alpha_max, max_step)
        
        # Binary search for alpha that decreases the objective
        best_alpha = 0.0
        best_obj = cur_c
        for trial_alpha in np.linspace(0, alpha_max, 50):
            h_trial = h_cur + trial_alpha * d
            h_trial = project_box_sum(h_trial)
            obj = exact_objective(h_trial)
            if obj < best_obj:
                best_obj = obj
                best_alpha = trial_alpha
        
        if best_alpha > 0:
            h_cur = project_box_sum(h_cur + best_alpha * d)
            new_c = exact_objective(h_cur)
            if new_c < cur_c - 1e-12:
                cur_c = new_c
            else:
                break
        else:
            break
    
    return h_cur, cur_c

def exact_objective(h):

    profile = np.correlate(h, 1.0 - h, mode='full') * dx

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(1, n):
        # tm[i] = popcount(i) % 2
        x = i
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        tm[i] = cnt % 2
    return tm

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            cnt = np.sum(mask)
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            if not np.any(mask):
                break
            cnt = np.sum(mask)
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        mask = (h > 0) if diff > 0 else (h < 1.0)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth(h0, tau, maxiter=100):
    n = len(h0)
    N = n
    tau_val = tau
    
    def obj_grad(x):
        dx = 2.0 / N
        r = np.correlate(x, 1.0 - x, mode='full')
        r_max = np.max(r)
        shifted = np.clip((r - r_max) / tau_val, -500, 500)
        exp_s = np.exp(shifted)
        log_sum = np.log(np.sum(exp_s))
        value = dx * (r_max + tau_val * log_sum)
        weights = exp_s / np.sum(exp_s)
        grad = np.zeros(N)
        for i in range(2 * N - 1):
            k = i - (N - 1)
            w = weights[i]
            if w < 1e-14:
                continue
            if k == 0:
                grad += w * (1.0 - 2.0 * x)
            elif k > 0:
                grad[k:N] += w * (1.0 - x[0:N-k])
                grad[0:N-k] -= w * x[k:N]
            else:
                a = -k
                grad[0:N-a] += w * (1.0 - x[a:N])
                grad[a:N] -= w * x[0:N-a]
        grad *= dx
        sv = np.sum(x) - N / 2.0
        value += 1e4 * sv * sv
        grad += 2e4 * sv
        return value, grad
    
    def eq_fun(x):
        return np.sum(x) - N / 2.0
    
    def eq_jac(x):
        return np.ones(N)
    
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_grad, h0, method='SLSQP', jac=True, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        return project_box_sum(res.x)
    except Exception:
        return None

    n = len(h)
    if n % 2 != 0:
        return None

    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_greedy_swap(b, max_rounds=500, rng_seed=42):
    n = len(b)
    cur = b.copy()
    cur_c = exact_objective(cur)
    rg = np.random.RandomState(rng_seed)
    for _ in range(max_rounds):
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-15
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = exact_objective(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                delta = cur_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i >= 0:
            cur[best_i] = 0.0
            cur[best_j] = 1.0
            cur_c -= best_delta
        else:
            break
    return cur, cur_c

One more optimization: in the LP, I can use the fact that the gradient vectors have a special structure. For lag k > 0, the gradient has nonzeros only in positions [0, n-1]. Specifically:
- g[k:n] += (1 - h[0:n-k])
- g[0:n-k] -= h[k:n]

So g[j] is nonzero for all j in [0, n-1] (the ranges [k, n-1] and [0, n-1-k] cover all positions for k < n/2, and even for k ≥ n/2, the union covers most positions). So the sparse structure doesn't help much.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            cnt = np.sum(mask)
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            if not np.any(mask):
                break
            cnt = np.sum(mask)
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        mask = (h > 0) if diff > 0 else (h < 1.0)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile_and_grads(h, n_active=8):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(profile)
    tol = 1e-8 * max(M, 1.0) + 1e-12
    active = np.where(profile >= M - tol)[0]
    if len(active) > n_active:
        active = active[np.argsort(-profile[active])[:n_active]]
    grads = []
    for idx in active:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif k > 0:
            g[k:n] += (1.0 - h[0:n - k])
            g[0:n - k] -= h[k:n]
        else:
            m = -k
            g[0:n - m] += (1.0 - h[m:n])
            g[m:n] -= h[0:n - m]
        grads.append(g * dx)
    return profile, active, grads, M

def active_set_lp_polish(h, max_iter=50, n_active=8, max_step=0.25):
    n = len(h)
    h_cur = h.copy()
    cur_c = exact_objective(h_cur)
    stuck = 0
    for it in range(max_iter):
        profile, active, grads, M = compute_profile_and_grads(h_cur, n_active)
        if len(active) == 0:
            break
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.array([np.append(g, -1.0) for g in grads])
        b_ub = np.array([-profile[idx] for idx in active])
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bnd = max_step
        bounds = [(-bnd, bnd)] * n + [(None, None)]
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
        except Exception:
            break
        if res.status != 0:
            break
        d = res.x[:n]
        if np.max(np.abs(d)) < 1e-10:
            stuck += 1
            if stuck > 3:
                break
            rng = np.random.RandomState(it + 7)
            d = rng.randn(n) * 0.05
            d -= d.mean()
        else:
            stuck = 0
        alpha_max = 1.0
        pos = d > 1e-12
        neg = d < -1e-12
        if np.any(pos):
            alpha_max = min(alpha_max, np.min((1.0 - h_cur[pos]) / d[pos]))
        if np.any(neg):
            alpha_max = min(alpha_max, np.min(-h_cur[neg] / d[neg]))
        alpha_max = min(max(alpha_max, 1e-6), 1.0)
        best_alpha = 0.0
        best_obj = cur_c

            obj = exact_objective(h_trial)

                best_alpha = alpha
        if best_alpha > 0 and best_obj < cur_c - 1e-13:
            h_cur = project_box_sum(h_cur + best_alpha * d)
            cur_c = exact_objective(h_cur)
        else:
            break
    return h_cur, cur_c

def slsqp_smooth(h0, tau, maxiter=80):
    n = len(h0)
    N = n
    tau_val = tau

    def obj_grad(x):
        dx = 2.0 / N
        r = np.correlate(x, 1.0 - x, mode='full')
        r_max = np.max(r)
        shifted = np.clip((r - r_max) / tau_val, -500, 500)
        exp_s = np.exp(shifted)
        log_sum = np.log(np.sum(exp_s))
        value = dx * (r_max + tau_val * log_sum)
        weights = exp_s / np.sum(exp_s)
        grad = np.zeros(N)
        for i in range(2 * N - 1):
            k = i - (N - 1)
            w = weights[i]
            if w < 1e-14:
                continue
            if k == 0:
                grad += w * (1.0 - 2.0 * x)
            elif k > 0:
                grad[k:N] += w * (1.0 - x[0:N - k])
                grad[0:N - k] -= w * x[k:N]
            else:
                a = -k
                grad[0:N - a] += w * (1.0 - x[a:N])
                grad[a:N] -= w * x[0:N - a]
        grad *= dx
        sv = np.sum(x) - N / 2.0
        value += 1e4 * sv * sv
        grad += 2e4 * sv
        return value, grad

    def eq_fun(x):
        return np.sum(x) - N / 2.0

    def eq_jac(x):
        return np.ones(N)

    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj_grad, h0, method='SLSQP', jac=True, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        return project_box_sum(res.x)
    except Exception:
        return None

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_greedy_swap(b, max_rounds=400, rng_seed=42):
    n = len(b)
    cur = b.copy()
    cur_c = exact_objective(cur)
    for _ in range(max_rounds):
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-15
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = exact_objective(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                delta = cur_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i >= 0:
            cur[best_i] = 0.0
            cur[best_j] = 1.0
            cur_c -= best_delta
        else:
            break
    return cur, cur_c

def thue_morse_seq(n):
    idx = np.arange(n)
    result = np.zeros(n)
    for b in range(8):
        result ^= ((idx >> b) & 1)
    return result.astype(np.float64)

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

def optimize_candidate(h_start, deadline, tag=""):

    c_cur = exact_objective(h_cur)
    # SLSQP phases
    tau_stages = [0.02, 0.005, 0.001, 0.0003, 0.0001]

    # Binary

            if b_c < c_cur:
