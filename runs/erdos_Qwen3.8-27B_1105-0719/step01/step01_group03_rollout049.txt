def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Fine correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            candidates = np.where(h < 1.0 - 1e-15)[0]
            if len(candidates) > 0:
                h[candidates] += diff2 / len(candidates)
        else:
            candidates = np.where(h > 1e-15)[0]
            if len(candidates) > 0:
                h[candidates] += diff2 / len(candidates)
    return np.clip(h, 0.0, 1.0)

    """Gradient of profile[m] w.r.t. h. m is 0..2n-2."""

    # Numerically stable log-sum-exp
    pm = p - p.max()
    return (1.0/beta) * (p.max() + np.log(np.sum(np.exp(beta * pm))))

    pm = p - p.max()
    exp_b = np.exp(beta * pm)
    w = exp_b / exp_b.sum()

    v = np.zeros(n)

        beta = beta_start + (beta_end - beta_start) * t_frac * t_frac  # quadratic ramp

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:

            remaining -= rem

            cands = np.where(h > 1e-15)[0]

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            cands = np.where(h < 1.0 - 1e-15)[0]
        else:
            cands = np.where(h > 1e-15)[0]
        if len(cands) > 0:
            h[cands] += diff2 / len(cands)
    return np.clip(h, 0.0, 1.0)


def profile_grad_single(h, m):
    n = len(h)
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[:n - d] += -dx * h[d:n]
        g[d:] += dx * (1.0 - h[:n - d])
    else:
        dp = -d
        g[:n - dp] += dx * (1.0 - h[dp:n])
        g[dp:] += -dx * h[:n - dp]
    return g


def smoothed_minimax(h0, beta_starts, n_iter, mu=0.87, alpha_init=0.004):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = exact_score(h)
    M = 2 * n - 1

    for it in range(n_iter):
        t_frac = (it + 1) / n_iter
        beta = beta_starts[0] + (beta_starts[1] - beta_starts[0]) * t_frac ** 1.5

        p = profile(h)
        pm = p - p.max()
        exp_b = np.exp(beta * pm)
        w = exp_b / exp_b.sum()

        g = np.zeros(n)
        for m in range(M):
            if w[m] > 1e-12:
                g += w[m] * profile_grad_single(h, m)
        g -= g.mean()

        alpha = alpha_init / (1.0 + 0.003 * it)
        v = mu * v - alpha * g
        h_new = repair_feasible(h + v)

        val = exact_score(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new

    return best_h, best_val


def active_set_descent(h0, n_iter=300, top_k=3):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = exact_score(h)

    for it in range(n_iter):
        p = profile(h)
        c = p.max()

        k = min(top_k, len(p))
        top_idx = np.argsort(p)[-k:]
        w = np.zeros(len(p))
        w_vals = np.maximum(p[top_idx] - p[top_idx].min() + 1e-10, 1e-10)
        for i, m in enumerate(top_idx):
            w[m] = w_vals[i]
        w /= w.sum()

        g = np.zeros(n)
        for m in top_idx:
            if w[m] > 1e-10:
                g += w[m] * profile_grad_single(h, m)
        g -= g.mean()

        norm_g = np.linalg.norm(g)
        if norm_g < 1e-14:
            break
        g /= norm_g

        moved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_try = repair_feasible(h - alpha * g)
            s = exact_score(h_try)
            if s < c - 1e-13:
                h = h_try
                if s < best_val:
                    best_val = s
                    best_h = h.copy()
                moved = True
                break
        if not moved:
            break

    return best_h, best_val


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def solve_slqp(h0, maxiter=200, tol=1e-10):
    n = len(h0)
    c0 = exact_score(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        p = np.correlate(h, 1 - h, mode="full") * dx
        return t - p

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -profile_grad_single(h, m)
            J[m, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            return h_new
    except Exception:
        pass
    return None


def random_binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    # Slight smoothing to make it continuous
    kernel = np.array([0.1, 0.3, 0.9, 0.3, 0.1])
    h = np.convolve(h, kernel / kernel.sum(), mode='same')
    return repair_feasible(h)


def cosine_start(n, f, amp, rng):

    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + amp * np.cos(2 * np.pi * f * x + phase)
    return repair_feasible(h)
