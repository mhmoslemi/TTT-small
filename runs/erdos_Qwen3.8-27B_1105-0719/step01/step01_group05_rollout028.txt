        return np.ones(n)

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()

    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = int(np.sum(mask))
            if count == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / count, 1.0)
        else:
            mask = h > 0.0
            count = int(np.sum(mask))
            if count == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / count, 0.0)

        else:

                    elif diff < 0 and h[j] > 0.0:

                        h[j] += adj
                        diff += adj

def compute_gradient_lag(h, lag, dx):
    """Gradient of P_lag w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[lag:] += (1.0 - h[:s]) * dx
            grad[:s] += -h[lag:n] * dx
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[a:] += (1.0 - h[:s]) * dx
            grad[:s] += -h[a:n] * dx
    return grad

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = np.max(profile)
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    value = pmax + tau * np.log(wsum)
    wnorm = weights / wsum
    
    grad = np.zeros(n)
    active = np.where(wnorm > 1e-8)[0]
    for m in active:
        lag = m - (n - 1)
        grad += wnorm[m] * compute_gradient_lag(x, lag, dx)
    return value, grad

def nesterov_pgd(h_init, deadline, target, tau_start=0.02, tau_end=0.00005, max_iters=5000):
    """Nesterov accelerated gradient with projection."""
    n = len(h_init)

        # Decay momentum slightly
        mom = momentum * (1.0 - 0.1 * frac)

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n

        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = int(np.sum(mask))
            if count == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / count, 1.0)
        else:
            mask = h > 0.0
            count = int(np.sum(mask))
            if count == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / count, 0.0)

def grad_lag(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[lag:] += (1.0 - h[:s]) * dx
            grad[:s] += -h[lag:n] * dx
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[a:] += (1.0 - h[:s]) * dx
            grad[:s] += -h[a:n] * dx
    return grad

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = np.max(profile)
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    value = pmax + tau * np.log(wsum)
    wnorm = weights / wsum
    grad = np.zeros(n)
    active = np.where(wnorm > 1e-8)[0]
    for m in active:
        lag = m - (n - 1)
        grad += wnorm[m] * grad_lag(x, lag, dx)
    return value, grad

        return np.sum(x) - target

        return np.ones(n)
