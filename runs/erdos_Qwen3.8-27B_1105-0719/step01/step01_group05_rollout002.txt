def penalized_fun(x, context):
    """x = [h_0, ..., h_{n-1}, c]
    context = [n, target, active_lags_array, c_target, penalty_weight]
    """
    n = context[0]
    target = context[1]
    active_lags = context[2]
    c_target = context[3]
    penalty = context[4]
    dx = 2.0 / n
    
    h = x[:n]
    c = x[-1]
    
    # Objective: c + penalty * sum_{active} (max(0, q_r - c_target))^2
    obj = c
    grad = np.zeros(x.shape)
    grad[-1] = 1.0
    
    for lag in active_lags:
        # Compute q at this lag
        if lag >= 0:
            s = n - lag
            if s <= 0:
                continue
            h_i = h[:s]
            h_j = h[lag:]
            q = np.dot(h_i, 1.0 - h_j) * dx
        else:
            a = -lag
            s = n - a
            if s <= 0:
                continue
            h_i = h[a:]
            h_j = h[:s]
            q = np.dot(h_i, 1.0 - h_j) * dx
        
        excess = max(0.0, q - c_target)
        if excess > 0:
            obj += penalty * excess * excess
            # Gradient of penalty term

            # d/dh_j: penalty * 2*excess * dx * (-h_i)
            if lag >= 0:
                grad[:s] += 2 * penalty * excess * (1.0 - h_j) * dx
                grad[lag:] -= 2 * penalty * excess * h_i * dx
