def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)

    score_parent, _ = exact_score(h_parent)
    best_h = h_parent.copy()
    best_c5 = score_parent

    # Phase 1: Smooth warm-up in q-space
    n = n_parent
    m = n // 2
    q_best = q_from_h(h_parent, n)

    for beta in [80, 200, 500, 1000]:
        if remaining() < 40:
            break
        h_q = build_h_from_q(q_best, n)
        q0 = q_best.copy()

        def q_val(x):
            return smooth_q_value_grad(x, n, beta)[0]

        def q_grad(x):
            return smooth_q_value_grad(x, n, beta)[1]

        try:
            res = minimize(q_val, q0, jac=q_grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * m,
                           options={'maxiter': 500, 'ftol': 1e-15})
            h_res = build_h_from_q(np.clip(res.x, 0, 1), n)
            s_res, _ = exact_score(h_res)
            if s_res < best_c5 - 1e-14:
                best_c5 = s_res
                best_h = h_res
                q_best = np.clip(res.x, 0, 1).copy()
        except Exception:
            pass

    # Phase 2: Active-set epigraph in q-space
    if remaining() > 80:
        h_q = build_h_from_q(q_best, n)
        corrs_q = compute_corrs(h_q)
        t0_init = corrs_q.max() * (2.0 / n) * (1.0 + 1e-3) + 1e-5
        q_es, c5_es = active_set_search(q_best, n, remaining(), t0_init)
        h_es = build_h_from_q(q_es, n)
        h_es = project_feasible(h_es, target_sum)
        s_es, _ = exact_score(h_es)
        if s_es < best_c5 - 1e-14:
            best_c5 = s_es
            best_h = h_es
            q_best = q_es.copy()

    # Phase 3: Full-space epigraph from best
    if remaining() > 60:
        h_full, c5_full = full_active_search(best_h, n, remaining())
        s_full, _ = exact_score(h_full)
        if s_full < best_c5 - 1e-14:
            best_c5 = s_full
            best_h = h_full

    # Phase 4: Smooth polish in full space
    if remaining() > 40:
        for beta in [500, 1500]:
            if remaining() < 20:
                break
            h_opt = smooth_full_solve(best_h, n, beta, target_sum, maxiter=400)
            s_opt, _ = exact_score(h_opt)
            if s_opt < best_c5 - 1e-14:
                best_c5 = s_opt
                best_h = h_opt

    # Phase 5: Try epigraph on full space again with fresh active set
    if remaining() > 50:
        h_full2, c5_full2 = full_active_search(best_h, n, remaining())
        s_full2, _ = exact_score(h_full2)
        if s_full2 < best_c5 - 1e-14:
            best_c5 = s_full2
            best_h = h_full2

    # Phase 6: Try smaller n values
    for n_try in [62, 64, 66, 68, 70]:
        if remaining() < 60:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        # Interpolate best_h to new grid
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        # Run epigraph on this smaller grid
        h_small, c5_small = full_active_search(h_interp, n_try, remaining())
        s_small, _ = exact_score(h_small)
        if s_small < best_c5 - 1e-10:
            best_c5 = s_small
            best_h = h_small

        # Also try q-space on smaller grid (if even)
        if n_try % 2 == 0 and remaining() > 40:
            q_small = q_from_h(h_small, n_try)
            corrs_s = compute_corrs(h_small)
            t0_s = corrs_s.max() * dx_try * (1.0 + 1e-3) + 1e-5
            q_es_s, c5_es_s = active_set_search(q_small, n_try, remaining(), t0_s)
            h_es_s = build_h_from_q(q_es_s, n_try)
            h_es_s = project_feasible(h_es_s, ts_try)
            s_es_s, _ = exact_score(h_es_s)
            if s_es_s < best_c5 - 1e-10:
                best_c5 = s_es_s
                best_h = h_es_s

    # Phase 7: Final polish - one more round of q-space epigraph + full epigraph
    n_final = len(best_h)
    ts_final = n_final / 2.0

    if remaining() > 30 and n_final % 2 == 0:
        q_final = q_from_h(best_h, n_final)
        h_qf = build_h_from_q(q_final, n_final)
        corrs_f = compute_corrs(h_qf)
        t0_f = corrs_f.max() * (2.0 / n_final) * (1.0 + 1e-3) + 1e-5
        q_res, c5_res = active_set_search(q_final, n_final, remaining(), t0_f)
        h_res = build_h_from_q(q_res, n_final)
        h_res = project_feasible(h_res, ts_final)
        s_res, _ = exact_score(h_res)
        if s_res < best_c5 - 1e-14:
            best_c5 = s_res
            best_h = h_res

    if remaining() > 20:
        h_fp, c5_fp = full_active_search(best_h, n_final, remaining())
        s_fp, _ = exact_score(h_fp)
        if s_fp < best_c5 - 1e-14:
            best_c5 = s_fp
            best_h = h_fp

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
