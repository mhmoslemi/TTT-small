def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        elif delta > 0 and delta < n:
            L = n - delta
            G[k, delta:n] += dx * (1.0 - h[0:L])
            G[k, 0:L] -= dx * h[delta:n]
        elif delta < 0 and (-delta) < n:
            d = -delta
            L = n - d
            G[k, 0:L] += dx * (1.0 - h[d:n])
            G[k, d:n] -= dx * h[0:L]
    return G

def solve_lp(h, radius, active_set):
    n = len(h)
    p = get_profile(h)
    G = get_lag_gradients(h)
    idx = active_set
    G_sub = G[idx, :]
    b = p[idx] - G_sub @ h
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lb = np.maximum(0.0, h - radius)
    ub = np.minimum(1.0, h + radius)
    bounds = list(zip(lb, ub)) + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return repair_feasible(res.x[:n])
    except Exception:
        pass
    return None

    stall = 0
    use_full = (2 * n - 1) <= 300
    multipliers = [1.0, 0.75, 0.55, 0.4, 0.28, 0.18, 0.1, 0.05, 0.02]
    while time.perf_counter() - t0 < max_time:
        p = get_profile(h)
        score = float(np.max(p))
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
        M = 2 * n - 1
        if use_full:
            h_cand = solve_lp_full(h, radius)
        else:
            active = np.argsort(p)[::-1][:min(M, 120)]
            h_cand = solve_lp_subset(h, radius, active)
        if h_cand is None:
            radius *= 0.5
            if radius < min_radius:
                break
            continue
        step = h_cand - h
        improved = False
        for mult in multipliers:
            h_trial = repair_feasible(h + mult * step)
            s = exact_score(h_trial)
            if s < best_score - 1e-13:
                h = h_trial
                best_h = h.copy()
                best_score = s
                radius = min(0.30, radius * 1.15)
                improved = True
                break
        if not improved:
            radius *= 0.65
            if radius < min_radius:
                break
    return best_h, best_score

def smooth_grad_descent(h_start, max_time, t0):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    s_param = 80.0
    while time.perf_counter() - t0 < max_time:
        prof = get_profile(h)
        G = get_lag_gradients(h)
        m = np.max(prof)
        w = np.exp(s_param * (prof - m))
        w /= w.sum()
        grad = G.T @ w
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        improved = False
        for a in [0.02, 0.01, 0.006, 0.003, 0.0015, 0.0008]:
            h_trial = repair_feasible(h - a * grad)
            s = exact_score(h_trial)
            if s < best_score - 1e-13:
                h = h_trial
                best_h = h.copy()
                best_score = s
                improved = True
                break
        if not improved:
            s_param *= 1.5
            if s_param > 500.0:
                break
    return best_h, best_score

def interleaved_descent(h_start, max_time, t0):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    phase = 0
    lp_t = max_time * 0.5
    sg_t = max_time * 0.5
    lp_start = time.perf_counter()
    sg_start = time.perf_counter()
    while time.perf_counter() - t0 < max_time:
        if phase == 0:
            if time.perf_counter() - lp_start > lp_t:
                h2, s2 = smooth_grad_descent(h, max_time, sg_start)
                if s2 < best_score - 1e-13:
                    best_score = s2
                    best_h = h2
                if exact_score(h) > best_score + 1e-12:
                    h = best_h.copy()
                phase = 1
                lp_start = time.perf_counter()
                continue
            h2, s2 = lp_descent(h, 3.0, time.perf_counter())
            if s2 < best_score - 1e-13:
                best_score = s2
                best_h = h2
            h = h2
        else:
            if time.perf_counter() - sg_start > sg_t:
                break
            h2, s2 = smooth_grad_descent(h, 3.0, time.perf_counter())
            if s2 < best_score - 1e-13:
                best_score = s2
                best_h = h2
            if exact_score(h2) > best_score + 1e-12:
                h = best_h.copy()
            else:
                h = h2
            phase = 0
            sg_start = time.perf_counter()
        time.sleep(0.01)
    return best_h, best_score

def anti_symmetrize(h):
    n = len(h)
    h_sym = 0.5 * (h + 1.0 - h[::-1].copy())
    return repair_feasible(h_sym)

def reverse_symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)

def full_symmetrize(h):
    h1 = anti_symmetrize(h)
    h2 = reverse_symmetrize(h)
    h3 = 0.5 * (h1 + h2)
    return repair_feasible(h3)

def interp_dim(h, new_len):
    n = len(h)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    h_new = np.interp(x_tgt, x_src, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_polish(h_start, K, maxiter=40):
    n = len(h_start)
    M = 2 * n - 1
    p = get_profile(h_start)
    active = np.argsort(p)[::-1][:min(K, M)]
    n_act = len(active)
    best_s = exact_score(h_start)

    def obj_func(x):
        return x[n]

    def obj_jac_func(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def cons_func(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        px = get_profile(h)
        ineq = t - px[active]
        return np.concatenate([[eq], ineq])

    def cons_jac_func(x):
        h = x[:n]
        G_full = get_lag_gradients(h)
        Ja = G_full[active, :]
        J = np.zeros((1 + n_act, n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, best_s + 0.5)]
    x0 = np.concatenate([h_start, [best_s + 1e-5]])
    try:
        result = minimize(obj_func, x0, jac=obj_jac_func, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'ineq', 'fun': cons_func, 'jac': cons_jac_func}],
                          options={'maxiter': maxiter, 'ftol': 1e-13})
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            return h_sol
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t0)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
        parent_n = len(parent)
    except Exception:
        parent = None
        parent_n = 140

    if parent is not None:
        best_h = repair_feasible(parent)
    else:
        best_h = np.full(140, 0.5)
        best_h[:70] = 1.0
        best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)

    # Verify gradients
    if tleft() > 5:
        if not verify_gradients(best_h):
            pass  # Continue anyway but be more careful

    # Symmetrizations
    for sym_func in [anti_symmetrize, reverse_symmetrize, full_symmetrize]:
        h_s = sym_func(best_h)
        s_s = exact_score(h_s)
        if s_s < best_score - 1e-13:
            best_h = h_s
            best_score = s_s

    # Phase 1: Interleaved LP + smooth gradient descent on parent dim
    if tleft() > 60:
        tl = min(200, tleft() * 0.40)
        h_r, s_r = interleaved_descent(best_h, tl, t0)
        if s_r < best_score - 1e-13:
            best_h = h_r
            best_score = s_r

    # Re-symmetrize after optimization
    for sym_func in [anti_symmetrize, full_symmetrize]:
        h_s = sym_func(best_h)
        s_s = exact_score(h_s)
        if s_s < best_score - 1e-13:
            best_h = h_s
            best_score = s_s

    # Phase 2: SLSQP with all constraints (if feasible size-wise)
    if tleft() > 30:
        n = len(best_h)
        M = 2 * n - 1
        K = min(M, 100) if M > 100 else M
        h_sol = slsqp_polish(best_h, K, maxiter=40)
        if h_sol is not None:
            s_sol = exact_score(h_sol)
            if s_sol < best_score - 1e-13:
                best_h = h_sol
                best_score = s_sol

    # Phase 3: Try different dimensions
    if tleft() > 50:
        base_n = len(best_h)
        scales = [0.45, 0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85, 0.9, 1.1, 1.2, 1.35, 1.5, 2.0]
        dims = []
        for sc in scales:
            d = int(round(base_n * sc))
            d = max(32, min(280, d))
            if d % 2 == 1:
                d += 1
            if d != base_n and d not in dims:
                dims.append(d)
        dims.sort()
        for dn in dims:
            if tleft() < 30:
                break
            h_i = interp_dim(best_h, dn)
            h_i = anti_symmetrize(h_i)
            tl = min(10, tleft() * 0.06)
            h_r, s_r = interleaved_descent(h_i, tl, time.perf_counter())
            if s_r < best_score - 1e-13:
                best_h = h_r
                best_score = s_r

    # Phase 4: Go back to best dimension and polish
    if tleft() > 25:
        n = len(best_h)
        tl = min(40, tleft() * 0.30)
        h_r, s_r = lp_descent(best_h, tl, t0)
        if s_r < best_score - 1e-13:
            best_h = h_r
            best_score = s_r

    # Phase 5: Multi-start perturbations
    if tleft() > 35:
        n = len(best_h)
        for p in range(8):
            if tleft() < 20:
                break
            eps = 0.0008 + 0.0015 * p
            z = np.random.randn(n)
            z -= z.mean()
            h_p = repair_feasible(best_h + eps * z)
            h_p = anti_symmetrize(h_p)
            tl = min(7, tleft() * 0.05)
            h_r, s_r = interleaved_descent(h_p, tl, time.perf_counter())
            if s_r < best_score - 1e-13:
                best_h = h_r
                best_score = s_r

    # Phase 6: Final SLSQP polish
    if tleft() > 15:
        n = len(best_h)
        M = 2 * n - 1
        K = min(M, 60)
        h_sol = slsqp_polish(best_h, K, maxiter=30)
        if h_sol is not None:
            s_sol = exact_score(h_sol)
            if s_sol < best_score - 1e-13:
                best_h = h_sol
                best_score = s_sol

    # Phase 7: Final LP micro-refinement
    if tleft() > 12:
        tl = min(10, tleft() * 0.4)
        h_r, s_r = lp_descent(best_h, tl, t0)
        if s_r < best_score - 1e-13:
            best_h = h_r
            best_score = s_r

    # Final: anti-symmetrize and check
    h_final_sym = anti_symmetrize(best_h)
    s_final_sym = exact_score(h_final_sym)
    if s_final_sym < best_score - 1e-13:
        best_h = h_final_sym
        best_score = s_final_sym

    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
