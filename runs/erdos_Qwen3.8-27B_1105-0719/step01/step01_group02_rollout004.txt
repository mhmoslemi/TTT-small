import numpy as np
from scipy.optimize import linprog
import time

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_value(h):
    return float(np.max(compute_profile(h)))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    return h

def profile_gradients(h):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= j + k < n:
                val += 1.0 - h[j + k]
            if 0 <= j - k < n:
                val -= h[j - k]
            g[j] = dx * val
        grads[m] = g
    return grads

def profile_gradients_fast(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((len(active_indices), n))
    for idx, m in enumerate(active_indices):
        k = m - (n - 1)
        g = np.zeros(n)
        j = np.arange(n)
        valid_plus = (j + k >= 0) & (j + k < n)
        valid_minus = (j - k >= 0) & (j - k < n)
        g[valid_plus] += 1.0 - h[j[valid_plus] + k]
        if np.any(valid_minus):
            g[valid_minus] -= h[j[valid_minus] - k]
        g *= dx
        grads[idx] = g
    return grads

def active_set_lp(h, tol=0.008, max_active=25, min_delta=0.0005):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    max_p = np.max(profile)
    active_mask = profile >= max_p - tol
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        return None
    if len(active_indices) > max_active:
        order = np.argsort(-profile[active_indices])
        active_indices = active_indices[order[:max_active]]
    A = profile_gradients_fast(h, active_indices)
    n_active = A.shape[0]
    n_vars = 2 * n + 1
    c = np.zeros(n_vars)
    c[-1] = -1.0
    A_ub = np.zeros((n_active, n_vars))
    for i in range(n_active):
        A_ub[i, :n] = A[i]
        A_ub[i, n:2*n] = -A[i]
        A_ub[i, 2*n] = 1.0
    b_ub = np.zeros(n_active)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = [0.0]
    U = np.array(1.0 - h)
    L = np.array(h)
    bounds = []
    for j in range(n):
        bounds.append((0.0, U[j] if U[j] > 0 else 0.0))
    for j in range(n):
        bounds.append((0.0, L[j] if L[j] > 0 else 0.0))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status != 0:
            return None
        delta = res.x[2 * n]
        if delta < min_delta:
            return None
        d = res.x[:n] - res.x[n:2*n]
        max_abs = np.max(np.abs(d))
        if max_abs < 1e-14:
            return None
        d = d / max_abs
        return d
    except Exception:
        return None

def line_search(h, d, cur_c, max_iter=50):
    n = len(h)
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    alpha_max = 1.0
    if np.any(pos_mask):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]) * 0.99)
    if np.any(neg_mask):
        alpha_max = min(alpha_max, np.min(-h[neg_mask] / d[neg_mask]) * 0.99)
    if alpha_max <= 0:
        return h, cur_c
    best_alpha = 0.0
    best_c = cur_c
    lo, hi = 0.0, alpha_max
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        h_try = h + mid * d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_box_sum(h_try, np.sum(h))
        c_try = c5_value(h_try)
        if c_try < best_c:
            best_c = c_try
            best_alpha = mid
        if c_try < cur_c:
            lo = mid
        else:
            hi = mid
        if hi - lo < alpha_max * 1e-6:
            break
    if best_alpha > 0:
        h_new = h + best_alpha * d
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_box_sum(h_new, np.sum(h))
        return h_new, c5_value(h_new)
    return h, cur_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=500, rng_seed=42):
    n = len(b)
    cur = b.copy()
    cur_c = c5_value(cur)
    rg = np.random.RandomState(rng_seed)
    for round_idx in range(max_rounds):
        best_delta = 0.0
        best_swap = None
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Random subset for speed
        if len(ones) * len(zeros) > 20000:
            ones_sample = rg.choice(ones, min(len(ones), 200), replace=False)
            zeros_sample = rg.choice(zeros, min(len(zeros), 200), replace=False)
        else:
            ones_sample = ones
            zeros_sample = zeros
        for i in ones_sample:
            for j in zeros_sample:
                cur[i] = 0.0
                cur[j] = 1.0
                c = c5_value(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                delta = cur_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (i, j)
        if best_swap is not None:
            cur[best_swap[0]] = 0.0
            cur[best_swap[1]] = 1.0
            cur_c -= best_delta
        else:
            break
    return cur, cur_c

def subgradient_step(h, tol=0.005, alpha=0.02):
    n = len(h)
    profile = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    max_p = np.max(profile)
    active_mask = profile >= max_p - tol / dx
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        return h
    A = profile_gradients_fast(h, active_indices)
    g = np.mean(A, axis=0)
    g_sum = np.sum(g)
    g -= g_sum / n
    h_new = h - alpha * g
    h_new = project_box_sum(h_new, np.sum(h))
    return h_new

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

def polish_continuous(h, deadline, min_remaining=30):
    n = len(h)
    cur_h = h.copy()
    cur_c = c5_value(cur_h)
    stag = 0
    while time.time() < deadline - min_remaining:
        d = active_set_lp(cur_h, tol=0.01, max_active=20, min_delta=0.0003)
        if d is not None:
            h_new, c_new = line_search(cur_h, d, cur_c)
            if c_new < cur_c - 1e-12:
                cur_h = h_new
                cur_c = c_new
                stag = 0
            else:
                stag += 1
                if stag > 5:
                    cur_h = subgradient_step(cur_h, tol=0.01, alpha=0.01)
                    c_new = c5_value(cur_h)
                    if c_new < cur_c - 1e-12:
                        cur_c = c_new
                        stag = 0
                    else:
                        stag += 1
        else:
            stag += 1
            if stag > 3:
                cur_h = subgradient_step(cur_h, tol=0.02, alpha=0.015)
                c_new = c5_value(cur_h)
                if c_new < cur_c - 1e-12:
                    cur_c = c_new
                    stag = 1
                else:
                    break
    return cur_h, cur_c

def polish_binary(b, deadline, min_remaining=30):
    cur = b.copy()
    cur_c = c5_value(cur)
    rounds = 0
    while time.time() < deadline - min_remaining:
        b2, c2 = binary_swap_improve(cur, max_rounds=100, rng_seed=rounds * 7 + 1)
        if c2 < cur_c - 1e-12:
            cur = b2
            cur_c = c2
            rounds += 1
        else:
            break
    return cur, cur_c
