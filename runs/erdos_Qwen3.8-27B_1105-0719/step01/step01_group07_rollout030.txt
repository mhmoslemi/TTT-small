def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    p = np.correlate(h, 1.0 - h, mode='full') * dx
    p_max = np.max(p)
    exp_vals = np.exp(tau * (p - p_max))
    w_sum = np.sum(exp_vals)
    weights = exp_vals / w_sum
    obj = p_max + np.log(w_sum) / tau
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        d = k - (n - 1)  # offset
        if d >= 0:
            # First term: m from d to n-1, contribution (1-h[m-d])
            ms = np.arange(d, n)
            grad[ms] += w * dx * (1.0 - h[ms - d])
            # Second term: m from 0 to n-1-d, contribution -h[m+d]
            ms2 = np.arange(0, n - d)
            grad[ms2] -= w * dx * h[ms2 + d]
        else:
            a = -d  # a > 0
            # First term: m from 0 to n-a-1, contribution (1-h[m+a])
            ms = np.arange(0, n - a)
            grad[ms] += w * dx * (1.0 - h[ms + a])
            # Second term: m from a to n-1, contribution -h[m-a]
            ms2 = np.arange(a, n)
            grad[ms2] -= w * dx * h[ms2 - a]
    return obj, grad
