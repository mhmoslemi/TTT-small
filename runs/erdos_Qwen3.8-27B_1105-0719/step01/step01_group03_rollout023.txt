    n = len(h)

def profile(h):
    n = len(h)
    dx = 2.0 / n

def gradient_profile(h):
    """Compute the gradient matrix G where G[m, j] = dr_m/dh_j"""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    
    # Zero lag: m = n-1
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    
    # Positive lags: d = 1, ..., n-1, corresponding to m = n-1+d
    for d in range(1, n):
        m = n - 1 + d
        # r_d = dx * sum_{i=0}^{n-d-1} h[i+d] * (1 - h[i])
        # dr_d/dh[j] for j >= d: dx * (1 - h[j-d])
        # dr_d/dh[j] for j < n-d: -dx * h[j+d]
        G[m, d:] += dx * (1.0 - h[:n-d])
        G[m, :n-d] += -dx * h[d:]
    
    # Negative lags: d = -1, ..., -(n-1), corresponding to m = n-1+d
    for s in range(1, n):
        m = n - 1 - s
        # r_{-s} = dx * sum_{i=0}^{n-s-1} h[i] * (1 - h[i+s])
        # dr_{-s}/dh[j] for j < n-s: dx * (1 - h[j+s])
        # dr_{-s}/dh[j] for j >= s: -dx * h[j-s]
        G[m, :n-s] += dx * (1.0 - h[s:])
        G[m, s:] += -dx * h[:n-s]
    
    return G

        else:
            idx = np.argmax(h)

    return h

def projected_gradient_descent(h0, max_iters=10000, tol=1e-12, restart_every=500):
    h = h0.copy()
    n = len(h)
    best = exact_score(h)
    best_h = h.copy()
    
    prev_h = None
    prev_grad = None
    no_improve_count = 0
    
    for it in range(max_iters):
        r = profile(h)
        c = r.max()
        
        if c < best - 1e-12:
            best = c
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Periodic restart
        if no_improve_count > restart_every:
            # Make a small random perturbation
            eps = 0.01 * (no_improve_count / restart_every)
            z = np.random.randn(n)
            z -= z.mean()
            h = project_feasible(h + eps * z)
            no_improve_count = 0
            prev_h = None
            prev_grad = None
            continue
        
        # Active set with adaptive threshold
        thresh = c - max(1e-5, 0.005 * c)
        active = r >= thresh
        if not active.any():
            active[r.argmax()] = True
        
        G = gradient_profile(h)
        G_act = G[active]
        w = np.maximum(r[active] - thresh, 1e-10)
        w = w / w.sum()
        grad = G_act.T @ w
        grad -= grad.mean()
        
        # BB step
        if prev_h is not None and prev_grad is not None:
            s_h = h - prev_h
            g_diff = grad - prev_grad
            ss = np.dot(s_h, s_h)
            yy = np.dot(g_diff, g_diff)
            if ss > 1e-15 and yy > 1e-15:
                alpha = ss / yy
                alpha = np.clip(alpha, 1e-6, 0.05)
            else:
                alpha = 0.005
        else:
            alpha = 0.005
        
        # Backtracking
        prev_h = h.copy()
        prev_grad = grad.copy()
        
        improved = False
        for _ in range(30):
            h_try = project_feasible(h - alpha * grad)
            c_try = exact_score(h_try)
            if c_try < best - 1e-12:
                h = h_try
                best = c_try
                best_h = h.copy()
                improved = True
                break
            alpha *= 0.5
        
        if not improved and alpha < 1e-10:
            # Stuck, do a restart
            z = np.random.randn(n)
            z -= z.mean()
            h = project_feasible(h + 0.005 * z)
            prev_h = None
            prev_grad = None

```python
def gradient_profile(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    
    # Zero lag: m=0, out_idx = n-1
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    
    # Positive lags: d = 1, ..., n-1, out_idx = n-1+d
    for d in range(1, n):
        row = n - 1 + d
        # j in [d, n-1]: first factor contribution
        G[row, d:] += dx * (1.0 - h[:n-d])
        # j in [0, n-1-d]: second factor contribution
        G[row, :n-d] += -dx * h[d:]
    
    # Negative lags: s = 1, ..., n-1, out_idx = n-1-s
    for s in range(1, n):
        row = n - 1 - s
        # j in [0, n-1-s]: first factor contribution
        G[row, :n-s] += dx * (1.0 - h[s:])
        # j in [s, n-1]: second factor contribution
        G[row, s:] += -dx * h[:n-s]
    
    return G
```
