Actually, I want to also try a slightly different approach for the 1D minimization. Instead of binary search, I can use the fact that the minimum of max of lines over [0,1] occurs at:
- x = 0
- x = 1  
- Or at an intersection of two lines that is in [0,1]

The value at the minimum is the minimum over all "critical points" of the function. A critical point is either:
- A vertex of the upper envelope in the interior of [0,1]
- An endpoint

For the upper envelope of K lines, there are at most K-1 vertices (in the worst case). Each vertex is an intersection of two lines.

I can find the minimum by:
1. Computing the upper envelope (this is O(K log K) using a convex hull trick, or O(K²) naively)
2. Evaluating at each vertex and the endpoints
3. Taking the minimum

For K = 2n-1 = 119, the naive O(K²) approach gives ~7000 intersection points to check. Each check is O(K) to evaluate the max. Total: O(K³) = ~1.7 million per coordinate. With n=60: 100 million per sweep. With 100 sweeps: 10 billion. Too slow in Python.

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    y[~np.isfinite(y)] = 0.5
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            delta = residual / len(eligible)
            h[eligible] = np.clip(h[eligible] + delta, 0.0, 1.0)
    return h


def cd_lagrangian(h0, n, lam, n_sweeps=80):
    dx = 2.0 / n
    h = h0.astype(float).copy()
    for sweep in range(n_sweeps):
        max_change = 0.0
        for j in range(n):
            old_val = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            a_k = c1 - c0
            b_k = c0
            lo_x, hi_x = 0.0, 1.0
            for _ in range(45):
                mid_x = (lo_x + hi_x) * 0.5
                vals = b_k + a_k * mid_x
                max_val = np.max(vals)
                active = (vals >= max_val - 1e-9)
                if not np.any(active):
                    break
                min_slope = np.min(a_k[active])
                max_slope = np.max(a_k[active])
                left_d = dx * min_slope + lam
                right_d = dx * max_slope + lam
                if left_d > 1e-13:
                    hi_x = mid_x
                elif right_d < -1e-13:
                    lo_x = mid_x
                else:
                    break
            x_opt = float(np.clip((lo_x + hi_x) * 0.5, 0.0, 1.0))
            if abs(x_opt - old_val) > max_change:
                max_change = abs(x_opt - old_val)
            h[j] = x_opt
        if max_change < 1e-13:
            break
    return h


def cd_with_constraint(h0, n, n_sweeps=60):
    target = n / 2.0
    best_h = project_feasible(h0, n)
    best_c5 = compute_c5(best_h)
    
    lam_lo, lam_hi = -3.0, 3.0
    for _ in range(35):
        lam = (lam_lo + lam_hi) * 0.5
        h_new = cd_lagrangian(best_h, n, lam, n_sweeps)
        s = np.sum(h_new)
        if s > target:
            lam_lo = lam
        else:
            lam_hi = lam
        h_proj = project_feasible(h_new, n)
        c5_new = compute_c5(h_proj)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_proj.copy()
    
    for lam in [0.0, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:
        h_new = cd_lagrangian(best_h, n, lam, n_sweeps)
        h_proj = project_feasible(h_new, n)
        c5_new = compute_c5(h_proj)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_proj.copy()
    
    return best_h, best_c5


def local_search_pairs(h, n, n_iters, rng, min_diff=0.01):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    improved_count = 0
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        if abs(best_h[i] - best_h[j]) < min_diff:
            continue
        h_new = best_h.copy()
        h_new[i], h_new[j] = best_h[j], best_h[i]
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            improved_count += 1
    return best_h, best_c5


def local_search_transfer(h, n, n_iters, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        cap = min(1.0 - best_h[i], best_h[j])
        if cap < 1e-6:
            continue
        tau = cap * rng.uniform(0.1, 1.0)
        h_new = best_h.copy()
        h_new[i] += tau
        h_new[j] -= tau
        if h_new[i] < -1e-12 or h_new[i] > 1.0 + 1e-12:
            continue
        if h_new[j] < -1e-12 or h_new[j] > 1.0 + 1e-12:
            continue
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5


def map_resolution(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for i in range(n_old):
        a = i * 2.0 / n_old
        b = (i + 1) * 2.0 / n_old
        for j in range(n_new):
            c = j * 2.0 / n_new
            d = (j + 1) * 2.0 / n_new
            overlap = max(0.0, min(b, d) - max(a, c))
            if overlap > 0:
                h_new[j] += h_old[i] * overlap / dx_new
    return h_new


def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and initial_h_values is not None:
