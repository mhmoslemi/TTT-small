def transfer_dimension(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    
    # Cumulative: F(x) = integral of h up to x
    # h is piecewise constant on [i*dx_src, (i+1)*dx_src]
    # F(j*dx_tgt) = sum of h_i * dx_src for i* dx < j*dx_tgt
    # More precisely: for target cell j, x goes from j*dx_tgt to (j+1)*dx_tgt
    # Result = (F((j+1)*dx_tgt) - F(j*dx_tgt)) / dx_tgt
    
    # Build F at target grid points
    # F at position x = sum_{i: (i+1)*dx_src <= x} h_i * dx_src + h_f * (x - f*dx_src) where f = floor(x/dx_src)
    
    edges = np.arange(n_tgt + 1) * dx_tgt
    F = np.zeros(n_tgt + 1)
    for j in range(n_tgt + 1):
        x = edges[j]
        # How much of each source cell is covered?
        # Source cell i covers [i*dx_src, (i+1)*dx_src]
        # Coverage of cell i by [0, x] is min(x, (i+1)*dx_src) - max(0, i*dx_src) if positive
        # = min(x - i*dx_src, dx_src) clipped to [0, dx_src]
        # Sum over i: sum_i h_i * max(0, min(x - i*dx_src, dx_src))
        # For x in [0, 2]:
        # Let p = x / dx_src (position in source grid)
        # Integer part: floor(p) cells are fully covered
        # Fractional part: frac(p) of the next cell
        p = x / dx_src
        i_full = int(np.floor(p))
        frac = p - i_full
        # Clamp
        i_full = min(i_full, n_src)
        if i_full >= n_src:
            F[j] = h_src.sum() * dx_src  # Full integral = n/2 * dx_src
        else:
            F[j] = (h_src[:i_full].sum() + frac * h_src[i_full] if i_full < n_src else h_src.sum()) * dx_src
    
    # Result: (F[j+1] - F[j]) / dx_tgt
    h_tgt = (F[1:] - F[:-1]) / dx_tgt
    return repair_feasible(h_tgt)

def sigmoid(x):

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break

            continue
        if delta >= 0:
            grad[:n - delta] += wk * (1.0 - h[delta:])
            grad[delta:] -= wk * h[:n - delta]
        else:
            m = -delta
            grad[m:] += wk * (1.0 - h[:n - m])
            grad[:n - m] -= wk * h[m:]
    return grad


def logit_project(z, n):
    target = n / 2.0

    def phi(c):
        return np.sum(sigmoid(z + c)) - target

    c_lo = float(z.min()) - 30.0
    c_hi = float(z.max()) + 30.0
    c = brentq(phi, c_lo, c_hi, xtol=1e-13, rtol=8.9e-16, maxiter=200)
    h = sigmoid(z + c)
    return h, c


def softmax_obj(z, beta, n):
    h, c = logit_project(z, n)
    C = compute_corr_values(h)
    C_max = C.max()
    exp_C = np.exp(beta * (C - C_max))
    return (C_max + np.log(exp_C.sum())) / beta


def softmax_grad(z, beta, n):
    h, c = logit_project(z, n)
    C = compute_corr_values(h)
    C_max = C.max()
    exp_C = np.exp(beta * (C - C_max))
    w = exp_C / exp_C.sum()
    G_h = compute_weighted_gradient(h, w)
    sp = h * (1.0 - h)
    T = sp.sum()
    if T < 1e-12:
        return np.zeros_like(z)
    G_z = sp * (G_h - np.dot(sp, G_h) / T)
    return G_z


def softmax_lbfgs(z0, beta, n, maxiter=2000, time_limit=60.0):
    t0 = time.perf_counter()

    def fun(z):
        return softmax_obj(z, beta, n)

    def jac(z):
        return softmax_grad(z, beta, n)

    f0 = fun(z0)
    try:
        res = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                       bounds=[(-200, 200)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxfun': 100000})
    except Exception:
        return None, None
    if time.perf_counter() - t0 > time_limit:
        return None, None
    h, c = logit_project(res.x, n)
    h = repair_feasible(h)
    score = exact_score(h)
    if np.isfinite(score) and score < f0:
        return h, score
    return h, score


def active_descent(h, max_steps=200, top_k=20, time_limit=30.0):
    n = len(h)
    t0 = time.perf_counter()
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)

    for step in range(max_steps):
        if time.perf_counter() - t0 > time_limit:
            break
        C = compute_corr_values(best)
        c_max = C.max()
        sorted_idx = np.argsort(C)[::-1]
        k = min(top_k, len(C))
        active = sorted_idx[:k]
        vals = C[active]
        sigma_w = max(0.002, 0.03 * c_max)
        weights = np.exp(-(c_max - vals) / sigma_w)
        weights /= weights.sum() + 1e-15
        w_full = np.zeros(len(C))
        w_full[active] = weights
        G_h = compute_weighted_gradient(best, w_full)
        direction = -G_h
        direction -= direction.mean()
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-14:
            break
        direction /= dir_norm
        best_alpha = None
        for alpha in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = repair_feasible(best + alpha * dir_norm * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-13:
                best_alpha = alpha
                best_score = score
                best = h_trial.copy()
                break
        if best_alpha is None:
            break

    return best, best_score


def transfer_dimension(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    src_edges = np.arange(n_src + 1) * dx_src
    src_F = np.concatenate([[0.0], np.cumsum(h_src * dx_src)])
    tgt_edges = np.arange(n_tgt + 1) * dx_tgt
    tgt_F = np.interp(tgt_edges, src_edges, src_F)
    h_tgt = (tgt_F[1:] - tgt_F[:-1]) / dx_tgt
    return repair_feasible(h_tgt)


def h_to_z(h):
    h_clipped = np.clip(h, 1e-10, 1.0 - 1e-10)
    z = np.log(h_clipped / (1.0 - h_clipped))
    z -= z.mean()
    return z


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    try:
        best_h = repair_feasible(np.asarray(initial_h_values, dtype=np.float64))
    except Exception:
        best_h = np.full(100, 0.5)

    best_score = exact_score(best_h)
    best_n = len(best_h)

    h_sym = repair_feasible(symmetrize(best_h))
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-13:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: Soft-max L-BFGS with beta annealing in logit space
    beta_schedule = [5, 15, 40, 100, 300, 1000, 3000, 10000, 30000, 100000]

    for beta in beta_schedule:
        if tl() < 90:
            break
        z0 = h_to_z(best_h)
        time_budget_phase = min(80.0, tl() - 60)
        h_new, score_new = softmax_lbfgs(z0, beta, best_n, maxiter=3000,
                                         time_limit=time_budget_phase)
        if h_new is not None and score_new < best_score - 1e-13:
            best_h = h_new
            best_score = score_new

    # Phase 2: Active descent polish
    if tl() > 40:
        h_ad, s_ad = active_descent(best_h, max_steps=300, top_k=25, time_limit=min(30, tl() - 20))
        if s_ad < best_score - 1e-13:
            best_h = h_ad
            best_score = s_ad

    # Phase 3: Another round of soft-max with higher beta
    if tl() > 60:
        for beta in [10000, 50000, 200000]:
            if tl() < 40:
                break
            z0 = h_to_z(best_h)
            h_new, score_new = softmax_lbfgs(z0, beta, best_n, maxiter=2000,
                                             time_limit=min(30, tl() - 30))
            if h_new is not None and score_new < best_score - 1e-13:
                best_h = h_new
                best_score = score_new

    # Phase 4: Active descent again
    if tl() > 30:
        h_ad, s_ad = active_descent(best_h, max_steps=200, top_k=30, time_limit=min(20, tl() - 15))
        if s_ad < best_score - 1e-13:
            best_h = h_ad
            best_score = s_ad

    # Phase 5: Dimension transfer + re-optimize
    if tl() > 80:
        n_cur = best_n
        target_dims = []
        for dn in [20, 40, 60]:
            nd = n_cur + dn
            if nd <= 250:
                target_dims.append(nd)
        if n_cur * 2 <= 250:
            target_dims.append(n_cur * 2)

        for nd in target_dims:
            if tl() < 50:
                break
            h_interp = transfer_dimension(best_h, nd)
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.03:
                z0 = h_to_z(h_interp)
                for beta in [50, 500, 5000]:
                    if tl() < 30:
                        break
                    h_new, score_new = softmax_lbfgs(z0, beta, nd, maxiter=1500,
                                                     time_limit=min(25, tl() - 25))
                    if h_new is not None and score_new < best_score - 1e-13:
                        best_h = h_new
                        best_score = score_new
                        best_n = nd
                        z0 = h_to_z(h_interp)
                    elif h_new is not None:
                        z0 = h_to_z(h_new)

                if tl() > 20:
                    h_ad, s_ad = active_descent(best_h if len(best_h) == nd else h_interp,
                                               max_steps=100, top_k=20,
                                               time_limit=min(15, tl() - 15))
                    if s_ad < best_score - 1e-13:
                        best_h = h_ad
                        best_score = s_ad
                        best_n = len(h_ad)

    # Phase 6: Multi-start perturbations
    if tl() > 40:
        n_pert = best_n
        for trial in range(12):
            if tl() < 25:
                break
            eps = rng.uniform(0.003, 0.015)
            z = rng.randn(n_pert)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            z0 = h_to_z(h_pert)
            h_new, score_new = softmax_lbfgs(z0, 1000, n_pert, maxiter=1500,
                                             time_limit=min(20, tl() - 20))
            if h_new is not None and score_new < best_score - 1e-13:
                best_h = h_new
                best_score = score_new
                best_n = n_pert
            if tl() > 10:
                h_ad, s_ad = active_descent(h_pert, max_steps=50, top_k=15,
                                           time_limit=min(10, tl() - 10))
                if s_ad < best_score - 1e-13:
                    best_h = h_ad
                    best_score = s_ad

    # Phase 7: Final active descent with more lags
    if tl() > 20:
        h_ad, s_ad = active_descent(best_h, max_steps=300, top_k=40, time_limit=min(15, tl() - 10))
        if s_ad < best_score - 1e-13:
            best_h = h_ad
            best_score = s_ad

    # Phase 8: Final SLSQP-style polish using scipy on h directly
    if tl() > 20:
        try:
            n = len(best_h)
            dx = 2.0 / n

            def obj_h(x):
                h = np.clip(x, 1e-10, 1.0 - 1e-10)
                C = np.correlate(h, 1.0 - h, mode="full") * dx
                C_max = C.max()
                beta_p = 5000.0
                exp_C = np.exp(beta_p * (C - C_max))
                return (C_max + np.log(exp_C.sum())) / beta_p

            def jac_h(x):
                h = np.clip(x, 1e-10, 1.0 - 1e-10)
                C = np.correlate(h, 1.0 - h, mode="full") * dx
                C_max = C.max()
                beta_p = 5000.0
                exp_C = np.exp(beta_p * (C - C_max))
                w = exp_C / exp_C.sum()
                return compute_weighted_gradient(h, w)

            res = minimize(obj_h, best_h, jac=jac_h, method='L-BFGS-B',
                           bounds=[(1e-10, 1 - 1e-10)] * n,
                           options={'maxiter': 500, 'ftol': 1e-15})
            h_sol = repair_feasible(res.x)
            s_sol = exact_score(h_sol)
            if s_sol < best_score - 1e-13:
                best_h = h_sol
                best_score = s_sol
        except Exception:
            pass

    # Final symmetrization check
    h_sym2 = repair_feasible(symmetrize(best_h))
    s_sym2 = exact_score(h_sym2)
    if s_sym2 < best_score - 1e-13:
        best_h = h_sym2
        best_score = s_sym2

    # Final repair and validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
