    # Actually lambda could be outside [-1, 1]
    # x_i = clip(h_i - lambda, 0, 1)
    # sum(x) is decreasing in lambda
    # At lambda = -inf, sum = n
    # At lambda = +inf, sum = 0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any tiny numerical issues
    diff = target - x.sum()
    if abs(diff) > 1e-12:

        if diff > 0:

            idx = np.argmax(room)
            x[idx] += min(diff, room[idx])
        else:
            idx = np.argmax(x)
            x[idx] += max(diff, -x[idx])

def get_lag_gradient_single(h, k):
    """Compute gradient of p_k with respect to h."""
    n = len(h)
    dx = 2.0 / n
    delta = k - (n - 1)
    g = np.zeros(n)
    if delta >= 0:
        if delta < n:
            g[delta:n] += dx
            g[delta:n] -= dx * h[0:n-delta]
            if n - delta > 0:
                g[0:n-delta] -= dx * h[delta:n]
    else:

        if d < n:
            g[0:n-d] += dx
            g[0:n-d] -= dx * h[d:n]
            g[d:n] -= dx * h[0:n-d]
    return g

    M = 2 * n - 1

    return h

def project_feasible(h, target=None):
    """Project onto {x: sum(x)=target, 0<=x<=1} via bisection."""
    n = len(h)
    if target is None:
        target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, x[idx] + diff)
    return x

def lse_obj_and_grad(x, tau):
    prof = get_profile(x)
    tp = tau * prof
    m = np.max(tp)
    exp_shifted = np.exp(tp - m)
    logsum = m + np.log(np.sum(exp_shifted))
    obj = logsum / tau
    weights = exp_shifted / np.sum(exp_shifted)
    G = get_all_lag_gradients(x)
    grad = weights @ G
    return obj, grad

def slsqp_lse(h_start, tau, maxiter=200, ftol=1e-12):
    n = len(h_start)
    target = n / 2.0

    def fun(x):
        v, _ = lse_obj_and_grad(x, tau)
        return v

    def jac(x):
        _, g = lse_obj_and_grad(x, tau)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    x0 = h_start.copy()
    try:
        res = minimize(fun, x0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair_feasible(res.x[:n])
    except Exception:
        pass
    return None

def slsqp_epigraph(h_start, t_start, active_lags, maxiter=100):
    n = len(h_start)
    M = 2 * n - 1
    target = n / 2.0
    nc = len(active_lags)

    def fun(x):
        return x[n]

    def jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - target
        prof = get_profile(h)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        G = get_all_lag_gradients(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + nc, n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 5.0)]
    x0 = np.concatenate([h_start, [t_start]])
    try:
        res = minimize(fun, x0, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair_feasible(res.x[:n])
    except Exception:
        pass
    return None

def subgradient_local(h_start, max_iters=100, time_limit=30.0, tolerance=1e-4):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()

    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        prof = get_profile(h)
        score = float(np.max(prof))
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()

        active_mask = prof > (score - tolerance)
        n_active = np.sum(active_mask)
        if n_active == 0:
            active_mask = prof > (score - 2 * tolerance)
            n_active = np.sum(active_mask)
        if n_active == 0:
            break

        G = get_all_lag_gradients(h)
        G_active = G[active_mask, :]
        grad = G_active.mean(axis=0)
        grad -= grad.mean()
        nrm = np.linalg.norm(grad)
        if nrm < 1e-15:
            break
        direction = -grad / nrm

        improved = False
        for alpha in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_trial = project_feasible(h + alpha * direction)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-13:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                improved = True
                break
        if not improved:
            break

    return best_h, best_score

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def structured_perturbation(h, n_modes=5, scale=0.02):
    n = len(h)
    x = np.arange(n)
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        a = np.random.randn() * scale
        b = np.random.randn() * scale
        pert += a * np.cos(2 * np.pi * m * x / n) + b * np.sin(2 * np.pi * m * x / n)
    pert -= pert.mean()
    return repair_feasible(h + pert)

def block_perturbation(h, scale=0.03):
    n = len(h)
    nblocks = np.random.randint(3, 8)
    block_size = n // nblocks
    pert = np.zeros(n)
    for i in range(nblocks):
        start = i * block_size
        end = start + block_size if i < nblocks - 1 else n
        pert[start:end] = np.random.randn() * scale
    pert -= pert.mean()
    return repair_feasible(h + pert)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start_global = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t_start_global)

    try:
        best_h = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        best_h = np.full(64, 0.5)

    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)

    h_sym = symmetrize(best_h)
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-14:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: Log-sum-exp SLSQP with increasing temperature
    tau_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0]
    for tau in tau_schedule:
        if tleft() < 35:
            break
        maxiter = int(min(200, max(40, tleft() * 0.15)))
        h_new = slsqp_lse(best_h.copy(), tau, maxiter=maxiter)
        if h_new is not None:
            s = exact_score(h_new)
            if s < best_score - 1e-13:
                best_h = h_new
                best_score = s
                h_sym2 = symmetrize(best_h)
                s2 = exact_score(h_sym2)
                if s2 < best_score - 1e-14:
                    best_h = h_sym2
                    best_score = s2

    # Phase 2: Subgradient local descent with adaptive tolerance
    if tleft() > 25:
        for tol in [0.05, 0.02, 0.01, 0.005]:
            if tleft() < 15:
                break
            tl = min(20.0, tleft() * 0.2)
            h_loc, s_loc = subgradient_local(best_h, max_iters=80, time_limit=tl, tolerance=tol)
            if s_loc < best_score - 1e-13:
                best_h = h_loc
                best_score = s_loc

    # Phase 3: Dynamic active-set epigraph polish
    if tleft() > 30:
        for round_idx in range(5):
            if tleft() < 20:
                break
            prof = get_profile(best_h)
            score = float(np.max(prof))
            # Active set: all lags within 1e-3 of max
            active = np.where(prof > score - 1e-3)[0]
            if len(active) > 80:
                active = np.argsort(prof)[::-1][:80]
            maxiter = int(min(120, max(30, tleft() * 0.25)))
            h_new = slsqp_epigraph(best_h, score + 1e-5, active, maxiter=maxiter)
            if h_new is not None:
                s = exact_score(h_new)
                if s < best_score - 1e-13:
                    best_h = h_new
                    best_score = s

    # Phase 4: Structured multi-start exploration
    if tleft() > 60:
        n_perturbs = int(min(15, max(5, tleft() // 10)))
        for p in range(n_perturbs):
            if tleft() < 30:
                break
            if p % 3 == 0:
                h_pert = structured_perturbation(best_h, n_modes=3 + p % 5, scale=0.015 + 0.005 * p)
            elif p % 3 == 1:
                h_pert = block_perturbation(best_h, scale=0.02 + 0.01 * p)
            else:
                z = np.random.randn(len(best_h))
                z -= z.mean()
                h_pert = repair_feasible(best_h + (0.01 + 0.005 * p) * z / np.std(z))

            s_pert = exact_score(h_pert)
            # Quick LSE polish from perturbation
            h_new = slsqp_lse(h_pert, 20.0, maxiter=60)
            if h_new is None:
                h_new = h_pert
            s = exact_score(h_new)
            # Subgradient polish
            h_loc, s_loc = subgradient_local(h_new, max_iters=30, time_limit=8.0, tolerance=0.01)
            if s_loc < s:
                s = s_loc
                h_new = h_loc
            if s < best_score - 1e-13:
                best_h = h_new
                best_score = s

    # Phase 5: Resolution search
    if tleft() > 50:
        pn = len(best_h)
        dims = sorted(set([pn - 10, pn - 5, pn + 5, pn + 10, pn + 20, pn + 40]))
        dims = [d for d in dims if 20 <= d <= 300]
        for new_n in dims:
            if tleft() < 25:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            h_new = slsqp_lse(h_interp, 30.0, maxiter=80)
            if h_new is None:
                h_new = h_interp
            s = exact_score(h_new)
            h_loc, s_loc = subgradient_local(h_new, max_iters=30, time_limit=8.0, tolerance=0.01)
            if s_loc < s:
                s = s_loc
                h_new = h_loc
            if s < best_score - 1e-13:
                best_h = h_new
                best_score = s

    # Phase 6: Final fine-grained polish
    if tleft() > 15:
        for tol in [0.003, 0.001, 0.0005, 0.0001]:
            if tleft() < 8:
                break
            tl = min(10.0, tleft() * 0.6)
            h_loc, s_loc = subgradient_local(best_h, max_iters=60, time_limit=tl, tolerance=tol)
            if s_loc < best_score - 1e-14:
                best_h = h_loc
                best_score = s_loc

    # Phase 7: Final epigraph with tight active set
    if tleft() > 10:
        prof = get_profile(best_h)
        score = float(np.max(prof))
        active = np.where(prof > score - 5e-4)[0]
        if len(active) < 5:
            active = np.argsort(prof)[::-1][:10]
        h_new = slsqp_epigraph(best_h, score + 1e-6, active, maxiter=60)
        if h_new is not None:
            s = exact_score(h_new)
            if s < best_score - 1e-14:
                best_h = h_new
                best_score = s

    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
