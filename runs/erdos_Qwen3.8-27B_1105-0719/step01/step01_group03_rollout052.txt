def project_onto_constraint(v):
    """Project v onto {0 <= h <= 1, sum(h) = n/2}."""
    n = len(v)
    target = n / 2.0
    # First clip to [0,1]
    h = np.clip(v, 0.0, 1.0)
    # Then adjust to meet the sum constraint
    # Use bisection on lambda: h_i = clip(v_i - lambda, 0, 1)
    # We need sum(h) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_mid = np.clip(v - mid, 0.0, 1.0)
        s = h_mid.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    # Final fixup
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            # Add to entries with most room
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                add = min(diff, room[i])
                h[i] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff >= -1e-14:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_mid = np.clip(h - mid, 0.0, 1.0)
        if h_mid.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def project_constrained(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_mid = np.clip(v - mid, 0.0, 1.0)
        if h_mid.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_profile(h, weights):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    grad = np.zeros(n)
    for m in range(M):
        wm = weights[m]
        if abs(wm) < 1e-15:
            continue
        d = m - (n - 1)
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n - d] += wm * dx * (-h[d:])
            grad[d:] += wm * dx * (1.0 - h[:n - d])
        else:
            dp = -d
            grad[:n - dp] += wm * dx * (1.0 - h[dp:])
            grad[dp:] += wm * dx * (-h[:n - dp])
    return grad

def lse_objective_and_grad(h, tau):
    n = len(h)
    profile = compute_profile(h)
    m = profile.max()
    scaled = tau * (profile - m)
    max_s = scaled.max()
    scaled = scaled - max_s
    exp_s = np.exp(scaled)
    s = exp_s.sum()
    weights = exp_s / s
    obj = m + (1.0 / tau) * (max_s + np.log(s))
    grad = gradient_profile(h, weights)
    return obj, grad

def lse_descent_with_momentum(h0, tau, n_iters, lr, momentum=0.9):
    n = len(h0)
    h = h0.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_actual = exact_score(h)
    for it in range(n_iters):
        obj, grad = lse_objective_and_grad(h, tau)
        vel = momentum * vel - lr * grad
        h_new = h + vel
        h_new = project_constrained(h_new, n)
        actual = exact_score(h_new)
        if actual < best_actual - 1e-12:
            best_actual = actual
            best_h = h_new.copy()
        h = h_new
        if it % 100 == 99:
            lr *= 0.92
    return best_h, best_actual

def lse_multistage(h0):
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    stages = [(5, 400, 0.008, 0.9), (20, 300, 0.004, 0.9), (80, 200, 0.0015, 0.85), (300, 150, 0.0008, 0.8)]
    for tau, nits, lr, mom in stages:
        h_new, c_new = lse_descent_with_momentum(h, tau, nits, lr, mom)
        if c_new < best_c - 1e-12:
            best_c = c_new
            h = h_new.copy()
        else:
            h = h_new.copy()
    return h, best_c

def solve_slsqp(h_start, maxiter=300, tol=1e-10):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)

    def obj_func(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_func(x):
        return x[:-1].sum() - n / 2.0

    def eq_grad(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_func(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_grad(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        J[:, -1] = 1.0
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                J[m, :n] = -dx * (1.0 - 2.0 * h)
            elif d > 0:
                J[m, :n - d] = dx * h[d:]
                J[m, d:n] = -dx * (1.0 - h[:n - d])
            else:
                dp = -d
                J[m, :n - dp] = -dx * (1.0 - h[dp:])
                J[m, dp:n] = dx * h[:n - dp]
        return J

    x0 = np.concatenate([h_start, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_func, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_grad}
    ]
    try:
        res = minimize(obj_func, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new < c0 + 0.05:
                return h_new
    except Exception:
        pass
    return None

def thue_morse_h(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 0 else 0.0
    return repair_feasible(h)

def walsh_h(n):
    h = np.zeros(n)
    for i in range(n):
        x = i
        val = 0
        while x > 0:
            val ^= 1
            x = (x & (x - 1))
        h[i] = 1.0 if val == 0 else 0.0
    return repair_feasible(h)

def block_pattern_h(n, block_size):
    h = np.zeros(n)
    for i in range(n):
        phase = (i // block_size) % 2
        h[i] = 1.0 if phase == 0 else 0.0
    return repair_feasible(h)

def smooth_pattern_h(n, freq, phase):
    i = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * i / n + phase)
    return repair_feasible(h)

def interpolate_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def local_subgrad_descent(h, max_iters=150):
    n = len(h)
    h = h.copy()
    best = exact_score(h)
    best_h = h.copy()
    for it in range(max_iters):
        profile = compute_profile(h)
        c = profile.max()
        if c < best - 1e-10:
            best = c
            best_h = h.copy()
        thresh = max(c - 1e-5, c - 0.001)
        active = profile >= thresh
        if not active.any():
            active[int(np.argmax(profile))] = True
        weights = np.zeros(2 * n - 1)
        weights[active] = 1.0 / active.sum()
        grad = gradient_profile(h, weights)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_try = h - alpha * grad
            h_try = project_constrained(h_try, n)
            s = exact_score(h_try)
            if s < best - 1e-12:
                h = h_try
                best = s
                best_h = h_try.copy()
                break
    return best_h, best
