import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    """Project h onto [0,1]^n ∩ {sum = target} using bisection."""
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if target is None:
        target = n / 2.0
    h[~np.isfinite(h)] = 0.5
    # Bisection for water-filling
    lo = -1.0 - h.max()
    hi = 1.0 - h.min() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        y = np.clip(h + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = y < 1.0 - 1e-10
        else:
            mask = y > 1e-10
        if mask.any():
            y[mask] += residual / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def binary_swap_search_1for1(b, deadline, max_rounds=200):
    """1-for-1 swap search for binary h."""
    n = len(b)
    curr_c = exact_objective(b)
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        best_improve = 1e-14
        best_swap = None
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                if curr_c - c > best_improve:
                    best_improve = curr_c - c
                    best_swap = (i, j)
        if best_swap is not None:

def exact_objective(h):
    n = len(h)

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.5
    lo = -2.0 - float(np.max(h)) if n > 0 else -1.0
    hi = 2.0 - float(np.min(h)) if n > 0 else 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(h + mid, 0.0, 1.0)
        s = float(np.sum(y))
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(y))
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = y < 1.0 - 1e-10
        else:
            mask = y > 1e-10
        if np.any(mask):
            y[mask] += residual / float(np.sum(mask))
            y = np.clip(y, 0.0, 1.0)
    return y

def subgradient_refine(h0, deadline, max_iter=500):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_objective(h)
    alpha = 0.05
    for it in range(max_iter):
        if time.time() > deadline - 3.0:
            break
        profile = exact_profile(h)
        max_val = float(np.max(profile))
        # Find active set (within 10% of max)
        threshold = max_val - 0.1 * max_val
        active = np.where(profile >= threshold)[0]
        # Compute gradient of active profile components
        g = np.zeros(n)
        for j in active:
            s = j - (n - 1)
            if s >= 0:
                # P_j = dx * sum_{i=s}^{n-1} h[i]*(1-h[i-s])
                for i in range(s, n):
                    g[i] += (1.0 - h[i - s])
                    g[i - s] -= h[i]
            else:
                t = -s
                for i in range(0, n - t):
                    g[i] += (1.0 - h[i + t])
                    g[i + t] -= h[i]
        g *= (2.0 / n)
        # Average gradient
        g /= float(len(active))
        # Subtract mean to preserve sum
        g -= np.mean(g)
        # Step
        h_new = h - alpha * g
        h_new = project_box_sum(h_new)
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-14:
            h = h_new
            curr_c = c_new
            alpha = min(alpha * 1.05, 0.2)
        else:
            alpha *= 0.7
            if alpha < 0.001:
                break
    return h, curr_c

    # Alternating
    h_alt = np.zeros(n)
    h_alt[::2] = 1.0
    if n % 2 == 1:
        h_alt = project_box_sum(h_alt)
    seeds.append(h_alt.copy())
    # Sine-based
    for freq in [2, 3, 4, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        h_sin = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        seeds.append(project_box_sum(h_sin))
    # Random
    seeds.append(random_binary(n, rng))
    seeds.append(random_binary(n, rng))
    return seeds

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5.0
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    # Load parent
    parent_h = None
    try:
        if 'initial_h_values' in globals():
            parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
            if len(parent_h) > 0:
                parent_h = project_box_sum(parent_h)
                pc = exact_objective(parent_h)
                if np.isfinite(pc) and pc < best_c:
                    best_c = pc
                    best_h = parent_h.copy()
                    best_n = len(parent_h)
    except Exception:
        pass

    if parent_h is None:
        parent_n = 48
        parent_h = project_box_sum(np.concatenate([np.ones(parent_n // 2), np.zeros(parent_n // 2)]))
        pc = exact_objective(parent_h)
        if pc < best_c:
            best_c = pc
            best_h = parent_h.copy()
            best_n = parent_n

    parent_n = len(parent_h)

    # Phase 1: Binary search at parent size
    if time.time() < deadline - 100:
        b = binary_threshold(parent_h)
        b, c_b = full_binary_search(b, deadline, do_3for3=True)
        if c_b < best_c:
            best_c = c_b
            best_h = b.copy()
            best_n = len(b)

    # Phase 2: Try multiple sizes
    sizes_to_try = []
    for s in range(32, 72, 2):
        if s != parent_n:
            sizes_to_try.append(s)
    # Also try parent_n + small variations
    for s in [parent_n - 2, parent_n + 2]:
        if 16 <= s <= 80 and s not in sizes_to_try:
            sizes_to_try.append(s)
    sizes_to_try = [s for s in sizes_to_try if s % 2 == 0 and 16 <= s <= 80]
    # Prioritize sizes near parent
    sizes_to_try.sort(key=lambda s: abs(s - parent_n))

    for n_try in sizes_to_try:
        if time.time() > deadline - 30:
            break
        remaining = deadline - time.time()
        if remaining < 40:
            break

        # Transfer and try binary
        tr = transfer_size(parent_h, n_try)
        b = binary_threshold(tr)
        time_per_size = min(80.0, remaining / 3.0)
        local_deadline = time.time() + time_per_size
        b, c_b = full_binary_search(b, local_deadline, do_3for3=(n_try <= 48))
        if c_b < best_c:
            best_c = c_b
            best_h = b.copy()
            best_n = n_try

        # Also try random starts
        if time.time() < deadline - 20 and n_try <= 56:
            for _ in range(3):
                if time.time() > deadline - 15:
                    break
                rb = random_binary(n_try, rng)
                rd = time.time() + min(30.0, (deadline - time.time()) / 2.0)
                rb, c_rb = full_binary_search(rb, rd, do_3for3=False)
                if c_rb < best_c:
                    best_c = c_rb
                    best_h = rb.copy()
                    best_n = n_try

    # Phase 3: Refined binary search on best found
    if best_h is not None and time.time() < deadline - 20:
        n_best = len(best_h)
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            # Already binary, do more 2-for-2
            b2 = best_h.copy()
            b2, c2 = binary_2for2(b2, deadline, max_rounds=150)
            if c2 < best_c:
                best_c = c2
                best_h = b2.copy()
            # More 1-for-1
            b3 = best_h.copy()
            b3, c3 = binary_1for1(b3, deadline, max_rounds=300)
            if c3 < best_c:
                best_c = c3
                best_h = b3.copy()
        else:
            # Continuous: try subgradient
            h_sg, c_sg = subgradient_refine(best_h, deadline, max_iter=200)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg.copy()
            # Threshold and binary search
            if len(best_h) % 2 == 0 and time.time() < deadline - 10:
                bt = binary_threshold(best_h)
                bt, c_bt = full_binary_search(bt, deadline, do_3for3=True)
                if c_bt < best_c:
                    best_c = c_bt
                    best_h = bt.copy()

    # Phase 4: Multi-start from best at same size
    if best_h is not None and time.time() < deadline - 15:
        n_best = len(best_h)
        for restart in range(5):
            if time.time() > deadline - 10:
                break
            # Perturb best and re-threshold
            perturbed = project_box_sum(best_h + rng.normal(0, 0.03 + 0.01 * restart, n_best))
            bp = binary_threshold(perturbed)
            bd = time.time() + min(25.0, (deadline - time.time()) / 2.0)
            bp, c_bp = full_binary_search(bp, bd, do_3for3=(n_best <= 48))
            if c_bp < best_c:
                best_c = c_bp
                best_h = bp.copy()

    # Phase 5: Try transferring to smaller sizes for final check
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        for n_small in [n_best - 2, n_best - 4, n_best - 6]:
            if n_small < 16 or n_small % 2 != 0:
                continue
            if time.time() > deadline - 8:
                break
            tr = transfer_size(best_h, n_small)
            bt = binary_threshold(tr)
            bd = time.time() + min(15.0, (deadline - time.time()) / 2.0)
            bt, c_bt = binary_1for1(bt, bd, max_rounds=200)
            bt, c_bt = binary_2for2(bt, bd, max_rounds=50)
            if c_bt < best_c:
                best_c = c_bt
                best_h = bt.copy()
                best_n = n_small

    # Final validation
    best_h = project_box_sum(best_h)
    final_c = exact_objective(best_h)
    if final_c < best_c:
        best_c = final_c
    # Verify feasibility
    n_final = len(best_h)
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)
    if abs(float(np.sum(best_h)) - n_final / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, n_final
