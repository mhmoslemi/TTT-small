def simulated_anneal(h, n_moves, T_start, T_end, deadline, rng):
    n = len(h)
    cur_c5 = float(compute_c5(h))
    best_c5 = cur_c5
    best_h = h.copy()
    no_improve = 0
    for m in range(n_moves):
        if time.time() > deadline:
            break
        progress = m / max(n_moves, 1)
        T = T_start * (T_end / T_start) ** progress
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if rng.random() < 0.5:
            inc, dec = i, j
        else:
            inc, dec = j, i
        cap = min(h[dec], 1.0 - h[inc])
        if cap < 1e-10:

        frac = p ** (1.0 + 3.0 * progress)

        h[dec] -= tau
        new_c5 = float(compute_c5(h))

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

def binary_round(h, n):
    """Round h to nearest binary vector with exactly n/2 ones."""
    target = n // 2
    order = np.argsort(h)[::-1]
    binary = np.zeros(n)

    # Verify sum
    s = float(np.sum(binary))
    if s != target:
        # Shouldn't happen, but just in case
        binary = project_feasible(h, n)
    return binary
