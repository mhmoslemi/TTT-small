    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    active = corr >= m - 1e-8
    if not np.any(active):
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
    weights = active / np.sum(active)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if not active[i]:
            continue
        k = i - (n - 1)
        w = weights[i]
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1-k:hi1+1-k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2+k:hi2+1+k]
    return grad
