    dx = 2.0 / n

                G[k, 0:n - delta] -= dx * h[delta:n]
        else:

                G[k, d:n] -= dx * h[0:n - d]

from scipy.optimize import minimize

    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]

def project_onto_feasible(h, S):
    """Project h onto {0 <= x <= 1, sum(x) = S} using bisection on lambda."""
    h = np.clip(h, 0.0, 1.0)
    # Binary search for lambda such that sum(clip(h - lambda, 0, 1)) = S
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    return project_onto_feasible(h, target)

    """Push values away from 0.5 toward 0 or 1."""
    h_new = 0.5 + gamma * (h - 0.5)
    return repair_feasible(h_new)

    n = len(h)

    h_new = project_onto_feasible(h - lr * grad, n / 2.0)

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:

        remaining = -diff
        for idx in order:

            h[idx] -= rem
            remaining -= rem

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def project_feasible(h, S):
    h = np.asarray(h, dtype=np.float64)
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        proj_sum = np.clip(h - mid, 0.0, 1.0).sum()
        if proj_sum > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def push_to_binary(h, gamma=5.0):
    h_new = 0.5 + gamma * (h - 0.5)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def threshold_to_binary(h):
    h_bin = (h > 0.5).astype(np.float64)
    return repair_feasible(h_bin)

def lse_optimize(h_start, betas, maxiter=40, penalty=0.0):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    for beta in betas:
        def obj(x):
            p = get_profile(x)
            bp = beta * p
            bm = bp.max()
            val = (bm + np.log(np.sum(np.exp(bp - bm)))) / beta
            if penalty > 0:
                val += penalty * np.sum(x * (1.0 - x))
            return val
        def obj_jac(x):
            p = get_profile(x)
            bp = beta * p
            bm = bp.max()
            w = np.exp(bp - bm)
            w /= w.sum()
            G = get_lag_gradients(x)
            g = w @ G
            if penalty > 0:
                g += penalty * (1.0 - 2.0 * x)
            return g
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_sol = repair_feasible(res.x)
                s = exact_score(h_sol)
                if s < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s
                h = h_sol
        except Exception:
            pass
    return best_h, best_score

def solve_epigraph(h_start, t_start, beta_pen=0.0, maxiter=50, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    prof = get_profile(h_start)
    if nc < M:
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta_pen * np.sum(h * (1.0 - h))

    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta_pen * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        pf = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        return np.concatenate([[eq], t - pf[active_lags]])

    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 1.0)]
    x0 = np.concatenate([h_start, [t_start]])
    try:
        result = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                         bounds=bounds,
                         constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                         options={'maxiter': maxiter, 'ftol': 1e-11})
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None

def discrete_sa(h0, time_limit=60, n_evals_max=2000000, T0=0.003, cooling=0.99995):
    n = len(h0)
    h = threshold_to_binary(h0)
    best_h = h.copy()
    best_score = exact_score(h)
    current_score = best_score
    ones = np.where(h == 1.0)[0].astype(np.int64).tolist()
    zeros = np.where(h == 0.0)[0].astype(np.int64).tolist()
    t0 = time.perf_counter()
    neval = 0
    T = T0
    T_min = 1e-7
    while time.perf_counter() - t0 < time_limit and neval < n_evals_max:
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        trial = h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        s_trial = exact_score(trial)
        neval += 1
        delta = s_trial - current_score
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, T_min)):
            h = trial
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            current_score = s_trial
            if s_trial < best_score:
                best_score = s_trial
                best_h = h.copy()
        T *= cooling
    return best_h, best_score

def local_gradient_descent(h_current, time_limit=20, n_active=30):

    it = 0

        prof = get_profile(h)
        score = float(np.max(prof))
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
        A = min(n_active, len(prof))
        active_idx = np.argsort(prof)[::-1][:A]
        G = get_lag_gradients(h)
        w = prof[active_idx].copy()
        w = np.maximum(w, 0.0)
        wsum = w.sum()
        if wsum < 1e-15:
            break
        w /= wsum
        direction = np.sum(w[:, None] * G[active_idx, :], axis=0)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = project_feasible(h + alpha * direction, n / 2.0)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                improved = True
                break
        if not improved:
            break
    return best_h, best_score
