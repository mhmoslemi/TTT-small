Let me verify the correlation mapping:
- np.correlate(a, v, 'full') where a and v have length n gives output of length 2n-1.
- The output at index k is sum_i a[i+k] * v[i] for valid i.
- More precisely, for np.correlate(h, 1-h, 'full'), output[k] = sum_i h[i + k - (n-1)] * (1-h[i]) for valid indices.

- For mode='full', the output has length 2*min(len(a),len(v))-1 = 2n-1 (when both have length n).
- The k-th element (0-indexed) is: sum over i of a[i] * v[i - k + (n-1)]... 

Actually, let me think about this differently. np.correlate(a, v, 'full') computes the cross-correlation. For two arrays of length n, the output at index k is:

output[k] = sum_i a[i] * v[::-1][i - k + (n-1)]

No, that's not right either. Let me just use the definition:

np.correlate(a, v, mode='full')[k] = sum_i a[i] * v[i + k - (len(v) - 1)]

where the sum is over all i where both indices are valid.

So for np.correlate(h, 1-h, 'full'):
output[k] = sum_i h[i] * (1-h[i + k - (n-1)])

where 0 <= i < n and 0 <= i + k - (n-1) < n.

Let me define d = k - (n-1). Then:
output[k] = sum_i h[i] * (1-h[i+d])

where 0 <= i < n and 0 <= i+d < n.

So for d > 0: i ranges from 0 to n-1-d
For d < 0: i ranges from -d to n-1
For d = 0: i ranges from 0 to n-1, and output = sum_i h[i]*(1-h[i])

This matches the strategy's description. Good.

Now for the gradient of output[k] with respect to h[j]:
output[k] = sum_i h[i] * (1-h[i+d])

d(output[k])/dh[j] = sum over all i where i=j or i+d=j of the derivative.
- If i=j: contributes (1-h[j+d])
- If i+d=j, i.e., i=j-d: contributes -h[j-d]

So d(output[k])/dh[j] = (1-h[j+d]) * [0<=j+d<n] - h[j-d] * [0<=j-d<n]

For d=0: d(output)/dh[j] = (1-h[j]) - h[j] = 1-2h[j]

For the symmetric parametrization:
- n = 2m

dh/dr[0] = 1 (for h[r])
dh/dr[n-1-r] = -1 (for h[n-1-r])

So d(output[k])/dq[r] = d(output[k])/dh[n-1-r] * (-1) + d(output[k])/dh[r] * 1

d(output[k])/dq[r] = d(output[k])/dh[r] * dh[r]/dq[r] + d(output[k])/dh[n-1-r] * dh[n-1-r]/dq[r]

d(output[k])/dq[r] = G[r] - G[n-1-r]

import numpy as np

def eval_score(h):
    """Exact score computation."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def corr_values(h):
    """Return all 2n-1 raw correlation values."""
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    """Build symmetric h from q. n = 2*len(q)."""
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    """Extract q from symmetric h. n = 2*len(q)."""
    n = len(h)
    m = n // 2
    q = np.zeros(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n-1-i]))
    return q

def project_feasible(x, target_sum):
    """Project x to {0<=h<=1, sum=target_sum}."""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_gradient_k(h, k):
    """Gradient of k-th correlation value (mode='full') w.r.t. h."""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # d/dh[i+d] contribution: -h[i] for each valid i
    # d/dh[i] contribution: (1-h[i+d]) for each valid i
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += -h[idx]
    return grad

def sym_corr_obj_grad(x, n_full):

    x = [q_0,...,q_{m-1}, t]

    w_sum = w.sum()

    else:
        w /= w_sum
    # Gradient w.r.t. t is 1 (for the smoothed part)

    # Gradient w.r.t. q

def make_smooth_func(beta):
    """Create smooth objective and gradient for symmetric variables."""
    def smooth_obj_grad(x, n_full):
        m = n_full // 2
        q = x[:m]
        h = build_sym_h(q)
        dx = 2.0 / n_full
        corrs_raw = corr_values(h)
        corrs = corrs_raw * dx
        max_c = corrs.max()
        shifted = corrs - max_c
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        val = max_c + np.log(w_sum) / beta
        grad_q = np.zeros(m)
        for k in range(2 * n_full - 1):
            if w[k] < 1e-14:
                continue
            g_full = corr_gradient_k(h, k)
            # d/dq[r] = G[r] - G[n-1-r]
            for r in range(m):
                grad_q[r] += w[k] * (g_full[r] - g_full[n_full - 1 - r])
        grad_q *= dx
        return val, grad_q
    return smooth_obj_grad

def smooth_sym_optimize(q0, n_full, beta_list, maxiter_per_beta, time_limit):
    """Optimize symmetric variables using smooth max."""
    m = n_full // 2
    q = q0.copy().clip(0, 1)
    h = build_sym_h(q)
    best_score = eval_score(h)
    best_q = q.copy()
    dx = 2.0 / n_full
    
    for beta in beta_list:
        time_start = time.time()
        if time_start > time_limit:
            break
        obj_grad = make_smooth_func(beta)
        
        def obj(x):
            return obj_grad(x, n_full)[0]
        
        def jac(x):
            return obj_grad(x, n_full)[1]
        
        bounds = [(0, 1)] * m
        try:
            res = minimize(obj, q, jac=jac, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter_per_beta,
                                                   'ftol': 1e-14, 'gtol': 1e-10})
            q_new = res.x.clip(0, 1)
        except Exception:
            q_new = q
        
        h_new = build_sym_h(q_new)
        score_new = eval_score(h_new)
        if score_new < best_score:
            best_score = score_new
            best_q = q_new.copy()
        q = q_new
    
    return best_q, best_score, build_sym_h(best_q)

def epigraph_sym_solve(q_center, n_full, active_indices, margin, maxiter, ftol, time_limit):
    """
    Solve epigraph subproblem in symmetric variables.
    min t s.t. t >= dx * corr_k(h(q)) for k in active_indices, 0<=q<=1
    """
    m = n_full // 2
    dx = 2.0 / n_full
    
    def get_h_from_x(x):
        return build_sym_h(x[:m])
    
    def objective(x):
        return x[m]  # just t
    
    def obj_grad(x):
        grad = np.zeros(m + 1)
        grad[m] = 1.0
        return grad
    
    constraints = []
    for k in active_indices:
        def con(x, k=k):
            h = build_sym_h(x[:m])
            corrs = corr_values(h)
            return x[m] - corrs[k] * dx
        def con_jac(x, k=k):
            h = build_sym_h(x[:m])
            g = corr_gradient_k(h, k)
            j = np.zeros(m + 1)
            # d(con)/dq[r] = -(dx)(G[r] - G[n-1-r])
            for r in range(m):
                j[r] = -dx * (g[r] - g[n_full - 1 - r])
            j[m] = 1.0
            return j
        constraints.append({'type': 'ineq', 'fun': con, 'jac': con_jac})
    
    x0 = np.zeros(m + 1)
    x0[:m] = q_center
    # Initial t: current max + small margin
    h_c = build_sym_h(q_center)
    corrs_c = corr_values(h_c) * dx
    x0[m] = corrs_c.max() + 1e-9
    
    bounds = [(0, 1)] * m + [(0, 1.0)]
    
    try:
        res = minimize(objective, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        q_out = res.x[:m].clip(0, 1)
    except Exception:
        q_out = q_center.copy()
    
    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)

def epigraph_full_solve(h_center, active_indices, maxiter, ftol, target_sum):
    """
    Solve epigraph subproblem in full variables.
    min t s.t. t >= dx * corr_k(h) for k in active_indices,
               sum(h) = target_sum, 0<=h<=1
    """
    n = len(h_center)
    dx = 2.0 / n
    
    def objective(x):
        return x[n]  # t
    
    def obj_grad(x):
        grad = np.zeros(n + 1)
        grad[n] = 1.0
        return grad
    
    constraints = []
    for k in active_indices:
        def con(x, k=k):
            h = x[:n]
            corrs = corr_values(h)
            return x[n] - corrs[k] * dx
        def con_jac(x, k=k):
            h = x[:n]
            g = corr_gradient_k(h, k)
            j = np.zeros(n + 1)
            j[:n] = -dx * g
            j[n] = 1.0
            return j
        constraints.append({'type': 'ineq', 'fun': con, 'jac': con_jac})
    
    # Equality: sum(h) = target_sum
    def eq_con(x):
        return np.sum(x[:n]) - target_sum
    def eq_con_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    constraints.append({'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac})
    
    x0 = np.zeros(n + 1)
    x0[:n] = h_center
    corrs_c = corr_values(h_center) * dx
    x0[n] = corrs_c.max() + 1e-9
    
    bounds = [(0, 1)] * n + [(0, 1.0)]
    
    try:
        res = minimize(objective, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        h_out = res.x[:n]
    except Exception:
        h_out = h_center.copy()
    
    h_out = project_feasible(h_out, target_sum)
    return h_out, eval_score(h_out)

def select_active(corrs_scaled, margin_frac=0.02, margin_abs=1e-6, neighbor_pad=3, max_k=80):
    """Select active lag indices based on proximity to max."""
    n = len(corrs_scaled)
    max_val = corrs_scaled.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = np.where(corrs_scaled >= threshold)[0]
    # Add neighbors
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < n:
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        # Keep the ones closest to max
        scored = [(corrs_scaled[k], k) for k in active_list]
        scored.sort(reverse=True)
        active_list = [k for _, k in scored[:max_k]]
        active_list.sort()
    return active_list

def transfer_to_n(h_old, n_new):
    """Transfer h from old resolution to new resolution."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, n_new / 2.0)
    return h_new

        if n_r % 2 == 0:
            h_r = make_binary_start(n_r, rng)
            h_r = 0.2 + 0.5 * h_r + 0.15 * rng.randn(n_r)
            h_r = project_feasible(h_r, n_r / 2.0)
            q_r = sym_from_h(h_r)
            if rem() > 80:
                tl = time.time() + min(55, rem() * 0.5)
                q_o, c5_o, h_o = smooth_sym_optimize(q_r, n_r, [100, 300], 150, tl)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()

                if c5_o < best_c5 + 0.005 and rem() > 40:
                    do_epig_sym_loop(h_o, max_rounds=10, per_solve_maxiter=40)
        else:
            h_r = make_binary_start(n_r, rng)
            h_r = 0.25 + 0.4 * h_r + 0.1 * rng.randn(n_r)
            h_r = project_feasible(h_r, n_r / 2.0)
            if rem() > 60:
                do_epig_full_loop(h_r, max_rounds=8, per_solve_maxiter=35)

    # Phase 6: Final polish on best
    if rem() > 30 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            if rem() > 50:
                tl = time.time() + min(40, rem() * 0.5)
                q_o, c5_o, h_o = smooth_sym_optimize(q_fin, n_fin, [600, 1200], 200, tl)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
            if rem() > 25:
                do_epig_sym_loop(best_h, max_rounds=15, per_solve_maxiter=50)
        else:
            if rem() > 30:
                do_epig_full_loop(best_h, max_rounds=10, per_solve_maxiter=40)

    # Final: one more epigraph pass with high maxiter
    if rem() > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            h_chk = build_sym_h(q_fin)
            corrs = corr_raw(h_chk) * (2.0 / n_fin)
            active = select_active(corrs, margin_frac=0.01, neighbor_pad=3, max_k=50)
            q_o, h_o, c5_o = solve_epigraph_sym(q_fin, n_fin, active, 80, 1e-12)
            h_o = build_sym_h(np.clip(q_o, 0, 1))
            c5_o = eval_score(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o

    # Validate and return
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = (0.5 + 0.3 * np.sin(2 * np.pi * x_fb)).clip(0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
