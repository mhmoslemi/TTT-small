```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    """Project x onto {h : 0<=h<=1, sum(h)=target_sum}."""
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    
    # Binary search for the shift that gives the right sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - h.sum()
    
    if abs(residual) > 1e-10:
        # Distribute residual among interior points
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += residual / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            # If still not exact, try again with remaining
            mask2 = (h > 1e-10) & (h < 1.0 - 1e-10)
            residual2 = target_sum - h.sum()
            if mask2.sum() > 0 and abs(residual2) > 1e-12:
                h[mask2] += residual2 / mask2.sum()
    
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    """Gradient of corr[k] with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    jdx = idx + offset
    grad[jdx] += (1.0 - h[idx])
    grad[idx] -= h[jdx]
    return grad


def slsqp_smooth(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx
    
    def con_fun(x):
        return x.sum() - target_sum
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_feasible(res.x, target_sum)
        return h_out, c5_of_h(h_out, dx)
    except Exception:
        return h0, c5_of_h(h0, dx)
