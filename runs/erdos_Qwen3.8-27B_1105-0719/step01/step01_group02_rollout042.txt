def active_set_direction(h, active_k_list, radius=0.1):
    n = len(h)
    m = len(active_k_list)
    
    # Build the gradient matrix: row k_a, col j gives ∂P(k_a)/∂h[j]
    # d has 2n+1 variables: d_pos[j], d_neg[j] for j in 0..n-1, plus t
    # Actually let's use: d[j] = d_pos[j] - d_neg[j], d_pos, d_neg ≥ 0
    # Variables: [d_pos[0], ..., d_pos[n-1], d_neg[0], ..., d_neg[n-1], t]
    # Total: 2n + 1
    
    # Objective: minimize t => c = [0, ..., 0, 1]
    c = np.zeros(2*n + 1)
    c[-1] = 1.0
    
    # Constraints: A_ub @ vars ≤ b_ub
    # For each active k: sum_j grad_k[j] * (d_pos[j] - d_neg[j]) ≤ t
    # i.e., sum_j grad_k[j]*d_pos[j] - sum_j grad_k[j]*d_neg[j] - t ≤ 0
    A_rows = []
    b_rows = []
    for k in active_k_list:
        grad = compute_grad_P(h, k)  # shape (n,)
        row = np.zeros(2*n + 1)
        row[0:n] = grad
        row[n:2*n] = -grad
        row[2*n] = -1.0  # -t ≤ 0 => grad·d - t ≤ 0
        A_rows.append(row)
        b_rows.append(0.0)
    
    # ∑ d[j] = 0 => ∑ d_pos[j] - ∑ d_neg[j] = 0
    # This is an equality constraint
    A_eq_row = np.zeros(2*n + 1)
    A_eq_row[0:n] = 1.0
    A_eq_row[n:2*n] = -1.0
    
    # ||d||_∞ ≤ radius: d_pos[j] ≤ radius, d_neg[j] ≤ radius
    # These are upper bounds on variables
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = A_eq_row.reshape(1, -1)
    b_eq = np.array([0.0])
    
    bounds = [(0, radius)] * (2*n) + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[0:n] - res.x[n:2*n]
    return d
```

def compute_grad_P(h, k):
    n = len(h)
    grad = np.zeros(n)
    # h[i+k] * (1-h[i]) summed over valid i
    # ∂/∂h[j] of h[i+k]*(1-h[i]):
    # j appears in h[i+k] when i = j-k (valid if 0 ≤ j-k < n): contributes (1-h[j-k])
    # j appears in h[i] when i = j (valid if 0 ≤ j+k < n): contributes -h[j+k]
    
    # First term: j-k in [0, n-1]
    if k >= 0:
        # j-k >= 0 => j >= k; j-k < n => j < n+k (always true since j < n)
        for j in range(k, n):
            grad[j] += (1 - h[j-k])
    else:
        # j-k >= 0 => j >= k (always true since j >= 0 > k)
        # j-k < n => j < n+k
        for j in range(0, n+k):
            grad[j] += (1 - h[j-k])
    
    # Second term: j+k in [0, n-1]
    if k >= 0:
        # j+k < n => j < n-k; j+k >= 0 always
        for j in range(0, n-k):
            grad[j] -= h[j+k]
    else:
        # j+k >= 0 => j >= -k; j+k < n always
        for j in range(-k, n):
            grad[j] -= h[j+k]
    
    return grad
```

import numpy as np
from scipy.optimize import linprog
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]) - 1e-15)
            if step <= 0:
                break
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]) + 1e-15)
            if step <= 0:
                break
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        mask = (h > 0) if diff > 0 else (h < 1.0)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_P(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            grad[k:n] += (1.0 - h[0:n - k])
            grad[0:n - k] -= h[k:n]
    else:
        m = -k
        if n - m > 0:
            grad[0:n - m] += (1.0 - h[m:n])
            grad[m:n] -= h[0:n - m]
    return grad

    if start_time is None:
        start_time = time.time()
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > start_time + time_limit:
            break
        
        profile = compute_profile(cur)
        max_p = np.max(profile)
        
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Precompute swap effects
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        effect_off = np.zeros((n_ones, profile_len))
        effect_on = np.zeros((n_zeros, profile_len))
        
        for ii in range(n_ones):
            i = ones[ii]
            for ki in range(profile_len):
                k = ki - (n - 1)
                d = 0.0
                if 0 <= i + k < n:
                    d += cur[i + k]
                if 0 <= i - k < n:
                    d -= (1.0 - cur[i - k])
                effect_off[ii][ki] = d
        
        for jj in range(n_zeros):
            j = zeros[jj]
            for ki in range(profile_len):
                k = ki - (n - 1)
                d = 0.0
                if 0 <= j + k < n:
                    d -= cur[j + k]
                if 0 <= j - k < n:
                    d += (1.0 - cur[j - k])
                effect_on[jj][ki] = d
        
        # Find best swap
        best_new_max = max_p
        best_ii, best_jj = -1, -1
        
        for ii in range(n_ones):
            new_profiles = profile[None, :] + effect_off[ii][None, :] + effect_on
            new_maxes = np.max(new_profiles, axis=1)
            min_mm = np.min(new_maxes)
            if min_mm < best_new_max - 1e-14:
                best_new_max = min_mm
                best_ii = ii
                best_jj = int(np.argmin(new_maxes))
        
        if best_ii >= 0:
            cur[ones[best_ii]] = 0.0
            cur[zeros[best_jj]] = 1.0
            cur_c = exact_objective(cur)
        else:
            break
    
    return cur, cur_c

def binary_2swap_improve(b, max_rounds=30, time_limit=None, start_time=None):
    """Try 2-swap moves (2 ones -> 0, 2 zeros -> 1)."""
    n = len(b)
    cur = b.copy()
    cur_c = exact_objective(cur)
    if start_time is None:
        start_time = time.time()
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > start_time + time_limit:
            break
        
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        profile = compute_profile(cur)
        max_p = np.max(profile)
        
        best_delta = 0.0
        best_combo = None
        
        # Sample pairs to keep it fast
        max_ones_try = min(len(ones), 40)
        max_zeros_try = min(len(zeros), 40)
        
        for ii in range(max_ones_try):
            for jj in range(ii + 1, max_ones_try):
                i1, i2 = ones[ii], ones[jj]
                found = False
                for kk in range(max_zeros_try):
                    for ll in range(kk + 1, max_zeros_try):
                        j1, j2 = zeros[kk], zeros[ll]
                        if j1 in (i1, i2) or j2 in (i1, i2):
                            continue
                        cur[i1] = 0.0
                        cur[i2] = 0.0
                        cur[j1] = 1.0
                        cur[j2] = 1.0
                        c = exact_objective(cur)
                        cur[i1] = 1.0
                        cur[i2] = 1.0
                        cur[j1] = 0.0
                        cur[j2] = 0.0
                        delta = cur_c - c
                        if delta > best_delta + 1e-14:
                            best_delta = delta
                            best_combo = (i1, i2, j1, j2)
                            found = True
                        if found and time_limit is not None and time.time() > start_time + time_limit:
                            break
                    if found and time_limit is not None and time.time() > start_time + time_limit:
                        break
        
        if best_combo is not None:
            i1, i2, j1, j2 = best_combo
            cur[i1] = 0.0
            cur[i2] = 0.0
            cur[j1] = 1.0
            cur[j2] = 1.0
            cur_c = exact_objective(cur)
        else:
            break
    
    return cur, cur_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

def continuous_refine(h, n_trials=3000, time_limit=None, start_time=None, seed=42):
    n = len(h)
    cur = h.copy()
    cur_c = exact_objective(cur)
    rg = np.random.RandomState(seed)
    if start_time is None:
        start_time = time.time()
    temp = max(0.002, 0.01 * cur_c)
    
    for t in range(n_trials):
        if time_limit is not None and time.time() > start_time + time_limit:
            break
        frac = t / float(n_trials)
        temp_t = temp * (1.0 - frac)
        i, j = rg.choice(n, 2, replace=False)
        amt = 0.1 * (0.3 + 0.7 * rg.random())
        if amt > cur[j] * 0.9:
            amt = cur[j] * 0.85
        if amt > (1.0 - cur[i]) * 0.9:
            amt = max(0.01, (1.0 - cur[i]) * 0.85)
        if amt < 0.003:
            continue
        cur[i] += amt
        cur[j] -= amt
        c = exact_objective(cur)
        if c < cur_c - 1e-13:
            cur_c = c
        else:
            delta = c - cur_c
            if rg.random() < np.exp(-delta / max(temp_t, 1e-9)):
                cur_c = c
            else:
                cur[i] -= amt
                cur[j] += amt
    
    return cur, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        parent = project_box_sum(parent)
        pc = exact_objective(parent)
        best_h = parent.copy()
        best_c = pc
        best_n = len(parent)
    except Exception:
        pass
    
    if best_h is None:
        n0 = 70
        h0 = np.zeros(n0)
        h0[::2] = 1.0
        best_h = project_box_sum(h0)
        best_c = exact_objective(best_h)
        best_n = n0
    
    # Multi-resolution strategy
    # Phase 1: Polish at parent size and nearby sizes
    sizes_phase1 = []
    if best_n > 0:
        sizes_phase1.append(best_n)
    for s in [70, 80, 60, 90, 100, 120]:
        if s not in sizes_phase1 and 40 <= s <= 300 and s % 2 == 0:
            sizes_phase1.append(s)
    
    for n in sizes_phase1:
        if time.time() > deadline - 80:
            break
        
        remaining = deadline - time.time() - 40
        if remaining < 30:
            break
        
        # Get candidate at this size
        if parent is not None and len(parent) != n:
            candidate = transfer_to_size(parent, n)
        elif parent is not None:
            candidate = parent.copy()
        else:
            candidate = best_h.copy() if len(best_h) == n else transfer_to_size(best_h, n)
        
        c_start = exact_objective(candidate)
        
        # Active-set LP polish (main optimization)
        time_budget = min(180, remaining * 0.5)
        candidate, c_refined = active_set_polish(
            candidate, n_active=10, radius=0.04, max_iters=400,
            time_limit=time_budget, start_time=time.time()
        )
        
        if c_refined < best_c:
            best_c = c_refined
            best_h = candidate.copy()
            best_n = n
        
        # Binary refinement
        if n % 2 == 0 and time.time() < deadline - 50:
            b = binary_threshold(candidate)
            if b is not None:
                rem2 = deadline - time.time() - 30
                b, b_c = fast_binary_swap(b, max_rounds=80,
                    time_limit=min(60, rem2 * 0.4), start_time=time.time())
                if b_c < best_c:
                    best_c = b_c
                    best_h = b.copy()
                    best_n = n
                
                # 2-swap improvement
                if time.time() < deadline - 40:
                    b2, b2_c = binary_2swap_improve(b, max_rounds=20,
                        time_limit=min(40, max(5, deadline - time.time() - 30)),
                        start_time=time.time())
                    if b2_c < best_c:
                        best_c = b2_c
                        best_h = b2.copy()
                        best_n = n
    
    # Phase 2: Polish at larger sizes for finer resolution
    if time.time() < deadline - 60:
        for n in [best_n + 20, best_n + 40]:
            if n > 400 or n % 2 != 0:
                continue
            if time.time() > deadline - 50:
                break
            candidate = transfer_to_size(best_h, n)
            c_start = exact_objective(candidate)
            time_budget = min(120, deadline - time.time() - 30)
            candidate, c_refined = active_set_polish(
                candidate, n_active=8, radius=0.03, max_iters=300,
                time_limit=time_budget, start_time=time.time()
            )
            if c_refined < best_c:
                best_c = c_refined
                best_h = candidate.copy()
                best_n = n
    
    # Phase 3: Final fine polish
    if time.time() < deadline - 20:
        rem = deadline - time.time() - 10
        if rem > 10:
            final_h, final_c = active_set_polish(
                best_h, n_active=12, radius=0.01, max_iters=200,
                time_limit=rem, start_time=time.time()
            )
            if final_c < best_c:
                best_c = final_c
                best_h = final_h
    
    # Final projection and evaluation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
