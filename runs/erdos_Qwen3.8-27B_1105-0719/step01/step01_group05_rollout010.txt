from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.minimum(h, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.maximum(h, 0.0)
    # Fix any tiny residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmax(h < 1.0) if diff > 0 else np.argmax(h > 0.0))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    target = 0.5 * n
    h = project_box_sum(h, target).copy()
    c5 = exact_objective(h)

        if deadline is not None and time.time() > deadline - 5:
            break

                    h[j] -= lam

                    h[j] += lam

    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
