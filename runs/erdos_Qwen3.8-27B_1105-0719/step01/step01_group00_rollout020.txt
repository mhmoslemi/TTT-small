    grad[idx] += (-h[idx + d])
    return grad


def h_grad_to_q_grad(grad_h, n):
    m = n // 2
    jq = np.arange(m)
    return grad_h[jq] - grad_h[n - 1 - jq]


def smooth_val_grad(q, n, beta):
    h = build_sym(n, q)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    scaled = corrs * dx
    lbs = logsumexp(beta * scaled)
    val = lbs / beta
    w = np.exp(beta * scaled - lbs)
    grad_h = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-15:
            continue
        grad_h += w[k] * grad_corr_k_h(h, k, n)
    grad_q = h_grad_to_q_grad(grad_h, n) * dx
    return val, grad_q


def smooth_optimize_step(q0, n, beta, maxiter, ftol):
    m = len(q0)
    bounds = [(0.0, 1.0)] * m
    res = minimize(smooth_val_grad, q0, args=(n, beta),
                   method='L-BFGS-B', bounds=bounds,
                   jac=True, options={'maxiter': maxiter, 'ftol': ftol})
    q_opt = np.clip(res.x, 0.0, 1.0)
    return q_opt


def select_active_set(scaled_corrs, margin_frac=0.02, max_active=60, neighbor_pad=2):
    s_max = scaled_corrs.max()
    margin = max(1e-8, margin_frac * s_max)
    active = set(np.where(scaled_corrs >= s_max - margin)[0].tolist())
    # Add neighbors
    expanded = set()
    for k in active:
        for pad in range(-neighbor_pad, neighbor_pad + 1):
            nk = k + pad
            if 0 <= nk < len(scaled_corrs):
                expanded.add(nk)
    if len(expanded) > max_active:
        # Keep the top ones by correlation value
        sorted_idx = np.argsort(scaled_corrs[sorted(expanded)])[::-1]
        active_set = sorted(expanded)[sorted_idx[:max_active]]
    else:
        active_set = sorted(expanded)
    return active_set


def epigraph_solve(q0, n, active_k, dx, t_upper, maxiter=40, ftol=1e-9):
    m = n // 2
    x0 = np.concatenate([q0, [t_upper]])
    bounds = [(0.0, 1.0)] * m + [(0.0, t_upper)]
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    # Constraint: t - dx * corr_k >= 0 for k in active_k
    ak = np.array(active_k)
    
    def cons_fun(x):
        h = build_sym(n, x[:-1])
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return x[-1] - dx * corrs[ak]
    
    def cons_jac(x):
        h = build_sym(n, x[:-1])
        jac_rows = np.zeros((len(ak), len(x)))
        for i, k in enumerate(ak):
            gh = grad_corr_k_h(h, k, n)
            gq = h_grad_to_q_grad(gh, n)
            jac_rows[i, :-1] = -dx * gq
            jac_rows[i, -1] = 1.0
        return jac_rows
    
    cons = [{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            q_opt = np.clip(res.x[:-1], 0.0, 1.0)
            return q_opt
    except Exception:
        pass
    return None
