import numpy as np
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(400):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def transfer_to_size(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)


def lse_profile_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp + 1e-300)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-12:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


def slsqp_one_tau(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        return lse_profile_and_grad(x, tau)[0]

    def grad_fn(x):
        return lse_profile_and_grad(x, tau)[1]

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(len(x))

    try:
        from scipy.optimize import minimize
        res = minimize(obj_fn, h0.copy(), method='SLSQP', jac=grad_fn,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_box_sum(np.array(res.x, dtype=np.float64))
    except Exception:
        return h0.copy()

            b[j] = 0.0
