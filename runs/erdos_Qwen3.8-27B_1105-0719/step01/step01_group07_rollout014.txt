import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            lo, hi = 0.0, 1.0 - h[mask].min()
            for _ in range(60):
                mid = (lo + hi) / 2.0
                if s + mid * cnt < target:
                    lo = mid
                else:
                    hi = mid
            h[mask] = np.minimum(1.0, h[mask] + lo)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            lo, hi = 0.0, h[mask].min()
            for _ in range(60):
                mid = (lo + hi) / 2.0
                if s - mid * cnt > target:
                    lo = mid
                else:
                    hi = mid
            h[mask] = np.maximum(0.0, h[mask] - lo)
    return np.clip(h, 0.0, 1.0)


def binary_from_continuous(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def binary_hill_climb(b, deadline, max_rounds=500):
    n = len(b)
    curr_energy = int(np.max(np.correlate(b, 1.0 - b, mode='full')))
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        best_improve = 1
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                e = int(np.max(np.correlate(b, 1.0 - b, mode='full')))
                b[i] = 1.0
                b[j] = 0.0
                if curr_energy - e > best_improve:
                    best_improve = curr_energy - e
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        b[i] = 0.0
        b[j] = 1.0
        curr_energy -= best_improve
        curr_c = exact_objective(b)
        ones = list(np.where(b == 1.0)[0])
        zeros = list(np.where(b == 0.0)[0])
    return b, curr_c


def binary_sa(b0, deadline, rng, max_iter=200000):
    n = len(b0)
    b = b0.copy()
    curr_energy = int(np.max(np.correlate(b, 1.0 - b, mode='full')))
    best_b = b.copy()
    best_energy = curr_energy
    best_c = exact_objective(b)
    T = 5.0
    cool = 0.9995
    n_iter = min(max_iter, max(1000, int((deadline - time.time()) * 3000)))
    ones_list = list(np.where(b == 1.0)[0])
    zeros_list = list(np.where(b == 0.0)[0])
    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        if it % 1000 == 0:
            T *= cool
        if not ones_list or not zeros_list:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        b[i] = 0.0
        b[j] = 1.0
        new_energy = int(np.max(np.correlate(b, 1.0 - b, mode='full')))
        delta = new_energy - curr_energy
        if delta <= 0 or rng.expovariate(1.0 / max(T, 0.01)) > delta:
            curr_energy = new_energy
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if curr_energy < best_energy:
                best_energy = curr_energy
                best_b = b.copy()
                best_c = exact_objective(b)
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c, best_energy


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            for k in range(n - s):
                grad[k] += w * dx * (1.0 - h[k + s])
            for k in range(s, n):
                grad[k] -= w * dx * h[k - s]
        else:
            a = -s
            for k in range(a, n):
                grad[k] += w * dx * (1.0 - h[k - a])
            for k in range(0, n - a):
                grad[k] -= w * dx * h[k + a]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumCon:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))


def slsqp_lse(h0, tau, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    f = LSEFunc(tau)
    con = SumCon(n / 2.0)
    cons = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(f.value, h0.copy(), method='SLSQP', jac=f.gradient,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(np.array(res.x, dtype=np.float64))
    except Exception:
        return h0.copy()


def refine_lse(h0, deadline, tau_schedule=None):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_objective(h)
    if tau_schedule is None:
        tau_schedule = [0.02, 0.006, 0.002, 0.0007, 0.0002, 0.00006]
    for tau in tau_schedule:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_lse(h, tau, maxiter=min(400, n * 4 + 30))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-14:
            h = h_new
            curr_c = c_new
    return h, curr_c


def epigraph_solve(h0, deadline, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    h0 = project_box_sum(h0.copy())
    t0 = float(exact_objective(h0)) * 1.01 + 1e-6
    x0 = np.concatenate([h0, [t0]])

    def fun(x):
        return float(x[-1])

    def fun_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_con(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    def ineq_con(x):
        h = x[:n]
        t = x[-1]
        prof = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - prof

    def ineq_jac(x):
        h = x[:n]
        J = np.zeros((n_lags, n + 1))
        for j in range(n_lags):
            s = j - (n - 1)
            if s >= 0:
                for k in range(n - s):
                    J[j, k] += dx * (1.0 - h[k + s])
                for k in range(s, n):
                    J[j, k] -= dx * h[k - s]
            else:
                a = -s
                for k in range(a, n):
                    J[j, k] += dx * (1.0 - h[k - a])
                for k in range(n - a):
                    J[j, k] -= dx * h[k + a]
            J[j, -1] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_con, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_con, 'jac': ineq_jac}
    ]
    try:
        res = minimize(fun, x0, method='SLSQP', jac=fun_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_box_sum(res.x[:n].copy())
        return h_new
    except Exception:
        return h0.copy()


def transfer_to_size(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)


def make_binary_start(n, rng, mode):
    if n % 2 != 0:
        n = n - 1
    m = n // 2
    if mode == 0:
        idx = rng.choice(n, m, replace=False)
    elif mode == 1:
        idx = np.arange(0, n, 2)[:m]
    elif mode == 2:
        idx = np.arange(0, m)
    elif mode == 3:
        block = max(2, n // 10)
        idx = []
        for i in range(0, n, 2 * block):
            idx.extend(range(i, min(i + block, n)))
        idx = idx[:m]
        if len(idx) < m:
            remaining = [i for i in range(n) if i not in set(idx)]
            idx.extend(rng.choice(remaining, m - len(idx), replace=False))
        idx = list(set(idx))[:m]
    elif mode == 4:
        idx = sorted(rng.choice(n, m, replace=False))
    else:
        idx = rng.choice(n, m, replace=False)
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load parent
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
    except Exception:
        pass

    if best_h is None:
        best_h = np.zeros(48)
        best_h[:24] = 1.0
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    parent_n = len(parent_h) if parent_h is not None else 48

    # Phase 1: Binary SA for multiple sizes
    sizes = sorted(set([parent_n, parent_n + 4, parent_n + 8, parent_n + 12,
                        max(24, parent_n - 4), max(24, parent_n - 8)]))
    sizes = [s for s in sizes if s % 2 == 0 and 20 <= s <= 80]

    binary_results = {}
    for n in sizes:
        if time.time() > deadline - 120:
            break
        # Multiple restarts
        for restart in range(8):
            if time.time() > deadline - 30:
                break
            mode = (restart + len(sizes)) % 5
            b_init = make_binary_start(n, rng, mode)
            # Quick hill climb first
            b_hc, c_hc = binary_hill_climb(b_init, deadline - 20, max_rounds=100)
            # Then SA from the hill-climb result
            b_sa, c_sa, e_sa = binary_sa(b_hc, deadline - 20, rng, max_iter=50000)
            # Final hill climb
            b_final, c_final = binary_hill_climb(b_sa, deadline - 10, max_rounds=200)
            if c_final < best_c - 1e-12:
                best_c = c_final
                best_h = b_final.copy()
            if n not in binary_results or c_final < binary_results[n][1]:
                binary_results[n] = (b_final, c_final)

    # Phase 2: Continuous refinement from parent and best binary
    if time.time() < deadline - 80:
        # Refine parent directly
        h_ref, c_ref = refine_lse(parent_h.copy(), deadline - 60)
        if c_ref < best_c - 1e-12:
            best_c = c_ref
            best_h = h_ref.copy()

    # Refine from best binary (as continuous)
    if time.time() < deadline - 60 and binary_results:
        best_bin_n = min(binary_results.keys(), key=lambda k: binary_results[k][1])
        best_bin = binary_results[best_bin_n][0]
        # Slightly perturb the binary to get a continuous starting point
        h_cont = best_bin + 0.01 * rng.randn(len(best_bin))
        h_cont = project_box_sum(h_cont)
        h_ref, c_ref = refine_lse(h_cont, deadline - 40)
        if c_ref < best_c - 1e-12:
            best_c = c_ref
            best_h = h_ref.copy()

    # Also try refining from transfer of best to different sizes
    if time.time() < deadline - 50:
        for n in [best_h.shape[0], best_h.shape[0] + 4, best_h.shape[0] - 4]:
            if n % 2 != 0 or n < 20 or n > 80:
                continue
            if time.time() > deadline - 15:
                break
            h_t = transfer_to_size(best_h, n)
            h_ref, c_ref = refine_lse(h_t, deadline - 10)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()

    # Phase 3: Epigraph polish on best continuous
    if time.time() < deadline - 20 and not np.all((best_h == 0.0) | (best_h == 1.0)):
        h_epi = epigraph_solve(best_h, deadline - 5, maxiter=200)
        c_epi = exact_objective(h_epi)
        if c_epi < best_c - 1e-12:
            best_c = c_epi
            best_h = h_epi.copy()

    # Phase 4: Binary polish from current best
    if time.time() < deadline - 10:
        n_best = len(best_h)
        if n_best % 2 == 0:
            b = binary_from_continuous(best_h)
            if b is not None:
                b, c_b = binary_hill_climb(b, deadline - 5, max_rounds=300)
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()
                # SA from this
                if time.time() < deadline - 5:
                    b_sa, c_sa, _ = binary_sa(b, deadline - 2, rng, max_iter=30000)
                    b_sa, c_sa2 = binary_hill_climb(b_sa, deadline - 1, max_rounds=100)
                    if c_sa2 < best_c - 1e-12:
                        best_c = c_sa2
                        best_h = b_sa.copy()

    # Phase 5: Final LSE polish
    if time.time() < deadline - 5 and not np.all((best_h == 0.0) | (best_h == 1.0)):
        h_final = slsqp_lse(best_h, 0.0005, maxiter=200)
        c_final = exact_objective(h_final)
        if c_final < best_c - 1e-12:
            best_c = c_final
            best_h = h_final.copy()

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    n_final = len(best_h)

    if abs(np.sum(best_h) - n_final / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    if np.max(best_h) > 1.0 + 1e-9 or np.min(best_h) < -1e-9:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, len(best_h)
