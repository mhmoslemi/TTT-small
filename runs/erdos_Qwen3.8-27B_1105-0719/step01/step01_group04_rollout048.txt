def exact_score(h):

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)

    s = h.sum()
    diff = target - s

        return h
    if diff > 0:
        idx = np.argsort(h)
        rem = diff

            h[i] += add

    else:

def compute_lag_values_and_grads(h):
    """Compute values and gradients for all lags.
    Returns values (2n-1,) and grads (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    # Lag 0
    values[n-1] = dx * np.sum(h * (1.0 - h))
    grads[n-1, :] = dx * (1.0 - 2.0 * h)
    
    for s in range(1, n):
        # Positive lag s: Q = dx * sum_{j=0}^{n-1-s} h[j]*(1-h[j+s])
        vals_pos = h[:n-s] * (1.0 - h[s:])
        values[n-1+s] = dx * np.sum(vals_pos)
        # Gradient
        # dQ/dh[k] for k in [0, n-1-s]: (1-h[k+s])
        grads[n-1+s, :n-s] += dx * (1.0 - h[s:])
        # dQ/dh[k] for k in [s, n-1]: -h[k-s]
        grads[n-1+s, s:] += dx * (-h[:n-s])
        
        # Negative lag -s: Q = dx * sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
        vals_neg = h[s:] * (1.0 - h[:n-s])
        values[n-1-s] = dx * np.sum(vals_neg)
        # Gradient
        # dQ/dh[k] for k in [s, n-1]: (1-h[k-s])
        grads[n-1-s, s:] += dx * (1.0 - h[:n-s])
        # dQ/dh[k] for k in [0, n-1-s]: -h[k]
        grads[n-1-s, :n-s] += dx * (-h[:n-s])
    
    return values, grads


def interpolate_h(h, new_n):
    """Interpolate h to a new grid of size new_n."""
    n = len(h)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_n) + 0.5) * 2.0 / new_n
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    y_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_tgt, x_ext, y_ext)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))


def solve_trust_qp(h_cur, values, grads, active_idx, lambda_trust):
    """Solve the trust-region QP subproblem using CVXPY.
    minimize t + lambda * ||h_new - h_cur||^2
    s.t. values[r] + grads[r] @ (h_new - h_cur) <= t  for r in active_idx
         sum(h_new) = n/2
         0 <= h_new <= 1
    Returns h_new or None."""
    if not HAS_CVX:
        return None
    n = len(h_cur)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Trust region penalty
    penalty = lambda_trust * cp.sum_squares(h_var - h_cur)
    objective = cp.Minimize(t_var + penalty)
    
    constraints = []
    for r in active_idx:
        lhs = values[r] + grads[r] @ (h_var - h_cur)
        constraints.append(lhs <= t_var)
    constraints.append(cp.sum(h_var) == n / 2.0)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CLARABINE)  # or CP_ECOS
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    
    if h_var.value is None:
        return None
    h_new = h_var.value.copy()
    return repair_feasible(h_new)

import numpy as np
import time
import math

try:
    import cvxpy as cp
    HAS_CVX = True
except Exception:
    HAS_CVX = False

from scipy.optimize import minimize


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(h)
        rem = diff
        for i in idx:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        idx = np.argsort(-h)
        rem = -diff
        for i in idx:
            if rem <= 1e-15:
                break
            sub = min(rem, h[i])
            h[i] -= sub
            rem -= sub
    err = target - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            for i in range(n):
                if h[i] < 1.0:
                    h[i] = min(1.0, h[i] + err)
                    break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    h[i] = max(0.0, h[i] + err)
                    break
    return h


def compute_lag_data(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    values[n - 1] = dx * np.sum(h * (1.0 - h))
    grads[n - 1, :] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        values[n - 1 + s] = dx * np.sum(h[:n - s] * (1.0 - h[s:]))
        grads[n - 1 + s, :n - s] += dx * (1.0 - h[s:])
        grads[n - 1 + s, s:] += dx * (-h[:n - s])
        values[n - 1 - s] = dx * np.sum(h[s:] * (1.0 - h[:n - s]))
        grads[n - 1 - s, s:] += dx * (1.0 - h[:n - s])
        grads[n - 1 - s, :n - s] += dx * (-h[:n - s])
    return values, grads


def interpolate_h(h, new_n):
    n = len(h)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_n) + 0.5) * 2.0 / new_n
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    y_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_tgt, x_ext, y_ext)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))


def solve_qp_cvx(h_cur, vals, grads, active_idx, lam):
    n = len(h_cur)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    objective = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h_cur))
    constraints = []
    for r in active_idx:
        constraints.append(vals[r] + grads[r] @ (h_var - h_cur) <= t_var)
    constraints.append(cp.sum(h_var) == n / 2.0)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CLARABINE)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return None
    if h_var.value is None:
        return None
    hv = np.asarray(h_var.value, dtype=np.float64)
    return repair_feasible(hv)


def make_qp_objective(h_cur, vals, grads, active_idx, lam):
    n = len(h_cur)
    m = len(active_idx)

    def qp_obj(x):
        t = x[n]
        h_new = x[:n]
        return t + lam * np.sum((h_new - h_cur) ** 2)

    def qp_obj_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 2.0 * lam * (x[:n] - h_cur)
        g[n] = 1.0
        return g

    def qp_eq_fun(x):
        return np.sum(x[:n]) - n / 2.0

    def qp_eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    def qp_ineq_fun(x):
        h_new = x[:n]
        t_val = x[n]
        lhs = vals[active_idx] + grads[active_idx] @ (h_new - h_cur)
        return t_val - lhs

    def qp_ineq_jac(x):
        g = np.zeros((m, n + 1))
        g[:, n] = 1.0
        g[:, :n] = -grads[active_idx, :]
        return g

    return qp_obj, qp_obj_jac, qp_eq_fun, qp_eq_jac, qp_ineq_fun, qp_ineq_jac


def solve_qp_slsqp(h_cur, vals, grads, active_idx, lam):
    n = len(h_cur)
    qp_obj, qp_obj_jac, qp_eq_fun, qp_eq_jac, qp_ineq_fun, qp_ineq_jac = \
        make_qp_objective(h_cur, vals, grads, active_idx, lam)
    c_max = np.max(vals[active_idx])
    x0 = np.concatenate([h_cur, [c_max + 0.1]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c_max + 2.0)]
    constraints = [
        {'type': 'eq', 'fun': qp_eq_fun, 'jac': qp_eq_jac},
        {'type': 'ineq', 'fun': qp_ineq_fun, 'jac': qp_ineq_jac}
    ]
    try:
        result = minimize(qp_obj, x0, jac=qp_obj_jac, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 150, 'ftol': 1e-13})
        if result.x is not None:
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None


def select_active(values, k_max=30, frac=0.003):
    c_max = np.max(values)
    threshold = c_max - frac * c_max
    candidates = np.where(values >= threshold)[0]
    if len(candidates) > k_max:
        order = np.argsort(values)[::-1][:k_max]
        return np.sort(order)
    if len(candidates) == 0:
        return np.argsort(values)[::-1][:5]
    return np.sort(candidates)


def trust_region_loop(h_init, max_iter=40, lam_init=0.05, k_active=25,
                      use_cvx=True, verbose=False):
    n = len(h_init)
    h = repair_feasible(h_init)
    best_h = h.copy()
    best_score = exact_score(h)
    lam = lam_init

    for it in range(max_iter):
        values, grads = compute_lag_data(h)
        c_max = float(np.max(values))
        active = select_active(values, k_max=k_active, frac=0.002)

        h_new = None
        if use_cvx and HAS_CVX:
            h_new = solve_qp_cvx(h, values, grads, active, lam)
        if h_new is None:
            h_new = solve_qp_slsqp(h, values, grads, active, lam)
        if h_new is None:
            break

        direction = h_new - h
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-10:
            break

        improved = False
        model_reduce = c_max - float(values[active][0] + grads[active][0] @ direction)
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            h_trial = repair_feasible(h + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                actual_reduce = best_score - score
                h = h_trial
                best_score = score
                best_h = h.copy()
                improved = True
                if model_reduce > 0:
                    ratio = actual_reduce / model_reduce
                    if ratio > 0.6:
                        lam = max(lam * 0.6, 0.002)
                    elif ratio < 0.1:
                        lam = min(lam * 1.5, 50.0)
                break

        if not improved:
            lam = min(lam * 1.8, 50.0)
            if lam > 20.0:
                break

    return best_h, best_score


def active_descent_polish(h_init, max_steps=50, sigma_scale=0.003):
    n = len(h_init)
    best = repair_feasible(h_init)
    best_score = exact_score(best)
    for _ in range(max_steps):
        values, grads = compute_lag_data(best)
        c = float(np.max(values))
        n_act = min(25, 2 * n - 1)
        idx = np.argsort(values)[::-1][:n_act]
        gaps = np.maximum(c - values[idx], 0.0)
        sigma = sigma_scale * c + 1e-10
        w = np.exp(-gaps / sigma)
        w /= w.sum() + 1e-15
        direction = -np.sum(w[:, None] * grads[idx, :], axis=0)
        direction -= np.mean(direction)
        improved = False
        for alpha in [0.02, 0.012, 0.007, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
            h_t = repair_feasible(best + alpha * direction)
            sc = exact_score(h_t)
            if sc < best_score - 1e-13:
                best = h_t
                best_score = sc
                improved = True
                break
        if not improved:
            break
    return best, best_score


def symmetrize(h):
    return repair_feasible((h + h[::-1]) / 2.0)


def gradient_fallback(h, n_steps=100):

    best_score = exact_score(best)
    for _ in range(n_steps):
