            # dG/dh[j]:
            #   if j <= n-d-1: +1-h[j+d]  (j acts as i)
            #   if j >= d: -h[j-d]  (j acts as i+d)
            d = -k
            idx1 = np.arange(0, n-d)  # j values where j<=n-d-1
            G[t, idx1] += (1.0 - h[idx1 + d])
            idx2 = np.arange(d, n)  # j values where j>=d
            G[t, idx2] -= h[idx2 - d]
    return G

def lp_descent_step(h, G, active_idx, radius=0.1):
    """Solve LP to find descent direction."""
    n = len(h)
    # Variables: d (n) + alpha (1) = n+1
    # Minimize alpha + small_reg * sum|d_j|
    # Subject to: G[t,:] · d <= alpha for t in active
    #             sum(d) = 0
    #             -radius <= d_j <= radius
    # 
    # To handle |d_j|, use d_j = pos_j - neg_j, pos,neg >= 0
    # Variables: pos(n) + neg(n) + alpha = 2n+1
    # This doubles the variables. For n=140, that's 281 vars, fine.
    
    # Actually, let's skip the L1 regularization and just use:
    # Variables: d(n) + alpha(1) = n+1
    # Minimize: alpha
    # Subject to: G[t,:] · d - alpha <= 0 for t in active
    #             sum(d) = 0
    #             -radius <= d_j <= radius
    
    # c for linprog: minimize c · x
    # x = [d_0, ..., d_{n-1}, alpha]
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize alpha
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active t: G[t,:] · d - alpha <= 0
    A_ub = np.zeros((len(active_idx), n+1))
    for i, t in enumerate(active_idx):
        A_ub[i, :n] = G[t, :]
        A_ub[i, n] = -1.0
    b_ub = np.zeros(len(active_idx))
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-radius, radius)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
        else:
            return None, None
    except Exception:
        return None, None

    """Compute (2n-1) x n gradient matrix of raw profile."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
