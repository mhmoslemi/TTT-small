     + h[(i-n+1)+t] for t in [n-1-i, 2n-2-i]
     + (1 - h[(j+n-1)-t]) for t in [j+1, j+n-1]
     - h[(j-n+1)+t] for t in [n-1-j, 2n-2-j]

This can be computed in O(n) using numpy slicing.

    delta = np.zeros(2*n - 1)
    # i as first factor: t in [i+1, i+n-1], value = h[i+n-1-t] - 1
    t_start, t_end = i+1, i+n  # exclusive end
    if t_start < 2*n-1 and t_end > 0:
        ts = max(0, t_start)
        te = min(2*n-1, t_end)
        # h[i+n-1-t] for t in [ts, te)
        # i+n-1-t ranges from i+n-1-ts to i+n-1-(te-1)
        # = i+n-1-max(0,i+1) to i+n-1-min(2n-1, i+n)-1
        # Since i < n: i+1 >= 1, so ts = i+1 (for i+1 <= 2n-2, which is always true)
