    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    return p_max + tau * log_sum

def _smooth_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    p = np.correlate(x, 1.0 - x, mode='full') * dx
    p_max = p.max()
    shifted = np.clip((p - p_max) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    weights = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        w = weights[t]
        if w < 1e-14:
            continue
        k = t - (n - 1)
        if k >= 0:
            if k == 0:
                grad[0:n] += w * (1.0 - 2.0 * x[0:n])
            else:
                grad[k:n] += w * (1.0 - x[0:n-k])
                grad[0:n-k] += w * (-x[k:n])
        else:
            m = -k
            grad[0:n-m] += w * (1.0 - x[m:n])
            grad[m:n] += w * (-x[0:n-m])
    grad *= dx
    return grad

def slsqp_polish(h0, tau, maxiter=80):
    n = len(h0)
    targets = n / 2.0
    def obj_and_grad(x):
        v = _smooth_obj(x, tau)
        g = _smooth_grad(x, tau)
        sv = np.sum(x) - targets
        v += 1e4 * sv * sv
        g = g + 2e4 * sv * np.ones(n)
        return v, g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - targets,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_and_grad, h0, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        return project_box_sum(res.x)
    except Exception:
        return None

def multi_start_binary(n, parent, deadline, seed_base=0, max_rounds=2000):
    best_h = None
    best_c = float('inf')
    initial_h = transfer_to_size(parent, n)
    candidates = []
    b0 = np.zeros(n)
    b0[initial_h >= 0.5] = 1.0
    if np.sum(b0) != n // 2:
        idx = np.argsort(-initial_h)
        b0 = np.zeros(n)
        b0[idx[:n//2]] = 1.0
    candidates.append(b0)
    for s in range(1, 6):
        candidates.append(perturb_binary(b0, n_swaps=5 * s, seed=seed_base + s))
    for s in range(6, 9):
        rng = np.random.RandomState(seed_base + s)
        h_rand = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_rand[idx] = 1.0
        candidates.append(h_rand)
    for ci, h_cand in enumerate(candidates):
        if deadline is not None and time.time() > deadline - 60:
            break
        n_rounds = max_rounds
        h_res, c_res = binary_guided_swap(h_cand, max_rounds=n_rounds,
                                          seed=seed_base + ci * 7,
                                          deadline=deadline)
        if c_res < best_c:
            best_c = c_res
            best_h = h_res.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s

        parent = None

    if best_h is None or parent is None:
        n_init = 140
        parent = np.ones(n_init) * 0.5
        parent = project_box_sum(parent)
        best_h = parent.copy()
        best_c = exact_objective(parent)

    sizes = [140, 138, 142, 60, 80, 100, 120, 160, 180, 70, 90, 110, 50, 200]
    seen_i = set()
    ordered_sizes = []
    if parent is not None:
        pn = len(parent)
        ordered_sizes.append(pn)
        seen_i.add(pn)
    for s in sizes:
        if s not in seen_i and s % 2 == 0 and 20 <= s <= 400:
            ordered_sizes.append(s)
            seen_i.add(s)

    for n in ordered_sizes:
        if time.time() > deadline - 120:
            break
        time_left = deadline - time.time()
        if n <= 80:
            max_r = 5000
        elif n <= 140:
            max_r = 3000
        else:
            max_r = 2000
        h_bin, c_bin = multi_start_binary(n, parent, deadline,
                                          seed_base=seed, max_rounds=max_r)
        if h_bin is not None and c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        if time.time() > deadline - 180:
            break
        if h_bin is not None and c_bin < best_c + 0.002:
            h_cont = h_bin.astype(np.float64)
            tau = 0.002
            h_pol = slsqp_polish(h_cont, tau, maxiter=60)
            if h_pol is not None:
                c_pol = exact_objective(h_pol)
                if c_pol < best_c:
                    best_c = c_pol
                    best_h = h_pol.copy()
            h_bin2 = np.zeros(n)
            h_bin2[h_pol >= 0.5] = 1.0
            if np.sum(h_bin2) != n // 2:
                idx = np.argsort(-h_pol)
                h_bin2 = np.zeros(n)
                h_bin2[idx[:n//2]] = 1.0
            h_res2, c_res2 = binary_guided_swap(h_bin2, max_rounds=1500,
                                                seed=seed + 1000, deadline=deadline)
            if c_res2 < best_c:
                best_c = c_res2
                best_h = h_res2.copy()

    if time.time() < deadline - 30 and best_h is not None:
        n_best = len(best_h)
        for tau in [0.001, 0.0005, 0.0002]:
            if time.time() > deadline - 15:
                break
            h_pol = slsqp_polish(best_h.astype(np.float64), tau, maxiter=50)
            if h_pol is not None:
                c_pol = exact_objective(h_pol)
                if c_pol < best_c:
                    best_c = c_pol
                    best_h = h_pol.copy()
        if np.sum(np.abs(best_h - np.round(best_h))) < 0.01 and len(best_h) % 2 == 0:
            h_b = np.round(best_h)
            if np.sum(h_b) == len(best_h) // 2:

                                                    seed=seed + 5000, deadline=deadline)

                    best_c = c_res3
                    best_h = h_res3.copy()

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
