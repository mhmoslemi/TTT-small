import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_profile_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    p = np.correlate(h, 1 - h, mode="full") * dx
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return p, G

def solve_sca_lp(h, p, G, radius):
    """Solve the SCA linear program. Returns proposed z or None."""
    n = len(h)
    M = len(p)
    
    # Variables: z_0, ..., z_{n-1}, t
    # Minimize t
    c = np.concatenate([np.zeros(n), [1.0]])
    
    # Inequality constraints: G[k] @ z - t <= G[k] @ h - p[k]
    # i.e., G[k] @ z - t <= G[k] @ (h) - p[k]
    # Wait: tangent_k(z) = p[k] + G[k] @ (z - h)
    # We want t >= tangent_k(z) for all k
    # i.e., -t + G[k] @ (z - h) <= -p[k]
    # i.e., G[k] @ z - t <= G[k] @ h - p[k]
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - p  # G[k] @ h - p[k] = -G[k] @ (h - h) = 0? No...
    
    # Wait: G[k] @ h - p[k]. Since p[k] = P_k(h) and G[k] is the gradient of P_k at h,
    # G[k] @ h is not necessarily related to p[k] in a simple way.
    # Actually, G[k] @ (z - h) is the linear approximation of P_k(z) - P_k(h).
    # So tangent_k(z) = p[k] + G[k] @ (z - h)
    # t >= tangent_k(z) → t >= p[k] + G[k] @ z - G[k] @ h
    # → -G[k] @ z + t >= p[k] - G[k] @ h
    # → G[k] @ z - t <= G[k] @ h - p[k]
    
    # G[k] @ h - p[k]: Let's compute this.
    # For k=0: G[0] @ h = dx * Σ (1-2h_i) * h_i = dx * Σ h_i - 2 dx * Σ h_i^2
    # p[0] = dx * Σ h_i(1-h_i) = dx * Σ h_i - dx * Σ h_i^2
    # G[0] @ h - p[0] = dx * Σ h_i - 2 dx * Σ h_i^2 - dx * Σ h_i + dx * Σ h_i^2 = -dx * Σ h_i^2
    # Which is negative (as expected, since the tangent is below the quadratic at the center)
    
    b_ub = G @ h - p
    
    # Trust region bounds
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi)) + [(0.0, None)]
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and np.isfinite(res.x).all():
            z = res.x[:n]
            t = res.x[n]
            if np.all(z >= -1e-10) and np.all(z <= 1 + 1e-10):
                return z, float(t)
    except Exception:
        pass
    return None, None

def sca_refine(h, max_iters=200, initial_radius=0.3, min_radius=1e-5):
    """Successive Convex Approximation with LP and line search."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    radius = initial_radius
    stagnation = 0
    
    for it in range(max_iters):
        p, G = build_profile_gradient(h)
        z_prop, t_prop = solve_sca_lp(h, p, G, radius)
        
        if z_prop is None:
            radius *= 0.5
            stagnation += 1
            if stagnation > 10:
                break
            continue
        
        # Symmetrize
        z_sym = (z_prop + z_prop[::-1]) / 2.0
        z_sym = repair_feasible(z_sym)

        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.025, 0.01]:
            candidate = h + alpha * (z_sym - h)
            candidate = repair_feasible(candidate)
            c = exact_score(candidate)
            if c < best_c - 1e-12:
                h = candidate
                best_c = c
                improved = True
                break
        
        if improved:
            radius = min(radius * 1.3, 1.0)
            stagnation = 0
        else:
            radius *= 0.6
            stagnation += 1
            if stagnation > 15:

                            improved = True

                        radius = min(radius * 1.5, 0.5)

            break
    
    return h, best_c

def active_set_descent(h, max_iters=100):
    """Subgradient descent on active set with line search."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    
    for _ in range(max_iters):
        p, G = build_profile_gradient(h)
        c = np.max(p)
        active_thresh = c - 1e-5
        active = np.where(p >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(p)])
        
        # Weighted average of active gradients
        w = p[active] - active_thresh + 1e-10
        w = w / w.sum()
        grad = G[active, :].T @ w
        grad -= grad.mean()  # project onto sum=0
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = h - alpha * grad
            h_try = repair_feasible(h_try)
            s = exact_score(h_try)
            if s < best_c - 1e-12:
                h = h_try
                best_c = s
                break
    
    return h, best_c

    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_profile_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    p = np.correlate(h, 1 - h, mode="full") * dx
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return p, G

def solve_sca_lp(h, p, G, radius):
    n = len(h)
    M = len(p)
    c_obj = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - p
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi)) + [(0.0, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and np.isfinite(res.x).all():
            z = res.x[:n]
            if np.all(z >= -1e-10) and np.all(z <= 1.0 + 1e-10):
                return z
    except Exception:
        pass
    return None

def sca_refine(h, max_iters=200, initial_radius=0.25, min_radius=1e-5, rng=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    radius = initial_radius
    stagnation = 0
    
    for it in range(max_iters):
        p, G = build_profile_gradient(h)
        z_prop = solve_sca_lp(h, p, G, radius)
        
        if z_prop is None:
            radius *= 0.5
            stagnation += 1
            if stagnation > 12:
                break
            continue
        
        z_sym = (z_prop + z_prop[::-1]) / 2.0
        z_sym = repair_feasible(z_sym)
        
        improved = False
        best_candidate = None
        best_candidate_c = best_c
        alphas = [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02]
        for alpha in alphas:
            candidate = h + alpha * (z_sym - h)
            candidate = repair_feasible(candidate)
            c_val = exact_score(candidate)
            if c_val < best_candidate_c:
                best_candidate_c = c_val
                best_candidate = candidate
            if c_val < best_c - 1e-13:
                break
        
        if best_candidate is not None and best_candidate_c < best_c - 1e-13:
            h = best_candidate
            best_c = best_candidate_c
            improved = True
            radius = min(radius * 1.4, 0.8)
            stagnation = 0
        else:
            radius *= 0.55
            stagnation += 1
            if stagnation > 8 and rng is not None:
                noise_scale = min(radius * 0.3, 0.05)
                z_noise = rng.randn(n) * noise_scale
                z_noise -= z_noise.mean()

                        stagnation = 0

def active_set_descent(h, max_iters=100):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    
    for _ in range(max_iters):
        p, G = build_profile_gradient(h)
        c_val = np.max(p)
        active_thresh = c_val - 1e-6
        active = np.where(p >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(p)])
        
        w = p[active] - active_thresh + 1e-10
        w = w / w.sum()
        grad = G[active, :].T @ w
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        
        found = False
        for alpha in [0.04, 0.025, 0.015, 0.01, 0.006, 0.003, 0.001]:
            h_try = h - alpha * grad
            h_try = repair_feasible(h_try)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                found = True
                break
        if not found:
            # Try symmetric gradient
            grad_sym = (grad + grad[::-1]) / 2.0
            grad_sym -= grad_sym.mean()
            ns = np.linalg.norm(grad_sym)
            if ns < 1e-14:
                break
            grad_sym /= ns

                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-13:
                    h = h_try
                    best_c = s
                    found = True
                    break
            if not found:

    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def penalty_refine(h, max_iters=60, c_target=None, rng=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    if c_target is None:
        c_target = best_c - 0.001
    
    for _ in range(max_iters):
        p, G = build_profile_gradient(h)
        # Penalty: sum of max(0, p_k - c_target)^2
        excess = np.maximum(0, p - c_target)
        grad_pen = G.T @ (2.0 * excess)
        grad_pen -= grad_pen.mean()
        norm = np.linalg.norm(grad_pen)
        if norm < 1e-14:
            # Try reducing target
            max_p = np.max(p)
            if max_p > c_target:
                c_target = max_p - 0.0005
            else:
                c_target -= 0.0005
            continue
        grad_pen /= norm
        found = False
        for alpha in [0.03, 0.02, 0.012, 0.008, 0.004, 0.002]:
            h_try = h - alpha * grad_pen
            h_try = repair_feasible(h_try)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                found = True
                break
        if not found:
            c_target = max(c_target - 0.001, best_c - 0.01)
    
    return h, best_c
