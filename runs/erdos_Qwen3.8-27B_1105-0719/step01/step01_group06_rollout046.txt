def build_gradient(h):
    """Build the gradient matrix G of shape (2n-1, n) where
    G[k, i] = d/dh[i] of profile[k].
    profile[k] = dx * sum_j h[j] * (1-h[j+k-(n-1)]) (with valid indices only)

    M = 2 * n - 1

            idx1 = np.arange(0, L)       # indices of h[i]
            idx2 = np.arange(c, c + L)   # indices of h[i+c]
            G[m, idx1] += dx * (1.0 - h[idx2])
            G[m, idx2] -= dx * h[idx1]
    
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: add to positions with most room
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        # Need to remove mass: remove from positions with most value
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

    minimize theta
    subject to: p[k] + G[k,:]·(x - h) ≤ theta for all k
                sum(x) = n/2
                max(0, h[i]-r) ≤ x[i] ≤ min(1, h[i]+r)
    
    Returns: (x_opt, theta_opt, success)
    """

    M = 2 * n - 1

    G = build_gradient(h)

        active = np.argsort(prof)[::-1][:max_lags]
        G_sel = G[active, :]
        p_sel = prof[active]
        # Also add any lags with profile close to max
        threshold = prof[active[-1]] + 1e-6
        extra = np.where((prof >= threshold) & (np.arange(M) not in active_set))[0]
        ...
    else:

        p_sel = prof

    # Variables: x[0..n-1], theta (index n)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize theta
    
    # Inequality: G_sel[k,:]·x - theta ≤ G_sel[k,:]·h - p_sel[k]
    A_ub = np.hstack([G_sel, -np.ones((len(G_sel), 1))])
    b_ub = G_sel @ h - p_sel
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # theta is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.success:

Hmm, but there's an issue. The LP is minimizing θ, and the constraint is G_sel[k,:]·(x-h) + p_sel[k] ≤ θ. At x=h, this gives θ ≥ max(p_sel). So the LP will always have θ ≥ max(p_sel) at the current point. The LP is looking for a point x (within the trust region) where the linearized profile has a smaller maximum.

If the LP returns x = h (no change), it means the current point is a local minimum of the linearized problem, and we need to either:
1. Increase the trust radius
2. Use a different method (gradient descent, SLSQP)
3. Try a different starting point

Let me also add a small perturbation to the objective to break ties:
c = ε * (x - h) + [0, ..., 0, 1]

This encourages the LP to move away from h when the objective is flat.

Actually, a better approach: add a small regularization to push toward "bimodal" values:
c[i] = -β * (2*h[i] - 1) for i = 0, ..., n-1 (pushes toward 0 or 1)
c[n] = 1

Wait, that's not quite right. If h[i] > 0.5, we want to push it toward 1, so the gradient of the "bimodality penalty" is positive (increasing h[i] reduces the penalty). If h[i] < 0.5, we want to push toward 0, so the gradient is negative.

The bimodality penalty is: β * sum(h[i] * (1-h[i])) = β * (sum(h[i]) - sum(h[i]^2))
The gradient w.r.t. h[i] is: β * (1 - 2*h[i])

So the objective coefficient for x[i] should be β * (1 - 2*h[i]) to minimize this penalty.

But wait, I'm MINIMIZING θ + β * sum(h[i]*(1-h[i])). The LP can only handle linear objectives. The term sum(h[i]*(1-h[i])) is quadratic, not linear.

So I can't directly add bimodality to the LP. But I can add a linear approximation:
At point h, the linearization of sum(h[i]*(1-h[i])) is:
sum(h[i]*(1-h[i])) + sum((1-2*h[i]) * (x[i] - h[i]))

So the linearized objective is:
θ + β * [sum(h[i]*(1-h[i])) + sum((1-2*h[i]) * (x[i] - h[i]))]
= θ + β * sum((1-2*h[i]) * x[i]) + const

The constant doesn't matter for the LP, so the objective coefficients are:
c[i] = β * (1 - 2*h[i]) for i = 0, ..., n-1
c[n] = 1

This is a valid linear objective! And it encourages bimodality in a first-order sense.

def gradient_step(h, step_size):
    """Take a gradient descent step on the max of the profile."""
    n = len(h)
    prof = get_profile(h)
    G = build_gradient(h)
    
    # Weighted average of active constraint gradients
    A = max(5, min(30, n // 4))
    active = np.argsort(prof)[::-1][:A]
    w = prof[active] - prof[active[-1]]  # relative weights
    w = np.maximum(w, 0)
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.ones(A) / A
    else:
        w /= wsum
    
    grad = w @ G[active, :]  # weighted gradient
    grad -= grad.mean()  # zero-mean (to preserve sum constraint)
    
    h_new = h - step_size * grad
    return repair_feasible(h_new)
