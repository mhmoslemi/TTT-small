def solve_qup(h, active_idx, R):
    n = len(h)
    profile = exact_profile(h)
    G = gradient_matrix(h)
    
    z = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(z) == n / 2.0,
        z >= 0,
        z <= 1,
        cp.norm(z - h, 2) <= R,
    ]
    for idx in active_idx:
        G_active = G[idx]
        p_val = profile[idx]
        # Linearized: p_val + G_active @ (z - h) <= t
        # = (p_val - G_active @ h) + G_active @ z <= t
        constraints.append(p_val - G_active @ h + G_active @ z <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try available solvers
    solved = False
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue
    if not solved:
        try:
            prob.solve()
            solved = True
        except Exception:
            return None
    
    if z.value is None:
        return None
    return project_box_sum(np.array(z.value, dtype=np.float64))

def binary_swap_search(b, deadline, max_rounds=200):

    zeros = set(np.where(b == 0.0)[0].tolist())

        else:

    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def binary_swap_search(b, deadline, max_rounds=200):
    n = len(b)
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        best_improve = 1e-13
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
                b[i] = 1.0
                b[j] = 0.0
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            break
    return b, curr_c

def pair_swap_continuous(h, deadline, max_rounds=50):
    """Try pairwise continuous swap: h[i]+delta, h[j]-delta preserving sum."""
    n = len(h)
    curr_c = exact_objective(h)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        best_improve = 1e-13
        best_move = None
        # Sample a subset of pairs for speed
        pairs_to_try = []
        for i in range(0, n, max(1, n // 24)):
            for j in range(0, n, max(1, n // 24)):
                if i != j:
                    pairs_to_try.append((i, j))
        if not pairs_to_try:
            pairs_to_try = [(0, 1)]
        for i, j in pairs_to_try:
            for delta in [0.05, 0.1, 0.15, -0.05, -0.1, -0.15]:
                if h[i] + delta < -0.01 or h[i] + delta > 1.01:
                    continue
                if h[j] - delta < -0.01 or h[j] - delta > 1.01:
                    continue
                h[i] += delta
                h[j] -= delta
                h_p = project_box_sum(h)
                c = exact_objective(h_p)
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_move = (i, j, delta, h_p.copy())
                h[i] -= delta
                h[j] += delta
        if best_move is not None:
            i, j, d, h_new = best_move
            h = h_new
            curr_c = exact_objective(h)
        else:
            break
    return h, curr_c

def make_starts(n, rng):
    starts = []
    x = (np.arange(n) + 0.5) * 2.0 / n
    # Sine waves
    for freq in [2, 3, 4, 5, 6, 7]:
        for phase in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0]:
            h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x + phase)
            starts.append(project_box_sum(np.clip(h, 0, 1)))
    # Block patterns
    h = np.zeros(n)
    block = max(1, n // 8)
    for i in range(0, n, 2 * block):
        h[i:i + block] = 1.0
    starts.append(project_box_sum(h))
    # Two blocks
    h = np.zeros(n)
    h[:n // 4] = 1.0
    h[3 * n // 4:] = 1.0
    starts.append(project_box_sum(h))
    # Random
    h = rng.uniform(0, 1, n)
    starts.append(project_box_sum(h))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Load parent
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            best_c = pc
            best_h = parent_h.copy()
            best_n = len(parent_h)
    except Exception:
        pass
    
    if best_h is None:
        n0 = 48
        best_h = make_starts(n0, rng)[0]
        best_c = exact_objective(best_h)
        best_n = n0
    
    parent_n = len(parent_h) if parent_h is not None else best_n
    
    # Sizes to explore
    size_candidates = set()
    size_candidates.add(parent_n)
    for s in [32, 36, 40, 44, 48, 52, 56, 64, 72, 80, 88, 96, 112, 128]:
        if 16 <= s <= 160:
            size_candidates.add(s)
    # Prefer parent size and nearby sizes first
    sizes_ordered = sorted(size_candidates, key=lambda s: (abs(s - parent_n), s))
    
    # Phase 1: QP refinement at multiple sizes
    for n in sizes_ordered:
        if time.time() > deadline - 35:
            break
        remaining = deadline - time.time()
        if remaining < 20:
            break
        
        # Get starting point
        if parent_h is not None:
            h_start = transfer_to_size(parent_h, n)
        else:
            h_start = make_starts(n, rng)[0]
        
        # Primary: QP refinement
        alloc = min(remaining * 0.5, 120.0)
        sub_deadline = min(deadline - 20, t_start + alloc)
        if sub_deadline > time.time() + 5:
            h_ref, c_ref = refine_qup(h_start, sub_deadline, max_rounds=25)
            if c_ref < best_c - 1e-10:
                best_c = c_ref
                best_h = h_ref.copy()
                best_n = n
        
        # Try a perturbed start
        if time.time() < deadline - 30:
            h_pert = project_box_sum(h_start + rng.normal(0, 0.04, n))
            sub_deadline2 = min(deadline - 15, time.time() + 30)
            if sub_deadline2 > time.time() + 5:
                h_ref2, c_ref2 = refine_qup(h_pert, sub_deadline2, max_rounds=15)
                if c_ref2 < best_c - 1e-10:
                    best_c = c_ref2
                    best_h = h_ref2.copy()
                    best_n = n
    
    # Phase 2: Binary refinement
    if time.time() < deadline - 15:
        bin_sizes = [best_n]
        for s in [48, 56, 64, 72, 80, 96]:
            if s != best_n and s % 2 == 0:
                bin_sizes.append(s)
        for n_b in bin_sizes:
            if time.time() > deadline - 10:
                break
            if n_b % 2 != 0:
                n_b -= 1
            if n_b < 16:
                continue
            h_src = transfer_to_size(best_h, n_b) if len(best_h) != n_b else best_h
            b = binary_threshold(h_src)
            if b is not None:
                sub_dl = min(deadline - 5, time.time() + 40)
                b_sol, c_b = binary_swap_search(b, sub_dl, max_rounds=120)
                if c_b < best_c - 1e-10:
                    best_c = c_b
                    best_h = b_sol.copy()
                    best_n = n_b
    
    # Phase 3: Pair-swap continuous refinement on best
    if time.time() < deadline - 10:
        h_ps, c_ps = pair_swap_continuous(best_h.copy(), deadline, max_rounds=30)
        if c_ps < best_c - 1e-10:
            best_c = c_ps
            best_h = h_ps.copy()
            best_n = len(h_ps)
    
    # Phase 4: Final QP polish
    if time.time() < deadline - 6:
        h_fin, c_fin = refine_qup(best_h, deadline, max_rounds=15)
        if c_fin < best_c - 1e-10:
            best_c = c_fin
            best_h = h_fin.copy()
            best_n = len(h_fin)
    
    # Final feasibility check
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)
    
    if not np.all(np.isfinite(best_h)):
        best_h = project_box_sum(np.clip(best_h, 0, 1))
        best_c = exact_objective(best_h)
        best_n = len(best_h)
    
    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)
    
    return best_h, best_c, best_n
