```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add 'diff' total, spread to entries with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff2)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        s = m - (n - 1)
        if s == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            G[m, 0:n-s] += dx * (1.0 - h[s:n])
            G[m, s:n] += -dx * h[0:n-s]
        else:
            a = -s
            G[m, a:n] += dx * (1.0 - h[0:n-a])
            G[m, 0:n-a] += -dx * h[a:n]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0
```

```python
def slp_propose(h, r):
    """Solve the linearized epigraph LP to propose a new point."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    G = build_gradient_matrix(h, n, dx)
    
    # Variables: z_0, ..., z_{n-1}, t (total n+1 variables)
    # Minimize t
    # Subject to:
    #   G[m] @ z - t <= G[m] @ h - profile[m] for all m
    #   sum(z) = n/2
    #   max(0, h_i - r) <= z_i <= min(1, h_i + r)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: G[m] @ z - t <= G[m]@h - profile[m]
    # A_ub @ x <= b_ub
    A_ub = np.zeros((M, n + 1))
    b_ub = np.zeros(M)
    for m in range(M):
        A_ub[m, :n] = G[m]
        A_ub[m, -1] = -1.0
        b_ub[m] = G[m] @ h - profile[m]
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lo = np.maximum(0.0, h - r)
    hi = np.minimum(1.0, h + r)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, profile.max() + 2.0))  # t bound
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n]
    return None
```

```python
def line_search(h_cur, h_prop, c_cur):
    """Exact line search from h_cur toward h_prop."""
    alphas = [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.025, 0.01]
    best_c = c_cur
    best_h = h_cur.copy()
    for alpha in alphas:
        candidate = h_cur + alpha * (h_prop - h_cur)
        candidate = repair_feasible(candidate)
        c = exact_score(candidate)
        if c < best_c - 1e-12:
            best_c = c
            best_h = candidate.copy()
    return best_h, best_c
```

```python
def slp_refine(h0, c0, max_iter=60, initial_r=0.2, min_r=1e-4, deadline=None):
    """Run SLP trust-region refinement."""
    h = h0.copy()
    c = c0
    r = initial_r
    stagnation = 0
    
    for it in range(max_iter):
        if deadline is not None and (deadline - time.perf_counter()) < 5:
            break
        
        h_prop = slp_propose(h, r)
        if h_prop is None:
            r *= 0.5
            stagnation += 1
            if stagnation > 8 or r < min_r:
                break
            continue
        
        h_new, c_new = line_search(h, h_prop, c)
        
        if c_new < c - 1e-11:
            h = h_new
            c = c_new
            stagnation = 0
            # Maybe increase r slightly if we made good progress
            if it % 5 == 0 and r < 0.3:
                r = min(r * 1.2, 0.35)
        else:
            r *= 0.7
            stagnation += 1
            if stagnation > 10 or r < min_r:
                break
    
    return h, c
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profil = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(profil))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64).ravel(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff2)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        s = m - (n - 1)
        if s == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            G[m, 0:n-s] += dx * (1.0 - h[s:n])
            G[m, s:n] += -dx * h[0:n-s]
        else:
            a = -s
            G[m, a:n] += dx * (1.0 - h[0:n-a])
            G[m, 0:n-a] += -dx * h[a:n]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def slp_propose(h, r):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    G = build_gradient_matrix(h, n, dx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    b_ub = np.zeros(M)
    for m in range(M):
        A_ub[m, :n] = G[m]
        A_ub[m, -1] = -1.0
        b_ub[m] = G[m] @ h - profile[m]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - r)
    hi = np.minimum(1.0, h + r)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, profile.max() + 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_prop = res.x[:n]
            if np.all(np.isfinite(z_prop)):
                return z_prop
    except Exception:
        pass
    return None

def line_search(h_cur, h_prop, c_cur):
    alphas = [1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01]
    best_c = c_cur
    best_h = h_cur.copy()
    for alpha in alphas:
        candidate = h_cur + alpha * (h_prop - h_cur)
        candidate = repair_feasible(candidate)
        c = exact_score(candidate)
        if c < best_c - 1e-12:
            best_c = c
            best_h = candidate.copy()
    return best_h, best_c

def slp_refine(h0, c0, max_iter=80, initial_r=0.15, min_r=5e-5, deadline_fn=None):
    h = h0.copy()
    c = c0
    r = initial_r
    stagnation = 0
    for it in range(max_iter):
        if deadline_fn is not None and deadline_fn() < 5.0:
            break
        h_prop = slp_propose(h, r)
        if h_prop is None:
            r *= 0.5
            stagnation += 1
            if stagnation > 10 or r < min_r:
                break
            continue
        h_new, c_new = line_search(h, h_prop, c)
        if c_new < c - 5e-13:
            h = h_new
            c = c_new
            stagnation = 0
            if it % 8 == 0 and r < 0.25:
                r = min(r * 1.15, 0.30)
        else:
            r *= 0.65
            stagnation += 1
            if stagnation > 12 or r < min_r:
                break
    return h, c

def slp_refine_multi(h0, c0, radii, deadline_fn=None):
    h = h0.copy()
    c = c0
    for r in radii:
        if deadline_fn is not None and deadline_fn() < 8.0:
            break
        h, c = slp_refine(h, c, max_iter=40, initial_r=r, min_r=r * 0.05, deadline_fn=deadline_fn)
    return h, c

def generate_wave(n, freq, amp, phase=0.0):
    x = (np.arange(n) + 0.5) / n
    vals = 0.5 + amp * np.cos(2 * np.pi * freq * x + phase)
    return repair_feasible(np.clip(vals, 0.0, 1.0))

def simulated_annealing(h0, c0, n_steps=2000, T0=0.01, T_end=0.0005, deadline_fn=None):
    h = h0.copy()
    c = c0
    n = len(h)
    best_h = h.copy()
    best_c = c
    for step in range(n_steps):
        if deadline_fn is not None and deadline_fn() < 5.0:
            break
        T = T0 * (T_end / T0) ** (step / max(1, n_steps))
        noise_scale = 0.01 + 0.02 * T / T0
        z = np.random.randn(n) * noise_scale
        z -= z.mean()
        candidate = repair_feasible(h + z)
        c_new = exact_score(candidate)
        delta = c_new - c
        if delta <= 0:
            h = candidate
            c = c_new
            if c < best_c:
                best_h = candidate.copy()
                best_c = c
        else:
            accept_prob = np.exp(-delta / max(T, 1e-10))
            if np.random.random() < accept_prob:
                h = candidate
                c = c_new
    return best_h, best_c
```
