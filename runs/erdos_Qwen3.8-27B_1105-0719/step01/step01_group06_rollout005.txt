def repair_feasible(h, target=None):

        target = n / 2.0

    if abs(diff) < 1e-14:
        return h

        room = 1.0 - h
        order = np.argsort(-room)

        for idx in order:
            if remaining <= 1e-15:
                break

            h[idx] += add
            remaining -= add

            remaining -= rem
