def profile_gradient(h, t, n):
    """Gradient of profile entry t with respect to h."""
    dx = 2.0 / n
    d = (n - 1) - t
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        # h[i]*(1-h[i+d]) for i=0..n-1-d
        grad[0:n-d] += (1.0 - h[d:n])
        grad[d:n] -= h[0:n-d]
    else:
        a = -d
        # h[i]*(1-h[i-a]) for i=a..n-1
        grad[a:n] += (1.0 - h[0:n-a])
        grad[0:n-a] -= h[a:n]
    grad *= dx
    return grad
```

        target = n / 2.0

    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to decrease sum
            mask = h > 1e-15
            if not mask.any():
                break

def binary_swap_improve(h, max_rounds=100):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    profile = np.correlate(h, 1-h, 'full')
    c = profile.max() * dx
    
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            for j in zeros:
                # Try swap
                h[i] = 0.0
                h[j] = 1.0
                new_profile = np.correlate(h, 1-h, 'full')
                new_c = new_profile.max() * dx
                if new_c < c - 1e-12:
                    c = new_c
                    improved = True
                    # Don't break - keep trying other swaps
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    return h, c
```

    n = len(h)
    dx = 2.0 / n
    d = (n - 1) - t

        grad[:] = 1.0 - 2.0 * h
    elif d > 0:

        grad[d:n] -= h[0:n-d]
    else:

        grad[a:n] += (1.0 - h[0:n-a])
        grad[0:n-a] -= h[a:n]
    grad *= dx
    return grad
