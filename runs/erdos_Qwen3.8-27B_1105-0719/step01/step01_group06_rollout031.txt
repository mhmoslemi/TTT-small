    n = len(h0)
    M = 2*n - 1
    dx = 2.0/n
    t0 = time.perf_counter()
    best_h = h0.copy()
    best_score = exact_score(h0)
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        
        h = best_h
        profile = get_profile(h)
        score = float(np.max(profile))
        G = build_gradient_matrix(h)
        
        # Build LP
        # Variables: [x_0, ..., x_{n-1}, theta], total n+1 vars
        c_obj = np.zeros(n+1)
        c_obj[n] = 1.0  # minimize theta

        A_ub = np.zeros((M, n+1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = profile - G @ h  # = profile - G @ h0 (since h = h0 at this point)

        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        # Bounds
        lb = np.maximum(0, h - r)
        ub = np.minimum(1, h + r)
        bounds = list(zip(lb, ub)) + [(None, None)]

import numpy as np
import time
from scipy.optimize import linprog


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for d in range(0, n):
        m = n - 1 - d
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        else:
            G[m, 0:n - d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n - d]
        m2 = n - 1 + d
        if d > 0:
            G[m2, 0:n - d] += -dx * h[d:n]
            G[m2, d:n] += dx * (1.0 - h[0:n - d])
    return G


def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h


def interpolate_to_dim(h, new_len):
    n = len(h)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def lp_trust_region(h0, r_init, time_limit, r_min=0.002):
    n = len(h0)
    M = 2 * n - 1
    t0 = time.perf_counter()
    best_h = h0.copy()
    best_score = exact_score(h0)
    r = r_init
    stall = 0
    max_stall = 20

    while time.perf_counter() - t0 < time_limit - 1.0 and r > r_min and stall < max_stall:
        if time.perf_counter() - t0 > time_limit - 2.0:
            break

        h = best_h
        profile = get_profile(h)
        G = build_gradient_matrix(h)

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = profile - G @ h

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        lb = np.maximum(0.0, h - r)
        ub = np.minimum(1.0, h + r)
        if lb.sum() > n / 2.0 + 1e-9 or ub.sum() < n / 2.0 - 1e-9:
            r *= 0.5
            stall += 1
            continue

        bounds = [(float(lb[i]), float(ub[i])) for i in range(n)] + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'max_iter': 20000})
        except Exception:
            r *= 0.5
            stall += 1
            continue

        if res.status != 0:
            r *= 0.5
            stall += 1
            continue

        x_opt = res.x[:n]
        step = x_opt - h
        step_norm = np.linalg.norm(step)

        if step_norm < 1e-10:
            r *= 0.5
            stall += 1
            continue

        improved = False
        for alpha in [1.0, 0.85, 0.7, 0.55, 0.4, 0.28, 0.18, 0.1, 0.05]:
            x_trial = repair_feasible(h + alpha * step)
            s_trial = exact_score(x_trial)
            if s_trial < best_score - 1e-13:
                best_h = x_trial
                best_score = s_trial
                improved = True
                r = min(0.45, r * 1.12)
                stall = 0
                break

        if not improved:
            active_idx = np.argsort(profile)[::-1][:min(25, M)]
            active_prof = profile[active_idx]
            w = np.maximum(active_prof - active_prof.min(), 0)

            if wsum < 1e-12:
                w = np.ones(len(active_idx)) / len(active_idx)

                        best_h = x_trial
                        best_score = s_trial
                        improved = True
                        stall = 0
                        break

        if not improved:
            r *= 0.6
            stall += 1

    return best_h, best_score


def subgradient_descent(h0, time_limit, step_scale=0.01):
    n = len(h0)
    M = 2 * n - 1
    t0 = time.perf_counter()
    best_h = h0.copy()
    best_score = exact_score(h0)
    step_size = step_scale

    while time.perf_counter() - t0 < time_limit - 1.0:
        h = best_h
        profile = get_profile(h)
        score = float(np.max(profile))
        if score < best_score:
            best_score = score
            best_h = h.copy()

        active_threshold = score - 0.005
        active_idx = np.where(profile >= active_threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(profile)[::-1][:5]

        G = build_gradient_matrix(h)
        subgrad = np.mean(G[active_idx, :], axis=0)
        subgrad -= subgrad.mean()
        ng = np.linalg.norm(subgrad)
        if ng < 1e-12:
            break
        subgrad /= ng

        improved = False
        for scale in [step_size, step_size * 0.5, step_size * 0.2, step_size * 0.05]:
            x_trial = repair_feasible(h - scale * subgrad)
            s_trial = exact_score(x_trial)
            if s_trial < best_score - 1e-13:
                best_h = x_trial
                best_score = s_trial
                step_size = min(0.05, step_size * 1.1)
                improved = True
                break

        if not improved:
            step_size *= 0.7
            if step_size < 1e-5:
                break

    return best_h, best_score


def make_symmetric(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)


def make_antisymmetric(h):
    h_anti = h[::-1].copy()
    return repair_feasible(h_anti)


def make_complement(h):
    h_comp = 1.0 - h
    return repair_feasible(h_comp)


def make_blend(h, w):
    h_blend = w * h + (1.0 - w) * (1.0 - h[::-1].copy())
    return repair_feasible(h_blend)


def make_oscillatory(n, freq, phase):

    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x + phase)
    return repair_feasible(h)
