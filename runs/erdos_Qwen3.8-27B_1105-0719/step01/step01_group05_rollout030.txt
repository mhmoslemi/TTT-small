            if improved:

import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def profile_and_grad_lag(h, lag_idx, n):
    """Compute profile value and gradient contribution for a specific lag."""
    dx = 2.0 / n
    n_minus_1 = n - 1
    lag = lag_idx - n_minus_1
    if lag >= 0:
        s = n - lag
        if s <= 0:
            return 0.0, np.zeros(n)
        # profile = sum_{i=0}^{s-1} h[i+lag] * (1-h[i])
        val = np.sum(h[lag:lag+s] * (1 - h[:s]))
        grad = np.zeros(n)
        grad[lag:lag+s] += (1 - h[:s])  # derivative w.r.t. h[i+lag]
        grad[:s] -= h[lag:lag+s]  # derivative w.r.t. h[i]
    else:
        a = -lag
        s = n - a
        if s <= 0:
            return 0.0, np.zeros(n)
        # profile = sum_{i=0}^{s-1} h[i] * (1-h[i+a])
        val = np.sum(h[:s] * (1 - h[a:a+s]))
        grad = np.zeros(n)
        grad[:s] += (1 - h[a:a+s])  # derivative w.r.t. h[i]
        grad[a:a+s] -= h[:s]  # derivative w.r.t. h[i+a]
    return val * dx, grad * dx

def smooth_max_profile(h, tau):
    """Smooth approximation of max profile using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    pmax = np.max(profile)
    if tau < 1e-14:
        return pmax
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    return pmax + tau * np.log(np.sum(exp_vals))

def smooth_max_grad(h, tau):
    """Gradient of the smooth max profile."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    pmax = np.max(profile)
    if tau < 1e-14:
        idx = np.argmax(profile)
        _, grad = profile_and_grad_lag(h, idx, n)
        return grad
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    active = weights > 1e-10
    for m in np.where(active)[0]:
        _, g = profile_and_grad_lag(h, m, n)
        grad += weights[m] * g
    return grad

def project_box_sum(h, target):
    n = len(h)
    h = np.clip(np.array(h, dtype=np.float64), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0: break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0: break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return h

def obj_and_grad(h, tau, target):
    n = len(h)
    val = smooth_max_profile(h, tau)
    grad = smooth_max_grad(h, tau)
    # Add penalty for constraint violation
    se = np.sum(h) - target
    val += 1e4 * se * se
    grad += 2e4 * se * np.ones(n)
    return val, grad

def slsqp_optimize(h_start, tau_list, deadline, target=None):
    n = len(h_start)
    if target is None:
        target = 0.5 * n
    best_h = project_box_sum(h_start, target)
    best_c = exact_objective(best_h)
    bounds = [(0.0, 1.0)] * n
    
    def eq_fun(x): return np.sum(x) - target
    def eq_jac(x): return np.ones(n)
    constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    
    for tau in tau_list:
        if time.time() > deadline - 5:
            break
        def og(x):
            return obj_and_grad(x, tau, target)
        try:
            maxiter = min(300, 3 * n + 50)
            res = minimize(og, best_h, method='SLSQP', bounds=bounds,
                         constraints=constraint, jac=True,
                         options={'maxiter': maxiter, 'ftol': 1e-15})
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                c = exact_objective(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
        except:
            pass
    return best_h, best_c

def binary_local_search(b, max_rounds=200, deadline=None):
    n = len(b)
    b = b.copy()
    c5 = exact_objective(b)
    for rnd in range(max_rounds):
        if deadline and time.time() > deadline - 5:
            break
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                nc = exact_objective(b)
                if nc < c5 - 1e-12:
                    c5 = nc
                    improved = True
                    ones = list(np.where(b > 0.5)[0])
                    zeros = list(np.where(b < 0.5)[0])
                    break
                else:
                    b[i] = 1.0
                    b[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return b, c5

def greedy_binary_construct(n, rng, n_restarts=20):
    """Greedy construction of binary sequence minimizing max profile."""
    best_b = None
    best_c = float('inf')
    target = n // 2
    
    for restart in range(n_restarts):
        b = np.zeros(n)
        remaining = list(range(n))
        rng.shuffle(remaining)
        
        for _ in range(target):
            best_pos = remaining[0]
            best_val = float('inf')
            # Try a subset of positions for speed
            candidates = remaining[:min(20, len(remaining))]
            for pos in candidates:
                b[pos] = 1.0
                c = exact_objective(b)
                b[pos] = 0.0
                if c < best_val:
                    best_val = c
                    best_pos = pos
            b[best_pos] = 1.0
            remaining.remove(best_pos)
        
        c = exact_objective(b)
        if c < best_c:
            best_c = c
            best_b = b.copy()
    
    return best_b, best_c

def continuous_refine_from_binary(b, deadline, target=None):
    """Refine a binary solution in the continuous space."""
    n = len(b)
    if target is None:
        target = 0.5 * n
    
    # Start with small perturbations from binary
    best_h = b.copy()
    best_c = exact_objective(b)
    
    tau_list = [0.01, 0.005, 0.001, 0.0003, 0.0001]
    h_opt, c_opt = slsqp_optimize(b, tau_list, deadline, target)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    
    # Try perturbations
    for pert_scale in [0.05, 0.1, 0.2]:
        if time.time() > deadline - 10:
            break
        h_pert = b + np.random.randn(n) * pert_scale
        h_pert = project_box_sum(h_pert, target)
        h_opt, c_opt = slsqp_optimize(h_pert, tau_list[:3], deadline, target)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c
