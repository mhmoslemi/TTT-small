from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break

        else:
            idx = int(np.argmax(h))

        if i == j:

        c_ij = h[i] + h[j]
        lo = max(0.0, c_ij - 1.0)
        hi = min(1.0, c_ij)
        if hi - lo < 1e-12:
            continue
        # Grid search
        grid = np.linspace(lo, hi, use_grid)
        best_val = c
        best_x = h[i]
        for x in grid:
            h[i] = x
            h[j] = c_ij - x
            # Check bounds
            if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                continue
            val = exact_score(h)
            if val < best_val:
                best_val = val
                best_x = x
        h[i] = best_x
        h[j] = c_ij - best_x
        if best_val < c - 1e-12:
            c = best_val
            n_stale = 0
        else:
            n_stale += 1
            if n_stale > 2000:
                break
        h = np.clip(h, 0.0, 1.0)
    
    h = repair_feasible(h)
    c = exact_score(h)
    return h, c

Term 1 is constant (doesn't depend on x).

Term 2: let S2 = Σ_{k: k∉{i,j}, k+s=i} h_k. Then Term 2 = S2 * (1-x) = S2 - S2*x.
Term 3: let S3 = Σ_{k: k∉{i,j}, k+s=j} h_k. Then Term 3 = S3 * (1-c+x) = S3(1-c) + S3*x.

- If i+s invalid: 0
