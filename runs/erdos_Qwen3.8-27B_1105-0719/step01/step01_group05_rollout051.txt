        else:
            s = -e
            for i in range(n - s):
                G[r, i] += dx * (1.0 - h[i + s])
                G[r, i + s] += -dx * h[i]
    return G

def slsqp_hard(h0, deadline, maxiter=300):
    n = len(h0)
    h0 = project_box_sum(h0, 0.5 * n)
    c0, profile0 = exact_objective(h0)
    c0_safe = c0 + 1e-8
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n + [(1e-6, 2.0)]

    def obj_fun(z):
        return z[-1]

    def obj_jac(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(z):
        return np.sum(z[:n]) - target

    def eq_jac(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    def ineq_fun(z):
        c_var = z[-1]
        h_var = z[:n]
        prof = np.correlate(h_var, 1.0 - h_var, mode='full') * (2.0 / n)
        return c_var - prof

    def ineq_jac(z):
        h_var = z[:n]
        G = compute_jacobian(h_var)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, -1] = 1.0
        return J

    constraints.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})

    z0 = np.concatenate([h0, [c0_safe]])
    best_h = h0.copy()
    best_c = c0

    try:
        res = minimize(obj_fun, z0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            cand = project_box_sum(res.x[:n], target)
            c_new, _ = exact_objective(cand)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = cand
    except Exception:
        pass
    return best_h, best_c

def active_set_polish(h, n_steps=200, deadline=None, step_scales=None):
    n = len(h)
    target = 0.5 * n
    h = project_box_sum(h, target)
    best_c, _ = exact_objective(h)
    if step_scales is None:
        step_scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    for step_idx in range(n_steps):
        if deadline is not None and time.time() > deadline - 2:
            break
        prof, _ = exact_objective(h)
        pmax = np.max(prof)
        threshold = pmax - pmax * 0.05
        active = np.where(prof >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(prof))]
        G = compute_jacobian(h)
        grad = -G[active, :].mean(axis=0)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_normed = grad / gnorm
        improved = False
        for scale in step_scales:
            step = scale * min(1.0, 0.5 / (gnorm + 1e-10))
            cand = project_box_sum(h + step * grad_normed, target)
            c_new, _ = exact_objective(cand)
            if c_new < best_c - 1e-14:
                best_c = c_new
                h = cand
                improved = True
                break
    return h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=50, rng=None, deadline=None):
    n = len(b)
    if n % 2 != 0 or b is None:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(0)
    c5, _ = exact_objective(b)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline - 5:
            break
        improved = False
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 5000:
            for i in ones:
                done = False
                for j in zeros:
                    b[i], b[j] = 0.0, 1.0
                    nc, _ = exact_objective(b)
                    if nc < c5 - 1e-14:
                        c5 = nc
                        improved = True
                        done = True
                        break
                    b[i], b[j] = 1.0, 0.0
                if done:
                    break
        else:
            n_tries = min(2000, n_pairs)
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc, _ = exact_objective(b)
                if nc < c5 - 1e-14:
                    c5 = nc
                    improved = True
                    ones = np.where(b > 0.5)[0]
                    zeros = np.where(b < 0.5)[0]
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        mid = n // 2
        h[mid:mid + q] = 1.0
    else:
        h = np.full(n, 0.5) + rng.randn(n) * 0.2
    return project_box_sum(h, target)

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def coordinate_descent(h, n_sweeps=20, deadline=None):
    n = len(h)
    target = 0.5 * n
    h = project_box_sum(h, target)
    best_c, _ = exact_objective(h)
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline - 3:
            break
        improved_any = False
        for i in range(n):
            if deadline is not None and time.time() > deadline - 3:
                break
            old_val = h[i]
            G_row_i = np.zeros(2 * n - 1)
            prof_old, _ = exact_objective(h)
            for e in range(-(n - 1), n):
                r = e + (n - 1)
                if e >= 0:
                    if 0 <= i - e < n - e:
                        a = (1.0 - h[i - e])
                        b_const = prof_old[r] / (2.0 / n) - h[i] * (1.0 - h[i - e])
                        G_row_i[r] = (2.0 / n) * a
                        b_const = prof_old[r] - (2.0 / n) * h[i] * (1.0 - h[i - e])
                    if 0 <= i < n - e:
                        G_row_i[r] += -(2.0 / n) * h[i + e]
                        b_const = prof_old[r] - (2.0 / n) * h[i + e] * (1.0 - h[i]) + (2.0 / n) * h[i + e] * (1.0 - h[i])
                else:
                    s = -e
                    if 0 <= i < n - s:
                        G_row_i[r] += (2.0 / n) * (1.0 - h[i + s])
                    if s <= i < n:
                        G_row_i[r] += -(2.0 / n) * h[i - s]
            # This approach is getting complicated; use simple projected gradient instead
            break
        # Fall back to subgradient step
        prof, _ = exact_objective(h)
        pmax = np.max(prof)
        active = np.where(prof >= pmax * 0.95)[0]
        G = compute_jacobian(h)
        grad = -G[active, :].mean(axis=0)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        for scale in [0.05, 0.02, 0.01, 0.005, 0.001]:
            step = scale / (gnorm + 1e-10)
            cand = project_box_sum(h + step * grad, target)
            c_new, _ = exact_objective(cand)
            if c_new < best_c - 1e-14:
                best_c = c_new
                h = cand
                improved_any = True
                break
        if not improved_any:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0, _ = exact_objective(init)
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 70
        best_h = make_start(n0, rng, mode=0)
        best_c, _ = exact_objective(best_h)

    n0 = len(best_h)

    # Phase 1: Hard-constraint SLSQP on parent
    if time.time() < deadline - 80:
        t_start = time.time()
        h_opt, c_opt = slsqp_hard(best_h, deadline, maxiter=300)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Active set polishing
    if time.time() < deadline - 60:
        h_opt, c_opt = active_set_polish(best_h, n_steps=300, deadline=deadline)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: SLSQP again after polishing
    if time.time() < deadline - 50:
        h_opt, c_opt = slsqp_hard(best_h, deadline, maxiter=200)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Binary optimization
    if n0 % 2 == 0 and time.time() < deadline - 40:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=80, rng=rng, deadline=deadline)
            if c_b < best_c:
                best_c = c_b
                best_h = b
            # SLSQP from binary
            if time.time() < deadline - 30:
                h_opt, c_opt = slsqp_hard(b, deadline, maxiter=200)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                # Polish
                if time.time() < deadline - 20:
                    h_opt2, c_opt2 = active_set_polish(h_opt, n_steps=100, deadline=deadline)
                    if c_opt2 < best_c:
                        best_c = c_opt2
                        best_h = h_opt2

    # Phase 5: Multi-size search
    sizes_to_try = []
    for s in range(30, 120, 2):
        sizes_to_try.append(s)
    sizes_to_try = [s for s in sizes_to_try if s != n0]
    # Sort by proximity to n0
    sizes_to_try.sort(key=lambda x: abs(x - n0))

    for n_try in sizes_to_try[:20]:
        if time.time() > deadline - 30:
            break
        # Try downsampled parent and random starts
        candidates = []
        if best_h is not None and len(best_h) != n_try:
            candidates.append(downsample(best_h, n_try))
        candidates.append(make_start(n_try, rng, mode=0))
        candidates.append(make_start(n_try, rng, mode=2))
        candidates.append(make_start(n_try, rng, mode=4))

        best_here = None
        best_here_c = float('inf')
        for cand in candidates:
            if time.time() > deadline - 15:
                break
            c_here, _ = exact_objective(cand)
            if c_here < best_here_c:
                best_here_c = c_here
                best_here = cand.copy()

        if best_here is not None:
            # SLSQP on best candidate
            if time.time() < deadline - 15:
                h_opt, c_opt = slsqp_hard(best_here, deadline, maxiter=150)
                if c_opt < best_here_c:
                    best_here_c = c_opt
                    best_here = h_opt
            # Polish
            if time.time() < deadline - 10:
                h_opt, c_opt = active_set_polish(best_here, n_steps=50, deadline=deadline)
                if c_opt < best_here_c:
                    best_here_c = c_opt
                    best_here = h_opt
            # Binary
            if n_try % 2 == 0 and time.time() < deadline - 8:
                b = binary_threshold(best_here)
                if b is not None:
                    b, c_b = binary_swap_improve(b, max_rounds=30, rng=rng, deadline=deadline)
                    if c_b < best_here_c:
                        best_here_c = c_b
                        best_here = b

            if best_here_c < best_c:
                best_c = best_here_c
                best_h = best_here

    # Phase 6: Final polishing
    if time.time() < deadline - 10:
        h_opt, c_opt = active_set_polish(best_h, n_steps=200, deadline=deadline)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if time.time() < deadline - 8:
        h_opt, c_opt = slsqp_hard(best_h, deadline, maxiter=150)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)

    return (best_h, best_c, best_n)
