def active_set_direction(h, active_lags, dx, max_iter_lp=1):
    n = len(h)
    nx = n + 1  # n components of d + 1 for alpha
    
    # Objective: minimize -alpha
    c = np.zeros(nx)
    c[-1] = -1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    A_rows = []
    b_rows = []
    
    for k in active_lags:
        g = gradient_lag_vec(h, k, dx)
        row = np.zeros(nx)
        row[:n] = g
        row[-1] = 1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    # Box on d: -h[j] <= d[j] <= 1-h[j]
    for j in range(n):
        # d[j] <= 1-h[j]
        row = np.zeros(nx)
        row[j] = 1.0
        A_rows.append(row)
        b_rows.append(1.0 - h[j])
        # -d[j] <= h[j]
        row = np.zeros(nx)
        row[j] = -1.0
        A_rows.append(row)
        b_rows.append(h[j])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nx))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds_lp = [(None, None)] * nx
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    
    if res.success:
        d = res.x[:n]
        alpha = res.x[-1]
        return d, alpha
    return None, 0.0
