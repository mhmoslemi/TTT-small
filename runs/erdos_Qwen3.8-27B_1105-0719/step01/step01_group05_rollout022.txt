def project_box_sum(h, target):

        j = zeros[rng.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        nc = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
        delta = nc - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = nc
            if nc < best_c:
                best_c = nc
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c

def make_binary_start(n, rng, mode=0):
    target = n // 2
    b = np.zeros(n)
    if mode == 0:
        b[::2] = 1.0
    elif mode == 1:
        b[:target] = 1.0
    elif mode == 2:
        idx = rng.choice(n, size=target, replace=False)
        b[idx] = 1.0
    elif mode == 3:
        phi = 0.6180339887498949
        pos = np.unique(np.floor(np.arange(target) * n * phi + 0.5).astype(int) % n)
        if len(pos) < target:
            rem = np.setdiff1d(np.arange(n), pos)
            extra = rng.choice(rem, size=target - len(pos), replace=False)
            pos = np.unique(np.concatenate([pos, extra]))
        b[pos[:target]] = 1.0
    elif mode == 4:
        q = n // 4
        b[:q] = 1.0
        b[2 * n // 3: 2 * n // 3 + q] = 1.0
        cur = int(np.sum(b))
        if cur > target:
            ones = np.where(b > 0.5)[0]
            b[rng.choice(ones, size=cur - target, replace=False)] = 0.0
        elif cur < target:
            zeros = np.where(b < 0.5)[0]
            b[rng.choice(zeros, size=target - cur, replace=False)] = 1.0
    elif mode == 5:
        # Two interleaved arithmetic progressions
        half = target // 2
        rem = target - half
        step1 = n // half if half > 0 else 1
        step2 = n // rem if rem > 0 else 1
        p1 = (np.arange(half) * step1) % n
        p2 = (np.arange(rem) * step2 + step1 // 2) % n
        pos = np.unique(np.concatenate([p1, p2]))
        if len(pos) < target:
            rem2 = np.setdiff1d(np.arange(n), pos)
            extra = rng.choice(rem2, size=target - len(pos), replace=False)
            pos = np.unique(np.concatenate([pos, extra]))
        b[pos[:target]] = 1.0
    else:
        idx = rng.choice(n, size=target, replace=False)
        b[idx] = 1.0
    return b

def smooth_obj_grad(x, tau, n):
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = np.max(profile)
    target = 0.5 * n
    se = np.sum(x) - target
    value = pmax
    if tau > 1e-12:
        shifted = (profile - pmax) / tau
        shifted = np.clip(shifted, -60, 60)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        value += tau * np.log(wsum)
        wnorm = weights / wsum
    else:
        wnorm = np.zeros(len(profile))
        wnorm[np.argmax(profile)] = 1.0
    
    grad = np.zeros(n)
    for m in range(len(profile)):
        if wnorm[m] < 1e-9:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            s = n - lag
            if s > 0:
                grad[:s] -= x[lag:]
                grad[lag:] += 1.0 - x[:s]
        else:
            a = -lag
            s = n - a
            if s > 0:
                grad[:s] += 1.0 - x[a:]
                grad[a:] -= x[:s]
    grad *= dx
    value += 1e5 * se * se
    grad += 2e5 * se
    below = np.maximum(0.0, -x)
    above = np.maximum(0.0, x - 1.0)
    value += 1e5 * (np.sum(below * below) + np.sum(above * above))
    grad = grad - 2e5 * below + 2e5 * above
    return value, grad

def continuous_refine(h_start, n, deadline, taus=None):
    if taus is None:
        taus = [0.05, 0.01, 0.003, 0.001, 0.0003, 0.0001]
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    h = h_start.copy()
    best_c = exact_objective(h)[0]
    
    for tau in taus:
        if time.time() > deadline - 5:
            break
        
        def obj_grad(x):
            return smooth_obj_grad(x, tau, n)
        
        try:
            maxit = max(50, min(300, n * 4))
            res = minimize(obj_grad, h, method='L-BFGS-B',
                          bounds=bounds, jac=True,
                          options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                c = exact_objective(cand)[0]
                if c < best_c - 1e-14:
                    best_c = c
                    h = cand
        except Exception:
            pass
    return h, best_c

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    
    # Initialize from parent if available
    n0 = 70
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init_p = project_box_sum(init, 0.5 * n0)
            c0, _ = exact_objective(init_p)
            best_h = init_p.copy()
            best_c = c0
    except (NameError, Exception):
        pass
    
    if best_h is None:
        b_init = make_binary_start(n0, rng, mode=3)
        best_h = b_init
        best_c = binary_c5(b_init)
    
    # Phase 1: Binary local search from initial
    if n0 % 2 == 0 and time.time() < deadline - 120:
        b = (best_h > 0.5).astype(np.float64)
        diff = n0 // 2 - int(np.sum(b))
        if diff > 0:
            zeros = np.where(b < 0.5)[0]
            if len(zeros) >= diff:
                b[rng.choice(zeros, size=diff, replace=False)] = 1.0
        elif diff < 0:
            ones = np.where(b > 0.5)[0]
            if len(ones) >= -diff:
                b[rng.choice(ones, size=-diff, replace=False)] = 0.0
        b, c_b = binary_local_search(b, deadline, max_kicks=60, kick_size=3, rng=rng)
        if c_b < best_c:
            best_c = c_b
            best_h = b.copy()
    
    # Phase 2: Simulated annealing from best
    if time.time() < deadline - 100:
        n_best = len(best_h)
        if n_best % 2 == 0:
            b_ann = (best_h > 0.5).astype(np.float64)
            if abs(np.sum(b_ann) - n_best // 2) < 0.1:
                b_ann, c_ann = binary_anneal(b_ann, deadline, iterations=30000, rng=rng)
                if c_ann < best_c:
                    best_c = c_ann
                    best_h = b_ann.copy()
                # Follow up with local search
                if time.time() < deadline - 80:
                    b_ls, c_ls = binary_local_search(b_ann, deadline, max_kicks=40, kick_size=2, rng=rng)
                    if c_ls < best_c:
                        best_c = c_ls
                        best_h = b_ls.copy()
    
    # Phase 3: Try multiple n values with diverse starts
    sizes = []
    for delta in [0, -2, 2, -4, 4, -6, 6, 8, 10, -8, 12, -10, 14, 16]:
        s = n0 + delta
        if 48 <= s <= 110 and s % 2 == 0:
            sizes.append(s)
    sizes = sorted(set(sizes), key=lambda x: abs(x - n0))
    
    for n_try in sizes:
        if time.time() > deadline - 40:
            break
        n_starts = 4 if n_try == n0 else 5
        for mode in range(n_starts):
            if time.time() > deadline - 30:
                break
            b_start = make_binary_start(n_try, rng, mode=mode)
            # Also try downsampled best
            if best_h is not None and len(best_h) != n_try:
                h_ds = downsample(best_h, n_try)
                b_ds = (h_ds > 0.5).astype(np.float64)
                if abs(np.sum(b_ds) - n_try // 2) < 0.1:
                    if binary_c5(b_ds) < binary_c5(b_start):
                        b_start = b_ds
            b_opt, c_opt = binary_local_search(b_start, deadline, max_kicks=50, kick_size=3, rng=rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = b_opt.copy()
    
    # Phase 4: Continuous refinement from best binary
    if time.time() < deadline - 40:
        h_cont = best_h.copy()
        n_cont = len(h_cont)
        h_cont, c_cont = continuous_refine(h_cont, n_cont, deadline,
                                           taus=[0.02, 0.005, 0.002, 0.0005])
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont.copy()
    
    # Phase 5: More binary optimization with larger kicks
    if len(best_h) % 2 == 0 and time.time() < deadline - 25:
        n_best = len(best_h)
        b = (best_h > 0.5).astype(np.float64)
        if abs(np.sum(b) - n_best // 2) < 0.1:
            b, c_b = binary_local_search(b, deadline, max_kicks=100, kick_size=5, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b.copy()
    
    # Phase 6: Anneal from best
    if len(best_h) % 2 == 0 and time.time() < deadline - 15:
        n_best = len(best_h)
        b_ann = (best_h > 0.5).astype(np.float64)
        if abs(np.sum(b_ann) - n_best // 2) < 0.1:
            b_ann, c_ann = binary_anneal(b_ann, deadline, iterations=20000, rng=rng)
            if c_ann < best_c:
                best_c = c_ann
                best_h = b_ann.copy()
            b_ls, c_ls = binary_local_search(b_ann, deadline, max_kicks=30, kick_size=2, rng=rng)
            if c_ls < best_c:
                best_c = c_ls
                best_h = b_ls.copy()
    
    # Final: continuous polish
    if time.time() < deadline - 8:
        h_final = best_h.copy()
        h_final, c_final = continuous_refine(h_final, len(h_final), deadline,
                                             taus=[0.001, 0.0003, 0.0001])
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
    
    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_n = len(best_h)
    
    return (best_h, final_c, best_n)
