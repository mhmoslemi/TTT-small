import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += w[k] * (1.0 - h[idx + offset])
        if offset != 0:
            grad[idx + offset] += w[k] * (-h[idx])
        else:
            grad[idx] += w[k] * (-h[idx])
    return grad * dx


def smooth_max_value(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def slsqp_opt(h0, dx, beta, target_sum, maxiter):
    n = len(h0)

    def obj(x):
        return smooth_max_value(x, dx, beta)

    def grad(x):
        return smooth_max_grad(x, dx, beta)

    def constr_fun(x):
        return x.sum() - target_sum

    def constr_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def compute_pos_effect(h, pos, n):
    """Effect vector: change in corr[k] when h[pos] increases by 1."""
    offsets = np.arange(2 * n - 1) - (n - 1)
    effect = np.zeros(2 * n - 1)
    valid_plus = (pos + offsets >= 0) & (pos + offsets < n)
    effect[valid_plus] += (1.0 - h[pos + offsets[valid_plus]])
    valid_minus = (offsets != 0) & (pos - offsets >= 0) & (pos - offsets < n)
    effect[valid_minus] -= h[pos - offsets[valid_minus]]
    return effect


def bit_swap_search(h0, dx, max_sweeps=200):
    n = len(h0)
    target_sum = int(round(n / 2.0))
    h = np.where(h0 > 0.5, 1.0, 0.0).astype(float)
    if h.sum() != target_sum:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if h.sum() > target_sum:
            excess = int(h.sum() - target_sum)
            h[ones[:excess]] = 0.0
        elif h.sum() < target_sum:
            deficit = int(target_sum - h.sum())
            h[zeros[:deficit]] = 1.0

    corrs = compute_corrs(h)
    best_M = corrs.max()
    best_h = h.copy()

    for sweep in range(max_sweeps):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        effects_ones = [compute_pos_effect(h, i, n) for i in ones]
        effects_zeros = [compute_pos_effect(h, j, n) for j in zeros]

        for ii in range(len(ones)):
            for jj in range(len(zeros)):
                i = ones[ii]
                j = zeros[jj]
                new_corrs = corrs - effects_ones[ii] + effects_zeros[jj]
                new_M = new_corrs.max()
                if new_M < best_M - 1e-14:
                    h[i] = 0.0
                    h[j] = 1.0
                    corrs = new_corrs
                    best_M = new_M
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return best_h, best_M * dx
