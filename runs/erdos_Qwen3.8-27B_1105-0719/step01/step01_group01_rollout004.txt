    lo = -5.0

        else:

    if abs(residual) > 1e-12:

        else:
            if residual > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + residual)

    n = len(h)
    hi_val = h[i]
    hj_val = h[j]
    dl = max(hi_val - 1.0, -hj_val)
    dh = min(hi_val, 1.0 - hj_val)
    if dh - dl < 1e-12:

def coordinate_descent(h, time_limit, seed, max_sweeps=200):
    """Coordinate descent with continuous pair transfers, random order."""
    rng = np.random.RandomState(seed)
    n = len(h)
    best_score = score_c5(h)
    deadline = time.time() + time_limit
    no_improve_streak = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline - 1.0:
            break
        sweep_improved = False
        perm = rng.permutation(n * (n - 1) // 2)
        idx = 0
        for p in perm:
            if time.time() > deadline - 0.5:
                break
            i = p // (n - 1)
            j = p % (n - 1)
            if j >= i:
                j += 1
            if i == j:
                continue
            delta = optimize_pair(h, i, j)
            if abs(delta) > 1e-10:
                h_new = h.copy()
                h_new[i] = h[i] - delta
                h_new[j] = h[j] + delta
                h_new = np.clip(h_new, 0.0, 1.0)

                    best_score = s_new
                    sweep_improved = True
        if sweep_improved:
            no_improve_streak = 0
        else:
            no_improve_streak += 1
            if no_improve_streak >= 3:
                break
    return h, best_score

def annealing_search(h0, time_limit, seed):
    """Simulated annealing with pair transfers."""
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    cur_score = score_c5(h)
    best_h = h.copy()
    best_score = cur_score
    deadline = time.time() + time_limit

    T = max(0.002, cur_score * 0.01)
    T_min = 1e-5

    while time.time() < deadline:
        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(max(0, h[i] - 1), min(h[i], 1 - h[j]))
        if delta < 1e-8:
            continue
        h_new = h.copy()
        h_new[i] = h[i] - delta
        h_new[j] = h[j] + delta
        h_new = np.clip(h_new, 0.0, 1.0)
        s_new = score_c5(h_new)
        d_s = s_new - cur_score
        if d_s < 0 or (T > T_min and rng.random() < np.exp(-d_s / T)):
            h = h_new
            cur_score = s_new
            if s_new < best_score:
                best_score = s_new
                best_h = h.copy()
        T *= 0.9995
        if T < T_min:
            break
    return best_h, best_score

def resample(h, n_new):
    """Resample h to new resolution via piecewise constant interpolation."""
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = n_new / 2.0
    h_new = project_feasible(h_new, target)
    return h_new
