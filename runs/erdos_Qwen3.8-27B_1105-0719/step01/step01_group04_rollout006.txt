def repair_feasible(vec):

def logit_gradient(h, z, c, G_h):
    """Chain rule: G_z_j = G_h_j - sigmoid'(z_j+c) * sum_i sigmoid'(z_i+c) G_h_i / T

    T = sig_prime.sum()
    if T < 1e-10:
        return G_h.copy()
    dot = sig_prime @ G_h
    return G_h - sig_prime * (dot / T)

    shifted = beta * (values - max_v)
    log_sum = np.log(np.sum(np.exp(shifted))) + beta * max_v
    return log_sum / beta

def active_descent_exact(h, max_steps=80, top_k=20):
    """Direct active descent on the exact objective."""
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for step in range(max_steps):
        values = compute_profile(best)

        # Select active lags
        n_active = min(top_k, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_active]
        
        # Compute weights based on gaps
        gaps = c_max - values[active_idx]
        sigma = max(0.005, np.std(gaps) + 1e-10)
        weights = np.exp(-gaps / sigma)
        weights /= weights.sum()
        
        # Compute weighted gradient
        full_weights = np.zeros(2 * n - 1)
        full_weights[active_idx] = weights
        direction = -compute_weighted_gradient(best, full_weights)
        
        # Remove mean to preserve integral
        direction -= direction.mean()
        
        # Line search
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:

def transfer_dimension(h_src, n_tgt):
    """Area-preserving transfer from h_src to a target dimension."""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    
    # Build cumulative function
    # F(x) = integral of h from 0 to x
    # Cell i covers [i*dx_src, (i+1)*dx_src) with height h_src[i]
    
    # For each target cell j, compute the average height
    x_src_edges = np.arange(n_src + 1) * dx_src
    x_tgt_edges = np.arange(n_tgt + 1) * dx_tgt
    
    # For each target cell, find overlap with source cells
    v = np.zeros(n_tgt)
    for j in range(n_tgt):
        x0 = x_tgt_edges[j]
        x1 = x_tgt_edges[j + 1]
        # Find source cells that overlap [x0, x1)
        i_start = int(np.floor(x0 / dx_src))
        i_end = int(np.ceil(x1 / dx_src))
        i_start = max(0, i_start)

            integral += h_src[i] * ovl
        v[j] = integral / dx_tgt
    v = np.clip(v, 0.0, 1.0)
    return v

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        for i in order:
            if diff <= 1e-15:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(-h)
        for i in order:
            if -diff <= 1e-15:
                break
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        for i in range(n):
            if residual > 0 and h[i] < 1.0:
                h[i] += residual
                break
            elif residual < 0 and h[i] > 0.0:
                h[i] += residual
                break
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_weighted_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grad = np.zeros(n)
    w0 = weights[n - 1]
    if abs(w0) > 1e-18:

    for delta in range(1, n):
        wp = weights[n - 1 + delta]
        if abs(wp) > 1e-18:
            idx = np.arange(n - delta)
            grad[idx] += wp * dx * (1.0 - h[idx + delta])
            grad[idx + delta] += -wp * dx * h[idx]
        wn = weights[n - 1 - delta]
        if abs(wn) > 1e-18:
            idx2 = np.arange(delta, n)
            grad[idx2] += wn * dx * (1.0 - h[idx2 - delta])
            grad[idx2 - delta] += -wn * dx * h[idx2]
    return grad

def softmax_val_and_grad(h, beta):
    values = compute_profile(h)
    max_v = values.max()
    shifted = beta * (values - max_v)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    weights = exp_s / Z
    obj = (np.log(Z) + beta * max_v) / beta
    grad = compute_weighted_gradient(h, weights)
    return obj, grad, values, weights

def transfer_dim(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    v = np.zeros(n_tgt)
    for j in range(n_tgt):
        x0 = j * dx_tgt
        x1 = (j + 1) * dx_tgt
        i_start = max(0, int(np.floor(x0 / dx_src)))
        i_end = min(n_src, int(np.ceil(x1 / dx_src)))
        total = 0.0
        for i in range(i_start, i_end):
            ovl = min(x1, (i + 1) * dx_src) - max(x0, i * dx_src)
            if ovl > 0:
                total += h_src[i] * ovl
        v[j] = total / dx_tgt
    return np.clip(v, 0.0, 1.0)
