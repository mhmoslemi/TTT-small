    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # all lag values

def compute_gradient_matrix(h):
    """Compute G[i,j] = dx * dF(s_i)/dh[j] where s_i = i - (n-1)"""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for i in range(M):
        s = i - (n - 1)
        # Case 1: h[j] appears as h[i+s] in F(s), contributing (1-h[i]) = (1-h[j-s])
        # valid j: max(0,s) <= j <= min(n-1, n-1+s)
        j_lo1 = max(0, s)
        j_hi1 = min(n - 1, n - 1 + s)
        # For these j, i_index = j - s, contribution is (1-h[j-s])
        for j in range(j_lo1, j_hi1 + 1):
            idx = j - s
            if 0 <= idx <= n - 1:
                G[i, j] += dx * (1.0 - h[idx])
        
        # Case 2: h[j] appears as h[i] in F(s), contributing -h[i+s] = -h[j+s]
        # valid j: max(0,-s) <= j <= min(n-1, n-1-s)
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        for j in range(j_lo2, j_hi2 + 1):
            idx = j + s
            if 0 <= idx <= n - 1:
                G[i, j] -= dx * h[idx]
    return G

import numpy as np
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(y, n):
    """Project y to feasible set: 0<=h<=1, sum(h)=n/2"""
    target = n / 2.0
    y = np.clip(y, -2.0, 3.0)  # initial clamping for bisection
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute residual among interior points
        interior = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            adj = residual / len(interior)
            h[interior] += adj
            h = np.clip(h, 0.0, 1.0)
        # If still not exact, adjust one point
        residual2 = target - np.sum(h)
        if abs(residual2) > 1e-14:
            # Find a point we can adjust
            if residual2 > 0:
                idx = np.argmax(h)
                # Can't increase a point at 1, so find one below 1
                candidates = np.where(h < 1.0 - 1e-14)[0]
            else:
                idx = np.argmin(h)
                candidates = np.where(h > 1e-14)[0]
            if len(candidates) > 0:
                i = candidates[0]
                h[i] += residual2
                h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h):
    """Build G[i,j] = dx * dF(s_i)/dh[j] where s_i = i - (n-1)"""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for i in range(M):
        s = i - (n - 1)
        # Case 1: h[j] as h[i+s], contribution (1-h[j-s])
        j_lo1 = max(0, s)
        j_hi1 = min(n - 1, n - 1 + s)
        if j_lo1 <= j_hi1:
            js = np.arange(j_lo1, j_hi1 + 1)
            idxs = js - s
            G[i, js] += dx * (1.0 - h[idxs])
        # Case 2: h[j] as h[i], contribution -h[j+s]
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            js = np.arange(j_lo2, j_hi2 + 1)
            idxs = js + s
            G[i, js] -= dx * h[idxs]
    return G

def slp_step(h, F, G, R, rho, active_indices=None):
    """One SLP step using CVXPY. Returns new h or None."""
    n = len(h)
    M = len(F)
    if active_indices is None:
        active_indices = np.arange(M)
    
    u = cp.Variable(n)
    v = cp.Variable()
    
    obj = cp.Minimize(v + rho * cp.sum_squares(u))
    constraints = [
        cp.sum(u) == 0,
        u >= -h + 1e-12,
        u <= 1.0 - h - 1e-12,
        cp.sum_squares(u) <= R**2,
    ]
    for i in active_indices:
        constraints.append(v >= F[i] + G[i, :] @ u)
    
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    
    if u.value is None:
        return None
    
    u_opt = u.value
    # Check bounds
    h_new = h + u_opt
    if np.any(h_new < -1e-8) or np.any(h_new > 1.0 + 1e-8):
        h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new

def slp_optimize(h, max_iter=50, R_init=0.05, rho_init=10.0, active_frac=0.5):
    """SLP optimization loop."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    R = R_init
    rho = rho_init
    consecutive_fails = 0
    
    for it in range(max_iter):
        F = compute_c5_all(best_h)
        G = build_gradient_matrix(best_h)
        
        # Select active lags (top fraction)
        M = len(F)
        n_active = max(3, int(active_frac * M))
        active_idx = np.argsort(F)[-n_active:]
        
        h_new = slp_step(best_h, F, G, R, rho, active_idx)
        if h_new is None:
            R *= 0.7
            consecutive_fails += 1
            if consecutive_fails > 10:
                break
            continue
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            R = min(R * 1.2, 0.3)
            consecutive_fails = 0
        else:
            R *= 0.6
            rho *= 1.5
            consecutive_fails += 1
            if consecutive_fails > 15 or R < 0.001:
                break
    
    return best_h, best_c5

def simulated_annealing(h, n, n_steps, T_init, T_min, cooling, rng):
    """SA with pair-transfer moves."""
    cur_h = h.copy()
    cur_c5 = compute_c5(h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    T = T_init
    
    for step in range(n_steps):
        T = max(T_min, T * cooling)
        
        # Choose move type
        if rng.random() < 0.5:
            # Random pair transfer
            i = rng.randint(n)
            j = rng.randint(n)
        else:
            # Targeted: pick from active lags
            F = compute_c5_all(cur_h)
            max_val = np.max(F)
            # Pick a random active lag
            active = np.where(F > max_val - 0.01)[0]
            lag_idx = rng.choice(active)
            s = lag_idx - (n - 1)
            # Pick a pair in this lag
            i_lo = max(0, -s)
            i_hi = min(n - 1, n - 1 - s)
            if i_lo >= i_hi:
                i = rng.randint(n)
                j = rng.randint(n)
            else:
                i_idx = rng.randint(i_lo, i_hi + 1)
                # Transfer from position i_idx+s to i_idx (to reduce the product)
                src = i_idx + s
                dst = i_idx
                if cur_h[src] > cur_h[dst]:
                    i, j = src, dst  # move mass from high to low
                else:
                    i, j = dst, src
                # Actually, to reduce h[src]*(1-h[dst]), we want to decrease h[src] and increase h[dst]
                # So transfer FROM src TO dst
                i, j = src, dst
        
        if i == j:
            continue
        
        # Transfer amount
        if cur_h[i] > cur_h[j]:
            max_delta = cur_h[i] * 0.3
        else:
            max_delta = (1.0 - cur_h[j]) * 0.3
        delta = rng.uniform(0.01, max(max_delta, 0.01))
        
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        delta_c5 = c5_new - cur_c5
        
        if delta_c5 < 0 or (T > T_min and rng.random() < np.exp(-delta_c5 / T)):
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    return best_h, best_c5

def resample(h_old, n_new):
    """Resample from n_old points to n_new points."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    h_new = project_feasible(h_new, n_new)
    return h_new

import numpy as np
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(y, n):
    target = n / 2.0
    y = np.array(y, dtype=float)
    lo = np.min(y) - 2.0
    hi = np.max(y) + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-11:
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            h[interior] += residual / len(interior)
            h = np.clip(h, 0.0, 1.0)
    residual2 = target - np.sum(h)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            cands = np.where(h < 1.0 - 1e-12)[0]
        else:
            cands = np.where(h > 1e-12)[0]
        if len(cands) > 0:
            h[cands[0]] += residual2
            h = np.clip(h, 0.0, 1.0)
    return h


def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for i in range(M):
        s = i - (n - 1)
        j_lo1 = max(0, s)
        j_hi1 = min(n - 1, n - 1 + s)
        if j_lo1 <= j_hi1:
            js = np.arange(j_lo1, j_hi1 + 1)
            idxs = js - s
            G[i, js] += dx * (1.0 - h[idxs])
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            js = np.arange(j_lo2, j_hi2 + 1)
            idxs = js + s
            G[i, js] -= dx * h[idxs]
    return G


def gradient_of_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted - np.max(shifted))
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for i in range(len(corr)):
        s_lag = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        j_lo1 = max(0, s_lag)
        j_hi1 = min(n - 1, n - 1 + s_lag)
        if j_lo1 <= j_hi1:
            js = np.arange(j_lo1, j_hi1 + 1)
            idxs = js - s_lag
            grad[js] += w * dx * (1.0 - h[idxs])
        j_lo2 = max(0, -s_lag)
        j_hi2 = min(n - 1, n - 1 - s_lag)
        if j_lo2 <= j_hi2:
            js = np.arange(j_lo2, j_hi2 + 1)
            idxs = js + s_lag
            grad[js] -= w * dx * h[idxs]
    return grad


def slp_step_cvx(h, F, G, R, rho, active_idx):
    n = len(h)
    u = cp.Variable(n)
    v = cp.Variable()
    obj = cp.Minimize(v + rho * cp.sum_squares(u))
    constraints = [
        cp.sum(u) == 0,
        u >= -h,
        u <= 1.0 - h,
        cp.sum_squares(u) <= R * R,
    ]
    for i in active_idx:
        constraints.append(v >= F[i] + G[i, :] @ u)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=20000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    if u.value is None:
        return None
    h_new = h + u.value
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new


def slp_optimize(h, max_iter=50, R_init=0.04, rho_init=10.0, active_frac=1.0):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    R = R_init
    rho = rho_init
    fails = 0
    for it in range(max_iter):

        G = build_gradient_matrix(best_h)
        M = len(F)
        n_active = max(4, int(active_frac * M))
        active_idx = np.argsort(F)[-n_active:]
        h_new = slp_step_cvx(best_h, F, G, R, rho, active_idx)
        if h_new is None:
            R *= 0.7
            fails += 1
            if fails > 12:
                break
            continue
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            R = min(R * 1.15, 0.25)
            fails = 0
        else:
            R *= 0.6
            rho *= 1.3
            fails += 1
            if fails > 15 or R < 0.002:
                break
    return best_h, best_c5


def gradient_step(h, step_size, beta=200.0):
    n = len(h)
    grad = gradient_of_max(h, beta)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

            h_opt, c5_opt = slp_optimize(best_h, max_iter=30, R_init=0.025, rho_init=20.0, active_frac=0.5)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Phase 4: Multiple restarts
    if tleft() > 100:
        starts = generate_starts(best_h, n, rng)
        n_restarts = min(len(starts), int(tleft() / 70))
        for r in range(n_restarts):
            if tleft() < 40:
                break
            h_s = starts[r]
            c5_s = compute_c5(h_s)
            n_sa = int(min(80000, tleft() * 40))
            T_i = max(0.005, c5_s * 0.04)
            h_opt, c5_opt = sa_explore(h_s, n, n_sa, T_i, 0.0003, 0.999, rng, beta=100.0)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
            if tleft() > 25:
                h_opt2, c5_opt2 = slp_optimize(best_h, max_iter=25, R_init=0.02, rho_init=25.0, active_frac=0.5)
                if c5_opt2 < best_c5:
                    best_h = h_opt2
                    best_c5 = c5_opt2

    # Phase 5: Resolution scanning
    if tleft() > 50:
        for n_try in [48, 72, 84, 40, 56, 96]:
            if tleft() < 35:
                break
            h_try = resample_h(best_h, n_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                n = n_try
            h_opt, c5_opt = slp_optimize(h_try, max_iter=30, R_init=0.035, rho_init=10.0, active_frac=0.6)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                n = n_try
            if tleft() > 20:
                n_sa = int(min(40000, tleft() * 30))
                h_opt2, c5_opt2 = sa_explore(best_h, n, n_sa, 0.004, 0.0002, 0.999, rng, beta=120.0)
                if c5_opt2 < best_c5:
                    best_h = h_opt2
                    best_c5 = c5_opt2
                    n = len(best_h)

    # Phase 6: Intensive final polish
    if tleft() > 20:
        h_opt, c5_opt = slp_optimize(best_h, max_iter=60, R_init=0.015, rho_init=50.0, active_frac=0.3)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 7: Final SA micro-polish
    if tleft() > 10:
        n_sa = int(min(50000, tleft() * 150))
        h_opt, c5_opt = sa_explore(best_h, len(best_h), n_sa, 0.003, 0.0001, 0.9995, rng, beta=500.0)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final: one more SLP
    if tleft() > 8:
        h_opt, c5_opt = slp_optimize(best_h, max_iter=30, R_init=0.01, rho_init=80.0, active_frac=0.25)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.copy(), best_c5, len(best_h)
