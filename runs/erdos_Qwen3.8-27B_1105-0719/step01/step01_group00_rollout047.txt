    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        i_max = n - 1 - offset
        if i_max < 0:
            return grad
        idx = np.arange(0, i_max + 1)
        grad[idx] += 1.0 - h[idx + offset]
        idx2 = np.arange(offset, n)
        grad[idx2] += -h[idx2 - offset]
    else:
        m = -offset
        idx = np.arange(m, n)
        grad[idx] += 1.0 - h[idx - m]
        idx2 = np.arange(0, n - m)
        grad[idx2] += -h[idx2 + m]
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(corrs) / beta * dx


def smooth_max_grad(h, dx, beta, top_m=None):
    corrs = compute_corrs(h)
    n = len(h)
    if top_m is not None and top_m < len(corrs):
        top_idx = np.argsort(corrs)[-top_m:]
        c_sub = corrs[top_idx]
        w = np.exp(beta * (c_sub - c_sub.max()))
        w /= w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] > 1e-10:
                grad += w[i] * grad_corr_k(h, k, n)
    else:
        c_shifted = corrs - corrs.max()
        w = np.exp(beta * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-10:
                grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    objf_val = lambda x: smooth_max_obj(x, dx, beta)
    objf_grad = lambda x: smooth_max_grad(x, dx, beta, top_m=20)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objf_val, h0, jac=objf_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def subgradient_polish(h0, dx, target_sum, max_iter=30000, initial_lr=0.05):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    lr = initial_lr
    no_improve = 0
    patience = 5000

    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience and it > 500:
                break

        threshold = max_c - 0.05
        active = np.where(corrs >= threshold)[0]
        if len(active) == 0:
            active = np.argmax(corrs)

        grad = np.zeros(n)
        for k in active:
            grad += grad_corr_k(h, k, n)
        grad /= len(active)
        grad *= dx

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        h_new = project_feasible(h - lr * grad, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-15:
            h = h_new
            lr = min(lr * 1.01, 0.2)
        else:
            lr *= 0.97
            if lr < 1e-8:
                break

        if it % 500 == 0:
            corrs_check = compute_corrs(h)
            if corrs_check.max() * dx < best_c5:
                best_c5 = corrs_check.max() * dx
                best_h = h.copy()

    return best_h, best_c5


def binary_round(h, target_sum):
    n = len(h)
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    n_ones = int(target_sum)
    h_bin[sorted_idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h_bin[sorted_idx[n_ones]] = frac
    return h_bin


def perturb_solution(h, target_sum, rng, strength=0.1):
    noise = rng.randn(len(h)) * strength
    return project_feasible(h + noise, target_sum)


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def optimize_single_start(h0, t0, budget_s, min_time=30):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    if remaining_time(t0, budget_s) < min_time:
        return best_h, best_c5

    # SLSQP continuation
    for beta in [30, 80, 200, 500, 1200, 3000, 6000]:
        if remaining_time(t0, budget_s) < min_time:
            break
        nit = 300 if beta < 1000 else 500
        h_new, c5_new = slsqp_optimize(best_h, dx, target_sum, beta, nit)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Subgradient polish
    if remaining_time(t0, budget_s) > min_time + 10:
        h_new, c5_new = subgradient_polish(best_h, dx, target_sum, 20000, 0.03)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final SLSQP with very high beta
    if remaining_time(t0, budget_s) > min_time + 5:
        h_new, c5_new = slsqp_optimize(best_h, dx, target_sum, 8000, 400)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Start from the initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Main optimization from initial
    best_h, best_c5 = optimize_single_start(best_h, t0, budget_s, min_time=80)

    # Phase 2: Try n=128 with interpolation
    if remaining_time(t0, budget_s) > 60:
        n_try = 128
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h, left=0.0, right=0.0)
        h_interp = project_feasible(h_interp, ts_try)
        h_new, c5_new = optimize_single_start(h_interp, t0, budget_s, min_time=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: Try n=127
    if remaining_time(t0, budget_s) > 50:
        n_try = 127
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h, left=0.0, right=0.0)
        h_interp = project_feasible(h_interp, ts_try)
        h_new, c5_new = optimize_single_start(h_interp, t0, budget_s, min_time=40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: Random perturbations of best
    for trial in range(8):
        if remaining_time(t0, budget_s) < 30:
            break
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_pert = perturb_solution(best_h, ts_b, rng, strength=0.05 + 0.02 * trial)
        h_new, c5_new = optimize_single_start(h_pert, t0, budget_s, min_time=25)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 5: Binarization + re-optimize
    if remaining_time(t0, budget_s) > 40:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_bin = binary_round(best_h, ts_b)
        h_new, c5_new = optimize_single_start(h_bin, t0, budget_s, min_time=25)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 6: Random binary starts
    for trial in range(20):
        if remaining_time(t0, budget_s) < 15:
            break
        n_b = len(best_h)
        ts_b = n_b / 2.0
        h_bin = make_binary_start(n_b, ts_b, rng)
        dx_b = 2.0 / n_b
        h_new, c5_new = slsqp_optimize(h_bin, dx_b, ts_b, 300, 400)
        h_new2, c5_new2 = slsqp_optimize(h_new, dx_b, ts_b, 2000, 500)
        if c5_new2 < best_c5:
            best_c5 = c5_new2
            best_h = h_new2

    # Phase 7: Final intensive polish on the best
    if remaining_time(t0, budget_s) > 20 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        # SLSQP with very high beta
        h_new, c5_new = slsqp_optimize(best_h, dx_b, ts_b, 10000, 600)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

        # Subgradient with very small step
        if remaining_time(t0, budget_s) > 10:
            h_new, c5_new = subgradient_polish(best_h, dx_b, ts_b, 30000, 0.01)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

        # One more SLSQP
        if remaining_time(t0, budget_s) > 5:
            h_new, c5_new = slsqp_optimize(best_h, dx_b, ts_b, 12000, 300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Final verification
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = c5_of_h(final_h, 2.0 / len(final_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h

    return best_h, best_c5, len(best_h)
