def exact_profile(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_value(h):

def project_feasible(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)

            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_gradient(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= h[lag:]
            grad[lag:] += 1.0 - h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - h[a:]
            grad[a:] -= h[:s]
    return grad * dx

def make_direction(h, grad, n):
    """Project -grad onto feasible directions (sum=0, respecting bounds)."""
    d = -grad.copy()
    # Remove mean to satisfy sum constraint
    d -= np.mean(d)
    # Respect bounds: if h[i]≈0, d[i] can't be negative; if h[i]≈1, d[i] can't be positive
    eps = 1e-10
    for _ in range(5):
        mask_low = (h < eps) & (d < 0)
        mask_high = (h > 1 - eps) & (d > 0)
        d[mask_low] = 0
        d[mask_high] = 0
        # Re-center
        d -= np.mean(d)
        d[mask_low] = 0  # Re-apply after re-centering
        d[mask_high] = 0
    return d

def line_search(h, d, c_base, n, alpha_max=None):
    """Find best step size along direction d."""
    if np.max(np.abs(d)) < 1e-15:
        return h, c_base
    # Find max step that stays in [0,1]
    if alpha_max is None:
        alpha_max = 1.0
        for i in range(n):
            if d[i] < -1e-15:
                alpha_max = min(alpha_max, h[i] / (-d[i]))
            elif d[i] > 1e-15:
                alpha_max = min(alpha_max, (1 - h[i]) / d[i])
        alpha_max = min(alpha_max, 1.0) * 0.99
    
    if alpha_max < 1e-15:
        return h, c_base
    
    # Scale direction
    d_scaled = d * alpha_max
    
    # Try a grid of step sizes
    best_c = c_base
    best_alpha = 0.0
    # Coarse grid
    fractions = [0.01, 0.03, 0.1, 0.3, 0.5, 0.7, 0.9]
    for frac in fractions:
        alpha = frac
        h_try = h + alpha * d_scaled
        h_try = project_feasible(h_try)
        c_try = c5_value(h_try)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_alpha = alpha
    
    if best_alpha == 0:
        # Try even smaller steps
        for frac in [0.003, 0.001]:
            alpha = frac
            h_try = h + alpha * d_scaled
            h_try = project_feasible(h_try)
            c_try = c5_value(h_try)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_alpha = alpha
    
    if best_alpha > 0:
        h_new = h + best_alpha * d_scaled
        h_new = project_feasible(h_new)
        return h_new, best_c
    return h, c_base

import numpy as np
import time

def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_value(h):
    return float(np.max(exact_profile(h)))

def project_feasible(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(h + t, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lag_gradient(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= h[lag:]
            grad[lag:] += 1.0 - h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - h[a:]
            grad[a:] -= h[:s]
    return grad * dx

def make_feasible_direction(h, g, n):
    d = -g.copy()
    d -= np.mean(d)
    eps = 1e-10
    for _ in range(8):
        d[(h < eps) & (d < 0)] = 0.0
        d[(h > 1.0 - eps) & (d > 0)] = 0.0
        d -= np.mean(d)
        d[(h < eps) & (d < 0)] = 0.0
        d[(h > 1.0 - eps) & (d > 0)] = 0.0
    return d

def line_search(h, d, c_base, n):
    max_abs_d = float(np.max(np.abs(d)))
    if max_abs_d < 1e-16:
        return h, c_base
    alpha_max = 0.99
    for i in range(n):
        if d[i] < -1e-15:
            alpha_max = min(alpha_max, h[i] / (-d[i]))
        elif d[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
    if alpha_max < 1e-14:
        return h, c_base
    d = d * alpha_max
    best_c = c_base
    best_h = h
    fracs = [0.5, 0.25, 0.1, 0.05, 0.02, 0.8, 0.95, 0.01]
    for f in fracs:
        h_try = h + f * d
        h_try = project_feasible(h_try)
        c_try = c5_value(h_try)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
    if best_h is not h:
        best_h = project_feasible(best_h)
        best_c = c5_value(best_h)
    return best_h, best_c

def optimize_step(h, profile, n, top_k_list, rng):
    c_base = float(np.max(profile))
    dx = 2.0 / n
    n_profile = len(profile)
    order = np.argsort(-profile)
    best_h = h
    best_c = c_base
    for K in top_k_list:
        if K > n_profile:
            K = n_profile
        active = order[:K]
        weights = profile[active]
        w = weights / np.sum(weights)
        grad = np.zeros(n)
        for k in range(K):
            lag = int(active[k]) - (n - 1)
            grad += w[k] * lag_gradient(h, lag, dx)
        d = make_feasible_direction(h, grad, n)
        h_try, c_try = line_search(h, d, c_base, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
    return best_h, best_c

def optimize_active_set(h, n, max_iters, deadline, rng, top_k_list=None):
    if top_k_list is None:
        top_k_list = [1, 2, 3, 5, 8]
    h = project_feasible(h, 0.5 * n)
    best_h = h.copy()
    best_c = c5_value(h)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        profile = exact_profile(h)
        c_cur = float(np.max(profile))
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()
        h_new, _ = optimize_step(h, profile, n, top_k_list, rng)
        if c5_value(h_new) < best_c:
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                d_rand = rng.randn(n)
                d_rand -= np.mean(d_rand)
                d_rand = make_feasible_direction(h, -d_rand, n)
                h_try, c_try = line_search(h, d_rand, best_c, n)
                if c_try < best_c - 1e-13:
                    h = h_try
                    best_c = c_try
                    best_h = h.copy()
                    stall = 0
                else:
                    h = project_feasible(h + rng.randn(n) * 0.01)
                    if c5_value(h) > best_c + 0.005:

                    stall = 0
    return best_h, best_c

def binary_round(h, n):
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap(h_b, rng, max_rounds=200):
    n = len(h_b)
    c_cur = c5_value(h_b)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h_b > 0.5)[0]
        zeros = np.where(h_b < 0.5)[0]
        n_pairs = len(ones) * len(zeros)
        ntries = min(n_pairs, 800)
        for _ in range(ntries):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h_b[i], h_b[j] = 0.0, 1.0
            c_new = c5_value(h_b)
            if c_new < c_cur - 1e-14:
                c_cur = c_new
                improved = True
                ones = np.where(h_b > 0.5)[0]
                zeros = np.where(h_b < 0.5)[0]
            else:
                h_b[i], h_b[j] = 1.0, 0.0
        if not improved:
            break
    return h_b, c_cur

def slsqp_refine(h, n, deadline, max_iter=150):
    best_h = project_feasible(h, 0.5 * n)
    best_c = c5_value(best_h)
    dx = 2.0 / n
    target = 0.5 * n

    def obj_and_grad(x):
        prof = np.correlate(x, 1.0 - x, mode='full') * dx
        pmax = float(np.max(prof))
        top5 = np.argsort(-prof)[:5]
        w = prof[top5]
        w = w / np.sum(w)
        grad = np.zeros(n)
        for k in range(len(top5)):
            lag = int(top5[k]) - (n - 1)
            grad += w[k] * lag_gradient(x, lag, dx)
        se = np.sum(x) - target
        val = pmax + 1e6 * se * se
        grad = grad + 2e6 * se
        below = np.maximum(0.0, -x)
        above = np.maximum(0.0, x - 1.0)
        val += 1e4 * (np.sum(below * below) + np.sum(above * above))
        grad += -2e4 * below + 2e4 * above
        return val, grad

    try:
        from scipy.optimize import minimize
        res = minimize(obj_and_grad, best_h, method='SLSQP',
                       jac=True, bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            cand = project_feasible(res.x, target)
            c_cand = c5_value(cand)
            if c_cand < best_c:
                best_c = c_cand
                best_h = cand
    except Exception:
        pass
    return best_h, best_c

        h += rng.randn(n) * 0.1
    else:
        h = np.full(n, 0.5)
