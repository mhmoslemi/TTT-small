            j = neg_grad[p]
            for delta in delta_options:
                if best_h[i] + delta > 1.0 or best_h[j] - delta < 0.0:
                    continue
                h_try = best_h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
                # also try opposite direction
                if best_h[i] - delta >= 0.0 and best_h[j] + delta <= 1.0:
                    h_try2 = best_h.copy()
                    h_try2[i] -= delta
                    h_try2[j] += delta
                    c5_try2 = compute_c5(h_try2)
                    if c5_try2 < best_c5 - 1e-14:
                        best_h = h_try2
                        best_c5 = c5_try2
                        improved = True
                        break
        if not improved:
            # try random pairs
            for _ in range(500):
                i, j = rng.choice(n, size=2, replace=False)
                if i == j:
                    continue
                d = rng.choice(delta_options)
                if best_h[i] + d > 1.0 or best_h[j] - d < 0.0:
                    continue
                h_try = best_h.copy()
                h_try[i] += d
                h_try[j] -= d
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_h = h_try
                    best_c5 = c5_try
                    break
    return best_h, best_c5


def fine_pair_search(h0, n, n_sweeps, rng=None):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    small_deltas = np.array([0.01, 0.02, 0.03, 0.05, 0.08, 0.12])
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                d = min(small_deltas)
                if best_h[i] + d > 1.0 or best_h[j] - d < 0.0:
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] -= d
                        h_try[j] += d
                    else:
                        continue
                else:
                    h_try = best_h.copy()
                    h_try[i] += d
                    h_try[j] -= d
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def swap_local_search(h0, n, n_iters, rng):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iters):
        i, j = rng.choice(n, size=2, replace=False)
        if abs(best_h[i] - best_h[j]) < 0.05:
            continue
        h_try = best_h.copy()
        h_try[i], h_try[j] = best_h[j], best_h[i]
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(60, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient with momentum
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        betas_pg = [15, 50, 150, 500, 2000]
        n_iter = min(8000, int(rem * 30))
        h_pg, c5_pg = projected_gradient(best_h, betas_pg, n_iter, momentum=0.8, lr0=0.4, rng=rng)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Phase 2: SLSQP refinement
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, [50, 200, 1000, 5000], max_iter=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Directed pair adjustment
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        n_sweeps = min(30, int(rem / 3))
        h_opt, c5_opt = directed_pair_search(best_h, n, n_sweeps, beta=500, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: SLSQP with high beta
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, [1000, 5000, 20000], max_iter=400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: More directed pair search at higher beta
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        n_sweeps = min(20, int(rem / 4))
        h_opt, c5_opt = directed_pair_search(best_h, n, n_sweeps, beta=2000, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Swap-based local search
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        n_iters = min(50000, int(rem * 40))
        h_opt, c5_opt = swap_local_search(best_h, n, n_iters, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Try other resolutions
    rem = budget_s - (time.time() - t0)
    n_try_list = [59, 61, 64, 72, 80]
    for n_try in n_try_list:
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c5_opt = slsqp_optimize(h_interp, [30, 200, 1000], max_iter=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n = n_try
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            h_opt2, c5_opt2 = directed_pair_search(h_opt, len(h_opt), min(15, int(rem/4)), beta=500, rng=rng)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2
                n = len(h_opt2)

    # Phase 8: Final SLSQP polish on best
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        n_f = len(best_h)
        h_opt, c5_opt = slsqp_optimize(best_h, [2000, 10000, 50000], max_iter=500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 9: Final fine pair search
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        n_f = len(best_h)
        n_sweeps = min(10, int(rem / 3))
        h_opt, c5_opt = directed_pair_search(best_h, n_f, n_sweeps, beta=5000, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 10: Final swap search
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        n_f = len(best_h)
        n_iters = min(80000, int(rem * 60))
        h_opt, c5_opt = swap_local_search(best_h, n_f, n_iters, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
