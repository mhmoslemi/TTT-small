where z = [h_0, ..., h_{n-1}, t].

The gradient:
df/dt = 1 - μ1 * sum_d max(0, C_d - t)
df/dh[i] = μ1 * sum_d max(0, C_d - t) * ∂C_d/∂h[i] + μ2 * (sum(h) - n/2)

This is more expensive per evaluation (n variables vs m variables) but gives more freedom.

import numpy as np

import time

def project_to_target(x, target_sum):

            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_grad_h(h, d, n):
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return g


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def sym_grad(h, d, n):
    m = n // 2
    gh = corr_grad_h(h, d, n)
    return gh[:m] - gh[n - 1:m - 1:-1]


def select_active(h, n, frac=0.5, margin=0.01, pad=2):
    cv = corr_full(h) * (2.0 / n)
    mx = cv.max()
    thresh = mx - max(1e-8, margin * mx)
    idx = np.where(cv >= thresh)[0]
    for a in list(idx):
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k <= 2 * n - 2:
                k = int(k)
                if k not in idx:
                    idx = np.append(idx, k)
    return np.sort(np.unique(idx))


class SymModel:
    def __init__(self, n, active_k, mu):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.ak = active_k
        self.mu = mu

    def fg(self, z):
        q = z[:self.m]
        t = z[self.m]
        h = build_sym(q, self.n)
        cv = corr_full(h) * self.dx
        vals = cv[self.ak]
        exc = np.maximum(0.0, vals - t)
        f = t + 0.5 * self.mu * np.dot(exc, exc)
        gf = np.zeros(self.m + 1)
        gf[self.m] = 1.0 - self.mu * exc.sum()
        gq = np.zeros(self.m)
        for i in range(len(self.ak)):
            if exc[i] > 0.0:
                d = self.ak[i] - (self.n - 1)
                gq += self.mu * exc[i] * sym_grad(h, d, self.n)
        gf[:self.m] = gq
        return f, gf


class FullModel:
    def __init__(self, n, active_k, mu, mu_sum):
        self.n = n
        self.dx = 2.0 / n
        self.ak = active_k
        self.mu = mu
        self.mu_sum = mu_sum
        self.target = n / 2.0

    def fg(self, z):
        h = z[:self.n]
        t = z[self.n]
        cv = corr_full(h) * self.dx
        vals = cv[self.ak]
        exc = np.maximum(0.0, vals - t)
        sdev = h.sum() - self.target
        f = t + 0.5 * self.mu * np.dot(exc, exc) + 0.5 * self.mu_sum * sdev * sdev
        gf = np.zeros(self.n + 1)
        gf[self.n] = 1.0 - self.mu * exc.sum()
        gh = np.zeros(self.n)
        for i in range(len(self.ak)):
            if exc[i] > 0.0:
                d = self.ak[i] - (self.n - 1)
                gh += self.mu * exc[i] * corr_grad_h(h, d, self.n)
        gf[:self.n] = gh + self.mu_sum * sdev
        return f, gf


def opt_sym(q0, n, active_k, mu, maxiter=300, gtol=1e-9):
    m = n // 2
    h0 = build_sym(q0, n)
    t0 = eval_c5(h0)
    model = SymModel(n, active_k, mu)
    z0 = np.concatenate([q0, [t0]])
    bounds = [(0.0, 1.0)] * m + [(0.0, t0 + 0.5)]
    try:
        res = minimize(model.fg, z0, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol})
        q_opt = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q0, eval_c5(h0)
    h_opt = build_sym(q_opt, n)
    return q_opt, eval_c5(h_opt)


def opt_full(h0, n, active_k, mu, mu_sum=500.0, maxiter=200, gtol=1e-9):
    model = FullModel(n, active_k, mu, mu_sum)
    t0 = eval_c5(h0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.5)]

                       bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol})

        h_opt = project_to_target(h_opt, n / 2.0)
    except Exception:
        return h0, eval_c5(h0)
    return h_opt, eval_c5(h_opt)


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)
