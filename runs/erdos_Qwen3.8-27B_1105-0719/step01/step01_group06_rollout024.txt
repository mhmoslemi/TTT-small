import numpy as np
import time

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def profile_gradient(h):
    """Compute d(profile[k])/d(h) for all k. Returns (M, n) array."""
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((M, n))
    for k in range(M):
        if k < n - 1:
            d = (n - 1) - k
            G[k, 0:n-d] = dx * (1.0 - h[d:n])
            G[k, d:n] = -dx * h[0:n-d]
        elif k > n - 1:
            d = k - (n - 1)
            G[k, d:n] = dx * (1.0 - h[0:n-d])
            G[k, 0:n-d] = -dx * h[d:n]
        else:
            G[k, :] = dx * (1.0 - 2.0 * h)
    return G

def lse_objective(h, beta):
    prof = get_profile(h)
    pmax = np.max(prof)
    if pmax > 50.0 / beta:
        return beta * pmax  # overflow guard (won't happen in practice)
    shifted = beta * (prof - pmax)
    shifted = np.clip(shifted, -60, 60)
    logsum = np.log(np.sum(np.exp(shifted)))
    return pmax + logsum / beta

def lse_gradient(h, beta):
    prof = get_profile(h)
    pmax = np.max(prof)
    shifted = beta * (prof - pmax)
    shifted = np.clip(shifted, -60, 60)
    w = np.exp(shifted)
    w /= w.sum()
    G = profile_gradient(h)
    return w @ G

def lse_step(h, beta, max_iter, time_limit, lr_init=0.01):
    """Accelerated gradient descent on LSE objective."""
    t0 = time.perf_counter()
    n = len(h)
    x = h.copy()
    v = np.zeros(n)  # momentum
    best_x = x.copy()
    best_f = lse_objective(x, beta)
    lr = lr_init
    momentum = 0.9
    
    for it in range(max_iter):
        if time.perf_counter() - t0 > time_limit:
            break
        
        f = lse_objective(x, beta)
        if f < best_f - 1e-14:
            best_f = f
            best_x = x.copy()
        
        grad = lse_gradient(x, beta)
        # Remove mean component (to preserve sum constraint approximately)
        grad -= grad.mean()
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        x_new = x + v
        
        # Project to [0, 1] and repair sum
        x_new = np.clip(x_new, 0.0, 1.0)
        x_new = repair_feasible(x_new)
        
        f_new = lse_objective(x_new, beta)
        
        # Backtracking if not improving
        if f_new > f + 1e-12:
            lr = max(lr * 0.5, 1e-8)
            v = momentum * v
            x_new = repair_feasible(x + v)
            f_new = lse_objective(x_new, beta)
            if f_new > f + 1e-12:
                x_new = x  # revert
                v *= 0.5
                continue
        
        x = x_new
        
        # Reduce momentum slightly over time
        momentum = max(0.5, momentum * 0.999)
    
    return best_x

def coordinate_exchange_step(h, score, n_pairs, time_limit):
    """Try coordinate pair exchanges with golden-section line search."""
    t0 = time.perf_counter()
    n = len(h)
    best_h = h.copy()
    best_score = score
    rng = np.random.default_rng(42)
    
    # Precompute gradient for guidance
    prof = get_profile(h)
    active = np.argsort(prof)[::-1][:min(20, len(prof))]
    G = profile_gradient(h)
    # Weighted gradient from top lags
    w = prof[active] - prof[active].min() + 1e-10
    w /= w.sum()
    grad_dir = w @ G[active]
    grad_dir -= grad_dir.mean()
    
    for trial in range(n_pairs):
        if time.perf_counter() - t0 > time_limit:
            break
        
        # Choose pair: mix of gradient-guided and random
        if trial % 3 == 0:
            # Gradient-guided: increase low-gradient, decrease high-gradient
            inc_idx = np.argmin(grad_dir)
            dec_idx = np.argmax(grad_dir)
            if inc_idx == dec_idx:
                inc_idx = (inc_idx + 1) % n
        else:
            inc_idx = rng.integers(n)
            dec_idx = rng.integers(n)
            if inc_idx == dec_idx:
                dec_idx = (dec_idx + 1) % n
        
        i, j = inc_idx, dec_idx
        if best_h[i] >= 1.0 - 1e-12 or best_h[j] <= 1e-12:
            continue
        
        # Feasible interval for alpha
        alpha_lo = max(-best_h[i], best_h[j] - 1.0)
        alpha_hi = min(1.0 - best_h[i], best_h[j])
        if alpha_hi - alpha_lo < 1e-10:
            continue
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = alpha_lo, alpha_hi
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        
        def eval_alpha(al):
            h_trial = best_h.copy()
            h_trial[i] += al
            h_trial[j] -= al
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = repair_feasible(h_trial)
            return exact_score(h_trial)
        
        fc = eval_alpha(c)
        fd = eval_alpha(d)
        
        for _ in range(12):
            if b - a < 1e-8:
                break
            if fc < fd:
                b = d
                d = c
                fd = fc
                c = b - gr * (b - a)
                fc = eval_alpha(c)
            else:
                a = c
                c = d
                fc = fd
                d = a + gr * (b - a)
                fd = eval_alpha(d)
        
        alpha_best = (a + b) / 2.0
        h_trial = best_h.copy()
        h_trial[i] += alpha_best
        h_trial[j] -= alpha_best
        h_trial = np.clip(h_trial, 0.0, 1.0)
        h_trial = repair_feasible(h_trial)
        s_trial = exact_score(h_trial)
        
        if s_trial < best_score - 1e-12:
            best_h = h_trial
            best_score = s_trial
            # Update gradient direction periodically
            if trial % 10 == 0:
                prof = get_profile(best_h)
                active = np.argsort(prof)[::-1][:min(20, len(prof))]
                G = profile_gradient(best_h)
                w = prof[active] - prof[active].min() + 1e-10
                w /= w.sum()
                grad_dir = w @ G[active]
                grad_dir -= grad_dir.mean()
    
    return best_h, best_score

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def random_perturbation(h, eps, seed=0):
    rng = np.random.default_rng(seed)
    z = rng.standard_normal(len(h))
    z -= z.mean()
    return repair_feasible(h + eps * z)

def binary_threshold(h):
    """Threshold h to binary (top n/2 values = 1)."""
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)[:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
