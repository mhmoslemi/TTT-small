def softmax_objective_and_grad(z, beta):

    n = len(h)
    dx = 2.0 / n
    
    # Compute all C_l values
    corr = np.correlate(h, 1.0 - h, mode="full")
    C = corr * dx
    L = 2 * n - 1
    
    # Soft-max weights
    beta_C = beta * C
    max_bc = np.max(beta_C)
    exp_vals = np.exp(beta_C - max_bc)
    w = exp_vals / np.sum(exp_vals)
    
    # Soft-max value
    J = (np.log(np.sum(exp_vals)) + max_bc) / beta
    
    # Gradient of C_l w.r.t. h
    # G_h[j] = dx * sum_l w_l * (dC_l/dh_j)
    G_h = np.zeros(n)
    for l in range(L):
        k = l - (n - 1)  # lag
        wl = w[l]
        if wl < 1e-15:
            continue
        if k >= 0:
            # h_j appears as h[i+k] when k <= j <= n-1: contribute (1-h_{j-k})
            # h_j appears in (1-h_i) when 0 <= j <= n-1-k: contribute -h_{j+k}
            if k <= n - 1:
                idx1 = np.arange(k, n)  # j values where h_j = h[i+k]
                G_h[idx1] += wl * dx * (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)  # j values where h_j in (1-h_i)
            G_h[idx2] += wl * dx * (-h[idx2 + k])
        else:
            m = -k
            # h_j appears as h[i-m] when 0 <= j <= n-1-m: contribute (1-h_{j+m})
            idx1 = np.arange(0, n - m)
            G_h[idx1] += wl * dx * (1.0 - h[idx1 + m])
            # h_j appears in (1-h_i) when m <= j <= n-1: contribute -h_{j-m}
            idx2 = np.arange(m, n)
            G_h[idx2] += wl * dx * (-h[idx2 - m])
    
    # Chain rule: G_z[j] = G_h[j] - sigma_prime[j] * sum(sigma_prime * G_h) / T
    s = np.sum(sigma_prime * G_h)
    G_z = G_h - sigma_prime * s / T

        for alpha in [0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001]:
            trial = repair_feasible(best + alpha * direction)
            sc = exact_score(trial)
            if sc < best_score - 1e-14:
                best = trial
                best_score = sc
                improved = True
                break
        if not improved:
            break
    return best, best_score


def symmetrize(h):
    return repair_feasible((h + h[::-1]) / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.perf_counter() - t_start)

    try:
        best_h = repair_feasible(np.asarray(initial_h_values, dtype=np.float64))
    except Exception:
        best_h = np.full(100, 0.5)

    best_score = exact_score(best_h)
    n0 = len(best_h)

    # Try symmetrization
    h_sym = symmetrize(best_h)
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-14:
        best_h = h_sym
        best_score = s_sym

    z = h_to_logit(best_h)

    # Phase 1: Beta-annealed logit optimization at parent dimension
    for beta in [8, 25, 60, 150, 400]:
        if remaining() < 35:
            break
        try:
            res = minimize(_obj_call, z, args=(beta,), jac=_grad_call,
                           method='L-BFGS-B',
                           options={'maxiter': 400, 'gtol': 1e-11, 'maxls': 50})
            z_new = res.x
        except Exception:
            continue
        h_new = logit_project(z_new)[0]
        h_new = repair_feasible(h_new)
        sc = exact_score(h_new)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc
        z = z_new

    # Phase 2: Active descent at parent dimension
    if remaining() > 25:
        h_ad, sc_ad = active_descent(best_h, max_steps=60)
        if sc_ad < best_score - 1e-14:
            best_h = h_ad
            best_score = sc_ad
            z = h_to_logit(best_h)

    # Phase 3: Higher dimension exploration
    if remaining() > 80:
        dims = []
        for dn in [50, 100, 150]:
            nd = n0 + dn
            if nd <= 350:
                dims.append(nd)
        if 2 * n0 <= 350:
            dims.append(2 * n0)
        for n_tgt in dims:
            if remaining() < 40:
                break
            h_interp = transfer_dim(best_h, n_tgt)
            z_interp = h_to_logit(h_interp)
            for beta in [50, 200]:
                if remaining() < 25:
                    break
                try:
                    res = minimize(_obj_call, z_interp, args=(beta,), jac=_grad_call,
                                   method='L-BFGS-B',
                                   options={'maxiter': 250, 'gtol': 1e-11, 'maxls': 40})
                    z_new = res.x
                except Exception:
                    continue
                h_new = logit_project(z_new)[0]
                h_new = repair_feasible(h_new)
                sc_hi = exact_score(h_new)
                # Also check at original dimension
                h_back = transfer_dim(h_new, n0)
                sc_back = exact_score(h_back)
                if sc_hi < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc_hi
                    n0 = n_tgt
                    z = z_new
                elif sc_back < best_score - 1e-14:
                    best_h = h_back
                    best_score = sc_back
                    z = h_to_logit(best_h)
                else:
                    z = z_new

    # Phase 4: Logit optimization at high beta after dimension transfer
    if remaining() > 40:
        n_cur = len(best_h)
        z = h_to_logit(best_h)
        for beta in [300, 800]:
            if remaining() < 25:
                break
            try:
                res = minimize(_obj_call, z, args=(beta,), jac=_grad_call,
                               method='L-BFGS-B',
                               options={'maxiter': 300, 'gtol': 1e-12, 'maxls': 40})
                z_new = res.x
            except Exception:
                continue
            h_new = logit_project(z_new)[0]
            h_new = repair_feasible(h_new)
            sc = exact_score(h_new)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc
            z = z_new

    # Phase 5: Active descent
    if remaining() > 20:
        h_ad, sc_ad = active_descent(best_h, max_steps=50)
        if sc_ad < best_score - 1e-14:
            best_h = h_ad
            best_score = sc_ad
            z = h_to_logit(best_h)

    # Phase 6: Multi-start with perturbations
    if remaining() > 50:
        for trial in range(6):
            if remaining() < 30:
                break
            eps = rng.uniform(0.003, 0.015)
            p = len(best_h)
            z_pert = z + eps * rng.randn(p)
            for beta in [100, 300]:
                if remaining() < 20:
                    break
                try:
                    res = minimize(_obj_call, z_pert, args=(beta,), jac=_grad_call,
                                   method='L-BFGS-B',
                                   options={'maxiter': 150, 'gtol': 1e-11, 'maxls': 30})
                    z_new = res.x
                except Exception:
                    continue
                h_new = logit_project(z_new)[0]
                h_new = repair_feasible(h_new)
                sc = exact_score(h_new)
                if sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc
                    z = z_new
                else:
                    z = z_new

    # Phase 7: Final active descent
    if remaining() > 15:
        h_ad, sc_ad = active_descent(best_h, max_steps=40)
        if sc_ad < best_score - 1e-14:
            best_h = h_ad
            best_score = sc_ad

    # Phase 8: One more high-beta logit pass
    if remaining() > 20:
        z = h_to_logit(best_h)
        try:
            res = minimize(_obj_call, z, args=(600,), jac=_grad_call,
                           method='L-BFGS-B',
                           options={'maxiter': 200, 'gtol': 1e-12, 'maxls': 40})
            z_new = res.x
            h_new = logit_project(z_new)[0]
            h_new = repair_feasible(h_new)
            sc = exact_score(h_new)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc
        except Exception:
            pass

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
