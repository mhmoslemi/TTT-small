    P = profile_full(h)
    T = P.max()
    A_idx = np.where(P > T - epsilon)[0]
    
    if len(A_idx) == 0:
        return None, None
    
    # Compute gradients for active lags
    # g_k is the gradient of P[k] w.r.t. h
    m = len(A_idx)
    G = np.zeros((m, n))  # G[k, :] = gradient of P at lag A_idx[k]
    
    for ki, idx in enumerate(A_idx):
        k = idx - (n - 1)  # lag
        g = np.zeros(n)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif k > 0:
            g[k:n] += (1.0 - h[0:n-k])
            g[0:n-k] -= h[k:n]
        else:
            a = -k
            g[0:n-a] += (1.0 - h[a:n])
            g[a:n] -= h[0:n-a]
        dx = 2.0 / n
        G[ki] = g * dx
    
    # LP: minimize t
    # vars: [d_0, ..., d_{n-1}, t]
    # constraints: P[k] + G[k,:] @ d <= t  =>  G[k,:] @ d - t <= -P[k]
    # sum(d) = 0
    # lb_i <= d_i <= ub_i
    # no bound on t (or t >= 0)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # G[k,:] @ d - t <= -P[k]
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P[A_idx]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    lb = np.maximum(-trust_r, -h)
    ub = np.minimum(trust_r, 1.0 - h)
    bounds = list(zip(lb, ub)) + [(None, None)]  # t is unbounded
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    t_val = res.x[n]

    h_new = project_box_sum(h + d)
    c_new = exact_objective(h_new)

def smooth_obj_grad(x, tau, n):
    dx = 2.0 / n
    raw = np.correlate(x, 1.0 - x, mode='full')
    r_max = np.max(raw)
    shifted = np.clip((raw - r_max) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    value = dx * (r_max + tau * log_sum)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)

            cur[j2] = 1.0
            c = exact_objective(cur)
            delta = cur_c - c
            if delta > best_delta:
                best_delta = delta
                best_move = (i, i2, j, j2, delta)
            cur[i] = 1.0
            cur[i2] = 1.0
            cur[j] = 0.0
            cur[j2] = 0.0
        if best_move is not None:
            i, i2, j, j2, delta = best_move
            cur[i] = 0.0
            cur[i2] = 0.0
            cur[j] = 1.0
            cur[j2] = 1.0
            cur_c -= delta
        else:
            break
    return cur, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
        parent = project_box_sum(parent)
        pc = exact_objective(parent)
        best_h = parent.copy()
        best_c = pc
        best_n = len(parent)
    except Exception:
        n0 = 140
        best_h = np.ones(n0) * 0.5
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)
        best_n = n0

    def time_left():
        return deadline - time.time()

    # Phase 1: Pair optimization from parent
    if time_left() > 60:
        h_cur = best_h.copy()
        for round_id in range(200):
            if time_left() < 60:
                break
            c_before = exact_objective(h_cur)
            c_after = pair_pass(h_cur, 3000, rng)
            if c_after < best_c - 1e-13:
                best_c = c_after
                best_h = h_cur.copy()
                best_n = len(h_cur)
            if c_before - c_after < 1e-12:
                if round_id > 20:
                    break

    # Phase 2: Smoothed SLSQP optimization
    if time_left() > 120:
        n = best_n
        h_cur = best_h.copy()
        tau_stages = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001]
        for tau in tau_stages:
            if time_left() < 60:
                break
            result = slsqp_smooth(h_cur, tau, maxiter=80)
            if result is not None:
                c_new = exact_objective(result)
                if c_new < exact_objective(h_cur) - 1e-12:
                    h_cur = result
            # Interleave pair optimization
            c_after = pair_pass(h_cur, 2000, rng)
            c_cur2 = exact_objective(h_cur)
            if c_cur2 < best_c - 1e-13:
                best_c = c_cur2
                best_h = h_cur.copy()
                best_n = len(h_cur)

    # Phase 3: Active-set LP
    if time_left() > 60:
        h_cur = best_h.copy()
        c_cur = exact_objective(h_cur)
        for round_id in range(10):
            if time_left() < 30:
                break
            h_new, c_new = active_set_lp(h_cur, epsilon=0.003, trust_r=0.05, max_lp_iter=5)
            if c_new < c_cur - 1e-14:
                h_cur = h_new
                c_cur = c_new
            else:
                h_new2, c_new2 = active_set_lp(h_cur, epsilon=0.001, trust_r=0.02, max_lp_iter=3)
                if c_new2 < c_cur - 1e-14:
                    h_cur = h_new2
                    c_cur = c_new2
                else:
                    break
            c_after = pair_pass(h_cur, 1500, rng)
            c_cur = exact_objective(h_cur)
            if c_cur < best_c - 1e-13:
                best_c = c_cur
                best_h = h_cur.copy()
                best_n = len(h_cur)

    # Phase 4: More pair optimization
    if time_left() > 60:
        h_cur = best_h.copy()
        for round_id in range(400):
            if time_left() < 30:
                break
            c_before = exact_objective(h_cur)
            pair_pass(h_cur, 5000, rng)
            c_after = exact_objective(h_cur)
            if c_after < best_c - 1e-13:
                best_c = c_after
                best_h = h_cur.copy()
                best_n = len(h_cur)
            if c_before - c_after < 1e-13:
                if round_id > 50:
                    break

    # Phase 5: Binary local search
    if time_left() > 90 and best_n % 2 == 0:
        h_cur = best_h.copy()
        b, b_c = binary_2swap_search(h_cur, max_rounds=300, rng_seed=seed)
        if b_c < best_c - 1e-13:
            best_c = b_c
            best_h = b.copy()
            best_n = len(b)
        # Also try: optimize the binary solution with pair moves (allows fractional)
        h_frac = b.astype(np.float64)
        c_pair = pair_pass(h_frac, 5000, rng)
        if c_pair < best_c - 1e-13:
            best_c = c_pair
            best_h = h_frac.copy()
            best_n = len(h_frac)

    # Phase 6: Multi-resolution
    if time_left() > 120:
        n_base = best_n
        try_sizes = [n_base]
        for s in [n_base - 10, n_base - 5, n_base + 5, n_base + 10, n_base + 20, n_base - 20]:
            if 30 <= s <= 400 and s % 2 == 0 and s not in try_sizes:
                try_sizes.append(s)
        for n_try in try_sizes:
            if time_left() < 60:
                break
            if n_try == n_base:
                continue
            h_try = transfer_to_size(best_h, n_try)
            c_try = exact_objective(h_try)
            c_try = pair_pass(h_try, 3000, rng)
            h_try = slsqp_smooth(h_try, 0.005, maxiter=60)
            if h_try is not None:
                c_try = exact_objective(h_try)
                c_try = pair_pass(h_try, 3000, rng)
                c_try = exact_objective(h_try)
            if time_left() > 30:
                h_new_try, c_new_try = active_set_lp(h_try, epsilon=0.003, trust_r=0.04, max_lp_iter=4)
                if c_new_try < c_try:
                    h_try = h_new_try
                    c_try = c_new_try
                    c_try = pair_pass(h_try, 2000, rng)
                    c_try = exact_objective(h_try)
            if c_try < best_c - 1e-13:
                best_c = c_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 7: Final polish
    if time_left() > 30:
        h_cur = best_h.copy()
        pair_pass(h_cur, 5000, rng)
        h_new, c_new = active_set_lp(h_cur, epsilon=0.002, trust_r=0.03, max_lp_iter=4)
        if c_new < exact_objective(h_cur):
            h_cur = h_new
            pair_pass(h_cur, 3000, rng)
        c_final = exact_objective(h_cur)
        if c_final < best_c - 1e-13:
            best_c = c_final
            best_h = h_cur.copy()

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
