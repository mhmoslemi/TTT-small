def safe_h(h, idx):
    n = len(h)
    mask = (idx >= 0) & (idx < n)
    out = np.zeros(len(idx), dtype=np.float64)
    out[mask] = h[idx[mask]]
    return out

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_objective(h):
    n = len(h)
    profile = compute_profile(h)
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)

        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

    n = len(h)

    for k in range(n):

    return F, m

    neg_mask = d < -1e-10

        hi_val = U

            lo_val = max(lo_val, float(np.max(vals)))
        if np.any(pos_mask):
            vals = (C - p0[pos_mask]) / d[pos_mask]
            hi_val = min(hi_val, float(np.min(vals)))
        return lo_val <= hi_val

    C_hi = C_current

        else:

    else:

                h_new[i] += lam

            h[j] -= best_lambda
            h = project_box_sum(h, target)
            p0 = compute_profile(h)
            c5 = float(np.max(p0))
        else:
            break

import numpy as np
from scipy.optimize import minimize
import time

def safe_h(h, idx):
    n = len(h)
    mask = (idx >= 0) & (idx < n)
    out = np.zeros(len(idx), dtype=np.float64)
    out[mask] = h[idx[mask]]
    return out

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    profile = compute_profile(h)
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def compute_F(h):
    n = len(h)
    m = np.arange(-(n-1), n)
    F = np.zeros((n, 2*n-1), dtype=np.float64)
    for k in range(n):
        F[k] = (1.0 - safe_h(h, k - m)) - safe_h(h, k + m)
    return F

def find_optimal_lambda(p0, d, L, U):
    if U - L < 1e-12:
        return 0.0
    neg_mask = d < -1e-12
    pos_mask = d > 1e-12
    if not np.any(neg_mask) and not np.any(pos_mask):
        return 0.0
    C_current = float(np.max(p0))
    min_per_m = np.where(pos_mask, p0 + d*U, np.where(neg_mask, p0 + d*L, p0))
    C_lb = float(np.max(min_per_m))
    if C_current <= C_lb + 1e-12:
        return 0.0
    C_lo = C_lb
    C_hi = C_current
    for _ in range(50):
        C_mid = (C_lo + C_hi) / 2.0
        lo_val = L
        hi_val = U
        if np.any(neg_mask):
            vals = (C_mid - p0[neg_mask]) / d[neg_mask]
            lv = float(np.max(vals))
            if lv > lo_val:
                lo_val = lv
        if np.any(pos_mask):
            vals = (C_mid - p0[pos_mask]) / d[pos_mask]
            hv = float(np.min(vals))
            if hv < hi_val:
                hi_val = hv
        if lo_val <= hi_val:
            C_hi = C_mid
        else:
            C_lo = C_mid
        if C_hi - C_lo < 1e-13:
            break
    lo_val = L
    hi_val = U
    if np.any(neg_mask):
        vals = (C_hi - p0[neg_mask]) / d[neg_mask]
        lv = float(np.max(vals))
        if lv > lo_val:
            lo_val = lv
    if np.any(pos_mask):
        vals = (C_hi - p0[pos_mask]) / d[pos_mask]
        hv = float(np.min(vals))
        if hv < hi_val:
            hi_val = hv
    if lo_val <= hi_val:
        lam = (lo_val + hi_val) / 2.0
    else:
        lam = 0.0
    if lam < L:
        lam = L
    elif lam > U:
        lam = U
    return lam

def pair_swap_optimize(h, deadline, max_rounds=500, verbose=False):
    n = len(h)
    h = h.copy()
    target = 0.5 * n
    p0 = compute_profile(h)
    c5 = float(np.max(p0))
    prev_c5 = c5
    stall_count = 0
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        F = compute_F(h)
        best_improvement = 0.0
        best_pair = None
        best_lambda = 0.0
        for i in range(n):
            for j in range(i+1, n):
                L = max(-h[i], h[j] - 1.0)
                U = min(1.0 - h[i], h[j])
                if U - L < 1e-10:
                    continue
                d = F[i] - F[j]
                if np.max(np.abs(d)) < 1e-10:
                    continue
                lam = find_optimal_lambda(p0, d, L, U)
                if abs(lam) < 1e-10:
                    continue
                h_new = h.copy()
                h_new[i] += lam
                h_new[j] -= lam
                h_new = project_box_sum(h_new, target)
                p_new = compute_profile(h_new)
                c5_new = float(np.max(p_new))
                if c5_new < c5 - 1e-12:
                    improvement = c5 - c5_new
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_pair = (i, j)
                        best_lambda = lam
        if best_pair is not None:
            i, j = best_pair
            h[i] += best_lambda
            h[j] -= best_lambda
            h = project_box_sum(h, target)
            p0 = compute_profile(h)
            c5 = float(np.max(p0))
            if c5 < prev_c5 - 1e-10:
                prev_c5 = c5
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 20:
                break
        else:
            break
    return h, c5

def _gradient_lag(x, lag, dx):
    n = len(x)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= x[lag:]
            grad[lag:] += 1.0 - x[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - x[a:]
            grad[a:] -= x[:s]
    return grad * dx

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = float(np.max(profile))
    if tau < 1e-12:
        m_idx = int(np.argmax(profile))
        lag = m_idx - (n - 1)
        grad = _gradient_lag(x, lag, dx)
        return pmax, grad
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    value = pmax + tau * np.log(wsum)
    wnorm = weights / wsum
    grad = np.zeros(n)
    active = wnorm > 1e-7
    for m in np.where(active)[0]:
        lag = m - (n - 1)
        grad += wnorm[m] * _gradient_lag(x, lag, dx)
    return value, grad

def slsqp_refine(h_start, tau_list, deadline, max_iter_per_tau=200):
    n = len(h_start)
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c = exact_objective(best_h)
    bounds = [(0.0, 1.0)] * n
    target = 0.5 * n

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

    for tau in tau_list:
        if time.time() > deadline - 5:
            break

        def obj_grad(x):
            val, grad = smooth_obj_and_grad(x, tau)
            se = np.sum(x) - target
            val += 1e5 * se * se
            grad = grad + 2e5 * se * np.ones(n)
            below = np.maximum(0.0, -x)
            above = np.maximum(0.0, x - 1.0)
            val += 1e5 * (np.sum(below * below) + np.sum(above * above))
            grad = grad - 2e5 * below + 2e5 * above
            return val, grad

        try:
            mi = max(50, min(max_iter_per_tau, n * 3 + 30))
            res = minimize(obj_grad, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraint, jac=True,
                           options={'maxiter': mi, 'ftol': 1e-14})
            if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                c = exact_objective(cand)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = cand
        except Exception:
            pass
    return best_h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=80, rng=None):
    n = len(b)
    if n % 2 != 0 or b is None:
        return b, exact_objective(b) if b is not None else (float('inf'), 0)
    if rng is None:
        rng = np.random.RandomState(0)
    c5 = exact_objective(b)
    for rnd in range(max_rounds):
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 4000:
            for i in ones:
                new_improved = False
                for j in zeros:
                    b[i] = 0.0
                    b[j] = 1.0
                    nc = exact_objective(b)
                    if nc < c5 - 1e-13:
                        c5 = nc
                        improved = True
                        new_improved = True
                        ones = list(np.where(b > 0.5)[0])
                        zeros = list(np.where(b < 0.5)[0])
                        break
                    b[i] = 1.0
                    b[j] = 0.0
                if new_improved:
                    break
        else:
            n_tries = 500
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc = exact_objective(b)
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = list(np.where(b > 0.5)[0])
                    zeros = list(np.where(b < 0.5)[0])
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        cnt = int(round(target))
        idx = rng.choice(n, size=min(cnt, n), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(2, 6)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[2 * n // 3: 2 * n // 3 + q] = 1.0
    elif mode == 5:
        h = np.full(n, 0.5) + rng.randn(n) * 0.15
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)
