                g[:s] += 1.0 - x[a:]
                g[a:] -= x[:s]
        grad += wnorm[m] * g * dx
    return value, grad

def simulated_annealing_binary(b, rng, time_limit=30, max_rounds=100000):
    """Simulated annealing on binary vectors."""
    n = len(b)

    T_end = 0.0001
    n_iter = 0
    
    while n_iter < max_rounds and (time.time() - t_start) < time_limit:
        n_iter += 1
        # Temperature schedule (exponential cooling)
        progress = min(1.0, n_iter / max_rounds)
        T = T_start * (T_end / T_start) ** progress
        
        # Random swap
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        b[j] = 1.0
        c_new, _ = exact_objective(b)

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c_cur = c_new
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0

def project_box_sum(h, target=None):

        best_h = make_start(n0, rng, mode=0)
        best_c, _ = exact_objective(best_h)

    n0 = len(best_h)

    # Phase 1: L-BFGS-B continuous refinement
    if time.time() < deadline - 100:
        h_opt, c_opt = lbfgsb_refine(best_h, n0, [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002], deadline, max_iter=250)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Binary + Simulated Annealing (multiple restarts)
    if n0 % 2 == 0 and time.time() < deadline - 80:
        b = to_binary(best_h)
        b, c_b = greedy_binary(b, rng, max_rounds=50, time_limit=5)
        b, c_b = sa_binary(b, rng, time_limit=25, max_iter=60000, t0_temp=0.006, t1_temp=0.00004)
        if c_b < best_c:
            best_c = c_b
            best_h = b.copy()

        # Restart SA from perturbed versions
        for restart in range(3):
            if time.time() > deadline - 60:
                break
            bp = perturb_binary(best_h, rng, n_swaps=3 + restart * 2)
            bp, cb = greedy_binary(bp, rng, max_rounds=30, time_limit=3)
            bp, cb = sa_binary(bp, rng, time_limit=10, max_iter=30000, t0_temp=0.004, t1_temp=0.00003)
            if cb < best_c:
                best_c = cb
                best_h = bp

    # Phase 3: Continuous L-BFGS-B from best binary
    if time.time() < deadline - 60:
        h_opt, c_opt = lbfgsb_refine(best_h, n0, [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005], deadline, max_iter=300)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

        # Another round of binary refinement
        if n0 % 2 == 0 and time.time() < deadline - 30:
            b = to_binary(best_h)
            b, cb = greedy_binary(b, rng, max_rounds=40, time_limit=3)
            b, cb = sa_binary(b, rng, time_limit=10, max_iter=40000, t0_temp=0.005, t1_temp=0.00003)
            if cb < best_c:
                best_c = cb
                best_h = b

    # Phase 4: Try other sizes
    sizes = [24, 28, 30, 32, 36, 40, 44, 48, 52, 56, 60, 64, 68, 72, 76, 80, 84, 88, 96]
    sizes.sort(key=lambda x: abs(x - n0))

    for n_try in sizes:
        if n_try == n0:
            continue
        if time.time() > deadline - 25:
            break

        for mode in [0, 2, 3, 5, 6, 1]:
            if time.time() > deadline - 12:
                break
            h_start = make_start(n_try, rng, mode=mode)
            if best_h is not None and len(best_h) != n_try:
                h_ds = downsample(best_h, n_try)
                if exact_objective(h_ds)[0] < exact_objective(h_start)[0]:
                    h_start = h_ds

            h_opt, c_opt = lbfgsb_refine(h_start, n_try, [0.01, 0.005, 0.002, 0.001], deadline, max_iter=150)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

            if n_try % 2 == 0 and c_opt < best_c + 0.003 and time.time() < deadline - 10:
                b = to_binary(h_opt)
                b, cb = greedy_binary(b, rng, max_rounds=30, time_limit=3)
                b, cb = sa_binary(b, rng, time_limit=6, max_iter=25000, t0_temp=0.005, t1_temp=0.00004)
                if cb < best_c:
                    best_c = cb
                    best_h = b

    # Phase 5: Final polish on the best solution found
    if time.time() < deadline - 8:
        n_best = len(best_h)
        h_opt, c_opt = lbfgsb_refine(best_h, n_best, [0.0005, 0.0002, 0.0001, 0.00005, 0.00002], deadline, max_iter=300)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

        if n_best % 2 == 0 and time.time() < deadline - 4:
            b = to_binary(best_h)
            b, cb = greedy_binary(b, rng, max_rounds=40, time_limit=2)
            b, cb = sa_binary(b, rng, time_limit=3, max_iter=20000, t0_temp=0.003, t1_temp=0.00002)
            if cb < best_c:
                best_c = cb
                best_h = b

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)

    return (best_h, final_c, len(best_h))
