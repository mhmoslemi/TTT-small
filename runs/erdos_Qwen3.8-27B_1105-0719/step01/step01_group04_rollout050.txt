            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - delta)
            values[n - 1 + delta] = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
            grads[n - 1 + delta, idx] += dx * (1.0 - h[idx + delta])
            grads[n - 1 + delta, idx + delta] += -dx * h[idx]
            idx2 = np.arange(delta, n)
            values[n - 1 - delta] = dx * np.sum(h[idx2] * (1.0 - h[idx2 - delta]))
            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
            grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    return values, grads

def project_feasible(h):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    n = len(h)
    target = n / 2.0
    # h_i = clip(h_i - mu, 0, 1) where mu is chosen so sum = target
    # Binary search on mu
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if h_proj.sum() > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h_proj = np.clip(h - mu, 0.0, 1.0)
    return repair_feasible(h_proj)

def solve_prox_step(h_cur, s, eta):
    """h_new = argmin (1/(2*eta))||h - h_cur||^2 + s·h s.t. 0<=h<=1, sum(h)=n/2
    = argmin (1/2)||h - (h_cur - eta*s)||^2 + constant s.t. constraints
    So it's just projecting (h_cur - eta*s) onto the feasible set!
    """
    h_unconstrained = h_cur - eta * s
    return project_feasible(h_unconstrained)

def minimax_proximal(h_init, max_iters=5000, eta_init=0.01, eta_min=0.0001,
                     tau_init=0.005, tau_min=0.0001, budget_fn=None, verbose=False):
    h = repair_feasible(h_init)
    best_h = h.copy()
    best_c = exact_score(h)
    n = len(h)
    eta = eta_init
    tau = tau_init
    no_improve = 0
    
    for it in range(max_iters):
        if budget_fn is not None and budget_fn() < 5:
            break
        
        values, grads = compute_lag_data(h)
        C = np.max(values)
        
        # Compute weights (softmax)
        shifted = (values - C) / tau
        w = np.exp(shifted)
        w_sum = w.sum()
        w /= w_sum
        
        s = w @ grads
        
        # Take proximal step
        h_new = solve_prox_step(h, s, eta)
        c_new = exact_score(h_new)
        
        improved = False
        if c_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            no_improve = 0
            eta = min(eta * 1.2, 0.05)
            improved = True
        elif c_new < C - 1e-8:
            h = h_new
            no_improve += 1
        else:
            no_improve += 1
            eta = max(eta / 2.0, eta_min)
            tau = max(tau * 0.9, tau_min)
        
        if no_improve > 100:
            # Try a different approach: random perturbation
            pert = np.random.uniform(-0.01, 0.01, n)
            h_try = repair_feasible(best_h + pert)
            c_try = exact_score(h_try)
            if c_try < best_c:
                best_h = h_try
                best_c = c_try
                h = h_try
                no_improve = 0
                eta = eta_init
            else:
                # Give up on this trajectory
                break
    
    return best_h, best_c

A better approach might be to use the **exact** gradient of a smoothed approximation:
f_τ(h) = τ * log(Σ_k exp(Q_k(h)/τ))

This is the log-sum-exp smoothing of the max. Its gradient is:
∇f_τ(h) = Σ_k (exp(Q_k/τ) / Σ_j exp(Q_j/τ)) * ∇Q_k(h)

Which is exactly the softmax-weighted average of gradients! So my method is doing gradient descent on the log-sum-exp smoothing. The nice property is that as τ→0, this approaches the true subgradient of the max.

For this smoothing, the convergence rate of gradient descent is O(τ² + 1/(T)) approximately (for smooth + convex parts). So smaller τ gives better final accuracy but slower per-step progress.

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h

def compute_lag_data(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    for delta in range(0, n):
        if delta == 0:
            values[n - 1] = dx * np.sum(h * (1.0 - h))
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - delta)
            values[n - 1 + delta] = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
            grads[n - 1 + delta, idx] += dx * (1.0 - h[idx + delta])
            grads[n - 1 + delta, idx + delta] += -dx * h[idx]
            idx2 = np.arange(delta, n)
            values[n - 1 - delta] = dx * np.sum(h[idx2] * (1.0 - h[idx2 - delta]))
            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
            grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    return values, grads

def project_feasible(p):
    n = len(p)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0

        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h_proj = np.clip(p - mu, 0.0, 1.0)
    return repair_feasible(h_proj)

def minimax_proximal(h_init, max_iters, tau_vals, eta_init, budget_fn):

    best_h = h.copy()
    best_c = exact_score(h)
    n = len(h)

    for tau in tau_vals:

def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def active_descent_polish(h_init, max_steps, budget_fn):
    h = repair_feasible(h_init)
    best_h = h.copy()
    best_c = exact_score(h)
    n = len(h)
    for step in range(max_steps):
        if budget_fn() < 10:
            break
        values, grads = compute_lag_data(h)
        C = np.max(values)
        n_active = min(20, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_active]
        gaps = np.maximum(C - values[active_idx], 0.0)
        tau_local = 0.003 * C + 1e-12
        weights = np.exp(-gaps / tau_local)
        weights /= weights.sum() + 1e-15
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        direction -= np.mean(direction)
        improved = False
        for alpha in [0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = repair_feasible(h + alpha * direction)
            score = exact_score(h_trial)
            if score < best_c - 1e-12:
                best_h = h_trial.copy()
                best_c = score
                h = h_trial
                improved = True
                break
            elif score < C - 1e-10:
                h = h_trial
        if not improved:
            for alpha in [0.005, 0.003, 0.001]:
                h_trial = repair_feasible(h - alpha * direction)
                score = exact_score(h_trial)
                if score < best_c - 1e-12:
                    best_h = h_trial.copy()
                    best_c = score
                    h = h_trial
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c
