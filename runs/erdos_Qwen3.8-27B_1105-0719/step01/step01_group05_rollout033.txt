        #   valid j: d ≤ j ≤ n-1
        #   Let k = j - d, so k ranges 0..n-d-1
        #   element = sum_k h[k+d] * (1-h[k]) * dx
        #   dq/dh[k+d] = dx * (1-h[k]) for k in 0..n-d-1
        #   dq/dh[k] = dx * (-h[k+d]) for k in 0..n-d-1
        
        s = n - a
        # h[j] * (1-h[j+a])
        grad[:s] += dx * (1.0 - h[a:a+s])  # derivative w.r.t. h[j] for j in 0..s-1
        grad[a:] -= dx * h[:s]  # derivative w.r.t. h[j+a] for j in 0..s-1
    return grad

def slsqp_direct(h_init, n, active_lags, c_init, deadline, max_iter=100, K_active=20):
    """Run SLSQP with direct inequality constraints for active lags."""
    target = 0.5 * n
    dx = 2.0 / n
    
    def obj_grad(z):
        h = z[:n]
        c = z[n]
        profile, _ = compute_profile_and_grads(h)
        # Objective: c
        obj = c
        grad = np.zeros(n + 1)
        grad[n] = 1.0
        return obj, grad
    
    # Equality constraint: sum(h) - target = 0
    def eq_fun(z):
        return np.sum(z[:n]) - target
    
    def eq_jac(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    # Inequality constraints: c - q_lag(h) >= 0 for each active lag
    # SLSQP convention: constraint function >= 0
    
    def make_ineq(lag):
        def fun(z):
            h = z[:n]
            c = z[n]
            dx = 2.0 / n
            # Compute q_lag
            profile = np.correlate(h, 1.0 - h, mode='full') * dx
            p = lag + (n - 1)  # convert to array index
            if 0 <= p < len(profile):
                return c - profile[p]
            return c
        def jac(z):
            h = z[:n]
            c = z[n]
            dx = 2.0 / n
            g = lag_gradient(h, lag, dx)
            j = np.zeros(n + 1)
            j[:n] = -g  # derivative of (c - q_lag) w.r.t. h is -grad(q_lag)
            j[n] = 1.0  # derivative w.r.t. c is 1
            return j
        return {'type': 'ineq', 'fun': fun, 'jac': jac}
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for lag in active_lags:
        constraints.append(make_ineq(lag))
    
    bounds = [(0.0, 1.0)] * n + [(c_init - 0.1, c_init)]
    
    z0 = np.concatenate([h_init, [c_init]])
    
    try:
        res = minimize(obj_grad, z0, method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=True,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], target)
            c_new, _ = exact_objective(h_new)
            return h_new, c_new
    except Exception:
        pass
    return h_init, exact_objective(h_init)[0]

            return np.sum(x) - target

        def eq_jac(x):
            return np.ones(n)

        constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
        mi = max(30, min(max_iter, n * 2 + 40))
        try:
            res = minimize(obj_grad, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraint, jac=True,
                           options={'maxiter': mi, 'ftol': 1e-13})
            if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                c, _ = exact_objective(cand)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = cand
        except Exception:
            pass
    return best_h, best_c

def slsqp_direct_constraints(h_start, n, active_lags, c_init, deadline, max_iter=80):
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, _ = exact_objective(best_h)
    target = 0.5 * n
    dx = 2.0 / n

    def obj_fun(z):
        return z[n]

    def obj_jac(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def eq_fun(z):
        return np.sum(z[:n]) - target

    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for lag in active_lags:
        p_idx = lag + (n - 1)
        if p_idx < 0 or p_idx >= 2 * n - 1:
            continue

        def make_ineq(lag_val, p_idx_val):
            def fun(z):
                h_z = z[:n]
                c_z = z[n]
                prof = np.correlate(h_z, 1.0 - h_z, mode='full') * (2.0 / n)
                return c_z - prof[p_idx_val]

            def jac(z):
                h_z = z[:n]
                g = lag_gradient(h_z, lag_val, 2.0 / n)
                j = np.zeros(n + 1)
                j[:n] = -g
                j[n] = 1.0
                return j

            return {'type': 'ineq', 'fun': fun, 'jac': jac}

        constraints.append(make_ineq(lag, p_idx))

    c_upper = c_init + 0.05
    bounds = [(0.0, 1.0)] * n + [(0.0, c_upper)]
    z0 = np.concatenate([best_h, [best_c + 1e-8]])

    try:
        res = minimize(obj_fun, z0, method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=obj_jac,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], target)
            c_new, _ = exact_objective(h_new)
            if c_new < best_c - 1e-14:
                return h_new, c_new
    except Exception:
        pass
    return best_h, best_c

def subgradient_descent(h_start, n, max_iter=200, deadline=None):
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, _ = exact_objective(best_h)
    target = 0.5 * n
    dx = 2.0 / n
    h = best_h.copy()

    for it in range(max_iter):
        if deadline and time.time() > deadline - 3:
            break
        profile = get_profile(h)
        pmax = np.max(profile)
        # Find active lags (within 5% of max)
        threshold = pmax * 0.95
        active_idx = np.where(profile >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(profile)])

        # Subgradient: average of gradients of active lags
        grad = np.zeros(n)
        for idx in active_idx:
            lag = idx - (n - 1)
            grad += lag_gradient(h, lag, dx)
        grad /= len(active_idx)

        # Remove mean component (to preserve sum)
        grad -= np.mean(grad)

        # Step size with decay
        alpha = 0.02 / (1.0 + 0.01 * it)

        # Try step
        for attempt in range(10):
            h_new = project_box_sum(h - alpha * grad, target)
            c_new, _ = exact_objective(h_new)
            if c_new < pmax - 0.5 * alpha * np.dot(grad, grad) * 0.1 or c_new < pmax:
                h = h_new
                break
            alpha *= 0.5

        c_curr, _ = exact_objective(h)
        if c_curr < best_c - 1e-14:
            best_c = c_curr
            best_h = h.copy()

    return best_h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=120, rng=None):
    n = len(b)
    if n % 2 != 0:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(0)
    c5, _ = exact_objective(b)
    for rnd in range(max_rounds):
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 5000:
            best_delta = 1e-14
            best_move = None
            for i in ones:
                for j in zeros:
                    b[i] = 0.0
                    b[j] = 1.0
                    nc, _ = exact_objective(b)
                    if nc < c5 - best_delta:
                        best_delta = c5 - nc
                        best_move = (i, j)
                    b[i] = 1.0
                    b[j] = 0.0
            if best_move is not None:
                b[best_move[0]] = 0.0
                b[best_move[1]] = 1.0
                c5 -= best_delta
                improved = True
        else:
            n_tries = 800
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc, _ = exact_objective(b)
                if nc < c5 - 1e-14:
                    c5 = nc
                    improved = True
                    ones = list(np.where(b > 0.5)[0])
                    zeros = list(np.where(b < 0.5)[0])
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5

def local_search_continuous(h, max_rounds=80, rng=None):
    n = len(h)
    target = 0.5 * n
    if rng is None:
        rng = np.random.RandomState(42)
    c_best, _ = exact_objective(h)
    h = h.copy()

    for rnd in range(max_rounds):
        improved = False
        for _ in range(300):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for amp in [0.01, 0.03, 0.05, 0.1, 0.2]:
                for sign in [1, -1]:
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + sign * amp, 0, 1)
                    h_new[j] = np.clip(h_new[j] - sign * amp, 0, 1)
                    h_new = project_box_sum(h_new, target)
                    c_new, _ = exact_objective(h_new)
                    if c_new < c_best - 1e-13:
                        c_best = c_new
                        h = h_new
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, c_best

def make_starts(n, rng, count=4):
    target = 0.5 * n
    starts = []
    h = np.zeros(n)
    h[::2] = 1.0
    starts.append(project_box_sum(h, target))
    h = np.zeros(n)
    h[:int(target)] = 1.0
    starts.append(project_box_sum(h, target))
    for _ in range(max(0, count - 3)):
        h = (rng.rand(n) > 0.5).astype(float)
        starts.append(project_box_sum(h, target))
    x = np.arange(n, dtype=float) / n
    h = 0.5 + 0.35 * np.cos(2 * np.pi * 2 * x + rng.uniform(0, 6.28))
    starts.append(project_box_sum(h, target))
    return starts

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def interp_between(ha, hb, n_alphas=10):
    n = len(ha)
    target = 0.5 * n
    best_h = project_box_sum(ha, target)
    best_c = exact_objective(best_h)[0]
    for a in np.linspace(0.1, 0.9, n_alphas):
        h = (1.0 - a) * ha + a * hb
        h = project_box_sum(h, target)
        c, _ = exact_objective(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 5
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load initial
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0, _ = exact_objective(init)
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        best_h = None

    if best_h is None:
        n0 = 70
        best_h = make_starts(n0, rng, 1)[0]
        best_c, _ = exact_objective(best_h)

    n0 = len(best_h)

    # Phase 1: Smooth SLSQP with tightening
    tau_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    if time.time() < deadline - 100:
        h_opt, c_opt = slsqp_smooth(best_h, n0, tau_list, deadline, max_iter=120)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Direct constrained SLSQP
    if time.time() < deadline - 80:
        for _ in range(3):
            active_lags, _ = get_active_lags(best_h, K=10)
            h_opt, c_opt = slsqp_direct_constraints(best_h, n0, active_lags, best_c, deadline, max_iter=60)
            if c_opt < best_c - 1e-14:
                best_c = c_opt
                best_h = h_opt
            else:
                break

    # Phase 3: Subgradient descent
    if time.time() < deadline - 60:
        h_sg, c_sg = subgradient_descent(best_h, n0, max_iter=150, deadline=deadline)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    # Phase 4: Local search continuous
    if time.time() < deadline - 50:
        h_ls, c_ls = local_search_continuous(best_h, max_rounds=60, rng=rng)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls

    # Phase 5: Binary refinement
    if n0 % 2 == 0 and time.time() < deadline - 40:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=150, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b
            # SLSQP from binary
            if time.time() < deadline - 25:
                tau_tight = [0.001, 0.0003, 0.0001, 0.00003]
                h_opt, c_opt = slsqp_smooth(b, n0, tau_tight, deadline, max_iter=100)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 6: Try other sizes
    sizes_to_try = sorted(set(
        [s for s in range(max(24, n0 - 8), n0 + 9, 2) if s != n0 and s % 2 == 0] +
        [28, 32, 36, 40, 44, 48, 52, 56, 60, 64, 68, 72, 76, 80, 84, 88, 92, 96, 100]
    ))

    for n_try in sizes_to_try:
        if time.time() > deadline - 20:
            break
        if abs(n_try - n0) > 20:
            continue
        # Downsampled parent
        h_ds = downsample(best_h, n_try)
        c_ds, _ = exact_objective(h_ds)

        # Random start
        h_rand = make_starts(n_try, rng, 2)[0]
        c_rand, _ = exact_objective(h_rand)

        candidates = []
        if c_ds < best_c + 0.03:
            candidates.append(h_ds)
        if c_rand < best_c + 0.03:
            candidates.append(h_rand)

        for h_start in candidates:
            if time.time() > deadline - 15:
                break
            h_opt, c_opt = slsqp_smooth(h_start, n_try, [0.005, 0.001, 0.0003, 0.0001], deadline, max_iter=80)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

            if n_try % 2 == 0 and c_opt < best_c + 0.01 and time.time() < deadline - 10:
                b = binary_threshold(h_opt)
                if b is not None:
                    b, c_b = binary_swap_improve(b, max_rounds=80, rng=rng)
                    if c_b < best_c:
                        best_c = c_b
                        best_h = b

    # Phase 7: Interpolation + refit
    if time.time() < deadline - 15:
        n_best = len(best_h)
        for _ in range(3):
            if time.time() > deadline - 10:
                break
            h_pert = best_h + rng.randn(n_best) * 0.02
            h_pert = project_box_sum(h_pert, 0.5 * n_best)
            h_interp, c_interp = interp_between(best_h, h_pert, n_alphas=8)
            if c_interp < best_c:
                best_c = c_interp
                best_h = h_interp
            h_opt, c_opt = slsqp_smooth(best_h, n_best, [0.0003, 0.0001, 0.00003], deadline, max_iter=80)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 8: Final binary + local search
    if time.time() < deadline - 5:
        n_best = len(best_h)
        if n_best % 2 == 0:
            b = binary_threshold(best_h)
            if b is not None:
                b, c_b = binary_swap_improve(b, max_rounds=60, rng=rng)
                if c_b < best_c:
                    best_c = c_b
                    best_h = b
        if time.time() < deadline - 3:
            h_ls, c_ls = local_search_continuous(best_h, max_rounds=30, rng=rng)
            if c_ls < best_c:
                best_c = c_ls
                best_h = h_ls

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)

    return (best_h, best_c, best_n)
