import numpy as np
import time
from scipy.optimize import linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(prof) * dx)

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_gradient_matrix(h):
    """Compute the (2n-1) x n gradient matrix."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n - 1)
        if d > 0:
            L = n - d
            # dP/dh[p] for p in [d, n-1]: += dx*(1-h[p-d])
            G[k, d:n] += dx * (1.0 - h[0:L])
            # dP/dh[p] for p in [0, n-d-1]: += -dx*h[p+d]
            G[k, 0:L] -= dx * h[d:n]
        elif d == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        else:
            c = -d
            L = n - c
            # dP/dh[p] for p in [0, n-c-1]: += dx*(1-h[p+c])
            G[k, 0:L] += dx * (1.0 - h[c:n])
            # dP/dh[p] for p in [c, n-1]: += -dx*h[p-c]
            G[k, c:n] -= dx * h[0:L]
    return G

def repair(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def solve_lp_trust(h0, radius, max_time_s=5.0):
    """Solve the linearized trust-region LP."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    prof = get_profile(h0)
    G = get_gradient_matrix(h0)
    
    # Current values at h0
    p_vals = np.correlate(h0, 1.0 - h0, mode='full') * dx
    
    # LP: minimize theta
    # subject to: G[k,:] @ h - theta <= G[k,:] @ h0 - p_vals[k] for all k
    #             0 <= h[i] <= 1
    #             sum(h) = n/2
    #             h0[i] - radius <= h[i] <= h0[i] + radius

    # Objective: minimize theta => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0  # theta is the last variable

    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - p_vals

    A_trust = np.zeros((2*n, n + 1))

    lb = np.maximum(0.0, h0 - radius)

                best_score = s_trial

                break
        if not improved:

            radius *= 0.7
            if no_improve_count > 15 or radius < 0.0005:
                break
    
    return best_h, best_score

    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n - 1)
        if d > 0:
            L = n - d
            G[k, d:n] += dx * (1.0 - h[0:L])
            G[k, 0:L] -= dx * h[d:n]
        elif d == 0:

            L = n - c
            G[k, 0:L] += dx * (1.0 - h[c:n])
