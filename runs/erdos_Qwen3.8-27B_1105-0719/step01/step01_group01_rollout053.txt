```python
def optimal_tau(c, d, cap):
    """Find tau in [0, cap] minimizing max(c + tau*d)"""
    n_c = len(c)
    # Candidate tau values
    candidates = [0.0, cap]
    
    # Find breakpoints where max changes
    max_c = np.max(c)
    active = np.where(c >= max_c - 1e-12)[0]
    inactive = np.where(c < max_c - 1e-12)[0]
    
    # For each active p and inactive q, find crossover
    for p in active:
        for q in inactive:
            denom = d[p] - d[q]
            if abs(denom) < 1e-15:
                continue
            tau = (c[q] - c[p]) / denom
            if 0 < tau < cap:
                candidates.append(tau)
    
    # Also need crossovers among active set members
    for i in range(len(active)):
        for j in range(i+1, len(active)):
            p, q = active[i], active[j]
            denom = d[p] - d[q]
            if abs(denom) < 1e-15:
                continue
            tau = (c[q] - c[p]) / denom
            if 0 < tau < cap:
                candidates.append(tau)
    
    # Evaluate at all candidates
    best_val = np.inf
    best_tau = 0.0
    for tau in candidates:
        val = np.max(c + tau * d)
        if val < best_val:
            best_val = val
            best_tau = tau
    
    return best_tau, best_val
```

```python
def compute_grad_masks(n):
    """Precompute boolean masks for gradient computation."""
    n_c = 2 * n - 1
    first_valid = np.zeros((n_c, n), dtype=bool)
    second_valid = np.zeros((n_c, n), dtype=bool)
    
    for k in range(n_c):
        d = k - (n - 1)
        if d >= 0:
            first_range = range(0, n - d)  # m from 0 to n-1-d
            second_range = range(d, n)  # m from d to n-1
        else:
            first_range = range(-d, n)  # m from -d to n-1
            second_range = range(0, n + d)  # m from 0 to n-1+d (note d<0 so n+d < n)
        
        for m in first_range:
            first_valid[k, m] = True
        for m in second_range:
            second_valid[k, m] = True
    
    return first_valid, second_valid
```

```python
def compute_A(h, n, first_valid, second_valid):
    """Compute gradient matrix A[k, m] = ∂c[k]/∂h[m]."""
    n_c = 2 * n - 1
    A = np.zeros((n_c, n))
    for k in range(n_c):
        d = k - (n - 1)
        # First term: (1-h[m+d]) for valid (k,m)
        fv = first_valid[k]
        indices = np.where(fv)[0]
```

```python
def optimal_tau(c, d, cap):

    n_c = len(c)
    
    # Candidate tau values: 0, cap, and breakpoints

    # tau = (c[q] - c[p]) / (d[p] - d[q])

    # And where at least one of p, q is "near" the maximum (to limit computation)
    
    # For efficiency, only consider pairs involving the top few c values
    sorted_idx = np.argsort(-c)
    top_k = min(30, n_c)  # consider top 30
    top_set = set(sorted_idx[:top_k].tolist())
    
    for p in range(n_c):
        if p not in top_set:
            continue
        for q in range(n_c):
            if q == p or q not in top_set:
                continue
            denom = d[p] - d[q]
            if abs(denom) < 1e-15:
                continue
            tau = (c[q] - c[p]) / denom
            if 1e-12 < tau < cap - 1e-12:
                candidates.append(tau)
    
    # Also add a few evenly spaced candidates for safety
    n_extra = 5
    for t in np.linspace(0, cap, n_extra + 1)[1:-1]:
        candidates.append(float(t))
    
    # Evaluate
    best_val = np.inf
    best_tau = 0.0
    for tau in candidates:
        val = np.max(c + tau * d)
        if val < best_val - 1e-15:
            best_val = val
            best_tau = tau
    
    return best_tau, best_val
```

This is O(|top_set|² + n_extra) per pair, which is about 30² = 900 per pair. With 400 candidate pairs (20×20 from top_i × top_j) plus 20 random, that's about 420 * 900 = 378K operations per iteration. Fast!

But wait, I'm only considering breakpoints among the top 30 c values. This might miss important breakpoints where a lower c value crosses above the max. Let me think...

If c[p] is much below max(c), it's unlikely to cross above in a small τ range. The crossing τ would be (c[q] - c[p]) / (d[p] - d[q]) where c[q] is near max and c[p] is low. For this to be in a small range, d[p] - d[q] would need to be large, which is unlikely for smooth functions.

In practice, for the small moves we're making (τ typically < 0.1), the relevant breakpoints are those where c values are close to each other. So considering only the top 30 should be fine for the first few iterations. As we converge, the active set stabilizes and this becomes even more accurate.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad_masks(n):
    n_c = 2 * n - 1
    first_valid = np.zeros((n_c, n), dtype=bool)
    second_valid = np.zeros((n_c, n), dtype=bool)
    for k in range(n_c):
        d = k - (n - 1)
        if d >= 0:
            first_valid[k, :n-d] = True
            second_valid[k, d:] = True
        else:
            first_valid[k, -d:] = True
            second_valid[k, :n+d] = True
    return first_valid, second_valid

def compute_A(h, n, first_valid, second_valid):
    n_c = 2 * n - 1
    A = np.zeros((n_c, n))
    d_vals = np.arange(n_c) - (n - 1)
    for k in range(n_c):
        d = d_vals[k]
        fv = first_valid[k]
        idx = np.where(fv)[0]
        if len(idx) > 0:
            A[k, idx] = 1.0 - h[idx + d]
        sv = second_valid[k]
        idx = np.where(sv)[0]
        if len(idx) > 0:
            A[k, idx] -= h[idx - d]
    return A

def optimal_tau(c, d, cap, top_k=25):
    n_c = len(c)
    if cap < 1e-14:
        return 0.0, np.max(c)
    
    candidates = [0.0, cap]
    
    # Find breakpoints among top values
    sorted_idx = np.argsort(-c)
    top = sorted_idx[:top_k]
    
    for ii in range(len(top)):
        p = top[ii]
        for jj in range(ii+1, len(top)):
            q = top[jj]
            denom = d[p] - d[q]
            if abs(denom) < 1e-15:
                continue
            tau = (c[q] - c[p]) / denom
            if 0 < tau < cap:
                candidates.append(tau)
    
    # Add some evenly spaced points
    for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
        candidates.append(frac * cap)
    
    best_val = np.inf
    best_tau = 0.0
    for tau in candidates:
        val = np.max(c + tau * d)
        if val < best_val:
            best_val = val
            best_tau = tau
    
    return best_tau, best_val

def project_feasible(y, n):
    target = n / 2.0
    y = np.clip(y, -10, 10)  # safety
    lo = np.min(y) - 2
    hi = np.max(y) + 2
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    h = np.clip(y - t, 0, 1)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0, 1)
    return h
