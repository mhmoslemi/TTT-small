def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h


def compute_lag_values_and_grads(h):

    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))

def trust_region_qp(h_cur, lambda_reg, active_indices=None, max_iter=30):
    """Solve the trust-region QP subproblem."""
    n = len(h_cur)
    dx = 2.0 / n
    values, grads = compute_lag_values_and_grads(h_cur)
    
    if active_indices is None:
        active_indices = np.arange(len(values))
    
    # Select active constraints
    idx = np.asarray(active_indices)
    vals_act = values[idx]
    grads_act = grads[idx, :]  # shape: (num_active, n)
    
    if not HAS_CP:
        return None
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Linearized constraints: vals_act + grads_act @ (h_var - h_cur) <= t_var
    # i.e., grads_act @ h_var <= t_var - vals_act + grads_act @ h_cur
    rhs = t_var - vals_act + grads_act @ h_cur
    
    # Objective: t + lambda * ||h - h_cur||^2
    obj = cp.Minimize(t_var + lambda_reg * cp.sum_squares(h_var - h_cur))
    
    constraints = [
        grads_act @ h_var <= rhs,
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            return None
    
    if h_var.value is None:
        return None
    
    return h_var.value.copy()

def interpolate_to_dimension(vec, new_len):

def repair_feasible(vec):
