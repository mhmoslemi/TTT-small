from scipy.optimize import minimize
from scipy.special import expit
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final fixup
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if abs(diff2) < 1e-15:
                break
            if diff2 > 0:
                add = min(diff2, 1.0 - h[i] + 1e-15)
                h[i] += add
                diff2 -= add
            else:
                remv = min(-diff2, h[i] + 1e-15)
                h[i] -= remv
                diff2 += remv
    return h

def logit_project(z, n):
    """Project unconstrained z to feasible h with sum = n/2."""
    # Find c such that sum(sigmoid(z_i + c)) = n/2
    # Use Newton's method
    c = 0.0
    for _ in range(50):
        h = expit(z + c)
        s = h.sum()
        if abs(s - n/2.0) < 1e-14:
            break
        dhdc = (h * (1.0 - h)).sum()
        if dhdc < 1e-15:
            break
        c -= (s - n/2.0) / dhdc
    h = expit(z + c)
    # Clip for safety
    h = np.clip(h, 1e-15, 1.0 - 1e-15)
    # Fix sum exactly
    diff = n/2.0 - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h, c

def softmax_objective_and_grad(h, beta):
    """Compute J_beta(h) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    values = corr * dx  # C_l values
    
    # Soft-max
    max_v = np.max(values)
    shifted = beta * (values - max_v)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    J = (max_v + np.log(Z)) / beta
    w = exp_vals / Z  # normalized weights
    
    # Gradient: dJ/dh_j = sum_l w_l * dC_l/dh_j
    # dC_l/dh_j for lag k = l - (n-1):
    # Need to sum over all lags
    grad = np.zeros(n)
    
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)  # lag
        wi = w[idx] * dx
        if k >= 0:
            # h_j contributes as first factor when j-k in [0, n-1-k]: j in [k, n-1]
            # Contribution: (1 - h_{j-k}) for j in [k, n-1]
            if k < n:
                j_range = np.arange(k, n)
                grad[j_range] += wi * (1.0 - h[j_range - k])
            # h_j contributes as second factor (1-h_j) when j in [0, n-1-k]
            # Contribution: -h_{j+k} for j in [0, n-1-k]
            if k < n:
                j_range2 = np.arange(0, n - k)
                grad[j_range2] += -wi * h[j_range2 + k]
        else:
            m = -k
            # h_j contributes as first factor when j in [m, n-1]
            # Contribution: (1 - h_{j-m}) for j in [m, n-1]
            j_range = np.arange(m, n)
            grad[j_range] += wi * (1.0 - h[j_range - m])
            # h_j contributes as second factor when j in [0, n-1-m]
            # Contribution: -h_{j+m} for j in [0, n-1-m]
            j_range2 = np.arange(0, n - m)
            grad[j_range2] += -wi * h[j_range2 + m]
    
    return J, grad

def logit_gradient(z, n, beta):
    """Compute gradient of J_beta(logit_project(z)) w.r.t. z."""
    h, c = logit_project(z, n)
    J, G_h = softmax_objective_and_grad(h, beta)
    
    # Chain rule: ∂J/∂z_k = G_h_k - σ'(z_k+c) * S / T
    # where S = sum_i σ'(z_i+c) * G_h_i, T = sum_i σ'(z_i+c)
    sig_prime = expit(z + c) * (1.0 - expit(z + c))
    T = sig_prime.sum()
    S = (sig_prime * G_h).sum()
    G_z = G_h - sig_prime * (S / T)
    
    return J, G_z

import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corp * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
                remv = min(rem, h[i])
                h[i] -= remv
                rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if abs(diff2) < 1e-15:
                break
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def logit_project(z, n):
    c = 0.0
    for _ in range(80):
        hc = expit(z + c)
        s = hc.sum()
        err = s - n / 2.0
        if abs(err) < 1e-14:
            break
        dhdc = (hc * (1.0 - hc)).sum()
        if dhdc < 1e-16:
            break
        c -= err / dhdc
    h = expit(z + c)
    h = np.clip(h, 1e-15, 1.0 - 1e-15)
    return h, c


def compute_correlation_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def softmax_grad_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    values = compute_correlation_values(h)
    max_v = np.max(values)
    shifted = beta * (values - max_v)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    w = exp_vals / Z
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        wi_dx = wi * dx
        k = idx - (n - 1)
        if k >= 0:
            m = k
            if m < n:
                j1 = np.arange(m, n)
                grad[j1] += wi_dx * (1.0 - h[j1 - m])
                j2 = np.arange(0, n - m)
                grad[j2] += -wi_dx * h[j2 + m]
        else:
            m = -k
            j1 = np.arange(0, n - m)
            grad[j1] += wi_dx * (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            grad[j2] += -wi_dx * h[j2 - m]
    return w, grad


def logit_obj_grad(z, n, beta):
    h, c = logit_project(z, n)
    _, G_h = softmax_grad_h(h, beta)
    values = compute_correlation_values(h)
    max_v = np.max(values)
    shifted = beta * (values - max_v)
    J = (max_v + np.log(np.exp(shifted).sum())) / beta
    sp = expit(z + c) * (1.0 - expit(z + c))
    T = sp.sum()
    if T < 1e-16:
        return J, np.zeros(n)
    S = (sp * G_h).sum()
    G_z = G_h - sp * (S / T)
    return J, G_z


def logit_obj_grad_wrapper(z, n, beta):
    J, G = logit_obj_grad(z, n, beta)
    return J, G


def h_to_z(h, eps=1e-6):
    h_clipped = np.clip(h, eps, 1.0 - eps)
    return np.log(h_clipped / (1.0 - h_clipped))


def logit_anneal(h_init, betas, maxiter_per_beta=50, budget_s=40):
    n = len(h_init)
    t0 = time.perf_counter()
    h_cur = repair_feasible(h_init)
    z_cur = h_to_z(h_cur)
    h_best = h_cur.copy()
    best_score = exact_score(h_best)

    def tl():
        return budget_s - (time.perf_counter() - t0)

    for beta in betas:
        if tl() < 5:
            break
        def obj(z):
            J, _ = logit_obj_grad(z, n, beta)
            return J
        def jac(z):
            _, G = logit_obj_grad(z, n, beta)
            return G
        try:
            res = minimize(obj, z_cur, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter_per_beta, 'gtol': 1e-10,
                                   'ftol': 1e-12, 'maxfun': maxiter_per_beta * 3})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new, _ = logit_project(res.x, n)
                h_new = repair_feasible(h_new)
                s = exact_score(h_new)
                if np.isfinite(s) and s < best_score - 1e-12:
                    best_score = s
                    h_best = h_new
                    h_cur = h_new
                    z_cur = h_to_z(h_new)
        except Exception:
            pass
    return h_best, best_score


def active_descent(h, max_steps=50, time_budget=30):
    n = len(h)
    h = repair_feasible(h).copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()

    for _ in range(max_steps):
        if time.perf_counter() - t0 > time_budget:
            break
        values = compute_correlation_values(h)
        c_max = np.max(values)
        n_active = min(20, 2 * n - 1)
        idx_sorted = np.argsort(values)[::-1][:n_active]
        spread = values[idx_sorted[0]] - values[idx_sorted[-1]]
        sigma = max(0.003, 0.1 * spread)
        gaps = values[idx_sorted] - c_max
        weights = np.exp(gaps / sigma)
        weights /= weights.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for ii, idx in enumerate(idx_sorted):
            wi_dx = weights[ii] * dx
            k = idx - (n - 1)
            if k >= 0:
                m = k
                if m < n:
                    j1 = np.arange(m, n)
                    grad[j1] += wi_dx * (1.0 - h[j1 - m])
                    j2 = np.arange(0, n - m)
                    grad[j2] += -wi_dx * h[j2 + m]
            else:
                m = -k
                j1 = np.arange(0, n - m)
                grad[j1] += wi_dx * (1.0 - h[j1 + m])
                j2 = np.arange(m, n)
                grad[j2] += -wi_dx * h[j2 - m]
        direction = -grad
        direction -= direction.mean()
        dn = np.linalg.norm(direction)
        if dn < 1e-14:

        direction /= dn
        improved = False

            if score < best_score - 1e-13:
                h = h_trial
                best_score = score
                improved = True
                break
        if not improved:

    return h, best_score


def transfer_dimension(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    cum = np.concatenate([[0.0], np.cumsum(h_src) * dx_src])
    x_edges = np.linspace(0.0, 2.0, n_tgt + 1)
    x_continuous = np.linspace(0.0, 2.0, n_src)
    h_flat = np.zeros(n_tgt)
    for j in range(n_tgt):
        a = x_edges[j]
        b = x_edges[j + 1]
        ia = a / dx_src
        ib = b / dx_src
        ia_f = int(np.floor(ia))
        ib_f = int(np.floor(ib))
        ia_f = min(ia_f, n_src - 1)
        ib_f = min(ib_f, n_src - 1)
        integral = 0.0
        for i in range(max(ia_f, 0), ib_f + 1):
            lo = max(a, i * dx_src)
            hi = min(b, (i + 1) * dx_src)
            if hi > lo:
                integral += h_src[i] * (hi - lo)
        h_flat[j] = integral / dx_tgt
    h_flat = np.clip(h_flat, 0.0, 1.0)
    return repair_feasible(h_flat)


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def compute_profile_and_gradients(h):
    n = len(h)
    dx = 2.0 / n
    values = compute_correlation_values(h)
    grads = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        if k >= 0:
            m = k
            if m < n:
                j1 = np.arange(m, n)
                grads[idx, j1] += (1.0 - h[j1 - m])
                j2 = np.arange(0, n - m)
                grads[idx, j2] += -h[j2 + m]
        else:
            m = -k
            j1 = np.arange(0, n - m)
            grads[idx, j1] += (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            grads[idx, j2] += -h[j2 - m]
    grads *= dx
    return values, grads


def slsqp_polish(h_start, maxiter=150):
    n = len(h_start)
    h_start = repair_feasible(h_start)

    def obj(x):
        h = x[:n]
        t = x[n]
        values = compute_correlation_values(h)
        penalty = np.sum(np.maximum(0, values - t) ** 2)
        return t + penalty

    def jac(x):
        h = x[:n]
        t = x[n]
        values = compute_correlation_values(h)
        _, grads = compute_profile_and_gradients(h)
        active = values > t
        g = np.zeros(n + 1)
        g[n] = 1.0
        if np.any(active):
            g[:n] = -np.sum(grads[active], axis=0)
        return g

    x0 = np.concatenate([h_start, [exact_score(h_start) + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [{'type': 'eq', 'fun': lambda x: x[:n].sum() - n / 2.0,
                    'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    try:
        res = minimize(obj, x0, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_sol = repair_feasible(res.x[:n])
            score = exact_score(h_sol)
            return h_sol, score
    except Exception:
        pass
    return None, None

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()

                break

                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def logit_project(z, n):
    c = 0.0
    for _ in range(80):
        hc = expit(z + c)
        s = hc.sum()
        err = s - n / 2.0
        if abs(err) < 1e-14:
            break
        dhdc = (hc * (1.0 - hc)).sum()
        if dhdc < 1e-16:
            break
        c -= err / dhdc
    h = expit(z + c)
    h = np.clip(h, 1e-15, 1.0 - 1e-15)
    return h, c


def compute_correlation_values(h):

    dx = 2.0 / n

def softmax_grad_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    values = compute_correlation_values(h)
    max_v = np.max(values)
    shifted = beta * (values - max_v)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    w = exp_vals / Z
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        wi_dx = wi * dx
        k = idx - (n - 1)
        if k >= 0:
            m = k

                j1 = np.arange(m, n)
                grad[j1] += wi_dx * (1.0 - h[j1 - m])
                j2 = np.arange(0, n - m)
                grad[j2] += -wi_dx * h[j2 + m]
        else:
            m = -k
            j1 = np.arange(0, n - m)
            grad[j1] += wi_dx * (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            grad[j2] += -wi_dx * h[j2 - m]
    return w, grad


def logit_obj_grad(z, n, beta):
    h, c = logit_project(z, n)
    _, G_h = softmax_grad_h(h, beta)
    values = compute_correlation_values(h)
    max_v = np.max(values)
    shifted = beta * (values - max_v)
    J = (max_v + np.log(np.exp(shifted).sum())) / beta
    sp = expit(z + c) * (1.0 - expit(z + c))
    T = sp.sum()
    if T < 1e-16:
        return J, np.zeros(n)
    S = (sp * G_h).sum()
    G_z = G_h - sp * (S / T)
    return J, G_z

def transfer_dimension(h_src, n_tgt):
    n_src = len(h_src)
