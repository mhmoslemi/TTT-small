def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def exact_profile(h):

    dx = 2.0 / n

def transfer_to_size(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)


def make_anti_symmetric(h_full, n):
    """Return anti-symmetric version of h_full for length n."""
    m = n // 2

        val = np.clip(val, 0.0, 1.0)
        result[i] = val

    return result

def coord_descent(h, max_rounds, deadline, anti=False, rng=None):
    """Coordinate descent using pair exchanges."""
    n = len(h)
    m = n // 2
    curr_c = exact_objective(h)
    
    if anti:
        # Only use pairs (i, n-1-i)
        pairs = [(i, n - 1 - i) for i in range(m)]
    else:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        improved = False
        if anti:
            # Random permutation of pairs
            start_pairs = pairs[:]
            if rng is not None:
                rng.shuffle(start_pairs)
        else:
            # Sample a random subset of pairs
            if rng is not None:
                n_sample = min(len(pairs), max(200, n * 20))
                start_pairs = [pairs[rng.randint(len(pairs))] for _ in range(n_sample)]
            else:
                start_pairs = pairs[:max(200, n * 20)]
        
        for (i, j) in start_pairs:
            if time.time() > deadline - 5:
                break
            h, curr_c = pair_step(h, i, j, curr_c)
            if h[i] != 0:  # some change indicator (not perfect)
                improved = True
        
        if not improved:
            break
    
    return h, curr_c
