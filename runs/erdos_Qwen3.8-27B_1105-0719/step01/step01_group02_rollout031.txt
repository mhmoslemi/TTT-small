from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw)) * dx

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    if target is None:
        target = n / 2.0
    # Bisection on offset
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_test)
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo+hi)/2.0, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_row_gradient(h, lag):
    """Compute gradient of raw[lag_index] w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    # raw[t] = sum_m h[m] * (1-h[m+t-(n-1)])
    # d/dh[j] of raw[t]:
    # = (1-h[j+t-(n-1)]) [from h[m] term where m=j]
    #   - h[j-(n-1-t)] [from (1-h[m+t-(n-1)]) term where m+t-(n-1)=j]
    s = lag  # signed lag
    # Term 1: h[j] appears when m=j, contributes (1-h[j+s]) if j+s in range
    for j in range(n):
        js = j + s
        if 0 <= js < n:
            grad[j] += (1.0 - h[js])
    # Term 2: (1-h[j]) appears when m+s=j, i.e., m=j-s, contributes -h[j-s] if j-s in range
    for j in range(n):
        jm = j - s
        if 0 <= jm < n:
            grad[j] -= h[jm]
    return grad

def active_descent(h, max_iter=50, top_k=20, lr=None):

    n = len(h)
    if lr is None:
        lr = 0.01 / n
    h_cur = h.copy()
    c_cur = exact_objective(h_cur)
    
    for it in range(max_iter):
        raw = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        raw_max = np.max(raw)
        
        # Select active lags
        if top_k < len(raw):
            top_idx = np.argsort(raw)[-top_k:]
        else:
            top_idx = np.arange(len(raw))
        
        # Compute weighted gradient
        weights = raw[top_idx]  # raw values as weights
        w_sum = np.sum(weights)
        if w_sum < 1e-15:
            break
        weights = weights / w_sum
        
        # Compute composite gradient
        grad = np.zeros(n)
        for w, t in zip(weights, top_idx):
            s = t - (n - 1)
            grad += w * get_row_gradient(h_cur, s)
        
        # Projected gradient descent
        step = h_cur - lr * grad
        h_new = project_box_sum(step)
        c_new = exact_objective(h_new)
        
        if c_new < c_cur - 1e-12:
            h_cur = h_new
            c_cur = c_new
            # Adaptive learning rate
            lr *= 1.1
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    
    return h_cur, c_cur

def binary_greedy_swap(b, max_rounds=2000):
    """Greedy binary swap optimization - more efficient than all-pairs"""
    n = len(b)
    cur = b.copy()
    cur_c = exact_objective(cur)
    
    for rnd in range(max_rounds):
        # Find the lag with maximum raw value
        raw = np.correlate(cur, 1.0 - cur, mode='full')
        max_t = np.argmax(raw)
        s = max_t - (n - 1)
        
        # Find the pair (i,j) that would most reduce the max lag
        # We want to swap a 1 to a 0 (or 0 to 1) to reduce raw[max_t]
        best_delta = 1e-15
        best_i, best_j = -1, -1
        
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try swapping: set cur[i]=0, cur[j]=1 (i was 1, j was 0)
        # Effect on raw[t]: 
        # For lag s: raw changes by removing contribution of position i (was 1, now 0)
        # and adding contribution of position j (was 0, now 1)
        
        # Delta for the max lag:
        # Removing i: -(1-cur[i+s]) + cur[i-s] (if in range)
        # Adding j: (1-cur[j+s]) - cur[j-s] (if in range)
        
        # But we need to check ALL lags, not just the max
        # For efficiency, just try a limited number of swaps
        
        # Try top candidates based on gradient at max lag
        grad = get_row_gradient(cur, s)
        # We want to decrease raw at max_t: swap 1→0 at high grad position, 0→1 at low grad position
        # (since we're moving mass from high-grad to low-grad positions)
        
        # Actually for binary, swapping i (1→0) and j (0→1):
        # delta_raw[t] = -(1-cur[i+s_t]) + cur[i-s_t] + (1-cur[j+s_t]) - cur[j-s_t]
        # where s_t = t - (n-1)
        
        # For the max lag specifically:
        # We want delta_raw[max_t] < 0
        # = -(1-cur[i+s]) + cur[i-s] + (1-cur[j+s]) - cur[j-s] < 0
        
        # Try a limited set of swaps
        # Sort ones by grad descending (most beneficial to turn off)
        # Sort zeros by grad ascending (most beneficial to turn on)
        ones_grad = grad[ones]
        zeros_grad = grad[zeros]
        ones_sorted = ones[np.argsort(-ones_grad)]
        zeros_sorted = zeros[np.argsort(zeros_grad)]
        
        n_try = min(30, len(ones_sorted), len(zeros_sorted))
        for a in range(n_try):
            for b_idx in range(n_try):
                i = ones_sorted[a]
                j = zeros_sorted[b_idx]
                if i == j:
                    continue
                cur[i] = 0.0
                cur[j] = 1.0
                c = exact_objective(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                delta = cur_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        
        if best_i >= 0:
            cur[best_i] = 0.0
            cur[best_j] = 1.0
            cur_c = exact_objective(cur)
        else:
            break
    
    return cur, cur_c

def binary_full_search(b, max_rounds=500):
    """Full O(n²) binary swap search"""
    n = len(b)
    cur = b.copy()
    cur_c = exact_objective(cur)

        best_i, best_j = -1, -1
        ones = np.where(cur == 1.0)[0]

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

def continuous_sa(h, n_trials=5000, rng_seed=42, max_time=30):
    """Simulated annealing on continuous h"""
    n = len(h)
    cur = h.copy()
    cur_c = exact_objective(cur)
    rg = np.random.RandomState(rng_seed)
    t_start = time.time()
    
    for t in range(n_trials):
        if time.time() > t_start + max_time:
            break
        frac = t / float(n_trials)
        temp = 0.005 * (1 - frac) + 0.0005
        
        i, j = rg.choice(n, 2, replace=False)
        amt = 0.05 + 0.15 * rg.random()
        if amt > cur[j] * 0.9:
            amt = cur[j] * 0.8 or 0.01
        if amt > (1.0 - cur[i]) * 0.9:
            amt = max(0.01, (1.0 - cur[i]) * 0.8)
        if amt < 0.005:
            continue
        
        cur[i] += amt
        cur[j] -= amt
        c = exact_objective(cur)
        
        if c < cur_c:
            cur_c = c
        else:
            if rg.random() < np.exp(-(c - cur_c) / temp):
                cur_c = c
            else:
                cur[i] -= amt
                cur[j] += amt
    
    cur = project_box_sum(cur)
    return cur, exact_objective(cur)

    n = len(b)
    cur = b.copy().astype(np.float64)
    cur_c = exact_objective(cur)
    raw = np.correlate(cur, 1.0 - cur, mode='full')
    dx = 2.0 / n
    
    s_vals = np.arange(2*n-1) - (n-1)
    
    for rnd in range(max_rounds):
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
