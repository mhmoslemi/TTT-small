import numpy as np
from scipy.optimize import minimize
from scipy.special import expit

_t_start = 0.0

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            if add > 0:
                h[i] += add
                rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            if remv > 0:
                h[i] -= remv
                rem -= remv
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            for i in range(n):
                if h[i] < 1.0 - 1e-15:
                    h[i] = min(1.0, h[i] + residual)
                    break
        else:
            for i in range(n):
                if h[i] > 1e-15:
                    h[i] = max(0.0, h[i] + residual)
                    break
    return h

def find_c(z, n):

    c = 0.0
    for _ in range(60):
        h = expit(z + c)
        s = h.sum()

            break
        sh = h * (1.0 - h)
        T = sh.sum()
        if T < 1e-15:
            break
        c += (target - s) / T
    return c

    c = find_c(z, n)
    return expit(z + c)

def h_to_logit(h):

    hc = np.clip(h, eps, 1.0 - eps)
    return np.log(hc / (1.0 - hc))

def compute_c_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_weighted_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grad = np.zeros(n)
    for k in range(L):
        w = weights[k]
        if abs(w) < 1e-16:
            continue
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                idx = np.arange(n - delta)
                grad[idx] += w * dx * (1.0 - h[idx + delta])
                idx2 = np.arange(delta, n)
                grad[idx2] -= w * dx * h[idx2 - delta]
        else:
            m = -delta
            if m < n:
                idx = np.arange(m, n)
                grad[idx] += w * dx * (1.0 - h[idx - m])
                idx2 = np.arange(n - m)
                grad[idx2] -= w * dx * h[idx2 + m]
    return grad

def softmax_obj_grad(z, beta):
    n = len(z)
    h = logit_to_h(z)
    c_vals = compute_c_values(h)
    mc = np.max(c_vals)
    ev = np.exp(beta * (c_vals - mc))
    se = ev.sum()
    J = mc + np.log(se) / beta
    w = ev / se
    G_h = compute_weighted_gradient(h, w)
    sh = h * (1.0 - h)
    T = sh.sum()
    if T < 1e-12:
        return J, np.zeros(n)
    S = np.dot(sh, G_h)
    G_z = G_h - sh * S / T
    return J, G_z

def power_mean_obj_grad(z, p):
    n = len(z)
    h = logit_to_h(z)
    c_vals = compute_c_values(h)
    c_pos = np.maximum(c_vals, 1e-10)
    cp = c_pos ** p
    sp = cp.sum()
    J = sp ** (1.0 / p)
    w = (c_pos ** (p - 1)) / sp ** ((p - 1) / p)
    G_h = compute_weighted_gradient(h, w)
    sh = h * (1.0 - h)
    T = sh.sum()
    if T < 1e-12:
        return J, np.zeros(n)
    S = np.dot(sh, G_h)
    G_z = G_h - sh * S / T
    return J, G_z

def lbfgs_logit(z, beta_levels, maxiter=120, flatten_alpha=0.0):
    """Optimize in logit space with beta annealing via L-BFGS."""
    n = len(z)
    best_z = z.copy()
    best_exact = exact_score(logit_to_h(z))

    for beta in beta_levels:
        def obj(zl):
            if flatten_alpha > 0:
                val, g = softmax_obj_grad(zl, beta)
                h = logit_to_h(zl)
                cv = compute_c_values(h)
                val = val + flatten_alpha * (cv.var() / cv.max())
                return val
            return softmax_obj_grad(zl, beta)[0]

        def grad(zl):
            _, g = softmax_obj_grad(zl, beta)
            return g

        try:
            res = minimize(obj, best_z, jac=grad, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-14,
                                    'gtol': 1e-9, 'maxls': 40})
            h_new = logit_to_h(res.x)
            sc = exact_score(h_new)
            if sc < best_exact:
                best_z = res.x.copy()
                best_exact = sc
        except Exception:
            pass
    return best_z, best_exact

                break
            h_up = interpolate_to_dim(best_h, nd)
            sc_up = exact_score(h_up)
            if sc_up <= best_score + 0.015:
                z_up = h_to_logit(h_up)
                z_sol, sc_sol = lbfgs_logit(z_up, [10, 25, 50, 100, 200], maxiter=150)
                if sc_sol < best_score:
                    best_h = logit_to_h(z_sol)
                    best_score = exact_score(best_h)
                    n_cur = len(best_h)
                h_ad, sc_ad = active_descent(best_h, max_steps=50, n_top=30)
                if sc_ad < best_score:
                    best_h = h_ad
                    best_score = sc_ad
                    n_cur = len(best_h)

    # Phase 6: Multi-start with structured perturbations
    if tl() > 100:
        for trial in range(8):
            if tl() < 35:
                break
            if trial % 4 == 0:
                freq = rng.uniform(0.5, 4.0)
                z_pert = h_to_logit(best_h) + 0.15 * np.sin(2 * np.pi * freq * np.arange(len(best_h)) / len(best_h))
            elif trial % 4 == 1:
                z_pert = h_to_logit(best_h) + rng.normal(0, 0.08, len(best_h))
            elif trial % 4 == 2:
                z_base = h_to_logit(best_h)
                bs = max(2, len(z_base) // 12)
                z_pert = z_base.copy()
                for b in range(0, len(z_base), bs):
                    z_pert[b:b + bs] += rng.uniform(-0.15, 0.15)
            else:
                z_pert = h_to_logit(best_h) + rng.normal(0, 0.04, len(best_h))
            z_sol, sc_sol = lbfgs_logit(z_pert, [30, 60, 120], maxiter=100)
            if sc_sol < best_score:
                best_h = logit_to_h(z_sol)
                best_score = exact_score(best_h)
                n_cur = len(best_h)

    # Phase 7: Final Adam refinement
    if tl() > 40:
        z = h_to_logit(best_h)
        z_sol, sc_sol = adam_logit(z, beta=300, n_iters=300, lr=0.01,
                                   flatten_alpha=0.05)
        if sc_sol < best_score:
            best_h = logit_to_h(z_sol)
            best_score = exact_score(best_h)

    # Phase 8: Final active descent
    if tl() > 15:
        h_ad, sc_ad = active_descent(best_h, max_steps=60, n_top=30)
        if sc_ad < best_score:
            best_h = h_ad
            best_score = sc_ad

    # Phase 9: One more L-BFGS on the polished result
    if tl() > 25:
        z = h_to_logit(best_h)
        z_sol, sc_sol = lbfgs_logit(z, [100, 200, 400], maxiter=100)
        if sc_sol < best_score:
            best_h = logit_to_h(z_sol)
            best_score = exact_score(best_h)
        h_ad, sc_ad = active_descent(best_h, max_steps=40, n_top=25)
        if sc_ad < best_score:
            best_h = h_ad
            best_score = sc_ad

    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
