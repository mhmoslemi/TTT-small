def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    try:
        val = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

    except Exception:

def repair_feasible(vec):

    sum_hi = np.clip(h + hi, 0.0, 1.0).sum()

        h = np.clip(h * scale, 0.0, 1.0)
    elif sum_hi <= target:

        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:

                h[i] += add
                remaining -= add

                remaining += rem

def build_profile_and_grad(h):

    dx = 2.0 / n
    M = 2 * n - 1
    P = np.zeros(M)
    G = np.zeros((M, n))

        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        ns = np.arange(n_lo, n_hi + 1)
        P[m] = dx * np.sum(h[ns + k] * (1.0 - h[ns]))
        j1_lo = max(0, n_lo + k)
        j1_hi = min(n - 1, n_hi + k)
        if j1_lo <= j1_hi:
            G[m, j1_lo:j1_hi + 1] += dx * (1.0 - h[j1_lo - k:j1_hi - k + 1])
        j2_lo = max(0, n_lo)
        j2_hi = min(n - 1, n_hi)
        if j2_lo <= j2_hi:
            G[m, j2_lo:j2_hi + 1] += -dx * h[j2_lo + k:j2_hi + k + 1]
    return P, G

            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_slp3, c_slp3 = slp_refine(h_pert, max_iters=50,
                                         time_limit=min(60, time_remaining()),
                                         t_start=t_start)
            if c_slp3 < best_c - 1e-12:
                best_h = h_slp3
                best_c = c_slp3

    # Stage 5: Subgradient descent from best
    if time_remaining() > 60:
        h_sg, c_sg = subgradient_descent(best_h, max_iters=150,
                                          time_limit=min(80, time_remaining()),
                                          t_start=t_start)
        if c_sg < best_c - 1e-12:
            best_h = h_sg
            best_c = c_sg

    # Stage 6: SLSQP polish again
    if time_remaining() > 60:
        h_sq2 = slsqp_polish(best_h, maxiter=400, tol=1e-13)
        if h_sq2 is not None:
            c_sq2 = exact_score(h_sq2)
            if c_sq2 < best_c - 1e-12:
                best_h = h_sq2
                best_c = c_sq2

    # Stage 7: Final SLP with very small radius
    if time_remaining() > 40:
        P, G = build_profile_and_grad(best_h)
        for radius_try in [0.05, 0.03, 0.015, 0.008, 0.004]:
            if time_remaining() < 10:
                break
            z_star, t_pred = solve_lp_slp(best_h, radius_try)
            if z_star is not None:
                h_ls, c_ls, _ = slp_line_search(best_h, z_star, best_c)
                if c_ls < best_c - 1e-13:
                    best_h = h_ls
                    best_c = c_ls

    # Stage 8: Try different dimensions
    if time_remaining() > 50:
        n_cur = len(best_h)
        dim_candidates = []
        for d in [n_cur - 2, n_cur - 4, n_cur + 2, n_cur + 4, n_cur + 8, n_cur - 8]:
            if 30 <= d <= 130 and d != n_cur:
                dim_candidates.append(d)
        dim_candidates = sorted(set(dim_candidates), key=lambda d: abs(d - n_cur))
        for new_n in dim_candidates[:3]:
            if time_remaining() < 25:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            c_interp = exact_score(h_interp)
            h_try, c_try = slp_refine(h_interp, max_iters=40,
                                       time_limit=min(30, time_remaining()),
                                       t_start=t_start)
            if c_try < best_c - 1e-12:
                best_h = h_try
                best_c = c_try
                n_best = new_n

    # Stage 9: Final SLSQP
    if time_remaining() > 30:
        h_final = slsqp_polish(best_h, maxiter=500, tol=1e-14)
        if h_final is not None:
            c_final_sq = exact_score(h_final)
            if c_final_sq < best_c - 1e-13:
                best_h = h_final
                best_c = c_final_sq

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    n_final = len(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, n_final
