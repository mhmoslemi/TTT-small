    if abs(diff2) > 1e-14:
        if diff2 > 0:
            for i in range(n):
                if h[i] < 1.0:
                    h[i] += min(diff2, 1.0 - h[i])
                    break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    h[i] += max(diff2, -h[i])
                    break
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_soft_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    values = np.correlate(h, 1.0 - h, mode="full") * dx
    max_v = values.max()
    shifted = beta * (values - max_v)
    exps = np.exp(shifted)
    s = exps.sum()
    obj = max_v + np.log(s) / beta
    w = exps / s
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for delta in range(-(n - 1), n):
        r = w[n - 1 + delta]
        if r < 1e-14:
            continue
        if delta == 0:
            grad += r * dx * (1.0 - 2.0 * h)
        elif delta > 0:
            m = n - delta
            grad[:m] += r * dx * one_minus_h[delta:]
            grad[delta:] += r * dx * (-h[:m])
        else:
            d = -delta
            m = n - d
            grad[d:] += r * dx * one_minus_h[:m]
            grad[:m] += r * dx * (-h[d:])
    return obj, grad


def make_soft_funcs(h0, beta):
    n = len(h0)
    target = n / 2.0

    def obj_func(x):
        val, _ = compute_soft_obj_and_grad(x, beta)
        return val

    def obj_grad(x):
        _, g = compute_soft_obj_and_grad(x, beta)
        return g

    def eq_fun(x):
        return x.sum() - target

    def eq_jac(x):
        return np.ones(n)

    return obj_func, obj_grad, eq_fun, eq_jac


def solve_smooth(h0, beta, maxiter=200):
    h0 = repair_feasible(h0)
    n = len(h0)
    obj_func, obj_grad, eq_fun, eq_jac = make_soft_funcs(h0, beta)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        result = minimize(
            obj_func, h0, jac=obj_grad,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
    except Exception:
        return None, None
    if result.x is not None:
        h_sol = repair_feasible(result.x)
        score = exact_score(h_sol)
        if np.isfinite(score):
            return h_sol, score
    return None, None


def cdf_transfer(h, new_len):
    n = len(h)
    dx_old = 2.0 / n
    cdf = np.cumsum(h) * dx_old
    cdf = np.concatenate([[0.0], cdf])
    x_new = (np.arange(new_len) + 0.5) * 2.0 / new_len
    new_h = np.zeros(new_len)
    for i in range(new_len):
        a = i * 2.0 / new_len
        b = (i + 1) * 2.0 / new_len
        mass_in = np.interp(b, cdf) - np.interp(a, cdf)
        new_h[i] = mass_in / (2.0 / new_len)
    return repair_feasible(np.clip(new_h, 0.0, 1.0))


def active_descent(h, max_steps=50):
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    for step in range(max_steps):
        values = compute_profile(best)
        c = values.max()
        n_active = min(20, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_active]
        gaps = np.maximum(c - values[active_idx], 0.0)
        sigma = 0.003 * c + 1e-10
        weights = np.exp(-gaps / sigma)
        weights /= weights.sum() + 1e-15
        dx = 2.0 / n
        one_minus_h = 1.0 - best
        direction = np.zeros(n)
        for delta in range(-(n - 1), n):
            r = sum(weights[k] for k in range(len(active_idx)) if active_idx[k] == n - 1 + delta)
            if r < 1e-14:
                continue
            if delta == 0:
                direction += r * dx * (1.0 - 2.0 * best)
            elif delta > 0:
                m = n - delta
                direction[:m] += r * dx * one_minus_h[delta:]
                direction[delta:] += r * dx * (-best[:m])
            else:
                d = -delta
                m = n - d
                direction[d:] += r * dx * one_minus_h[:m]
                direction[:m] += r * dx * (-best[d:])
        direction -= direction.mean()
        improved = False
        for alpha in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        if not improved:
            break
    return best, best_score


def symmetrize(h):
    return repair_feasible((h + h[::-1]) / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    rng = np.random.RandomState(seed)

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_h = repair_feasible(parent.copy())
    else:
        best_h = np.full(100, 0.5)

    best_score = exact_score(best_h)
    best_n = len(best_h)

    # Try symmetric variants
    h_sym = symmetrize(best_h)
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-13:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: Soft optimization with increasing beta at current dimension
    beta_schedule = [15, 30, 50, 80, 130, 200, 350, 500]
    for b in beta_schedule:
        if tl() < 60:
            break
        maxiter = 150 if len(best_h) <= 120 else 120
        h_sol, score = solve_smooth(best_h, b, maxiter=maxiter)
        if h_sol is not None and score < best_score - 1e-13:
            best_h = h_sol
            best_score = score
            best_n = len(h_sol)

    # Phase 2: Active descent polish
    if tl() > 30:
        h_ref, score_ref = active_descent(best_h, max_steps=40)
        if score_ref < best_score - 1e-14:
            best_h = h_ref
            best_score = score_ref

    # Phase 3: Dimension transfer exploration
    if tl() > 80:
        n_cur = best_n
        dims_to_try = []
        for factor in [0.75, 0.65, 1.15, 1.3, 1.5]:
            nd = int(round(n_cur * factor))
            nd = max(40, min(nd, 220))
            if nd != n_cur and nd not in dims_to_try:
                dims_to_try.append(nd)
        dims_to_try.sort()

        for nd in dims_to_try:
            if tl() < 50:
                break
            h_interp = cdf_transfer(best_h, nd)
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.03:
                beta_sched_short = [30, 60, 120, 250]
                for b in beta_sched_short:
                    if tl() < 40:
                        break
                    h_sol, score = solve_smooth(h_interp, b, maxiter=100)
                    if h_sol is not None and score < best_score - 1e-13:
                        best_h = h_sol
                        best_score = score
                        best_n = len(h_sol)
                        h_interp = h_sol
                # Quick descent on this dimension
                if tl() > 30:
                    h_ref, score_ref = active_descent(h_interp, max_steps=25)
                    if score_ref < best_score - 1e-14:
                        best_h = h_ref
                        best_score = score_ref
                        best_n = len(h_ref)

    # Phase 4: Random perturbation multi-start
    if tl() > 60:
        n_best = len(best_h)
        for trial in range(12):
            if tl() < 40:
                break
            sigma = rng.uniform(0.003, 0.015)
            z = rng.randn(n_best)
            z -= z.mean()
            h_pert = repair_feasible(best_h + sigma * z)
            s_pert = exact_score(h_pert)
            if s_pert < best_score + 0.02:
                h_sol, score = solve_smooth(h_pert, beta=80, maxiter=100)
                if h_sol is not None and score < best_score - 1e-13:
                    best_h = h_sol
                    best_score = score
                    best_n = len(h_sol)

    # Phase 5: Deep optimization on best with high beta
    if tl() > 40:
        for b in [300, 500, 800]:
            if tl() < 30:
                break
            h_sol, score = solve_smooth(best_h, b, maxiter=200)
            if h_sol is not None and score < best_score - 1e-13:
                best_h = h_sol
                best_score = score
                best_n = len(h_sol)

    # Phase 6: Final active descent
    if tl() > 20:
        h_ref, score_ref = active_descent(best_h, max_steps=50)
        if score_ref < best_score - 1e-14:
            best_h = h_ref
            best_score = score_ref

    # Phase 7: One more smooth pass at very high beta
    if tl() > 25:
        h_sol, score = solve_smooth(best_h, beta=1000, maxiter=150)
        if h_sol is not None and score < best_score - 1e-13:
            best_h = h_sol
            best_score = score
            best_n = len(h_sol)

    # Final active descent
    if tl() > 15:
        h_ref, score_ref = active_descent(best_h, max_steps=30)
        if score_ref < best_score - 1e-14:
            best_h = h_ref
            best_score = score_ref

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
