- Pick a 1 and a 0, swap them, check if max overlap decreased.
- This is O(n^2) per swap, times O(n) swaps per round, for O(n^3) per round.
- For n=70, that's about 343,000 operations per round, which is fine.

Actually, for the binary case, I can compute the overlap more efficiently. When I swap positions i (was 1) and j (was 0), the change in overlap at lag k is:
- At lag k > 0:
  - If i+k = j: we lose the contribution h[i]*(1-h[j]) = 1*1 = 1, and gain h'[i]*(1-h'[j])... wait, after swap h'[i]=0, h'[j]=1. So the new contribution at position i is h'[i]*(1-h'[i+k]) = 0*(...) = 0, and at position j is h'[j]*(1-h'[j+k]).

def exact_objective(h):
    """Compute C5 = max of correlation profile * dx"""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    """Project h onto [0,1]^n with sum = target"""
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute among non-boundary elements
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def correlation_gradients(h):
    """
    Compute the gradient of each correlation component.
    Returns: grad_matrix of shape (2n-1, n) where grad_matrix[i, m] = dR[i]/dh[m]
    Also returns the raw correlation values R (before *dx).
    """
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2*n - 1, n))
    
    for i in range(2*n - 1):
        k = i - (n - 1)
        if k >= 0:
            # R[i] = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
            # d/dh[m]: (1-h[m-k]) for m in [k, n-1] minus h[m+k] for m in [0, n-1-k]
            s = n - k  # number of terms
            # First part: m in [k, n-1], contribution (1-h[m-k])
            G[i, k:] = 1.0 - h[:s]
            # Second part: m in [0, n-1-k], contribution -h[m+k]
            G[i, :s] -= h[k:]
        else:
            # k < 0, let s = -k
            s = -k
            # R[i] = sum_{n=s}^{n-1} h[n-s]*(1-h[n])
            # d/dh[m]: (1-h[m+s]) for m in [0, n-1-s] minus h[m-s] for m in [s, n-1]
            num = n - s  # number of terms
            # First part: m in [0, n-1-s], contribution (1-h[m+s])
            G[i, :num] = 1.0 - h[s:]
            # Second part: m in [s, n-1], contribution -h[m-s]
            G[i, s:] -= h[:num]
    
    return G, R

def slsqp_constrained(h_start, n, deadline, max_iters=100):
    """
    Solve min c s.t. c - profile[i] >= 0 for all i, sum(h) = n/2, 0<=h<=1.
    Variables: [h_0, ..., h_{n-1}, c]
    """
    dx = 2.0 / n
    target = 0.5 * n
    c0 = exact_objective(h_start)
    c_init = c0 + 1e-6
    
    def objective(z):
        return z[-1]  # c
    
    def objective_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def equality_fun(z):
        return np.sum(z[:n]) - target
    
    def equality_jac(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    # Precompute gradient matrix structure for constraints
    # For constraint i: g_i(z) = c - dx * R_i(h) >= 0
    # Jacobian row: [-dx * G[i,:], 1]
    
    n_cons = 2 * n - 1
    G, R = correlation_gradients(z[:n] if 'z' in dir() else h_start)
    
    def constraint_fun(z):
        h = z[:n]
        c = z[-1]
        R = np.correlate(h, 1.0 - h, mode='full')
        return c - R * dx  # shape (2n-1,)
    
    def constraint_jac(z):
        h = z[:n]
        G, R = correlation_gradients(h)
        # Jacobian is (2n-1, n+1): rows are [-dx*G[i,:], 1]
        J = np.zeros((2*n - 1, n + 1))
        J[:, :n] = -dx * G
        J[:, n] = 1.0
        return J
    
    bounds_h = [(0.0, 1.0)] * n
    bounds_c = [(1e-8, 1.5)]
    bounds = bounds_h + bounds_c
    
    constraints = [
        {'type': 'eq', 'fun': equality_fun, 'jac': equality_jac},
        {'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac}
    ]
    
    z0 = np.append(h_start, c_init)
    
    result = minimize(
        objective, z0, method='SLSQP', jac=objective_grad,
        bounds=bounds, constraints=constraints,
        options={'maxiter': max_iters, 'ftol': 1e-10}
    )
    
    if result.x is not None and np.all(np.isfinite(result.x)):
        h_new = project_box_sum(result.x[:n], target)
        c_new = exact_objective(h_new)
        if c_new < c0 - 1e-12:
            return h_new, c_new
    return h_start, c0

def gradient_descent_step(h, n, deadline, step_sizes=None):
    """
    Try to reduce the max overlap by moving in the negative gradient
    direction of the active constraint.
    """
    if step_sizes is None:
        step_sizes = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]
    
    dx = 2.0 / n
    target = 0.5 * n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(profile)
    max_idx = int(np.argmax(profile))
    
    # Get gradient of the active constraint
    G, R = correlation_gradients(h)
    grad = G[max_idx, :] * dx  # gradient of the active profile component
    
    # Also consider top few active constraints
    top_k = min(5, 2*n-1)
    top_indices = np.argsort(-profile)[:top_k]
    top_weights = profile[top_indices] / profile[top_indices].sum()
    grad_weighted = np.zeros(n)
    for w, idx in zip(top_weights, top_indices):
        grad_weighted += w * G[idx, :] * dx
    
    best_h = h
    best_c = c_max
    
    for alpha in step_sizes:
        for direction in [-grad_weighted, -grad]:
            h_try = h + alpha * direction
            h_try = project_box_sum(h_try, target)
            c_try = exact_objective(h_try)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try.copy()
    
    return best_h, best_c

def make_pattern(n, mode, rng):

    elif mode == 4:
        # Two blocks
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[n//2: n//2+q] = 1.0
    elif mode == 5:
        # Smooth random
        h = np.full(n, 0.5) + rng.randn(n) * 0.2
    elif mode == 6:
        # Thinning of alternating
        h = np.zeros(n)
        for i in range(0, n, 3):
            h[i] = 1.0
    elif mode == 7:
        # Thulpen-block with period 5
        pat = np.array([1,1,0,0,0])
        h = np.tile(pat, n // len(pat) + 1)[:n]
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)

def transfer_resolution(h_src, n_dest):
    """Transfer a solution from one resolution to another using integration."""
    n_src = len(h_src)
    if n_dest == n_src:

    n_samples = max(500, 20 * n_dest)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_dense = np.linspace(0.5/n_src, 1 - 0.5/n_src, n_samples)
    h_dense = np.interp(x_dense, x_src, h_src)
    
    # Integrate over each destination cell
    x_cell_centers = (np.arange(n_dest) + 0.5) / n_dest
    cell_width = 1.0 / n_dest
    h_dest = np.zeros(n_dest)
    for i in range(n_dest):
        lo = i * cell_width
        hi = (i + 1) * cell_width
        mask = (x_dense >= lo) & (x_dense <= hi)
        if np.sum(mask) > 0:
            h_dest[i] = np.mean(h_dense[mask])
        else:
            h_dest[i] = np.interp(x_cell_centers[i], x_src, h_src)
    
    return project_box_sum(h_dest, 0.5 * n_dest)
