        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

            mask = x > 1e-12
        if np.any(mask):

    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    lo = -2.0

            lo = mid
        else:
            hi = mid

def epigraph_solve(h0, deadline, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    h0 = project_box_sum(h0.copy())
    c0 = exact_objective(h0)
    x0 = np.append(h0, c0 + 0.01)
    
    def fun(x):
        return x[n]
    
    def grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    constraints = []
    for k in range(n):
        # Constraint: t - f_k(h) >= 0
        k_val = k
        def cfun(x, k=k_val):
            h = x[:n]
            t = x[n]
            if k == 0:
                val = np.sum(h * (1.0 - h))
            else:
                val = np.sum(h[k:] * (1.0 - h[:n-k]))
            return t - dx * val
        def cjac(x, k=k_val):
            h = x[:n]
            g = np.zeros(n + 1)
            if k == 0:
                g[:n] = -dx * (1.0 - 2.0 * h)
            else:
                g[k:n] = -dx * (1.0 - h[:n-k])
                g[:n-k] += -dx * h[k:n]
            g[n] = 1.0
            return g
        constraints.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})
    
    # Equality constraint: sum(h) = n/2
    def efun(x):
        return np.sum(x[:n]) - n / 2.0
    def ejac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    constraints.append({'type': 'eq', 'fun': efun, 'jac': ejac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    try:
        res = minimize(fun, x0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        h_new = project_box_sum(np.array(res.x[:n], dtype=np.float64))
        return h_new
    except Exception:
        return h0

            if c_new < best_c:
                best_c = c_new
                best_b = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
        T *= cool
    return best_b, best_c


class TFunc:
    def __init__(self, n):
        self.n = n
    def value(self, x):
        return x[self.n]
    def gradient(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g


class EqConstraint:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0
    def jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


class IneqConstraint:
    def __init__(self, n, k):
        self.n = n
        self.k = k
        self.dx = 2.0 / n
    def fun(self, x):
        h = x[:self.n]
        t = x[self.n]
        if self.k == 0:
            val = np.sum(h * (1.0 - h))
        else:
            val = np.sum(h[self.k:] * (1.0 - h[:self.n - self.k]))
        return t - self.dx * val
    def jac(self, x):
        h = x[:self.n]
        g = np.zeros(self.n + 1)
        if self.k == 0:
            g[:self.n] = -self.dx * (1.0 - 2.0 * h)
        else:
            g[self.k:self.n] = -self.dx * (1.0 - h[:self.n - self.k])
            g[:self.n - self.k] += -self.dx * h[self.k:self.n]
        g[self.n] = 1.0
        return g


def epigraph_solve(h0, deadline, maxiter=200):
    n = len(h0)
    h0 = project_box_sum(h0.copy())
    c0 = exact_objective(h0)
    x0 = np.append(h0, c0 + 0.005)
    tfunc = TFunc(n)
    eqcon = EqConstraint(n)
    constraints = []
    for k in range(n):
        constraints.append({'type': 'ineq', 'fun': IneqConstraint(n, k).fun, 'jac': IneqConstraint(n, k).jac})
    constraints.append({'type': 'eq', 'fun': eqcon.fun, 'jac': eqcon.jac})
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = minimize(tfunc.value, x0.copy(), method='SLSQP', jac=tfunc.gradient,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        h_new = project_box_sum(np.array(res.x[:n], dtype=np.float64))
        return h_new
    except Exception:
        return h0


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = 3 + rng.randint(0, 4)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)
