def exact_objective(h):

    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break

        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break

        f_val = float(np.max(P + b * d))
        if f_val < best_f:

            best_d = d
    return best_d, best_f

    """Optimize h using pair exchanges with exact 1D line search."""
    n = len(h)
    h = project_box_sum(h.copy())
    curr_c = float(np.max(get_profile(h)))
    local_best = h.copy()
    local_best_c = curr_c
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 15:
            break
        improved = False
        # Try all pairs (or a random subset for large n)
        if n <= 60:
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
            rng.shuffle(pairs)
        else:
            # Random subset
            pairs = [(int(rng.randint(n)), int(rng.randint(n))) for _ in range(n*n)]
            pairs = [(i,j) for i,j in pairs if i < j]
        
        for (i, j) in pairs:
            if time.time() > deadline - 15:
                break
            P = get_profile(h)
            b = compute_pair_gradient(h, i, j)
            L = max(-h[i], h[j] - 1.0)
            U = min(1.0 - h[i], h[j])
            if U - L < 1e-12:
                continue
            best_d, best_f = solve_1d_pair(P, b, L, U)
            if best_f < curr_c - 1e-13 and abs(best_d) > 1e-12:
                h[i] += best_d
                h[j] -= best_d
                h = project_box_sum(h)
                curr_c = float(np.max(get_profile(h)))
                improved = True
                if curr_c < local_best_c:
                    local_best_c = curr_c
                    local_best = h.copy()
        if not improved:
            break
    
    return local_best, local_best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def binary_swap_search(b, deadline, max_rounds=200):
    n = len(b)
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b, curr_c
