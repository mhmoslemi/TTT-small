        except:
            return None
    if h_var.value is None:
        return None
    return np.asarray(h_var.value).flatten()

                lam *= 2  # smaller steps
                K = min(2*n-1, K + 5)
                if stuck_count > 5:

                continue

        accepted = False
        direction = h_qp - h
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-10:
            stuck_count += 1
            lam *= 0.5  # try bigger steps
            if stuck_count > 5:
                break
            continue
        
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:

                h = h_trial
                best_h = h_trial.copy()
                best_score = score
                accepted = True
                break

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h, kind='stable')
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h, kind='stable')
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            cands = np.where(h < 1.0 - 1e-12)[0]
        else:
            cands = np.where(h > 1e-12)[0]
        if len(cands) > 0:
            spread = residual / len(cands)
            for i in cands:
                if residual > 0:
                    h[i] = min(1.0, h[i] + spread)
                else:
                    h[i] = max(0.0, h[i] + spread)
    return h

    n = len(h_cur)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    obj = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h_cur))
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    for idx in active_idx:
        constraints.append(values[idx] + grads[idx] @ (h_var - h_cur) <= t_var)
    prob = cp.Problem(obj, constraints)
    solvers = [cp.ECOS, cp.CVXOPT, cp.PDLP, cp.SCS]
    for s in solvers:
        try:

            if h_var.value is not None:
                return np.asarray(h_var.value).flatten().copy()
        except Exception:
            continue
    return None

                h_trial = repair_feasible(h + alpha * direction)
                score = exact_score(h_trial)
                if score < best_score - 1e-13:
                    h = h_trial.copy()
                    best = h_trial.copy()
                    best_score = score
                    improved = True
                    break
            if not improved:
                sigma_base *= 0.7
                if sigma_base < 0.0002:
                    break

    return best, best_score


def coord_descent_step(h, values, grads, time_budget=15):
    """Coordinate-wise best response for the most active lag."""
    t_start = time.perf_counter()
    h = repair_feasible(h.copy())
    n = len(h)
    best = h.copy()
    best_score = exact_score(h)

    for step in range(40):
        if time.perf_counter() - t_start > time_budget:
            break
        values, grads = compute_profile_and_gradients(h)
        C = np.max(values)
        if C < best_score - 1e-13:
            best_score = C
            best = h.copy()

        r_star = int(np.argmax(values))
        g = grads[r_star].copy()
        g -= np.mean(g)
        for alpha in [0.03, 0.015, 0.008, 0.004, 0.002]:
            h_trial = repair_feasible(h + alpha * (-g))
            score = exact_score(h_trial)
            if score < best_score - 1e-13:
                h = h_trial.copy()
                best = h_trial.copy()
                best_score = score
                break
        else:
            break

    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_h = repair_feasible(parent)
    else:
        best_h = np.full(140, 0.5)

    h_sym = repair_feasible((best_h + best_h[::-1]) / 2.0)
    if exact_score(h_sym) < exact_score(best_h) - 1e-12:
        best_h = h_sym

    best_score = exact_score(best_h)
    best_n = len(best_h)

    # Phase 1: Trust-region QP (main optimization)
    if time_left() > 150:
        tr_budget = min(450, time_left() * 0.45)
        h_tr, score_tr = trust_region_optimize(best_h, max_iters=80, lam_init=0.4, time_budget=tr_budget)
        if score_tr < best_score - 1e-12:
            best_h = h_tr
            best_score = score_tr
            best_n = len(h_tr)

    # Phase 2: Active descent polish
    if time_left() > 40:
        h_ad, score_ad = active_descent_polish(best_h, max_steps=60, time_budget=min(50, time_left() * 0.15))
        if score_ad < best_score - 1e-12:
            best_h = h_ad
            best_score = score_ad
            best_n = len(h_ad)

    # Phase 3: Coordinate descent refinement
    if time_left() > 30:
        h_cd, score_cd = coord_descent_step(best_h, None, None, time_budget=20)
        if score_cd < best_score - 1e-12:
            best_h = h_cd
            best_score = score_cd
            best_n = len(h_cd)

    # Phase 4: Trust-region with different lambda values
    if time_left() > 100:
        for lam_val in [0.1, 1.5, 4.0]:
            if time_left() < 50:
                break
            h_tr, score_tr = trust_region_optimize(best_h, max_iters=40, lam_init=lam_val,
                                                    time_budget=min(120, time_left() * 0.25))
            if score_tr < best_score - 1e-12:
                best_h = h_tr
                best_score = score_tr
                best_n = len(h_tr)
            # Polish after each
            if time_left() > 30:
                h_ad, score_ad = active_descent_polish(best_h, max_steps=30, time_budget=20)
                if score_ad < best_score - 1e-12:
                    best_h = h_ad
                    best_score = score_ad
                    best_n = len(h_ad)

    # Phase 5: Dimension exploration
    if time_left() > 120:
        n_cur = len(best_h)
        dims = sorted(set(n_cur + dn for dn in [-20, -10, -5, 5, 10, 20, 30]
                          if 40 <= n_cur + dn <= 200))
        for nd in dims:
            if time_left() < 60:
                break
            if nd == n_cur:
                continue
            h_interp = interpolate_to_dimension(best_h, nd)
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.004:
                h_tr, score_tr = trust_region_optimize(h_interp, max_iters=35, lam_init=0.3,
                                                        time_budget=min(90, time_left() * 0.3))
                if time_left() > 20:
                    h_ad, score_ad = active_descent_polish(h_tr, max_steps=25, time_budget=15)
                    if score_ad < best_score - 1e-12:
                        best_h = h_ad
                        best_score = score_ad
                        best_n = len(h_ad)
                        n_cur = best_n
                elif score_tr < best_score - 1e-12:
                    best_h = h_tr
                    best_score = score_tr
                    best_n = len(h_tr)
                    n_cur = best_n

    # Phase 6: Multi-start perturbation
    rng = np.random.RandomState(seed)
    if time_left() > 80:
        for trial in range(6):
            if time_left() < 35:
                break
            eps = rng.uniform(0.003, 0.015)
            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_tr, score_tr = trust_region_optimize(h_pert, max_iters=30, lam_init=0.5,
                                                    time_budget=min(70, time_left() * 0.3))
            if score_tr < best_score - 1e-12:
                best_h = h_tr
                best_score = score_tr
                best_n = len(h_tr)

    # Phase 7: Final active descent + trust-region
    if time_left() > 25:
        h_ad, score_ad = active_descent_polish(best_h, max_steps=50, time_budget=min(25, time_left() * 0.4))
        if score_ad < best_score - 1e-12:
            best_h = h_ad
            best_score = score_ad
            best_n = len(h_ad)

    if time_left() > 30:
        h_tr, score_tr = trust_region_optimize(best_h, max_iters=30, lam_init=0.2,
                                                time_budget=min(35, time_left() - 5))
        if score_tr < best_score - 1e-12:
            best_h = h_tr
            best_score = score_tr
            best_n = len(h_tr)

    # Final polish
    if time_left() > 10:
        h_ad, score_ad = active_descent_polish(best_h, max_steps=20, time_budget=min(10, time_left() - 2))
        if score_ad < best_score - 1e-12:
            best_h = h_ad
            best_score = score_ad

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)

    try:
        verify_score = evaluate_erdos_solution(best_h)
        if verify_score < best_score:
            best_score = verify_score
    except Exception:
        pass

    return best_h, best_score, len(best_h)
