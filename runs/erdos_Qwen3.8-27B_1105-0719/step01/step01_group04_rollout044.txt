def repair_feasible(vec):

def compute_profile(h):

def trust_region_step(h_cur, maxiter=50, time_limit=None, t_start=None):
    """Trust-region method using CVXPY for the subproblem."""
    n = len(h_cur)
    h_cur = repair_feasible(h_cur)
    best = h_cur.copy()
    best_score = exact_score(best)
    
    # Initial parameters
    lam = 0.1  # regularization parameter
    top_k = 30  # number of active constraints
    
    for it in range(maxiter):

                break

        # Build CVXPY problem

        t_var = cp.Variable()
        
        # Objective: t + lam * ||h_var - h_cur||^2
        obj = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h_cur))
        
        # Constraints

            sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1,
            t_var >= 0

            g_row = grads[idx, :]

            constraints.append(q_val + g_row @ (h_var - h_cur) <= t_var)
        
        prob = cp.Problem(obj, constraints)

            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
