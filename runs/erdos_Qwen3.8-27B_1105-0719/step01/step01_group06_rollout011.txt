                L = n - d
                G[k, 0:L] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:L]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def solve_lp_subproblem(h0, radius, max_lags=None):
    n = len(h0)
    M = 2 * n - 1
    p = get_profile(h0)
    G = get_lag_gradients(h0)
    if max_lags is not None and max_lags < M:
        active = np.argsort(p)[::-1][:max_lags]
    else:
        active = np.arange(M)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for i, k in enumerate(active):
        A_ub[i, :n] = G[k]
        A_ub[i, n] = -1.0
        b_ub[i] = G[k] @ h0 - p[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_sol = repair_feasible(res.x[:n])
            return h_sol
    except Exception:
        pass
    return None

def lp_trust_region(h_start, max_time, radius_init=0.15, max_lags=None):
    h = h_start.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    radius = radius_init
    stall = 0
    while time.perf_counter() - t0 < max_time:
        h_sol = solve_lp_subproblem(h, radius, max_lags=max_lags)
        if h_sol is None:
            radius *= 0.7
            stall += 1
            if stall > 12:
                break
            continue
        s = exact_score(h_sol)
        if s < best_score - 5e-13:
            best_h = h_sol.copy()
            best_score = s
            h = h_sol.copy()
            radius = min(0.40, radius * 1.15)
            stall = 0
        else:
            stall += 1
            radius *= 0.62
            if radius < 0.002 or stall > 20:
                break
    return best_h, best_score

def gradient_descent(h_start, max_time, lr_init=0.01, top_frac=0.4):
    h = h_start.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    lr = lr_init
    no_improve = 0
    while time.perf_counter() - t0 < max_time and no_improve < 30:
        p = get_profile(h)
        score = float(np.max(p))
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_k = max(8, int(len(p) * top_frac))
        active = np.argsort(p)[::-1][:top_k]
        w = np.exp((p[active] - p.max()) / max(0.03, (p[active].max() - p[active].min()) * 0.3 + 1e-10))
        w /= w.sum()
        G = get_lag_gradients(h)
        grad = np.sum(w[:, None] * G[active, :], axis=0)
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        improved = False
        for mult in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05, lr*0.02]:
            h_trial = repair_feasible(h - mult * grad)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                improved = True
                break
        if improved:
            lr = min(0.04, lr * 1.15)
            no_improve = 0
        else:
            lr *= 0.6
            if lr < 0.0005:
                break
    return best_h, best_score

def interpolate_to_dim(h, new_len):
    n = len(h)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_polish(h_start, t_start, maxiter=50, subset=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = subset if subset > 0 else M
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def obj_fn(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        return t - prof[active_lags]

    def cons_grad(x):
        h = x[:n]
        G = get_lag_gradients(h)
        J = np.zeros((len(active_lags), n + 1))
        J[:, :n] = -G[active_lags, :]
        J[:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 1.0)]
    x0 = np.concatenate([h_start, [t_start + 1e-6]])
    cons = [{'type': 'ineq', 'fun': cons_fun, 'jac': cons_grad}]
    try:
        result = minimize(obj_fn, x0, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-10})
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            return h_sol
    except Exception:
        pass
    return None

def build_fallback(n=64):
    h = np.zeros(n)
    for i in range(n):
        x = (i + 0.5) * 2.0 / n
        v = 0.5 + 0.3 * np.cos(2 * np.pi * x / 2.0)
        h[i] = v
    return repair_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_global = time.perf_counter()

    def _tl():
        return budget_s - (time.perf_counter() - t_global)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = build_fallback(64)

    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_score = exact_score(parent)

    # Symmetrize
    h_sym = repair_feasible(0.5 * (parent + parent[::-1]))
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-13:
        best_h = h_sym
        best_score = s_sym

    # Complement-symmetry transform
    h_comp = repair_feasible(0.5 * (parent + 1.0 - parent[::-1]))
    s_comp = exact_score(h_comp)
    if s_comp < best_score - 1e-13:
        best_h = h_comp
        best_score = s_comp

    # Phase 1: LP trust-region, moderate radius
    if _tl() > 50:
        h1, s1 = lp_trust_region(best_h, max_time=min(90, _tl() * 0.30), radius_init=0.18)
        if s1 < best_score - 1e-12:
            best_h, best_score = h1, s1

    # Phase 2: Gradient descent for fine structure
    if _tl() > 30:
        h2, s2 = gradient_descent(best_h, max_time=min(35, _tl() * 0.20), lr_init=0.008)
        if s2 < best_score - 1e-12:
            best_h, best_score = h2, s2

    # Phase 3: LP with larger radius for escape
    if _tl() > 40:
        h3, s3 = lp_trust_region(best_h, max_time=min(60, _tl() * 0.25), radius_init=0.30)
        if s3 < best_score - 1e-12:
            best_h, best_score = h3, s1 if False else s3

    # Phase 4: SLSQP full-constraint polish
    if _tl() > 30:
        mi = min(60, max(20, int(_tl() * 0.2)))
        h4 = slsqp_polish(best_h, best_score, maxiter=mi)
        if h4 is not None:
            s4 = exact_score(h4)
            if s4 < best_score - 1e-12:
                best_h, best_score = h4, s4

    # Phase 5: LP again with tight radius
    if _tl() > 25:
        h5, s5 = lp_trust_region(best_h, max_time=min(30, _tl() * 0.25), radius_init=0.06)
        if s5 < best_score - 1e-12:
            best_h, best_score = h5, s5

    # Phase 6: Dimension exploration
    if _tl() > 60:
        pn = len(best_h)
        dims = set()
        for sc in [0.6, 0.75, 0.85, 1.15, 1.25, 1.4, 1.7, 2.0]:
            d = int(round(pn * sc))
            if 32 <= d <= 280:
                dims.add(d)
        for d in [48, 64, 80, 96, 112, 128, 144, 160, 176, 192, 208, 224, 240, 256]:
            dims.add(d)
        dims.discard(pn)
        dims = sorted(dims)
        for new_n in dims:
            if _tl() < 30:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_int = exact_score(h_interp)
            # LP on new dimension
            t_alloc = min(35, _tl() * 0.15)
            h_new, s_new = lp_trust_region(h_interp, max_time=t_alloc, radius_init=0.20)
            if s_new < best_score - 1e-12:
                best_h, best_score = h_new, s_new
            # Quick gradient on new dim if budget allows
            if _tl() > 20:
                h_gd, s_gd = gradient_descent(h_new if h_new is not None else h_interp,
                                              max_time=min(15, _tl() * 0.08), lr_init=0.006)
                if s_gd < best_score - 1e-12:
                    best_h, best_score = h_gd, s_gd

    # Phase 7: Multi-start structured perturbations on best dimension
    if _tl() > 50:
        n = len(best_h)
        for p in range(12):
            if _tl() < 20:
                break
            if p == 0:
                z = np.random.randn(n)
                z -= z.mean()
                h_pert = repair_feasible(best_h + 0.025 * z)
            elif p == 1:
                h_pert = repair_feasible(np.roll(best_h, 1))
            elif p == 2:
                h_pert = repair_feasible(np.roll(best_h, -1))
            elif p == 3:
                h_pert = repair_feasible(best_h[::-1])
            elif p == 4:
                freq = 2 * np.pi * np.random.randint(2, max(3, n // 3)) / n
                z = np.sin(freq * np.arange(n))
                z -= z.mean()
                h_pert = repair_feasible(best_h + 0.04 * z)
            elif p == 5:
                z = np.sin(2 * np.pi * 3 * np.arange(n) / n) + 0.7 * np.sin(2 * np.pi * 5 * np.arange(n) / n)
                z -= z.mean()
                h_pert = repair_feasible(best_h + 0.03 * z)
            elif p == 6:
                h_pert = best_h.copy()
                blk = max(1, n // 5)
                h_pert[:blk] += 0.03
                h_pert[-blk:] -= 0.03
                h_pert = repair_feasible(h_pert)
            elif p == 7:
                x = np.linspace(-1, 1, n)
                h_s = 1.0 / (1.0 + np.exp(-5.0 * x))
                h_pert = repair_feasible(0.75 * best_h + 0.25 * repair_feasible(h_s))
            elif p == 8:
                # Piecewise constant perturbation
                nb = max(2, n // 8)
                h_pert = best_h.copy()
                for b in range(nb):
                    sl = slice(b * n // nb, (b + 1) * n // nb)
                    h_pert[sl] += 0.015 * np.random.choice([-1, 1])
                h_pert = repair_feasible(h_pert)
            elif p == 9:
                # Logistic mix
                x = np.linspace(0, 1, n)
                h_l = 1.0 / (1.0 + np.exp(-8 * (x - 0.5)))
                h_pert = repair_feasible(0.85 * best_h + 0.15 * repair_feasible(h_l))
            elif p == 10:
                # Random smooth (sinc kernel)
                z = np.random.randn(n)
                kernel = np.sinc(np.arange(-10, 11) / 5.0)
                from scipy.ndimage import uniform_filter1d
                z = uniform_filter1d(z, size=5)
                z -= z.mean()
                h_pert = repair_feasible(best_h + 0.012 * z)
            else:
                # Two-scale perturbation
                z1 = np.sin(2 * np.pi * np.random.randint(1, n // 4) * np.arange(n) / n)
                z2 = np.random.randn(n)
                from scipy.ndimage import uniform_filter1d
                z2 = uniform_filter1d(z2, size=max(3, n // 20))
                z = z1 + 0.5 * z2
                z -= z.mean()
                h_pert = repair_feasible(best_h + 0.02 * z)

            s_pert = exact_score(h_pert)
            t_alloc = min(25, _tl() * 0.12)
            h_new, s_new = lp_trust_region(h_pert, max_time=t_alloc, radius_init=0.12)
            if s_new < best_score - 1e-12:
                best_h, best_score = h_new, s_new
            # Quick GD follow-up
            if _tl() > 15:
                h_gd, s_gd = gradient_descent(h_new, max_time=min(10, _tl() * 0.06), lr_init=0.005)
                if s_gd < best_score - 1e-12:
                    best_h, best_score = h_gd, s_gd

    # Phase 8: Final LP tightening
    if _tl() > 15:
        h_fin, s_fin = lp_trust_region(best_h, max_time=min(15, _tl() * 0.5), radius_init=0.04)
        if s_fin < best_score - 1e-12:
            best_h, best_score = h_fin, s_fin

    # Phase 9: Final gradient micro-refinement
    if _tl() > 10:
        h_fg, s_fg = gradient_descent(best_h, max_time=min(8, _tl() * 0.5), lr_init=0.003)
        if s_fg < best_score - 1e-12:
            best_h, best_score = h_fg, s_fg

    # Final repair and evaluation
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)

    return (best_h, final_score, len(best_h))
