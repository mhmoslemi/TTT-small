    #   contribution: h[i-offset] * (1-h[i])

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def coord_optimize_sweep(h, dx, beta, lam, n_sweeps=100):
    """Coordinate descent with smooth max + Lagrange multiplier for sum constraint."""
    n = len(h)
    h = h.copy().astype(float)
    
    for sweep in range(n_sweeps):
        for i in range(n):
            corrs = compute_corrs(h)
            # Compute b[k] = dc_k/dh_i and a[k] = c_k - h[i]*b[k]
            b = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                offset = k - (n - 1)
                if 0 <= i + offset < n:
                    b[k] += (1.0 - h[i + offset])
                if 0 <= i - offset < n:
                    b[k] -= h[i - offset]
            
            a = corrs - h[i] * b
            
            # 1D convex minimization: min_{x in [0,1]} (1/beta)*logsumexp(beta*(a+b*x))*dx + lam*x
            # Derivative: dx * sum(w_k * b_k) + lam, where w_k = softmax(beta*(a+b*x))
            vals0 = beta * a
            m0 = vals0.max()
            w0 = np.exp(vals0 - m0)
            d0 = dx * (w0 * b).sum() / w0.sum() + lam
            
            if d0 >= 0:
                h[i] = 0.0
                continue
            
            vals1 = beta * (a + b)
            m1 = vals1.max()
            w1 = np.exp(vals1 - m1)
            d1 = dx * (w1 * b).sum() / w1.sum() + lam
            
            if d1 <= 0:
                h[i] = 1.0
                continue
            
            # Bisection
            lo_x, hi_x = 0.0, 1.0
            for _ in range(40):
                mid_x = (lo_x + hi_x) / 2.0
                vals = beta * (a + b * mid_x)
                m = vals.max()
                w = np.exp(vals - m)
                d = dx * (w * b).sum() / w.sum() + lam
                if d > 0:
                    hi_x = mid_x
                else:
                    lo_x = mid_x
            h[i] = (lo_x + hi_x) / 2.0
        
        # Project to maintain feasibility
        h = project_feasible(h, n / 2.0)
        # Update lambda
        h = project_feasible(h, n / 2.0)
    
    return h

                if 0 <= i + offset < n:
                    b[k] += (1.0 - h[i + offset])

            w0 = np.exp(vals0 - m0)

            vals1 = beta * (a + b)

def nesterov_smooth(h0, dx, target_sum, max_iter, beta):
    """Nesterov accelerated gradient for smooth max."""
    n = len(h0)
    h = h0.copy().astype(float)
    y = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    t = 1.0
    
    for it in range(max_iter):
        corrs = compute_corrs(y)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                offset = k - (n - 1)
                i_lo = max(0, -offset)
                i_hi = min(n - 1, n - 1 - offset)
                if i_lo <= i_hi:
                    idx = np.arange(i_lo, i_hi + 1)
                    grad[idx + offset] += w[k] * (1.0 - y[idx])
                    grad[idx] += -w[k] * y[idx + offset]
        grad *= dx
        
        alpha = 1.0 / (1.0 + 0.05 * t)
        h_new = y - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h_new + momentum * (h_new - h)
        t = t_new
        h = h_new
        
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5


def binary_sa(h0, dx, target_sum, max_iters, rng, T_start=0.5, T_end=0.01):
    """Simulated annealing on binary solutions."""
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    excess = h.sum() - target_sum
    if excess > 0:
        ones = np.where(h == 1.0)[0]
        rng.shuffle(ones)
        h[ones[:int(abs(excess))]] = 0.0
    elif excess < 0:
        zeros = np.where(h == 0.0)[0]
        rng.shuffle(zeros)
        h[zeros[:int(abs(excess))]] = 1.0
    
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur_c5 = best_c5
    
    for it in range(max_iters):
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_of_h(h, dx)
        
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5


def lbfgs_smooth(h0, dx, target_sum, beta, maxiter):
    """L-BFGS-B with penalty for sum constraint."""
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx + 500.0 * (x.sum() - target_sum) ** 2
    
    def grad(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)

                i_lo = max(0, -offset)

                    grad[idx + offset] += w[k] * (1.0 - x[idx])
                    grad[idx] += -w[k] * x[idx + offset]
        grad *= dx
        grad += 1000.0 * (x.sum() - target_sum)
        return grad

    try:

    return best_h, best_c5, n_final
