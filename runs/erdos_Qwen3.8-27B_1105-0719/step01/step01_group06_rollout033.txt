def compute_G(h):
    """Compute full gradient matrix G where G[m, i] = dp_m/dh_i."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                idx = np.arange(delta, n)
                G[k, idx] += dx * (1.0 - h[idx - delta])
                idx2 = np.arange(0, n - delta)
                G[k, idx2] -= dx * h[idx2 + delta]
        else:
            d = -delta
            if d < n:
                idx = np.arange(0, n - d)
                G[k, idx] += dx * (1.0 - h[idx + d])
                idx2 = np.arange(d, n)
                G[k, idx2] -= dx * h[idx2 - d]

    Returns optimal delta."""

    hi = 1.0 - h_i  # delta such that h_i + delta = 1
    
    # f(delta) = max_m(p[m] + a[m]*delta) - convex piecewise linear
    # Use ternary search
    for _ in range(n_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(p + a * m1)
        f2 = np.max(p + a * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)

def lse_slsqp(h_start, beta, maxiter=100, time_limit=30):
    """Optimize log-sum-exp surrogate using SLSQP."""
    n = len(h_start)
    t0 = time.perf_counter()
    
    def obj(h):
        prof = get_profile(h)
        # Log-sum-exp with numerical stability
        max_p = np.max(prof)
        if max_p > 1e15:
            return max_p
        return max_p + np.log(np.sum(np.exp(beta * (prof - max_p)))) / beta
    
    def obj_grad(h):
        prof = get_profile(h)
        max_p = np.max(prof)
        if max_p > 1e15:
            return np.zeros(n)
        exp_terms = np.exp(beta * (prof - max_p))
        w = exp_terms / np.sum(exp_terms)
        G = compute_G(h)
        return G.T @ w
    
    def eq_con(h):
        return h.sum() - n / 2.0
    
    def eq_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    x0 = h_start.copy()
    
    try:
        result = minimize(
            obj, x0, jac=obj_grad,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x)
            return h_sol
    except Exception:
        pass
    return None

def epigraph_solve(h_start, t_start, maxiter=80, n_constraints=0):
    """SLSQP epigraph with top-K constraints."""
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    
    prof = get_profile(h_start)
    if nc < M:
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)
    
    def obj(x):
        return x[n]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = get_profile(h)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        G = compute_G(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 1.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-11}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            return h_sol
    except Exception:
        pass
    return None

def interpolate_to_dim(vec, new_len):
    n = len(vec)

    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len

    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_G(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                idx = np.arange(delta, n)
                G[k, idx] += dx * (1.0 - h[idx - delta])
                idx2 = np.arange(0, n - delta)
                G[k, idx2] -= dx * h[idx2 + delta]
        else:
            d = -delta
            if d < n:
                idx = np.arange(0, n - d)
                G[k, idx] += dx * (1.0 - h[idx + d])
                idx2 = np.arange(d, n)
                G[k, idx2] -= dx * h[idx2 - d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)

def solve_1d_concave_max(p, a, lo, hi, n_iters=60):
    """Solve min_delta max_m(p[m] + a[m]*delta) for delta in [lo, hi]."""
    if lo >= hi:
        return 0.0
    for _ in range(n_iters):
        m1 = lo + (hi - lo) * 0.382
        m2 = hi - (hi - lo) * 0.382
        f1 = np.max(p + a * m1)
        f2 = np.max(p + a * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)

    """Swap-based coordinate descent that preserves sum constraint."""
    n = len(h)
    M = 2 * n - 1

import numpy as np

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_G(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                idx = np.arange(delta, n)
                G[k, idx] += dx * (1.0 - h[idx - delta])
                idx2 = np.arange(0, n - delta)
                G[k, idx2] -= dx * h[idx2 + delta]
        else:
            d = -delta
            if d < n:
                idx = np.arange(0, n - d)
                G[k, idx] += dx * (1.0 - h[idx + d])
                idx2 = np.arange(d, n)
                G[k, idx2] -= dx * h[idx2 - d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def solve_1d_minmax(p, a, lo, hi, n_iters=55):
    """Solve min_delta max_m(p[m]+a[m]*delta) for delta in [lo,hi]."""
    if lo >= hi:
        return 0.0
    for _ in range(n_iters):
        m1 = lo + (hi - lo) * 0.381966
        m2 = hi - (hi - lo) * 0.381966
        f1 = np.max(p + a * m1)
        f2 = np.max(p + a * m2)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)

def epigraph_solve(h_start, t_start, maxiter=60, n_constraints=50):

    nc = min(n_constraints, M)

        return x[n]

        g = np.zeros(n + 1)
        g[n] = 1.0

        eq = h.sum() - n / 2.0

        G = compute_G(h)

        J[1:, n] = 1.0

    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,

        pert = np.zeros(n)

            amt = rng.uniform(0.01, 0.04)
            pert[a] += amt

    elif mode == 3:
        # Block perturbation
        pert = np.zeros(n)
        bl = rng.randint(n // 10, n // 4)

        mask[start:start+bl] = False
        pert[mask] = -pert[start:start+bl].sum() / mask.sum()
    else:
        pert = rng.uniform(0.003, 0.01) * rng.randn(n)
    
    pert -= pert.mean()
    return repair_feasible(h + pert)

        n = len(best_h)

        for r in range(n_restarts):
            if tleft() < 25:
                break
            mode = r % 5
            h_pert = generate_structured_perturbation(best_h, seed + r * 7, mode)
            # Swap descent from perturbation
            tl_r = min(6, tleft() * 0.1)
            h_rd, s_rd = swap_descent(h_pert, max_iters=10000, time_limit=tl_r, seed=seed + 50 + r)
            if s_rd < best_score - 1e-12:
                best_h = h_rd
                best_score = s_rd
            # Also try from the perturbation with epigraph
            if s_rd < best_score + 0.002 and tleft() > 15:
                h_ep = epigraph_solve(h_rd, s_rd + 1e-6, maxiter=30, n_constraints=40)
                if h_ep is not None:
                    s_ep = exact_score(h_ep)
                    if s_ep < best_score - 1e-12:
                        best_h = h_ep
                        best_score = s_ep
                    if tleft() > 8:
                        h_p, s_p = swap_descent(h_ep, max_iters=5000, time_limit=min(3, tleft()*0.1), seed=seed+80+r)
                        if s_p < best_score - 1e-12:
                            best_h = h_p
                            best_score = s_p
    
    # Phase 6: Try different dimensions
    if tleft() > 45:
        pn = len(best_h)
        dims = []
        for d in [pn - 8, pn - 4, pn + 4, pn + 8, pn + 16, pn + 32]:
            if 40 <= d <= 250 and d != pn:
                dims.append(d)
        if pn * 2 <= 200:
            dims.append(pn * 2)
        dims = sorted(set(dims))
        
        for new_n in dims:
            if tleft() < 25:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            # Swap descent in new dimension
            tl_d = min(8, tleft() * 0.15)
            h_nd, s_nd = swap_descent(h_interp, max_iters=20000, time_limit=tl_d, seed=seed + 200)
            if s_nd < best_score - 1e-12:
                best_h = h_nd
                best_score = s_nd
    
    # Phase 7: Final intensive swap descent
    if tleft() > 15:
        tl_final = min(15, tleft() * 0.6)
        h_fin, s_fin = swap_descent(best_h, max_iters=50000, time_limit=tl_final, seed=seed + 300)
        if s_fin < best_score - 1e-13:
            best_h = h_fin
            best_score = s_fin
    
    # Phase 8: Final structured polish
    if tleft() > 8:
        tl_st = min(6, tleft() * 0.5)
        h_st, s_st = swap_descent_structured(best_h, time_limit=tl_st, seed=seed + 400)
        if s_st < best_score - 1e-13:
            best_h = h_st
            best_score = s_st
    
    # Final repair and score
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    
    return (best_h, final_score, len(best_h))
