def gradient_of_lag(h, n, dx, d):
    """Compute gradient of C_d with respect to h."""
    G = np.zeros(n)
    if d == 0:
        G[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        G[0:n-d] += dx * (1.0 - h[d:n])
        G[n-d:n] += -dx * h[0:d]
    else:
        dp = -d
        G[dp:n] += dx * (1.0 - h[0:n-dp])
        G[0:dp] += -dx * h[dp:n]
    return G

def subgradient_direction(h, n, dx, top_m=5, sharpness=50.0):
    """Compute a (sub)gradient direction using top-m active constraints."""
    profile = np.correlate(h, 1 - h, mode="full") * dx
    c = np.max(profile)
    
    idx = np.argsort(-profile)[:top_m]
    weights = np.exp((profile[idx] - c) * sharpness)
    weights /= weights.sum()
    
    G = np.zeros(n)
    for w, mi in zip(weights, idx):
        d = mi - (n - 1)
        G += w * gradient_of_lag(h, n, dx, d)
    
    G -= G.mean()  # project out the mean
    return G, c, profile

def momentum_subgradient(h, max_iters=10000, top_m=3, sharpness=50.0,
                         init_alpha=0.005, decay=0.9995, beta=0.8):
    """Subgradient method with Nesterov momentum."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    
    v = np.zeros(n)
    alpha = init_alpha
    no_improve = 0
    
    for it in range(max_iters):
        # Nesterov lookahead
        h_lookahead = repair_feasible(h - beta * v)
        G, c, _ = subgradient_direction(h_lookahead, n, dx, top_m, sharpness)
        
        # Update

        h = repair_feasible(h - alpha * v)
        c = exact_score(h)

            best_h = h.copy()
            no_improve = 0

sum_d sum_i h[i](1-h[i+d]) / (2n-1)

This is still non-convex.

I think the best approach is still the subgradient + SLSQP combination. Let me finalize.

Actually, one more idea: **alternating optimization**. Fix all but two coordinates, optimize those two. Since fixing all but two coordinates makes the objective a function of two variables (which is a piecewise-convex function), we can find the optimal values by checking the breakpoints.

For two coordinates h[i] and h[j], the objective as a function of (h[i], h[j]) is:
max_d (a_d * h[i] + b_d * h[j] + c_d)

where a_d, b_d, c_d depend on the other coordinates. The minimum of this over the region {h[i] + h[j] = s, 0 <= h[i] <= 1, 0 <= h[j] <= 1} (where s is determined by the sum constraint) can be found by:
1. Parameterize h[j] = s - h[i]
2. The objective becomes max_d ((a_d - b_d) * h[i] + b_d * s + c_d), which is a piecewise linear convex function of h[i]
3. Find the minimum of this piecewise linear function

This is a 1D problem! The minimum of a piecewise linear convex function occurs at a breakpoint or at a boundary. The breakpoints are where (a_i - b_i) * x + b_i * s + c_i = (a_j - b_j) * x + b_j * s + c_j.

This approach is exact (for the two-coordinate subproblem) and efficient. Let me implement it!

The algorithm:
1. For each pair (i, j) where h[i] != 0 and h[j] != 1 (so they can be changed):
   a. Compute a_d, b_d, c_d for all d
   b. Find the optimal h[i] (and hence h[j] = s - h[i])
   c. If it improves the objective, accept
2. Repeat until no improvement

This is O(n^2 * n * n) = O(n^4) per full pass (n^2 pairs, n lags, n breakpoints to check). For n=66, that's about 19 million ops per pass. Might be too slow for Python.

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_of_lag(h, n, dx, d):
    G = np.zeros(n)
    if d == 0:
        G[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        G[0:n-d] = dx * (1.0 - h[d:n])
        G[n-d:n] = -dx * h[0:d]
    else:
        dp = -d
        G[dp:n] = dx * (1.0 - h[0:n-dp])
        G[0:dp] = -dx * h[dp:n]
    return G

def subgradient_direction(h, n, dx, top_m=3, sharpness=80.0):
    profile = np.correlate(h, 1 - h, mode="full") * dx
    c = np.max(profile)
    idx = np.argsort(-profile)[:top_m]
    w = np.exp((profile[idx] - c) * sharpness)
    w /= w.sum()
    G = np.zeros(n)
    for wi, mi in zip(w, idx):
        d = mi - (n - 1)
        G += wi * gradient_of_lag(h, n, dx, d)
    G -= G.mean()
    return G, c

def momentum_subgradient(h, max_iters=20000, top_m=3, sharpness=80.0,
                         init_alpha=0.004, decay=0.9998, beta=0.7, time_limit_s=None, t0=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    v = np.zeros(n)
    alpha = init_alpha
    no_improve = 0
    for it in range(max_iters):
        if time_limit_s is not None and (time.perf_counter() - t0) > time_limit_s:
            break
        h_la = repair_feasible(h - beta * v)
        G, c = subgradient_direction(h_la, n, dx, top_m, sharpness)
        v = beta * v + G
        h = repair_feasible(h - alpha * v)
        c = exact_score(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        alpha *= decay
        if no_improve > 3000:
            v = np.zeros(n)
            no_improve = 0
            alpha = max(alpha, 0.003)
    return best_h, best_c

def binary_sa(h, max_iters=30000, T_start=0.004, T_end=0.00005, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h)
    h_b = (h > 0.5).astype(np.float64)
    n_ones = int(h_b.sum())
    target = n // 2
    if n_ones > target:
        ones = np.where(h_b == 1)[0]
        rem = rng.choice(ones, n_ones - target, replace=False)
        h_b[rem] = 0
    elif n_ones < target:
        zeros = np.where(h_b == 0)[0]
        add = rng.choice(zeros, target - n_ones, replace=False)
        h_b[add] = 1
    h = h_b.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    cur_c = best_c
    T = T_start
    cool = (T_end / T_start) ** (1.0 / (max_iters + 1))
    for it in range(max_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c_new = exact_score(h)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        delta = c_new - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = c_new
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c

def binary_greedy(h, max_passes=5, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    h = (h > 0.5).astype(np.float64)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rem = rng.choice(ones, n_ones - target, replace=False)
        h[rem] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        add = rng.choice(zeros, target - n_ones, replace=False)
        h[add] = 1
    best_h = h.copy()
    best_c = exact_score(h)
    for _ in range(max_passes):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        order = rng.permutation(len(ones) * len(zeros))
        for idx in order:
            ii = idx // len(zeros)
            jj = idx % len(zeros)
            a, b = ones[ii], zeros[jj]
            h[a] = 0.0
            h[b] = 1.0
            c = exact_score(h)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                order = rng.permutation(len(ones) * len(zeros))
            else:
                h[a] = 1.0
                h[b] = 0.0
        if not improved:
            break
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_epigraph(h_start, maxiter=400, tol=1e-11):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        hh = x[:-1]
        t = x[-1]
        profile = np.correlate(hh, 1 - hh, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        hh = x[:-1]
        J = np.zeros((M, n + 1))
        J[:, -1] = 1.0
        for m in range(M):
            d = m - (n - 1)
            J[m, :n] = -gradient_of_lag(hh, n, dx, d)
        return J
    c0 = exact_score(h_start)
    x0 = np.concatenate([h_start, [c0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            if h_new.sum() > 0 and abs(h_new.sum() - n/2.0) < 1e-8:
                return h_new
    except Exception:
        pass
    return None
