        if 0 <= jk2 < n:

        return val
    
    # Find breakpoints of the max of linear functions

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h):
    """Project h to the set {x: sum(x) = len(x)/2, 0 <= x <= 1}."""
    n = len(h)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h.copy()
    lo, hi = -1.5, 1.5

        g = np.sum(np.clip(h - mid, 0.0, 1.0))
        if g > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

    n = len(h)

    slopes = np.zeros(n_corr)

        if dm == 0:
            continue  # handled separately

            slope -= h[jk2]
