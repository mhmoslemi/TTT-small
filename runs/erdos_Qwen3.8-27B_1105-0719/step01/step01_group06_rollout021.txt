    h = np.array(h, dtype=np.float64).copy()

    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0

    if abs(diff) < 1e-14:
        return h

    tau_hi = 0.0

def compute_direction(h, profile, top_k=10):
    """Compute a descent direction based on active lags."""
    n = len(h)
    dx = 2.0 / n
    # Get active lags
    active = np.argsort(profile)[::-1][:top_k]

    grad = np.zeros(n)
    for k in active:

            if delta < n:
                grad[delta:n] += dx * (1.0 - h[0:n-delta])
                grad[0:n-delta] -= dx * h[delta:n]

            if d < n:

    d = -grad
    d -= d.mean()  # Ensure zero sum again
    
    norm = np.linalg.norm(d)
    if norm < 1e-15:

    return d / norm

def block_direction(h, block_size=7, shift=1):
    """Create a direction that shifts mass from one block to the next."""
    n = len(h)
    d = np.zeros(n)
    for start in range(0, n - block_size, shift):
        # Transfer mass from block [start, start+block_size) to [start+shift, start+shift+block_size)
        src = slice(start, start + block_size)
        dst = slice(start + shift, start + shift + block_size)
        # Only if both are valid
        d[src] -= 1.0
        d[dst] += 1.0
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return None
    return d / norm

    # Based on which position has higher "pressure"

    return d

def compute_subgradient_direction(h, profile, top_k=15, weight_power=2.0):
    """Compute weighted subgradient direction for active/near-active lags."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    # Weight by how close to max
    max_val = profile.max()
    active = np.argsort(profile)[::-1][:top_k]
    weights = np.zeros(M)
    for idx in active:
        if profile[idx] > max_val * 0.9:  # within 10% of max
            weights[idx] = (profile[idx] / max_val) ** weight_power
    
    # Weighted gradient
    grad = np.zeros(n)
    for k in active:
        if weights[k] > 0:
            grad += weights[k] * lag_gradient(h, k, n, dx)
    
    # Project to zero-sum
    grad -= grad.mean()
    
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return -grad / norm  # Negative gradient = descent direction

    delta = k - (n - 1)

            g[0:n - delta] -= dx * h[delta:n]
    else:
        dd = -delta
        if dd < n:
            g[0:n - dd] += dx * (1.0 - h[dd:n])
            g[dd:n] -= dx * h[0:n - dd]
    return g

                        s = s_new
                        improved = True
                        break
                    tried += 1
                    if tried > 30:
                        break
                if improved:
                    break

        # Random pairwise
        if not improved and stall % 4 == 1:
            for _ in range(8):
                i, j = np.random.choice(n, 2, replace=False)
                h_new, s_new = pairwise_line_search(h, int(i), int(j))
                if h_new is not None and s_new < s - 1e-13:
                    h = h_new
                    s = s_new
                    improved = True
                    break

        # Low-frequency direction
        if not improved and stall % 5 == 2:
            freqs = np.arange(n)
            d = np.sin(2.0 * np.pi * freqs / n)
            d -= d.mean()
            nr = np.linalg.norm(d)
            if nr > 1e-15:
                d /= nr
                h_new, s_new = line_search(h, d)
                if h_new is not None and s_new < s - 1e-13:
                    h = h_new
                    s = s_new
                    improved = True

        if improved:
            stall = 0
        else:
            stall += 1

        if stall > MAX_STALL:
            # Perturb and restart local search
            eps = 0.01 + 0.02 * (stall % 3)
            z = np.random.randn(n)
            z -= z.mean()
            h_pert = repair_feasible(h + eps * z)
            s_pert = exact_score(h_pert)
            # Try to refine
            prof_p = get_profile(h_pert)
            d = subgradient_dir(h_pert, prof_p, top_k=10)
            if d is not None:
                h_new, s_new = line_search(h_pert, d)
                if h_new is not None and s_new < s - 1e-13:
                    h = h_new
                    s = s_new
                    stall = 0
            if stall > MAX_STALL:
                stall = MAX_STALL - 10  # Prevent infinite loop

    return h, s

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t_start)

    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(140, 0.5)

    n = len(parent)
    best_h = repair_feasible(parent)
    best_score = exact_score(best_h)

    # Symmetrize
    h_sym = repair_feasible(0.5 * (best_h + best_h[::-1].copy()))
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-13:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: Main optimization at parent dimension
    best_h, best_score = optimize_at_dim(best_h, tleft, max_time=500)

    # Phase 2: Try a few perturbation restarts
    for attempt in range(5):
        if tleft() < 60:
            break
        eps = 0.005 + 0.01 * attempt
        z = np.random.randn(n)
        z -= z.mean()
        h_pert = repair_feasible(best_h + eps * z)
        s_pert = exact_score(h_pert)
        h_opt, s_opt = optimize_at_dim(h_pert, tleft, max_time=80)
        if s_opt < best_score - 1e-13:
            best_h = h_opt
            best_score = s_opt

    # Phase 3: Try different dimensions
    if tleft() > 60:
        candidate_dims = []
        for dd in [-8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 20, 28, 40]:
            nn = n + dd
            if 40 <= nn <= 300 and nn % 2 == 0:
                candidate_dims.append(nn)
        # Also try some "nice" dimensions
        for nn in [128, 144, 160, 168, 192, 224, 256]:
            if nn % 2 == 0 and nn not in candidate_dims and 40 <= nn <= 300:
                candidate_dims.append(nn)
        candidate_dims = sorted(set(candidate_dims))

        for nn in candidate_dims:
            if tleft() < 40:
                break
            h_interp = interpolate_dim(best_h, nn)
            h_opt, s_opt = optimize_at_dim(h_interp, tleft, max_time=60)
            if s_opt < best_score - 1e-13:
                best_h = h_opt
                best_score = s_opt

    # Phase 4: Final refinement at the best dimension
    if tleft() > 30:
        best_h, best_score = optimize_at_dim(best_h, tleft, max_time=min(120, tleft() - 10))

    # Final repair
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)

    return (best_h, final_score, len(best_h))
