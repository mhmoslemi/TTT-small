def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    y = np.clip(y, -2.0, 3.0)  # initial clip to avoid extreme values
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        # Spread residual among interior points
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(interior) > 0:
            h[interior] += residual / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
        else:
            # Spread among all
            h += residual / n
            h = np.clip(h, 0.0, 1.0)
    return h

def anneal_exchange(h, current_c5, n_moves, rng, t0=0.05, tf=0.001):
    """Simulated annealing with exchange moves."""
    best_h = h.copy()
    best_c5 = current_c5
    curr_h = h.copy()
    curr_c5 = current_c5
    n = len(h)
    
    for mv in range(n_moves):
        progress = mv / max(n_moves - 1, 1)
        T = t0 * math.exp(math.log(tf / t0) * progress)
        
        i, j = rng.choice(n, size=2, replace=False)
        # Random direction
        if rng.random() < 0.5:
            inc, dec = i, j
        else:
            inc, dec = j, i
        
        cap = min(1.0 - curr_h[inc], curr_h[dec])
        if cap < 1e-12:
            continue
        
        # Random tau (biased towards smaller)
        tau = cap * (rng.random() ** 0.5)
        
        h_new = curr_h.copy()
        h_new[inc] += tau
        h_new[dec] -= tau
        if h_new[inc] > 1.0 + 1e-10 or h_new[dec] < -1e-10:
            continue
        
        c5_new = compute_c5(h_new)
        delta = c5_new - curr_c5
        
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            curr_h = h_new
            curr_c5 = c5_new
            if curr_c5 < best_c5 - 1e-14:

                best_c5 = curr_c5

def map_resolution(h_old, n_new):
    """Map h_old to new resolution n_new preserving mass."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    h_new = np.zeros(n_new)
    for j in range(n_new):
        # New interval: [j*dx_new, (j+1)*dx_new)
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        total = 0.0
        length = 0.0
        for i in range(n_old):
            old_start = i * dx_old
            old_end = (i + 1) * dx_old
            overlap = min(x_end, old_end) - max(x_start, old_start)
            if overlap > 0:
                total += h_old[i] * overlap
                length += overlap
        if length > 0:
            h_new[j] = total / length
        else:
            h_new[j] = 0.5
    return project_feasible(h_new, n_new)
