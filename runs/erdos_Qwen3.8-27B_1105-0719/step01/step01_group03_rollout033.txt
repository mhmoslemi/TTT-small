- P_n = {v : sum v_i = n/2, 0 <= v_i <= 1}

For n odd: fill (n-1)/2 entries with 1, one entry with 0.5, rest with 0.

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add more; add to entries with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in order:
            add = min(remaining, capacity[i])
            h[i] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to remove; remove from entries with most value
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            rem = min(remaining, h[i])
            h[i] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, tau):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    M = len(profile)
    
    # Compute weights with numerical stability
    mu = np.max(profile)
    u = np.exp((profile - mu) / tau)
    w = u / u.sum()
    
    # Compute gradient: sum_m w_m * grad p_m
    grad = np.zeros(n)
    
    for m in range(M):
        if w[m] < 1e-15:
            continue
        d = m - (n - 1)
        wm = w[m]
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                # p_d has term h[i+d]*(1-h[i])
                # d(p_d)/d(h[i]) = -dx * h[i+d]
                # d(p_d)/d(h[i+d]) = dx * (1-h[i])
                grad[i] += wm * (-dx) * h[i + d]
                grad[i + d] += wm * dx * (1.0 - h[i])
        else:
            s = -d
            for i in range(n - s):
                # p_{-s} has term h[i]*(1-h[i+s])
                # d/d(h[i]) = dx*(1-h[i+s])
                # d/d(h[i+s]) = -dx*h[i]
                grad[i] += wm * dx * (1.0 - h[i + s])
                grad[i + s] += wm * (-dx) * h[i]
    
    return grad

def fw_vertex(g, n):
    """Find v in P_n that minimizes g·v.
    P_n = {v : sum v_i = n/2, 0 <= v_i <= 1}
    """
    target = n / 2.0
    v = np.zeros(n)
    # Sort by g ascending (we want to put mass where g is smallest)
    order = np.argsort(g)
    remaining = target
    for i in order:
        if remaining >= 1.0 - 1e-12:
            v[i] = 1.0
            remaining -= 1.0
        else:
            v[i] = remaining
            remaining = 0.0
            break
    return v

def fw_anneal(h0, tau_start, tau_end, max_iter, deadline_check):
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    
    n = len(h)
    tau = tau_start
    
    for it in range(max_iter):
        if deadline_check() <= 5:
            break
        
        # Anneal tau
        progress = (it + 1) / max_iter
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute gradient
        grad = smooth_grad(h, tau)
        
        # FW vertex
        v = fw_vertex(grad, n)
        
        # Direction
        direction = v - h
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            break
        
        # Line search
        best_alpha = 0.0
        best_val = exact_score(h)
        
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.7, 0.4, 0.15, 0.05, 0.8, 0.6]:
            h_try = h + alpha * direction
            h_try = repair_feasible(h_try)
            val = exact_score(h_try)
            if val < best_val - 1e-12:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            h = repair_feasible(h + best_alpha * direction)
            c = exact_score(h)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
        else:
            # Try smaller steps
            for alpha in [0.02, 0.01, 0.005]:
                h_try = repair_feasible(h + alpha * direction)
                val = exact_score(h_try)
                if val < best_c - 1e-12:
                    h = h_try
                    best_c = val
                    best_h = h.copy()
                    break
    
    return best_h, best_c

def active_set_descent(h0, max_iter, time_fn):
    """Subgradient descent focusing on active constraints."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iter):
        if time_fn() <= 2:
            break
        
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        
        # Active set: lags within threshold of max
        thresh = c - max(1e-4, 0.01 * c)
        active = np.where(profile >= thresh)[0]
        
        # Weighted gradient
        weights = np.zeros(2 * n - 1)
        active_weights = profile[active] - thresh + 1e-6
        weights[active] = active_weights
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for m in active:
            d = m - (n - 1)
            wm = weights[m]
            if d == 0:
                grad += wm * dx * (1.0 - 2.0 * h)
            elif d > 0:
                grad[:n-d] += wm * (-dx) * h[d:]
                grad[d:] += wm * dx * (1.0 - h[:n-d])
            else:
                s = -d
                grad[:n-s] += wm * dx * (1.0 - h[s:])
                grad[s:] += wm * (-dx) * h[:n-s]
        
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Backtracking
        alpha = 0.05
        improved = False
        for _ in range(15):
            h_new = project_polytope(h - alpha * grad, n)
            c_new = exact_score(h_new)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
        
        if not improved:
            # Try random perturbation to escape
            z = np.random.randn(n) * 0.01
            h_try = project_polytope(h + z, n)
            if exact_score(h_try) < best_c:
                h = h_try
                best_c = exact_score(h)
                best_h = h.copy()
    
    return best_h, best_c

def solve_epigraph_slsqp(h0, maxiter=200):
    """SLSQP on the epigraph formulation."""
    n = len(h0)
    dx = 2.0 / n
    c0 = exact_score(h0)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = compute_profile(h)
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        n_local = len(h)
        M = 2 * n_local - 1
        J = np.zeros((M, n_local + 1))
        for m in range(M):
            d = m - (n_local - 1)
            if d == 0:
                J[m, :n_local] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                J[m, :n_local-d] += dx * (-h[d:])
                J[m, d:n_local] += dx * (1.0 - h[:n_local-d])
            else:
                s = -d
                J[m, :n_local-s] += dx * (1.0 - h[s:])
                J[m, s:n_local] += dx * (-h[:n_local-s])
            J[m, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new < c0 + 1e-8:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def generate_starts(h0, rng):
    """Generate diverse starting points."""
    n = len(h0)
    starts = [h0.copy()]
    
    # Symmetrized
    h_sym = repair_feasible((h0 + h0[::-1]) / 2.0)
    starts.append(h_sym)
    
    # Power transforms
    for p in [2.0, 3.0, 0.5, 1.5]:
        h_c = (h0 - 0.5) * 2.0  # center to [-1, 1]
        h_c = np.clip(h_c, -0.99, 0.99)
        h_p = np.sign(h_c) * np.abs(h_c) ** (1.0 / p)
        h_p = h_p * 0.5 + h0.mean() * 0.5  # rough rescale
        h_p = repair_feasible(h0 * 0.3 + (0.5 + 0.5 * np.sign(h_c) * np.abs(h_c)**(1.0/p)) * 0.7)
        starts.append(h_p)
    
    # Random perturbations
    for eps in [0.05, 0.1, 0.15]:
        z = rng.randn(n)
        z -= z.mean()
        starts.append(repair_feasible(h0 + eps * z))
    
    # Binary pattern based on h0 rank
    order = np.argsort(h0)
    h_bin = np.zeros(n)
    h_bin[order[n//2:]] = 1.0
    h_bin = repair_feasible(h_bin)
    starts.append(h_bin)
    
    # Blend with binary
    for frac in [0.3, 0.5, 0.7]:
        starts.append(repair_feasible(frac * h0 + (1-frac) * h_bin))
    
    return starts

    n = len(vec)

    target = n / 2.0

        return h

            h[i] += a
            rem -= a
            if rem <= 1e-14:
