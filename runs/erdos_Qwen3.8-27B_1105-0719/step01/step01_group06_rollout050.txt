```python
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.special import expit
import time

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    # For each lag m = 0..M-1, the shift is d = m - (n-1)
    for m in range(M):
        d = m - (n - 1)
        if d > 0:
            # p_m = dx * sum_{i=0}^{n-d-1} h[i+d] * (1 - h[i])
            L = n - d
            # dp/dh[i+d] += dx * (1-h[i]) for i=0..L-1
            G[m, d:n] += dx * (1.0 - h[0:L])
            # dp/dh[i] += -dx * h[i+d] for i=0..L-1
            G[m, 0:L] -= dx * h[d:n]
        elif d < 0:
            c = -d
            # p_m = dx * sum_{i=0}^{n-c-1} h[i] * (1 - h[i+c])
            L = n - c
            G[m, 0:L] += dx * (1.0 - h[c:n])
            G[m, c:n] -= dx * h[0:L]
        else:
            # p_m = dx * sum_{i=0}^{n-1} h[i] * (1 - h[i])
            G[m, :] = dx * (1.0 - 2.0 * h)
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def solve_lp_direction(h0, delta):
    """Solve the linearized LP to get a direction from h0."""
    n = len(h0)
    M = 2 * n - 1
    p = get_profile(h0)
    G = get_gradient_matrix(h0)
    
    # Variables: x = [h_0, ..., h_{n-1}, t]
    # min t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: G@h - t <= G@h0 - p
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h0 - p
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - delta)
        hi = min(1.0, h0[i] + delta)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'maxiter': 5000})
    
    if result.success:
        return result.x[:n]
    return None
```
