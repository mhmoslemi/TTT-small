def alem_step(h, G, P, C, R, n_active):
    n = len(h)
    target = n / 2.0
    m = 2 * n - 1
    
    # Select active constraints
    slacks = C - P  # non-negative, 0 for active
    active_idx = np.argsort(slacks)[:n_active]
    
    # Build the QP
    z = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(z) == target,
        z >= 0,
        z <= 1,
        cp.norm(z - h) <= R,
    ]

    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if z.value is not None:

    except:
        pass

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_feasible(h):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    # Bisection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            for i in range(n):
                if result[i] < 1.0:
                    add = min(residual, 1.0 - result[i])
                    result[i] += add
                    residual -= add
                    if abs(residual) < 1e-12:
                        break
        else:
            for i in range(n):
                if result[i] > 0.0:
                    sub = min(-residual, result[i])
                    result[i] -= sub
                    residual += sub
                    if abs(residual) < 1e-12:
                        break
    return result

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradients(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for ell in range(-(n-1), n):
        j = ell + (n - 1)
        i_lo1 = max(0, -ell)
        i_hi1 = min(n - 1, n - 1 - ell)
        if i_lo1 <= i_hi1:
            k_idx = np.arange(i_lo1, i_hi1 + 1)
            G[j, k_idx] += dx * (1.0 - h[k_idx + ell])
        i_lo2 = max(0, ell)
        i_hi2 = min(n - 1, n - 1 + ell)
        if i_lo2 <= i_hi2:
            k_idx = np.arange(i_lo2, i_hi2 + 1)
            G[j, k_idx] += -dx * h[k_idx - ell]
    return G

def alem_step(h, R, n_active, deadline):
    n = len(h)
    P = compute_profile(h)
    C = float(np.max(P))
    G = compute_gradients(h)
    
    slacks = C - P
    active_idx = np.argsort(slacks)[:min(n_active, 2*n-1)]
    
    z = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(z) == n / 2.0,
        z >= 0,
        z <= 1,
        cp.norm(z - h) <= R,
    ]
    
    for j in active_idx:
        constraints.append(G[j] @ z - t <= G[j] @ h - P[j])
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PIPS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if z.value is not None:
                    return np.array(z.value).flatten(), C
        except Exception:
            continue
    
    return None, C

Delta formula for swapping h[i]: 1→0 and h[j]: 0→1:

For each shift ell:
- Effect of h[i] going 1→0:
  - As first factor (h[i]): removes h[i](1-h[i+ell]) = 1*(1-h[i+ell]) → delta = -(1-h[i+ell]) if i+ell valid
  - As second factor (1-h[i]): removes h[i-ell]*(1-h[i]) = h[i-ell]*0 = 0, adds h[i-ell]*(1-0) = h[i-ell] → delta = +h[i-ell] if i-ell valid

- Effect of h[j] going 0→1:
  - As first factor (h[j]): adds h[j](1-h[j+ell]) = 1*(1-h[j+ell]) → delta = +(1-h[j+ell]) if j+ell valid

I2 = 1 if 0 ≤ i-ell < n (and i-ell is in valid range for second factor)

import numpy as np
import time
import cvxpy as cp

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_feasible(h):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2, 0, 1)
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            for i in range(n):
                if result[i] < 1.0:
                    add = min(residual, 1.0 - result[i])
                    result[i] += add
                    residual -= add
                    if abs(residual) < 1e-12:
                        break
        else:
            for i in range(n):
                if result[i] > 0.0:
                    sub = min(-residual, result[i])
                    result[i] -= sub
                    residual += sub
                    if abs(residual) < 1e-12:
                        break
    return result

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradients(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for ell in range(-(n - 1), n):
        j = ell + (n - 1)
        lo1 = max(0, -ell)
        hi1 = min(n - 1, n - 1 - ell)
        if lo1 <= hi1:
            k_idx = np.arange(lo1, hi1 + 1)
            G[j, k_idx] += dx * (1.0 - h[k_idx + ell])
        lo2 = max(0, ell)
        hi2 = min(n - 1, n - 1 + ell)
        if lo2 <= hi2:
            k_idx = np.arange(lo2, hi2 + 1)
            G[j, k_idx] += -dx * h[k_idx - ell]
    return G

def alem_solve(h, R, n_active, solver_list):
    n = len(h)
    P = compute_profile(h)
    C = float(np.max(P))
    G = compute_gradients(h)
    m = 2 * n - 1
    n_act = min(n_active, m)
    slacks = C - P
    active_idx = np.argsort(slacks)[:n_act]
    z = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(z) == n / 2.0,
        z >= 0,
        z <= 1,
        cp.norm(z - h) <= R,
    ]
    for j in active_idx:
        constraints.append(G[j] @ z - t <= G[j] @ h - P[j])
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if z.value is not None:
                    return np.array(z.value).flatten(), C
        except Exception:
            continue
    return None, C

def line_search_direct(h, z, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = exact_objective(h)
    alphas = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        y = project_feasible(h + alpha * (z - h))
        c = exact_objective(y)
        if c < best_c - 1e-12:
            best_h = y
            best_c = c
            break
    return best_h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def binary_steepest(b, deadline, max_rounds=5000):
    n = len(b)
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is None:
            no_improve += 1
            if no_improve > 3:
                break
            continue
        no_improve = 0
        i, j = best_swap
        b[i] = 0.0
        b[j] = 1.0
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        curr_c = exact_objective(b)
    return b, curr_c

def binary_double_swap(b, deadline, max_rounds=200):
    n = len(b)
    curr_c =
exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        best_improve = 1e-14
        best_move = None
        m = len(ones)
        for a in range(m):
            for c_idx in range(a + 1, m):
                i1, i2 = ones[a], ones[c_idx]
                for b_idx in range(m):
                    for d_idx in range(b_idx + 1, m):
                        j1, j2 = zeros[b_idx], zeros[d_idx]
                        b[i1] = 0.0
                        b[i2] = 0.0
                        b[j1] = 1.0
                        b[j2] = 1.0
                        c = exact_objective(b)
                        b[i1] = 1.0
                        b[i2] = 1.0
                        b[j1] = 0.0
                        b[j2] = 0.0
                        improve = curr_c - c
                        if improve > best_improve:
                            best_improve = improve
                            best_move = (i1, i2, j1, j2)
        if best_move is None:
            break
        i1, i2, j1, j2 = best_move
        b[i1] = 0.0
        b[i2] = 0.0
        b[j1] = 1.0
        b[j2] = 1.0
        ones = list(np.where(b == 1.0)[0])
        zeros = list(np.where(b == 0.0)[0])

def binary_sa(b, deadline, max_iters=20000, T_start=0.008, T_end=0.0005):
    n = len(b)
    rng = np.random.RandomState(12345)
    curr_c = exact_objective(b)
    best_b = b.copy()
    best_c = curr_c
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for it in range(max_iters):
        if time.time() > deadline - 4:
            break
        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** progress
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        new_c = exact_objective(b)
        delta = new_c - curr_c
        if delta < -1e-14 or (delta <= 0.01 and rng.random() < np.exp(-delta / T)):
            curr_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if curr_c < best_c - 1e-14:
                best_c = curr_c
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c

def transfer_to_size(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return project_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_feasible(h_new)

def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        return h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.randint(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        return project_feasible(np.clip(h, 0, 1))
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
        return project_feasible(h)
    elif mode == 3:
        return project_feasible(rng.uniform(0, 1, n))
    elif mode == 4:
        h = np.zeros(n)
        block = max(1, n // 10)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        return project_feasible(h)
    elif mode == 5:
        h = np.zeros(n)
        k = n // 2
        golden = 0.6180339887498949
        for i in range(k):
            pos = int((i * golden) % 1.0 * n)
            h[pos] = 1.0
        return project_feasible(h)
    else:
        return project_feasible(np.full(n, 0.5))

def gradient_descent_step(h, P, G, n_active, lr, deadline):
    n = len(h)
    C = float(np.max(P))
    slacks = C - P
    active_idx = np.argsort(slacks)[:n_active]
    grad = np.zeros(n)
    for j in active_idx:
        grad += G[j]
    grad /= len(active_idx)
    neg_grad = h - lr * grad
    new_h = project_feasible(neg_grad)
    return new_h

            if b_trans is not None:
                b_trans, c_trans = binary_steepest(b_trans, deadline, max_rounds=1000)
                if c_trans < best_c - 1e-13:
                    best_c = c_trans
                    best_h = b_trans.copy()

    # Final validation
    best_h = project_feasible(best_h)
    best_c = exact_objective(best_h)

    return best_h, best_c, len(best_h)
