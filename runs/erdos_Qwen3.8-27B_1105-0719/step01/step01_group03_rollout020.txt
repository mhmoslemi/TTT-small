        grad = G.T @ wp
        m_s = b1 * m_s + (1.0 - b1) * grad
        v_s = b2 * v_s + (1.0 - b2) * grad * grad
        m_hat = m_s / (1.0 - b1 ** (t + 1))
        v_hat = v_s / (1.0 - b2 ** (t + 1))
        h_try = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_try = repair_feasible(h_try)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
        h = h_try
        if (t + 1) % 40 == 0:
            tau *= tau_decay
            lr *= lr_decay
    return best_h, best_c

def coord_descent(h0, max_sweeps=40, grid_coarse=150, grid_fine=400):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best = exact_score(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            profile = compute_profile(h)
            G = grad_profile(h, n, dx)
            a = G[:, i].copy()
            c = profile - a * h[i]
            v_grid = np.linspace(0.0, 1.0, grid_coarse)
            f = np.max(a[:, None] * v_grid[None, :] + c[:, None], axis=0)
            v1 = v_grid[np.argmin(f)]
            lo = max(0.0, v1 - 1.0 / grid_coarse)
            hi = min(1.0, v1 + 1.0 / grid_coarse)
            v_grid2 = np.linspace(lo, hi, grid_fine)
            f2 = np.max(a[:, None] * v_grid2[None, :] + c[:, None], axis=0)
            v_opt = v_grid2[np.argmin(f2)]
            if abs(v_opt - h[i]) < 1e-11:
                continue
            h_save = h.copy()
            h[i] = v_opt
            h = repair_feasible(h)
            c_new = exact_score(h)
            if c_new < best - 1e-14:
                best = c_new
                improved = True
            else:
                h = h_save
        if not improved:
            break
    return h, best

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof

    def ineq_jac(x):
        h = x[:-1]
        G = grad_profile(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            return h_new
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.perf_counter() - t_start

    def time_left():
        return budget_s - elapsed()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 10:
        parent = np.full(64, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent
    best_n = len(parent)

    # Phase 1: Adam LS from parent with different parameters
    configs = [
        (500, 0.1, 0.92, 0.04, 0.85),
        (400, 0.06, 0.93, 0.05, 0.88),
        (350, 0.12, 0.90, 0.03, 0.82),
        (600, 0.05, 0.94, 0.04, 0.90),
    ]
    for iters, tau0, td, lr0, lrd in configs:
        if time_left() < 30:
            break
        h_a, c_a = adam_optimize(parent, iters=iters, tau0=tau0, tau_decay=td, lr0=lr0, lr_decay=lrd)
        if c_a < best_c - 1e-13:
            best_h = h_a
            best_c = c_a
            best_n = len(h_a)

    # Phase 2: Coordinate descent polish
    if time_left() > 20:
        h_cd, c_cd = coord_descent(best_h, max_sweeps=35)
        if c_cd < best_c - 1e-14:
            best_h = h_cd
            best_c = c_cd

    # Phase 3: More Adam from polished solution
    if time_left() > 30:
        h_a2, c_a2 = adam_optimize(best_h, iters=300, tau0=0.03, tau_decay=0.95, lr0=0.02, lr_decay=0.9)
        if c_a2 < best_c - 1e-14:
            best_h = h_a2
            best_c = c_a2
        h_cd2, c_cd2 = coord_descent(best_h, max_sweeps=40)
        if c_cd2 < best_c - 1e-14:
            best_h = h_cd2
            best_c = c_cd2

    # Phase 4: Try perturbed starts to escape local optima
    if time_left() > 80:
        for eps in [0.03, 0.06, 0.1, 0.15, 0.25]:
            if time_left() < 40:
                break
            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_a, c_a = adam_optimize(h_pert, iters=300, tau0=0.08, tau_decay=0.93, lr0=0.04, lr_decay=0.87)
            if c_a < best_c - 1e-13:
                h_cd, c_cd = coord_descent(h_a, max_sweeps=30)
                if c_cd < best_c - 1e-14:
                    best_h = h_cd
                    best_c = c_cd
                else:
                    best_h = h_a
                    best_c = c_a

    # Phase 5: Try different dimensions
    if time_left() > 80:
        n_cur = best_n
        dim_candidates = []
        for dn in [-6, -4, -2, 2, 4, 6, 10, 16, 24, -12, -20]:
            nc = n_cur + dn
            if 30 <= nc <= 200:
                dim_candidates.append(nc)
        dim_candidates = sorted(set(dim_candidates), key=lambda d: abs(d - n_cur))
        for new_n in dim_candidates[:5]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_a, c_a = adam_optimize(h_interp, iters=300, tau0=0.08, tau_decay=0.93, lr0=0.04, lr_decay=0.87)
            if c_a < best_c - 1e-13:
                h_cd, c_cd = coord_descent(h_a, max_sweeps=30)
                if c_cd < best_c - 1e-14:
                    best_h = h_cd
                    best_c = c_cd
                    best_n = new_n
                else:
                    best_h = h_a
                    best_c = c_a
                    best_n = new_n

    # Phase 6: SLSQP polish
    if time_left() > 30:
        h_sol = slsqp_solve(best_h, maxiter=150)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-14:
                best_h = h_sol
                best_c = c_sol

    # Phase 7: Final coordinate descent
    if time_left() > 15:
        h_cd, c_cd = coord_descent(best_h, max_sweeps=50, grid_coarse=200, grid_fine=600)
        if c_cd < best_c - 1e-14:
            best_h = h_cd
            best_c = c_cd

    # Phase 8: Final Adam with very low temperature
    if time_left() > 20:
        h_a, c_a = adam_optimize(best_h, iters=200, tau0=0.01, tau_decay=0.97, lr0=0.005, lr_decay=0.95)
        if c_a < best_c - 1e-14:
            best_h = h_a
            best_c = c_a

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
