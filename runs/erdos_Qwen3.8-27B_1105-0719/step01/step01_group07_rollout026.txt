def project_feasible(h):
    """Project h onto {0<=x<=1, sum(x)=n/2} using bisection on lambda."""
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h[~np.isfinite(h)] = 0.0
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h + lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = result < 1.0 - 1e-10
            cnt = int(np.sum(mask))
            if cnt > 0:
                result[mask] = np.minimum(1.0, result[mask] + residual / cnt)
        else:
            mask = result > 1e-10
            cnt = int(np.sum(mask))
            if cnt > 0:
                result[mask] = np.maximum(0.0, result[mask] + residual / cnt)
    return np.clip(result, 0.0, 1.0)

def gradient_of_profile_entry(h, j):
    """Gradient of P_j(h) = dx * sum h[i]*(1-h[i+lag])"""
    n = len(h)
    dx = 2.0 / n
    lag = j - (n - 1)

        if i == q:
            grad[i] += dx * (1.0 - 2.0 * h[i])
        else:
            grad[i] += dx * (1.0 - h[q])
            grad[q] -= dx * h[i]
    return grad

def subgradient_at_max(h):
    """Compute a subgradient of C(h) = max_j P_j(h)"""
    profile = exact_profile(h)
    m = np.max(profile)
    # Active set: entries within small tolerance of max
    active = np.where(profile >= m - 1e-8)[0]
    grad = np.zeros(len(h))
    for j in active:
        grad += gradient_of_profile_entry(h, j)

    """
    Solve the bundle QP subproblem.
    bundle: list of (h_k, c_k, g_k) where c_k = C(h_k), g_k = subgradient at h_k
    Q: regularization parameter
    Returns: proposed h_new, t_new
    """
    n = len(h_current)
    target = n / 2.0
    
    # Variables: [h_0, ..., h_{n-1}, t] total n+1
    # Objective: Q * (1/2) * ||h - h_current||^2 + t
    # In linprog form (linear objective): we can't do quadratic directly
    # Instead, use the linearized bundle model:
    # min t
    # s.t. c_k + g_k^T (h - h_k) <= t for all k in bundle
    #      sum(h) = target

    #      ||h - h_current||_2 <= R (trust region)

    #      sum(h) = target
    #      0 <= h_i <= 1

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
