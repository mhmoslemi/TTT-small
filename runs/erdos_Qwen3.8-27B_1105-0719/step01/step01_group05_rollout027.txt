def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)

        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break

            mask = h < 1.0
            s = np.sum(mask)
            if s == 0:

            h[mask] = np.minimum(h[mask] + diff / s, 1.0)
        else:
            mask = h > 0.0
            s = np.sum(mask)
            if s == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / s, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        # Distribute remaining
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)

def gradient_active_lag(h, lag, dx):
    """Compute gradient of p_lag(h) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s <= 0:
            return grad
        # ∂/∂h[j] = (1-h[j-lag]) for j in [lag, n-1]
        #            -h[j+lag] for j in [0, s-1]
        grad[lag:lag+s] += (1.0 - h[:s])  # j-lag ranges from 0 to s-1
        grad[:s] -= h[lag:]  # j+lag ranges from lag to n-1
    else:
        j = -lag
        s = n - j
        if s <= 0:
            return grad
        # ∂/∂h[k] = (1-h[k+j]) for k in [0, s-1]
        #            -h[k-j] for k in [j, n-1]
        grad[:s] += (1.0 - h[j:j+s])
        grad[j:j+s] -= h[:s]
    return grad * dx

def smooth_log_sum_exp(h, tau, dx):
    """Compute smooth approximation of max profile and its gradient."""
    n = len(h)
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    pmax = np.max(profile)
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    value = pmax + tau * np.log(wsum)
    wnorm = weights / wsum
    grad = np.zeros(n)
    active = wnorm > 1e-6
    for m in np.where(active)[0]:
        lag = m - (n - 1)
        grad += wnorm[m] * gradient_active_lag(h, lag, dx)
    return value, grad

    best_h = project_box_sum(h_start, target)
    best_c = exact_objective(best_h)[0]
    
    def obj_grad(x):
        val, grad = smooth_log_sum_exp(x, tau, dx)
        # Add constraint penalty
        se = np.sum(x) - target
        val += 1e4 * se * se
        grad = grad + 2e4 * se * np.ones(n)
        return val, grad
    
    bounds = [(0.0, 1.0)] * n

        zpvar = cp.Variable(s, nonneg=True)
        for i in range(s):
            constraints.append(zp[i] <= h[i+k])
            constraints.append(zp[i] <= 1 - h[i])
            constraints.append(zp[i] >= h[i+k] - h[i])
            constraints.append(zp[i] >= 0)
        constraints.append(zp @ np.ones(s) <= t / dx)
        
        # Negative lag -k: z[i] = h[i] * (1-h[i+k]) for i = 0..s-1
        zn = cp.Variable(s, nonneg=True)
        for i in range(s):
            constraints.append(zn[i] <= h[i])
            constraints.append(zn[i] <= 1 - h[i+k])
            constraints.append(zn[i] >= h[i] - h[i+k])
            constraints.append(zn[i] >= 0)
        constraints.append(zn @ np.ones(s) <= t / dx)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        # Try different solvers
        for solver in [cp.SCS, cp.CBC, cp.GLPK, cp.SCIP]:
            try:
                prob.solve(solver=solver, time_limit=max(10, int(deadline - time.time()) - 10))
                if h.value is not None and np.all(np.isfinite(h.value)):
                    h_sol = np.round(h.value)
                    # Verify
                    h_sol = np.clip(h_sol, 0, 1)
                    if abs(h_sol.sum() - target) > 0.01:
                        continue
                    c, _ = exact_objective(h_sol)
                    return h_sol, c
            except Exception:
                continue
    except Exception:
        pass
    
    return None, float('inf')

def upsample(h, new_n):
    """Upsample h to new_n points maintaining integral."""
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def downsample(h, new_n):
    """Downsample h to new_n points."""
    n = len(h)
    if new_n >= n:
        return upsample(h, new_n)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def make_structured_start(n, rng, mode=0):
    """Create structured starting points."""
    target = 0.5 * n
    if mode == 0:
        # Block pattern: [1...1 0...0]
        h = np.zeros(n)
        h[:target] = 1.0
    elif mode == 1:
        # Two blocks
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[n//2:n//2+q] = 1.0
    elif mode == 2:
        # Three blocks
        h = np.zeros(n)
        q = n // 6
        h[:q] = 1.0
        h[n//3:n//3+q] = 1.0
        h[2*n//3:2*n//3+q] = 1.0
    elif mode == 3:
        # Random
        h = (rng.rand(n) > 0.5).astype(float)
    elif mode == 4:
        # Quasi-periodic: alternating blocks of different sizes
        h = np.zeros(n)
        pattern = [2, 1, 1, 2, 2, 1, 2, 1, 1, 2]  # block sizes
        pos = 0
        val = 1
        while pos < n:
            bs = pattern[pos % len(pattern)]
            val = 1 - val
            h[pos:min(pos+bs, n)] = val
            pos += bs
    elif mode == 5:
        # Cosine
        x = np.arange(n)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * 2 * x / n)
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)

def slsqp_multi_tau(h_start, n, deadline):
    """SLSQP with decreasing temperature."""
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c = exact_objective(best_h)[0]
    
    tau_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for tau in tau_list:
        if time.time() > deadline - 10:
            break
        h_opt, c_opt = slsqp_refine(best_h, n, tau, deadline, max_iter=200)
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c
